14/// Count the entries of `git status --porcelain -z` for a work tree. Untracked files are
15/// counted as files (`--untracked-files=all`), because a directory of untracked work is the
16/// thing a move must not lose and "1 untracked" for a directory of forty files would
17/// understate it.
18pub fn dirty_state(worktree: &Path) -> Result<DirtyState> {
19    let bytes = git::status_porcelain_z(worktree, "all")?;
20    let mut state = parse_status_z(&bytes);
21    state.in_progress = operation_in_progress(worktree);
22    Ok(state)
23}

25/// The dirty state of many work trees, measured at once.
26///
27/// Each measurement is a `git status` subprocess of a few tens of milliseconds and none of
28/// them depends on another, so they are taken on a small pool of threads rather than one
29/// after the next. The measurement that motivated this: on a repository with 128
30/// registered work trees the sequence cost 6.2–9.9 s (n=3, shared machine), which was
31/// most of what the whole topology cost.
32///
33/// A `None` in the input means "do not measure this one"; a `None` in the output means it
34/// was not measured, or git refused to say. The answers come back in the order the work
35/// trees were given, whatever order they finished in, so what this feeds is byte-identical
36/// to reading them one at a time.
37///
38/// ```
39/// use majordomus_cli::worktree::state::dirty_states;
40/// let repo = tempfile::tempdir().unwrap();
41/// let init = std::process::Command::new("git")
42///     .args(["init", "-q"])
43///     .current_dir(repo.path())
44///     .status()
45///     .unwrap();
46/// assert!(init.success());
47/// std::fs::write(repo.path().join("new.txt"), "work").unwrap();
48///
49/// let states = dirty_states(&[Some(repo.path()), None, Some(repo.path())]);
50/// assert_eq!(states.len(), 3);
51/// assert!(states[1].is_none(), "a None in is a None out");
52/// let first = states[0].as_ref().expect("a git work tree is measured");
53/// assert_eq!(first.untracked, 1);
54/// assert!(!first.clean);
55/// assert_eq!(states[2], states[0], "answers come back in input order");
56/// ```
57pub fn dirty_states(worktrees: &[Option<&Path>]) -> Vec<Option<DirtyState>> {
58    let mut out: Vec<Option<DirtyState>> = vec![None; worktrees.len()];
59    let wanted: Vec<usize> = worktrees
60        .iter()
61        .enumerate()
62        .filter_map(|(i, w)| w.map(|_| i))
63        .collect();
64    // One measurement is not worth a thread, and none is not worth a channel.
65    if wanted.len() < 2 {
66        for i in wanted {
67            out[i] = worktrees[i].and_then(|p| dirty_state(p).ok());
68        }
69        return out;
70    }
71    let next = AtomicUsize::new(0);
72    let (tx, rx) = mpsc::channel();
73    std::thread::scope(|scope| {
74        for _ in 0..probe_threads(wanted.len()) {
75            let tx = tx.clone();
76            let next = &next;
77            let wanted = &wanted;
78            scope.spawn(move || loop {
79                let Some(&i) = wanted.get(next.fetch_add(1, Ordering::Relaxed)) else {
80                    return;
81                };
82                let measured = worktrees[i].and_then(|p| dirty_state(p).ok());
83                if tx.send((i, measured)).is_err() {
84                    return;
85                }
86            });
87        }
88        // The receiver below ends when the last sender is gone, so this one must go first.
89        drop(tx);
90        for (i, measured) in rx {
91            out[i] = measured;
92        }
93    });
94    out
95}

97/// How many work trees to measure at once. Each measurement waits on a git subprocess
98/// rather than on this process's CPU, but the pool is still the machine's parallelism and
99/// no more: a development machine runs other people's sessions, and a pool that
100/// oversubscribes it makes their commands slower to make this one faster.
101fn probe_threads(work: usize) -> usize {
102    std::thread::available_parallelism()
103        .map(|n| n.get())
104        .unwrap_or(4)
105        .clamp(1, 16)
106        .min(work)
107}

109/// Parse the NUL-separated porcelain v1 status. Renames are disabled by the caller
110/// (`--no-renames`), so every entry is `XY path\0` and no entry is followed by a second
111/// path.
112///
113/// ```
114/// use majordomus_cli::worktree::state::parse_status_z;
115/// let s = parse_status_z(b"M  a\0 M b\0?? c\0UU d\0MM e\0");
116/// assert_eq!((s.staged, s.unstaged, s.untracked, s.conflicted), (2, 2, 1, 1));
117/// assert!(!s.clean);
118/// assert!(parse_status_z(b"").clean);
119/// ```
120pub fn parse_status_z(bytes: &[u8]) -> DirtyState {
121    let mut state = DirtyState::default();
122    for entry in bytes.split(|b| *b == 0).filter(|e| e.len() >= 3) {
123        let x = entry[0] as char;
124        let y = entry[1] as char;
125        match (x, y) {
126            ('?', '?') => state.untracked += 1,
127            ('!', '!') => {}
128            ('U', _) | (_, 'U') | ('A', 'A') | ('D', 'D') => state.conflicted += 1,
129            _ => {
130                if x != ' ' {
131                    state.staged += 1;
132                }
133                if y != ' ' {
134                    state.unstaged += 1;
135                }
136            }
137        }
138    }
139    state.clean =
140        state.staged == 0 && state.unstaged == 0 && state.untracked == 0 && state.conflicted == 0;
141    state
142}

144/// The operation git is in the middle of in this work tree, if any: what `git status`
145/// would report as "rebase in progress" and so on, read from the per-worktree git
146/// directory the way git itself does.
147pub fn operation_in_progress(worktree: &Path) -> Option<String> {
148    let dir = git_dir_of(worktree)?;
149    let probes: &[(&str, &str)] = &[
150        ("rebase-merge", "rebase"),
151        ("rebase-apply", "rebase"),
152        ("MERGE_HEAD", "merge"),
153        ("CHERRY_PICK_HEAD", "cherry-pick"),
154        ("REVERT_HEAD", "revert"),
155        ("BISECT_LOG", "bisect"),
156    ];
157    probes
158        .iter()
159        .find(|(file, _)| dir.join(file).exists())
160        .map(|(_, op)| (*op).to_string())
161}

163/// The git directory of one work tree, read the way git records it rather than by asking
164/// git for it: `.git` is that directory in the primary checkout and a file holding
165/// `gitdir: <path>` in a linked one. `git rev-parse --absolute-git-dir` answers the same
166/// question, but it is a subprocess, and a subprocess is what this costs — 128 of the
167/// cheapest possible `git rev-parse` took 1.5–2.2 s on this machine (n=3) — paid once per
168/// work tree for a question one `stat` and one small read answer. Git is asked only when
169/// `.git` is not there to read, so a checkout arranged some other way still gets an answer.
170fn git_dir_of(worktree: &Path) -> Option<PathBuf> {
171    let dot = worktree.join(".git");
172    match std::fs::metadata(&dot) {
173        Ok(m) if m.is_dir() => return Some(dot),
174        Ok(_) => {
175            if let Some(p) = std::fs::read_to_string(&dot)
176                .ok()
177                .as_deref()
178                .and_then(gitdir_link)
179            {
180                return Some(if p.is_absolute() { p } else { worktree.join(p) });
181            }
182        }
183        Err(_) => {}
184    }
185    let out = git::try_run(worktree, &["rev-parse", "--absolute-git-dir"]).ok()?;
186    if out.status != Some(0) {
187        return None;
188    }
189    Some(PathBuf::from(out.text().ok()?))
190}

192/// The path a linked work tree's `.git` file names, when it names one. Git writes exactly
193/// one line, `gitdir: <path>`, and the path is normally absolute.
194///
195/// ```
196/// use majordomus_cli::worktree::state::gitdir_link;
197/// use std::path::PathBuf;
198/// assert_eq!(gitdir_link("gitdir: /a/.git/worktrees/x\n"), Some(PathBuf::from("/a/.git/worktrees/x")));
199/// assert_eq!(gitdir_link("gitdir: ../elsewhere"), Some(PathBuf::from("../elsewhere")));
200/// assert_eq!(gitdir_link("gitdir:\n"), None);
201/// assert_eq!(gitdir_link("something else\n"), None);
202/// assert_eq!(gitdir_link(""), None);
203/// ```
204pub fn gitdir_link(text: &str) -> Option<PathBuf> {
205    let rest = text.lines().next()?.strip_prefix("gitdir:")?.trim();
206    if rest.is_empty() {
207        None
208    } else {
209        Some(PathBuf::from(rest))
210    }
211}

214#[derive(Debug, Clone, PartialEq, Eq)]
215pub struct BranchRef {
216    /// Short name.
217    pub name: String,
218    /// The commit.
219    pub head: String,

222    /// The work tree holding it, when one does.

226/// Every local branch, in one subprocess: name, commit, upstream with its ahead/behind
227/// distance, and the work tree holding it.
228pub fn branches(primary: &Path) -> Result<Vec<BranchRef>> {
229    let out = git::run(primary, &["for-each-ref", "refs/heads", BRANCH_FORMAT])?;
230    Ok(parse_branches(&String::from_utf8_lossy(&out.stdout)))
231}

233/// One local branch, read now: the same fields as [`branches`], for a caller that must not act
234/// on a reading taken earlier. `None` when the branch no longer exists.
235///
236/// `for-each-ref refs/heads/<name>` also matches the branches below `<name>/`, so the result is
237/// filtered to the exact name.
238///
239/// ```
240/// use majordomus_cli::worktree::state::branch;
241/// let repo = tempfile::tempdir().unwrap();
242/// let git = |args: &[&str]| {
243///     let ok = std::process::Command::new("git")
244///         .args(["-c", "user.name=t", "-c", "user.email=t@t", "-c", "commit.gpgsign=false"])
245///         .args(args)
246///         .current_dir(repo.path())
247///         .status()
248///         .unwrap();
249///     assert!(ok.success(), "git {args:?}");
250/// };
251/// git(&["init", "-q", "-b", "trunk"]);
252/// git(&["commit", "-q", "--allow-empty", "-m", "root"]);
253/// git(&["branch", "feat/sub"]);
254///
255/// // `refs/heads/feat` matches feat/sub as well; the exact-name filter drops it.
256/// assert!(branch(repo.path(), "feat").unwrap().is_none());
257/// let feat = branch(repo.path(), "feat/sub").unwrap().expect("feat/sub exists");
258/// assert_eq!(feat.name, "feat/sub");
259/// assert!(feat.upstream.is_none());
260/// assert_eq!(feat.head.len(), 40);
261/// assert!(branch(repo.path(), "trunk").unwrap().unwrap().worktree.is_some());
262/// assert!(branch(repo.path(), "gone").unwrap().is_none());
263/// ```
264pub fn branch(primary: &Path, name: &str) -> Result<Option<BranchRef>> {
265    let pattern = format!("refs/heads/{name}");
266    let out = git::run(primary, &["for-each-ref", pattern.as_str(), BRANCH_FORMAT])?;
267    Ok(parse_branches(&String::from_utf8_lossy(&out.stdout))
268        .into_iter()
269        .find(|b| b.name == name))
270}

272/// The `for-each-ref` format [`parse_branches`] reads.
273const BRANCH_FORMAT: &str = "--format=%(refname:short)%00%(objectname)%00%(upstream:short)%00%(upstream:track,nobracket)%00%(worktreepath)%00";
274
275/// Parse the format above: five NUL-terminated fields per branch, one branch per line.
276///
277/// ```
278/// use majordomus_cli::worktree::state::parse_branches;
279/// let b = parse_branches("feature/x\0abc\0origin/feature/x\0ahead 2, behind 1\0/a/foo-wt/feature/x\0\nmain\0def\0\0\0/a/foo\0\n");
280/// assert_eq!(b.len(), 2);
281/// let u = b[0].upstream.as_ref().unwrap();
282/// assert_eq!((u.ahead, u.behind, u.gone), (Some(2), Some(1), false));
283/// assert!(b[1].upstream.is_none());
284/// ```
285pub fn parse_branches(text: &str) -> Vec<BranchRef> {
286    let mut out = Vec::new();
287    for line in text.split('\n') {
288        let fields: Vec<&str> = line.split('\0').collect();
289        if fields.len() < 5 || fields[0].is_empty() {
290            continue;
291        }
292        let upstream = if fields[2].is_empty() {
293            None
294        } else {
295            let track = fields[3].trim();
296            let mut ahead = None;
297            let mut behind = None;
298            let mut gone = false;
299            for part in track.split(',').map(str::trim) {
300                if part == "gone" {
301                    gone = true;
302                } else if let Some(n) = part.strip_prefix("ahead ") {
303                    ahead = n.trim().parse().ok();
304                } else if let Some(n) = part.strip_prefix("behind ") {
305                    behind = n.trim().parse().ok();
306                }
307            }
308            if !gone {
309                ahead = ahead.or(Some(0));
310                behind = behind.or(Some(0));
311            }
312            Some(UpstreamState {
313                name: fields[2].to_string(),
314                ahead,
315                behind,
316                gone,
317            })
318        };
319        out.push(BranchRef {
320            name: fields[0].to_string(),
321            head: fields[1].to_string(),
322            upstream,
323            worktree: if fields[4].is_empty() {
324                None
325            } else {
326                Some(PathBuf::from(fields[4]))
327            },
328        });
329    }
330    out
331}

333/// Every local branch reachable from `trunk` or from the remote-tracking branch `trunk`
334/// follows. A local trunk that has fallen behind its remote — a primary checkout nobody has
335/// pulled, because a session holds uncommitted work in it — would otherwise call every branch
336/// that landed since unmerged, and cleanup would offer nothing; a branch the remote trunk
337/// contains is published and integrated, which is the safety cleanup asks for.
338///
339/// ```
340/// use majordomus_cli::worktree::state::merged_into_trunk;
341/// use std::process::Command;
342/// let dir = tempfile::tempdir().unwrap();
343/// let git = |args: &[&str]| {
344///     let ok = Command::new("git")
345///         .arg("-C")
346///         .arg(dir.path())
347///         .args(args)
348///         .env("GIT_AUTHOR_NAME", "t")
349///         .env("GIT_AUTHOR_EMAIL", "t@example.com")
350///         .env("GIT_COMMITTER_NAME", "t")
351///         .env("GIT_COMMITTER_EMAIL", "t@example.com")
352///         .status()
353///         .unwrap()
354///         .success();
355///     assert!(ok, "git {args:?}");
356/// };
357/// git(&["init", "-q", "-b", "main"]);
358/// git(&["commit", "-q", "--allow-empty", "-m", "base"]);
359/// git(&["branch", "landed"]);
360/// // no upstream here: the local trunk alone decides, and it contains `landed`
361/// let merged = merged_into_trunk(dir.path(), "main").unwrap();
362/// assert!(merged.contains("landed"));
363/// ```
364pub fn merged_into_trunk(primary: &Path, trunk: &str) -> Result<BTreeSet<String>> {
365    let mut merged = merged_into(primary, trunk)?;
366    let upstream = format!("{trunk}@{{upstream}}");
367    if let Ok(out) = git::run(
368        primary,
369        &[
370            "rev-parse",
371            "--abbrev-ref",
372            "--symbolic-full-name",
373            &upstream,
374        ],
375    ) {
376        let name = String::from_utf8_lossy(&out.stdout).trim().to_string();
377        if !name.is_empty() {
378            merged.extend(merged_into(primary, &name)?);
379        }
380    }
381    Ok(merged)
382}

384/// Every local branch reachable from `trunk`, in one subprocess.
385pub fn merged_into(primary: &Path, trunk: &str) -> Result<BTreeSet<String>> {
386    let out = git::run(
387        primary,
388        &[
389            "branch",
390            "--list",
391            "--merged",
392            trunk,
393            "--format=%(refname:short)",
394        ],
395    )?;
396    Ok(String::from_utf8_lossy(&out.stdout)
397        .lines()
398        .map(str::trim)
399        .filter(|l| !l.is_empty())
400        .map(str::to_string)
401        .collect())
402}

404/// The issue ids this repository's project model declares: the file names under
405/// `.ai/repo/project/issues/`, read from the primary checkout. A filesystem read, not an
406/// index build, so the topology stays cheap.
407pub fn issue_ids(primary: &Path) -> Vec<String> {
408    let dir = primary.join(".ai/repo/project/issues");
409    let Ok(entries) = std::fs::read_dir(dir) else {
410        return Vec::new();
411    };
412    let mut ids: Vec<String> = entries
413        .filter_map(|e| e.ok())
414        .filter_map(|e| {
415            let name = e.file_name().to_string_lossy().to_string();
416            name.strip_suffix(".yaml").map(str::to_string)
417        })
418        .filter(|id| !id.eq_ignore_ascii_case("readme"))
419        .collect();
420    crate::order::canonical(&mut ids);
421    ids
422}

424/// The issue a branch provably names: a path component equal to an issue id, or beginning
425/// with the id followed by `-`. Nothing else counts.
426///
427/// ```
428/// use majordomus_cli::worktree::state::issue_of;
429/// let ids = vec!["I0042".to_string(), "I1003".to_string()];
430/// assert_eq!(issue_of("feature/I1003-graph-kinds", &ids).as_deref(), Some("I1003"));
431/// assert_eq!(issue_of("fix/I0042", &ids).as_deref(), Some("I0042"));
432/// assert_eq!(issue_of("feature/graph-kinds", &ids), None);
433/// assert_eq!(issue_of("feature/I10030", &ids), None);
434/// ```
435pub fn issue_of(branch: &str, ids: &[String]) -> Option<String> {
436    for component in branch.split('/') {
437        for id in ids {
438            if component == id
439                || component
440                    .strip_prefix(id.as_str())
441                    .is_some_and(|rest| rest.starts_with('-'))
442            {
443                return Some(id.clone());
444            }
445        }
446    }
447    None
448}
