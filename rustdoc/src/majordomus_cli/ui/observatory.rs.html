1//! The Observatory: `/ui/observatory`, the repository and its mesh drawn as rotating
2//! scenes for a wall display.
3//!
4//! Nothing here is a second model. The screen asks the capabilities the other screens ask
5//! ([`SOURCES`]), projects each answer into the small, bounded model one scene draws
6//! ([`Projection`]), and puts that projection on the page as data. The script
7//! (`share/ui/observatory.js`) draws it and decides nothing: which scenes exist, what each
8//! needs, whether it is available and why not are all said here, once, in [`SCENES`].
9//!
10//! Three rules hold for every scene:
11//!
12//! - **a mark is a fact or it is decoration.** A node, a pulse, a bar or a ring is drawn from
13//!   a value a source answered; the slow background motion is decoration and the page says so;
14//! - **absent is not zero.** A scene whose source did not answer, or answered that it is not
15//!   running here, is unavailable with the source's own reason, and is never drawn empty;
16//! - **what is projected is what the screen may show.** An event is projected as its kind,
17//!   its stream and its clock, never its body: an intent's text, a path or a host name stays
18//!   in the capability that owns it, behind the link the scene carries.
19//!
20//! A published snapshot of the UI (ADR 0141) carries this page like any other: the
21//! projection is then the one reading taken at the published commit, the script reads nothing
22//! again, and the page says it is a snapshot. A process that runs no mesh, which is what
23//! reads a snapshot, has no mesh scene to show and says why.

254/// The answers the Observatory projects from, by source.
255type Answers = std::collections::BTreeMap<Source, Answer>;

257/// Read every source and project it.
258pub fn project(ctx: &Context) -> (Projection, Answers) {
259    let answers = ask_all(ctx, SOURCES);
260    let projection = project_from(
261        &answers,
262        crate::peers::rfc3339(std::time::SystemTime::now()),
263    );
264    (projection, answers)
265}
266
267/// The projection of a set of answers: a function of them alone, so the same answers give
268/// the same projection.
269pub fn project_from(answers: &Answers, observed_at: String) -> Projection {
270    let sources = SOURCES
271        .iter()
272        .filter_map(|s| answers.get(s))
273        .map(|a| SourceStanding {
274            capability: a.source.capability,
275            state: if a.value.is_ok() {
276                "answered"
277            } else {
278                "failed"
279            },
280            reason: a.value.as_ref().err().cloned(),
281            took_ms: a.took.as_millis() as u64,
282        })
283        .collect::<Vec<_>>();
284    let scenes = SCENES
285        .iter()
286        .map(|scene| {
287            let (mut model, mut reason) = match unanswered(scene, answers) {
288                Some(reason) => (None, Some(reason)),
289                None => match model(scene.id, answers, &sources) {
290                    Ok(model) => (Some(model), None),
291                    Err(reason) => (None, Some(reason)),
292                },
293            };
294            // a scene of the mesh in a process that runs none may draw the recording the
295            // repository carries, as a replay; a source that failed is not "no mesh here"
296            let mut replay = None;
297            if model.is_none() && no_mesh_here(scene, answers) {
298                if let Some((recorded, standing)) = recorded(scene.id, answers) {
299                    model = Some(recorded);
300                    reason = None;
301                    replay = Some(standing);
302                }
303            }
304            SceneProjection {
305                id: scene.id,
306                label: scene.label,
307                character: scene.character,
308                requires: scene.requires.iter().map(|s| s.capability).collect(),
309                meaning: scene.meaning,
310                href: scene.href,
311                dwell_seconds: scene.dwell_seconds,
312                available: model.is_some(),
313                reason,
314                model: model.unwrap_or(Value::Null),
315                replay,
316            }
317        })
318        .collect();
319    let revision = answers
320        .get(&REPOSITORY)
321        .and_then(|a| a.value.as_ref().ok())
322        .and_then(|v| v.pointer("/repository/git/head"))
323        .and_then(Value::as_str)
324        .map(|h| h[..12.min(h.len())].to_string());
325    Projection {
326        schema: SCHEMA,
327        observed_at,
328        revision,
329        sources,
330        scenes,
331    }
332}

334/// The first required source that did not answer, as the reason the scene is unavailable.
335fn unanswered(scene: &Scene, answers: &Answers) -> Option<String> {
336    scene.requires.iter().find_map(|s| match answers.get(s) {
337        None => Some(format!("{} was not asked", s.capability)),
338        Some(a) => a
339            .value
340            .as_ref()
341            .err()
342            .map(|e| format!("{} did not answer: {e}", s.capability)),
343    })
344}

346fn answer(answers: &Answers, source: Source) -> &Value {
347    answers
348        .get(&source)
349        .and_then(|a| a.value.as_ref().ok())
350        .unwrap_or(&Value::Null)

353fn items<'a>(value: &'a Value, pointer: &str) -> &'a [Value] {
354    value
355        .pointer(pointer)
356        .and_then(Value::as_array)
357        .map(Vec::as_slice)
358        .unwrap_or(&[])
359}

361fn word(value: &Value, key: &str) -> String {
362    value

365        .unwrap_or_default()
366        .to_string()

369/// A source that says it is not running here: its own reason, or the plain fact.
370/// Whether every source the scene requires answered, and one of them answered that no mesh
371/// runs in this process: the one case a recording may stand in for.
372fn no_mesh_here(scene: &Scene, answers: &Answers) -> bool {
373    crate::capability::builtin::observatory::RECORDED.contains(&scene.id)
374        && scene.requires.iter().all(|s| {
375            answers.get(s).is_some_and(|a| {
376                a.value
377                    .as_ref()
378                    .is_ok_and(|v| v.get("active") == Some(&Value::Bool(false)))
379            })
380        })
381}

383/// The repository's recording of one scene, with when it was recorded.
384fn recorded(id: &str, answers: &Answers) -> Option<(Value, ReplayStanding)> {
385    let replay = answers.get(&REPLAY)?.value.as_ref().ok()?;
386    let recording = replay.get("recording")?;
387    let model = recording.get("scenes")?.get(id)?.clone();
388    let standing = ReplayStanding {
389        recorded_at: recording.get("recorded_at")?.as_str()?.to_string(),
390        revision: recording
391            .get("revision")
392            .and_then(Value::as_str)
393            .map(str::to_string),
394        sources: recording
395            .pointer("/manifest/sources")?
396            .as_array()?
397            .iter()
398            .filter_map(|v| v.as_str().map(str::to_string))
399            .collect(),
400    };
401    Some((model, standing))
402}

404fn inactive(value: &Value, capability: &str) -> Option<String> {
405    (value.get("active").and_then(Value::as_bool) != Some(true)).then(|| {
406        match value.get("reason").and_then(Value::as_str) {
407            Some(reason) => format!("{capability}: {reason}"),
408            None => format!("{capability} reports no mesh runtime active in this process"),
409        }
410    })
411}

413/// The first twelve characters of an identity: enough to tell two apart on a wall, and the
414/// whole of what a mark is labelled with.
415fn short(id: &str) -> String {
416    id.chars().take(12).collect()
417}
418
419/// The last twelve characters of a compound identity (`node-runtime`, `node-runtime/session`):
420/// the part that tells two runtimes of one machine apart, which its first characters do not.
421fn tail(id: &str) -> String {
422    let plain: Vec<char> = id.chars().filter(char::is_ascii_alphanumeric).collect();
423    plain[plain.len().saturating_sub(12)..].iter().collect()
424}

426/// The marks of a page of `mesh.events`, newest first and at most [`BOUND`]: an event is a mark
427/// by its identity (stream and sequence), its kind, its node, its logical clock and its time,
428/// and never its body. One that states no sequence has no identity and is not drawn under an
429/// invented one. The scene and `observatory.events` both project through this, so a mark the
430/// stream delivers is the mark the page would have drawn.
431pub fn event_marks(listed: &[Value]) -> Vec<Value> {
432    listed
433        .iter()
434        .rev()
435        .take(BOUND)
436        .filter_map(|e| {
437            let stream = word(e, "stream");
438            let seq = e.get("seq").and_then(Value::as_u64)?;
439            Some(serde_json::json!({
440                "id": format!("{stream}:{seq}"),
441                "kind": e.pointer("/body/kind").and_then(Value::as_str).unwrap_or("unknown"),
442                "node": short(&stream),
443                "lamport": e.get("lamport").and_then(Value::as_u64),
444                "at": e.get("at").and_then(Value::as_u64),
445            }))
446        })
447        .collect()
448}

450/// The model one scene draws, or why there is nothing true to draw.
451fn model(id: &str, answers: &Answers, sources: &[SourceStanding]) -> Result<Value, String> {
452    match id {
453        "mesh" => {
454            let peers = answer(answers, PEERS);
455            if let Some(reason) = inactive(peers, PEERS.capability) {
456                return Err(reason);
457            }
458            let machines = items(peers, "/machines");
459            if machines.is_empty() {
460                return Err("mesh.peers knows no machine".into());
461            }
462            let nodes: Vec<Value> = machines
463                .iter()
464                .take(BOUND)
465                .map(|m| {
466                    let runtimes: Vec<Value> = items(m, "/runtimes")
467                        .iter()
468                        .take(BOUND)
469                        .map(|r| {
470                            serde_json::json!({
471                                "id": tail(&word(r, "runtime")),
472                                "liveness": word(r, "liveness"),
473                                "this": r.get("this_runtime").and_then(Value::as_bool).unwrap_or(false),
474                                "sessions": items(r, "/sessions").len(),
475                            })
476                        })
477                        .collect();
478                    serde_json::json!({
479                        "id": word(m, "node"),
480                        "label": word(m, "name"),
481                        "local": m.get("local").and_then(Value::as_bool),
482                        "runtimes": runtimes,
483                    })
484                })
485                .collect();
486            Ok(serde_json::json!({
487                "machines": nodes,
488                "known": machines.len(),
489                "refused": items(peers, "/refused").len(),
490            }))
491        }
492        "activity" => {
493            let events = answer(answers, EVENTS);
494            if let Some(reason) = inactive(events, EVENTS.capability) {
495                return Err(reason);
496            }
497            let marks = event_marks(items(events, "/events"));
498            Ok(serde_json::json!({
499                "events": marks,
500                "lamport": events.get("lamport").and_then(Value::as_u64),
501            }))
502        }
503        "intents" => {
504            let roadmap = answer(answers, ROADMAP);
505            let milestones = items(roadmap, "/milestones");
506            if milestones.is_empty() {
507                return Err("plan.roadmap holds no milestone".into());
508            }
509            let nodes: Vec<Value> = milestones
510                .iter()
511                .take(BOUND)
512                .map(|m| {
513                    let id = word(m, "id");
514                    serde_json::json!({
515                        "id": id,
516                        "label": word(m, "title"),
517                        "status": word(m, "status"),
518                        "done": m.pointer("/counts/by_status/DONE").and_then(Value::as_u64),
519                        "total": m.pointer("/counts/total").and_then(Value::as_u64),
520                        "depends_on": items(m, "/depends_on"),
521                        "href": format!("{}?milestone={id}", super::roadmap::ROUTE),
522                    })
523                })
524                .collect();
525            Ok(serde_json::json!({
526                "milestones": nodes,
527                "now": roadmap.get("now"),
528                "next": roadmap.get("next"),
529            }))
530        }
531        "tokens" => {
532            let runs = items(answer(answers, ECONOMICS), "/runs");
533            if runs.is_empty() {
534                return Err(
535                    "economics.runs holds no recorded run, so there is no usage to draw".into(),
536                );
537            }
538            Ok(serde_json::json!({ "runs": runs.iter().take(BOUND).collect::<Vec<_>>() }))
539        }
540        "heartbeat" => {
541            let health = answer(answers, HEALTH);
542            let checks = items(health, "/checks");
543            if checks.is_empty() {
544                return Err("health.report lists no dimension".into());
545            }
546            Ok(serde_json::json!({
547                "status": health.get("status"),
548                "checks": checks.iter().take(BOUND).map(|c| serde_json::json!({
549                    "id": word(c, "id"),
550                    "label": word(c, "title"),
551                    "status": word(c, "status"),
552                })).collect::<Vec<_>>(),
553            }))
554        }
555        "delivery" => {
556            let releases = items(answer(answers, RELEASES), "/releases");
557            if releases.is_empty() {
558                return Err("distribution.releases records no release".into());
559            }
560            // the next version is the trunk's own claim, drawn only when its source answered
561            let version = answer(answers, VERSION);
562            Ok(serde_json::json!({
563                "releases": releases.iter().take(BOUND).map(|r| serde_json::json!({
564                    "version": word(r, "version"),
565                    "published_at": word(r, "published_at"),
566                    "yanked": r.get("yanked").and_then(Value::as_bool).unwrap_or(false),
567                })).collect::<Vec<_>>(),
568                "count": releases.len(),
569                "next": version.get("next"),
570                "changes": version.get("changes").and_then(Value::as_array).map(Vec::len),
571            }))
572        }
573        "coop" => {
574            let state = answer(answers, STATE);
575            if let Some(reason) = inactive(state, STATE.capability) {
576                return Err(reason);
577            }
578            let sessions = items(state, "/state/sessions");
579            if sessions.is_empty() {
580                return Err("mesh.state holds no session".into());
581            }
582            let of = |list: &[Value], key: &str, session: &str| {
583                list.iter().filter(|c| word(c, key) == session).count()
584            };
585            let claims = items(state, "/state/claims");
586            let reviews = items(state, "/state/reviews");
587            Ok(serde_json::json!({
588                "sessions": sessions.iter().take(BOUND).map(|s| {
589                    let key = word(s, "key");
590                    serde_json::json!({
591                        "id": tail(&key),
592                        "node": short(&word(s, "node")),
593                        "state": word(s, "state"),
594                        "claims": of(claims, "session", &key),
595                        "reviews": of(reviews, "session", &key),
596                    })
597                }).collect::<Vec<_>>(),
598                "claims": claims.len(),
599                "overlaps": items(state, "/state/overlaps").len(),
600                "handovers": items(state, "/state/handovers").len(),
601                "reviews": reviews.len(),
602            }))
603        }
604        "ambient" => Ok(serde_json::json!({
605            "sources": sources.iter().map(|s| serde_json::json!({
606                "capability": s.capability,
607                "state": s.state,
608            })).collect::<Vec<_>>(),
609        })),
610        other => Err(format!("no projection is written for the scene {other}")),
611    }
612}

614/// The Observatory screen: the stage the script draws on, the projection it draws, and the
615/// same facts as a table for a reader without the script.
616pub fn screen(ctx: &Context, query: &[(String, String)], lib: &str, kiosk: bool) -> View {
617    // the scene a reader asked for by name; a name that is no scene selects nothing
618    let selected = query
619        .iter()
620        .find(|(k, _)| k == "scene")
621        .and_then(|(_, v)| SCENES.iter().find(|s| s.id == v))
622        .map(|s| s.id);
623    let (projection, answers) = project(ctx);
624    let available = projection.scenes.iter().filter(|s| s.available).count();
625    let top = hero(
626        "eye",
627        "violet",
628        "Observatory",
629        "The repository and its mesh, drawn from what the sources answered.",
630        &format!(
631            "{available} of {} scenes have something true to draw; the others say why not.",
632            projection.scenes.len()
633        ),
634        Vec::new(),
635    );
636
637    let stage = el("section")
638        .class("mj-obs")
639        .attr("aria-label", "Observatory")
640        .attr(
641            "data-observatory",
642            serde_json::to_string(&projection).unwrap_or_else(|_| "{}".into()),
643        )
644        .attr("data-observatory-lib", lib)
645        .attr_if("data-observatory-scene", selected)
646        .attr("data-observatory-mode", if kiosk { "kiosk" } else { "screen" })
647        .child(
648            el("canvas")
649                .class("mj-obs-canvas")
650                .attr("role", "img")
651                .attr(
652                    "aria-label",
653                    "The scenes listed below, drawn; every fact on it is in the table",
654                )
655                .flag("hidden"),
656        )
657        .child(
658            el("p")
659                .class("mj-obs-noscript mj-ui-dim")
660                .text("The scenes are drawn by the page's script. Without it, the table below holds every fact a scene would show."),
661        );
662
663    let rows = projection
664        .scenes
665        .iter()
666        .map(|s| {
667            el("tr")
668                .attr("data-scene", s.id)
669                .child(el("td").child(link(format!("{ROUTE}?scene={}", s.id), s.label)))
670                .child(el("td").text(match (s.available, s.character) {
671                    (true, Character::Semantic) => "available",
672                    (true, Character::Ambient) => "decorative",
673                    (false, _) => "unavailable",
674                }))
675                .child(el("td").text(s.reason.clone().unwrap_or_else(|| s.meaning.to_string())))
676                .child(el("td").class("mj-ui-mono").text(if s.requires.is_empty() {
677                    "none".to_string()
678                } else {
679                    s.requires.join(", ")
680                }))
681                .child(el("td").child(link(s.href, "rows")))
682        })
683        .collect::<Vec<_>>();
684    let asked: Vec<&Answer> = SOURCES.iter().filter_map(|s| answers.get(s)).collect();
685    let scenes = panel(
686        "Scenes",
687        "Every scene, whether it has something true to draw, and what its marks mean.",
688        table(
689            &[
690                "Scene",
691                "State",
692                "Meaning, or why not",
693                "Requires",
694                "As rows",
695            ],
696            rows,
697            "No scene is declared.",
698        ),
699        &asked,
700    );
701
702    let other = if kiosk {
703        link(ROUTE, "Open it as a screen of the UI")
704    } else {
705        link(
706            KIOSK,
707            "Open it as a kiosk: the same display, filling the window, unattended",
708        )
709    };
710    View::new(
711        if kiosk {
712            "Observatory kiosk"
713        } else {
714            "Observatory"
715        },
716        el("div")
717            .class("mj-ui-stack")
718            .child(top)
719            .child(stage)
720            .child(el("p").class("mj-ui-dim").child(other))
721            .child(scenes),
722    )
723}

733        Answer {
734            source,
735            route: None,
736            exposure: Exposure::default(),
737            value,
738            took: Duration::from_millis(3),

743    fn all(with: &[(Source, Result<Value, String>)]) -> Answers {
744        SOURCES
745            .iter()
746            .map(|s| {
747                let value = with
748                    .iter()
749                    .find(|(w, _)| w == s)
750                    .map(|(_, v)| v.clone())
751                    .unwrap_or(Ok(serde_json::json!({})));
752                (*s, answered(*s, value))
753            })
754            .collect()
755    }
