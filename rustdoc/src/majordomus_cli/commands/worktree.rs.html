1//! `majordomus worktree`: the command line's rendering of the worktree service.
2//!
3//! Every subcommand calls [`crate::worktree::WorktreeService`] directly — the same service
4//! the capability registry's `worktree.*` handlers call — and prints the same typed
5//! answers. It does not build the repository's index: a guard that runs on every commit
6//! and a status that runs on every directory change cannot afford a two-second startup,
7//! and nothing about the topology is in the index anyway. The one exception is
8//! `create --issue`, which reads the issue's record from the index to name the branch.
9//!
10//! Nothing here decides anything: no path is derived, no standing is judged, no safety
11//! check is made in this file. It reads arguments, calls the service, and prints. The human
12//! form and the JSON form are two renderings of one typed answer.
13

27/// Run `majordomus worktree`.
28pub fn run(args: WorktreeArgs) -> Result<u8> {
29    let start = match &args.repo.repo {
30        Some(p) => p.clone(),
31        None => std::env::current_dir().map_err(|e| Error::io(".", e))?,
32    };
33    let stdout = std::io::stdout();
34    let mut out = stdout.lock();
35    let format = args.format;
36    let svc = || WorktreeService::open(&start).map_err(refuse);
37    match args.command {
38        None | Some(WorktreeCommand::Status) => status(&svc()?, format, &mut out),
39        Some(WorktreeCommand::List) => list(&svc()?, format, &mut out),
40        Some(WorktreeCommand::Topology) => topology(&svc()?, format, &mut out),
41        Some(WorktreeCommand::Root) => root(&svc()?, format, &mut out),
42        Some(WorktreeCommand::Path { branch }) => path(&svc()?, &branch, &mut out),
43        Some(WorktreeCommand::Inspect { branch }) => inspect(&svc()?, &branch, format, &mut out),
44        Some(WorktreeCommand::Create {
45            branch,
46            base,
47            issue,
48        }) => {
49            let branch = match (branch, issue) {
50                (Some(b), _) => b,
51                (None, Some(id)) => issue_branch(&args.repo, &id)?,
52                (None, None) => {
53                    return Err(Error::Refused {
54                        code: crate::cli::EXIT_USAGE,
55                        reason: "worktree create: a branch or --issue is required".into(),
56                    })
57                }
58            };
59            create(&svc()?, &branch, base, false, format, &mut out)
60        }
61        Some(WorktreeCommand::Ensure { branch, base }) => {
62            create(&svc()?, &branch, base, true, format, &mut out)
63        }
64        Some(WorktreeCommand::Migrate {
65            plan,
66            dry_run,
67            allow_copy,
68            only,
69            include_ephemeral,
70        }) => migrate_cmd(
71            &svc()?,
72            !(plan || dry_run),
73            MigrationOptions {
74                allow_copy,
75                only,
76                include_ephemeral,
77            },
78            format,
79            &mut out,
80        ),
81        Some(WorktreeCommand::Validate) => doctor(&svc()?, format, true, &mut out),
82        Some(WorktreeCommand::Doctor) => doctor(&svc()?, format, false, &mut out),
83        Some(WorktreeCommand::Guard { quiet }) => guard(&svc()?, format, quiet, &mut out),
84        Some(WorktreeCommand::Repair { dry_run }) => repair(&svc()?, dry_run, format, &mut out),
85        Some(WorktreeCommand::Remove { selector, force }) => {
86            remove(&svc()?, &selector, force, format, &mut out)
87        }
88        Some(WorktreeCommand::Cleanup) => cleanup(&svc()?, format, &mut out),
89        Some(WorktreeCommand::Branches { without_worktree }) => {
90            branches(&svc()?, without_worktree, &mut out)
91        }
92    }
93}

97fn w(out: &mut Out<'_>, s: impl AsRef<str>) -> Result<()> {
98    writeln!(out, "{}", s.as_ref()).map_err(Error::Transport)
99}

101fn json<T: Serialize>(out: &mut Out<'_>, v: &T) -> Result<()> {
102    let value = serde_json::to_value(v).unwrap_or(Value::Null);
103    w(
104        out,
105        serde_json::to_string_pretty(&value).unwrap_or_else(|_| value.to_string()),
106    )
107}

109/// A refusal from the service becomes this executable's error with the service's own words
110/// and its own exit code. Nothing is rephrased and no debug struct reaches a person.
111fn refuse(e: crate::worktree::WorktreeError) -> Error {
112    Error::Refused {
113        code: e.exit_code(),
114        reason: e.to_string(),
115    }

119    head.as_deref()

121        .unwrap_or_else(|| "-".into())

133fn root(svc: &WorktreeService, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
134    let t = svc.topology(Detail::Fast).map_err(refuse)?;
135    match format {
136        OutputFormat::Json => json(out, &t.container)?,
137        // One path and nothing else: this output is consumed by `cd "$(...)"`.
138        OutputFormat::Text => w(out, &t.container.path)?,
139    }
140    Ok(0)

149fn render_worktree_line(w_: &WorktreeState) -> String {
150    let standing = w_.standing.as_str().to_uppercase();
151    let here = if w_.current { "  (you are here)" } else { "" };
152    let mut line = format!("{standing:<10} {:<44} {}{here}", w_.label, w_.path);
153    if let Some(e) = &w_.expected_path {
154        if w_.standing == Standing::Misplaced {
155            line.push_str(&format!("\n           -> belongs at {e}"));
156        }
157    }
158    if let Some(d) = &w_.dirty {
159        if !d.clean {
160            line.push_str(&format!("\n           dirty: {}", d.summary()));
161        }
162    }
163    if let Some(u) = &w_.upstream {
164        if u.gone || u.ahead.unwrap_or(0) > 0 || u.behind.unwrap_or(0) > 0 {
165            line.push_str(&format!(
166                "\n           upstream {}: {}",
167                u.name,
168                if u.gone {
169                    "gone".to_string()
170                } else {
171                    format!(
172                        "ahead {}, behind {}",
173                        u.ahead.unwrap_or(0),
174                        u.behind.unwrap_or(0)
175                    )
176                }
177            ));
178        }
179    }
180    if let Some(i) = &w_.issue {
181        line.push_str(&format!("\n           issue {i}"));
182    }
183    line
184}

186fn render_diagnostic(d: &TopologyDiagnostic) -> String {
187    let mut s = format!(
188        "{:<8} {}  {}",
189        d.severity.as_str().to_uppercase(),
190        d.code.as_str(),
191        d.message
192    );
193    if let Some(p) = &d.path {
194        s.push_str(&format!("\n         at {p}"));
195    }
196    if let Some(e) = &d.expected {
197        s.push_str(&format!("\n         expected {e}"));
198    }
199    s.push_str(&format!("\n         [remedy: {}]", d.remedy));
200    s
201}

203fn header(t: &RepositoryTopology, out: &mut Out<'_>) -> Result<()> {
204    w(
205        out,
206        format!("repository  {}", t.repository.primary_worktree),
207    )?;
208    w(
209        out,
210        format!(
211            "container   {}{}",
212            t.container.path,
213            if t.container.exists {
214                ""
215            } else {
216                "  (not created yet)"
217            }
218        ),
219    )?;
220    w(
221        out,
222        format!(
223            "trunk       {}  ({})",
224            t.trunk.branch.as_deref().unwrap_or("(unknown)"),
225            trunk_source_word(t.trunk.source)
226        ),
227    )?;
228    Ok(())
229}

231fn trunk_source_word(source: crate::worktree::TrunkSource) -> &'static str {
232    use crate::worktree::TrunkSource::*;
233    match source {
234        RemoteHead => "the remote's HEAD",
235        DefaultBranchConfig => "init.defaultBranch",
236        ConventionalName => "the conventional name",
237        PrimaryCheckout => "the primary checkout's branch",
238        Unknown => "unknown",
239    }

242fn list(svc: &WorktreeService, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
243    let t = svc.topology(Detail::Full).map_err(refuse)?;
244    match format {
245        OutputFormat::Json => json(out, &t.worktrees)?,
246        OutputFormat::Text => {
247            header(&t, out)?;
248            w(out, "")?;
249            for w_ in &t.worktrees {
250                w(out, render_worktree_line(w_))?;
251            }
252            w(out, "")?;
253            w(out, tallies_line(&t))?;
254        }
255    }
256    Ok(if t.valid { 0 } else { EXIT_REFUSED })
257}

259fn tallies_line(t: &RepositoryTopology) -> String {
260    let ta = &t.tallies;
261    format!(
262        "{} worktree(s): {} canonical, {} misplaced, {} detached, {} missing; {} branch(es), {} without a worktree, {} cleanup-eligible; {} error(s), {} warning(s)",
263        ta.worktrees, ta.canonical, ta.misplaced, ta.detached, ta.missing, ta.branches,
264        ta.branches_without_worktree, ta.cleanup_eligible, ta.errors, ta.warnings
265    )
266}

268fn topology(svc: &WorktreeService, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
269    let t = svc.topology(Detail::Full).map_err(refuse)?;
270    match format {
271        OutputFormat::Json => json(out, &t)?,
272        OutputFormat::Text => {
273            header(&t, out)?;
274            w(out, "")?;
275            w(out, "worktrees")?;
276            for w_ in &t.worktrees {
277                w(
278                    out,
279                    format!("  {}", render_worktree_line(w_).replace('\n', "\n  ")),
280                )?;
281            }
282            let without: Vec<&BranchState> =
283                t.branches.iter().filter(|b| b.worktree.is_none()).collect();
284            if !without.is_empty() {
285                w(out, "")?;
286                w(out, "branches without a worktree")?;
287                for b in without {
288                    w(
289                        out,
290                        format!(
291                            "  {:<44} {}{}",
292                            b.name,
293                            short(&Some(b.head.clone())),
294                            if b.cleanup_eligible {
295                                "  merged, cleanup-eligible"
296                            } else if b.merged_into_trunk == Some(true) {
297                                "  merged"
298                            } else {
299                                ""
300                            }
301                        ),
302                    )?;
303                }
304            }
305            if !t.diagnostics.is_empty() {
306                w(out, "")?;
307                w(out, "diagnostics")?;
308                for d in &t.diagnostics {
309                    w(
310                        out,
311                        format!("  {}", render_diagnostic(d).replace('\n', "\n  ")),
312                    )?;
313                }
314            }
315            w(out, "")?;
316            w(out, tallies_line(&t))?;
317        }
318    }
319    Ok(if t.valid { 0 } else { EXIT_REFUSED })
320}

322fn status(svc: &WorktreeService, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
323    let s: StatusReport = svc.status().map_err(refuse)?;
324    match format {
325        OutputFormat::Json => json(out, &s)?,
326        OutputFormat::Text => {
327            let wt = &s.worktree;
328            w(out, format!("branch      {}", wt.label))?;
329            w(
330                out,
331                format!(
332                    "worktree    {} {}",
333                    if s.canonical { "ok " } else { "!! " },
334                    wt.standing.as_str()
335                ),
336            )?;
337            w(out, format!("path        {}", wt.path))?;
338            if let Some(e) = &wt.expected_path {
339                if !s.canonical {
340                    w(out, format!("expected    {e}"))?;
341                }
342            }
343            w(out, format!("head        {}", short(&wt.head)))?;
344            w(out, format!("dirty       {}", dirty_word(&wt.dirty)))?;
345            if let Some(u) = &wt.upstream {
346                w(
347                    out,
348                    format!(
349                        "upstream    {} ({})",
350                        u.name,
351                        if u.gone {
352                            "gone".into()
353                        } else {
354                            format!(
355                                "ahead {}, behind {}",
356                                u.ahead.unwrap_or(0),
357                                u.behind.unwrap_or(0)
358                            )
359                        }
360                    ),
361                )?;
362            }
363            if let Some(i) = &wt.issue {
364                w(out, format!("issue       {i}"))?;
365            }
366            w(out, format!("container   {}", s.container.path))?;
367            w(
368                out,
369                format!(
370                    "trunk       {} ({})",
371                    s.trunk.branch.as_deref().unwrap_or("(unknown)"),
372                    trunk_source_word(s.trunk.source)
373                ),
374            )?;
375            for d in wt
376                .diagnostics
377                .iter()
378                .filter(|d| d.severity != Severity::Info)
379            {
380                w(out, render_diagnostic(d))?;
381            }
382            if s.repository_errors > 0 {
383                w(
384                    out,
385                    format!(
386                        "repository  {} topology error(s) (majordomus worktree doctor)",
387                        s.repository_errors
388                    ),
389                )?;
390            }
391        }
392    }
393    Ok(if s.canonical { 0 } else { EXIT_REFUSED })
394}

396fn inspect(
397    svc: &WorktreeService,
398    branch: &str,
399    format: OutputFormat,
400    out: &mut Out<'_>,
401) -> Result<u8> {
402    let r = svc.inspect(branch).map_err(refuse)?;
403    match format {
404        OutputFormat::Json => json(out, &r)?,
405        OutputFormat::Text => {
406            w(
407                out,
408                format!(
409                    "branch      {}{}",
410                    r.branch,
411                    if r.branch_exists {
412                        ""
413                    } else {
414                        "  (does not exist yet)"
415                    }
416                ),
417            )?;
418            w(out, format!("expected    {}", r.expected_path))?;
419            match &r.worktree {
420                Some(wt) => {
421                    w(
422                        out,
423                        format!("worktree    {} ({})", wt.path, wt.standing.as_str()),
424                    )?;
425                    w(out, format!("head        {}", short(&wt.head)))?;
426                    w(out, format!("dirty       {}", dirty_word(&wt.dirty)))?;
427                }
428                None => w(
429                    out,
430                    format!(
431                        "worktree    none{}",
432                        if r.destination_exists {
433                            "  (something occupies the canonical path)"
434                        } else {
435                            ""
436                        }
437                    ),
438                )?,
439            }
440            for d in r
441                .diagnostics
442                .iter()
443                .filter(|d| d.severity != Severity::Info)
444            {
445                w(out, render_diagnostic(d))?;
446            }
447            if r.worktree.is_none() && !r.destination_exists {
448                w(
449                    out,
450                    format!("create      majordomus worktree create {}", r.branch),
451                )?;
452            }
453        }
454    }
455    Ok(0)
456}

458fn doctor(
459    svc: &WorktreeService,
460    format: OutputFormat,
461    quiet: bool,
462    out: &mut Out<'_>,
463) -> Result<u8> {
464    let t = svc.topology(Detail::Fast).map_err(refuse)?;
465    match format {
466        OutputFormat::Json => json(out, &t.diagnostics)?,
467        OutputFormat::Text => {
468            if !quiet {
469                header(&t, out)?;
470                w(out, "")?;
471            }
472            for d in t
473                .diagnostics
474                .iter()
475                .filter(|d| !quiet || d.severity == Severity::Error)
476            {
477                w(out, render_diagnostic(d))?;
478            }
479            w(
480                out,
481                format!(
482                    "worktree topology: {} — {} error(s), {} warning(s)",
483                    if t.valid { "valid" } else { "INVALID" },
484                    t.tallies.errors,
485                    t.tallies.warnings
486                ),
487            )?;
488        }
489    }
490    Ok(if t.valid { 0 } else { EXIT_REFUSED })
491}

493fn guard(
494    svc: &WorktreeService,
495    format: OutputFormat,
496    quiet: bool,
497    out: &mut Out<'_>,
498) -> Result<u8> {
499    let v = svc.guard().map_err(refuse)?;
500    match format {
501        OutputFormat::Json => json(out, &v)?,
502        OutputFormat::Text => {
503            if v.ok {
504                if !quiet {
505                    w(
506                        out,
507                        format!(
508                            "worktree guard: ok — {} at {}{}",
509                            v.branch.as_deref().unwrap_or("(detached)"),
510                            v.path,
511                            if v.exempt { " (exempt)" } else { "" }
512                        ),
513                    )?;
514                }
515            } else {
516                w(out, "worktree guard: REFUSED")?;
517                if let Some(r) = &v.reason {
518                    w(out, render_diagnostic(r))?;
519                }
520            }
521        }
522    }
523    Ok(if v.ok { 0 } else { EXIT_REFUSED })
524}

526fn cleanup(svc: &WorktreeService, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
527    let t = svc.topology(Detail::Full).map_err(refuse)?;
528    let eligible: Vec<&BranchState> = t.branches.iter().filter(|b| b.cleanup_eligible).collect();
529    match format {
530        OutputFormat::Json => json(out, &eligible)?,
531        OutputFormat::Text => {
532            if eligible.is_empty() {
533                w(out, "nothing is cleanup-eligible: no branch is both merged into the trunk and clean or unchecked-out")?;
534            } else {
535                w(out, "merged into the trunk, and clean or not checked out — nothing is deleted here:")?;
536                for b in &eligible {
537                    w(
538                        out,
539                        format!(
540                            "  {:<44} {}",
541                            b.name,
542                            match &b.worktree {
543                                Some(p) => format!(
544                                    "majordomus worktree remove {}; then git branch -d {}   ({p})",
545                                    b.name, b.name
546                                ),
547                                None => format!("git branch -d {}", b.name),
548                            }
549                        ),
550                    )?;
551                }
552            }
553        }
554    }
555    Ok(0)
556}

558fn branches(svc: &WorktreeService, without_worktree: bool, out: &mut Out<'_>) -> Result<u8> {
559    let t = svc.topology(Detail::Fast).map_err(refuse)?;
560    for b in t
561        .branches
562        .iter()
563        .filter(|b| !without_worktree || b.worktree.is_none())

566    }
567    Ok(0)

572fn create(
573    svc: &WorktreeService,
574    branch: &str,
575    base: Option<String>,
576    ensure: bool,
577    format: OutputFormat,
578    out: &mut Out<'_>,
579) -> Result<u8> {
580    let request = CreateRequest {
581        branch: branch.to_string(),
582        base,
583    };
584    let report = if ensure {
585        svc.ensure(&request)
586    } else {
587        svc.create(&request)
588    }
589    .map_err(refuse)?;
590    match format {
591        OutputFormat::Json => json(out, &report)?,
592        OutputFormat::Text => {
593            if report.container_created {
594                w(out, format!("created the container {}", report.container))?;
595            }
596            if report.existed {
597                w(out, format!("exists  {}", report.path))?;
598            } else {
599                w(out, format!("created {}", report.path))?;
600            }
601            match (&report.base, report.branch_created) {
602                (Some(b), true) => w(out, format!("branch  {} (new, from {b})", report.branch))?,
603                _ => w(out, format!("branch  {}", report.branch))?,
604            }
605            w(out, report.envrc.describe())?;
606            w(out, report.envrc_local.describe())?;
607            w(
608                out,
609                format!("cd \"$(majordomus worktree path {})\"", report.branch),
610            )?;
611        }
612    }
613    Ok(0)
614}

616/// The branch an issue's work happens on: `feature/<id>-<slug>`, from the issue record
617/// this repository keeps. The id is the provable link the topology reads back.
618fn issue_branch(repo: &crate::cli::RepoArgs, id: &str) -> Result<String> {
619    let app = crate::app::App::load(repo)?;
620    let object = app
621        .index()
622        .objects
623        .iter()
624        .find(|o| o.kind == "issue" && o.identity == id)
625        .ok_or_else(|| Error::Refused {
626            code: crate::worktree::EXIT_MISSING,
627            reason: format!(
628                "no issue '{id}' in this repository's project model; nothing was created and no issue number was invented"
629            ),
630        })?;
631    let slug = object
632        .metadata
633        .get("slug")
634        .and_then(Value::as_str)
635        .map(str::to_string)
636        .unwrap_or_else(|| id.to_lowercase());
637    Ok(format!("feature/{id}-{slug}"))
638}

640fn migrate_cmd(
641    svc: &WorktreeService,
642    apply: bool,
643    options: MigrationOptions,
644    format: OutputFormat,
645    out: &mut Out<'_>,
646) -> Result<u8> {
647    let plan = if apply {
648        migrate::apply(svc, &options).map_err(refuse)?
649    } else {
650        migrate::plan_with(svc, options.include_ephemeral).map_err(refuse)?
651    };
652    match format {
653        OutputFormat::Json => json(out, &plan)?,
654        OutputFormat::Text => render_plan(&plan, out)?,
655    }
656    // A plan never fails for having found something to do. An apply that left a step
657    // blocked or failed says so through its exit code.
658    Ok(if apply && (plan.blocked > 0 || plan.failed > 0) {
659        EXIT_REFUSED
660    } else {
661        0
662    })
663}

665fn render_plan(plan: &MigrationPlan, out: &mut Out<'_>) -> Result<()> {
666    let verb = if plan.applied {
667        "migrated"
668    } else {
669        "to migrate"
670    };
671    if plan.steps.is_empty() {
672        w(
673            out,
674            format!(
675                "nothing {verb}: every worktree with a branch is at its canonical path under {}",
676                plan.container
677            ),
678        )?;
679    }
680    for step in &plan.steps {
681        w(out, "")?;
682        w(out, step.branch.as_str())?;
683        w(out, format!("  from    {}", step.from))?;
684        w(out, format!("  to      {}", step.to))?;
685        w(out, format!("  head    {}", short(&step.head)))?;
686        w(out, format!("  state   {}", step.dirty.summary()))?;
687        let action = match step.action {
688            MigrationAction::Move => "move",
689            MigrationAction::MoveViaStaging => "move out of the container path, then into it",
690            MigrationAction::CopyAndRepair => {
691                "copy across filesystems, repair, verify, remove the original"
692            }
693            MigrationAction::None => "none",
694        };
695        match step.outcome {
696            StepOutcome::Planned => w(out, format!("  action  {action}"))?,
697            StepOutcome::Moved => {
698                w(
699                    out,
700                    format!(
701                        "  outcome moved and verified ({action}){}",
702                        step.message
703                            .as_deref()
704                            .map(|m| format!(": {m}"))
705                            .unwrap_or_default()
706                    ),
707                )?;
708                if let Some(envrc) = &step.envrc {
709                    w(out, format!("  {}", envrc.describe()))?;
710                }
711            }
712            StepOutcome::Blocked => {
713                w(out, "  outcome BLOCKED")?;
714                for b in &step.blockers {
715                    w(
716                        out,
717                        format!("    {}", render_diagnostic(b).replace('\n', "\n    ")),
718                    )?;
719                }
720                if let Some(m) = &step.message {
721                    w(out, format!("    {m}"))?;
722                }
723            }
724            StepOutcome::Failed => {
725                w(
726                    out,
727                    format!(
728                        "  outcome FAILED: {}",
729                        step.message.as_deref().unwrap_or("-")
730                    ),
731                )?;
732                for d in &step.differences {
733                    w(out, format!("    {d}"))?;
734                }
735            }
736        }
737    }
738    if !plan.exceptions.is_empty() {
739        w(out, "")?;
740        w(out, "not migrated by design")?;
741        for d in &plan.exceptions {
742            w(
743                out,
744                format!("  {}", render_diagnostic(d).replace('\n', "\n  ")),
745            )?;
746        }
747    }
748    w(out, "")?;
749    if plan.applied {
750        w(
751            out,
752            format!(
753                "{} step(s): {} moved and verified, {} blocked, {} failed",
754                plan.steps.len(),
755                plan.moved,
756                plan.blocked,
757                plan.failed
758            ),
759        )?;
760        if let Some(p) = &plan.moved_current {
761            w(out, format!("the worktree you were in moved: cd \"{p}\""))?;
762        }
763    } else {
764        w(
765            out,
766            format!(
767                "{} step(s): {} movable, {} blocked; nothing was changed. `majordomus worktree migrate` applies the movable ones",
768                plan.steps.len(),
769                plan.movable,
770                plan.blocked
771            ),
772        )?;
773    }
774    Ok(())
775}

777fn repair(
778    svc: &WorktreeService,
779    dry_run: bool,
780    format: OutputFormat,
781    out: &mut Out<'_>,
782) -> Result<u8> {
783    let report = svc.repair(dry_run).map_err(refuse)?;
784    match format {
785        OutputFormat::Json => json(out, &report)?,
786        OutputFormat::Text => {
787            if report.pruned.is_empty() {
788                w(
789                    out,
790                    "nothing to prune: every registered worktree still exists",
791                )?;
792            } else {
793                for line in &report.pruned {
794                    w(out, line)?;
795                }
796                w(
797                    out,
798                    format!(
799                        "{} registration(s) {}",
800                        report.pruned.len(),
801                        if report.applied {
802                            "dropped"
803                        } else {
804                            "would be dropped; nothing was changed"
805                        }
806                    ),
807                )?;
808            }
809            for line in &report.repaired {
810                w(out, line)?;
811            }
812            if report.applied {
813                w(
814                    out,
815                    "administrative links repaired where git found them stale",
816                )?;
817            }
818        }
819    }
820    Ok(0)
821}
822
823fn remove(
824    svc: &WorktreeService,
825    selector: &str,
826    force: bool,
827    format: OutputFormat,
828    out: &mut Out<'_>,
829) -> Result<u8> {
830    // Say what is about to go, and what it holds, before it goes: a removal that prints its
831    // subject only afterwards has taught its lesson too late. The service refuses dirty
832    // work without `force` regardless; this line is for the person who typed `--force`.
833    if format == OutputFormat::Text {
834        let record = svc.resolve(selector).map_err(refuse)?;
835        let held = record
836            .branch
837            .clone()
838            .unwrap_or_else(|| "a detached HEAD".into());
839        let dirty = if record.path.is_dir() {
840            crate::worktree::state::dirty_state(&record.path)
841                .map(|d| d.summary())
842                .unwrap_or_else(|_| "unknown".into())
843        } else {
844            "directory missing".into()
845        };
846        w(
847            out,
848            format!(
849                "removing {} (holding {held}; {dirty}){}",
850                record.path.display(),
851                if force { "; forced" } else { "" }
852            ),
853        )?;
854    }
855    let report = svc.remove(selector, force).map_err(refuse)?;
856    match format {
857        OutputFormat::Json => json(out, &report)?,
858        OutputFormat::Text => {
859            w(out, format!("removed {}", report.path))?;
860            if let Some(b) = &report.branch {
861                w(
862                    out,
863                    format!("branch  {b} still exists (git branch -d {b} removes it)"),
864                )?;
865            }
866        }
867    }
868    Ok(0)
869}
