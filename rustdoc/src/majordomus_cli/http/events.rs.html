1//! The live channel: one WebSocket over which this process reports what its executions are
2//! doing.
3//!
4//! It is an adapter and holds no execution logic. It subscribes to
5//! [`crate::execution::ExecutionStore`], renders what arrives, and closes; every decision
6//! about what an execution is, what may happen to it and what an event means was already
7//! taken in [`crate::execution`]. That is what makes an audit trail, a notifier or a
8//! second protocol something that subscribes here rather than something that has to be
9//! threaded through the engine.
10//!
11//! # The protocol
12//!
13//! ```text
14//! GET /events                                  every execution of this process, live
15//! GET /events?execution=<id>                   one execution, its whole retained history, then live
16//! GET /events?execution=<id>&after=<sequence>  the same, from a cursor: what a reconnect asks for
17//! GET /events?execution=<a>&execution=<b>      several executions, each its history, then live
18//! ```
19//!
20//! The subscription is the request target, so there is nothing to send and nothing to
21//! acknowledge: a reconnect is a new URL with the sequence the client last saw, and the
22//! server replays the gap before it goes live.
23//!
24//! Sequences are dense per execution, so a connection that follows several executions keeps
25//! a cursor for each. The one `after` starts every one of those cursors: each execution
26//! named is replayed from the events after it, up to [`MAX_REPLAY`] each. A client that
27//! reconnects to several executions sends the lowest sequence it saw among them and drops,
28//! per execution, what it already holds; with one execution that is exactly the sequence it
29//! last saw, as it always was. What the server writes is one JSON document
30//! per text frame, each with a `type` and a `data`: either a
31//! [`crate::execution::ExecutionEvent`], or one of the [`StreamMessage`] control messages
32//! below. A client that does not recognise a `type` must ignore that frame; that is what
33//! lets a variant be added without a version change.
34//!
35//! The server pings on a heartbeat and never reads a frame. A client that goes away is
36//! noticed within [`LOOK`] where the server can ask the connection's socket (the read
37//! deadline's sweeper finds it, `super::deadline`), and otherwise when the next write
38//! fails, which the heartbeat guarantees happens; a client that sends anything is ignored,
39//! and the reason is in [`super::ws`]. Beside the ping, which is a
40//! control frame a browser never shows to a page, the same quiet interval writes a
41//! `stream.heartbeat` text frame, so a client can tell a quiet channel from a dead one.
42//!
43//! A replay is paged until it is caught up: a cursor further behind than one page
44//! ([`MAX_REPLAY`]) receives every retained event after it, not the first page and then
45//! the live stream with the rest missing. An event whose frame would exceed
46//! [`ws::MAX_FRAME_BYTES`] is written with its unbounded value — a completed execution's
47//! `output`, a created one's `input` — replaced by a [`Withheld`] reference naming where the
48//! whole value is read, rather than refused and taken for a client that went away.
49

81/// A message about the stream itself, rather than about an execution.
82///
83/// Same envelope as an event — a `type` and a `data` — so a client has one reader.
84#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
85#[serde(tag = "type", content = "data")]
86pub enum StreamMessage {
87    /// The first frame of every connection: what was subscribed to, what was replayed, and
88    /// what the client should expect.
89    #[serde(rename = "stream.ready")]
90    Ready {
91        /// The event protocol this server speaks, [`PROTOCOL_VERSION`].
92        protocol_version: String,
93        /// What this connection follows.
94        scope: Scope,
95        /// The executions followed, for a scoped connection.
96        #[serde(skip_serializing_if = "Vec::is_empty")]
97        executions: Vec<String>,
98        /// How many retained events were written before this connection went live, over
99        /// every execution it follows.
100        replayed: usize,
101        /// Whether the history before the replay had already been dropped by the store's
102        /// bound, for any execution followed, so a client knows its picture starts in the
103        /// middle.
104        truncated: bool,
105        /// How many seconds of quiet before the server pings.
106        heartbeat_seconds: u64,
107    },
108    /// The channel is quiet and alive: written with the ping on every quiet heartbeat, as
109    /// a text frame a page can see, so a client that has heard nothing for a few heartbeats
110    /// knows the connection is dead and reconnects from its cursor.
111    #[serde(rename = "stream.heartbeat")]
112    Heartbeat {
113        /// When it was written, RFC 3339 in UTC.
114        at: String,
115    },
116    /// The client fell behind and events were dropped rather than buffered. It must read
117    /// the snapshot and the history again, from the last sequence it saw.
118    #[serde(rename = "stream.lagged")]
119    Lagged {
120        /// How many events were dropped.
121        dropped: u64,
122    },
123    /// The server is closing this connection.
124    #[serde(rename = "stream.closing")]
125    Closing {
126        /// Why.
127        reason: String,
128    },
129}

131/// What a connection follows.
132#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
133#[serde(rename_all = "snake_case")]
134pub enum Scope {

136    All,
137    /// The executions named in the request target.
138    Executions,

141/// What an event carries in place of a value too large for one frame: why it was withheld,
142/// how large it is, and where the whole value is read.
143///
144/// The event is still the same typed event, with its sequence, so nothing about the
145/// execution's story is lost — only the value is read elsewhere, from the snapshot.
146///
147/// ```
148/// use majordomus_cli::http::events::Withheld;
149/// let w = Withheld {
150///     withheld: "too_large".into(),
151///     bytes: 2_000_000,
152///     limit: 1_048_576,
153///     itself: "/api/v1/executions/get?id=e1".into(),
154/// };
155/// let json = serde_json::to_value(&w).unwrap();
156/// assert_eq!(json["self"], "/api/v1/executions/get?id=e1", "the reference is `self` on the wire");
157/// assert!(json.get("itself").is_none());
158/// assert_eq!(serde_json::from_value::<Withheld>(json).unwrap(), w);
159/// ```
160#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
161pub struct Withheld {
162    /// Why: always `too_large`.
163    pub withheld: String,
164    /// The size of the value as JSON, in bytes.
165    pub bytes: usize,
166    /// The largest frame this server writes, [`ws::MAX_FRAME_BYTES`].
167    pub limit: usize,
168    /// The snapshot that carries the whole value: `GET /api/v1/executions/get?id=…`.
169    #[serde(rename = "self")]
170    pub itself: String,
171}

188impl Accepted {
189    /// Serve the connection on an upgraded socket until it ends. Blocking; the caller runs
190    /// it on a thread of its own.
191    pub fn serve(self, socket: Box<dyn std::io::Write + Send>) {
192        self.stream.run(socket, || false);
193    }
194
195    /// [`Accepted::serve`], also ending as soon as `hung_up` says the client has gone: asked
196    /// every [`LOOK`] while the channel is quiet, so a closed channel stops counting as use
197    /// of the server within a second, not at its next write (I2131).
198    ///
199    /// A client that has already gone ends the channel at the first look, long before the
200    /// heartbeat would have found out by writing:
201    ///
202    /// ```
203    /// use std::sync::Arc;
204    /// use std::time::{Duration, Instant};
205    /// use majordomus_cli::execution::{ExecutionStore, Limits};
206    /// use majordomus_cli::http::events::{accept, PATH};
207    /// use majordomus_cli::http::Request;
208    ///
209    /// let store = Arc::new(ExecutionStore::new(Limits::default()));
210    /// let req = Request::parse_target("GET", PATH, vec![]).with_headers(vec![
211    ///     ("Upgrade".into(), "websocket".into()),
212    ///     ("Connection".into(), "Upgrade".into()),
213    ///     ("Sec-WebSocket-Key".into(), "dGhlIHNhbXBsZSBub25jZQ==".into()),
214    ///     ("Sec-WebSocket-Version".into(), "13".into()),
215    /// ]);
216    /// let accepted = accept(&store, &req, "/api/v1/executions/get").expect("an upgrade");
217    /// let started = Instant::now();
218    /// accepted.serve_until(Box::new(Vec::new()), || true);
219    /// assert!(started.elapsed() < Duration::from_secs(5), "a gone client ends the channel");
220    /// ```
221    pub fn serve_until(self, socket: Box<dyn std::io::Write + Send>, hung_up: impl Fn() -> bool) {
222        self.stream.run(socket, hung_up);
223    }
224}

226/// How often a quiet channel asks whether its client has hung up: the idle timer's own tick,
227/// so a channel that closes is no longer counted as use by the next time the timer looks.
228pub const LOOK: Duration = Duration::from_secs(1);

230struct LiveStream {
231    store: Arc<ExecutionStore>,
232    filter: Filter,
233    after: u64,
234    /// How long the channel stays quiet before it pings and writes a heartbeat.
235    heartbeat: Duration,
236    /// The snapshot route a [`Withheld`] reference names, from the registry.
237    snapshot: String,
238}

240/// Decide what a request to [`PATH`] means: an accepted connection, or the response to
241/// send instead.
242///
243/// The origin is checked here for the same reason it is checked for a state-changing
244/// request: a page on another origin can open a WebSocket to this server, and the
245/// same-origin policy does not stop it. This does.
246///
247/// `snapshot` is the route of `executions.get`, as the registry declares it: what a
248/// [`Withheld`] reference names as the place the whole value is read.
249pub fn accept(
250    store: &Arc<ExecutionStore>,
251    req: &Request,
252    snapshot: &str,
253) -> Result<Accepted, Response> {
254    if !ws::is_upgrade(req) {
255        return Err(Response::error(
256            426,
257            "upgrade_required",
258            &format!(
259                "{PATH} is a WebSocket. Open it with a WebSocket client; the same events are readable over HTTP at {}executions/events",
260                crate::capability::model::HttpExposure::PREFIX
261            ),
262        ));
263    }
264    if let Some(origin) = req.header("origin") {
265        let host = req.header("host").unwrap_or_default();
266        let ours = [format!("http://{host}"), format!("https://{host}")];
267        if !ours.iter().any(|a| a == origin) {
268            tracing::warn!(
269                origin = origin,
270                "a live channel from another origin was refused"
271            );
272            return Err(Response::error(
273                403,
274                "forbidden",
275                &format!("a WebSocket from origin '{origin}' is refused: this server accepts one from its own origin only"),
276            ));
277        }
278    }
279    let accept = match ws::handshake(req) {
280        ws::Handshake::Accept(value) => value,
281        ws::Handshake::Refuse { status, reason } => {
282            return Err(Response::error(status, "invalid_input", &reason))
283        }
284    };
285    if OPEN.load(Ordering::Relaxed) >= MAX_CONNECTIONS {
286        return Err(Response::error(
287            503,
288            "unavailable",
289            &format!("this process serves {MAX_CONNECTIONS} live channels at once and they are all in use"),
290        ));
291    }
292    let mut executions = Vec::new();
293    let mut after = 0u64;
294    for (key, value) in &req.query {
295        match key.as_str() {
296            "execution" => match ExecutionId::parse(value) {
297                // named twice is followed once, and replayed once
298                Some(id) if executions.contains(&id) => {}
299                Some(id) => executions.push(id),
300                None => {
301                    return Err(Response::error(
302                        400,
303                        "invalid_input",
304                        &format!("'{value}' is not an execution id"),
305                    ))
306                }
307            },
308            "after" => match value.parse::<u64>() {
309                Ok(n) => after = n,
310                Err(_) => {
311                    return Err(Response::error(
312                        400,
313                        "invalid_input",
314                        &format!("after '{value}' is not a sequence number"),
315                    ))
316                }
317            },
318            other => {
319                return Err(Response::error(
320                    400,
321                    "invalid_input",
322                    &format!("unknown parameter '{other}'; {PATH} takes 'execution' and 'after'"),
323                ))
324            }
325        }
326    }
327    let filter = if executions.is_empty() {
328        Filter::All
329    } else {
330        Filter::Only(executions)
331    };
332    Ok(Accepted {
333        accept,
334        stream: LiveStream {
335            store: Arc::clone(store),
336            filter,
337            after,
338            heartbeat: HEARTBEAT,
339            snapshot: snapshot.to_string(),
340        },
341    })
342}

344impl LiveStream {
345    fn run(self, mut socket: Box<dyn std::io::Write + Send>, hung_up: impl Fn() -> bool) {
346        let open = OPEN.fetch_add(1, Ordering::Relaxed) + 1;
347        tracing::debug!(connections = open, "a live channel opened");
348        let (subscriber, rx) = self.store.subscribe(self.filter.clone());
349        // subscribe first, replay second, and skip in the live stream anything the replay
350        // already wrote: that is what leaves no gap between the history and the stream and
351        // no event written twice. Sequences are dense per execution, so the mark of what was
352        // written is one cursor per execution followed, each starting at `after`: a single
353        // mark across executions would take one execution's events for ones already written
354        // as soon as another had gone further.
355        let mut written: HashMap<ExecutionId, u64> = HashMap::new();
356        let mut replayed = Vec::new();
357        let mut truncated = false;
358        match &self.filter {
359            Filter::Only(ids) => {
360                // each cursor is set as its replay is collected rather than as it is written:
361                // a socket that fails mid-replay ends the connection before the live stream
362                // reads a cursor, so the two only differ on a connection that is already gone
363                for id in ids {
364                    let (events, cut) = self.replay(id);
365                    truncated |= cut;
366                    let cursor = events.iter().map(|e| e.sequence).fold(self.after, u64::max);
367                    replayed.extend(events);
368                    written.insert(id.clone(), cursor);
369                }
370            }
371            // an all-executions connection has no cursor to replay from; a client that needs
372            // history asks for it per execution over HTTP
373            Filter::All => {}
374        }
375        let ready = StreamMessage::Ready {
376            protocol_version: PROTOCOL_VERSION.to_string(),
377            scope: match self.filter {
378                Filter::All => Scope::All,
379                Filter::Only(_) => Scope::Executions,
380            },
381            executions: match &self.filter {
382                Filter::All => Vec::new(),
383                Filter::Only(ids) => ids.iter().map(|i| i.to_string()).collect(),
384            },
385            replayed: replayed.len(),
386            truncated,
387            heartbeat_seconds: HEARTBEAT.as_secs(),
388        };
389        let mut alive = write(&mut socket, &ready);
390        for event in replayed {
391            if !alive {
392                break;
393            }
394            alive = self.write_event(&mut socket, &event);
395        }
396        // quiet since: the heartbeat is written after this long without a frame, and the
397        // wait for the next delivery is cut into looks at whether the client has hung up
398        let mut quiet = std::time::Instant::now();
399        while alive {
400            let until_heartbeat = self.heartbeat.saturating_sub(quiet.elapsed());
401            match rx.recv_timeout(until_heartbeat.min(LOOK)) {
402                Ok(Delivery::Event(event)) => {
403                    // only a scoped connection has cursors; an unscoped one replayed nothing
404                    if let Some(cursor) = written.get_mut(&event.execution_id) {
405                        if event.sequence <= *cursor {
406                            continue; // the replay already wrote it
407                        }
408                        *cursor = event.sequence;
409                    }
410                    alive = self.write_event(&mut socket, &event);
411                    quiet = std::time::Instant::now();
412                }
413                Ok(Delivery::Lagged { dropped }) => {
414                    alive = write(&mut socket, &StreamMessage::Lagged { dropped });
415                    quiet = std::time::Instant::now();
416                }
417                Err(std::sync::mpsc::RecvTimeoutError::Timeout) if hung_up() => {
418                    tracing::debug!("the client of a live channel hung up");
419                    break;
420                }
421                Err(std::sync::mpsc::RecvTimeoutError::Timeout)
422                    if quiet.elapsed() < self.heartbeat => {}
423                Err(std::sync::mpsc::RecvTimeoutError::Timeout) => {
424                    // the ping keeps intermediaries and the protocol honest; the heartbeat
425                    // is the frame a page can see, so a dead channel is told from a quiet one
426                    alive = ws::write_ping(&mut socket).is_ok()
427                        && write(
428                            &mut socket,
429                            &StreamMessage::Heartbeat {
430                                at: crate::peers::rfc3339(std::time::SystemTime::now()),
431                            },
432                        );
433                    quiet = std::time::Instant::now();
434                }
435                Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => {
436                    let _ = write(
437                        &mut socket,
438                        &StreamMessage::Closing {
439                            reason: "this server is shutting down".into(),
440                        },
441                    );
442                    let _ = ws::write_close(&mut socket, 1001, "going away");
443                    break;
444                }
445            }
446        }
447        self.store.unsubscribe(subscriber);
448        let open = OPEN.fetch_sub(1, Ordering::Relaxed) - 1;
449        tracing::debug!(connections = open, "a live channel closed");
450    }
451
452    /// Every retained event of one execution after the cursor, page by page until caught
453    /// up, and whether the history before the first had already been dropped.
454    ///
455    /// One page is [`MAX_REPLAY`]; the whole replay is bounded by the store's own
456    /// retention, so the loop ends. Taking only the first page would leave the events
457    /// between it and the moment of subscription unwritten, with `stream.ready` claiming
458    /// nothing was missing.
459    fn replay(&self, id: &ExecutionId) -> (Vec<Arc<ExecutionEvent>>, bool) {
460        let Some(first) = self.store.events(id, self.after, MAX_REPLAY) else {
461            return (Vec::new(), false);
462        };
463        let truncated = first.truncated;
464        let mut more = first.more;
465        let mut cursor = first.last_sequence;
466        let mut events = first.events;
467        while more {
468            let Some(page) = self.store.events(id, cursor, MAX_REPLAY) else {
469                break;
470            };
471            if page.events.is_empty() {
472                break;
473            }
474            more = page.more;
475            cursor = page.last_sequence;
476            events.extend(page.events);
477        }
478        (events, truncated)
479    }
480
481    /// Write one execution event; one too large for a frame is written with its unbounded
482    /// value replaced by a [`Withheld`] reference, so the channel stays open.
483    fn write_event(
484        &self,
485        socket: &mut Box<dyn std::io::Write + Send>,
486        event: &ExecutionEvent,
487    ) -> bool {
488        let Ok(text) = serde_json::to_string(event) else {
489            tracing::error!("an execution event could not be rendered as JSON");
490            return true;
491        };
492        if text.len() <= ws::MAX_FRAME_BYTES {
493            return write_frame(socket, &text);
494        }
495        match self.withhold(event) {
496            Some(lighter) => write(socket, &lighter),
497            None => {
498                // no payload but the two unbounded values can reach the bound; one that
499                // does is skipped rather than taken for a client that went away, and the
500                // client's gap check reads it from the history over HTTP
501                tracing::warn!(
502                    event = %event.event_id,
503                    bytes = text.len(),
504                    "an execution event too large for one frame was not written"
505                );
506                true
507            }
508        }
509    }
510
511    /// The event with its unbounded value replaced by a reference, when it has one.
512    fn withhold(&self, event: &ExecutionEvent) -> Option<ExecutionEvent> {
513        let reference = |value: &serde_json::Value| {
514            let bytes = serde_json::to_string(value).map_or(0, |t| t.len());
515            serde_json::to_value(Withheld {
516                withheld: "too_large".into(),
517                bytes,
518                limit: ws::MAX_FRAME_BYTES,
519                itself: format!(
520                    "{}?id={}",
521                    self.snapshot,
522                    super::router::percent_encode(event.execution_id.as_str())
523                ),
524            })
525            .unwrap_or_default()
526        };
527        let payload = match &event.payload {
528            EventPayload::Completed { output } => EventPayload::Completed {
529                output: reference(output),
530            },
531            EventPayload::Created {
532                capability,
533                title,
534                input,
535            } => EventPayload::Created {
536                capability: capability.clone(),
537                title: title.clone(),
538                input: reference(input),
539            },
540            _ => return None,
541        };
542        Some(ExecutionEvent {
543            payload,
544            ..event.clone()
545        })
546    }
547}

549/// Write one value as a text frame; `false` when the client is gone.
550fn write<T: Serialize>(socket: &mut Box<dyn std::io::Write + Send>, value: &T) -> bool {
551    let Ok(text) = serde_json::to_string(value) else {
552        tracing::error!("an execution event could not be rendered as JSON");
553        return true;
554    };
555    write_frame(socket, &text)
556}
557
558/// Write one rendered document as a text frame; `false` when the client is gone.
559fn write_frame(socket: &mut Box<dyn std::io::Write + Send>, text: &str) -> bool {
560    match ws::write_text(socket.as_mut(), text) {
561        Ok(()) => true,
562        Err(e) => {
563            tracing::debug!("a live channel went away: {e}");
564            false
565        }
566    }

1141    /// A `Write` that keeps the WebSocket frames it was given and then goes away, so the
1142    /// connection loop can be run without a socket. It decodes only what this server
1143    /// writes: unmasked, unfragmented frames.
1144    #[derive(Clone)]
1145    struct SharedSink {
1146        bytes: Arc<std::sync::Mutex<Vec<u8>>>,
1147        /// How many frames it accepts before the writes start failing, as a closed tab's
1148        /// socket does.
1149        accepts: Arc<AtomicUsize>,
1150    }
1151
1152    impl SharedSink {
1153        fn new(accepts: usize) -> Self {
1154            SharedSink {
1155                bytes: Arc::new(std::sync::Mutex::new(Vec::new())),
1156                accepts: Arc::new(AtomicUsize::new(accepts)),
1157            }
1158        }
1159
1160        fn frames(&self) -> Vec<String> {
1161            let bytes = self.bytes.lock().unwrap();
1162            let mut out = Vec::new();
1163            let mut i = 0usize;
1164            while i + 2 <= bytes.len() {
1165                let opcode = bytes[i] & 0x0f;
1166                let short = bytes[i + 1] as usize;
1167                let (len, header) = match short {
1168                    126 => (u16::from_be_bytes([bytes[i + 2], bytes[i + 3]]) as usize, 4),
1169                    127 => (
1170                        u64::from_be_bytes(bytes[i + 2..i + 10].try_into().unwrap()) as usize,
1171                        10,
1172                    ),
1173                    n => (n, 2),
1174                };
1175                let start = i + header;
1176                let end = start + len;
1177                if end > bytes.len() {
1178                    break;
1179                }
1180                if opcode == 0x1 {
1181                    out.push(String::from_utf8_lossy(&bytes[start..end]).into_owned());
1182                }
1183                i = end;
1184            }
1185            out
1186        }
1187    }
1188
1189    impl std::io::Write for SharedSink {
1190        fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
1191            if self.accepts.load(Ordering::SeqCst) == 0 {
1192                return Err(std::io::Error::new(
1193                    std::io::ErrorKind::BrokenPipe,
1194                    "the client went away",
1195                ));
1196            }
1197            self.accepts.fetch_sub(1, Ordering::SeqCst);
1198            self.bytes.lock().unwrap().extend_from_slice(buf);
1199            Ok(buf.len())
1200        }
1201        fn flush(&mut self) -> std::io::Result<()> {
1202            Ok(())
1203        }
1204    }
