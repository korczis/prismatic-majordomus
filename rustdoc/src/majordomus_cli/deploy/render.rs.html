1//! The provider artifacts, rendered from the canonical deployment object: the container
2//! image definition, the ignore file that bounds its build context, and the provider
3//! configuration.
4//!
5//! None of the three is authoritative and none is written by hand. Each carries a header
6//! naming the object it came from and the command that regenerates it, and
7//! `majordomus generate --check` refuses a tree where one of them was edited. That is the
8//! whole point of the kind: the port, the routes, the resources, the machine count and the
9//! build inputs are stated once, and these files are views of that one statement.
10
11use crate::generate::{Artifact, ArtifactFormat};

13use super::Deployment;

15/// Where the image definition goes.
16pub const DOCKERFILE: &str = "deploy/Dockerfile";
17/// Where the build context's ignore file goes. Docker reads it from the context root, so
18/// it lives at the repository root whatever the image definition's own path is.
19pub const DOCKERIGNORE: &str = ".dockerignore";
20/// Where the provider configuration goes. Fly reads `fly.toml` from the working directory.
21pub const FLY_TOML: &str = "fly.toml";
22
23/// The Rust toolchain the builder stage pins. A floating tag would make the image a
24/// different thing on different days, which is the opposite of what a deployment is for.
25pub const RUST_IMAGE: &str = "rust:1.85-alpine3.21";
26/// The runtime base, pinned for the same reason.
27pub const RUNTIME_IMAGE: &str = "alpine:3.21";
28/// The unprivileged user the deployed process runs as.
29pub const RUN_USER: &str = "majordomus";
30/// Where the layer and the tool distribution live inside the image.
31pub const APP_DIR: &str = "/app";
32
33/// The comment header every generated artifact carries: what wrote it, from what, and how
34/// to change it. A contributor who edits the output is told which file to change instead.
35fn header(comment: &str, source: &str, what: &str) -> String {
36    // The first line is the one every generated artifact of this repository opens with;
37    // `generate --check` reads exactly that, and this renderer predates the contract.
38    format!(
39        "{comment} {header}\n\
40         {comment} {what}\n\
41         {comment} GENERATED by `majordomus generate deployment` from {source}.\n\
42         {comment} Do not edit: `majordomus generate --check` refuses a hand edit. Change\n\
43         {comment} the deployment object and regenerate.\n\n",
44        header = crate::generate::HEADER
45    )
46}

48/// The directory of the package the builder stage compiles: the one input that is a build
49/// input rather than something the running process reads.
50fn package_dir(d: &Deployment) -> Option<&str> {
51    d.build
52        .inputs
53        .iter()
54        .map(String::as_str)
55        .find(|p| p.rsplit('/').next() == Some(d.build.package.as_str()))
56}
57
58/// The inputs the running process reads: everything the build context carries except the
59/// package the builder compiles.
60fn runtime_inputs(d: &Deployment) -> Vec<&str> {
61    let pkg = package_dir(d);
62    d.build
63        .inputs
64        .iter()
65        .map(String::as_str)
66        .filter(|p| Some(*p) != pkg)
67        .collect()
68}

70/// The container image, as a multi-stage build: a builder that holds the toolchain and a
71/// runtime that holds the binary, the layer and nothing else. The runtime carries no
72/// compiler, no cargo registry, no repository history and no shell history — what is not
73/// copied in cannot leak out.
74pub fn dockerfile(d: &Deployment, source: &str) -> String {
75    let mut out = header(
76        "#",
77        source,
78        "The container image of this repository's deployment.",
79    );
80    let pkg = package_dir(d).unwrap_or("apps");
81    out.push_str(&format!(
82        "FROM {RUST_IMAGE} AS build\n\
83         RUN apk add --no-cache musl-dev\n\
84         WORKDIR /src\n\
85         COPY {pkg} ./{pkg}\n\
86         RUN cargo build --locked --{profile} --manifest-path {pkg}/Cargo.toml --package {package} --bin {binary}\n\n",
87        profile = d.build.profile,
88        package = d.build.package,
89        binary = d.build.binary,
90    ));
91    out.push_str(&format!(
92        "FROM {RUNTIME_IMAGE}\n\
93         # An unprivileged user, which is also why the declared port is above 1024.\n\
94         RUN adduser -D -H -u 10001 {RUN_USER}\n\
95         WORKDIR {APP_DIR}\n\
96         COPY --from=build /src/{pkg}/target/{profile}/{binary} /usr/local/bin/{binary}\n",
97        profile = d.build.profile,
98        binary = d.build.binary,
99    ));
100    for input in runtime_inputs(d) {
101        out.push_str(&format!("COPY {input} {APP_DIR}/{input}\n"));
102    }
103    out.push_str(&format!(
104        "USER {RUN_USER}\n\
105         EXPOSE {port}\n\
106         # The address, the port and the interface are the object's; this command names the\n\
107         # deployment and reads them from it. Discovery is a filesystem walk because the\n\
108         # image carries the layer without the repository that tracks it.\n\
109         ENTRYPOINT [\"{binary}\"]\n\
110         CMD [\"serve\", \"--discovery\", \"filesystem\", \"--deployment\", \"{id}\"]\n",
111        port = d.listen.port,
112        binary = d.build.binary,
113        id = d.id,
114    ));
115    out
116}

118/// The build context: nothing but the declared inputs. Everything is excluded and each
119/// input is let back in, so a file that appears in the repository tomorrow does not appear
120/// in the image tomorrow.
121pub fn dockerignore(d: &Deployment, source: &str) -> String {
122    let mut out = header(
123        "#",
124        source,
125        "What the deployment's build context carries: only the declared inputs.",
126    );
127    out.push_str("*\n");
128    for input in &d.build.inputs {
129        out.push_str(&format!("!{input}\n"));
130    }
131    // The image definition is not an input of the deployment; it is a view of it. It is
132    // let back in because a builder that reads the context needs the file it builds from.
133    out.push_str(&format!("!{DOCKERFILE}\n"));
134    out.push_str(
135        "\n# and never these, wherever they appear inside an input\n\
136         **/.git\n\
137         **/target\n\
138         **/node_modules\n\
139         **/.DS_Store\n\
140         .ai/local\n",
141    );
142    out
143}

145/// The provider configuration. Every value here is read from the object; nothing is stated
146/// twice, and the `[build]` section points at the generated image definition.
147pub fn fly_toml(d: &Deployment, source: &str) -> String {
148    let mut out = header(
149        "#",
150        source,
151        "The Fly.io configuration of this repository's deployment.",
152    );
153    out.push_str(&format!(
154        "app = \"{app}\"\n\
155         primary_region = \"{region}\"\n\n\
156         [build]\n\
157         dockerfile = \"{DOCKERFILE}\"\n\n\
158         [http_service]\n\
159         internal_port = {port}\n\
160         force_https = {https}\n\
161         auto_stop_machines = \"{stop}\"\n\
162         auto_start_machines = {start}\n\
163         min_machines_running = {min}\n\n",
164        app = d.application,
165        region = d.region,
166        port = d.listen.port,
167        https = d
168            .provider
169            .fly
170            .as_ref()
171            .and_then(|f| f.force_https)
172            .unwrap_or(true),
173        stop = if d.machines.autostop { "stop" } else { "off" },
174        start = d.machines.autostart,
175        min = d.machines.min_running,
176    ));
177    if let Some(c) = d.provider.fly.as_ref().and_then(|f| f.concurrency) {
178        out.push_str(&format!(
179            "  [http_service.concurrency]\n  type = \"requests\"\n  soft_limit = {}\n  hard_limit = {}\n\n",
180            c.soft_limit, c.hard_limit
181        ));
182    }
183    out.push_str(&format!(
184        "  [[http_service.checks]]\n\
185         \x20 grace_period = \"{grace}s\"\n\
186         \x20 interval = \"{interval}s\"\n\
187         \x20 method = \"GET\"\n\
188         \x20 timeout = \"{timeout}s\"\n\
189         \x20 path = \"{liveness}\"\n\n\
190         \x20 [[http_service.checks]]\n\
191         \x20 grace_period = \"{grace}s\"\n\
192         \x20 interval = \"{interval}s\"\n\
193         \x20 method = \"GET\"\n\
194         \x20 timeout = \"{timeout}s\"\n\
195         \x20 path = \"{readiness}\"\n\n",
196        grace = d.health.grace_seconds.unwrap_or(2),
197        interval = d.health.interval_seconds.map(|v| v.get()).unwrap_or(15),
198        timeout = d.health.timeout_seconds.map(|v| v.get()).unwrap_or(2),
199        liveness = d.health.liveness,
200        readiness = d.health.readiness,
201    ));
202    out.push_str(&format!(
203        "[[vm]]\n\
204         cpu_kind = \"{kind}\"\n\
205         cpus = {cpus}\n\
206         memory_mb = {memory}\n",
207        kind = match d.resources.cpu_kind {
208            super::CpuKind::Shared => "shared",
209            super::CpuKind::Performance => "performance",
210        },
211        cpus = d.resources.cpus,
212        memory = d.resources.memory_mb,
213    ));
214    out
215}

217/// The three artifacts of one deployment.
218pub fn artifacts(d: &Deployment, source: &str) -> Vec<Artifact> {
219    vec![
220        Artifact {
221            path: DOCKERFILE.into(),
222            document: "deployment".into(),
223            format: ArtifactFormat::Text,
224            schema: None,
225            source: source.into(),
226            content: dockerfile(d, source),
227        },
228        Artifact {
229            path: DOCKERIGNORE.into(),
230            document: "deployment".into(),
231            format: ArtifactFormat::Text,
232            schema: None,
233            source: source.into(),
234            content: dockerignore(d, source),
235        },
236        Artifact {
237            path: FLY_TOML.into(),
238            document: "deployment".into(),
239            format: ArtifactFormat::Text,
240            schema: None,
241            source: source.into(),
242            content: fly_toml(d, source),
243        },
244    ]
245}
