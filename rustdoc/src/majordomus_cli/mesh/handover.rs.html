1//! Handovers across the mesh. A handover is written on one machine by `majordomus
2//! handover` into that checkout's `.ai/local/state/handovers/`; this module is the bridge
3//! between that record and the journal, in both directions:
4//!
5//! - **publish** reads a local record — its front matter's facts and its Markdown body —
6//!   into the [`HandoverBody`] a journal event carries, bounded, identified by the digest
7//!   of its body;
8//! - **materialize** writes a consumed handover into the consuming checkout's handovers
9//!   directory as a record of the same schema, so `majordomus handover --resolve` on a
10//!   checkout of the same branch finds it through the ordinary resolution. It names the
11//!   repository the way the resolver compares it (the local git common directory) and the
12//!   origin runtime as its worktree — another checkout's record on the same branch, tier 1
13//!   — and never copies a path of the author's disk.
14//!
15//! Materializing the same handover twice writes one file.
16//!
17//! The publishing half needs nothing but the record on disk, which is what the round trip
18//! below exercises: the newest record of a checkout, read into the bounded body a journal
19//! event carries, identified by the digest of that body and carrying none of the paths of
20//! the machine that wrote it.
21//!
22//! ```
23//! use majordomus_cli::mesh::handover::{directory, latest, to_body};
24//!
25//! let checkout = tempfile::tempdir().unwrap();
26//! let dir = directory(checkout.path());
27//! std::fs::create_dir_all(&dir).unwrap();
28//! std::fs::write(
29//!     dir.join("20260915T100000Z--feature-x--abcdef1--0011.md"),
30//!     "---\nschema_version: 1\ntask_id: t-1\nbranch: feature/x\nhead: abcdef1234\n\
31//!      worktree: /checkout/of/the/author\n---\n\n# Objective\nship\n",
32//! )
33//! .unwrap();
34//!
35//! let newest = latest(checkout.path()).unwrap();
36//! let body = to_body(&newest, Some("#184".into()), None).unwrap();
37//! assert_eq!(body.branch.as_deref(), Some("feature/x"));
38//! assert_eq!(body.issue.as_deref(), Some("#184"), "the publisher's to say");
39//! // the author's disk never travels: the record's worktree is not part of the body
40//! assert!(!serde_json::to_string(&body).unwrap().contains("/checkout/of/the/author"));
41//! ```

49/// Where a checkout's handover records live. It is under `.ai/local/`, which is never
50/// committed, so a handover is a fact about one checkout until something — this module, or
51/// a person — moves it; both halves of this module agree on the location by calling here
52/// rather than by each spelling the path.
53///
54/// ```
55/// use majordomus_cli::mesh::handover::directory;
56///
57/// let root = tempfile::tempdir().unwrap();
58/// let dir = directory(root.path());
59/// assert!(dir.ends_with(".ai/local/state/handovers"));
60/// assert!(dir.starts_with(root.path()), "always inside the checkout it belongs to");
61/// ```
62pub fn directory(root: &Path) -> PathBuf {
63    root.join(".ai/local/state/handovers")
64}

66/// The newest handover record of a checkout: names begin with a compact UTC timestamp, so
67/// the greatest name is the latest record.
68///
69/// Sorting by name rather than by modification time is deliberate — a file copied or
70/// restored keeps the moment its handover describes, which is the question being asked,
71/// and not the moment the bytes landed. Anything that is not a handover of this checkout
72/// is passed over rather than reported: a file that is not Markdown, one whose name does
73/// not begin with a timestamp, and a symbolic link out of the directory, which is how
74/// something outside the checkout would otherwise get itself published.
75///
76/// ```
77/// use majordomus_cli::mesh::handover::{directory, latest};
78///
79/// let root = tempfile::tempdir().unwrap();
80/// let dir = directory(root.path());
81/// std::fs::create_dir_all(&dir).unwrap();
82/// assert!(latest(root.path()).is_err(), "an empty directory holds no newest record");
83///
84/// std::fs::write(dir.join("20260101T000000Z--a--0000000--00.md"), "---\n---\nold").unwrap();
85/// std::fs::write(dir.join("20260915T100000Z--b--0000000--01.md"), "---\n---\nnew").unwrap();
86/// std::fs::write(dir.join("README.md"), "not a handover").unwrap();
87/// let newest = latest(root.path()).unwrap();
88/// assert!(newest.ends_with("20260915T100000Z--b--0000000--01.md"));
89/// ```
90pub fn latest(root: &Path) -> Result<PathBuf, String> {
91    let dir = directory(root);
92    let real_dir = dir
93        .canonicalize()
94        .map_err(|e| format!("{}: {e}", dir.display()))?;
95    let entries = std::fs::read_dir(&dir).map_err(|e| format!("{}: {e}", dir.display()))?;
96    entries
97        .filter_map(|e| e.ok().map(|e| e.path()))
98        .filter(|p| {
99            p.extension().is_some_and(|x| x == "md")
100                && p.file_name()
101                    .and_then(|n| n.to_str())
102                    .is_some_and(|n| n.as_bytes().first().is_some_and(u8::is_ascii_digit))
103                // A symbolic link out of the directory is not a handover of this checkout:
104                // nothing outside it is ever published.
105                && p.canonicalize().is_ok_and(|real| real.starts_with(&real_dir))
106        })
107        .max()
108        .ok_or_else(|| format!("no handover record under {}", dir.display()))
109}

111/// A front-matter value from a peer: one line, no quote that could end the value.
112fn one_line(value: &str) -> String {
113    value
114        .chars()
115        .filter(|c| !c.is_control() && *c != '"')
116        .collect()
117}

119/// Split a record into its front matter (flat `key: value` lines; list items are skipped)
120/// and its body.
121///
122/// This is a reader of one shape, not a YAML parser: a handover's front matter is flat by
123/// contract, and the few facts the mesh needs from it are scalars. Nesting and list items
124/// are skipped rather than refused, so a record that carries more than this module
125/// understands still publishes.
126///
127/// ```
128/// use majordomus_cli::mesh::handover::read;
129///
130/// let dir = tempfile::tempdir().unwrap();
131/// let path = dir.path().join("record.md");
132/// std::fs::write(
133///     &path,
134///     "---\nbranch: feature/x\nchanged_files:\n  - apps/x.rs\n---\n\n# Objective\nship\n",
135/// )
136/// .unwrap();
137///
138/// let (front, body) = read(&path).unwrap();
139/// assert_eq!(front["branch"], "feature/x");
140/// assert!(!front.contains_key("apps/x.rs"), "a list item is not a fact of the front matter");
141/// assert!(body.starts_with("# Objective"));
142///
143/// // a file with no front matter is not a handover record
144/// std::fs::write(&path, "# Objective\nship\n").unwrap();
145/// assert!(read(&path).is_err());
146/// ```
147pub fn read(path: &Path) -> Result<(BTreeMap<String, String>, String), String> {
148    let text = std::fs::read_to_string(path).map_err(|e| format!("{}: {e}", path.display()))?;
149    split(&text).ok_or_else(|| format!("{}: no front matter", path.display()))
150}

152fn split(text: &str) -> Option<(BTreeMap<String, String>, String)> {
153    let rest = text.strip_prefix("---\n")?;
154    let end = rest.find("\n---\n")?;
155    let mut front = BTreeMap::new();
156    for line in rest[..end].lines() {
157        if line.starts_with(' ') || line.starts_with('-') {
158            continue;
159        }
160        if let Some((key, value)) = line.split_once(':') {
161            let value = value.trim().trim_matches('"').to_string();
162            front.insert(key.trim().to_string(), value);
163        }
164    }
165    Some((front, rest[end + 5..].trim_start_matches('\n').to_string()))
166}

168/// The journal body of a local record. `issue` and `milestone` are the publisher's to say:
169/// a handover record has no field for either.
170///
171/// What crosses is the body and a few facts about the work — never a path of the author's
172/// disk, because the receiving checkout is somewhere else and a path from another machine
173/// is at best noise. The body is bounded at [`MAX_HANDOVER_BYTES`] and an empty one is
174/// refused, since a handover whose body says nothing hands nothing over. A placeholder
175/// front-matter value (`none`, `DETACHED`) is read as the absence it stands for.
176///
177/// ```
178/// use majordomus_cli::mesh::handover::to_body;
179/// use majordomus_cli::mesh::journal::HandoverBody;
180///
181/// let dir = tempfile::tempdir().unwrap();
182/// let path = dir.path().join("20260915T100000Z--feature-x--abcdef1--0011.md");
183/// std::fs::write(
184///     &path,
185///     "---\ntask_id: t-1\nbranch: DETACHED\nhead: abcdef1234\nworktree: /home/someone/repo\n\
186///      ---\n\n# Objective\nship\n",
187/// )
188/// .unwrap();
189///
190/// let body = to_body(&path, Some("#184".into()), None).unwrap();
191/// assert_eq!(body.task.as_deref(), Some("t-1"));
192/// assert_eq!(body.branch, None, "DETACHED is the absence of a branch, not a branch");
193/// assert_eq!(body.issue.as_deref(), Some("#184"));
194/// assert_eq!(body.id, HandoverBody::digest_of(&body.body));
195/// assert!(!serde_json::to_string(&body).unwrap().contains("/home/someone"));
196///
197/// // a record whose body says nothing hands nothing over
198/// std::fs::write(&path, "---\ntask_id: t-1\n---\n\n").unwrap();
199/// assert!(to_body(&path, None, None).is_err());
200/// ```
201pub fn to_body(
202    path: &Path,
203    issue: Option<String>,
204    milestone: Option<String>,
205) -> Result<HandoverBody, String> {
206    let (front, body) = read(path)?;
207    if body.trim().is_empty() {
208        return Err(format!("{}: the handover has no body", path.display()));
209    }
210    if body.len() > MAX_HANDOVER_BYTES {
211        return Err(format!(
212            "{}: a body of {} bytes exceeds the {MAX_HANDOVER_BYTES}-byte bound a mesh handover carries",
213            path.display(),
214            body.len()
215        ));
216    }
217    let fact = |key: &str| {
218        front
219            .get(key)
220            .filter(|v| !v.is_empty() && *v != "none" && *v != "NONE")
221            .cloned()
222    };
223    Ok(HandoverBody {
224        id: HandoverBody::digest_of(&body),
225        task: fact("task_id"),
226        issue,
227        milestone,
228        branch: fact("branch").filter(|b| b != "DETACHED"),
229        head: fact("head"),
230        created_at: fact("created_at"),
231        name: path
232            .file_name()
233            .and_then(|n| n.to_str())
234            .map(str::to_string),
235        body,
236    })
237}

239/// The git common directory of `root`, spelled as the shell resolver spells the local
240/// repository id: absolute as git prints it, or joined to the root when relative.
241fn repository_id(root: &Path) -> Result<String, String> {
242    let out = std::process::Command::new("git")
243        .arg("-C")
244        .arg(root)
245        .args(["rev-parse", "--git-common-dir"])
246        .output()
247        .map_err(|e| format!("git: {e}"))?;
248    if !out.status.success() {
249        return Err(format!("{} is not a git work tree", root.display()));
250    }
251    let dir = String::from_utf8_lossy(&out.stdout).trim().to_string();
252    Ok(if dir.starts_with('/') {
253        dir
254    } else {
255        format!("{}/{dir}", root.display())
256    })
257}

259/// Write a consumed handover into `root`'s handovers directory, or return the record that
260/// already holds it. The record's worktree is `mesh:<origin runtime>`: it resolves as
261/// another checkout's handover on the same branch.
262///
263/// This is the half that turns a message into continuity: after it, `majordomus handover
264/// --resolve` on this checkout finds the other machine's handover through the ordinary
265/// resolution and nothing downstream has to know the mesh exists. It is written to be
266/// repeatable — consuming the same handover twice yields the one record — and defensively,
267/// because every value in it was written by another runtime: the front matter and the file
268/// name admit only what their parts are, so neither a key nor a path can be smuggled in
269/// through a branch name or a timestamp.
270///
271/// ```
272/// use majordomus_cli::mesh::handover::{directory, materialize, to_body};
273/// use majordomus_cli::mesh::journal::StreamId;
274/// use majordomus_cli::mesh::state::HandoverView;
275///
276/// let tmp = tempfile::tempdir().unwrap();
277/// let root = tmp.path().join("checkout");
278/// std::fs::create_dir_all(&root).unwrap();
279/// assert!(std::process::Command::new("git")
280///     .arg("-C").arg(&root).args(["init", "-q"]).status().unwrap().success());
281///
282/// let src = tmp.path().join("20260915T100000Z--feature-x--abcdef1--0011.md");
283/// std::fs::write(&src, "---\nbranch: feature/x\nhead: abcdef1234\n---\n\n# Objective\nship\n")
284///     .unwrap();
285/// let stream = StreamId::parse(&format!("{}-{}-{}", "a".repeat(32), "b".repeat(16), "c".repeat(16)))
286///     .unwrap();
287/// let view = HandoverView {
288///     id: to_body(&src, None, None).unwrap().id,
289///     stream,
290///     runtime: format!("{}-{}", "a".repeat(32), "b".repeat(16)),
291///     published_lamport: 1,
292///     handover: to_body(&src, None, None).unwrap(),
293///     consumed_by: vec![],
294/// };
295///
296/// let written = materialize(&root, &view).unwrap();
297/// assert_eq!(written.parent(), Some(directory(&root).as_path()));
298/// assert_eq!(materialize(&root, &view).unwrap(), written, "one handover, one record");
299/// assert!(std::fs::read_to_string(&written).unwrap().contains("worktree: mesh:"));
300/// ```
301pub fn materialize(root: &Path, view: &HandoverView) -> Result<PathBuf, String> {
302    let dir = directory(root);
303    std::fs::create_dir_all(&dir).map_err(|e| format!("{}: {e}", dir.display()))?;
304    let marker = format!("mesh_handover: {}", view.id);
305    if let Ok(entries) = std::fs::read_dir(&dir) {
306        for entry in entries.flatten() {
307            let path = entry.path();
308            if path.extension().is_some_and(|x| x == "md")
309                && std::fs::read_to_string(&path).is_ok_and(|t| t.lines().any(|l| l == marker))
310            {
311                return Ok(path);
312            }
313        }
314    }
315    // Everything below comes from another runtime. Validation refused multi-line values at
316    // ingest; this writes defensively anyway, because a file name and a front matter built
317    // from a peer's words are exactly where a path or a key would be smuggled.
318    let h = &view.handover;
319    let branch = one_line(h.branch.as_deref().unwrap_or("DETACHED"));
320    let head = one_line(h.head.as_deref().unwrap_or("NONE"));
321    let created = one_line(
322        &h.created_at
323            .clone()
324            .unwrap_or_else(|| crate::peers::rfc3339(std::time::SystemTime::now())),
325    );
326    let mut text = String::new();
327    text.push_str("---\nschema_version: 1\n");
328    text.push_str(&format!("created_at: {created}\n"));
329    text.push_str(&format!(
330        "task_id: {}\n",
331        one_line(h.task.as_deref().unwrap_or("none"))
332    ));
333    text.push_str("profile: none\n");
334    text.push_str(&format!("owner: \"mesh:{}\"\n", view.runtime));
335    text.push_str(&format!("repository_id: {}\n", repository_id(root)?));
336    text.push_str(&format!("worktree: mesh:{}\n", view.runtime));
337    text.push_str(&format!(
338        "branch: {branch}\nhead: {head}\nworking_tree: clean\nchanged_files:\n"
339    ));
340    text.push_str(&format!("{marker}\nmesh_origin: {}\n", view.runtime));
341    if let Some(issue) = &h.issue {
342        text.push_str(&format!("issue: \"{}\"\n", one_line(issue)));
343    }
344    if let Some(milestone) = &h.milestone {
345        text.push_str(&format!("milestone: \"{}\"\n", one_line(milestone)));
346    }
347    text.push_str("---\n\n");
348    text.push_str(&h.body);
349    // The file name admits only what its parts are: digits and T/Z of a timestamp, hex of a
350    // commit, a branch key of [A-Za-z0-9_-] (no dots, so never `..`), hex of the id.
351    let mut compact: String = created
352        .chars()
353        .filter(|c| c.is_ascii_digit() || *c == 'T' || *c == 'Z')
354        .take(16)
355        .collect();
356    if compact.is_empty() {
357        compact = "00000000T000000Z".into();
358    }
359    let branch_key: String = branch
360        .chars()
361        .map(|c| {
362            if c.is_ascii_alphanumeric() || c == '_' || c == '-' {
363                c
364            } else {
365                '-'
366            }
367        })
368        .take(64)
369        .collect();
370    let mut head7: String = head
371        .chars()
372        .filter(char::is_ascii_hexdigit)
373        .take(7)
374        .collect();
375    if head7.is_empty() {
376        head7 = "0000000".into();
377    }
378    let name = format!(
379        "{compact}--mesh--{branch_key}--{head7}--{}.md",
380        &view.id[..16]
381    );
382    let path = dir.join(&name);
383    if name.contains('/') || path.parent() != Some(dir.as_path()) {
384        return Err(format!(
385            "refusing to write a handover outside {}",
386            dir.display()
387        ));
388    }
389    let tmp = dir.join(format!(".tmp.mesh.{}", &view.id[..16]));
390    std::fs::write(&tmp, text).map_err(|e| format!("{}: {e}", tmp.display()))?;
391    #[cfg(unix)]
392    {
393        use std::os::unix::fs::PermissionsExt;
394        let _ = std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o600));
395    }
396    std::fs::rename(&tmp, &path).map_err(|e| format!("{}: {e}", path.display()))?;
397    Ok(path)
398}
