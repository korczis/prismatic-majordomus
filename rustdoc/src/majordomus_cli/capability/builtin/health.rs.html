1//! The `health` module: whether what this process serves is healthy, answered by the
2//! engines that already decide it rather than by checks written a second time here.
3//!
4//! The module is `health` and not `system` because the benchmark projection reserves
5//! `system` for the transports' own targets (`system.http.index`, `system.mcp.ping`) and
6//! buckets its coverage by that name. A capability module called `system` would have its
7//! lines tallied as transport targets and silently vanish from the per-transport
8//! denominators; `reserved_namespace_is_not_a_module` holds that shut.
9//!
10//! Every check delegates: the index's own diagnostics say whether the layer read cleanly,
11//! the benchmark projection's coverage says whether every executable capability is timed,
12//! `generate --check` — the same comparison the command runs — says whether the committed
13//! projections are stale, and [`crate::capability::builtin::server::standing_at`] — the
14//! reading `server.status` answers from — says where the shared server of this checkout
15//! stands. A check that computed its own verdict would be a second opinion, and two
16//! opinions drift.
17//!
18//! # Four readings of "ready", and why none of them is this one
19//!
20//! This crate answers "ready" four times, about four different subjects. The full table,
21//! with what asks each one, is in `docs/MCP.md`; in short:
22//!
23//! - [`crate::http::Served::ready`] — is a *surface's* producer's output on disk? Per
24//!   surface, and what the home page renders as `not built`.
25//! - [`Readiness`] (`health.ready`, `GET /api/v1/ready`) — can *this process* answer a
26//!   request? Local initialisation only, never a dependency: a readiness probe that
27//!   contacts another service fails a deployment for something that is not this process.
28//! - [`crate::environment::ServiceAvailability`] — does *anything* accept a connection at
29//!   the address the lease published? One bounded TCP connect, no DNS, because it runs on
30//!   a shell prompt.
31//! - [`crate::capability::builtin::server::ServerStanding`] — is what answers at that
32//!   address *current*: this checkout's server, from the file on disk, at this version?
33//!
34//! The `server` check below is the fourth, and it is the only one of them that a health
35//! report should carry: the first is per surface, the second is the report's own caller,
36//! and the third cannot tell a live server from a socket somebody else holds.

60/// Where one dimension of the system stands. Ordered by severity, so the worst check
61/// decides the whole.
62#[derive(
63    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
64)]
65#[serde(rename_all = "snake_case")]
66pub enum HealthStatus {
67    /// The engine that decides this dimension is satisfied.
68    Ok,
69    /// Nothing is broken and something is worth looking at.
70    Warn,
71    /// The engine that decides this dimension is not satisfied.
72    Fail,
73    /// This process cannot decide it, and says so rather than reporting `ok`.
74    Unknown,
75}
76
77impl HealthStatus {
78    /// The word as serialised.
79    ///
80    /// ```
81    /// use majordomus_cli::capability::builtin::health::HealthStatus;
82    /// assert_eq!(HealthStatus::Warn.as_str(), "warn");
83    /// ```
84    pub fn as_str(self) -> &'static str {
85        match self {
86            HealthStatus::Ok => "ok",
87            HealthStatus::Warn => "warn",
88            HealthStatus::Fail => "fail",
89            HealthStatus::Unknown => "unknown",
90        }
91    }
92
93    /// The worse of two: the whole is as good as its worst check, and `unknown` is worse
94    /// than `warn` because an undecided dimension is not a healthy one.
95    ///
96    /// ```
97    /// use majordomus_cli::capability::builtin::health::HealthStatus::*;
98    /// assert_eq!(Ok.worse(Warn), Warn);
99    /// assert_eq!(Fail.worse(Unknown), Unknown);
100    /// ```
101    pub fn worse(self, other: Self) -> Self {
102        if other > self {
103            other
104        } else {
105            self
106        }
107    }
108}

110/// One dimension of the system, decided by one engine, with what a reader needs to check
111/// it themselves.
112#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
113pub struct HealthCheck {
114    /// A stable id, `[a-z][a-z0-9-]*`.
115    pub id: String,
116    /// The short name.
117    pub title: String,
118    /// Where it stands.
119    pub status: HealthStatus,
120    /// One line: what the engine decided and on what.
121    pub detail: String,
122    /// The engine that decided it, named so the reader knows what to fix.
123    pub decided_by: String,
124    /// What reproduces the verdict: a command, or a path.
125    #[serde(default, skip_serializing_if = "Vec::is_empty")]
126    pub evidence: Vec<String>,
127    /// Findings behind the verdict, when the engine produced any: the diagnostic
128    /// messages, the stale files, the missing coverage lines.
129    #[serde(default, skip_serializing_if = "Vec::is_empty")]
130    pub findings: Vec<String>,
131}

133/// The health of what this process serves.
134#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
135pub struct Health {
136    /// The worst check's status.
137    pub status: HealthStatus,
138    /// How many checks stand where, by status word.
139    pub tallies: BTreeMap<String, usize>,
140    /// Every dimension, in a stable order.
141    pub checks: Vec<HealthCheck>,
142}

144/// The answer to "is this process alive": the cheapest true thing this executable can
145/// say about itself. Deliberately not a summary of anything — a probe that grew an
146/// opinion would stop reporting the one fact the platform polls it for.
147#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
148pub struct Liveness {
149    /// Always `true`. A process that could not answer would not answer at all, which is
150    /// the signal.
151    pub alive: bool,
152    /// This executable's version, so a rolling deployment can tell which build answered.
153    pub version: String,
154}
155
156/// The answer to "can this process serve traffic": the local initialisation a request
157/// would need, and nothing beyond this process. A readiness check that grew a dependency
158/// probe fails a deployment because an unrelated service is down.
159#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
160pub struct Readiness {
161    /// Whether every part below is in place.
162    pub ready: bool,
163    /// This executable's version.
164    pub version: String,
165    /// How many capabilities the registry holds; zero would mean nothing to serve.
166    pub capabilities: usize,
167    /// How many objects the index holds. Read from the index this process built at
168    /// start-up: it is already in memory, and reading it walks nothing.
169    pub objects: usize,
170    /// Whether the layer read cleanly. A degraded layer is still served — the diagnostics
171    /// are the point — so this reports rather than refuses.
172    pub layer: HealthStatus,
173}
174
175/// Liveness. No filesystem, no index traversal, no network: two fields this process can
176/// answer with while doing anything else.
177fn liveness(_: &Context, _: Empty) -> Result<Liveness, CapabilityError> {
178    Ok(Liveness {
179        alive: true,
180        version: crate::VERSION.into(),
181    })
182}

184/// Readiness. Only what this process needs in order to answer a request, all of it already
185/// resident: the registry it built, the index it built, and how that index read. Nothing
186/// here contacts a provider, a database or another service, and nothing here walks a tree.
187fn readiness(ctx: &Context, _: Empty) -> Result<Readiness, CapabilityError> {
188    let capabilities = ctx.registry.summary().total;
189    let objects = ctx.index.objects.len();
190    Ok(Readiness {
191        ready: capabilities > 0,
192        version: crate::VERSION.into(),
193        capabilities,
194        objects,
195        layer: match ctx.index.state {
196            State::Degraded => HealthStatus::Warn,
197            State::Ok => HealthStatus::Ok,
198        },
199    })
200}

202/// Record one dimension, and say on the execution's stream that it was decided.
203///
204/// The report is the same whoever asked for it: a `GET /api/v1/health`, an MCP tool call
205/// and a browser watching an execution all run this function, and the reporting is silent
206/// for the two of them that are not executions. That is the whole of what it took to make
207/// an existing capability observable — no second implementation, no second route, and no
208/// change to what it answers.
209fn record(checks: &mut Vec<HealthCheck>, p: &crate::execution::Progress, check: HealthCheck) {
210    p.step_done(
211        &check.id,
212        check.status != HealthStatus::Fail,
213        Some(check.detail.clone()),
214    );
215    checks.push(check);
216    p.progress(checks.len() as u64, None, "dimension(s) decided");
217}

219fn health(ctx: &Context, _: Empty) -> Result<Health, CapabilityError> {
220    let index = &ctx.index;
221    let mut checks = Vec::new();
222
223    // --- the layer, as the index read it
224    let errors = index
225        .diagnostics
226        .iter()
227        .filter(|d| d.severity == Severity::Error)
228        .count();
229    let warnings = index
230        .diagnostics
231        .iter()
232        .filter(|d| d.severity == Severity::Warning)
233        .count();
234    record(
235        &mut checks,
236        &ctx.progress,
237        HealthCheck {
238            id: "layer".into(),
239            title: "The layer as it was read".into(),
240            status: match (index.state, warnings) {
241                (State::Degraded, _) => HealthStatus::Fail,
242                (State::Ok, 0) => HealthStatus::Ok,
243                (State::Ok, _) => HealthStatus::Warn,
244            },
245            detail: format!(
246                "{} object(s) indexed; {errors} error(s), {warnings} warning(s)",
247                index.objects.len()
248            ),
249            decided_by: "the index's own diagnostics".into(),
250            evidence: vec!["majordomus capabilities validate".into()],
251            findings: index
252                .diagnostics
253                .iter()
254                .filter(|d| d.severity != Severity::Info)
255                .map(|d| {
256                    format!(
257                        "{}: {} ({})",
258                        d.code,
259                        d.message,
260                        d.path.as_deref().unwrap_or("-")
261                    )
262                })
263                .collect(),
264        },
265    );
266
267    // --- the registry: it exists, therefore it validated; a registry that does not build
268    // stops the process before any transport is bound
269    let summary = ctx.registry.summary();
270    record(
271        &mut checks,
272        &ctx.progress,
273        HealthCheck {
274            id: "registry".into(),
275            title: "The capability registry".into(),
276            status: HealthStatus::Ok,
277            detail: format!(
278                "{} capabilities ({} builtin, {} declarative) across {} modules; fingerprint {}",
279                summary.total,
280                summary.builtin,
281                summary.declarative,
282                summary.modules,
283                &ctx.registry.fingerprint()[..12.min(ctx.registry.fingerprint().len())]
284            ),
285            decided_by:
286                "the registry builder; it refuses to build on a duplicate or a malformed exposure"
287                    .into(),
288            evidence: vec!["majordomus capabilities validate".into()],
289            findings: Vec::new(),
290        },
291    );
292
293    // --- the scope: what a worker reads of this repository and what it never reads
294    let tally = &index.scoped.tally;
295    record(
296        &mut checks,
297        &ctx.progress,
298        HealthCheck {
299            id: "scope".into(),
300            title: "The declared scope".into(),
301            status: HealthStatus::Ok,
302            detail: format!(
303                "{} of {} tracked file(s) in scope, {} out, read from {}",
304                tally.r#in, tally.files, tally.out, index.repository.scope_path
305            ),
306            decided_by: "the scope declaration the index was built under".into(),
307            evidence: vec!["majordomus scope".into()],
308            findings: Vec::new(),
309        },
310    );
311
312    // --- git
313    let (git_status, git_detail) = match &index.repository.git {
314        GitState::Available(info) => {
315            let head = info.head.as_deref().unwrap_or("(unborn)");
316            (
317                if info.working_tree == "clean" {
318                    HealthStatus::Ok
319                } else {
320                    HealthStatus::Warn
321                },
322                format!(
323                    "{} at {}, working tree {}",
324                    info.branch.as_deref().unwrap_or("(detached)"),
325                    &head[..12.min(head.len())],
326                    info.working_tree
327                ),
328            )
329        }
330        GitState::Unavailable { reason } => (HealthStatus::Unknown, reason.clone()),
331    };
332    record(
333        &mut checks,
334        &ctx.progress,
335        HealthCheck {
336            id: "git".into(),
337            title: "Version control".into(),
338            status: git_status,
339            detail: git_detail,
340            decided_by: "git, as the index asked it once at startup".into(),
341            evidence: vec!["git status".into()],
342            findings: Vec::new(),
343        },
344    );
345
346    // --- benchmark coverage, from the projection that defines the targets
347    let projection = BenchmarkProjection::from_context(ctx);
348    let coverage = Coverage::compute(ctx, &projection);
349    let total = coverage.tallies.get("total").cloned().unwrap_or_default();
350    record(
351        &mut checks,
352        &ctx.progress,
353        HealthCheck {
354            id: "benchmark-coverage".into(),
355            title: "Benchmark coverage".into(),
356            status: if total.missing > 0 {
357                HealthStatus::Fail
358            } else if total.waived > 0 {
359                HealthStatus::Warn
360            } else {
361                HealthStatus::Ok
362            },
363            detail: format!(
364                "{} of {} target(s) covered, {} missing, {} waived",
365                total.covered, total.required, total.missing, total.waived
366            ),
367            decided_by:
368                "the benchmark projection's coverage, the same one `bench coverage --check` reads"
369                    .into(),
370            evidence: vec!["majordomus bench coverage --check".into()],
371            findings: coverage
372                .lines
373                .iter()
374                .filter(|l| matches!(l.state, crate::bench::CoverageState::Missing))
375                .map(|l| format!("{} on {}: no case", l.subject, l.transport.name()))
376                .collect(),
377        },
378    );
379
380    // --- the committed registry manifest against the registry this process built.
381    // Only that one artifact: it is derived from the code alone, and rendering it costs
382    // nothing but serialising the descriptors already in memory. Rendering the OpenAPI
383    // document, the reference and the benchmark matrix on every health call would rebuild
384    // canonical state per request, which this executable does not do; `generate --check`
385    // is the complete answer and this check names it.
386    let root = std::path::Path::new(&index.repository.root);
387    let manifest = generate::Document::new(
388        "registry",
389        generate::REGISTRY_SCHEMA,
390        "the canonical capability registry",
391        generate::registry_manifest(&ctx.registry),
392    )
393    .artifacts(crate::VERSION);
394    let manifest = manifest
395        .iter()
396        .find(|a| a.format == generate::ArtifactFormat::Json)
397        .map(|a| a.content.clone())
398        .unwrap_or_default();
399    let path = format!("{}/registry.json", generate::OUT_DIR);
400    let committed = std::fs::read_to_string(root.join(&path));
401    let (status, detail, findings) = match &committed {
402        Ok(text) if *text == manifest => (
403            HealthStatus::Ok,
404            format!("{path} matches the registry this process built"),
405            Vec::new(),
406        ),
407        Ok(_) => (
408            HealthStatus::Fail,
409            format!("{path} differs from the registry this process built"),
410            vec![format!("{path} (differs)")],
411        ),
412        Err(_) => (
413            HealthStatus::Unknown,
414            format!("{path} could not be read"),
415            vec![format!("{path} (missing)")],
416        ),
417    };
418    record(
419        &mut checks,
420        &ctx.progress,
421        HealthCheck {
422            id: "generated-registry".into(),
423            title: "Committed projections".into(),
424            status,
425            detail,
426            decided_by:
427                "the same rendering `majordomus generate` writes, compared with what is committed"
428                    .into(),
429            evidence: vec!["majordomus generate --check".into()],
430            findings,
431        },
432    );
433
434    // --- the shared server of this checkout
435    //
436    // Not a second opinion and not a second reading: `server::standing_at` is what answers
437    // `GET /api/v1/server`, and this reports what it says. It costs one small file and,
438    // when the lease names an address, one probe bounded by `lease::PROBE_TIMEOUT`; the
439    // five-second cache on `health.report` is as fresh as a file other processes write can
440    // usefully be, and `server.status` is the uncached answer for a caller that needs one.
441    let checkout = root.canonicalize().unwrap_or_else(|_| root.to_path_buf());
442    let local_half = server::local_half(&checkout);
443    let lease_path = crate::lease::lease_file(&checkout, &local_half);
444    let (standing, reason, lease) = server::standing_at(&checkout, &local_half);
445    let detail = match lease.document() {
446        Some(doc) => format!(
447            "{}: {}, version {}, pid {}, since {}",
448            standing.as_str(),
449            doc.url.as_deref().unwrap_or("no address published yet"),
450            doc.version.as_deref().unwrap_or("unstated"),
451            doc.pid,
452            doc.started_at
453        ),
454        None if standing == ServerStanding::Absent => format!(
455            "absent: no lease at {}, so nothing serves this checkout",
456            lease_path.display()
457        ),
458        None => format!(
459            "{}: the lease at {} holds no document",
460            standing.as_str(),
461            lease_path.display()
462        ),
463    };
464    // Whose report this is. Everything above describes the checkout's server, read from the
465    // lease — and a process whose lease was taken over describes, accurately, a server that
466    // is not itself. The one caller that has reached the wrong process is the one caller
467    // who cannot afford to be told that the right server is fine, so this outranks the
468    // standing rather than qualifying it.
469    let orphaned = crate::lease::was_lost();
470    let mut findings: Vec<String> = reason.into_iter().collect();
471    if orphaned {
472        findings.push(
473            "this process no longer holds the lease: it serves the sessions it already had \
474             and takes on no new ones; what is described above is another process"
475                .into(),
476        );
477    }
478    record(
479        &mut checks,
480        &ctx.progress,
481        HealthCheck {
482            id: "server".into(),
483            title: "The shared server".into(),
484            status: if orphaned {
485                HealthStatus::Warn
486            } else {
487                match standing {
488                    // A checkout nobody serves is not an unhealthy one: nothing is running
489                    // and nothing claims to be. A lease still binding resolves itself
490                    // within the bind grace, and is reported as stale by this same engine
491                    // when it does not. Neither is a fault of what this process serves.
492                    ServerStanding::Absent | ServerStanding::Starting | ServerStanding::Ready => {
493                        HealthStatus::Ok
494                    }
495                    // A server answering from code this tree no longer has says yesterday's
496                    // truth in today's words, and a lease naming one that answers for
497                    // nobody sends the next reader at a dead address. Both are somebody's
498                    // to clear; neither stops this process from serving, so neither is a
499                    // failure.
500                    ServerStanding::Outdated | ServerStanding::Stale => HealthStatus::Warn,
501                }
502            },
503            detail: if orphaned {
504                format!("{detail}; and this process is not it")
505            } else {
506                detail
507            },
508            decided_by:
509                "the decision `server.status` makes, from this checkout's lease and one probe of the server it names"
510                    .into(),
511            evidence: vec!["majordomus serve status".into()],
512            findings,
513        },
514    );
515
516    // --- the peers attached to this process
517    let peers = ctx.peers.list();
518    record(
519        &mut checks,
520        &ctx.progress,
521        HealthCheck {
522            id: "peers".into(),
523            title: "Attached clients".into(),
524            status: HealthStatus::Ok,
525            detail: format!("{} peer(s) attached to this process", peers.len()),
526            decided_by: "the in-memory peer board of this process".into(),
527            evidence: vec!["majordomus_peers".into()],
528            findings: Vec::new(),
529        },
530    );
531
532    let status = checks
533        .iter()
534        .fold(HealthStatus::Ok, |acc, c| acc.worse(c.status));
535    let mut tallies: BTreeMap<String, usize> = BTreeMap::new();
536    for c in &checks {
537        *tallies.entry(c.status.as_str().into()).or_insert(0) += 1;
538    }
539    Ok(Health {
540        status,
541        tallies,
542        checks,
543    })
544}

604    /// Liveness says one thing and reads nothing: the counters prove it moved no
605    /// canonical state, and the answer is the same whatever the layer holds.
606    #[test]
607    fn liveness_is_the_same_answer_whatever_the_layer_holds() {
608        let small = SyntheticRepository::new(Shape {
609            rules: 1,
610            ..Shape::default()
611        })
612        .expect("a synthetic repository");
613        let large = SyntheticRepository::new(Shape {
614            rules: 40,
615            ..Shape::default()
616        })
617        .expect("a synthetic repository");
618        let a = liveness(&small.context().expect("a context"), Empty {}).expect("alive");
619        let b = liveness(&large.context().expect("a context"), Empty {}).expect("alive");
620        assert_eq!(a, b);
621        assert!(a.alive);
622        assert_eq!(a.version, crate::VERSION);
623    }
624
625    /// Readiness reads what this process already holds — the registry it built and the
626    /// index it built — and reports the layer rather than refusing over it.
627    #[test]
628    fn readiness_reports_what_this_process_already_holds() {
629        let repo = SyntheticRepository::new(Shape::default()).expect("a synthetic repository");
630        let ctx = repo.context().expect("a context");
631        let r = readiness(&ctx, Empty {}).expect("ready");
632        assert!(r.ready);
633        assert_eq!(r.capabilities, ctx.registry.summary().total);
