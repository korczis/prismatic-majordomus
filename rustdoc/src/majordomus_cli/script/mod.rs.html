1//! The engine a scripted capability runs in (ADR 0069).
2//!
3//! A capability may be scripted: declared in a file of the layer, with a Rhai script as its
4//! handler. This module is the one place a Rhai [`Engine`] is built, so every script runs
5//! under the same sandbox and nothing else in the crate configures an interpreter of its own.
6//!
7//! What the sandbox guarantees, each proved by a test below:
8//!
9//! - **no eval.** Rhai's own `eval` is disabled, so a script is compiled from its tracked file
10//!   and never from a string assembled at run time (`project.no-network-no-eval`, and the
11//!   exception ADR 0069 declares in its terms).
12//! - **no hidden clock.** The crate is built with `no_time`: time reaches a script through its
13//!   context, so the same script over the same inputs produces the same output.
14//! - **bounded work.** Operations, call depth, expression depth and the sizes of strings, arrays
15//!   and maps are limited, so a runaway script ends with a named error instead of a hang.
16//! - **cancellation.** Every operation checks the execution's cancel flag, so a script stops
17//!   when its execution is asked to stop, without a thread being killed.
18//! - **deterministic maps.** A Rhai object map iterates in key order, whatever order its keys
19//!   were inserted in.
20//!
21//! The engine registers no filesystem, process or network function. A script reaches the
22//! repository only through capabilities, which a later change binds; `import` resolves no
23//! module, and `print` and `debug` write nowhere until the caller routes them.
24
25// The engine is wired to a capability by the next change of this milestone; until then only
26// the tests below build one.
27#![allow(dead_code)]

29use rhai::{Dynamic, Engine};

31/// How much one script execution may do before it is stopped.
32///
33/// The defaults are deliberately generous for repository automation — a script orchestrates
34/// capabilities, it does not compute — and small enough that an accidental infinite loop ends
35/// in well under a second.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub(crate) struct Limits {
38    /// Rhai operations (roughly, evaluated expressions and statements).
39    pub(crate) operations: u64,
40    /// Nested function calls.
41    pub(crate) call_levels: usize,
42    /// Expression nesting at the top level of a script.
43    pub(crate) expression_depth: usize,
44    /// Expression nesting inside a function body.
45    pub(crate) function_expression_depth: usize,
46    /// Characters in one string.
47    pub(crate) string_size: usize,
48    /// Elements in one array.
49    pub(crate) array_size: usize,
50    /// Entries in one object map.
51    pub(crate) map_size: usize,
52}
53
54impl Default for Limits {
55    fn default() -> Self {
56        Limits {
57            operations: 1_000_000,
58            call_levels: 64,
59            expression_depth: 64,
60            function_expression_depth: 32,
61            string_size: 1 << 20,
62            array_size: 100_000,
63            map_size: 100_000,
64        }
65    }
66}
67
68/// The value a cancelled script's evaluation returns as its termination token.
69const CANCELLED: &str = "cancelled";
70
71/// Build the sandboxed engine every scripted capability runs in.
72///
73/// `cancelled` is asked on every operation: once it answers true, the script stops at that
74/// operation with [`rhai::EvalAltResult::ErrorTerminated`]. An execution passes its progress
75/// handle's flag; a compilation, which runs nothing, passes `|| false`.
76///
77/// `print` and `debug` write nowhere until the caller routes them: the shared server speaks
78/// MCP on its standard output, and a script's line there would corrupt the protocol.
79pub(crate) fn engine(
80    limits: &Limits,
81    cancelled: impl Fn() -> bool + Send + Sync + 'static,
82) -> Engine {
83    let mut engine = Engine::new();
84    engine
85        .set_max_operations(limits.operations)
86        .set_max_call_levels(limits.call_levels)
87        .set_max_expr_depths(limits.expression_depth, limits.function_expression_depth)
88        .set_max_string_size(limits.string_size)
89        .set_max_array_size(limits.array_size)
90        .set_max_map_size(limits.map_size);
91    // Compiled from a tracked file, never from a string: `eval` does not exist here.
92    engine.disable_symbol("eval");
93    // `import` would read a file from the filesystem at large; modules resolve nowhere.
94    engine.set_module_resolver(rhai::module_resolvers::DummyModuleResolver::new());
95    engine.on_print(|_| {});
96    engine.on_debug(|_, _, _| {});
97    engine.on_progress(move |_| cancelled().then(|| Dynamic::from(CANCELLED)));
98    engine
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use rhai::EvalAltResult;
105    use std::sync::atomic::{AtomicBool, Ordering};
106    use std::sync::{Arc, Mutex};
107
108    fn sandbox() -> Engine {
109        engine(&Limits::default(), || false)
110    }
111
112    #[test]
113    fn a_map_iterates_in_key_order_whatever_order_it_was_built_in() {
114        let e = sandbox();
115        let keys: Vec<String> = e
116            .eval::<rhai::Map>(r#"let m = #{}; m.zulu = 1; m.alpha = 2; m.mike = 3; m"#)
117            .expect("a map")
118            .keys()
119            .map(|k| k.to_string())
120            .collect();
121        assert_eq!(keys, ["alpha", "mike", "zulu"]);
122        // and a script sees the same order
123        let listed = e
124            .eval::<String>(
125                r#"let m = #{ zulu: 1, alpha: 2 }; let s = ""; for k in m.keys() { s += k; } s"#,
126            )
127            .expect("keys in a script");
128        assert_eq!(listed, "alphazulu");
129    }

131    #[test]
132    fn eval_is_not_a_function_a_script_can_call() {
133        let err = sandbox()
134            .compile(r#"eval("40 + 2")"#)
135            .expect_err("eval must not compile");
136        assert!(err.to_string().contains("eval"), "{err}");
137    }

150    #[test]
151    fn a_runaway_script_is_stopped_by_its_operation_limit() {
152        let limits = Limits {
153            operations: 10_000,
154            ..Limits::default()
155        };
156        let err = engine(&limits, || false)
157            .eval::<Dynamic>("let x = 0; loop { x += 1; }")
158            .expect_err("an endless loop must end");
159        assert!(
160            matches!(*err, EvalAltResult::ErrorTooManyOperations(..)),
161            "{err}"
162        );
163    }
164
165    #[test]
166    fn a_cancelled_execution_stops_at_its_next_operation() {
167        // the flag is raised part way through, as a cancel request arrives while it runs
168        let cancel = Arc::new(AtomicBool::new(false));
169        let raised = Arc::clone(&cancel);
170        let mut e = engine(&Limits::default(), move || cancel.load(Ordering::SeqCst));
171        e.register_fn("stop", move || raised.store(true, Ordering::SeqCst));
172        let err = e
173            .eval::<Dynamic>("let x = 0; loop { x += 1; if x == 100 { stop(); } }")
174            .expect_err("a cancelled script must stop");
175        match *err {
176            EvalAltResult::ErrorTerminated(token, _) => {
177                assert_eq!(token.into_string().unwrap_or_default(), CANCELLED)
178            }
179            other => panic!("expected termination, got {other}"),
180        }
181    }

209    #[test]
210    fn print_and_debug_write_nowhere_until_the_caller_routes_them() {
211        // unrouted: the call succeeds and nothing reaches this process's standard output
212        let quiet = sandbox()
213            .eval::<i64>(r#"print("hidden"); debug("hidden"); 1"#)
214            .expect("print is a no-op");
215        assert_eq!(quiet, 1);
216        // routed: the caller's sink receives the line
217        let lines = Arc::new(Mutex::new(Vec::<String>::new()));
218        let sink = Arc::clone(&lines);
219        let mut e = sandbox();
220        e.on_print(move |l| sink.lock().unwrap().push(l.to_string()));
221        let routed = e.eval::<i64>(r#"print("seen"); 1"#).expect("print runs");
222        assert_eq!(routed, 1);
223        assert_eq!(*lines.lock().unwrap(), ["seen"]);
224    }
