1//! The version a landing allocates, applied where the integration executor already
2//! refreshes a pull request (ADR 0134, slice B).
3//!
4//! Two seams and nothing else. Classification ([`super::relation`]) asks
5//! [`version_only_conflict`]: under `release.allocation: landing` a conflict on the manifest
6//! that vanishes once the head carries master's version is not a person's to resolve, because
7//! the head's own number is no longer an author's statement. The refresh ([`super::drain`])
8//! calls [`resolve_version_conflict`] when the merge of master stops, and [`allocate_in`]
9//! after the merge and before the derive: master's version raised by what
10//! [`crate::release::landing::next_for`] decides for the head. The head the required check
11//! judges is then the head that lands, carrying the number it lands with.
12//!
13//! Every answer is computed from master as it is at that refresh. A retry, a second refresh
14//! after master moved, or a recovery after a crash recomputes from the new master and
15//! overwrites the head's number, so no integration raises a raised version again.
16

27/// The release policy master commits, at `master` in the clone at `root`.
28///
29/// Read from the commit, never from a working tree: the relation of a head to master is
30/// cached by the pair of commit ids as a fact forever ([`super::relation`]), so whatever it
31/// depends on must be a fact of those commits; and the trunk's policy is the one a landing
32/// onto it obeys, whatever the executor's own checkout says. A directory with no
33/// `.ai/manifest.yaml`, and a master that commits no policy, allocate nothing: the default,
34/// `off`. A committed policy that cannot be read is an error, never `off`, because a refresh
35/// made without its allocation lands a head that owes a number.
36///
37/// ```text
38/// policy_at(Path::new("/tmp"), "master")  ->  Ok(ReleasePolicy { allocation: Off, .. })
39/// ```
40pub fn policy_at(root: &Path, master: &str) -> Result<ReleasePolicy, String> {
41    if !root.join(".ai/manifest.yaml").is_file() {
42        return Ok(ReleasePolicy::default());
43    }
44    let repo = crate::repository::Repository::open(root)
45        .map_err(|e| format!("the repository could not be opened: {e}"))?;
46    let rel = repo
47        .section_path(crate::policy::POLICY_SECTION)
48        .ok_or_else(|| "the manifest names no policy section".to_string())?;
49    let Some(text) = show(root, master, &rel) else {
50        return Ok(ReleasePolicy::default());
51    };
52    crate::metadata::yaml::parse_into::<crate::policy::Policy>(&text)
53        .map(|p| p.release)
54        .map_err(|e| format!("{master}:{rel} could not be read: {e}"))
55}

57/// Whether master allocates at landing. An unreadable policy answers no, which is the
58/// stricter answer for a classification: the version conflict stays a person's.
59pub fn allocates_at(root: &Path, master: &str) -> bool {
60    policy_at(root, master).is_ok_and(|p| p.allocation == Allocation::Landing)
61}

72/// An answer that may have failed, with what it was about: the one error sentence this
73/// module writes for the file system, the version writer and git alike.
74fn about<T, E: std::fmt::Display>(
75    r: Result<T, E>,
76    what: impl std::fmt::Display,
77) -> Result<T, String> {
78    r.map_err(|e| format!("{what}: {e}"))
79}

81/// A fresh directory under `under` for one merge, with the manifest's parent made.
82fn scratch(under: &Path) -> Result<Scratch, String> {
83    static NEXT: AtomicU64 = AtomicU64::new(0);
84    let dir = under.join(format!(
85        "majordomus-allocate-{}-{}",
86        std::process::id(),
87        NEXT.fetch_add(1, Ordering::Relaxed)
88    ));
89    let manifest = dir.join(MANIFEST);
90    let parent = manifest.parent().unwrap_or(&dir);
91    about(std::fs::create_dir_all(parent), parent.display())?;
92    Ok(Scratch(dir))
93}

95/// `manifest` with its package version set to `to`, by the one writer of that line, in `dir`.
96fn with_version(dir: &Path, manifest: &str, to: &str) -> Result<String, String> {
97    let path = dir.join(MANIFEST);
98    about(std::fs::write(&path, manifest), path.display())
99        .and_then(|()| about(version::write(dir, to), "the version could not be written"))
100        .and_then(|_| about(std::fs::read_to_string(&path), path.display()))
101}

103/// The manifest a three-way merge of `branch` and `trunk` over `base` gives once the branch
104/// carries the trunk's version, or `None` when it still conflicts: something besides the
105/// version line conflicts, and that stays a person's.
106///
107/// ```text
108/// let base = "[package]\nversion = \"0.1.0\"\n\n[dependencies]\na = \"1\"\n";
109/// let branch = "[package]\nversion = \"0.2.0\"\n\n[dependencies]\na = \"1\"\nb = \"1\"\n";
110/// let trunk = "[package]\nversion = \"0.3.0\"\n\n[dependencies]\na = \"1\"\n";
111/// merged_on_trunk_version(base, branch, trunk)  ->  Ok(Some(version 0.3.0, with b = "1"))
112/// ```
113pub fn merged_on_trunk_version(
114    base: &str,
115    branch: &str,
116    trunk: &str,
117) -> Result<Option<String>, String> {
118    let Some(to) = version::declared_in(trunk) else {
119        return Ok(None);
120    };
121    scratch(&std::env::temp_dir()).and_then(|s| merged_in(&s.0, &to, base, branch, trunk))
122}
123
124/// [`merged_on_trunk_version`] in `dir`, the branch carrying `to`: every file the merge
125/// reads is written there, and `git merge-file` runs there.
126fn merged_in(
127    dir: &Path,
128    to: &str,
129    base: &str,
130    branch: &str,
131    trunk: &str,
132) -> Result<Option<String>, String> {
133    let current = with_version(dir, branch, to)?;
134    let files = [
135        ("current", current.as_str()),
136        ("base", base),
137        ("other", trunk),
138    ];
139    for (name, text) in files {
140        about(std::fs::write(dir.join(name), text), name)?;
141    }
142    about(
143        Command::new("git")
144            .current_dir(dir)
145            .args(["merge-file", "-p", "-q", "current", "base", "other"])
146            .output(),
147        "git merge-file could not run",
148    )
149    .and_then(|out| merge_verdict(out.status.code(), &out.stdout, &out.stderr))
150}

152/// What `git merge-file`'s exit says: the merged text, `None` for a conflict (a positive
153/// status is the number of conflicts git left), and an error for anything else, a signal
154/// included.
155fn merge_verdict(
156    code: Option<i32>,
157    stdout: &[u8],
158    stderr: &[u8],
159) -> Result<Option<String>, String> {
160    match code {
161        Some(0) => Ok(Some(String::from_utf8_lossy(stdout).into_owned())),
162        Some(n) if (1..=127).contains(&n) => Ok(None),
163        _ => Err(format!(
164            "git merge-file failed: {}",
165            String::from_utf8_lossy(stderr).trim()
166        )),
167    }
168}

170fn show(root: &Path, rev: &str, path: &str) -> Option<String> {
171    let out = Command::new("git")
172        .arg("-C")
173        .arg(root)
174        .args(["show", &format!("{rev}:{path}")])
175        .output()
176        .ok()?;

179        .then(|| String::from_utf8_lossy(&out.stdout).into_owned())
180}

182/// Whether a merge of `head` into `master` conflicts on the manifest only because the two
183/// declare different versions: the conflict vanishes once the head carries master's. Read
184/// from git alone, as [`super::relation`] reads everything else, and `false` whenever any of
185/// the three manifests cannot be read.
186pub fn version_only_conflict(root: &Path, master: &str, head: &str) -> bool {
187    let Some(base) = Command::new("git")
188        .arg("-C")
189        .arg(root)
190        .args(["merge-base", master, head])
191        .output()
192        .ok()
193        .filter(|o| o.status.success())
194        .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string())
195    else {
196        return false;
197    };
198    match (
199        show(root, &base, MANIFEST),
200        show(root, master, MANIFEST),
201        show(root, head, MANIFEST),
202    ) {
203        (Some(base), Some(trunk), Some(branch)) => {
204            matches!(merged_on_trunk_version(&base, &branch, &trunk), Ok(Some(_)))
205        }
206        _ => false,
207    }
208}

210/// The version a landing of `head` onto `master` allocates, when master allocates at
211/// landing and the head, which contains master, does not carry it: such a head owes a
212/// refresh although nothing of master is missing from it (ADR 0134, slice B2). `None` when
213/// master does not allocate, when the head carries the allocated version, and when the
214/// allocation cannot be decided, which the refresh would refuse in its own words.
215///
216/// Every input is a fact of the two commits, as [`super::relation`] requires of what it
217/// caches: master's committed policy, its manifest and registry, the head's.
218pub fn owed_version(root: &Path, master: &str, head: &str) -> Option<String> {
219    let policy = policy_at(root, master)
220        .ok()
221        .filter(|p| p.allocation == Allocation::Landing)?;
222    let decision = landing::next_for(root, master, head, &policy).ok()?;
223    let carried = show(root, head, MANIFEST).and_then(|m| version::declared_in(&m));
224    (carried.as_deref() != Some(decision.next_version.as_str())).then_some(decision.next_version)
225}

227/// In `dir`, where a merge of master into the head stopped: when the manifest is the only
228/// path left unmerged and its conflict is the version line alone, resolve it to the merge
229/// with master's version and stage it. `Ok(false)` leaves the merge as it stopped, for the
230/// caller to refuse in its own words.
231pub fn resolve_version_conflict(dir: &Path) -> Result<bool, String> {
232    let git = |args: &[&str]| -> Option<String> {
233        let out = Command::new("git")
234            .arg("-C")
235            .arg(dir)
236            .args(args)
237            .output()
238            .ok()?;
239        out.status
240            .success()
241            .then(|| String::from_utf8_lossy(&out.stdout).into_owned())
242    };
243    let unmerged: Vec<String> = git(&["diff", "--name-only", "--diff-filter=U"])
244        .unwrap_or_default()
245        .lines()
246        .map(str::to_string)
247        .collect();
248    if unmerged != [MANIFEST] {
249        return Ok(false);
250    }
251    // stage 1 is the merge base, 2 the head being refreshed, 3 master
252    let stage = |n: u8| git(&["show", &format!(":{n}:{MANIFEST}")]);
253    let (Some(base), Some(branch), Some(trunk)) = (stage(1), stage(2), stage(3)) else {
254        return Ok(false);
255    };
256    merged_on_trunk_version(&base, &branch, &trunk).and_then(|merged| {
257        let Some(merged) = merged else {
258            return Ok(false);
259        };
260        std::fs::write(dir.join(MANIFEST), merged).map_err(|e| format!("{MANIFEST}: {e}"))?;
261        git(&["add", "--", MANIFEST])
262            .map(|_| true)
263            .ok_or_else(|| format!("git add {MANIFEST} failed"))
264    })
265}

267/// In `dir`, where master has just been merged into `head` and nothing is committed yet:
268/// write the version a landing of `head` onto `master` allocates, decided by
269/// [`landing::next_for`] in the clone at `root`. `None` when the policy does not allocate at
270/// landing; an allocation that cannot be decided is an error, never a head left on a number
271/// nobody chose.
272pub fn allocate_in(
273    root: &Path,
274    dir: &Path,
275    master: &str,
276    head: &str,
277    policy: &ReleasePolicy,
278) -> Result<Option<LandingDecision>, String> {
279    if policy.allocation != Allocation::Landing {
280        return Ok(None);
281    }
282    let decision = landing::next_for(root, master, head, policy)
283        .map_err(|e| format!("the version could not be allocated: {e}"))?;
284    version::write(dir, &decision.next_version)
285        .map_err(|e| format!("the allocated version could not be written: {e}"))?;
286    Ok(Some(decision))
287}

289/// The trailer line a refresh commit carries for `decision`: the number, what selected it
290/// and the trunk it was raised from, so `git log` answers which landing took which number.
291///
292/// ```text
293/// Version-Allocated: 0.22.0 (minor from 0.21.0)
294/// ```
295pub fn trailer(decision: &LandingDecision) -> String {
296    format!(
297        "{TRAILER}: {} ({} from {})",
298        decision.next_version,
299        decision.selected.as_str(),
300        decision.trunk_version
301    )
302}

607    /// A head and master that both moved the version on `topic` and master, with the head's
608    /// manifest as `head_manifest` and master's as `trunk_manifest`; the merge of master into
609    /// the head is left stopped.
610    fn stopped(
611        head_manifest: Option<&str>,
612        trunk_manifest: Option<&str>,
613    ) -> crate::synthetic::SyntheticRepository {
614        let repo = trunk("  allocation: landing\n  floor: minor\n");
615        let root = repo.root();
616        let commit = |manifest: Option<&str>, message: &str| {
617            match manifest {
618                Some(m) => write(root, MANIFEST, m),
619                None => ok(root, &["rm", "-q", MANIFEST]),
620            }
621            ok(root, &["commit", "-qam", message]);
622        };
623        ok(root, &["checkout", "-q", "-b", "topic"]);
624        commit(head_manifest, "the head");
625        ok(root, &["checkout", "-q", "master"]);
626        commit(trunk_manifest, "master");
627        ok(root, &["checkout", "-q", "topic"]);
628        assert!(
629            !git(root, &["merge", "-q", "master"]).status.success(),
630            "the merge stops"
631        );
632        repo
633    }
