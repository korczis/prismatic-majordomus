1//! The lifecycle stage a task stands at, derived from the answers to the completion policy
2//! and never written down by anybody.
3//!
4//! # Why the stage is derived
5//!
6//! A lifecycle written as a state a worker sets — `implementing`, `validated`, `deployed`
7//! — is a claim, and the artifact a worker learns to advance. The alternative is a pile of
8//! booleans (`tests_done`, `deployed`) that a report toggles. Both are the boolean soup
9//! this repository refuses: canonical facts first, status derived second.
10//!
11//! So the stage is a fold over the policy's stages in order. A stage whose every question
12//! passes or is exempt is behind the task; the first stage with a question still owed is
13//! where the task stands; a stage with a question that *refuses* — a failing gate, stale
14//! evidence — is where it is blocked. `completed` is the word for a task that is past every
15//! stage, and nothing else earns it.
16//!
17//! ```
18//! use majordomus_cli::gates::{DoneQuestion, GateStatus, StageDecl, StageState, derive_stage};
19//!
20//! let stages = vec![
21//!     StageDecl { id: "build".into(), title: "Build".into(), summary: "s".into() },
22//!     StageDecl { id: "ship".into(), title: "Ship".into(), summary: "s".into() },
23//! ];
24//! let q = |id: &str, status: GateStatus| DoneQuestion {
25//!     id: id.into(), stage: if id == "committed" { "build".into() } else { "ship".into() },
26//!     question: id.into(), status, evidence: "e".into(), source: "s".into(), remediation: "r".into(),
27//! };
28//!
29//! // committed passes, ci never reported: the task stands at Ship, pending
30//! let s = derive_stage(&stages, &[q("committed", GateStatus::Pass), q("ci", GateStatus::Queued)]);
31//! assert_eq!(s.id, "ship");
32//! assert_eq!(s.state, StageState::Pending);
33//! assert_eq!(s.owing, ["ci"]);
34//! assert!(!s.complete);
35//!
36//! // a refusal anywhere is where the task is blocked, even behind a later pending one
37//! let s = derive_stage(&stages, &[q("committed", GateStatus::Stale), q("ci", GateStatus::Queued)]);
38//! assert_eq!((s.id.as_str(), s.state), ("build", StageState::Blocked));
39//!
40//! // every question answered: complete, and the stage is the last one
41//! let s = derive_stage(&stages, &[q("committed", GateStatus::Pass), q("ci", GateStatus::Exempt)]);
42//! assert!(s.complete);
43//! assert_eq!(s.state, StageState::Complete);
44//! ```

53/// Where one stage stands, folded from the answers to its questions and never set by
54/// anybody. A refusal anywhere in the stage makes it blocked whatever else passed; a
55/// stage with nothing owed is complete; one with an unanswered question is pending; and
56/// a stage the change asks nothing of is empty, which neither blocks nor completes.
57///
58/// ```
59/// use majordomus_cli::gates::StageState;
60///
61/// assert_eq!(StageState::Blocked.as_str(), "blocked");
62/// // serialised as the same word the report prints, so no reader meets two spellings
63/// assert_eq!(serde_json::to_string(&StageState::Pending).unwrap(), "\"pending\"");
64/// assert_eq!(serde_json::from_str::<StageState>("\"empty\"").unwrap(), StageState::Empty);
65/// ```
66#[derive(
67    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
68)]
69#[serde(rename_all = "lowercase")]
70pub enum StageState {
71    /// Every question of the stage passes or is exempt.
72    Complete,
73    /// A question of the stage has not been answered: queued or unknown.
74    Pending,
75    /// A question of the stage refuses: failing, stale, or blocked by one that is.
76    Blocked,
77    /// The stage declares no question that applies here.
78    Empty,
79}
80
81impl StageState {
82    /// The word, as serialised: the same spelling the report carries, so a surface that
83    /// prints the state and one that reads it back from JSON agree on the text.
84    ///
85    /// ```
86    /// use majordomus_cli::gates::StageState;
87    ///
88    /// assert_eq!(StageState::Complete.as_str(), "complete");
89    /// for state in [StageState::Complete, StageState::Pending, StageState::Blocked, StageState::Empty] {
90    ///     assert_eq!(serde_json::to_value(state).unwrap(), state.as_str());
91    /// }
92    /// ```
93    pub fn as_str(self) -> &'static str {
94        match self {
95            Self::Complete => "complete",
96            Self::Pending => "pending",
97            Self::Blocked => "blocked",
98            Self::Empty => "empty",
99        }
100    }
101}

103/// One stage of the lifecycle, with what its questions said. Every stage of the policy
104/// gets one, in the policy's order, so a reader sees the stages behind the task as well
105/// as the one it stands at; `owing` is written only when something is owed.
106///
107/// ```
108/// use majordomus_cli::gates::{StageReport, StageState};
109///
110/// let report = StageReport {
111///     id: "commit".into(),
112///     title: "Commit and push".into(),
113///     state: StageState::Complete,
114///     questions: vec!["committed".into(), "pushed".into()],
115///     owing: Vec::new(),
116/// };
117/// let json = serde_json::to_value(&report).unwrap();
118/// assert_eq!(json["state"], "complete");
119/// assert!(json.get("owing").is_none(), "nothing owed is nothing written");
120/// ```
121#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
122pub struct StageReport {
123    /// The stage's identity.
124    pub id: String,
125    /// The stage as a heading.
126    pub title: String,
127    /// Where it stands.
128    pub state: StageState,
129    /// The questions of this stage, in policy order.
130    pub questions: Vec<String>,
131    /// The questions still owed or refusing, in policy order.
132    #[serde(default, skip_serializing_if = "Vec::is_empty")]
133    pub owing: Vec<String>,
134}

136/// Where the task stands: the one stage a reader acts on, and every stage behind and
137/// ahead of it. Produced by [`derive_stage`] and by nothing else, so the stage a report
138/// names is always the fold over the answers and never a word somebody wrote down.
139///
140/// ```
141/// use majordomus_cli::gates::{DoneQuestion, GateStatus, LifecycleStage, StageDecl, StageState, derive_stage};
142///
143/// let stages = [StageDecl { id: "build".into(), title: "Build".into(), summary: "s".into() }];
144/// let owed = DoneQuestion {
145///     id: "committed".into(), stage: "build".into(), question: "Is it committed?".into(),
146///     status: GateStatus::Queued, evidence: "owed".into(), source: "s".into(), remediation: "git commit".into(),
147/// };
148/// let stage: LifecycleStage = derive_stage(&stages, &[owed]);
149/// assert_eq!((stage.id.as_str(), stage.state), ("build", StageState::Pending));
150/// assert_eq!(stage.owing, ["committed"]);
151/// assert!(!stage.complete, "a question still owed is a task not completed");
152/// assert_eq!(stage.stages.len(), 1);
153/// ```
154#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
155pub struct LifecycleStage {
156    /// The stage the task stands at: the first one blocked, else the first one pending,
157    /// else the last stage of the policy.
158    pub id: String,
159    /// That stage as a heading.
160    pub title: String,
161    /// Where it stands.
162    pub state: StageState,
163    /// True only when every stage is complete or empty. The word `completed` is earned
164    /// here and nowhere else.
165    pub complete: bool,
166    /// The questions still owed or refusing at the stage the task stands at.
167    #[serde(default, skip_serializing_if = "Vec::is_empty")]
168    pub owing: Vec<String>,
169    /// Every stage of the policy, in order.
170    pub stages: Vec<StageReport>,
171}
172
173/// Fold the answered questions over the policy's stages. The task stands at the first
174/// stage that is blocked, else the first that is pending, else the last stage of the
175/// policy; it is complete only when every stage is complete or empty. A question naming a
176/// stage the policy does not declare is ignored, and no policy at all is no completion.
177///
178/// ```
179/// use majordomus_cli::gates::{DoneQuestion, GateStatus, StageDecl, StageState, derive_stage};
180///
181/// let decl = |id: &str| StageDecl { id: id.into(), title: id.into(), summary: "s".into() };
182/// let q = |id: &str, stage: &str, status: GateStatus| DoneQuestion {
183///     id: id.into(), stage: stage.into(), question: id.into(), status,
184///     evidence: "e".into(), source: "s".into(), remediation: "r".into(),
185/// };
186///
187/// // a later stage that refuses is where the task is, even behind an earlier pending one
188/// let s = derive_stage(&[decl("a"), decl("b")], &[q("x", "a", GateStatus::Queued), q("y", "b", GateStatus::Fail)]);
189/// assert_eq!((s.id.as_str(), s.state), ("b", StageState::Blocked));
190/// assert_eq!(s.stages[0].state, StageState::Pending);
191///
192/// // an empty stage is not owed, and a policy with no stage completes nothing
193/// let s = derive_stage(&[decl("a"), decl("b")], &[q("x", "a", GateStatus::Pass)]);
194/// assert!(s.complete && s.stages[1].state == StageState::Empty);
195/// assert!(!derive_stage(&[], &[]).complete);
196/// ```
197pub fn derive_stage(stages: &[StageDecl], questions: &[DoneQuestion]) -> LifecycleStage {
198    let reports: Vec<StageReport> = stages
199        .iter()
200        .map(|s| {
201            let mine: Vec<&DoneQuestion> = questions.iter().filter(|q| q.stage == s.id).collect();
202            let owing: Vec<String> = mine
203                .iter()
204                .filter(|q| !matches!(q.status, GateStatus::Pass | GateStatus::Exempt))
205                .map(|q| q.id.clone())
206                .collect();
207            let state = if mine.is_empty() {
208                StageState::Empty
209            } else if mine.iter().any(|q| q.status.refuses()) {
210                StageState::Blocked
211            } else if owing.is_empty() {
212                StageState::Complete
213            } else {
214                StageState::Pending
215            };
216            StageReport {
217                id: s.id.clone(),
218                title: s.title.clone(),
219                state,
220                questions: mine.iter().map(|q| q.id.clone()).collect(),
221                owing,
222            }
223        })
224        .collect();
225
226    let at = reports
227        .iter()
228        .find(|r| r.state == StageState::Blocked)
229        .or_else(|| reports.iter().find(|r| r.state == StageState::Pending))
230        .or_else(|| reports.last());
231    let complete = reports
232        .iter()
233        .all(|r| matches!(r.state, StageState::Complete | StageState::Empty))
234        && !reports.is_empty();
235    match at {
236        Some(r) => LifecycleStage {
237            id: r.id.clone(),
238            title: r.title.clone(),
239            state: if complete {
240                StageState::Complete
241            } else {
242                r.state
243            },
244            complete,
245            owing: r.owing.clone(),
246            stages: reports.clone(),
247        },
248        None => LifecycleStage {
249            id: String::new(),
250            title: String::new(),
251            state: StageState::Empty,
252            complete: false,
253            owing: Vec::new(),
254            stages: reports,
255        },
256    }
257}

259#[cfg(test)]
260mod tests {
261    use super::*;
262
263    fn decl(id: &str) -> StageDecl {
264        StageDecl {
265            id: id.into(),
266            title: id.to_uppercase(),
267            summary: "s".into(),
268        }
269    }
270    fn q(id: &str, stage: &str, status: GateStatus) -> DoneQuestion {
271        DoneQuestion {
272            id: id.into(),
273            stage: stage.into(),
274            question: id.into(),
275            status,
276            evidence: "e".into(),
277            source: "s".into(),
278            remediation: "r".into(),
279        }
280    }
281
282    #[test]
283    fn unknown_is_pending_never_complete() {
284        let s = derive_stage(&[decl("a")], &[q("x", "a", GateStatus::Unknown)]);
285        assert_eq!(s.state, StageState::Pending);
286        assert!(!s.complete);
287        assert_eq!(s.owing, ["x"]);
288    }
289
290    #[test]
291    fn a_blocked_later_stage_wins_over_an_earlier_pending_one() {
292        let s = derive_stage(
293            &[decl("a"), decl("b")],
294            &[
295                q("x", "a", GateStatus::Queued),
296                q("y", "b", GateStatus::Fail),
297            ],
298        );
299        assert_eq!((s.id.as_str(), s.state), ("b", StageState::Blocked));
300        assert_eq!(s.stages[0].state, StageState::Pending);
301    }
302
303    #[test]
304    fn an_empty_stage_neither_blocks_nor_completes_on_its_own() {
305        let s = derive_stage(&[decl("a"), decl("b")], &[q("x", "a", GateStatus::Pass)]);
306        assert!(s.complete, "an empty stage is not owed");
307        assert_eq!(s.stages[1].state, StageState::Empty);
308        let none = derive_stage(&[], &[]);
309        assert!(!none.complete, "no policy is no completion");
310    }
311
312    #[test]
313    fn the_order_is_the_policy_s_and_a_question_of_no_stage_is_ignored() {
314        let s = derive_stage(
315            &[decl("b"), decl("a")],
316            &[
317                q("x", "a", GateStatus::Queued),
318                q("y", "b", GateStatus::Queued),
319                q("z", "zz", GateStatus::Fail),
320            ],
321        );
322        assert_eq!(s.id, "b", "the first pending stage in policy order");
323        assert_eq!(
324            s.stages.iter().map(|r| r.id.as_str()).collect::<Vec<_>>(),
325            ["b", "a"]
326        );
327    }
328
329    #[test]
330    fn a_state_s_word_is_the_word_it_is_serialised_as() {
331        for state in [
332            StageState::Complete,
333            StageState::Pending,
334            StageState::Blocked,
335            StageState::Empty,
336        ] {
337            assert_eq!(serde_json::to_value(state).unwrap(), state.as_str());
338            assert_eq!(
339                serde_json::from_value::<StageState>(state.as_str().into()).unwrap(),
340                state
341            );
342        }
343        assert_eq!(StageState::Blocked.as_str(), "blocked");
344    }
