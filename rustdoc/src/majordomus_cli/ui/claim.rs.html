1//! One claim in full, for the Evidence screen's inspector (`/ui/evidence?claim=<id>`): what it
2//! says and where, the state its proof is in and what that state means, the run behind it,
3//! what changed since, how to see it again, every run the ledger holds for it, and the other
4//! claims the same test proves, each a link to its own inspector.
5//!
6//! Read from `evidence.claim`, the one capability that joins a claim to its execution; a claim
7//! the matrix does not declare is its refusal, said as such, never an empty proof.
8

18/// A claim id as the matrix spells them (`a-finding-names-how-to-see-it-again`): letters,
19/// digits, `-`, `_` and `.`, and nothing a link could carry anything else in.
20pub fn is_claim_id(id: &str) -> bool {
21    !id.is_empty()
22        && id.len() <= 128
23        && id
24            .bytes()
25            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.'))
26}

28/// The claim a query names, when it names one.
29pub fn picked(query: &[(String, String)]) -> Option<String> {
30    query
31        .iter()
32        .find(|(k, _)| k == "claim")
33        .map(|(_, v)| v.trim().to_string())
34        .filter(|v| is_claim_id(v))
35}

37/// Where a claim's inspector is.
38pub fn href(id: &str) -> String {
39    format!("{}?claim={id}#claim", super::evidence::ROUTE)
40}

42/// Ask `evidence.claim` for `id`.
43pub fn ask(ctx: &Context, id: &str) -> Answer {
44    ask_input(ctx, "evidence.claim", serde_json::json!({ "claim": id }))
45}

47fn strings(v: &Value, k: &str) -> Vec<String> {
48    v.get(k)
49        .and_then(Value::as_array)
50        .map(|a| {
51            a.iter()
52                .filter_map(Value::as_str)
53                .map(str::to_string)
54                .collect()
55        })
56        .unwrap_or_default()
57}

59fn mono(t: String) -> El {
60    el("span").class("mj-ui-mono").text(t)
61}
62
63/// How many of the ledger's runs of a claim's test the inspector lists, newest first.
64const RUNS: usize = 10;
65
66/// Seconds as a run recorded them, or `unknown` when it recorded none.
67fn seconds(run: &Value) -> String {
68    run.get("seconds")
69        .filter(|s| !s.is_null())
70        .map_or_else(|| "unknown".into(), Value::to_string)
71}
72
73/// The first ten characters of a run's commit, or `unknown` when it recorded none.
74fn commit(run: &Value) -> String {
75    let c = text(run, "commit");
76    match c.get(..10) {
77        Some(short) => short.to_string(),
78        None if c.is_empty() => "unknown".into(),
79        None => c,
80    }
81}

83/// The inspector of the claim `answer` is about. The document that says it, the code that
84/// implements it and the test that settles it lead to their technical views when the index
85/// holds them (`ctx` asks it).
86pub fn inspector(ctx: &Context, answer: &Answer, id: &str) -> El {
87    let body = unanswered(answer).unwrap_or_else(|| {
88        // `unanswered` has spoken for an answer that failed; this one answered
89        let c = answer.ok().cloned().unwrap_or(Value::Null);
90        let fact = |label: &str, value: El| {
91            el("div")
92                .class("mj-ui-fact")
93                .child(el("dt").text(label))
94                .child(el("dd").child(value))
95        };
96        let run = c.get("execution").cloned().unwrap_or(Value::Null);
97        let last = if run.is_null() {
98            el("span").class("mj-ui-dim").text("never run")
99        } else {
100            el("span")
101                .child(word_badge(&text(&run, "outcome")))
102                .child(mono(format!(
103                    " {} s · {} · {}",
104                    seconds(&run),
105                    commit(&run),
106                    value(&run, "working_tree")
107                )))
108        };
109        let runs = c
110            .get("executions")
111            .and_then(Value::as_array)
112            .cloned()
113            .unwrap_or_default();
114        let history = runs
115            .iter()
116            .rev()
117            .take(RUNS)
118            .map(|e| {
119                el("tr")
120                    .child(el("td").child(word_badge(&text(e, "outcome"))))
121                    .child(el("td").class("mj-ui-mono").text(seconds(e)))
122                    .child(el("td").class("mj-ui-mono").text(commit(e)))
123                    .child(
124                        el("td")
125                            .class("mj-ui-mono mj-ui-dim")
126                            .text(value(e, "working_tree")),
127                    )
128                    .child(
129                        el("td")
130                            .class("mj-ui-mono mj-ui-dim")
131                            .text(value(e, "at").replace('T', " ")),
132                    )
133            })
134            .collect::<Vec<_>>();
135        let changed = strings(&c, "changed");
136        let also = strings(&c, "also_proves");
137        let reproduce = text(&c, "reproduce");
138        el("section")
139            .class("mj-ui-detail")
140            .attr("id", "claim")
141            .attr("aria-label", format!("Claim {id}"))
142            .child(
143                el("div")
144                    .class("mj-ui-detail-head")
145                    .child(el("p").class("mj-ui-mono mj-ui-dim").text(id.to_string()))
146                    .child(el("h3").class("mj-ui-detail-title").text(text(&c, "claim")))
147                    .child(
148                        el("p")
149                            .class("mj-ui-badges")
150                            .child(state(&c, "state"))
151                            .child(state(&c, "status"))
152                            .child(
153                                el("a")
154                                    .class("mj-ui-more")
155                                    .attr("href", super::evidence::ROUTE)
156                                    .text("Close"),
157                            ),
158                    ),
159            )
160            .child(el("p").class("mj-ui-lead").text(text(&c, "meaning")))
161            .child(
162                el("dl")
163                    .class("mj-ui-facts")
164                    .child(fact("Said in", {
165                        let source = text(&c, "source");
166                        object_if(ctx, &format!("majordomus://document/{source}"), &source)
167                    }))
168                    .child(fact("Implemented by", {
169                        let code = text(&c, "implementation");
170                        object_if(ctx, &format!("majordomus://implementation/{code}"), &code)
171                    }))
172                    .child(fact("Settled by", {
173                        let test = text(&c, "test_path");
174                        object_if(ctx, &format!("majordomus://test/{test}"), &test)
175                    }))
176                    .child(fact("Last run", last))
177                    .child(fact(
178                        "Reproduce",
179                        if reproduce.is_empty() {
180                            el("span").class("mj-ui-dim").text("none")
181                        } else {
182                            el("code").class("mj-ui-mono").text(reproduce)
183                        },
184                    )),
185            )
186            .child(
187                el("h4")
188                    .class("mj-ui-panel-title")
189                    .text("Changed since it was proven"),
190            )
191            .child(if changed.is_empty() {
192                nothing("Nothing it names has changed since its proof.")
193            } else {
194                el("ul")
195                    .class("mj-ui-list")
196                    .children(changed.into_iter().map(|p| el("li").child(mono(p))))
197            })
198            .child(
199                el("h4")
200                    .class("mj-ui-panel-title")
201                    .text("Also proved by the same test"),
202            )
203            .child(if also.is_empty() {
204                nothing("The test proves this claim alone.")
205            } else {
206                el("p").class("mj-ui-badges").children(also.iter().map(|a| {
207                    el("a")
208                        .class("mj-ui-chip mj-ui-chip--link")
209                        .attr("href", href(a))
210                        .child(el("code").class("mj-ui-chip-value").text(a.clone()))
211                }))
212            })
213            .child(
214                el("h4")
215                    .class("mj-ui-panel-title")
216                    .text("Runs the ledger holds"),
217            )
218            .child(table(
219                &["Outcome", "Seconds", "Commit", "Tree", "At"],
220                history,
221                "The ledger holds no run of its test.",
222            ))
223            .children(more(RUNS.min(runs.len()), runs.len(), None))
224    });
225    panel(
226        "Claim",
227        "One claim of the product, with what proves it and how to see it again.",
228        body,
229        &[answer],
230    )
231}

255    /// A claim that never ran, names no reproduction, changed nothing since and shares its
256    /// test with no other claim says each of those, rather than drawing an empty field.
257    #[test]
258    fn a_claim_never_run_says_what_it_lacks() {
259        let repo = repo();
260        let ctx = repo.context().expect("a context");
261        let html = inspector(
262            &ctx,
263            &answer(Ok(
264                serde_json::json!({ "id": "c-2", "claim": "Alone", "state": "not_run" }),
265            )),
266            "c-2",
267        )
268        .render();
269        for want in [
270            "never run",
271            ">none<",
272            "Nothing it names has changed since its proof.",
273            "The test proves this claim alone.",
274        ] {
275            assert!(html.contains(want), "no {want}: {html}");
276        }
277        let html = inspector(
278            &ctx,
279            &answer(Ok(
280                serde_json::json!({ "id": "c-3", "claim": "Timed nowhere",
281                "execution": { "outcome": "pass", "commit": "0123456789abcdef" } }),
282            )),
283            "c-3",
284        )
285        .render();
286        assert!(html.contains("0123456789"), "{html}");
287        assert!(
288            html.contains(" unknown s · 0123456789 · none"),
289            "a run without its seconds or its tree says so: {html}"
290        );
291    }

294    fn a_claim_shows_its_proof_what_changed_and_the_claims_its_test_also_proves() {
295        let a = answer(Ok(serde_json::json!({
296            "id": "c-1", "claim": "A finding names how to see it again", "status": "guaranteed",
297            "source": "docs/DOCTRINE.md", "implementation": "scripts/x", "test_path": "test/cases/138.sh",
298            "state": "stale", "meaning": "The proof is older than what it is about.",
299            "execution": { "outcome": "pass", "seconds": 6, "commit": "a04b65c9110a74dc", "working_tree": "dirty" },
300            "changed": ["docs/DOCTRINE.md"], "reproduce": "bash test/run.sh 138",
301            "executions": [{ "outcome": "fail", "seconds": 5, "commit": "1111111111aa", "working_tree": "clean", "at": "2026-10-01T00:00:00Z" },
302                           { "outcome": "pass", "seconds": 6, "commit": "a04b65c9110a", "working_tree": "dirty", "at": "2026-10-02T00:00:00Z" }],
303            "also_proves": ["authored-files-are-english"]
304        })));
305        let repo = repo();
306        let ctx = repo.context().expect("a context");
307        let html = inspector(&ctx, &a, "c-1").render();
308        for want in [
309            "id=\"claim\"",
310            "The proof is older than what it is about.",
311            "bash test/run.sh 138",
312            "docs/DOCTRINE.md",
313            "/ui/evidence?claim=authored-files-are-english#claim",
314            "a04b65c911",
315        ] {
316            assert!(html.contains(want), "no {want}: {html}");
317        }
318        match (html.find("2026-10-02"), html.find("2026-10-01")) {
319            (Some(newer), Some(older)) => assert!(newer < older, "the newest run first"),
320            other => panic!("both runs are listed: {other:?}"),
321        }
322    }

324    /// What the claim names leads to its technical view when the index holds it, and is
325    /// text when it does not, so no link answers 404; more runs than it lists are counted.
326    #[test]
327    fn what_a_claim_names_is_a_link_when_the_index_holds_it() {
328        let repo = repo();
329        let ctx = repo.context().expect("a context");
330        let runs: Vec<Value> = (0..12)
331            .map(|i| serde_json::json!({ "outcome": "pass", "seconds": i, "commit": "abcdefabcdef", "at": format!("2026-10-{:02}T00:00:00Z", i + 1) }))
332            .collect();
333        let html = inspector(
334            &ctx,
335            &answer(Ok(serde_json::json!({
336                "id": "c-4", "claim": "Linked", "source": "docs/DOC_0.md",
337                "test_path": "test/cases/no_such_case.sh", "executions": runs
338            }))),
339            "c-4",
340        )
341        .render();
342        assert!(
343            html.contains(
344                "href=\"/cockpit/object?uri=majordomus%3A%2F%2Fdocument%2Fdocs%2FDOC_0.md\""
345            ),
346            "the document the index holds is a link: {html}"
347        );
348        assert!(
349            html.contains("test/cases/no_such_case.sh") && !html.contains("no_such_case.sh\""),
350            "a test the index does not hold is text: {html}"
351        );
352        assert!(html.contains("The first 10 of 12; 2 more"), "{html}");
353    }

355    #[test]
356    fn a_claim_the_matrix_does_not_declare_is_said_as_its_refusal_and_a_hostile_id_never_asked() {
357        let repo = repo();
358        let ctx = repo.context().expect("a context");
359        let refused = inspector(

368        assert_eq!(
369            picked(&q("a-finding.names_it")),
370            Some("a-finding.names_it".into())
371        );
372        assert_eq!(picked(&q("x\" onclick=\"y")), None);
373    }

375    /// A run's seconds and commit are shown as recorded: a short commit whole, and one a run
376    /// did not record as unknown.
377    #[test]
378    fn a_run_says_what_it_did_not_record() {
379        assert_eq!(seconds(&serde_json::json!({ "seconds": 6 })), "6");
380        assert_eq!(seconds(&serde_json::json!({ "seconds": null })), "unknown");
381        assert_eq!(seconds(&serde_json::json!({})), "unknown");
382        assert_eq!(
383            commit(&serde_json::json!({ "commit": "0123456789abcdef" })),
384            "0123456789"
385        );
386        assert_eq!(commit(&serde_json::json!({ "commit": "abc" })), "abc");
387        assert_eq!(commit(&serde_json::json!({})), "unknown");
388    }
