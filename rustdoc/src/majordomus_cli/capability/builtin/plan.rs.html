1//! The `plan` module: the milestone and issue model's *derivations*, projected.
2//!
3//! The records themselves have always been served — `majordomus://issue/I0001` returns the
4//! file through `objects.get` — and that is the least interesting half of the model. What
5//! makes it worth having is everything nobody authored: which issues are READY, which are
6//! BLOCKED and on what, the topological waves the graph allows to run at once, the roadmap
7//! order the milestone graph implies, the milestone a worker is on and the one issue to
8//! take next. Until this module those lived only in `lib/plan.sh`, so an agent asking the
9//! shared MCP server what to work on could be handed 184 documents and no answer.
10//!
11//! Every capability here reads [`crate::plan::Plan`], built on the call out of records the
12//! index already holds. Nothing is cached: the plan changes under the process — a
13//! transition writes a lifecycle marker into a record between two calls — and a `next`
14//! that answered from a snapshot would send two workers to one issue.
15//!
16//! One capability here writes: `plan.transition` moves an issue through `start`, `verify`
17//! and `done`. It is the first development operation of the runtime, and it is here because
18//! [ADR 0040] decided that development semantics are capabilities of this registry and every
19//! surface a consumer of them — not because the registry's read-only habit was an accident.
20//! Until it landed, an agent could be told what to work on and had no way to say it had
21//! started; the derivation and the transition were in two programs, and only one of them was
22//! reachable from MCP, HTTP or the Cockpit.
23//!
24//! What it does not do is own the semantics. The guards are [`crate::plan::check`], the
25//! resulting status is computed by the one `derive` every reader uses, and the record it
26//! writes is the same `.ai/repo/project/issues/<id>.yaml` `lib/plan.sh` writes, to the byte.
27//! `plan evidence` stays a command-line operation for now: it appends a block rather than
28//! setting a field, and it is the next transition to converge, not this one.
29//!
30//! [ADR 0040]: https://github.com/korczis/prismatic-majordomus/blob/master/.ai/repo/adrs/0040-development-semantics-are-capabilities-of-one-runtime.md

58/// The derived plan of the repository this process serves. Built on the call: the plan
59/// changes under the process.
60///
61/// A repository with no `.ai/repo/project/` answers with an empty plan rather than a
62/// refusal. The shell tool exits 12 there because it is telling a person where the model
63/// would live; a projection is answering what this repository holds, and "no milestones and
64/// no issues" is that answer. Only a named record that does not exist is a `not found`.
65fn plan_of(ctx: &Context) -> Result<Plan, CapabilityError> {
66    Ok(Plan::build(&ctx.index))
67}

71/// The milestone the benchmark cases name, declared once because three case sets need it.
72///
73/// A repository with no plan still has to be timed on these capabilities — the coverage
74/// gate counts targets, not repositories — and its OpenAPI operation still has to show an
75/// example of every parameter it declares, because the examples *are* the cases and case
76/// 92 refuses a parameter no case ever sets. So the fallback names a milestone that exists
77/// nowhere: the empty answer is the operation being measured there, and it is the answer a
78/// caller filtering on a milestone that does not exist gets.
79fn case_milestone(plan: &Plan) -> String {
80    plan.milestones
81        .first()
82        .map_or_else(|| "M000".to_string(), |m| m.id.clone())
83}

98    /// Only issues in this execution wave.

100}

317fn plan_status(
318    ctx: &Context,
319    input: PlanMilestoneFilter,
320) -> Result<PlanStatusReport, CapabilityError> {
321    let plan = plan_of(ctx)?;
322    let only = input.milestone.as_deref();
323    if let Some(m) = only {
324        if plan.milestone(m).is_none() {
325            return Err(CapabilityError::NotFound(format!("no milestone '{m}'")));
326        }
327    }
328    let mut counts = PlanCounts {
329        total: 0,
330        required: 0,
331        by_status: plan.statuses.issue.iter().map(|s| (s.clone(), 0)).collect(),
332    };
333    for i in &plan.issues {
334        if only.is_some_and(|m| i.milestone != m) {
335            continue;
336        }
337        counts.total += 1;
338        *counts.by_status.entry(i.status.clone()).or_insert(0) += 1;
339    }
340    counts.required = counts.total - counts.by_status.get("CANCELLED").copied().unwrap_or(0);
341    Ok(PlanStatusReport {
342        milestones: plan
343            .milestones
344            .iter()
345            .filter(|m| only.is_none_or(|only| m.id == only))
346            .map(|m| PlanMilestoneProgress {
347                id: m.id.clone(),
348                status: m.status.clone(),
349                title: m.title.clone(),
350                counts: m.counts.clone(),
351            })
352            .collect(),
353        next_ready: plan.next_ready(only).cloned(),
354        project: plan.project.clone(),
355        statuses: plan.statuses.clone(),
356        counts,
357    })
358}

360fn plan_issues(ctx: &Context, input: PlanIssueFilter) -> Result<PlanIssueList, CapabilityError> {
361    let plan = plan_of(ctx)?;
362    if let Some(s) = input.status.as_deref() {
363        if !plan.statuses.issue.iter().any(|v| v == s) {
364            return Err(CapabilityError::InvalidInput(format!(
365                "'{s}' is not an issue status; the vocabulary is {}",
366                plan.statuses.issue.join(", ")
367            )));
368        }
369    }
370    let issues: Vec<PlanIssue> = plan
371        .issues
372        .iter()
373        .filter(|i| input.milestone.as_deref().is_none_or(|m| i.milestone == m))
374        .filter(|i| input.status.as_deref().is_none_or(|s| i.status == s))
375        .filter(|i| input.wave.is_none_or(|w| i.wave == w))
376        .cloned()
377        .collect();
378    Ok(PlanIssueList {
379        total: issues.len(),
380        issues,
381        statuses: plan.statuses,
382    })
383}

385fn plan_waves(
386    ctx: &Context,
387    input: PlanMilestoneFilter,
388) -> Result<PlanWaveReport, CapabilityError> {
389    let plan = plan_of(ctx)?;
390    let only = input.milestone.as_deref();
391    let view = |w: &PlanWave| PlanWaveView {
392        wave: w.wave,
393        issues: plan
394            .issues
395            .iter()
396            .filter(|i| w.issues.contains(&i.id))
397            .filter(|i| only.is_none_or(|m| i.milestone == m))
398            .cloned()
399            .collect(),
400    };
401    Ok(PlanWaveReport {
402        waves: plan
403            .waves
404            .iter()
405            .map(view)
406            .filter(|w| !w.issues.is_empty())
407            .collect(),
408        serialised_by_scope: plan
409            .findings
410            .iter()
411            .filter(|f| f.code == "scope_conflict")
412            .cloned()
413            .collect(),
414    })
415}

417fn plan_next(ctx: &Context, input: PlanMilestoneFilter) -> Result<PlanNextIssue, CapabilityError> {
418    let plan = plan_of(ctx)?;
419    let issue = plan.next_ready(input.milestone.as_deref()).cloned();
420    Ok(PlanNextIssue {
421        reason: issue.is_none().then(|| {
422            "no issue is READY; every one of them waits on a dependency or on its milestone's gate"
423                .to_string()
424        }),
425        active_milestone: input
426            .milestone
427            .unwrap_or_else(|| plan.project.active_milestone.clone()),
428        issue,
429    })
430}

432fn plan_roadmap(ctx: &Context, _: Empty) -> Result<PlanRoadmap, CapabilityError> {
433    let plan = plan_of(ctx)?;
434    let ordered: Vec<PlanMilestone> = plan.roadmap().into_iter().cloned().collect();
435    let mut open = ordered
436        .iter()
437        .filter(|m| !matches!(m.status.as_str(), "DONE" | "CANCELLED" | "SUPERSEDED"));
438    let now = open
439        .next()
440        .filter(|m| m.blocked_by.is_empty())
441        .map(|m| m.id.clone());
442    let next = if now.is_some() {
443        open.next().map(|m| m.id.clone())
444    } else {
445        // The first open milestone is itself blocked: it is not "now", it is "next".
446        ordered
447            .iter()
448            .find(|m| !matches!(m.status.as_str(), "DONE" | "CANCELLED" | "SUPERSEDED"))
449            .map(|m| m.id.clone())
450    };
451    Ok(PlanRoadmap {
452        milestones: ordered,
453        now,
454        next,
455    })
456}

458fn plan_validate(ctx: &Context, _: Empty) -> Result<PlanValidation, CapabilityError> {
459    let plan = plan_of(ctx)?;
460    Ok(PlanValidation {
461        valid: plan.failures() == 0,
462        milestones: plan.milestones.len(),
463        issues: plan.issues.len(),
464        failures: plan.failures(),
465        warnings: plan.warnings(),
466        findings: plan.findings,
467    })
468}

470fn plan_record(ctx: &Context, input: PlanRecordInput) -> Result<PlanRecord, CapabilityError> {
471    let plan = plan_of(ctx)?;
472    if let Some(m) = plan.milestone(&input.id) {
473        let issues = plan
474            .issues
475            .iter()
476            .filter(|i| i.milestone == m.id)
477            .cloned()
478            .collect();
479        return Ok(PlanRecord::PlanMilestone {
480            milestone: Box::new(m.clone()),
481            issues,
482        });
483    }
484    if let Some(i) = plan.issue(&input.id) {
485        let depends_on = i
486            .depends_on
487            .iter()
488            .filter_map(|d| plan.issue(d))
489            .cloned()
490            .collect();
491        return Ok(PlanRecord::PlanIssue {
492            issue: Box::new(i.clone()),
493            depends_on,
494        });
495    }
496    Err(CapabilityError::NotFound(format!(
497        "no milestone or issue '{}'",
498        input.id
499    )))
500}

679#[derive(Debug, Serialize, Deserialize, JsonSchema)]
680#[serde(deny_unknown_fields)]
681/// The input of `plan.transition`: which issue to move, and which way.
682///
683/// Two fields and no third. There is no timestamp — the move is stamped with the moment it
684/// happened, never with one the caller supplies, because a lifecycle a caller can backdate
685/// is not a record of what happened. There is no force or reason field either: a transition
686/// the model refuses is refused, and the way to make it legal is to satisfy the guard rather
687/// than to ask past it.
688///
689/// ```
690/// use majordomus_cli::capability::builtin::plan::PlanTransitionInput;
691/// use majordomus_cli::plan::Transition;
692///
693/// // The wire form an MCP client or an HTTP POST sends.
694/// let input: PlanTransitionInput =
695///     serde_json::from_str(r#"{"issue":"I0001","transition":"done"}"#).unwrap();
696/// assert_eq!(input.issue, "I0001");
697/// assert_eq!(input.transition, Transition::Done);
698/// assert_eq!(input.transition.field(), "completed_at");
699/// assert_eq!(input.transition.event(), "plan_done");
700///
701/// // Unknown fields are refused rather than ignored: a caller that misspells `transition`
702/// // is told so instead of having its move silently dropped.
703/// assert!(serde_json::from_str::<PlanTransitionInput>(
704///     r#"{"issue":"I0001","transition":"done","force":true}"#).is_err());
705/// ```
706pub struct PlanTransitionInput {
707    /// The issue to move, by its id — which is also its file name.
708    pub issue: String,
709    /// Which move: `start`, `verify` or `done`.
710    pub transition: Transition,
711}
712
713#[derive(Debug, Serialize, Deserialize, JsonSchema)]
714/// What one transition did.
715///
716/// `from` and `to` are both derived statuses, read before and after the write. They are
717/// reported rather than assumed because the move does not determine the status on its own:
718/// `done` on a record whose evidence is incomplete leaves it in VERIFY, and the caller is
719/// told that rather than told it succeeded.
720///
721/// ```
722/// use majordomus_cli::capability::builtin::plan::PlanTransitionResult;
723///
724/// // What a `done` looks like when the record's required evidence is not all present: the
725/// // call succeeded, the field was written, and the issue is still in VERIFY. A caller that
726/// // read only the absence of an error would report this as finished; a caller that reads
727/// // `to` knows better, which is why the field is here.
728/// let held: PlanTransitionResult = serde_json::from_str(r#"{
729///     "issue":"I0001","transition":"done","from":"VERIFY","to":"VERIFY",
730///     "field":"completed_at","at":"2026-09-11T12:00:00Z","event":"plan_done"
731/// }"#).unwrap();
732/// assert_eq!(held.to, "VERIFY", "the status is derived from the record, not from the move");
733///
734/// // `next_ready` is absent when the plan has none left to hand out, which is ordinary
735/// // rather than an error, so it is omitted rather than spelled as an empty string.
736/// assert!(held.next_ready.is_none());
737/// ```
738pub struct PlanTransitionResult {
739    /// The issue that moved.
740    pub issue: String,
741    /// The move that was performed.
742    pub transition: Transition,
743    /// The status the issue was in before.
744    pub from: String,
745    /// The status it is in now, derived from the record as written.
746    pub to: String,
747    /// The field that was stamped.
748    pub field: String,
749    /// The timestamp written into it, and into `updated_at`.
750    pub at: String,
751    /// The ledger event appended.
752    pub event: String,
753    /// The issue a worker should take next, derived after the move. Absent when the plan
754    /// has none — which after a `start` is the ordinary case, because the issue just taken
755    /// is no longer READY.
756    #[serde(default, skip_serializing_if = "Option::is_none")]
757    pub next_ready: Option<String>,
758}
759
760/// The benchmark cases of the one capability here that writes.
761///
762/// Every case is a refusal, and that is deliberate rather than a gap. A case is *executed* —
763/// by `majordomus bench` against this repository, and by the HTTP and MCP suites against a
764/// fixture — so a case that moved an issue would change the plan every time the suite ran,
765/// and a measurement whose cost is a changed record is not a measurement. The runner states
766/// the principle that makes this sound: *"a refusal is an answer, and answering is the work
767/// a benchmark exists to measure"*, and only `Internal` is fatal.
768///
769/// The refusal path is also the honest thing to time: it builds the plan, resolves the issue
770/// and evaluates the guard, which is everything the write path does except the two syscalls
771/// at the end.
772///
773/// The refusal must be a *refusal* and not a lookup miss. An earlier version named an issue
774/// that could not exist, which answered 404 rather than 422, and `tests/http_serve.rs` holds
775/// every route to answering its own cases — 200, or 422 for a command that turns a caller
776/// down. Choosing the move from the record keeps the case inside that contract.
777///
778/// So a plan with no issue yields no case, and the handler is not bent to make one: naming
779/// an issue the plan does not have is `NotFound` whether or not the plan holds other issues,
780/// and a 422 that appeared only when the plan is empty would make the same request's answer
781/// depend on records it never named. The descriptor declares the requirement conditional on
782/// `BenchmarkPrecondition::PlanHoldsAnIssue` instead, and coverage reports it inapplicable in
783/// such a repository — and missing, as before, in one whose plan holds an issue.
784impl BenchmarkCases for PlanTransitionInput {
785    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
786        let plan = Plan::build(ctx.index);
787        // The move that this issue's own status refuses. Every issue has one: a READY issue
788        // cannot be verified, and an issue in any other status cannot be started. So the
789        // case is chosen from the record rather than from a status this repository happens
790        // to hold today, and it stays a refusal whatever the plan looks like when it runs.
791        let refused_move = |i: &PlanIssue| {
792            if i.status == "READY" {
793                Transition::Verify
794            } else {
795                Transition::Start
796            }
797        };
798        plan.issues
799            .first()
800            .map(|i| {
801                vec![NamedCase::new(
802                    "refused-by-status",
803                    PlanTransitionInput {
804                        issue: i.id.clone(),
805                        transition: refused_move(i),
806                    },
807                )]
808            })
809            .unwrap_or_default()
810    }
811}

813fn plan_transition(
814    ctx: &Context,
815    input: PlanTransitionInput,
816) -> Result<PlanTransitionResult, CapabilityError> {
817    let plan = Plan::build(&ctx.index);
818    let before = plan
819        .issue(&input.issue)
820        .ok_or_else(|| {
821            CapabilityError::NotFound(format!(
822                "no issue '{}'; plan.issues lists every one",
823                input.issue
824            ))
825        })?
826        .status
827        .clone();
828
829    // The record's own file, as the index recorded it. Located here rather than composed
830    // from the id so that there is one answer to where a record lives.
831    let rel = ctx
832        .index
833        .objects
834        .iter()
835        .find(|o| o.kind == crate::plan::ISSUE && o.identity == input.issue)
836        .map(|o| o.provenance.path.clone())
837        .ok_or_else(|| {
838            CapabilityError::NotFound(format!(
839                "issue '{}' is in the plan but no object carries its file",
840                input.issue
841            ))
842        })?;
843
844    let root = Path::new(&ctx.index.repository.root);
845    let events = ctx
846        .index
847        .share
848        .as_ref()
849        .map(|s| s.join("events.yaml"))
850        .ok_or_else(|| {
851            CapabilityError::Refused(
852                "this index was built without a share directory, so the event vocabulary cannot be read; a transition may not write an event it cannot validate".into(),
853            )
854        })?;
855    let vocabulary = crate::ledger::Vocabulary::load(&events)
856        .map_err(|e| CapabilityError::Refused(e.to_string()))?;
857
858    let at = crate::ledger::now();
859    crate::plan::transition(
860        root,
861        &root.join(&rel),
862        &ctx.index,
863        &plan,
864        &input.issue,
865        input.transition,
866        &at,
867        &vocabulary,
868        &ctx.index.repository.git,
869    )
870    .map_err(|e| match e {
871        TransitionError::NoSuchIssue(_) => CapabilityError::NotFound(e.to_string()),
872        TransitionError::Refused(_) => CapabilityError::Refused(e.to_string()),
873        TransitionError::Write { .. } => CapabilityError::Refused(e.to_string()),
874    })?;
875
876    // What the plan says now, through the same derivation every reader uses.
877    let after = Plan::after(&ctx.index, &input.issue, input.transition, &at);
878    Ok(PlanTransitionResult {
879        to: after
880            .issue(&input.issue)
881            .map(|i| i.status.clone())
882            .unwrap_or_default(),
883        next_ready: after.next_ready(None).map(|i| i.id.clone()),
884        issue: input.issue,
885        transition: input.transition,
886        from: before,
887        field: input.transition.field().to_string(),
888        at,
889        event: input.transition.event().to_string(),
890    })
891}
