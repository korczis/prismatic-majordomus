2//! the committed ones still match the registry. Check mode writes nothing.

11/// Run `majordomus generate`.
12pub fn run(args: GenerateArgs) -> Result<u8> {
13    let app = App::load(&args.repo)?;
14    // Before anything is planned, and before anything is judged: an executable of another
15    // generation rewrites every artifact from a model this tree does not have, and nothing
16    // in the bytes says so, so the damage reads as an ordinary regeneration. Where the
17    // repository being generated is the one that declares this executable, the two must be
18    // the same generation.
19    refuse_foreign_generation(&app)?;
20    let targets: &[Target] = match args.target {
21        GenerateTarget::All => Target::ALL,
22        GenerateTarget::Openapi => &[Target::OpenApi],
23        GenerateTarget::Docs => &[Target::Docs],
24        GenerateTarget::Benchmarks => &[Target::Benchmarks],
25        GenerateTarget::Registry => &[Target::Registry],
26        GenerateTarget::Allow => &[Target::Allow],
27        GenerateTarget::Providers => &[Target::Providers],
28        GenerateTarget::Site => &[Target::Site],
29        GenerateTarget::Distribution => &[Target::Distribution],
30        GenerateTarget::Manifest => &[Target::Manifest],
31        GenerateTarget::Web => &[Target::Web],
32        GenerateTarget::Changelog => &[Target::Changelog],
33        GenerateTarget::Deployment => &[Target::Deployment],
34        GenerateTarget::Graph => &[Target::Graph],
35        GenerateTarget::Design => &[Target::Design],
36        GenerateTarget::Economics => &[Target::Economics],
37    };
38    let artifacts = generate::plan(&app, targets)?;
39    // the contract before the bytes: an artifact that carries no provenance, declares an
40    // encoding its suffix contradicts, or breaks the schema it names is never written and
41    // never reported in sync
42    let schemas = generate::GeneratedSchemas::load(&app.share.generated_schemas_dir())?;
43    generate::verify(&artifacts, &schemas)?;
44    let root = args
45        .out
46        .clone()
47        .unwrap_or_else(|| app.repository.root().to_path_buf());
48    let stdout = std::io::stdout();
49    let mut out = stdout.lock();
50    if args.check {
51        generate::check(&root, &artifacts)?;
52        for a in &artifacts {
53            writeln!(
54                out,
55                "OK   generated   {} — {}, matches the registry",
56                a.path,
57                a.schema.as_deref().unwrap_or(a.format.suffix())
58            )
59            .map_err(Error::Transport)?;
60        }
61        writeln!(out, "generate --check: in sync").map_err(Error::Transport)?;
62    } else {
63        for path in generate::write(&root, &artifacts)? {
64            writeln!(
65                out,
66                "{}",
67                path.strip_prefix(&root).unwrap_or(&path).display()
68            )
69            .map_err(Error::Transport)?;
70        }
71    }
72    Ok(0)
73}

75/// Refuse when this executable is not of the generation of the tree it was asked to derive.
76///
77/// The question is only asked where the repository carries *this* crate's manifest: a
78/// foreign repository has no opinion about this executable, and generating there is exactly
79/// what a released binary is for. A manifest at the same path that declares some other
80/// package — a test fixture's crate, a repository that keeps its own tool at that path — is
81/// a foreign repository too, whatever it states.
82///
83/// Where the repository does declare this crate, three things are asked in turn, and any
84/// one of them answered is enough to proceed:
85///
86/// 1. **The version.** Every artifact carries `majordomus-cli <version>` in its provenance
87///    header, so a binary of another version restamps all of them with a version the tree
88///    does not have. That is the 0.2.0-into-0.3.1 incident, and the cheapest question.
89/// 2. **The generation.** Was this executable built from *these* sources? [`crate::GENERATION`]
90///    against the same digest taken over the tree's crate. Equal means it is this tree's
91///    executable by construction — which is the ordinary case, because `scripts/derive`
92///    builds it from the tree it is about to derive, and it stays true when a session edits
93///    a capability and rebuilds.
94/// 3. **The model.** When the sources differ, the executable may still be interchangeable:
95///    a build from a sibling worktree whose crate differs by a comment produces the same
96///    artifacts. So the registry it projects is compared against the registry the tree has
97///    committed at `docs/generated/registry.json` — its builtin half, since the declarative
98///    kinds in that document are the repository's content and not this executable's model.
99///    Equal means the same model, observed rather than assumed, and a correctly-matching
100///    prebuilt `MAJORDOMUS_BIN` keeps working without the ten-minute rebuild several
101///    sessions set it to avoid.
102///
103/// None of the three is a verdict about the *tree*. If this executable's model disagrees
104/// with a tree it was not built from, the tree is not the thing that is out of date, and
105/// saying it was is how this became destructive: the operator's next move is to "fix" the
106/// tree by running the command that overwrites it. The refusal names the executable, exits
107/// [`REFUSED`](crate::error::Error::Refused) rather than the contract-unmet code a stale
108/// artifact uses, and writes nothing.
109fn refuse_foreign_generation(app: &App) -> Result<()> {
110    let crate_dir = app.repository.root().join(CRATE_DIR);
111    let manifest = crate_dir.join("Cargo.toml");
112    let Ok(text) = std::fs::read_to_string(&manifest) else {
113        return Ok(());
114    };
115    if package_field(&text, "name").as_deref() != Some(CRATE_NAME) {
116        return Ok(());
117    }
118    // A manifest with no crate root beside it is not a crate this executable could have
119    // been built from, and a generation computed over it is a hash of a few files rather
120    // than of a model. The tree that does this on purpose is a fixture: a repository
121    // written to say "the crate lives here", so that the generator projects the design into
122    // it. Reading that as a foreign generation refused three cases for a property their
123    // subject does not have — and it was the same refusal that protects a real checkout, so
124    // it could not be told apart from the thing it exists to catch.
125    //
126    // The test is the crate root — `src/lib.rs` or `src/main.rs` — and not the presence of
127    // `src/`, because this command writes `src/web/tokens.css` and `src/design/tokens.yaml`
128    // into the tree it generates: a directory this executable created a moment ago is not
129    // evidence that the tree carries the crate. A real checkout always has the root; a
130    // distribution carries no manifest and left above.
131    if !crate_dir.join("src/lib.rs").is_file() && !crate_dir.join("src/main.rs").is_file() {
132        return Ok(());
133    }
134    if let Some(declared) = package_version(&text) {
135        if declared != crate::VERSION {
136            return Err(Error::Refused {
137                code: REFUSED,
138                reason: format!(
139                    "this executable is {running} and {CRATE_DIR}/Cargo.toml declares \
140                     {declared}: generating would stamp every artifact with a version this \
141                     tree does not have. The tree is not stale; this executable is. \
142                     Rebuild it — `cargo build --manifest-path {CRATE_DIR}/Cargo.toml` — or \
143                     run `scripts/derive` with MAJORDOMUS_BIN unset, which builds it before \
144                     it generates.",
145                    running = crate::VERSION,
146                ),
147            });
148        }
149    }
150    let tree = crate::generation::crate_generation(&crate_dir);
151    if tree.as_deref() == Some(crate::GENERATION) {
152        return Ok(());
153    }
154    if projects_the_committed_registry(app) {
155        return Ok(());
156    }
157    Err(Error::Refused {
158        code: REFUSED,
159        reason: format!(
160            "this executable is not of this tree's generation, and deriving with it would \
161             rewrite every artifact from a model this tree does not have.\n  \
162             executable  {exe}\n              generation {ours}, built from commit {commit}\n  \
163             tree        {root}\n              generation {theirs}, over {CRATE_DIR}/{{{inputs}}}\n\
164             The tree is not stale; this executable is, so nothing was written and nothing \
165             was judged. Rebuild it — `cargo build --manifest-path {CRATE_DIR}/Cargo.toml` \
166             — or run `scripts/derive` with MAJORDOMUS_BIN unset, which builds it before it \
167             generates. Do not run `majordomus generate` with this executable: that is the \
168             move that destroys the artifacts.",
169            exe = std::env::current_exe()
170                .map(|p| p.display().to_string())
171                .unwrap_or_else(|_| "(unknown path)".into()),
172            ours = short(crate::GENERATION),
173            commit = short(crate::COMMIT),
174            root = app.repository.root().display(),
175            theirs = tree.as_deref().map_or("unreadable", short),
176            inputs = crate::generation::GENERATION_INPUTS.join(", "),
177        ),
178    })
179}

181/// Does this executable project the registry this tree has committed?
182///
183/// The registry manifest is the model every other artifact is a projection of, and
184/// `docs/generated/registry.json` is that model as the tree carries it. Comparing them
185/// asks whether this executable and the one that wrote the tree's artifacts agree about
186/// what the repository *is* — a property of the model, not of a string.
187///
188/// Two kinds of member are dropped before comparing, and for the same reason: neither is
189/// this executable's model. The provenance members are the wrapper [`generate::Document`]
190/// adds — `generator` in particular carries the version, which rule 1 has already decided.
191/// The index-derived members are the repository's own content: `declarative_kinds` is
192/// whichever kinds the tree has objects of, so a session that adds a document of a new kind
193/// makes the committed registry legitimately stale, and comparing that member would report
194/// the *executable* as foreign for something the executable had no part in. Any other
195/// member is compared, so one added later makes this answer `false` and the derivation
196/// refuse — the safe direction.
197fn projects_the_committed_registry(app: &App) -> bool {
198    let path = app
199        .repository
200        .root()
201        .join(generate::OUT_DIR)
202        .join("registry.json");
203    let Ok(text) = std::fs::read_to_string(&path) else {
204        return false;
205    };
206    let Ok(serde_json::Value::Object(committed)) = serde_json::from_str(&text) else {
207        return false;
208    };
209    let serde_json::Value::Object(ours) = generate::registry_manifest(&app.context.registry) else {
210        return false;
211    };
212    match (
213        serde_json::to_string(&model_only(committed)),
214        serde_json::to_string(&model_only(ours)),
215    ) {
216        (Ok(theirs), Ok(ours)) => theirs == ours,
217        _ => false,
218    }
219}
220
221/// A registry document with everything that is not this executable's model removed.
222fn model_only(members: serde_json::Map<String, serde_json::Value>) -> serde_json::Value {
223    serde_json::Value::Object(
224        members
225            .into_iter()
226            .filter(|(k, _)| {
227                !PROVENANCE_MEMBERS.contains(&k.as_str())
228                    && !INDEX_DERIVED_MEMBERS.contains(&k.as_str())
229            })
230            .collect(),
231    )
232}

234/// The prefix of a digest a diagnostic names — a generation, or the commit a build recorded.
235/// The whole value identifies the thing; twelve hex characters distinguish it, which is what
236/// a person comparing two lines of a refusal needs.
237fn short(digest: &str) -> &str {
238    if digest.len() <= SHORT_DIGEST {
239        digest
240    } else {
241        &digest[..SHORT_DIGEST]
242    }
243}
244
245/// How much of a digest [`short`] keeps.
246const SHORT_DIGEST: usize = 12;
247
248/// The members `generate::Document` writes above a document's own: provenance, not model.
249const PROVENANCE_MEMBERS: &[&str] = &["schema", "generated", "generator"];
250
251/// The members of the registry document that come from the index rather than from the
252/// executable: a property of the repository being derived, and no evidence about which
253/// executable is deriving it.
254const INDEX_DERIVED_MEMBERS: &[&str] = &["declarative_kinds"];
255
256/// The exit code of a refusal, from the exit-code contract in `docs/CLI.md`: the command
257/// declined. Not `CONTRACT_UNMET`, which is what a stale artifact reports — telling them
258/// apart is what lets `scripts/derive-check` stop printing "run scripts/derive" at an
259/// operator whose executable, not whose tree, is the thing out of date.
260const REFUSED: u8 = 15;
261
262/// Where this crate lives in the repository that declares it.
263const CRATE_DIR: &str = "apps/majordomus-cli";
264
265/// The name of the crate this executable was built from, as its own manifest declares it.
266const CRATE_NAME: &str = env!("CARGO_PKG_NAME");

268/// The `version` of the manifest's `[package]` table: the one Rust reader of the authority,
269/// [`crate::release::version::declared_in`], and never a second parse of it here.
270fn package_version(text: &str) -> Option<String> {
271    crate::release::version::declared_in(text)
272}
273
274/// One quoted scalar of the manifest's `[package]` table. Read without a TOML parser,
275/// because the fields wanted here are quoted scalars in a table whose header is
276/// unambiguous, and a dependency to read them would be a dependency this crate argues
277/// against carrying.
278fn package_field(text: &str, key: &str) -> Option<String> {
279    let mut in_package = false;
280    for line in text.lines() {
281        let line = line.trim();
282        if line.starts_with('[') {
283            in_package = line == "[package]";
284            continue;
285        }
286        if !in_package {
287            continue;
288        }
289        let Some(rest) = line.strip_prefix(key) else {
290            continue;
291        };
292        let Some(rest) = rest.trim_start().strip_prefix('=') else {
293            continue;
294        };
295        let rest = rest.trim();
296        let value = rest.split('#').next().unwrap_or(rest).trim();
297        return Some(value.trim_matches('"').to_string());
298    }
299    None
300}

314    #[test]
315    fn the_provenance_members_are_exactly_what_the_document_wrapper_adds() {
316        // The model comparison subtracts these from the committed registry document. If the
317        // wrapper ever adds a fourth, this fails here rather than by quietly comparing a
318        // provenance line as if it were part of the model.
319        let document = crate::generate::Document::new(
320            "registry",
321            crate::generate::REGISTRY_SCHEMA,
322            "a source line",
323            serde_json::json!({ "modules": [] }),
324        );
325        let stamped = document.stamped("9.9.9");
326        let added: Vec<&str> = stamped
327            .as_object()
328            .expect("a stamped document is an object")
329            .keys()
330            .filter(|k| k.as_str() != "modules")
331            .map(String::as_str)
332            .collect();
333        assert_eq!(added, PROVENANCE_MEMBERS);
334    }

362    #[test]
363    fn a_manifest_declaring_another_crate_is_a_foreign_repository() {
364        // A fixture's crate at the path this repository keeps its own tool: the version it
365        // states is that crate's, and no opinion about this executable.
366        let fixture = "[package]\nname = \"fixture\"\nversion = \"1.0.0\"\n";
367        assert_ne!(package_field(fixture, "name").as_deref(), Some(CRATE_NAME));
368        let own = format!("[package]\nname = \"{CRATE_NAME}\"\nversion = \"0.3.1\"\n");
369        assert_eq!(package_field(&own, "name").as_deref(), Some(CRATE_NAME));
370        // `name` is not the prefix of some other key that happens to start with it
371        assert_eq!(
372            package_field("[package]\nnamespace = \"x\"\n", "name"),
373            None
374        );
375    }

395    /// The guard's subject is a crate, and a directory holding only a manifest is not one.
396    /// Both directions, because the exemption is only safe while the loaded case still

398    #[test]

405            format!(
406                "[package]\nname = \"{CRATE_NAME}\"\nversion = \"{}\"\n",
407                crate::VERSION
408            ),
409        )
410        .expect("manifest");
411
412        assert!(
413            crate::generation::crate_generation(&crate_dir).is_some(),
414            "a manifest alone still hashes to something, which is the trap"
415        );
416        assert_ne!(
417            crate::generation::crate_generation(&crate_dir).as_deref(),
418            Some(crate::GENERATION),
419            "and that something is not this executable's generation"
420        );
421        assert!(
422            !crate_dir.join("src/lib.rs").is_file() && !crate_dir.join("src/main.rs").is_file(),
423            "the fixture under test carries no crate root"
424        );
425
426        // and what this command writes into the tree it generates does not make one: a
427        // `src/` full of generated stylesheets is still not a crate
428        std::fs::create_dir_all(crate_dir.join("src/web")).expect("src/web");
429        std::fs::write(crate_dir.join("src/web/tokens.css"), ":root{}\n").expect("token css");
430        assert!(
431            !crate_dir.join("src/lib.rs").is_file() && !crate_dir.join("src/main.rs").is_file(),
432            "a directory the generator created is not evidence of the crate"
433        );
434
435        // with a crate root beside it, the generation is the tree's and differs from ours:
436        // the case the guard exists for, still a refusal
437        std::fs::write(crate_dir.join("src/main.rs"), "fn main() {}\n").expect("source");
438        assert!(crate_dir.join("src/main.rs").is_file());
439        assert_ne!(
440            crate::generation::crate_generation(&crate_dir).as_deref(),
441            Some(crate::GENERATION)
442        );
443    }
