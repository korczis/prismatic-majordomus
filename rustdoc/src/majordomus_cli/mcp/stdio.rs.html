1//! The stdio transport: one JSON message per line in, one per line out. Nothing but
2//! protocol messages is ever written to the output; diagnostics go through `tracing`,
3//! which the executable points at stderr. EOF on input ends the session cleanly, and so
4//! does a closed output, because both mean the client has gone.
5//!
6//! The loop does not know what answers a message: a local [`Server`], or a bridge to the
7//! shared server of another process, is a function from a message to an optional
8//! response, and the loop drives whichever it is given.
9
10use std::io::{BufRead, Write};
11
12use serde_json::Value;
13
14use crate::error::{Error, Result};
15
16use super::protocol::Server;

18/// Serve until the input ends, answering every message through `handle`. Returns the
19/// number of messages answered.
20///
21/// ```
22/// use majordomus_cli::mcp::stdio;
23/// use serde_json::json;
24/// let input = b"{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"ping\"}\nnot json\n\n";
25/// let mut out = Vec::new();
26/// let n = stdio::serve(&input[..], &mut out, |m| Some(json!({ "jsonrpc": "2.0", "id": m["id"], "result": {} }))).unwrap();
27/// let lines: Vec<&str> = std::str::from_utf8(&out).unwrap().lines().collect();
28/// assert_eq!(n, 2, "one answer, one parse error");
29/// assert!(lines[0].contains("\"result\""));
30/// assert!(lines[1].contains("-32700"));
31/// ```
32pub fn serve<R, W, H, M>(input: R, mut output: W, mut handle: H) -> Result<usize>
33where
34    R: BufRead,
35    W: Write,
36    H: FnMut(Value) -> Option<M>,
37    M: serde::Serialize,
38{
39    let mut answered = 0;
40    for line in input.lines() {
41        let line = match line {
42            Ok(l) => l,
43            Err(e) if e.kind() == std::io::ErrorKind::InvalidData => {
44                write_line(&mut output, &Server::parse_error("input is not UTF-8"))?;
45                continue;
46            }
47            Err(e) => return Err(Error::Transport(e)),
48        };
49        if line.trim().is_empty() {
50            continue;
51        }
52        let written = match serde_json::from_str::<Value>(&line) {
53            Ok(message) => match handle(message) {
54                Some(response) => write_line(&mut output, &response),
55                None => continue,
56            },
57            Err(e) => write_line(&mut output, &Server::parse_error(&e.to_string())),
58        };
59        match written {
60            Ok(()) => answered += 1,
61            Err(Error::Transport(e)) if e.kind() == std::io::ErrorKind::BrokenPipe => {
62                tracing::info!("client closed the output; stopping");
63                return Ok(answered);
64            }
65            Err(e) => return Err(e),
66        }
67    }
68    tracing::info!(answered, "input ended; stopping");
69    Ok(answered)
70}

72fn write_line<W: Write, M: serde::Serialize>(output: &mut W, message: &M) -> Result<()> {
73    let mut text = serde_json::to_string(message).map_err(|e| Error::Protocol {
74        reason: e.to_string(),
75    })?;
76    text.push('\n');
77    output
78        .write_all(text.as_bytes())
79        .map_err(Error::Transport)?;
80    output.flush().map_err(Error::Transport)
81}
