1//! One milestone in full, for the Roadmap's inspector (`/ui/roadmap?milestone=M004`): what it
2//! holds and where each issue stands, what can start now, the longest chain of open work
3//! through it, the intents its issues serve, and the milestones it waits for and unblocks.
4//!
5//! Read from `plan.roadmap` and `plan.issues`, both of which answer in milliseconds, so the
6//! inspector is part of the first paint. Every issue is a link to its own inspector on the PR
7//! Drain, and every milestone a link to its own here: a reader walks the plan by clicking it.
8//!
9//! What the milestone is *for* — its problem, the outcome it promises, where things stand and
10//! where they should, what is in and out of its scope, and what would prove it — is the
11//! record's own prose, asked of `objects.get` and drawn by [`super::prose`]: markup as markup,
12//! and every issue or milestone the prose names a link to it.
13

23/// A plan milestone id: letters, digits, `-` and `_`, as the plan's own ids are (`M004`,
24/// `development-runtime`), and nothing a link could carry anything else in.
25///
26/// ```text
27/// assert!(is_milestone_id("M004") && is_milestone_id("development-runtime"));
28/// assert!(!is_milestone_id("M0\"4") && !is_milestone_id(""));
29/// ```
30pub fn is_milestone_id(id: &str) -> bool {
31    !id.is_empty()
32        && id.len() <= 64
33        && id
34            .bytes()
35            .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_')
36}

38/// The milestone a query names, when it names one.
39pub fn picked(query: &[(String, String)]) -> Option<String> {
40    query
41        .iter()
42        .find(|(k, _)| k == "milestone")
43        .map(|(_, v)| v.trim().to_string())
44        .filter(|v| is_milestone_id(v))
45}

56fn strings(v: &Value, k: &str) -> Vec<String> {
57    v.get(k)
58        .and_then(Value::as_array)
59        .map(|a| {
60            a.iter()
61                .filter_map(Value::as_str)
62                .map(str::to_string)
63                .collect()
64        })
65        .unwrap_or_default()
66}

72/// The longest chain of open issues through a milestone, first to last: each issue depends on
73/// the one before it, as the plan records `depends_on`, and none of them is done. The order a
74/// reader would have to work through, at the least, before the milestone can close. Issues of
75/// other milestones are not part of it; a cycle, which the plan's validator refuses, ends a
76/// chain rather than looping.
77pub fn critical_path(issues: &[Value]) -> Vec<String> {
78    let open: BTreeMap<String, Vec<String>> = issues
79        .iter()
80        .filter(|i| !closed(&text(i, "status")))
81        .map(|i| (text(i, "id"), strings(i, "depends_on")))
82        .collect();
83    fn longest(
84        id: &str,
85        open: &BTreeMap<String, Vec<String>>,
86        memo: &mut BTreeMap<String, Vec<String>>,
87        on_stack: &mut BTreeSet<String>,
88    ) -> Vec<String> {
89        if let Some(found) = memo.get(id) {
90            return found.clone();
91        }
92        if !on_stack.insert(id.to_string()) {
93            return Vec::new();
94        }
95        let mut best: Vec<String> = Vec::new();
96        for dep in open.get(id).into_iter().flatten() {
97            if open.contains_key(dep) {
98                let chain = longest(dep, open, memo, on_stack);
99                if chain.len() > best.len() {
100                    best = chain;
101                }
102            }
103        }
104        on_stack.remove(id);
105        best.push(id.to_string());
106        memo.insert(id.to_string(), best.clone());
107        best
108    }
109    let mut memo = BTreeMap::new();
110    let mut best: Vec<String> = Vec::new();
111    for id in open.keys() {
112        let chain = longest(id, &open, &mut memo, &mut BTreeSet::new());
113        if chain.len() > best.len() {
114            best = chain;
115        }
116    }
117    best
118}

136/// The inspector of milestone `id`.
137/// The prose sections of a milestone's record, in the order a reader asks: why, what for,
138/// from where to where, within what bounds, and proven how. A section the record does not
139/// carry is not drawn.
140const NARRATIVE: &[(&str, &str)] = &[
141    ("problem", "The problem"),
142    ("outcome", "The outcome"),
143    ("current_state", "Where it stands"),
144    ("desired_state", "Where it should stand"),
145    ("scope", "In scope"),
146    ("non_scope", "Out of scope"),
147    ("acceptance_criteria", "Proven when"),
148];
149
150/// The record's narrative as headed prose: a string as a paragraph, a list as a list.
151fn narrative(record: &Value, known: &super::prose::Known) -> Vec<El> {
152    let mut out = Vec::new();
153    for (key, heading) in NARRATIVE {
154        let drawn = match record.get(*key) {
155            Some(Value::String(prose)) if !prose.trim().is_empty() => {
156                super::prose::paragraph(prose, known)
157            }
158            Some(Value::Array(_)) if !strings(record, key).is_empty() => {
159                super::prose::bullets(&strings(record, key), known)
160            }
161            _ => continue,
162        };
163        out.push(el("h4").class("mj-ui-panel-title").text(*heading));
164        out.push(drawn);
165    }
166    out
167}

169/// The milestone inspector with the record's own prose: `record` is `objects.get` for the
170/// milestone, when the caller asked it.
171pub fn inspector_with(roadmap: &Answer, issues: &Answer, record: &Answer, id: &str) -> El {
172    let known = super::prose::Known {
173        issues: list(issues, "/issues")
174            .iter()
175            .map(|i| text(i, "id"))
176            .collect(),
177        milestones: list(roadmap, "/milestones")
178            .iter()
179            .map(|m| text(m, "id"))
180            .collect(),
181    };
182    let told = record
183        .ok()
184        .and_then(|r| r.get("metadata"))
185        .map(|metadata| narrative(metadata, &known))
186        .unwrap_or_default();
187    let base = inspector(roadmap, issues, id);
188    match told.is_empty() {
189        true => base,
190        false => panel(
191            "What it is for",
192            "The milestone's own record: why it exists, what it promises, and what would prove it.",
193            told.into_iter()
194                .fold(el("div").class("mj-ui-stack"), El::child),
195            &[record],
196        )
197        .attr("id", "milestone-record")
198        .child(base),
199    }
200}

202pub fn inspector(roadmap: &Answer, issues: &Answer, id: &str) -> El {
203    let body = unanswered(roadmap)
204        .or_else(|| unanswered(issues))
205        .unwrap_or_else(|| {
206            let Some(m) = list(roadmap, "/milestones")
207                .into_iter()
208                .find(|m| text(m, "id") == id)
209            else {
210                return nothing(&format!("{id} is not a milestone of the plan."));
211            };
212            let held: BTreeSet<String> = strings(&m, "issues").into_iter().collect();
213            let mine: Vec<Value> = list(issues, "/issues")
214                .into_iter()
215                .filter(|i| held.contains(&text(i, "id")) || text(i, "milestone") == id)
216                .collect();
217            let ready: Vec<String> = mine
218                .iter()
219                .filter(|i| text(i, "status").eq_ignore_ascii_case("READY"))
220                .map(|i| text(i, "id"))
221                .collect();
222            let path = critical_path(&mine);
223            let serves: Vec<String> = mine
224                .iter()
225                .flat_map(|i| {
226                    strings(i, "serves")
227                        .into_iter()
228                        .map(|s| super::intent::intent_of(&s).to_string())
229                })
230                .collect::<BTreeSet<_>>()
231                .into_iter()
232                .collect();
233            let rows = mine
234                .iter()
235                .map(|i| {
236                    let iid = text(i, "id");
237                    el("tr")
238                        .child(
239                            el("td").child(
240                                el("a")
241                                    .class("mj-ui-row-link")
242                                    .attr("href", issue_href(&iid))
243                                    .child(
244                                        el("span").class("mj-ui-mono mj-ui-dim").text(iid.clone()),
245                                    )
246                                    .child(
247                                        el("span").class("mj-ui-cell-title").text(text(i, "title")),
248                                    ),
249                            ),
250                        )
251                        .child(el("td").child(state(i, "status")))
252                        .child(
253                            el("td")
254                                .class("mj-ui-mono")
255                                .text(super::view::value(i, "priority")),
256                        )
257                        .child(
258                            el("td").class("mj-ui-mono").text(
259                                i.get("wave")
260                                    .filter(|w| !w.is_null())
261                                    .map_or_else(|| "none".into(), Value::to_string),
262                            ),
263                        )
264                        .child(el("td").child(chips(&strings(i, "blocked_by"), issue_href)))
265                })
266                .collect::<Vec<_>>();
267            let fact = |label: &str, value: El| {
268                el("div")
269                    .class("mj-ui-fact")
270                    .child(el("dt").text(label))
271                    .child(el("dd").child(value))
272            };
273            let count = |s: &str| {
274                m.pointer(&format!("/counts/by_status/{s}"))
275                    .and_then(Value::as_u64)
276            };
277            el("section")
278                .class("mj-ui-detail")
279                .attr("id", "milestone")
280                .attr("aria-label", format!("Milestone {id}"))
281                .child(
282                    el("div")
283                        .class("mj-ui-detail-head")
284                        .child(el("p").class("mj-ui-mono mj-ui-dim").text(format!(
285                            "{id} · order {} · {}",
286                            m.get("order").map(Value::to_string).unwrap_or_else(|| "?".into()),
287                            text(&m, "priority")
288                        )))
289                        .child(el("h3").class("mj-ui-detail-title").text(text(&m, "title")))
290                        .child(
291                            el("p")
292                                .class("mj-ui-badges")
293                                .child(state(&m, "status"))
294                                .child(
295                                    el("a")
296                                        .class("mj-ui-more")
297                                        .attr(
298                                            "href",
299                                            format!("{}#milestone-{id}", super::roadmap::ROUTE),
300                                        )
301                                        .text("Close"),
302                                ),
303                        ),
304                )
305                .child(
306                    el("dl")
307                        .class("mj-ui-facts")
308                        .child(fact(
309                            "Done",
310                            meter(
311                                count("DONE"),
312                                m.pointer("/counts/total").and_then(Value::as_u64),
313                            ),
314                        ))
315                        .child(fact("Ready now", chips(&ready, issue_href)))
316                        .child(fact(
317                            "Blocked",
318                            el("span").class("mj-ui-mono").text(
319                                count("BLOCKED").map_or_else(|| "none".into(), |n| n.to_string()),
320                            ),
321                        ))
322                        .child(fact(
323                            "Waits for",
324                            chips(&strings(&m, "depends_on"), href_owned),
325                        ))
326                        .child(fact(
327                            "Blocked by",
328                            chips(&strings(&m, "blocked_by"), href_owned),
329                        ))
330                        .child(fact(
331                            "Unblocks",
332                            chips(&strings(&m, "dependents"), href_owned),
333                        ))
334                        .child(fact("Serves", chips(&serves, super::intent::href))),
335                )
336                .child(el("h4").class("mj-ui-panel-title").text("Critical path"))
337                .child(if path.is_empty() {
338                    nothing("No issue of it is open.")
339                } else {
340                    el("ol")
341                        .class("mj-ui-badges")
342                        .attr("aria-label", "The longest chain of open issues")
343                        .children(
344                            path.iter()
345                                .map(|i| el("li").child(chip(issue_href(i), i.clone()))),
346                        )
347                })
348                .child(el("h4").class("mj-ui-panel-title").text("Issues"))
349                .child(table(
350                    &["Issue", "Status", "Priority", "Wave", "Blocked by"],
351                    rows,
352                    "The milestone holds no issue.",
353                ))
354        });
355    panel(
356        "Milestone",
357        "One milestone of the plan: what it holds, what can start, and what it waits for.",
358        body,
359        &[roadmap, issues],
360    )
361}

413    /// The record's own prose is drawn above the plan's arithmetic: every section it carries,
414    /// markup as markup, and an issue or milestone it names as a link to it. A section it does
415    /// not carry is not drawn, and a record nobody could read leaves the inspector as it was.
416    #[test]
417    fn the_records_prose_is_drawn_with_its_markup_and_its_links() {
418        let (roadmap, issues) = plan();
419        let record = answer(
420            "objects.get",
421            serde_json::json!({ "metadata": {
422                "id": "M004",
423                "problem": "The shell tool is slow: `doctor` takes ten seconds, see I0002.",
424                "outcome": "Every command is **benchmarked**, after M003.",
425                "current_state": "   ",
426                "scope": ["lib", "bin/majordomus"],
427                "non_scope": [],
428                "acceptance_criteria": ["A regression is refused, as I0003 proves"]
429            }}),
430        );
431        let html = inspector_with(&roadmap, &issues, &record, "M004").render();
432        for expected in [
433            "The problem",
434            "<code class=\"mj-ui-code\">doctor</code>",
435            "?issue=I0002#issue\">I0002</a>",
436            "The outcome",
437            "<strong>benchmarked</strong>",
438            "?milestone=M003#milestone\">M003</a>",
439            "In scope",
440            "<li>bin/majordomus</li>",
441            "Proven when",
442            "?issue=I0003#issue\">I0003</a>",
443            "id=\"milestone-record\"",
444            // and the plan's own inspector is still there, below it
445            "Critical path",
446        ] {
447            assert!(html.contains(expected), "{expected} is missing from {html}");
448        }
449        for absent in ["Where it stands", "Out of scope", "Where it should stand"] {
450            assert!(
451                !html.contains(absent),
452                "{absent} was drawn from nothing: {html}"
453            );
454        }
455
456        let unread = Answer {
457            value: Err("no such object".into()),
458            ..answer("objects.get", Value::Null)
459        };
460        let plain = inspector_with(&roadmap, &issues, &unread, "M004").render();
461        assert_eq!(plain, inspector(&roadmap, &issues, "M004").render());
462        let bare = answer(
463            "objects.get",
464            serde_json::json!({ "metadata": { "id": "M004" } }),
465        );
466        assert_eq!(
467            inspector_with(&roadmap, &issues, &bare, "M004").render(),
468            inspector(&roadmap, &issues, "M004").render()
469        );
470    }
