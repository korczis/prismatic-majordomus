1//! Zed: what it reads from a project, and the projection of the environment onto it.
2//!
3//! Verified against Zed 1.23.2 — the installed application and the source at that tag:
4//! `crates/settings_content/src/project.rs` (the keys `.zed/settings.json` may carry),
5//! `crates/task/src/task_template.rs` (the shape of `.zed/tasks.json`),
6//! `crates/prompt_store/src/prompts.rs` (the instruction files, first match wins), and the
7//! documentation at <https://zed.dev/docs/tasks>, <https://zed.dev/docs/ai/instructions>,
8//! <https://zed.dev/docs/ai/skills> and <https://zed.dev/docs/ai/mcp>.

44/// `.zed/tasks.json`: one task per recipe, run as `just <recipe>` from the project root —
45/// the canonical command, never a copy of its body. `None` when there is nothing to run.
46pub fn tasks_artifact(env: &EditorEnvironment, version: &str) -> Option<Artifact> {
47    if env.tasks.is_empty() {
48        return None;
49    }
50    let tasks: Vec<Value> = env
51        .tasks
52        .iter()
53        .map(|t| {
54            let mut task = json!({
55                "label": format!("just {}", t.recipe),
56                "command": "just",
57                "args": [t.recipe],
58            });
59            if let Some(g) = &t.group {
60                task["tags"] = json!([g]);
61            }
62            task
63        })
64        .collect();
65    // `{:#}` is serde_json's pretty form of a value, and a value cannot fail to print
66    let body = format!("{:#}", Value::Array(tasks));
67    Some(Artifact::verbatim(
68        TASKS,
69        format!("{DOCUMENT_PREFIX}zed-tasks"),
70        ArtifactFormat::Json,
71        None,
72        JUSTFILE_SOURCE,
73        format!("{}{body}\n", jsonc_banner(JUSTFILE_SOURCE, version)),
74    ))
75}

77/// What Zed does with each capability here.
78pub fn adapter(env: &EditorEnvironment) -> EditorAdapter {
79    let skills = env
80        .skills
81        .iter()
82        .filter(|s| super::is_skill_name(&s.id))
83        .map(|s| format!("{SKILLS_DIR}/{}/SKILL.md", s.id))
84        .collect();
85    let tasks = if env.tasks.is_empty() {
86        Vec::new()
87    } else {
88        vec![TASKS.to_string()]
89    };
90    EditorAdapter {
91        id: ID.into(),
92        title: "Zed".into(),
93        verified_against: VERIFIED_AGAINST.into(),
94        capabilities: vec![
95            EditorCapability {
96                id: "instructions".into(),
97                support: EditorSupport::Supported,
98                artifacts: vec!["AGENTS.md".into()],
99                evidence: "zed.dev/docs/ai/instructions; prompts.rs RULES_FILE_NAMES".into(),
100                note: "Zed reads the provider bootstrap AGENTS.md, which `majordomus generate providers` already derives from the policy; nothing is added for Zed.".into(),
101            },
102            EditorCapability {
103                id: "mcp".into(),
104                support: EditorSupport::RequiresUserAction,
105                artifacts: Vec::new(),
106                evidence: "zed.dev/docs/ai/mcp; project.rs ContextServerSettingsContent".into(),
107                note: "A project `context_servers` entry needs a command Zed can start, and the repository's launcher is the relative path bin/majordomus-mcp, whose resolution Zed does not document. Add it in your own settings: {\"context_servers\": {\"majordomus\": {\"command\": \"<absolute path to this checkout>/bin/majordomus-mcp\", \"args\": []}}}.".into(),
108            },
109            EditorCapability {
110                id: "settings".into(),
111                support: EditorSupport::Unsupported,
112                artifacts: Vec::new(),
113                evidence: "project.rs ProjectSettingsContent".into(),
114                note: "No repository policy differs from Zed's defaults, so no .zed/settings.json is written; theme, fonts, keys and agent settings are personal and never projected.".into(),
115            },
116            EditorCapability {
117                id: "skills".into(),
118                support: EditorSupport::Supported,
119                artifacts: skills,
120                evidence: "zed.dev/docs/ai/skills".into(),
121                note: "Read from .agents/skills/<name>/SKILL.md in trusted worktrees; each file points at the canonical definition.".into(),
122            },
123            EditorCapability {
124                id: "tasks".into(),
125                support: if env.workflows_resolved == Some(false) {
126                    EditorSupport::Unknown
127                } else {
128                    EditorSupport::Supported
129                },
130                artifacts: tasks,
131                evidence: "zed.dev/docs/tasks; task_template.rs TaskTemplate".into(),
132                note: "Every public recipe of the tracked justfile without [confirm], run as `just <recipe>` from the project root.".into(),
133            },
134        ],
135    }
136}

138/// Which instruction file Zed reads in this checkout. A file earlier in its list than
139/// AGENTS.md hides the bootstrap from Zed entirely.
140pub fn instruction_findings(root: &Path) -> Vec<EditorDiagnostic> {
141    let first = INSTRUCTION_FILES
142        .iter()
143        .find(|f| root.join(f).is_file())
144        .copied();
145    match first {
146        Some("AGENTS.md") | None => Vec::new(),
147        Some(other) => vec![EditorDiagnostic {
148            severity: Severity::Warning,
149            code: "zed-reads-another-instruction-file".into(),
150            adapter: Some(ID.into()),
151            path: Some(other.into()),
152            message: format!(
153                "Zed reads the first of its instruction files that exists, and here that is {other}, not AGENTS.md"
154            ),
155            remedy: format!("fold {other} into the policy's projections or remove it, so that Zed reads AGENTS.md"),
156        }],
157    }
158}

160#[cfg(test)]
161mod tests {
162    use super::*;
163    use crate::editors::TaskRef;
164
165    #[test]
166    fn a_task_runs_the_recipe_and_carries_no_shell() {
167        let env = EditorEnvironment {
168            tasks: vec![TaskRef {
169                recipe: "test".into(),
170                group: Some("test".into()),
171                group_rank: 0,
172            }],
173            ..Default::default()
174        };
175        let a = tasks_artifact(&env, "0.0.0").unwrap();
176        let body = crate::editors::jsonc_body(&a.content).unwrap();
177        let v: Value = serde_json::from_str(body).unwrap();
178        assert_eq!(
179            v,
180            json!([{"label": "just test", "command": "just", "args": ["test"], "tags": ["test"]}])
181        );
182    }
183
184    #[test]
185    fn no_recipe_no_file() {
186        assert!(tasks_artifact(&EditorEnvironment::default(), "0.0.0").is_none());
187    }
188
189    #[test]
190    fn a_rules_file_earlier_in_zeds_list_is_reported() {
191        let dir = tempfile::tempdir().unwrap();
192        std::fs::write(dir.path().join("AGENTS.md"), "x").unwrap();
193        assert!(instruction_findings(dir.path()).is_empty());
194        std::fs::write(dir.path().join(".cursorrules"), "x").unwrap();
195        let found = instruction_findings(dir.path());
196        assert_eq!(found[0].path.as_deref(), Some(".cursorrules"));
197    }
198
199    #[test]
200    fn the_adapter_says_unknown_when_the_justfile_could_not_be_read() {
201        let unread = EditorEnvironment {
202            workflows_resolved: Some(false),
203            ..Default::default()
204        };
205        let tasks = |env: &EditorEnvironment| {
206            adapter(env)
207                .capabilities
208                .into_iter()
209                .find(|c| c.id == "tasks")
210                .unwrap()
211        };
212        assert_eq!(tasks(&unread).support, EditorSupport::Unknown);
213        assert!(tasks(&unread).artifacts.is_empty());
214
215        let read = EditorEnvironment {
216            workflows_resolved: Some(true),
217            tasks: vec![TaskRef {
218                recipe: "loose".into(),
219                group: None,
220                group_rank: usize::MAX,
221            }],
222            skills: vec![crate::editors::SkillRef {
223                id: "ok".into(),
224                title: "Ok".into(),
225                description: "d".into(),
226                path: ".ai/repo/skills/ok/SKILL.md".into(),
227                sha256: String::new(),
228            }],
229            ..Default::default()
230        };
231        assert_eq!(tasks(&read).support, EditorSupport::Supported);
232        assert_eq!(tasks(&read).artifacts, [TASKS]);
233        // a recipe in no group carries no tag
234        let a = tasks_artifact(&read, "0.0.0").unwrap();
235        assert!(!a.content.contains("\"tags\""), "{}", a.content);
236    }
