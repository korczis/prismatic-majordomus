1//! The `design` module: the design system as a capability.
2//!
3//! Four questions, answered from the declaration compiled into this executable — the
4//! same one every stylesheet was projected from: what the design is (its fingerprint, its
5//! identity, its theme contract, where it is projected), what tokens it holds, what one
6//! token means, and whether the colours it pairs can be read on one another. The MCP tool,
7//! the resource, the HTTP route, the OpenAPI operation and the Cockpit's Design page are
8//! projections of these; none of them holds a token of its own.
9//!
10//! Three of the four read nothing at all: the design is the tool's, not the supervised
11//! repository's, and it ships inside the binary. Whether the *served* stylesheets carry the
12//! same fingerprint is a question for the page (`--mj-design` against
13//! `design.system.design`), which the Cockpit asks on every load.
14//!
15//! `design.kit` and `design.kit.adopt` are about the supervised repository rather than
16//! this tool's own design (I4334, ADR 0230): the first judges the repository's user
17//! interface against a kit profile the distribution ships (`share/design/kits/`), signal by
18//! signal, and the second declares that profile in the repository — its rule, the
19//! enforcement entries that run the check in CI and before every push, and `ui.kit` in the
20//! policy's `ui:` block, which the design licence declares and both read. The judgement is
21//! [`crate::design::kit`]'s; these declare it once.
22//!
23//! `design.contrast` is the exception, and reads only the distribution's own share
24//! directory: the pair a ratio is computed for is stated by the primitives that consume the
25//! declaration, and those stylesheets ship beside `kinds.yaml` and the obligation
26//! vocabulary. It is located the way every other reader locates it
27//! ([`crate::share::Share`]), never discovered in the supervised repository.
28//!
29//! `design.licence` is the one that reads the supervised repository, and the one file
30//! beyond it the caller names: whether a public tree holds a licensed kit's bytes, judged
31//! from kit-evidence/v1 fingerprints ([`crate::design::licence`]). It has no HTTP route: it
32//! opens a file by the path its input gives, and that is a local caller's act.
33//!
34//! `design.kit` has none for the same reason. Its input `evidence` names a file the process
35//! opens, and a query parameter that names a file on the server would hand any HTTP caller,
36//! a remote one included, an oracle over the server's files. The registry binds one input
37//! type to every transport and cannot withhold one field from one of them, so the judgement
38//! is the command line's and the MCP tool's: the command line for CI and the pre-push hook,
39//! MCP for an agent beside them.

243fn system(_: &Context, _: Empty) -> Result<DesignReport, CapabilityError> {
244    let design = compiled()?;
245    let states = design.status.states.iter().map(|(_, w)| w.len()).sum();
246    let read_by = |path: &str| -> &'static str {
247        match path {
248            render::THEME_CSS | render::SURFACE_CSS | render::STATUS_CSS => {
249                "both Tailwind builds: the site's and the Cockpit's"
250            }
251            render::TOKENS_CSS => "the pages the executable renders itself, and the Swagger shell",
252            render::COMPILED_YAML => "this executable: the design capabilities and the Cockpit",
253            render::COMPILED_MARK => "the Cockpit's shell",
254            render::SITE_JSON => "the site's templates and the browser probes",
255            _ => "a reader",
256        }
257    };
258    Ok(DesignReport {
259        fingerprint: design.fingerprint(),
260        design: design.short_fingerprint(),
261        schema: design.schema,
262        source: SOURCE.into(),
263        identity: design.identity.clone(),
264        font: design.font.clone(),
265        theme: ThemeView {
266            class: design.theme.class.clone(),
267            storage_key: design.theme.storage_key.clone(),
268            bootstrap: design.theme_bootstrap(),
269        },
270        viewports: design.viewports.clone(),
271        tallies: TokenTallies {
272            palette: design.palette.len(),
273            roles: design.roles.len(),
274            statuses: design.status.roles.len(),
275            states,
276            type_steps: design.type_.scale.len(),
277            aliases: design.alias.flowbite.len(),
278        },
279        projections: [
280            render::THEME_CSS,
281            render::SURFACE_CSS,
282            render::STATUS_CSS,
283            render::TOKENS_CSS,
284            render::COMPILED_YAML,
285            render::COMPILED_MARK,
286            render::SITE_JSON,
287            "docs/generated/design.json",
288            "docs/generated/design.yaml",
289            "docs/generated/design.md",
290        ]
291        .iter()
292        .map(|p| ProjectionView {
293            path: p.to_string(),
294            read_by: read_by(p).to_string(),
295        })
296        .collect(),
297    })
298}

300fn tokens(_: &Context, input: TokensInput) -> Result<TokenList, CapabilityError> {
301    let design = compiled()?;
302    let tokens: Vec<Token> = design
303        .tokens()
304        .into_iter()
305        .filter(|t| input.kind.is_none_or(|k| t.kind == k))
306        .collect();
307    Ok(TokenList {
308        fingerprint: design.fingerprint(),
309        total: tokens.len(),
310        tokens,
311    })
312}

314fn explain(_: &Context, input: ExplainTokenInput) -> Result<Token, CapabilityError> {
315    let design = compiled()?;
316    if input.token.trim().is_empty() {
317        return Err(CapabilityError::Refused(
318            "name a token: a role, a status, a state word, a type step, or its custom property"
319                .into(),
320        ));
321    }
322    design.explain(&input.token).ok_or_else(|| {
323        CapabilityError::NotFound(format!(
324            "no token named '{}'; `design.tokens` lists every one",
325            input.token
326        ))
327    })
328}

330/// The stylesheets that state which foreground is rendered on which ground, read from the
331/// distribution's share directory. A distribution that ships none states no pair, which is
332/// reported as a source that was not found rather than as a clean measurement.
333fn consumers(root: &std::path::Path) -> Result<Vec<(String, String)>, CapabilityError> {
334    let share = crate::share::Share::locate(None, root).map_err(|e| {
335        CapabilityError::Internal(format!(
336            "the distribution's share directory was not found: {e}"
337        ))
338    })?;
339    let mut out = Vec::new();
340    for name in contrast::CONSUMERS {
341        let path = share.dir().join(name);
342        match std::fs::read_to_string(&path) {
343            Ok(text) => out.push((format!("share/{name}"), text)),
344            Err(e) => {
345                return Err(CapabilityError::Internal(format!(
346                    "{}: {e}; the design's own primitives are what state which colour is rendered on which ground",
347                    path.display()
348                )))
349            }
350        }
351    }
352    Ok(out)
353}

355fn kit_share(ctx: &Context, root: &std::path::Path) -> Result<std::path::PathBuf, CapabilityError> {
356    crate::share::Share::locate(ctx.index.share.as_deref(), root)
357        .map(|s| s.dir().to_path_buf())
358        .map_err(|e| {
359            CapabilityError::Internal(format!(
360                "the distribution's share directory, which holds the kit profiles, was not found: {e}"
361            ))
362        })
363}

365fn kit_check(ctx: &Context, input: KitInput) -> Result<KitReport, CapabilityError> {
366    let root = std::path::PathBuf::from(&ctx.index.repository.root);
367    let share = kit_share(ctx, &root)?;
368    let evidence = input.evidence.as_deref().map(std::path::Path::new);
369    kit::inspect(&root, &share, evidence).map_err(CapabilityError::Internal)
370}
371
372fn kit_adopt(ctx: &Context, input: KitAdoptInput) -> Result<AdoptReport, CapabilityError> {
373    if input.profile.trim().is_empty() {
374        return Err(CapabilityError::InvalidInput(
375            "name a kit profile of share/design/kits/, such as flowbite-pro".into(),
376        ));
377    }
378    let root = std::path::PathBuf::from(&ctx.index.repository.root);
379    let share = kit_share(ctx, &root)?;
380    kit::adopt(&root, &share, input.profile.trim(), input.dry_run).map_err(adopt_error)
381}

383/// An adoption's error in the capability's categories, one for one.
384fn adopt_error(e: AdoptError) -> CapabilityError {
385    match e {
386        AdoptError::NotFound(m) => CapabilityError::NotFound(m),
387        AdoptError::Refused(m) => CapabilityError::Refused(m),
388        AdoptError::Internal(m) => CapabilityError::Internal(m),
389    }
390}

392fn readable(ctx: &Context, _: Empty) -> Result<ContrastReport, CapabilityError> {
393    let design = compiled()?;
394    let root = std::path::PathBuf::from(&ctx.index.repository.root);
395    Ok(contrast::measure(design, &consumers(&root)?))
396}

414
415fn licensed(ctx: &Context, input: LicenceInput) -> Result<LicenceReport, CapabilityError> {
416    let repository = ctx.repository().map_err(|e| {
417        CapabilityError::Internal(format!("the repository could not be re-read: {e}"))
418    })?;
419    let home = std::env::var_os("HOME").map(std::path::PathBuf::from);
420    let tightened = std::env::var(licence::VISIBILITY_ENV).ok();
421    licence::assess(
422        &repository,
423        input.evidence.as_deref(),
424        home.as_deref(),
425        tightened.as_deref(),
426    )
427    .map_err(licence_error)
428}

430/// A licence assessment's error in the capability's categories: evidence that cannot be
431/// read is the caller's input, a policy that cannot be used is a refusal, and a tree that
432/// cannot be listed is the handler's own failure.
433fn licence_error(e: LicenceError) -> CapabilityError {
434    match e {
435        LicenceError::Evidence(m) => CapabilityError::InvalidInput(m),
436        LicenceError::Policy(m) => CapabilityError::Refused(m),
437        LicenceError::Tree(m) => CapabilityError::Internal(m),
438    }
439}

805    /// A context over `repo` whose index names `share` as the distribution's share.
806    fn context_over(
807        repo: &crate::synthetic::SyntheticRepository,
808        share: std::path::PathBuf,
809    ) -> std::sync::Arc<Context> {
810        let mut index = repo.index().expect("an index");
811        index.share = Some(share);
812        let registry = crate::capability::CapabilityRegistry::builder()
813            .with_modules(crate::capability::builtin::modules())
814            .with_index(&index)
815            .build()
816            .expect("a registry");
817        std::sync::Arc::new(Context::new(
818            std::sync::Arc::new(index),
819            std::sync::Arc::new(registry),
820        ))
821    }
