1//! direnv's approval of a worktree's `.envrc`, carried to the path the worktree now has.
2//!
3//! direnv approves an `.envrc` by path and content: the same file at a new path is a file it
4//! has not seen, and it refuses to load it until a person runs `direnv allow` there. A
5//! worktree this module creates or moves therefore starts blocked, and the first `cd` into it
6//! is answered with `direnv: error .envrc is blocked` — in a repository with sixty linked
7//! worktrees, the recurring "direnv does not work again". Measured on 2026-09-09: of the 37
8//! worktrees that had an `.envrc`, 36 were blocked and the primary checkout was the one
9//! approved.
10//!
11//! The approval is carried, not granted. The person approved the `.envrc` of the primary
12//! checkout; a worktree whose `.envrc` is byte-for-byte that file gets the same approval at
13//! its own path, at the moment the path comes into being. A worktree whose `.envrc` differs
14//! — an older branch, a branch that is not theirs — is left as direnv leaves it, and the
15//! report says so, because approving content the person has not seen is the one thing
16//! `direnv allow` exists to prevent. When the primary checkout's own `.envrc` is not
17//! approved, nothing is approved anywhere: the person does not load this file, and a
18//! worktree is not the place to start.
19//!
20//! Every outcome is reported and none is fatal. A worktree that exists and is blocked is
21//! still a worktree; what this adds is that a person learns it from the report rather than
22//! from the next `cd`.
23

25use std::process::Command;

30/// What became of the `.envrc` of a worktree that was just created, found, or moved.
31#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
32#[serde(rename_all = "snake_case", tag = "outcome")]
33pub enum EnvrcApproval {
34    /// direnv accepted it: the next `cd` loads the environment.
35    Approved,
36    /// The worktree has no `.envrc`; there is nothing to approve and nothing is blocked.
37    NoEnvrc,
38    /// direnv is not on the PATH; nothing was done, and nothing is blocked either, because
39    /// nothing would load the file.
40    DirenvAbsent,
41    /// The worktree's `.envrc` is not the primary checkout's, so the approval given there
42    /// does not carry. `direnv allow` in the worktree, after reading it, is the person's.
43    Differs,
44    /// The primary checkout's `.envrc` is not approved, so there is no approval to carry.
45    NotApprovedInPrimary,
46    /// direnv was asked and refused, or could not be run; the message is its own.
47    Failed {
48        /// direnv's standard error, or the error running it.
49        message: String,
50    },
51}
52
53impl EnvrcApproval {
54    /// One line for a person, after the path the worktree is at.
55    pub fn describe(&self) -> String {
56        match self {
57            Self::Approved => {
58                "envrc   approved for direnv (the primary checkout's, at this path)".into()
59            }
60            Self::NoEnvrc => "envrc   none; nothing for direnv to load".into(),
61            Self::DirenvAbsent => "envrc   direnv is not installed; nothing to approve".into(),
62            Self::Differs => {
63                "envrc   differs from the primary checkout's and stays blocked; read it, then \
64                 `direnv allow` there"
65                    .into()
66            }
67            Self::NotApprovedInPrimary => {
68                "envrc   the primary checkout's is not approved, so nothing was approved here; \
69                 `direnv allow` in the primary checkout first"
70                    .into()
71            }
72            Self::Failed { message } => format!("envrc   direnv refused: {message}"),
73        }
74    }
75}

77/// Carry the primary checkout's approval to the `.envrc` of `worktree`, with the direnv on
78/// the PATH. Never fails: the worst outcome is a worktree left as direnv left it, reported.
79pub fn approve(primary: &Path, worktree: &Path) -> EnvrcApproval {
80    match direnv_on_path() {
81        Some(direnv) => approve_with(&direnv, primary, worktree),
82        None => EnvrcApproval::DirenvAbsent,
83    }
84}
85
86/// The same, with a given direnv executable. What the tests drive, with a direnv of their
87/// own making, so the decision is proved without depending on the machine's.
88pub fn approve_with(direnv: &Path, primary: &Path, worktree: &Path) -> EnvrcApproval {
89    let envrc = worktree.join(".envrc");
90    let Ok(here) = std::fs::read(&envrc) else {
91        return EnvrcApproval::NoEnvrc;
92    };
93    let Ok(there) = std::fs::read(primary.join(".envrc")) else {
94        return EnvrcApproval::Differs;
95    };
96    if here != there {
97        return EnvrcApproval::Differs;
98    }
99    match allowed_in(direnv, primary) {
100        Some(true) => {}
101        Some(false) => return EnvrcApproval::NotApprovedInPrimary,
102        None => {
103            return EnvrcApproval::Failed {
104                message: "`direnv status` in the primary checkout did not say whether its \
105                          .envrc is allowed"
106                    .into(),
107            }
108        }
109    }
110    // The absolute path, and run from the worktree: direnv resolves a bare `allow` against
111    // its working directory, and the absolute argument says which file is meant even so.
112    let output = Command::new(direnv)
113        .arg("allow")
114        .arg(&envrc)
115        .current_dir(worktree)
116        .output();
117    match output {
118        Ok(o) if o.status.success() => EnvrcApproval::Approved,
119        Ok(o) => {
120            let stderr = String::from_utf8_lossy(&o.stderr).trim().to_string();
121            EnvrcApproval::Failed {
122                message: if stderr.is_empty() {
123                    format!("exit {}", o.status)
124                } else {
125                    stderr
126                },
127            }
128        }
129        Err(e) => EnvrcApproval::Failed {
130            message: e.to_string(),
131        },
132    }
133}

135/// Whether direnv has the `.envrc` of `dir` approved, read from `direnv status` run there.
136/// `None` when the answer is not in the output: an unknown is never taken for a yes.
137fn allowed_in(direnv: &Path, dir: &Path) -> Option<bool> {
138    let output = Command::new(direnv)
139        .arg("status")
140        .current_dir(dir)
141        .output()
142        .ok()?;
143    let text = String::from_utf8_lossy(&output.stdout);
144    // `Found RC allowed 0` is allowed; 1 is not allowed; 2 is denied. The line is direnv's,
145    // and has been since the allow/deny split; anything else is an unknown.
146    text.lines()
147        .find_map(|l| l.trim().strip_prefix("Found RC allowed "))
148        .and_then(|v| v.trim().parse::<u8>().ok())
149        .map(|v| v == 0)
150}
151
152fn direnv_on_path() -> Option<PathBuf> {
153    let path = std::env::var_os("PATH")?;
154    std::env::split_paths(&path)
155        .map(|d| d.join("direnv"))
156        .find(|p| p.is_file())
157}

164    /// A direnv of the test's own: answers `status` with the given allowed value and records
165    /// every `allow` it is asked for in a file, one path per line.
166    fn fake_direnv(dir: &Path, allowed: u8, allow_exit: i32) -> (PathBuf, PathBuf) {
167        let record = dir.join("allowed.log");
168        let script = dir.join("direnv");
169        std::fs::write(
170            &script,
171            format!(
172                "#!/bin/sh\ncase \"$1\" in\n  status) printf 'Loaded RC allowed 0\\nFound RC allowed {allowed}\\n' ;;\n  allow) printf '%s\\n' \"$2\" >> '{}'; exit {allow_exit} ;;\n  *) exit 2 ;;\nesac\n",
173                record.display()
174            ),
175        )
176        .unwrap();
177        std::fs::set_permissions(&script, std::fs::Permissions::from_mode(0o755)).unwrap();
178        // Linux refuses to exec a file that any process holds open for writing (ETXTBSY). A
179        // test thread that forks a child while the write above has the file open leaves that
180        // child holding the descriptor until it execs, so the first exec of this fake could
181        // fail, and `approve_with` read the failure as "direnv did not say" (master's rust job
182        // in run 36557029377). One exec that succeeds proves no writer is left: the write has
183        // closed, and no later fork can inherit it.
184        wait_until_executable(&script);
185        (script, record)
186    }
187
188    fn wait_until_executable(script: &Path) {
189        for _ in 0..100 {
190            match Command::new(script).arg("probe").output() {
191                Err(e) if e.raw_os_error() == Some(libc::ETXTBSY) => {
192                    std::thread::sleep(std::time::Duration::from_millis(20))
193                }
194                _ => return,
195            }
196        }
197        panic!("{} stayed busy for writing", script.display());
198    }

200    fn checkout(root: &Path, name: &str, envrc: Option<&str>) -> PathBuf {
201        let dir = root.join(name);
202        std::fs::create_dir_all(&dir).unwrap();
203        if let Some(text) = envrc {
204            std::fs::write(dir.join(".envrc"), text).unwrap();
205        }
206        dir
207    }

210    fn the_primary_checkouts_approval_is_carried_to_an_identical_envrc() {
211        let tmp = tempfile::tempdir().unwrap();

236    /// Linux alone refuses to exec a file held open for writing, so this holds there: the
237    /// fake is run only once no writer holds it, rather than failing on the first try.
238    #[cfg(target_os = "linux")]
239    #[test]
240    fn a_fake_still_open_for_writing_is_waited_for_before_it_is_run() {
241        let tmp = tempfile::tempdir().unwrap();
242        let (direnv, _) = fake_direnv(tmp.path(), 0, 0);
243        let held = std::fs::OpenOptions::new()
244            .write(true)
245            .open(&direnv)
246            .unwrap();
247        let busy = Command::new(&direnv).arg("probe").output().unwrap_err();
248        assert_eq!(busy.raw_os_error(), Some(libc::ETXTBSY), "{busy}");
249        let release = std::thread::spawn(move || {
250            std::thread::sleep(std::time::Duration::from_millis(200));
251            drop(held);
252        });
253        wait_until_executable(&direnv);
254        let after = Command::new(&direnv).arg("probe").output();
255        release.join().unwrap();
256        assert!(
257            after.is_ok(),
258            "the fake was handed on while still busy: {after:?}"
259        );
260    }
