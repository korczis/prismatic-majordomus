1//! `majordomus intent`: the intents, through the registry's own `intent.*` capabilities.
2//!
3//! This file renders; it decides nothing. Every answer is the output of the capability the
4//! CLI exposure names, executed through the one executor, so the command line, the HTTP
5//! routes and the MCP tools cannot answer the same question differently. The two verdicts
6//! — `validate` and `preflight` — are read out of that output and turned into exit 10.
7
8use std::io::Write;

10use serde_json::{json, Value};

13use crate::capability::{CapabilityError, Context};
14use crate::cli::{IntentArgs, IntentCommand, OutputFormat};

17/// The exit code when `validate` finds a failure or `preflight` refuses.
18pub const EXIT_INVALID: u8 = 10;
19
20/// Run `majordomus intent`.
21pub fn run(args: IntentArgs) -> Result<u8> {
22    let app = App::load(&args.repo)?;
23    let format = args.format;
24    match &args.command {
25        IntentCommand::List => {
26            let v = call(&app.context, &["intent", "list"], json!({}))?;
27            emit(format, &v, list_text)?;
28            Ok(0)
29        }
30        IntentCommand::Show { id } => {
31            let v = call(&app.context, &["intent", "show"], json!({ "id": id }))?;
32            emit(format, &v, show_text)?;
33            Ok(0)
34        }
35        IntentCommand::Validate => {
36            let v = call(&app.context, &["intent", "validate"], json!({}))?;
37            emit(format, &v, validate_text)?;
38            Ok(if v["valid"].as_bool() == Some(true) {
39                0
40            } else {
41                EXIT_INVALID
42            })
43        }
44        // the coverage is a projection of records the load already read: it has no refusal
45        // of its own, so the only failure is the write, which is returned as it is
46        IntentCommand::Coverage => call(&app.context, &["intent", "coverage"], json!({}))
47            .and_then(|v| emit(format, &v, coverage_text))
48            .map(|()| 0),
49        IntentCommand::Preflight { issue, paths } => {
50            let mut input = json!({ "paths": paths.join(",") });
51            if let Some(issue) = issue {
52                input["issue"] = json!(issue);
53            }
54            let v = call(&app.context, &["intent", "preflight"], input)?;
55            emit(format, &v, preflight_text)?;
56            Ok(if v["verdict"] == "refused" {
57                EXIT_INVALID
58            } else {
59                0
60            })
61        }
62        IntentCommand::Realization { intent } => {
63            let mut input = json!({});
64            if let Some(intent) = intent {
65                input["intent"] = json!(intent);
66            }
67            let v = call(&app.context, &["intent", "realization"], input)?;
68            emit(format, &v, realization_text)?;
69            let regressed = v["findings"]
70                .as_array()
71                .into_iter()
72                .flatten()
73                .any(|f| f["code"] == "closed_work_contradicted");
74            Ok(if regressed { EXIT_INVALID } else { 0 })
75        }
76        IntentCommand::Explain { id } => {
77            let v = call(&app.context, &["intent", "explain"], json!({ "id": id }))?;
78            emit(format, &v, explain_text)?;
79            Ok(0)
80        }
81    }
82}

84/// Each intent with how far reality is from it and who realises it, then every unit of work
85/// with its strongest link or the reason it has none.
86fn realization_text(v: &Value) -> String {
87    let mut out = Vec::new();
88    for i in v["intents"].as_array().into_iter().flatten() {
89        out.push(format!(
90            "{}  {}  {}/{} met  {}",
91            s(i, "intent"),
92            s(i, "stage"),
93            i["met"],
94            i["criteria"],
95            s(i, "title")
96        ));
97        for c in i["unmet"].as_array().into_iter().flatten() {
98            let issues: Vec<&str> = c["issues"]
99                .as_array()
100                .into_iter()
101                .flatten()
102                .filter_map(Value::as_str)
103                .collect();
104            out.push(format!(
105                "  unmet     {}  {}{}",
106                s(c, "id"),
107                s(c, "state"),
108                if issues.is_empty() {
109                    String::new()
110                } else {
111                    format!("  served by {}", issues.join(" "))
112                }
113            ));
114        }
115        for w in i["work"].as_array().into_iter().flatten() {
116            let providers: Vec<&str> = w["providers"]
117                .as_array()
118                .into_iter()
119                .flatten()
120                .filter_map(Value::as_str)
121                .collect();
122            out.push(format!(
123                "  work      {} {}  {}  {}  handovers {}{}",
124                s(w, "kind"),
125                s(w, "id"),
126                s(w, "outcome"),
127                s(w, "provenance"),
128                w["handovers"],
129                if providers.is_empty() {
130                    String::new()
131                } else {
132                    format!("  by {}", providers.join(", "))
133                }
134            ));
135        }
136    }
137    let work = v["work"].as_array().cloned().unwrap_or_default();
138    if !work.is_empty() {
139        out.push(String::new());
140    }
141    for w in &work {
142        let unit = &w["work"];
143        let link = w["links"].as_array().and_then(|l| l.first()).map(|l| {
144            format!(
145                "{} via {} {} ({})",
146                s(l, "intent"),
147                s(l, "issue"),
148                s(l, "via"),
149                s(l, "provenance")
150            )
151        });
152        out.push(format!(
153            "{} {}  {}  {}",
154            s(unit, "kind"),
155            s(unit, "id"),
156            s(unit, "outcome"),
157            link.unwrap_or_else(|| format!("unlinked: {}", s(w, "unlinked")))
158        ));
159    }
160    findings_text(&mut out, &v["findings"]);
161    out.push(format!(
162        "{} intent(s), {} unit(s) of work, {} serving no intent",
163        v["intents"].as_array().map_or(0, Vec::len),
164        work.len(),
165        v["orphans"]
166    ));
167    out.join("\n")
168}

170fn explain_text(v: &Value) -> String {
171    let i = &v["intent"];
172    let mut out = vec![
173        format!("{}  {}", s(i, "id"), s(i, "title")),
174        String::new(),
175        format!("  {}", s(i, "statement")),
176        String::new(),
177    ];
178    for b in v["because"].as_array().into_iter().flatten() {
179        out.push(format!("  - {}", b.as_str().unwrap_or("")));
180    }
181    out.join("\n")
182}

184fn call(ctx: &Context, path: &[&str], input: Value) -> Result<Value> {
185    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
186    let id = ctx
187        .registry
188        .by_cli(&words)
189        .map(|c| c.id.to_string())
190        .ok_or_else(|| Error::Protocol {
191            reason: format!(
192                "no capability is exposed as `majordomus {}`",
193                path.join(" ")
194            ),
195        })?;
196    ctx.execute(&id, input).map_err(|e| match e {
197        CapabilityError::NotFound(reason) => Error::NotFound { reason },
198        other => Error::Protocol {
199            reason: other.to_string(),
200        },
201    })
202}

204fn emit(format: OutputFormat, v: &Value, text: impl Fn(&Value) -> String) -> Result<()> {
205    let body = match format {
206        // `{:#}` is serde_json's own pretty writer, the one `to_string_pretty` runs, so the
207        // bytes are the same; serialising a `Value` cannot fail, so there is no fallback
208        OutputFormat::Json => format!("{v:#}"),
209        OutputFormat::Text => text(v),
210    };
211    writeln!(std::io::stdout().lock(), "{body}").map_err(Error::Transport)
212}

218fn list_text(v: &Value) -> String {
219    let intents = v["intents"].as_array().cloned().unwrap_or_default();
220    let width = intents
221        .iter()
222        .map(|i| s(i, "id").len())
223        .max()
224        .unwrap_or(2)
225        .max(2);
226    let mut out = vec![format!(
227        "{:<width$}  {:<10}  {:<11}  {:<7}  TITLE",
228        "ID", "STAGE", "VERDICT", "MET"
229    )];
230    for i in &intents {
231        let total = i["satisfaction"].as_array().map_or(0, Vec::len);
232        out.push(format!(
233            "{:<width$}  {:<10}  {:<11}  {:<7}  {}",
234            s(i, "id"),
235            s(i, "stage"),
236            s(&i["verdict"], "state"),
237            format!("{}/{total}", i["met"]),
238            s(i, "title"),
239        ));
240    }
241    out.push(String::new());
242    out.push(format!("{} intent(s)", v["count"]));
243    out.join("\n")
244}

246/// Every criterion with the work that carries it, then every issue with the reason it exists.
247fn coverage_text(v: &Value) -> String {
248    let criteria = v["criteria"].as_array().cloned().unwrap_or_default();
249    let mut out = vec![format!("{:<28}  {:<9}  ISSUES", "CRITERION", "STRENGTH")];
250    for c in &criteria {
251        let issues: Vec<&str> = c["issues"]
252            .as_array()
253            .into_iter()
254            .flatten()
255            .map(|i| s(i, "id"))
256            .collect();
257        out.push(format!(
258            "{:<28}  {:<9}  {}",
259            format!("{}#{}", s(c, "intent"), s(c, "criterion")),
260            s(c, "strength"),
261            if issues.is_empty() {
262                "—".into()
263            } else {
264                issues.join(" ")
265            },
266        ));
267    }
268    let issues = v["issues"].as_array().cloned().unwrap_or_default();
269    let mut origins: std::collections::BTreeMap<&str, usize> = std::collections::BTreeMap::new();
270    for i in &issues {
271        *origins.entry(s(i, "origin")).or_default() += 1;
272    }
273    out.push(String::new());
274    out.push(format!(
275        "{} criterion(s); {} issue(s): {}",
276        criteria.len(),
277        issues.len(),
278        origins
279            .iter()
280            .map(|(o, n)| format!("{n} {o}"))
281            .collect::<Vec<_>>()
282            .join(", "),
283    ));
284    out.join("\n")
285}

287fn show_text(v: &Value) -> String {
288    let mut out = vec![
289        format!("{}  {}", s(v, "id"), s(v, "title")),
290        String::new(),
291        format!("  {}", s(v, "statement")),
292        String::new(),
293        format!(
294            "  stage {}   verdict {}   source {}",
295            s(v, "stage"),
296            s(&v["verdict"], "state"),
297            s(v, "source")
298        ),
299    ];
300    for inv in v["invariants"].as_array().into_iter().flatten() {
301        out.push(format!("  invariant   {}", inv.as_str().unwrap_or("")));
302    }
303    for m in v["milestones"].as_array().into_iter().flatten() {
304        out.push(format!(
305            "  milestone   {}  {}",
306            s(m, "id"),
307            m["status"].as_str().unwrap_or("unresolved")
308        ));
309    }
310    for c in v["satisfaction"].as_array().into_iter().flatten() {
311        out.push(format!(
312            "  criterion   {}  {}{}  {} {}{}",
313            s(c, "id"),
314            s(c, "state"),
315            // a met criterion names the verdict it rests on: `proven` and `inputs_unchanged`
316            // are two answers, never one tick
317            match (c["met"].as_bool(), c["proof"].as_str()) {
318                (Some(true), Some(p)) => format!(" ({p})"),
319                _ => String::new(),
320            },
321            s(c, "evidence"),
322            s(c, "ref"),
323            c["reproduce"]
324                .as_str()
325                .map(|r| format!("  [reproduce: {r}]"))
326                .unwrap_or_default()
327        ));
328    }
329    for r in v["verdict"]["reasons"].as_array().into_iter().flatten() {
330        out.push(format!(
331            "  held back   {}  {} {}",
332            s(r, "criterion"),
333            s(r, "evidence"),
334            s(r, "state"),
335        ));
336    }
337    for g in v["governance"].as_array().into_iter().flatten() {
338        out.push(format!("  governance  {}", g.as_str().unwrap_or("")));
339    }
340    out.join("\n")
341}

356fn validate_text(v: &Value) -> String {
357    let mut out = Vec::new();
358    findings_text(&mut out, &v["findings"]);
359    out.push(format!(
360        "{} intent(s), {} failure(s), {} warning(s): {}",
361        v["intents"],
362        v["failures"],
363        v["warnings"],
364        if v["valid"].as_bool() == Some(true) {
365            "valid"
366        } else {
367            "invalid"
368        }
369    ));
370    out.join("\n")
371}

381/// The verdict, each issue with its own, each intent the work is held to with what it asks
382/// of the worker, then every refusal with its cause.
383fn preflight_text(v: &Value) -> String {
384    let mut out = vec![format!("verdict     {}", s(v, "verdict"))];
385    for i in v["issues"].as_array().into_iter().flatten() {
386        let serves = words(&i["serves"]);
387        out.push(format!(
388            "issue       {}  {}  milestone {}{}",
389            s(i, "issue"),
390            s(i, "verdict"),
391            s(i, "milestone"),
392            if serves.is_empty() {
393                String::new()
394            } else {
395                format!("  serves {}", serves.join(" "))
396            }
397        ));
398    }
399    for i in v["intents"].as_array().into_iter().flatten() {
400        out.push(format!(
401            "intent      {}  {}  {}",
402            s(i, "id"),
403            s(i, "stage"),
404            s(i, "title")
405        ));
406        out.push(format!("  statement   {}", s(i, "statement")));
407        for c in i["criteria"].as_array().into_iter().flatten() {
408            out.push(format!("  criterion   {}  {}", s(c, "id"), s(c, "state")));
409        }
410        for inv in words(&i["invariants"]) {
411            out.push(format!("  invariant   {inv}"));
412        }
413        for n in words(&i["non_goals"]) {
414            out.push(format!("  non-goal    {n}"));
415        }
416        let critique = &i["critique"];
417        if critique.is_object() {
418            let open: Vec<&str> = critique["open_blocking"]
419                .as_array()
420                .into_iter()
421                .flatten()
422                .map(|f| s(f, "id"))
423                .collect();
424            out.push(format!(
425                "  critique    reviewed at {} by {}; open blocking: {}",
426                s(critique, "reviewed_at"),
427                s(critique, "reviewed_by"),
428                if open.is_empty() {
429                    "none".to_string()
430                } else {
431                    open.join(", ")
432                }
433            ));
434        } else {
435            out.push("  critique    none recorded".into());
436        }
437        let gap = &i["gap"];
438        if gap.is_object() {
439            let conditions: Vec<String> = gap["conditions"]
440                .as_array()
441                .into_iter()
442                .flatten()
443                .map(|c| format!("{} {}", s(c, "criterion"), s(c, "state_text")))
444                .collect();
445            out.push(format!(
446                "  gap         observed at {}: {}",
447                s(gap, "observed_at"),
448                conditions.join(", ")
449            ));
450        }
451    }
452    for g in words(&v["governance"]) {
453        out.push(format!("governance  {g}"));
454    }
455    for r in v["refusals"].as_array().into_iter().flatten() {
456        out.push(format!(
457            "refusal     {}  {}",
458            s(r, "cause"),
459            s(r, "message")
460        ));
461    }
462    out.join("\n")
463}

470    /// Every part of an answer has its line: an issue serving nothing, an intent with
471    /// non-goals, an open blocking finding and a recorded gap, one with no critique at all,
472    /// the governance and each refusal with its cause.
473    #[test]
474    fn the_preflight_text_says_every_part_of_the_answer() {
475        let answer = json!({
476            "verdict": "refused",
477            "issues": [
478                {"issue": "I1", "verdict": "serves", "milestone": "m", "serves": ["x#case"]},
479                {"issue": "I2", "verdict": "refused", "milestone": "m", "serves": []}
480            ],
481            "intents": [
482                {"id": "x", "stage": "active", "title": "The x", "statement": "x holds",
483                 "criteria": [{"id": "case", "state": "current"}],
484                 "invariants": ["never y"], "non_goals": ["not z"],
485                 "critique": {"reviewed_at": "c0ffee", "reviewed_by": "a reviewer",
486                              "open_blocking": [{"id": "F1"}, {"id": "F2"}]},
487                 "gap": {"observed_at": "2026-10-06T00:00:00Z",
488                         "conditions": [{"criterion": "case", "state_text": "not met"}]}},
489                {"id": "w", "stage": "active", "title": "The w", "statement": "w holds",
490                 "criteria": [], "invariants": [], "non_goals": [], "critique": null,
491                 "gap": null}
492            ],
493            "governance": ["rule:project.alpha"],
494            "refusals": [{"cause": "issue_serves_nothing", "message": "I2 serves nothing"}]
495        });
496        let text = preflight_text(&answer);
497        for line in [
498            "verdict     refused",
499            "issue       I1  serves  milestone m  serves x#case",
500            "issue       I2  refused  milestone m",
501            "intent      x  active  The x",
502            "  statement   x holds",
503            "  criterion   case  current",
504            "  invariant   never y",
505            "  non-goal    not z",
506            "  critique    reviewed at c0ffee by a reviewer; open blocking: F1, F2",
507            "  gap         observed at 2026-10-06T00:00:00Z: case not met",
508            "intent      w  active  The w",
509            "  critique    none recorded",
510            "governance  rule:project.alpha",
511            "refusal     issue_serves_nothing  I2 serves nothing",
512        ] {
513            assert!(
514                text.lines().any(|l| l == line),
515                "{line:?} missing from\n{text}"
516            );
517        }
518        // an issue that serves nothing says no `serves` at all
519        assert!(text
520            .lines()
521            .any(|l| l.ends_with("milestone m") && l.contains("I2")));
522    }

524    #[test]
525    fn each_verb_reaches_its_capability_and_an_unexposed_one_is_named() {
526        let repo = SyntheticRepository::small().unwrap();
527        let ctx = repo.context().unwrap();
528
529        // a repository with no intent answers the list, and validates
530        let list = call(&ctx, &["intent", "list"], json!({})).unwrap();
531        assert_eq!(list["count"], 0);
532        // the JSON the command prints is byte for byte serde_json's pretty rendering
533        assert_eq!(
534            format!("{list:#}"),
535            serde_json::to_string_pretty(&list).unwrap()
536        );
537        let valid = call(&ctx, &["intent", "validate"], json!({})).unwrap();
538        assert_eq!(valid["valid"], true);
539
540        match call(&ctx, &["intent", "absent"], json!({})) {
541            Err(Error::Protocol { reason }) => assert_eq!(
542                reason,
543                "no capability is exposed as `majordomus intent absent`"
544            ),
545            other => panic!("an unexposed verb answered: {other:?}"),
546        }
547        // the capability's own not-found is the command line's not-found, exit 12
548        match call(&ctx, &["intent", "show"], json!({ "id": "absent" })) {
549            Err(e @ Error::NotFound { .. }) => {
550                assert_eq!(e.exit_code(), 12);
551                assert!(e.to_string().contains("no intent 'absent'"), "{e}");
552            }
553            other => panic!("an absent intent answered: {other:?}"),
554        }
555        // every other refusal is a protocol error carrying the capability's message
556        match call(&ctx, &["intent", "preflight"], json!({ "paths": "" })) {
557            Err(Error::Protocol { reason }) => {
558                assert!(reason.contains("name the issue"), "{reason}")
559            }
560            other => panic!("a preflight of nothing answered: {other:?}"),
561        }
562    }

566        let intent = json!({
