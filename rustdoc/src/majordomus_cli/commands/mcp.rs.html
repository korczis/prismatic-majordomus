1//! `majordomus mcp`: serve one MCP client over stdio. By default the process joins the
2//! repository's one shared server: it starts it when nobody has (loopback HTTP with
3//! Swagger UI, OpenAPI and MCP over HTTP beside its own stdio session) and bridges its
4//! stdio to it when someone has. `--standalone` serves the client alone, as the first
5//! version of this executable did; `--inspect` prints what would be served and stops.

28/// Run `majordomus mcp`.
29pub fn run(args: McpArgs) -> Result<u8> {
30    if args.inspect {
31        let app = App::load(&args.repo)?;
32        return inspect(&Surface::new(app.context.clone()), args.format);
33    }
34    match args.transport {
35        Transport::Stdio => {}
36    }
37    let stdin = std::io::stdin();
38    let stdout = std::io::stdout();
39    if args.standalone {
40        let mut server = standalone(&args.repo)?;
41        tracing::info!("standalone: no shared server, no HTTP, no peers");
42        stdio::serve(stdin.lock(), stdout.lock(), |m| server.handle(m))?;
43        return Ok(0);
44    }
45    let repo = Repository::discover(&start_dir(&args.repo)?)?;
46    let mut session = Session::open(args, repo)?;
47    stdio::serve(stdin.lock(), stdout.lock(), |m| session.handle(m))?;
48    session.finish();
49    Ok(0)
50}

52/// A server answering one stdio session alone: no shared server, no HTTP, no peers.
53fn standalone(repo: &RepoArgs) -> Result<Server> {
54    let app = App::load(repo)?;
55    // a stdio session outlives commits like any other: it follows the repository too
56    let live = Arc::new(Live::watching(repo.clone(), app.context.clone()));
57    let peer = live.current().peers.attach(PeerTransport::Stdio);
58    Ok(Server::new(
59        Surface::new(Arc::clone(&live)).for_peer(peer),
60        crate::VERSION,
61    ))
62}

64fn start_dir(args: &RepoArgs) -> Result<std::path::PathBuf> {
65    match &args.repo {
66        Some(p) => Ok(p.clone()),
67        None => std::env::current_dir().map_err(|e| Error::io(".", e)),
68    }
69}

71/// The stdio session of this process, answered locally by the shared server this
72/// process runs, or forwarded to the one another process runs.
73struct Session {
74    args: McpArgs,
75    repo: Repository,
76    backend: Backend,
77}

79enum Backend {
80    Local(Box<Local>),
81    Remote {
82        bridge: Arc<Mutex<Bridge>>,
83        heartbeat: Heartbeat,
84    },
85    /// The shared server cannot be used (the lease cannot be written or replaced, the
86    /// server cannot start): this client is served alone, as `--standalone` would, and
87    /// the log said why.
88    Alone(Box<Server>),
89}

91/// This process is the shared server; its own stdio is answered here.
92struct Local {
93    server: Server,
94    peer: PeerId,
95    shared: SharedServer,
96}

98struct Heartbeat {
99    stop: Arc<AtomicBool>,
100    thread: Option<JoinHandle<()>>,
101}
102
103impl Heartbeat {
104    fn start(bridge: Arc<Mutex<Bridge>>) -> Self {
105        let stop = Arc::new(AtomicBool::new(false));
106        let flag = Arc::clone(&stop);
107        let thread = std::thread::Builder::new()
108            .name("mcp-heartbeat".into())
109            .spawn(move || {
110                let tick = Duration::from_millis(500);
111                let mut waited = Duration::ZERO;
112                while !flag.load(Ordering::SeqCst) {
113                    std::thread::sleep(tick);
114                    waited += tick;
115                    if waited >= HEARTBEAT {
116                        waited = Duration::ZERO;
117                        if let Err(e) = lock(&bridge).heartbeat() {
118                            tracing::debug!("heartbeat: {e}");
119                        }
120                    }
121                }
122            })
123            .ok();
124        Heartbeat { stop, thread }
125    }
126
127    fn stop(mut self) {
128        self.stop.store(true, Ordering::SeqCst);
129        if let Some(t) = self.thread.take() {
130            let _ = t.join();
131        }
132    }
133}

135impl Session {
136    fn open(args: McpArgs, repo: Repository) -> Result<Self> {
137        // the policy declares the lease timings, and it is read before the first election
138        // reads them: a timing read first is the compiled default for the life of the process
139        // (I2129). A policy that does not parse leaves the defaults; `doctor` reports it.
140        let _ = crate::policy::LoadedPolicy::load(&repo);
141        let backend = match lease::elect(&repo) {
142            Ok(Role::Server(lease)) => match Self::serve(&args, lease, None) {
143                Ok(backend) => backend,
144                Err(e) => Self::alone(&args, None, &format!("starting it failed: {e}"))?,
145            },
146            Ok(Role::Peer { url }) => Self::attach(url),
147            Err(e) => Self::alone(&args, None, &e.to_string())?,
148        };
149        Ok(Session {
150            args,
151            repo,
152            backend,
153        })
154    }
155
156    /// The shared server cannot be used: serve this client alone and say why. The layer
157    /// itself still has to load; when it does not, that error is the answer, not this.
158    fn alone(args: &McpArgs, resume: Option<ClientInfo>, why: &str) -> Result<Backend> {
159        let mut server = standalone(&args.repo)?;
160        tracing::warn!(
161            "cannot use the shared server: {why}; serving this client alone (no HTTP, no Swagger UI, no peers), as --standalone would"
162        );
163        if let Some(client) = resume {
164            server.resume(client);
165        }
166        Ok(Backend::Alone(Box::new(server)))
167    }
168
169    /// Become the shared server: load the layer, bind, and answer this stdio locally.
170    fn serve(
171        args: &McpArgs,
172        lease: Lease,
173        resume: Option<crate::peers::ClientInfo>,
174    ) -> Result<Backend> {
175        // the lease is kept young while the layer loads, so that a peer waiting on it
176        // never mistakes a slow start for an abandoned one
177        lease.keep_alive();
178        let app = App::load(&args.repo)?;
179        let live = Arc::new(Live::watching(args.repo.clone(), app.context.clone()));
180        let ctx = live.current();
181        let root = std::path::Path::new(&app.context.index.repository.root);
182        let (http_host, _) = crate::cli::checkout_http_host(args.http_host.as_deref(), root);
183        // the port this checkout asks for on every start, unless one was given (I2702)
184        let http_port = args
185            .http_port
186            .unwrap_or_else(|| crate::cli::checkout_port(root));
187        let shared = SharedServer::start(
188            Arc::clone(&live),
189            crate::VERSION,
190            &http_host,
191            http_port,
192            true,
193            None,
194            lease,
195            Some(app.share.dir()),
196        )?;
197        let peer = ctx.peers.attach(PeerTransport::Stdio);
198        let mut server = Server::new(Surface::new(live).for_peer(peer.clone()), crate::VERSION)
199            .with_endpoint(Some(shared.url()));
200        if let Some(client) = resume {
201            server.resume(client);
202        }
203        Ok(Backend::Local(Box::new(Local {
204            server,
205            peer,
206            shared,
207        })))
208    }
209
210    /// Attach to the shared server another process runs.
211    fn attach(url: String) -> Backend {
212        tracing::info!(
213            url = %url,
214            "a shared server for this repository is already running at {url} (swagger ui {url}{swagger}); bridging this stdio session to it",
215            swagger = crate::http::swagger::SWAGGER_PATH
216        );
217        let bridge = Arc::new(Mutex::new(Bridge::new(url)));
218        let heartbeat = Heartbeat::start(Arc::clone(&bridge));
219        Backend::Remote { bridge, heartbeat }
220    }
221
222    fn handle(&mut self, message: Value) -> Option<Reply> {
223        match &mut self.backend {
224            Backend::Local(local) => local.server.handle(message),
225            Backend::Alone(server) => server.handle(message),
226            Backend::Remote { bridge, .. } => {
227                let answer = lock(bridge).handle(&message);
228                match answer {
229                    Ok(v) if message["method"] == "initialize" => {
230                        v.map(|v| Reply::Value(told_of_version(v, crate::VERSION)))
231                    }
232                    Ok(v) => v.map(Reply::Value),
233                    Err(e) if e.calls_for_an_election() => self.failover(message, e),
234                    // the server is there and would not take this request: that is its
235                    // answer, and the client is given it instead of an election
236                    Err(e) => refused(&message, &e),
237                }
238            }
239        }
240    }
241
242    /// The server this session was bridged to is gone: elect again, and either become
243    /// the server (carrying the client's session over) or attach to whoever did.
244    ///
245    /// The message is sent again only when the cause says that cannot run it twice
246    /// ([`BridgeError::may_resend`]); otherwise the election still happens, for the
247    /// messages that follow, and the client is told this one's outcome is unknown.
248    fn failover(&mut self, message: Value, cause: BridgeError) -> Option<Reply> {
249        let resend = cause.may_resend(&message);
250        tracing::warn!("{cause}; electing again");
251        let (client, announcements) = match &self.backend {
252            Backend::Remote { bridge, .. } => {
253                let b = lock(bridge);
254                (b.client().cloned(), b.announcements().cloned().collect())
255            }
256            Backend::Local(_) | Backend::Alone(_) => (None, Vec::new()),
257        };
258        match lease::elect(&self.repo) {
259            Ok(Role::Server(lease)) => {
260                match Self::serve(&self.args, lease, client.clone()) {
261                    Ok(backend) => {
262                        self.replace(backend);
263                        tracing::info!(
264                            "took over as the shared server; this session continues locally"
265                        );
266                        // what the client said it was working on, said again on the board it
267                        // now serves itself: an announcement outlives the server it was made
268                        // to. Every claim it holds, not the newest — a client that carried
269                        // one of three onto its own board would be understating itself to
270                        // every peer that arrives after the takeover.
271                        if let Backend::Local(local) = &self.backend {
272                            for a in &announcements {
273                                let intent = a["intent"].as_str().unwrap_or_default().trim();
274                                if intent.is_empty() {
275                                    continue;
276                                }
277                                let scope: Vec<String> = a["scope"]
278                                    .as_array()
279                                    .map(|s| {
280                                        s.iter()
281                                            .filter_map(|v| v.as_str())
282                                            .map(|v| v.trim().to_string())
283                                            .filter(|v| !v.is_empty())
284                                            .collect()
285                                    })
286                                    .unwrap_or_default();
287                                let claim =
288                                    a["claim"].as_str().map(str::trim).filter(|c| !c.is_empty());
289                                local.server.surface().context().peers.announce_claim(
290                                    &local.peer,
291                                    claim,
292                                    intent,
293                                    scope,
294                                );
295                            }
296                            if !announcements.is_empty() {
297                                tracing::info!(
298                                    claims = announcements.len(),
299                                    "the client's announcement was carried onto this server's board"
300                                );
301                            }
302                        }
303                        if resend {
304                            self.handle(message)
305                        } else {
306                            outcome_unknown(&message, &cause)
307                        }
308                    }
309                    Err(e) => {
310                        tracing::error!("cannot take over as the shared server: {e}");
311                        let reason = format!("{cause}; and taking over failed: {e}");
312                        self.settle_alone(message, client, &reason, (!resend).then_some(&cause))
313                    }
314                }
315            }
316            Ok(Role::Peer { url }) => {
317                let Backend::Remote { bridge, .. } = &self.backend else {
318                    return unavailable(&message, &cause.to_string());
319                };
320                let mut b = lock(bridge);
321                b.move_to(url.clone());
322                let retry =
323                    b.reinitialize()
324                        .and_then(|()| if resend { b.handle(&message) } else { Ok(None) });
325                match retry {
326                    Ok(_) if !resend => {
327                        tracing::info!(url = %url, "re-attached to the shared server");
328                        outcome_unknown(&message, &cause)
329                    }
330                    Ok(v) => {
331                        tracing::info!(url = %url, "re-attached to the shared server");
332                        v.map(Reply::Value)
333                    }
334                    Err(_) if !resend => outcome_unknown(&message, &cause),
335                    Err(e) if !e.calls_for_an_election() => refused(&message, &e),
336                    Err(e) => unavailable(&message, &e.to_string()),
337                }
338            }
339            Err(e) => {
340                let reason = format!("{cause}; electing again failed: {e}");
341                self.settle_alone(message, client, &reason, (!resend).then_some(&cause))
342            }
343        }
344    }
345
346    /// Replace the backend, stopping the heartbeat of a bridge that is left behind.
347    fn replace(&mut self, backend: Backend) {
348        let old = std::mem::replace(&mut self.backend, backend);
349        if let Backend::Remote { heartbeat, .. } = old {
350            heartbeat.stop();
351        }
352    }
353
354    /// Neither serving nor attaching worked: serve this client alone when the layer
355    /// loads, and answer the request with an error naming every failure when it does not.
356    /// `withheld` is the failure that may have run the message already: it is then not
357    /// run here, and the client is told its outcome is unknown.
358    fn settle_alone(
359        &mut self,
360        message: Value,
361        client: Option<ClientInfo>,
362        reason: &str,
363        withheld: Option<&BridgeError>,
364    ) -> Option<Reply> {
365        match Self::alone(&self.args, client, reason) {
366            Ok(backend) => {
367                self.replace(backend);
368                match withheld {
369                    Some(cause) => outcome_unknown(&message, cause),
370                    None => self.handle(message),
371                }
372            }
373            Err(e) => {
374                tracing::error!("cannot serve this client alone either: {e}");
375                match withheld {
376                    Some(cause) => outcome_unknown(&message, cause),
377                    None => unavailable(&message, reason),
378                }
379            }
380        }
381    }
382
383    /// The client has gone. A server lingers while other peers are attached; a bridge
384    /// tells the server its session is over.
385    fn finish(self) {
386        match self.backend {
387            Backend::Local(local) => {
388                let Local {
389                    server,
390                    peer,
391                    shared,
392                } = *local;
393                // Detached, not closed: the stdio client that just went may be a client
394                // somebody is restarting, and it comes back to its own episode by name.
395                // `shared.stop()` below closes whatever is still held, which is the moment
396                // at which nothing can come back any more (ADR 0103).
397                server.surface().context().episodes.detach(&peer);
398                server.surface().context().peers.detach(&peer);
399                drop(server);
400                shared.wait_until_peers_leave();
401                shared.stop();
402            }
403            Backend::Remote { bridge, heartbeat } => {
404                heartbeat.stop();
405                lock(&bridge).close();
406                tracing::info!("bridge closed");
407            }
408            Backend::Alone(_) => {}
409        }
410    }
411}

413/// The `initialize` answer a bridged client relays, with what its client must know when the
414/// shared server is a newer release than this executable (`mine`): the answers come from
415/// the newer server, while this process — the bridge, its command line — is the older one
416/// until it is started from the newer executable. Said in the instructions, which every
417/// client shows its model, because an older client that attached in silence is how a
418/// version difference went unseen (I2163). Any other answer is relayed as it is.
419fn told_of_version(mut answer: Value, mine: &str) -> Value {
420    let Some(result) = answer.get_mut("result").and_then(Value::as_object_mut) else {
421        return answer;
422    };
423    let served = result
424        .get("serverInfo")
425        .and_then(|info| info.get("version"))
426        .and_then(Value::as_str)
427        .map(str::to_string);
428    if lease::version_verdict(served.as_deref(), mine) != lease::VersionVerdict::ServerNewer {
429        return answer;
430    }
431    let theirs = served.unwrap_or_default();
432    let note = format!(
433        "This client is majordomus {mine}; the shared server it is bridged to runs {theirs}, a newer release. Its answers are the newer server's; this client process is the older one until it is started from the newer executable."
434    );
435    tracing::warn!("{note}");
436    let instructions = result
437        .get("instructions")
438        .and_then(Value::as_str)
439        .map(|i| format!("{note}\n\n{i}"))
440        .unwrap_or(note);
441    result.insert("instructions".into(), Value::String(instructions));
442    answer
443}

445/// The JSON-RPC answer for a request that no server could take; nothing for a notification.
446fn unavailable(message: &Value, reason: &str) -> Option<Reply> {
447    let id = message.get("id").cloned().filter(|i| !i.is_null())?;
448    Some(Reply::Value(json!({
449        "jsonrpc": "2.0",
450        "id": id,
451        "error": { "code": -32603, "message": format!("shared server unavailable: {reason}") }
452    })))
453}
454
455/// The JSON-RPC answer for a request that was sent, got no answer, and is not sent again
456/// because it may have run: its outcome is unknown, and the answer says so with its id.
457/// Nothing for a notification.
458fn outcome_unknown(message: &Value, cause: &BridgeError) -> Option<Reply> {
459    let id = message.get("id").cloned().filter(|i| !i.is_null())?;
460    tracing::warn!(request = %id, "{cause}; the request may have run, so it is not sent again");
461    Some(Reply::Value(json!({
462        "jsonrpc": "2.0",
463        "id": id,
464        "error": {
465            "code": -32603,
466            "message": format!(
467                "outcome unknown: request {id} was sent to the shared server and no answer came \
468                 ({cause}); it may have run, so it was not sent again"
469            ),
470            "data": { "request_id": id, "outcome": "unknown" }
471        }
472    })))
473}

475/// The JSON-RPC answer for a request the server answered with a refusal that calls for no
476/// election (a 4xx other than 404 or 409): that refusal, named. Nothing for a notification.
477fn refused(message: &Value, cause: &BridgeError) -> Option<Reply> {
478    let id = message.get("id").cloned().filter(|i| !i.is_null())?;
479    let status = match cause {
480        BridgeError::Rejected { status, .. } => Some(*status),
481        BridgeError::Unreachable { .. } | BridgeError::Unanswered { .. } => None,
482    };
483    let text = match status {
484        Some(413) => format!("payload too large: request {id} was refused: {cause}"),
485        _ => format!("request {id} was refused: {cause}"),
486    };
487    Some(Reply::Value(json!({
488        "jsonrpc": "2.0",
489        "id": id,
490        "error": { "code": -32600, "message": text, "data": { "request_id": id, "status": status } }
491    })))
492}

498/// The word an inspection puts before a diagnostic: the verdict words every other report

503        Severity::Warning => "WARN",

508/// What an inspection says about the projection, one line each: who answers and with which
509/// protocol versions, how many tools carry each effect, the tools that write the
510/// repository, where each declared client's configuration stands, and what that leaves to
511/// be done.
512fn projection_lines(p: &crate::capability::builtin::mcp::McpProjection) -> Vec<String> {
513    // an effect and a standing are written as the words they serialise to, which are the
514    // words every other surface shows
515    fn word<T: serde::Serialize>(value: T) -> String {
516        serde_json::to_value(value)
517            .ok()
518            .and_then(|v| v.as_str().map(String::from))
519            .unwrap_or_default()
520    }
521    let mut lines = vec![
522        format!("server      {} {}", p.server.name, p.server.version),
523        format!("protocol    {}", p.protocol_versions.join(", ")),
524    ];
525    lines.extend(
526        p.effects
527            .iter()
528            .map(|e| format!("effect      {:<20} {} tool(s)", word(e.effect), e.tools)),
529    );
530    lines.extend(p.writers.iter().map(|name| format!("writes      {name}")));
531    lines.extend(
532        p.clients
533            .iter()
534            .map(|c| format!("client      {:<24} {}", c.config, word(c.standing))),
535    );
536    lines.extend(
537        p.findings
538            .iter()
539            .map(|f| format!("WARN {:<22} - — {} ({})", f.code, f.message, f.remedy)),
540    );
541    lines
542}

544fn inspect(surface: &Surface, format: OutputFormat) -> Result<u8> {
545    let stdout = std::io::stdout();
546    let mut out = stdout.lock();
547    let index = surface.index();
548    // the projection as the capability every other surface asks describes it, so that what
549    // this prints about the protocol, the effects and the clients is not a second account
550    let described = crate::capability::builtin::mcp::describe(
551        &surface.context(),
552        crate::capability::builtin::mcp::McpProjectionInput::default(),
553    );
554    match format {
555        OutputFormat::Json => {
556            let v = json!({
557                "repository": surface.repository_info().map_err(|e| Error::Protocol { reason: e.to_string() })?,
558                "mcp": described,
559                "resources": &*surface.resources(),
560                "tools": &*surface.tools(),
561            });
562            writeln!(
563                out,
564                "{}",
565                serde_json::to_string_pretty(&v).map_err(|e| Error::Protocol {
566                    reason: e.to_string()
567                })?
568            )
569            .map_err(Error::Transport)?;
570        }
571        OutputFormat::Text => {
572            // the lines first, then one write: an inspection is printed whole or not at all
573            let mut lines = vec![
574                format!("repository  {}", index.repository.root),
575                format!("discovery   {}", index.repository.discovery),
576                format!(
577                    "state       {}",
578                    match index.state {
579                        crate::index::State::Ok => "ok",
580                        crate::index::State::Degraded => "degraded",
581                    }
582                ),
583            ];
584            lines.extend(
585                index
586                    .kinds()
587                    .into_iter()
588                    .map(|(kind, n)| format!("kind        {kind:<12} {n}")),
589            );
590            let summary = surface.registry().summary();
591            lines.push(format!(
592                "capabilities {} ({} builtin, {} declarative)",
593                summary.total, summary.builtin, summary.declarative
594            ));
595            lines.extend(projection_lines(&described));
596            lines.extend(
597                surface
598                    .resources()
599                    .iter()
600                    .map(|r| format!("resource    {}", r.uri)),
601            );
602            lines.extend(
603                surface
604                    .tools()
605                    .iter()
606                    .map(|t| format!("tool        {}", t.name)),
607            );
608            lines.extend(index.diagnostics.iter().map(|d| {
609                format!(
610                    "{:<4} {:<22} {} — {}",
611                    level(d.severity),
612                    d.code,
613                    d.path.as_deref().unwrap_or("-"),
614                    d.message
615                )
616            }));
617            writeln!(out, "{}", lines.join("\n")).map_err(Error::Transport)?;
618        }
619    }
620    Ok(if index.errors() > 0 { 10 } else { 0 })
621}

648    fn older_client_told_test() {
649        // bridged to a newer release, the client is told, before what the server said
650        let told = told_of_version(initialized("0.21.0"), "0.20.0");
651        let instructions = told["result"]["instructions"].as_str().unwrap();
652        assert!(instructions.starts_with("This client is majordomus 0.20.0"));
653        assert!(instructions.contains("runs 0.21.0, a newer release"));
654        assert!(instructions.ends_with("The projection of a repository's AI layer."));
655        // the same release, or an older one, is relayed as it is
656        assert_eq!(
657            told_of_version(initialized("0.20.0"), "0.20.0"),
658            initialized("0.20.0")
659        );
660        assert_eq!(
661            told_of_version(initialized("0.19.1"), "0.20.0"),
662            initialized("0.19.1")
663        );
664        // and an error answer has nothing to be told in
665        let error = json!({"jsonrpc": "2.0", "id": 1, "error": {"code": -32600}});
666        assert_eq!(told_of_version(error.clone(), "0.20.0"), error);
667    }

673    #[test]
674    fn a_request_whose_outcome_is_unknown_is_answered_with_its_id_and_a_notification_is_not() {
675        let cause = BridgeError::Unanswered {
676            url: url(),
677            reason: "timed out".into(),
678        };
679        let request = json!({ "jsonrpc": "2.0", "id": 7, "method": "tools/call" });
680        let Some(Reply::Value(answer)) = outcome_unknown(&request, &cause) else {
681            panic!("a request is answered");
682        };
683        assert_eq!(answer["id"], 7);
684        assert_eq!(answer["error"]["code"], -32603);
685        assert_eq!(answer["error"]["data"]["outcome"], "unknown");
686        assert_eq!(answer["error"]["data"]["request_id"], 7);
687        let text = answer["error"]["message"].as_str().unwrap();
688        assert!(text.starts_with("outcome unknown: request 7"), "{text}");
689        assert!(text.contains("timed out"), "the cause is named: {text}");
690        let notification = json!({ "jsonrpc": "2.0", "method": "notifications/initialized" });
691        assert!(outcome_unknown(&notification, &cause).is_none());
692        let null_id = json!({ "jsonrpc": "2.0", "id": null, "method": "ping" });
693        assert!(outcome_unknown(&null_id, &cause).is_none());
694    }

696    #[test]
697    fn a_refusal_is_answered_as_that_refusal_with_its_status() {
698        let request = json!({ "jsonrpc": "2.0", "id": 9, "method": "tools/call" });
699        let answer = |cause: &BridgeError| match refused(&request, cause) {
700            Some(Reply::Value(v)) => v,
701            _ => panic!("a request is answered with a value"),
702        };
703        let too_large = answer(&BridgeError::Rejected {
704            url: url(),
705            status: 413,
706            body: "too_large".into(),
707        });
708        assert_eq!(too_large["error"]["code"], -32600);
709        assert_eq!(too_large["error"]["data"]["status"], 413);
710        let text = too_large["error"]["message"].as_str().unwrap();
711        assert!(text.starts_with("payload too large: request 9"), "{text}");
712
713        let bad = answer(&BridgeError::Rejected {
714            url: url(),
715            status: 400,
716            body: "invalid".into(),
717        });
718        assert_eq!(bad["error"]["data"]["status"], 400);
719        assert!(bad["error"]["message"]
720            .as_str()
721            .unwrap()
722            .starts_with("request 9 was refused"));
723
724        // a failure that carries no status is still named, with no status claimed for it
725        for cause in [
726            BridgeError::Unreachable {
727                url: url(),
728                reason: "connection refused".into(),
729            },
730            BridgeError::Unanswered {
731                url: url(),
732                reason: "timed out".into(),
733            },
734        ] {
735            let v = answer(&cause);
736            assert_eq!(v["error"]["data"]["status"], Value::Null, "{v}");
737            assert!(v["error"]["message"].as_str().unwrap().contains(&url()));
738        }
739        let notification = json!({ "jsonrpc": "2.0", "method": "notifications/cancelled" });
740        let rejected = BridgeError::Rejected {
741            url: url(),
742            status: 400,
743            body: String::new(),
744        };
745        assert!(refused(&notification, &rejected).is_none());
746    }
