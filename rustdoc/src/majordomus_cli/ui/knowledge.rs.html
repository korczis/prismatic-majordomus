1//! Knowledge: what the deriver left for review and whether it is still writing. Read from
2//! `knowledge_base.status` and `knowledge_base.candidates`.
3//!
4//! The screen shows the queue and names the commands a reviewer runs. It promotes and rejects
5//! nothing: a screen reads and never executes (ADR 0142), and promotion is a person's act.
6//! The table is bounded, because the queue is not: what is left out is counted and said.

27/// How many candidates, and how many subjects, the screen draws. The queue has no bound of
28/// its own beyond the policy's cap, and a screen answers within its budget whatever the queue
29/// holds; the newest are drawn and the rest are counted.
30pub const SHOWN: usize = 50;

32/// The commands a reviewer runs, in the order a review takes them. Text, never a control:
33/// the screen is a reading.
34const REVIEW: &[(&str, &str)] = &[
35    (
36        "majordomus knowledge candidates",
37        "the queue, with the branch each record came from",
38    ),
39    (
40        "majordomus knowledge observe --kind <kind> --subject <subject> \"<statement>\"",
41        "what was noticed is written down, so the next derivation has it",

43    (
44        "majordomus knowledge record <id>",
45        "one record with every reference it names resolved",
46    ),
47    (
48        "majordomus knowledge promote <id> < evidence.md",
49        "the record becomes curated knowledge, with the evidence a person wrote",

51    (
52        "majordomus knowledge reject <id> --reason \"<why>\"",
53        "the record is kept as rejected, with the reason",

55    (

57        "every record is valid, evidenced and resolves what it names",

61/// The Knowledge screen.
62pub fn screen(ctx: &Context) -> View {
63    let answers = ask_all(ctx, &[STATUS, CANDIDATES]);
64    let status = &answers[&STATUS];
65    let queue = &answers[&CANDIDATES];
66
67    let top = hero(
68        "sparkles",
69        "orange",
70        "Knowledge",
71        "What the work left behind, awaiting a reviewer.",
72        "The records the deriver wrote down from closed sessions that no person has promoted or rejected yet, what each is about, how to review them, and whether learning keeps up with the sessions that close.",
73        vec![
74            kpi("Awaiting review", &queue.read("/total")),
75            kpi("On this branch", &queue.read("/on_this_branch")),
76            kpi(UNKNOWN_BRANCH, &queue.read("/unattributed")),
77            ledger_kpi(
78                NOT_LEARNED,
79                status,
80                kpi(NOT_LEARNED, &status.read("/closed_without_derivation")),
81            ),
82            ledger_kpi(
83                "Newest session",
84                status,
85                kpi_state("Newest session", &status.read("/freshness")),
86            ),
87        ],
88    );
89
90    // In the order a reviewer reads: what awaits them, what it is about, how to review it,
91    // and last whether the queue is still being filled.
92    let grid = el("div")
93        .class("mj-ui-grid")
94        .children(queue_panels(queue))
95        .child(panel(
96            "How to review",
97            "The commands a reviewer runs, in order. This screen only reads the queue; promoting and rejecting are a person's act.",
98            el("div").class("mj-ui-stack").child(review()).child(manual()),
99            &[],
100        ))
101        .child(panel(
102            "Is learning up to date?",
103            "Whether the deriver has written down what the newest closed session left behind, as knowledge status judges it.",
104            unanswered(status).unwrap_or_else(|| deriver(status)),
105            &[status],
106        ));
107
108    View::new(
109        "Knowledge",
110        el("div").class("mj-ui-stack").child(top).child(grid),
111    )
112}

114/// The queue's panels: what awaits review and what it is about, or, when the queue answered
115/// that nothing does, one sentence in their place. Two empty tables say the same thing twice
116/// and say neither why nor how a record would arrive.
117fn queue_panels(queue: &Answer) -> Vec<El> {
118    if queue_empty(queue) {
119        return vec![wide(panel(
120            "Awaiting review",
121            "The candidates no reviewer has promoted or rejected yet.",
122            empty_queue(),
123            &[queue],
124        ))];
125    }
126    vec![
127        wide(panel(
128            "Awaiting review",
129            "The newest candidates, each leading to its record.",
130            unanswered(queue).unwrap_or_else(|| candidates(queue)),
131            &[queue],
132        )),
133        wide(panel(
134            "By subject",
135            "The records derived from observations, grouped by what they are about.",
136            unanswered(queue).unwrap_or_else(|| subjects(queue)),
137            &[queue],
138        )),
139    ]
140}

142/// Whether the queue answered and holds nothing: a total of zero and no candidate. A queue
143/// that did not answer is not empty; it is unanswered, and its panels say so.
144fn queue_empty(queue: &Answer) -> bool {
145    queue
146        .ok()
147        .is_some_and(|answer| answer.get("total").and_then(Value::as_u64) == Some(0))
148        && list(queue, "/candidates").is_empty()
149}

155        .class("mj-ui-prose")

158        .text(" records what was noticed.")

161/// The judgement of the deriver: when it last wrote, when an episode last closed, why the
162/// newest episode is judged as it is, and every finding the status reports.
163fn deriver(status: &Answer) -> El {
164    let Some(answer) = status.ok() else {
165        return nothing("The status did not answer.");
166    };
167    if !ledger_recorded(answer) {
168        return no_ledger();
169    }
170    let mark = |key: &str| -> String {
171        answer
172            .get(key)
173            .map(|m| {
174                let at = text(m, "ts");
175                let episode = text(m, "episode");
176                if episode.is_empty() {
177                    at
178                } else {
179                    format!("{at}, session {episode}")
180                }
181            })
182            .filter(|line| !line.is_empty())
183            .unwrap_or_else(|| "never, in this checkout".to_string())
184    };
185    let judged = answer.get("judged").and_then(Value::as_bool);
186    let stopped = answer.get("stopped_writer").and_then(Value::as_bool);
187    let verdict = match (judged, stopped) {
188        (Some(true), Some(true)) => "no: sessions close and nothing was learned from the newest",
189        (Some(true), _) => "yes: the newest closed session was learned from",
190        (Some(false), _) => "not judged; the findings say why",
191        (None, _) => "the status carries no judgement",
192    };
193    let fact = |name: &str, value: String| {
194        el("div")
195            .class("mj-ui-fact")
196            .child(el("dt").text(name))
197            .child(el("dd").text(value))
198    };
199    let facts = el("dl")
200        .class("mj-ui-facts")
201        .child(fact("Up to date", verdict.to_string()))
202        .child(fact("Last learned from", mark("last_derived")))
203        .child(fact("Last session closed", mark("last_closed")))
204        .child(fact("Why", text(answer, "freshness_reason")));
205    let findings = list(status, "/findings");
206    let body = el("div").class("mj-ui-stack").child(facts);
207    if findings.is_empty() {
208        return body;
209    }
210    body.child(
211        el("ul").class("mj-ui-prose-list").children(
212            findings
213                .iter()
214                .filter_map(Value::as_str)
215                .map(|finding| el("li").text(finding)),
216        ),
217    )
218}

220/// What a checkout without a session ledger is told instead of a judgement: the published
221/// snapshot and a fresh clone carry none, and a value read from a ledger that is not there
222/// would read as "never" or "0" about the repository when it is only about this checkout.
223const NO_LEDGER: &str = "no ledger";
224
225/// The tile that counts candidates whose branch this checkout does not know.
226const UNKNOWN_BRANCH: &str = "Unknown branch";
227
228/// The tile that counts closed sessions no derivation followed.
229const NOT_LEARNED: &str = "Not learned from";
230
231/// Whether the status read any line of the session ledger: a close, a derivation, or a
232/// close no derivation names. The status's own `present` also counts candidates, which are
233/// tracked records and are read in any checkout; the ledger is local, so it is asked apart.
234fn ledger_recorded(answer: &Value) -> bool {
235    answer.get("last_closed").is_some_and(|m| !m.is_null())
236        || answer.get("last_derived").is_some_and(|m| !m.is_null())
237        || answer
238            .get("closed_without_derivation")
239            .and_then(Value::as_u64)
240            .is_some_and(|n| n > 0)
241}
242
243/// A tile whose value is read from the session ledger: as drawn when the status read a
244/// ledger, and as [`NO_LEDGER`] when it read none, so a snapshot never states a count or a
245/// freshness it could not observe.
246fn ledger_kpi(label: &str, status: &Answer, drawn: El) -> El {
247    match status.ok() {
248        Some(answer) if !ledger_recorded(answer) => kpi(
249            label,
250            &Reading {
251                source: STATUS,
252                pointer: "/last_closed",
253                state: Availability::Available(Value::String(NO_LEDGER.to_string())),
254            },
255        ),
256        _ => drawn,
257    }
258}

260/// The deriver's panel where no session ledger is here: what that means, and where the
261/// judgement is made instead. Prose, not facts: there is nothing to tabulate.
262fn no_ledger() -> El {
263    el("p")
264        .class("mj-ui-prose")
265        .text("No session ledger is recorded in this checkout, so whether learning keeps up is not judged here. The published snapshot and a fresh clone carry none; in a working checkout, ")
266        .child(el("code").class("mj-ui-code").text("majordomus knowledge status"))
267        .text(" judges it, and this panel shows when the deriver last wrote.")
268}

270/// The review, as the commands it is made of: each step is the command as code, which breaks
271/// anywhere at phone width, and what it does as a sentence at reading size. The command is
272/// what a reviewer copies and the sentence what they read; neither is a headline value.
273fn review() -> El {
274    el("ol")
275        .class("mj-ui-prose-list")
276        .children(REVIEW.iter().map(|(command, does)| {
277            el("li")
278                .child(el("code").class("mj-ui-code").text(*command))
279                .text(format!(": {does}."))
280        }))
281}

303/// The subjects, the first [`SHOWN`] of them, each with how many records await review, how
304/// many were rejected and kept, and how many episodes observed it.
305fn subjects(queue: &Answer) -> El {
306    let groups = list(queue, "/groups");
307    let size = |group: &Value, key: &str| -> String {
308        group
309            .get(key)
310            .and_then(Value::as_array)
311            .map_or(0, Vec::len)
312            .to_string()
313    };
314    let rows = groups
315        .iter()
316        .take(SHOWN)
317        .map(|group| {
318            el("tr")
319                .child(el("td").text(text(group, "about")))
320                .child(el("td").class("mj-ui-dim").text(text(group, "route")))
321                .child(el("td").text(size(group, "candidates")))
322                .child(el("td").text(size(group, "rejected")))
323                .child(el("td").text(size(group, "episodes")))
324        })
325        .collect::<Vec<_>>();
326    let shown = rows.len();
327    let body = el("div").class("mj-ui-stack").child(table(
328        &["About", "Route", "Awaiting", "Rejected", "Episodes"],
329        rows,
330        "No record derived from an observation awaits review.",
331    ));
332    match more(shown, groups.len(), None) {
333        Some(line) => body.child(line),
334        None => body,
335    }
336}

338/// The candidates, the newest [`SHOWN`] of them first. The source orders by id, which begins
339/// with the episode, so the last of its list are the newest; the order is the source's and is
340/// only read from its other end.
341fn candidates(queue: &Answer) -> El {
342    let all = list(queue, "/candidates");
343    let rows = all
344        .iter()
345        .rev()
346        .take(SHOWN)
347        .map(|record| {
348            let id = text(record, "id");
349            let uri = text(record, "uri");
350            let from = record
351                .get("branch")
352                .and_then(Value::as_str)
353                .filter(|branch| !branch.is_empty())
354                .map_or_else(|| "not known in this checkout".to_string(), str::to_string);
355            el("tr")
356                .child(
357                    el("td").child(object(&uri, id)).child(
358                        el("span")
359                            .class("mj-ui-cell-title")
360                            .text(text(record, "title")),
361                    ),
362                )
363                .child(el("td").class("mj-ui-dim").text(text(record, "class")))
364                .child(el("td").class("mj-ui-dim").text(from))
365                .child(
366                    el("td")
367                        .child(state(record, "freshness"))
368                        .child(el("span").class("mj-ui-dim").text(waited(record))),
369                )
370        })
371        .collect::<Vec<_>>();
372    let shown = rows.len();
373    let body = el("div").class("mj-ui-stack").child(table(
374        &["Record", "Class", "From", "Waiting"],
375        rows,
376        "No candidate awaits review.",
377    ));
378    match more(shown, all.len(), None) {
379        Some(line) => body.child(line),
380        None => body,
381    }
382}

384/// How long a record has waited, in the largest whole unit: the source measured it in
385/// minutes, and says in `freshness_reason` when it could not.
386fn waited(record: &Value) -> String {
387    match record.get("age_minutes").and_then(Value::as_i64) {
388        Some(minutes) if minutes >= 2 * 24 * 60 => format!(" {} d", minutes / (24 * 60)),
389        Some(minutes) if minutes >= 2 * 60 => format!(" {} h", minutes / 60),
390        Some(minutes) => format!(" {minutes} min"),
391        None => format!(" {}", text(record, "freshness_reason")),
392    }
393}

403    /// kit lays tiles out at no less than 9.5rem (`mj-ui-strip`), which is a tile of about

421    fn candidate(n: usize) -> Value {
422        json!({
423            "id": format!("s-20261013{n:06}-aaaa-{n:012x}"),
