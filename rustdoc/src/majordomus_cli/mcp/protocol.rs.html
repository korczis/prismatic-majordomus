1//! JSON-RPC 2.0 and the MCP methods this server answers, over a [`Surface`]. Transport
2//! agnostic: a message in, zero or one message out. The subset spoken is the server side
3//! of the protocol: `initialize`, `ping`, `resources/list`, `resources/read`,
4//! `resources/templates/list`, `tools/list`, `tools/call`. Prompts are not advertised.
5//! Nothing here decides what a tool may change: a tool is a capability, its effect is the
6//! capability's, and its annotations are the capability's classified hints.
7

17/// One outgoing message. Most are built values; the two listings are prepared once and
18/// carried as raw JSON, so answering `tools/list` or `resources/list` copies bytes and
19/// builds nothing. A transport serialises a reply with `serde_json::to_string`; a caller
20/// that needs the data as a value takes [`Reply::into_value`].
21#[derive(Debug)]
22pub enum Reply {
23    /// A built message.
24    Value(Value),
25    /// A response whose result is prepared JSON.
26    Prepared {
27        /// The request id.
28        id: Value,
29        /// The serialised result object.
30        result: Arc<Box<RawValue>>,
31    },
32    /// The responses of a batch.
33    Batch(Vec<Reply>),
34}
35
36impl Reply {
37    /// The reply as a value (a prepared result is parsed).
38    pub fn into_value(self) -> Value {
39        match self {
40            Reply::Value(v) => v,
41            Reply::Prepared { id, result } => json!({
42                "jsonrpc": "2.0",
43                "id": id,
44                "result": serde_json::from_str::<Value>(result.get()).unwrap_or(Value::Null),
45            }),
46            Reply::Batch(items) => Value::Array(items.into_iter().map(Reply::into_value).collect()),
47        }
48    }
49}
50
51impl Serialize for Reply {
52    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
53        match self {
54            Reply::Value(v) => v.serialize(serializer),
55            Reply::Prepared { id, result } => {
56                let mut m = serializer.serialize_map(Some(3))?;
57                m.serialize_entry("jsonrpc", "2.0")?;
58                m.serialize_entry("id", id)?;
59                m.serialize_entry("result", &**result)?;
60                m.end()
61            }
62            Reply::Batch(items) => items.serialize(serializer),
63        }
64    }
65}

161    /// Handle one decoded message. A notification yields `None`; a batch yields a batch of
162    /// the responses its requests produced, or `None` when it held only notifications.
163    pub fn handle(&mut self, message: Value) -> Option<Reply> {
164        if let Some(peer) = self.surface.peer() {
165            self.surface.context().peers.touch(peer);
166            // ... and the same message is the heartbeat of whatever episode this connection
167            // holds. A client that is working never has to send anything it would not
168            // otherwise send in order to keep its episode alive, which is the difference
169            // between a lifecycle and a thing clients have to remember to do (ADR 0103).
170            self.surface.context().episodes.touch(peer);
171        }
172        match message {
173            Value::Array(batch) => {
174                if batch.is_empty() {
175                    return Some(Reply::Value(error(
176                        Value::Null,
177                        INVALID_REQUEST,
178                        "empty batch",
179                    )));
180                }
181                let responses: Vec<Reply> = batch
182                    .into_iter()
183                    .filter_map(|m| self.handle_one(m))
184                    .collect();
185                (!responses.is_empty()).then_some(Reply::Batch(responses))
186            }
187            other => self.handle_one(other),
188        }
189    }
190
191    fn handle_one(&mut self, message: Value) -> Option<Reply> {
192        let Value::Object(msg) = message else {
193            return Some(Reply::Value(error(
194                Value::Null,
195                INVALID_REQUEST,
196                "a message is a JSON object",
197            )));
198        };
199        let id = msg.get("id").cloned();
200        let Some(method) = msg.get("method").and_then(Value::as_str) else {
201            // A response or a malformed message; a server never answers a response.
202            return id
203                .filter(|i| !i.is_null())
204                .map(|i| Reply::Value(error(i, INVALID_REQUEST, "missing method")));
205        };
206        let params = msg.get("params").cloned().unwrap_or(Value::Null);
207        let Some(id) = id.filter(|i| !i.is_null()) else {
208            self.notification(method, &params);
209            return None;
210        };
211        tracing::debug!(operation = method, "request");
212        // the two listings are prepared once and copied, never rebuilt
213        match method {
214            "tools/list" => {
215                return Some(Reply::Prepared {
216                    id,
217                    result: self.surface.tools_result(),
218                })
219            }
220            "resources/list" => {
221                return Some(Reply::Prepared {
222                    id,
223                    result: self.surface.resources_result(),
224                })
225            }
226            _ => {}
227        }
228        Some(Reply::Value(match self.request(method, &params) {
229            Ok(result) => json!({ "jsonrpc": "2.0", "id": id, "result": result }),
230            Err((code, message)) => error(id, code, &message),
231        }))
232    }
233
234    fn notification(&mut self, method: &str, _params: &Value) {
235        match method {
236            "notifications/initialized" => self.initialized = true,
237            other => tracing::debug!(operation = other, "notification ignored"),
238        }
239    }

241    fn request(&mut self, method: &str, params: &Value) -> Result<Value, (i64, String)> {
242        if !METHODS.contains(&method) {
243            return Err((METHOD_NOT_FOUND, format!("method not found: {method}")));
244        }
245        match method {
246            "initialize" => Ok(self.initialize(params)),
247            "ping" => Ok(json!({})),
248            "resources/list" => Ok(
249                json!({ "resources": Value::Array(self.surface.resources_json().as_ref().clone()) }),
250            ),
251            "resources/templates/list" => Ok(json!({ "resourceTemplates": [] })),
252            "resources/read" => {
253                let uri = params
254                    .get("uri")
255                    .and_then(Value::as_str)
256                    .ok_or_else(|| (INVALID_PARAMS, "params.uri is required".to_string()))?;
257                match self.surface.read(uri) {
258                    Ok(c) => Ok(
259                        json!({ "contents": [{ "uri": c.uri, "mimeType": c.media_type, "text": c.text }] }),
260                    ),
261                    Err(SurfaceError::UnknownResource(u)) => {
262                        Err((RESOURCE_NOT_FOUND, format!("resource not found: {u}")))
263                    }
264                    Err(SurfaceError::Internal(e)) => Err((INTERNAL_ERROR, e)),
265                    Err(e) => Err((INVALID_PARAMS, e.to_string())),
266                }
267            }
268            "tools/list" => {
269                Ok(json!({ "tools": Value::Array(self.surface.tools_json().as_ref().clone()) }))
270            }
271            "tools/call" => {
272                let name = params
273                    .get("name")
274                    .and_then(Value::as_str)
275                    .ok_or_else(|| (INVALID_PARAMS, "params.name is required".to_string()))?;
276                let args = params
277                    .get("arguments")
278                    .cloned()
279                    .unwrap_or_else(|| json!({}));
280                if !args.is_object() {
281                    return Err((
282                        INVALID_PARAMS,
283                        "params.arguments must be an object".to_string(),
284                    ));
285                }
286                match self.surface.call(name, &args) {
287                    Ok(ToolOutcome::Ok(value)) => {
288                        // compact: a client that wants the data reads structuredContent
289                        let text = value.to_string();
290                        Ok(
291                            json!({ "content": [{ "type": "text", "text": text }], "structuredContent": value, "isError": false }),
292                        )
293                    }
294                    // the category travels beside the text, so a client branches on a word
295                    // and never parses a sentence; no structuredContent, because the
296                    // tool's output schema describes a success
297                    Ok(ToolOutcome::Refused { code, reason }) => Ok(json!({
298                        "content": [{ "type": "text", "text": reason }],
299                        "isError": true,
300                        "_meta": { "majordomus": { "error": { "code": code } } }
301                    })),
302                    Err(SurfaceError::Internal(e)) => Err((INTERNAL_ERROR, e)),
303                    Err(e) => Err((INVALID_PARAMS, e.to_string())),
304                }
305            }
306            other => Err((METHOD_NOT_FOUND, format!("method not found: {other}"))),
307        }
308    }

310    fn initialize(&mut self, params: &Value) -> Value {
311        let asked = params
312            .get("protocolVersion")
313            .and_then(Value::as_str)
314            .unwrap_or_default();
315        let version = PROTOCOL_VERSIONS
316            .iter()
317            .find(|v| **v == asked)
318            .copied()
319            .unwrap_or(PROTOCOL_VERSIONS[0]);
320        let client = ClientInfo::from_initialize(params);
321        if let Some(peer) = self.surface.peer() {
322            self.surface.context().peers.identify(peer, client.clone());
323            tracing::info!(peer = %peer, client = %client.name, version = %client.version, "peer initialized");
324        }
325        self.client = Some(client);
326        json!({
327            "protocolVersion": version,
328            "capabilities": {
329                "resources": { "subscribe": false, "listChanged": false },
330                "tools": { "listChanged": false }
331            },
332            "serverInfo": { "name": SERVER_NAME, "title": SERVER_TITLE, "version": self.version },
333            "instructions": self.instructions(),
334        })
335    }

337    /// The text a client reads after `initialize`: what is served, what of it writes, where
338    /// the shared server is, and who else is attached.
339    pub fn instructions(&self) -> String {
340        let index = self.surface.index();
341        let registry = self.surface.registry();
342        let summary = registry.summary();
343        let mut text = format!(
344            "{} This repository: {} object(s), {} capabilities ({} tools, {} resources), index state {}. Resources are majordomus://<kind>/<identity>; majordomus://repository carries the diagnostics; majordomus_capabilities lists every capability with its projections. {}",
345            crate::about::SUMMARY,
346            index.objects.len(),
347            summary.total,
348            summary.mcp_tools,
349            summary.mcp_resources,
350            match index.state { crate::index::State::Ok => "ok", crate::index::State::Degraded => "degraded" },
351            effects_sentence(&registry),
352        );
353        if let Some(url) = &self.endpoint {
354            text.push_str(&format!(
355                // "of this checkout", not "for this repository": a linked worktree is a
356                // checkout with a server of its own, and on 2026-09-11 this repository had
357                // seven of them. The sentence a client reads on every initialize was the
358                // one that told nine agents they had seen each other (ADR 0044).
359                " This session belongs to the shared server of this checkout at {url}: its home page at {url}/ lists every surface, Swagger UI is {url}/swagger, OpenAPI {url}/openapi.json, MCP over HTTP {url}/mcp."
360            ));
361        }
362        if let Some(peer) = self.surface.peer() {
363            let board = &self.surface.context().peers;
364            text.push_str(&format!(
365                // Only this checkout's board is named here: the gathered one costs a lease
366                // read and a probe per registered checkout, and `initialize` is on the path
367                // of every client's first millisecond. The words point at the capability
368                // that pays for the wide answer rather than paying for it here.
369                " You are peer {peer} on this checkout's board; attached here: {}. majordomus_peers lists every worker of this repository — this board and the board of every other checkout of it, each peer stamped with the worktree it is working in; call majordomus_announce with your intent and the paths you expect to touch so that the other clients (Claude, Codex, Gemini, ...) can avoid colliding with you.",
370                board.summary()
371            ));
372            // The nudge that matters is delivered here, on every initialize, which is also
373            // every reconnect. A session whose transport was re-established keeps its work
374            // and loses its place on the board, and nothing else in the protocol will ever
375            // mention it again: this repository spent three hours with a session invisible
376            // to eight others for exactly that reason. So the instructions say what is
377            // true of THIS caller rather than what is true in general.
378            let peers = board.list();
379            if !peers
380                .iter()
381                .any(|p| Some(&p.id) == self.surface.peer() && p.announcement.is_some())
382            {
383                text.push_str(" You have not announced anything. If you have worked in this repository before in another session, the board does not know it: announce now, before you start, and again if this connection is ever re-established — your peer id is a position on this board, handed out again on a reconnect, and never your identity.");
384            }
385            let silent = peers
386                .iter()
387                .filter(|p| {
388                    p.attached && p.announcement.is_none() && Some(&p.id) != self.surface.peer()
389                })
390                .count();
391            if silent > 0 {
392                text.push_str(&format!(
393                    " {silent} attached peer(s) have announced nothing, so the board understates who is here."
394                ));
395            }
396            let departed = peers.iter().filter(|p| !p.attached).count();
397            if departed > 0 {
398                text.push_str(&format!(
399                    " {departed} peer(s) on the board have gone but their announcements stand: what they claimed is still claimed until somebody says otherwise."
400                ));
401            }
402        }
403        text
404    }

407/// What the `initialize` instructions say about writing, read off the registry instead of
408/// stated by hand.
409///
410/// The sentence used to be the constant "Nothing here writes to the repository." It was true
411/// when it was written and false from [ADR 0040] onwards: `plan.transition` stamps a field
412/// into a tracked issue record and appends a ledger event, and it is exposed as an MCP tool.
413/// An agent that reads its instructions and believes them would then call a writer thinking
414/// it was reading — the one thing the instructions exist to prevent. So the sentence is
415/// derived: every capability whose MCP exposure is a tool and whose effect is
416/// [`Effect::RepositoryMutation`] is named, in the registry's own order, and only a registry
417/// that holds no such tool gets the old sentence back.
418///
419/// The wording itself is [`crate::about::writes`], the one sentence the tool uses about its
420/// own writers; this function only decides which tools it names.
421///
422/// The assertion is `the_instructions_never_claim_read_only_while_a_writer_is_exposed`.
423///
424/// [ADR 0040]: ../../../.ai/repo/adrs/0040-development-semantics-are-capabilities-of-one-runtime.md
425/// [`Effect::RepositoryMutation`]: crate::capability::model::Effect::RepositoryMutation
426pub(crate) fn effects_sentence(registry: &crate::capability::CapabilityRegistry) -> String {
427    let writers: Vec<&str> = registry
428        .iter()
429        .filter(|c| c.execution.effect == crate::capability::model::Effect::RepositoryMutation)
430        .filter_map(|c| c.exposure.mcp.as_ref()?.tool.as_deref())
431        .collect();
432    crate::about::writes(&writers)
433}

435pub(crate) fn resource_json(r: &super::surface::Resource) -> Value {
436    let mut v = json!({ "uri": r.uri, "name": r.name, "mimeType": r.media_type, "_meta": { "majordomus": r.meta } });
437    if let Some(t) = &r.title {
438        v["title"] = Value::String(t.clone());
439    }
440    if let Some(d) = &r.description {
441        v["description"] = Value::String(d.clone());
442    }
443    v
444}

446pub(crate) fn tool_json(t: &super::surface::Tool) -> Value {
447    json!({
448        "name": t.name, "title": t.title, "description": t.description,
449        "inputSchema": t.input_schema,
450        "outputSchema": t.output_schema,
451        "_meta": { "majordomus": { "id": t.id, "effect": t.effect } },
452        "annotations": {
453            "readOnlyHint": t.hints.read_only, "destructiveHint": t.hints.destructive,
454            "idempotentHint": t.hints.idempotent, "openWorldHint": t.hints.open_world
455        }
456    })
457}

459fn error(id: Value, code: i64, message: &str) -> Value {
460    // the one place a JSON-RPC error is made: it may echo a tool name or an argument
461    let message = crate::redaction::redact_secrets(message).text;
462    json!({ "jsonrpc": "2.0", "id": id, "error": { "code": code, "message": message } })
463}

469    /// The `initialize` instructions are the one thing every client reads before it does
470    /// anything, and for most of this server's life they ended with the constant "Nothing
471    /// here writes to the repository." That sentence stopped being true when the first
472    /// mutating capability was exposed as a tool, and nothing said so: a constant cannot go
473    /// stale loudly. So the sentence is derived, and this is the assertion that it stays
474    /// derived — if a tool that writes the repository is exposed and the instructions still
475    /// promise a read-only surface, this fails.
476    #[test]
477    fn the_instructions_never_claim_read_only_while_a_writer_is_exposed() {
478        let registry = crate::capability::CapabilityRegistry::builder()
479            .with_builtin(crate::capability::builtin::all())
480            .build()
481            .unwrap();
482        let writers: Vec<&str> = registry
483            .iter()
484            .filter(|c| c.execution.effect == crate::capability::model::Effect::RepositoryMutation)
485            .filter_map(|c| c.exposure.mcp.as_ref()?.tool.as_deref())
486            .collect();
487        let sentence = effects_sentence(&registry);
488
489        if writers.is_empty() {
490            assert_eq!(sentence, "Nothing here writes to the repository.");
491        } else {
492            assert!(
493                !sentence.contains("Nothing here writes"),
494                "{} mutating tool(s) are exposed ({}) and the instructions still tell every \
495                 client the surface writes nothing: {sentence}",
496                writers.len(),
497                writers.join(", ")
498            );
499            for w in &writers {
500                assert!(
501                    sentence.contains(w),
502                    "the instructions do not name the writing tool {w}: {sentence}"
503                );
504            }
505        }
506    }
507
508    /// A tool's annotations are what a client reads to decide whether to ask a person
509    /// first, and for most of this server's life three of the four were constants:
510    /// never destructive, always idempotent. The first tool that unlinked files was
511    /// announced as both. They are now the capability's classified hints, and this is the
512    /// assertion that they stay so: every tool of the real registry is rendered, and each
513    /// annotation is compared with the policy of the capability behind it.
514    #[test]
515    fn a_tools_annotations_are_its_capabilitys_hints_and_a_writer_is_never_announced_as_safe() {
516        let registry = crate::capability::CapabilityRegistry::builder()
517            .with_builtin(crate::capability::builtin::all())
518            .build()
519            .unwrap();
520        let mut writers = 0;
521        for c in registry.iter() {
522            let Some(tool) = super::super::surface::Tool::of(c) else {
523                continue;
524            };
525            let v = tool_json(&tool);
526            let a = &v["annotations"];
527            let h = c.execution.hints();
528            assert_eq!(a["readOnlyHint"], h.read_only, "{}", tool.name);
529            assert_eq!(a["destructiveHint"], h.destructive, "{}", tool.name);
530            assert_eq!(a["idempotentHint"], h.idempotent, "{}", tool.name);
531            assert_eq!(a["openWorldHint"], h.open_world, "{}", tool.name);
532            assert_eq!(
533                v["_meta"]["majordomus"]["effect"],
534                serde_json::to_value(c.execution.effect).unwrap(),
535                "{}",
536                tool.name
537            );
538            if c.execution.effect == crate::capability::model::Effect::RepositoryMutation {
539                writers += 1;
540                assert_eq!(
541                    a["readOnlyHint"], false,
542                    "{} writes the repository",
543                    tool.name
544                );
545                assert_eq!(
546                    a["destructiveHint"], true,
547                    "{} writes the repository",
548                    tool.name
549                );
550                assert_eq!(
551                    a["idempotentHint"], false,
552                    "{} writes the repository",
553                    tool.name
554                );
555            }
556            if a["readOnlyHint"] == true {
557                assert_eq!(c.execution.effect, crate::capability::model::Effect::Read);
558            }
559        }
560        assert!(
561            writers > 0,
562            "the registry exposes no writing tool, so this proved nothing about one"
563        );
564    }

566    /// The sentence over a registry this repository does not happen to hold, so that the
567    /// assertion above cannot pass merely because the real registry has a writer today: a
568    /// registry with nothing in it writes nothing, and says so.
569    #[test]
570    fn a_registry_with_no_writer_still_says_nothing_writes() {

572            .build()
573            .unwrap();
574        assert_eq!(
575            effects_sentence(&empty),
576            "Nothing here writes to the repository."
577        );
578    }
