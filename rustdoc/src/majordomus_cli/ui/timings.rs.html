1//! How long CI takes, for the CI/CD screen: `integration.timings`, measured from the recorded
2//! forge observation. Every check with its runs, failures and median, 90th percentile and
3//! longest duration, each median drawn as a bar against the slowest; and the open heads that
4//! waited longest for CI, each a link to its pull request on the PR Drain.
5//!
6//! Nothing is estimated: a run without a start or an end is counted beside the figures and
7//! never drawn as zero, and a checkout that recorded no observation says so.
8

16/// The source.
17pub const TIMINGS: Source = Source::of("integration.timings");
18
19/// How many of the longest-waiting heads are listed.
20const HEADS: usize = 10;
21
22/// Seconds as a reader says them: `42 s`, `7 min 05 s`, `1 h 02 min`.
23///
24/// ```text
25/// assert_eq!(duration(42), "42 s");
26/// assert_eq!(duration(425), "7 min 05 s");
27/// assert_eq!(duration(3720), "1 h 02 min");
28/// ```
29pub fn duration(s: u64) -> String {
30    match s {
31        0..=59 => format!("{s} s"),
32        60..=3599 => format!("{} min {:02} s", s / 60, s % 60),
33        _ => format!("{} h {:02} min", s / 3600, (s % 3600) / 60),
34    }
35}
36
37fn num(v: &Value, k: &str) -> Option<u64> {
38    v.get(k).and_then(Value::as_u64)
39}
40
41/// A duration the answer gave, or `unknown` when it gave none: never a zero it did not say.
42fn seconds(s: Option<u64>) -> String {
43    s.map_or_else(|| "unknown".into(), duration)
44}
45
46/// A count the answer gave, or `unknown`.
47fn count(n: Option<u64>) -> String {
48    n.map_or_else(|| "unknown".into(), |n| n.to_string())
49}
50
51/// A bar of `value` against `max`, drawn on the server in the given tone, with its value as
52/// text beside it so the bar is never the only way to read it.
53fn bar(value: Option<u64>, max: u64, tone: &str) -> El {
54    // ui-integrity: a width, not a count shown: an unknown value draws an empty bar, and the
55    // text beside every bar says unknown; anything measured is at least visible
56    let width = value
57        .and_then(|v| (v * 100).checked_div(max))
58        .map_or(0, |w| w.max(1));
59    el("span")
60        .class(format!("mj-tone--{tone}"))
61        .attr("aria-hidden", "true")
62        .raw(format!(
63            r#"<svg viewBox="0 0 100 6" width="120" height="8" preserveAspectRatio="none" focusable="false"><rect x="0" y="0" width="100" height="6" rx="3" fill="currentColor" opacity="0.15"/><rect x="0" y="0" width="{width}" height="6" rx="3" fill="currentColor"/></svg>"#
64        ))
65}

67/// The panel.
68pub fn panel_of(timings: &Answer) -> El {
69    let body = unanswered(timings).unwrap_or_else(|| {
70        let v = timings.ok();
71        if v.and_then(|v| v.get("observed")) != Some(&Value::Bool(true)) {
72            return nothing(&format!(
73                "No forge observation is recorded here, so no run is measured: {}",
74                v.and_then(|v| v.get("reason"))
75                    .and_then(Value::as_str)
76                    .unwrap_or("no reason was given")
77            ));
78        }
79        let checks = list(timings, "/timings/checks");
80        if checks.is_empty() {
81            return nothing(
82                "No check run on an open pull request has both a start and an end in the observation; run majordomus prs refresh to record them.",
83            );
84        }
85        // the slowest median the answer gave; a check without one draws no bar
86        let slowest = checks
87            .iter()
88            .filter_map(|c| num(c, "median_seconds"))
89            .max()
90            .unwrap_or(1);
91        let check_rows = checks
92            .iter()
93            .map(|c| {
94                let median = num(c, "median_seconds");
95                let failed = num(c, "failed");
96                let failing = failed.is_some_and(|f| f > 0);
97                el("tr")
98                    .child(el("td").class("mj-ui-mono").text(text(c, "name")))
99                    .child(
100                        el("td")
101                            .child(bar(median, slowest, if failing { "rose" } else { "blue" }))
102                            .child(el("span").class("mj-ui-mono").text(format!(" {}", seconds(median)))),
103                    )
104                    .child(el("td").class("mj-ui-mono").text(seconds(num(c, "p90_seconds"))))
105                    .child(el("td").class("mj-ui-mono").text(seconds(num(c, "slowest_seconds"))))
106                    .child(el("td").class("mj-ui-mono").text(format!(
107                        "{} · {} failed",
108                        count(num(c, "runs")),
109                        count(failed)
110                    )))
111            })
112            .collect::<Vec<_>>();
113        let heads = list(timings, "/timings/heads");
114        let longest = heads
115            .iter()
116            .filter_map(|h| num(h, "wall_seconds"))
117            .max()
118            .unwrap_or(1);
119        let head_rows = heads
120            .iter()
121            .take(HEADS)
122            .map(|h| {
123                el("tr")
124                    .child(el("td").child(match num(h, "number") {
125                        Some(n) => el("a")
126                            .attr("href", format!("{}?pr={n}#pr", super::prs::ROUTE))
127                            .text(format!("#{n}")),
128                        None => el("span").class("mj-ui-dim").text("unnumbered"),
129                    }))
130                    .child(
131                        el("td")
132                            .child(bar(
133                                num(h, "wall_seconds"),
134                                longest,
135                                if h.get("failed") == Some(&Value::Bool(true)) { "rose" } else { "violet" },
136                            ))
137                            .child(el("span").class("mj-ui-mono").text(format!(" {}", seconds(num(h, "wall_seconds"))))),
138                    )
139                    .child(el("td").class("mj-ui-mono mj-ui-dim").text(format!(
140                        "{} · {}",
141                        text(h, "slowest_check"),
142                        seconds(num(h, "slowest_seconds"))
143                    )))
144            })
145            .collect::<Vec<_>>();
146        el("div")
147            .class("mj-ui-stack")
148            .child(strip(vec![
149                kpi("Runs measured", &timings.read("/timings/measured_runs")),
150                kpi("Not measured", &timings.read("/timings/unmeasured_runs")),
151                kpi("Observed", &timings.read("/timings/observed_at")),
152            ]))
153            .child(table(
154                &["Check", "Median", "90th percentile", "Longest", "Runs"],
155                check_rows,
156                "No check was measured.",
157            ))
158            .child(el("h3").class("mj-ui-panel-title").text("Longest waits"))
159            .child(table(
160                &["Pull request", "Wall time", "Slowest check"],
161                head_rows,
162                "No open head has a measured check.",
163            ))
164    });
165    panel(
166        "Build timings",
167        "How long each check takes on the open pull requests, measured from when each run started to when it completed.",
168        body,
169        &[timings],
170    )
171}

173#[cfg(test)]
174mod tests {
175    use super::*;
176    use crate::ui::source::Exposure;
177    use std::time::Duration;

179    fn answer(value: Value) -> Answer {
180        Answer {
181            source: TIMINGS,
182            route: Some("GET /api/v1/pull-requests/timings".into()),
183            exposure: Exposure::default(),
184            value: Ok(value),
185            took: Duration::from_millis(1),
186        }
187    }
188
189    /// A check or a head the answer gives without its numbers is drawn as unknown and
190    /// unnumbered, never as a zero it did not say.
191    #[test]
192    fn what_the_answer_leaves_out_is_unknown_never_zero() {
193        let html = panel_of(&answer(serde_json::json!({ "observed": true, "timings": {
194            "observed_at": "2026-10-11T12:00:00Z",
195            "checks": [{ "name": "site" }],
196            "heads": [{ "head_sha": "a", "slowest_check": "site", "failed": false }]
197        }})))
198        .render();
199        assert!(html.contains("unknown"), "{html}");
200        assert!(html.contains("unnumbered"), "{html}");
201        assert!(!html.contains(">0 s<"), "{html}");
202    }
203
204    #[test]
205    fn a_duration_reads_as_a_person_says_it() {
206        assert_eq!(duration(42), "42 s");
207        assert_eq!(duration(425), "7 min 05 s");
208        assert_eq!(duration(3720), "1 h 02 min");
209    }
210
211    #[test]
212    fn the_checks_and_the_longest_waits_are_drawn_with_their_numbers() {
213        let html = panel_of(&answer(serde_json::json!({ "observed": true, "timings": {
214            "observed_at": "2026-10-11T12:00:00Z", "measured_runs": 3, "unmeasured_runs": 1,
215            "checks": [
216                { "name": "site", "runs": 2, "passed": 2, "failed": 0, "median_seconds": 2400, "p90_seconds": 2460, "slowest_seconds": 2460 },
217                { "name": "rust", "runs": 1, "passed": 0, "failed": 1, "median_seconds": 600, "p90_seconds": 600, "slowest_seconds": 600 }
218            ],
219            "heads": [{ "number": 952, "head_sha": "a", "started_at": "x", "completed_at": "y",
220                        "wall_seconds": 2460, "slowest_check": "site", "slowest_seconds": 2460, "failed": false }]
221        }})))
222        .render();
223        for want in [
224            "site",
225            "40 min 00 s",
226            "41 min 00 s",
227            "1 failed",
228            "/ui/prs?pr=952#pr",
229            "<svg",
230        ] {
231            assert!(html.contains(want), "no {want}: {html}");
232        }
233        assert!(
234            html.contains("mj-tone--rose"),
235            "a check that failed is drawn as one"
236        );
237    }
238
239    #[test]
240    fn no_observation_and_no_measured_run_say_so_and_draw_nothing() {
241        let none = panel_of(&answer(
242            serde_json::json!({ "observed": false, "reason": "no forge observation is recorded" }),
243        ))
244        .render();
245        assert!(none.contains("so no run is measured: no forge observation is recorded"));
246        assert!(!none.contains("<svg"));
247        let empty = panel_of(&answer(serde_json::json!({ "observed": true, "timings": {
248            "observed_at": "t", "measured_runs": 0, "unmeasured_runs": 4, "checks": [], "heads": [] }}))).render();
249        assert!(empty.contains("has both a start and an end"));
250        assert!(!empty.contains("<svg"));
251    }

253    /// A head whose checks failed is drawn in the failing tone even when no check's median
254    /// is.
255    #[test]
256    fn a_failed_head_is_drawn_as_one() {

262                        "slowest_check": "rust", "slowest_seconds": 600, "failed": true }]
263        }})))
264        .render();
265        assert!(html.contains("/ui/prs?pr=960#pr"), "{html}");
266        assert!(html.contains("mj-tone--rose"), "{html}");
267    }
