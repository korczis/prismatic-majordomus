1//! `majordomus skills`: every skill as a proven capability, through the registry's own
2//! `skills.*` capabilities.
3//!
4//! This file renders; it decides nothing. Every answer is the output of the capability the
5//! command line exposure names, run through the one executor, so the command line, the HTTP
6//! routes and the MCP tools cannot answer the same question differently. The one verdict,
7//! `verify`, is read out of that output and turned into exit 10.
8//!
9//! The text rendering keeps the four facts apart. `proven` and `inputs_unchanged` print
10//! differently, and an orphan prints which half it lacks, because a column that showed a tick
11//! for "a test names it" would be the badge whose derivation cannot be inspected.
12

22/// The exit code when `verify` finds a failure.
23pub const EXIT_INVALID: u8 = 10;
24
25/// Run `majordomus skills`.
26pub fn run(args: SkillsArgs) -> Result<u8> {
27    let app = App::load(&args.repo)?;
28    let stdout = std::io::stdout();
29    answer(&app.context, &args.command, args.format, &mut stdout.lock())
30}

32/// Answer one `skills` subcommand from `ctx`, writing it to `out` in `format`.
33fn answer<W: Write>(
34    ctx: &Context,
35    command: &SkillsCommand,
36    format: OutputFormat,
37    out: &mut W,
38) -> Result<u8> {
39    match command {
40        SkillsCommand::Status => {
41            let v = call(ctx, &["skills", "status"], json!({}))?;
42            emit(out, format, &v, status_text)?;
43            Ok(0)
44        }
45        SkillsCommand::Explain { id } => {
46            let v = call(ctx, &["skills", "explain"], json!({ "id": id }))?;
47            emit(out, format, &v, explain_text)?;
48            Ok(0)
49        }
50        SkillsCommand::Verify => {
51            let v = call(ctx, &["skills", "verify"], json!({}))?;
52            emit(out, format, &v, verify_text)?;
53            Ok(if v["valid"].as_bool() == Some(true) {
54                0
55            } else {
56                EXIT_INVALID
57            })
58        }
59    }
60}

62fn call(ctx: &Context, path: &[&str], input: Value) -> Result<Value> {
63    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
64    let id = ctx
65        .registry
66        .by_cli(&words)
67        .map(|c| c.id.to_string())
68        .ok_or_else(|| Error::Protocol {
69            reason: format!(
70                "no capability is exposed as `majordomus {}`",
71                path.join(" ")
72            ),
73        })?;
74    ctx.execute(&id, input).map_err(|e| match e {
75        CapabilityError::NotFound(reason) => Error::NotFound { reason },
76        other => Error::Protocol {
77            reason: other.to_string(),
78        },
79    })
80}

82fn emit<W: Write>(
83    out: &mut W,
84    format: OutputFormat,
85    v: &Value,
86    text: impl Fn(&Value) -> String,
87) -> Result<()> {
88    let body = match format {
89        // `{:#}` is serde_json's own pretty writer, the one `to_string_pretty` runs, so the
90        // bytes are the same; serialising a `Value` cannot fail, so there is no fallback
91        OutputFormat::Json => format!("{v:#}"),
92        OutputFormat::Text => text(v),
93    };
94    writeln!(out, "{body}").map_err(Error::Transport)
95}

97fn s<'a>(v: &'a Value, k: &str) -> &'a str {
98    v[k].as_str().unwrap_or("")
99}
100
101fn yes(b: &Value) -> &'static str {
102    if b.as_bool() == Some(true) {
103        "yes"
104    } else {
105        "no"
106    }
107}

109fn status_text(v: &Value) -> String {
110    let skills = v["skills"].as_array().cloned().unwrap_or_default();
111    let width = skills
112        .iter()
113        .map(|i| s(i, "id").len())
114        .max()
115        .unwrap_or(2)
116        .max(2);
117    let mut out = vec![format!(
118        "{:<width$}  {:<10}  {:<12}  {:<16}  {:<10}  {:<8}  USED",
119        "ID", "STATUS", "STANDING", "TESTED", "DOCUMENTED", "ENFORCED"
120    )];
121    for i in &skills {
122        out.push(format!(
123            "{:<width$}  {:<10}  {:<12}  {:<16}  {:<10}  {:<8}  {}",
124            s(i, "id"),
125            s(i, "status"),
126            s(i, "standing"),
127            i["tested"]["state"].as_str().unwrap_or(""),
128            yes(&i["documented"]["documented"]),
129            yes(&i["enforced"]["enforced"]),
130            yes(&i["used"]["used"]),
131        ));
132    }
133    out.push(String::new());
134    out.push(format!("{} skill(s)", v["count"]));
135    out.join("\n")
136}

138fn explain_text(v: &Value) -> String {
139    let mut out = vec![
140        format!("{}  {}", s(v, "id"), s(v, "title")),
141        String::new(),
142        format!("  {}", s(v, "description")),
143        String::new(),
144        format!(
145            "  status {}   version {}   standing {}   source {}",
146            s(v, "status"),
147            v["version"],
148            s(v, "standing"),
149            s(v, "path")
150        ),
151    ];
152    if let Some(p) = v["provenance"].as_object() {
153        out.push(format!(
154            "  provenance  {} {} {}",
155            p.get("origin").and_then(Value::as_str).unwrap_or(""),
156            p.get("ledger").and_then(Value::as_str).unwrap_or(""),
157            p.get("decision").and_then(Value::as_str).unwrap_or("")
158        ));
159    }
160    out.push(format!(
161        "  tested      {}",
162        v["tested"]["state"].as_str().unwrap_or("")
163    ));
164    for t in v["tested"]["tests"].as_array().into_iter().flatten() {
165        out.push(format!(
166            "    test      {}  {}{}",
167            s(t, "path"),
168            s(t, "state"),
169            t["reproduce"]
170                .as_str()
171                .map(|r| format!("  [reproduce: {r}]"))
172                .unwrap_or_default()
173        ));
174    }
175    out.push(format!(
176        "  documented  {}  {}",
177        yes(&v["documented"]["documented"]),
178        s(&v["documented"], "page")
179    ));
180    let gates: Vec<&str> = v["enforced"]["gates"]
181        .as_array()
182        .into_iter()
183        .flatten()
184        .filter_map(Value::as_str)
185        .collect();
186    out.push(format!(
187        "  enforced    {}  doctrine {}  gates {}",
188        yes(&v["enforced"]["enforced"]),
189        v["enforced"]["doctrine"].as_str().unwrap_or("-"),
190        if gates.is_empty() {
191            "-".to_string()
192        } else {
193            gates.join(", ")
194        }
195    ));
196    out.push(format!("  used        {}", yes(&v["used"]["used"])));
197    for i in v["used"]["invocations"].as_array().into_iter().flatten() {
198        out.push(format!(
199            "    invoked   {}:{}  {}",
200            s(i, "path"),
201            i["line"],
202            s(i, "reference")
203        ));
204    }
205    findings_text(&mut out, &v["findings"]);
206    out.join("\n")
207}
208
209fn findings_text(out: &mut Vec<String>, findings: &Value) {
210    for f in findings.as_array().into_iter().flatten() {
211        out.push(format!(
212            "{} {}  {}: {}  [reproduce: {}]",
213            s(f, "level").to_uppercase(),
214            s(f, "code"),
215            s(f, "subject"),
216            s(f, "message"),
217            s(f, "reproduce"),
218        ));
219    }
220}
221
222fn verify_text(v: &Value) -> String {
223    let mut out = Vec::new();
224    findings_text(&mut out, &v["findings"]);
225    out.push(format!(
226        "{} skill(s), {} failure(s), {} warning(s): {}",
227        v["skills"],
228        v["failures"],
229        v["warnings"],
230        if v["valid"].as_bool() == Some(true) {
231            "valid"
232        } else {
233            "invalid"
234        }
235    ));
236    out.join("\n")
237}

248    /// A sink that refuses every write, the way a closed pipe does.
249    struct Closed;
250
251    impl Write for Closed {
252        fn write(&mut self, _: &[u8]) -> std::io::Result<usize> {
253            Err(std::io::Error::new(
254                std::io::ErrorKind::BrokenPipe,
255                "closed",
256            ))
257        }
258        fn flush(&mut self) -> std::io::Result<()> {
259            Ok(())
260        }
261    }

376    /// Each verb reaches its capability and prints what it answered; the verdict is the
377    /// exit code; a missing skill is not found; a sink that refuses the answer is an error.
378    #[test]
379    fn each_verb_answers_through_its_capability_and_a_closed_sink_is_an_error() {
380        let repo = with_skill();
381        let ctx = repo.context().unwrap();
382        let status = SkillsCommand::Status;
383        let explain = SkillsCommand::Explain { id: "alpha".into() };
384
385        let mut out = Vec::new();
386        assert_eq!(
387            answer(&ctx, &status, OutputFormat::Json, &mut out).unwrap(),
388            0
389        );
390        let v: Value = serde_json::from_slice(&out).unwrap();
391        assert_eq!(v["count"], 1);
392        assert_eq!(
393            String::from_utf8(out).unwrap(),
394            format!("{}\n", serde_json::to_string_pretty(&v).unwrap())
395        );
396
397        let mut out = Vec::new();
398        assert_eq!(
399            answer(&ctx, &explain, OutputFormat::Text, &mut out).unwrap(),
400            0
401        );
402        assert!(String::from_utf8(out)
403            .unwrap()
404            .starts_with("alpha  Skill alpha\n"));
405
406        // nothing tests or invokes alpha, so it is an orphan and the verdict refuses
407        let mut out = Vec::new();
408        assert_eq!(
409            answer(&ctx, &SkillsCommand::Verify, OutputFormat::Text, &mut out).unwrap(),
410            EXIT_INVALID
411        );
412        assert!(String::from_utf8(out).unwrap().ends_with(": invalid\n"));
413
414        match answer(
415            &ctx,
416            &SkillsCommand::Explain {
417                id: "absent".into(),
418            },
419            OutputFormat::Text,
420            &mut Vec::new(),
421        ) {
422            Err(e @ Error::NotFound { .. }) => assert_eq!(e.exit_code(), 12),
423            other => panic!("an absent skill answered {other:?}"),
424        }
425
426        for command in [&status, &explain, &SkillsCommand::Verify] {
427            assert!(
428                matches!(
429                    answer(&ctx, command, OutputFormat::Text, &mut Closed),
430                    Err(Error::Transport(_))
431                ),
432                "{command:?}"
433            );
434        }
435    }

437    /// A capability that refuses is a protocol error carrying its message, for every verb; a
438    /// registry exposing no `skills` command is named as such; no repository fails first.
439    #[test]
440    fn a_refusal_an_unexposed_verb_and_no_repository_are_each_an_error() {
441        let repo = with_skill();
442        let ledger = repo.root().join(crate::evidence::LEDGER_PATH);
443        std::fs::create_dir_all(ledger.parent().unwrap()).unwrap();
444        std::fs::write(&ledger, "not a ledger").unwrap();
445        let ctx = repo.context().unwrap();
446        for command in [SkillsCommand::Status, SkillsCommand::Verify] {
447            match answer(&ctx, &command, OutputFormat::Json, &mut Vec::new()) {
448                Err(Error::Protocol { reason }) => {
449                    assert!(
450                        reason.contains("not a ledger this version can read"),
451                        "{reason}"
452                    )
453                }
454                other => panic!("{command:?} answered over an unreadable ledger: {other:?}"),
455            }
456        }
457
458        let index = repo.index().unwrap();
459        let registry = CapabilityRegistry::builder()
460            .with_modules(
461                builtin::modules()
462                    .into_iter()
463                    .filter(|m| m.id.as_str() != "skills")
464                    .collect(),
465            )
466            .build()
467            .unwrap();
468        let bare = Context::new(Arc::new(index), Arc::new(registry));
469        match answer(
470            &bare,
471            &SkillsCommand::Status,
472            OutputFormat::Json,
473            &mut Vec::new(),
474        ) {
475            Err(Error::Protocol { reason }) => {
476                assert_eq!(
477                    reason,
478                    "no capability is exposed as `majordomus skills status`"
479                )
480            }
481            other => panic!("a registry with no skills answered {other:?}"),
482        }
483
484        let nowhere = tempfile::tempdir().unwrap();
485        let args = SkillsArgs {
486            repo: RepoArgs {
487                repo: Some(nowhere.path().to_path_buf()),
488                discovery: DiscoveryMode::Vcs,
489                strict: false,
490                share: None,
491            },
492            command: SkillsCommand::Status,
493            format: OutputFormat::Json,
494        };
495        assert!(run(args).is_err());
496    }
