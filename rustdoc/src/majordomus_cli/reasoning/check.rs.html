1//! `reasoning.check`: the executable half of the rules `project.advisors-are-optional`
2//! and `project.review-is-recorded-not-claimed`. Each check reads files and says what it
3//! found; none of them asks an advisor anything.
4//!
5//! * `catalogue` — the advisor catalogue resolves against the provider table and the
6//!   model catalogue (one inventory, referenced, never restated);
7//! * `adapters` — every adapter the catalogue names has its transport module, so the Node
8//!   layer is derived from the catalogue rather than keeping a list of its own;
9//! * `coupling` — the provider-independent code (this module, its capability, its CLI
10//!   rendering, its Cockpit page, the transport driver) names no advisor, adapter,
11//!   executable, provider, vendor or model of the catalogue as a string literal;
12//! * `ci` — no CI workflow or gate names a model vendor's credential variable, so no
13//!   required gate can depend on a live model;
14//! * `claims` — no document asserts a claim the catalogue forbids ("consensus
15//!   guarantees ...") or that an advisor is required;
16//! * `records` — every stored record is readable, every consultation names an advisor
17//!   its plan selected and one this repository knows, every conclusion's computed review
18//!   matches the consultations it cites.
19//!
20//! The lifecycle: load the catalogue, its references and the store, then [`run`] every
21//! check over a repository root; an empty list is a clean repository.
22//!
23//! ```
24//! use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
25//! use majordomus_cli::reasoning::check::run;
26//! use majordomus_cli::reasoning::store::Store;
27//!
28//! let root = tempfile::tempdir().unwrap();
29//! let catalogue = AdvisorCatalogue {
30//!     forbidden_claims: vec!["consensus guarantees correctness".into()],
31//!     ..Default::default()
32//! };
33//! let loaded = Store::at(root.path()).load();
34//! assert!(run(root.path(), &catalogue, &References::default(), &loaded).is_empty());
35//!
36//! std::fs::write(root.path().join("README.md"), "Consensus guarantees correctness.\n").unwrap();
37//! let findings = run(root.path(), &catalogue, &References::default(), &loaded);
38//! assert_eq!(findings.len(), 1);
39//! assert_eq!((findings[0].check.as_str(), findings[0].subject.as_str()), ("claims", "README.md"));
40//! ```

51/// One finding of a reasoning check: which check, where, and what is wrong.
52///
53/// `check` is one of [`CHECKS`]; `subject` is a repository-relative path, a record id or
54/// an advisor; `message` is one sentence saying what the rule refuses. Findings are
55/// reported, never repaired.
56///
57/// ```
58/// use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
59/// use majordomus_cli::reasoning::check::{run, ReasoningFinding, CHECKS};
60/// use majordomus_cli::reasoning::store::Loaded;
61///
62/// let root = tempfile::tempdir().unwrap();
63/// let loaded = Loaded { records: vec![], unreadable: vec!["t/broken.json".into()] };
64/// let found = run(root.path(), &AdvisorCatalogue::default(), &References::default(), &loaded);
65/// let expected = ReasoningFinding {
66///     check: "records".into(),
67///     subject: "t/broken.json".into(),
68///     message: "a reasoning record that does not parse".into(),
69/// };
70/// assert_eq!(found, [expected]);
71/// assert!(CHECKS.contains(&found[0].check.as_str()));
72/// ```
73#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
74pub struct ReasoningFinding {
75    /// The check that found it.
76    pub check: String,
77    /// Where: a file, a record, an advisor.
78    pub subject: String,
79    /// What is wrong.
80    pub message: String,
81}

96fn files_under(root: &Path, rel: &str, ext: &[&str]) -> Vec<std::path::PathBuf> {
97    let path = root.join(rel);
98    // ordered by construction: a directory listing has no order of its own
99    let mut out = std::collections::BTreeSet::new();
100    if path.is_file() {
101        out.insert(path);
102    } else if let Ok(entries) = std::fs::read_dir(&path) {
103        for e in entries.flatten() {
104            let p = e.path();
105            if p.is_dir() {
106                if let Ok(r) = p.strip_prefix(root) {
107                    out.extend(files_under(root, &r.display().to_string(), ext));
108                }
109            } else if p
110                .extension()
111                .and_then(|x| x.to_str())
112                .is_some_and(|x| ext.contains(&x))
113            {
114                out.insert(p);
115            }
116        }
117    }
118    out.into_iter().collect()
119}

121/// The names the independent code must not spell.
122///
123/// Every advisor's id, adapter, executable, provider, vendor and model, and the peer
124/// adapter when peers are declared: each once, in canonical order. These are the
125/// literals the `coupling` check looks for.
126///
127/// ```
128/// use majordomus_cli::reasoning::catalogue::AdvisorCatalogue;
129/// use majordomus_cli::reasoning::check::catalogue_names;
130///
131/// let catalogue: AdvisorCatalogue = majordomus_cli::metadata::yaml::parse_into(
132///     "version: 1\nadvisors:\n  - id: alpha\n    title: Alpha\n    transport: cli\n    adapter: alpha-cli\n    executable: alpha\n",
133/// )
134/// .unwrap();
135/// assert_eq!(catalogue_names(&catalogue), ["alpha", "alpha-cli"]);
136/// ```
137pub fn catalogue_names(catalogue: &AdvisorCatalogue) -> Vec<String> {
138    let mut names = Vec::new();
139    for a in &catalogue.advisors {
140        names.push(a.id.clone());
141        names.push(a.adapter.clone());
142        names.extend(a.executable.clone());
143        names.extend(a.provider.clone());
144        names.extend(a.vendor.clone());
145        names.extend(a.model.clone());
146    }
147    if let Some(p) = &catalogue.peers {
148        names.push(p.adapter.clone());
149    }
150    crate::order::canonical(&mut names);
151    names.dedup();
152    names
153}

155/// String literals of a source naming any of `names`: `"name"` or `'name'`, exactly.
156///
157/// A name inside prose, a path or a longer literal is not a hit: only a literal that is
158/// the name and nothing else couples code to an advisor. Each name is reported once, in
159/// the order of `names`.
160///
161/// ```
162/// use majordomus_cli::reasoning::check::literals_naming;
163///
164/// let names = vec!["alpha".to_string(), "beta".to_string()];
165/// let source = "let a = \"alpha\"; let b = 'alpha'; // the beta advisor\nimport('./beta.mjs')";
166/// assert_eq!(literals_naming(source, &names), ["alpha"]);
167/// ```
168pub fn literals_naming(text: &str, names: &[String]) -> Vec<String> {
169    let mut hits = Vec::new();
170    for n in names {
171        for quote in ['"', '\''] {
172            let needle = format!("{quote}{n}{quote}");
173            if text.contains(&needle) && !hits.contains(n) {
174                hits.push(n.clone());
175            }
176        }
177    }
178    hits
179}

181/// Run every check over the repository at `root`, in the order of [`CHECKS`].
182///
183/// It reads the catalogue, the adapter modules, the provider-independent sources, the CI
184/// workflows, the documentation and the loaded records, and asks no advisor anything.
185/// The findings come back in check order; an empty list means every rule holds.
186///
187/// ```
188/// use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
189/// use majordomus_cli::reasoning::check::run;
190/// use majordomus_cli::reasoning::record::ReasoningRecord;
191/// use majordomus_cli::reasoning::store::Loaded;
192///
193/// let record: ReasoningRecord = serde_json::from_value(serde_json::json!({
194///     "schema": "majordomus/reasoning-record", "version": 1, "id": "consultation-1",
195///     "task": "t", "recorded_at": "2026-10-01T00:00:01Z",
196///     "body": {
197///         "kind": "consultation", "plan": "plan-0", "advisor": "alpha", "status": "timeout",
198///     },
199/// }))
200/// .unwrap();
201/// let loaded = Loaded { records: vec![record], unreadable: vec![] };
202/// let root = tempfile::tempdir().unwrap();
203/// let found = run(root.path(), &AdvisorCatalogue::default(), &References::default(), &loaded);
204/// let messages: Vec<&str> = found.iter().map(|f| f.message.as_str()).collect();
205/// assert_eq!(
206///     messages,
207///     [
208///         "names advisor 'alpha', which the catalogue does not declare",
209///         "cites plan plan-0, which is not recorded",
210///     ]
211/// );
212/// ```
213pub fn run(
214    root: &Path,
215    catalogue: &AdvisorCatalogue,
216    refs: &References,
217    loaded: &Loaded,
218) -> Vec<ReasoningFinding> {
219    let mut out = Vec::new();
220    let mut push = |check: &str, subject: String, message: String| {
221        out.push(ReasoningFinding {
222            check: check.into(),
223            subject,
224            message,
225        })
226    };
227
228    for d in catalogue.diagnostics(refs) {
229        push("catalogue", "share/advisors.yaml".into(), d);
230    }
231
232    if root.join(ADAPTER_DIR).is_dir() {
233        let mut adapters: Vec<&str> = catalogue
234            .advisors
235            .iter()
236            .map(|a| a.adapter.as_str())
237            .collect();
238        if let Some(p) = &catalogue.peers {
239            adapters.push(&p.adapter);
240        }
241        for adapter in adapters {
242            let module = format!("{ADAPTER_DIR}/{adapter}.mjs");
243            if !root.join(&module).is_file() {
244                push(
245                    "adapters",
246                    module.clone(),
247                    format!("the catalogue names adapter '{adapter}' and no transport module {module} implements it"),
248                );
249            }
250        }
251    }
252
253    let names = catalogue_names(catalogue);
254    for rel in INDEPENDENT_SOURCES {
255        for file in files_under(root, rel, &["rs", "mjs"]) {
256            let Ok(text) = std::fs::read_to_string(&file) else {
257                continue;
258            };
259            for n in literals_naming(&text, &names) {
260                push(
261                    "coupling",
262                    file.strip_prefix(root).unwrap_or(&file).display().to_string(),
263                    format!("provider-independent code names '{n}'; it must reach advisors through the catalogue"),
264                );
265            }
266        }
267    }
268
269    let credentials: Vec<&str> = refs
270        .models
271        .vendors
272        .iter()
273        .filter_map(|v| v.credential_env.as_deref())
274        .collect();
275    let mut ci_files = files_under(root, ".github/workflows", &["yml", "yaml"]);
276    ci_files.extend(files_under(root, ".ai/repo/ci/gates.yaml", &["yaml"]));
277    for file in ci_files {
278        let Ok(text) = std::fs::read_to_string(&file) else {
279            continue;
280        };
281        for var in &credentials {
282            if text.contains(var) {
283                push(
284                    "ci",
285                    file.strip_prefix(root)
286                        .unwrap_or(&file)
287                        .display()
288                        .to_string(),
289                    format!(
290                        "CI names the model credential {var}; no gate may depend on a live model"
291                    ),
292                );
293            }
294        }
295    }
296
297    let mut phrases: Vec<String> = catalogue
298        .forbidden_claims
299        .iter()
300        .map(|p| p.to_lowercase())
301        .collect();
302    for a in &catalogue.advisors {
303        let t = a.title.to_lowercase();
304        phrases.push(format!("requires {t}"));
305        phrases.push(format!("{t} is required"));
306    }
307    let mut docs = files_under(root, "docs", &["md"]);
308    docs.extend(files_under(root, "site/content", &["md"]));
309    docs.extend(files_under(root, ".ai/repo", &["md"]));
310    docs.push(root.join("README.md"));
311    for file in docs {
312        let Ok(text) = std::fs::read_to_string(&file) else {
313            continue;
314        };
315        let lower = text.to_lowercase();
316        for p in &phrases {
317            if lower.contains(p.as_str()) {
318                push(
319                    "claims",
320                    file.strip_prefix(root)
321                        .unwrap_or(&file)
322                        .display()
323                        .to_string(),
324                    format!("asserts '{p}', a claim the reasoning design refuses"),
325                );
326            }
327        }
328    }
329
330    for u in &loaded.unreadable {
331        push(
332            "records",
333            u.clone(),
334            "a reasoning record that does not parse".into(),
335        );
336    }
337    for r in &loaded.records {
338        match &r.body {
339            ReasoningBody::Consultation(c) => {
340                let known = catalogue.get(&c.advisor).is_some() || c.advisor.starts_with("peer:");
341                if !known {
342                    push(
343                        "records",
344                        r.id.clone(),
345                        format!(
346                            "names advisor '{}', which the catalogue does not declare",
347                            c.advisor
348                        ),
349                    );
350                }
351                match loaded.get(&c.plan).map(|p| &p.body) {
352                    Some(ReasoningBody::Plan(p)) => {
353                        if !p.plan.selected.iter().any(|s| s.advisor == c.advisor) {
354                            push(
355                                "records",
356                                r.id.clone(),
357                                format!(
358                                    "claims advisor '{}', which plan {} did not select",
359                                    c.advisor, c.plan
360                                ),
361                            );
362                        }
363                    }
364                    _ => push(
365                        "records",
366                        r.id.clone(),
367                        format!("cites plan {}, which is not recorded", c.plan),
368                    ),
369                }
370            }
371            ReasoningBody::Conclusion(k) => {
372                let mut computed: Vec<String> = Vec::new();
373                for id in &k.input.consultations {
374                    match loaded.get(id).map(|x| &x.body) {
375                        Some(ReasoningBody::Consultation(c))
376                            if c.status == ConsultationStatus::Completed =>
377                        {
378                            if !computed.contains(&c.advisor) {
379                                computed.push(c.advisor.clone());
380                            }
381                        }
382                        _ => push(
383                            "records",
384                            r.id.clone(),
385                            format!("cites {id}, which is not a completed consultation"),
386                        ),
387                    }
388                }
389                crate::order::canonical(&mut computed);
390                if computed != k.reviewed_by || k.independent_review_count != k.reviewed_by.len() {
391                    push(
392                        "records",
393                        r.id.clone(),
394                        format!(
395                            "claims review by [{}] ({}), and the consultations it cites show [{}]",
396                            k.reviewed_by.join(", "),
397                            k.independent_review_count,
398                            computed.join(", ")
399                        ),
400                    );
401                }
402            }
403            _ => {}
404        }
405    }
406    out
407}

423    #[test]
424    fn a_literal_is_a_name_and_a_word_inside_prose_is_not() {
425        let names = vec!["alpha".to_string()];
426        assert_eq!(literals_naming("let x = \"alpha\";", &names), ["alpha"]);
427        assert_eq!(
428            literals_naming("import('./alpha.mjs')", &names),
429            Vec::<String>::new()
430        );
431        assert_eq!(
432            literals_naming("// the alpha advisor", &names),
433            Vec::<String>::new()
434        );
435    }

437    /// The anti-coupling proof for this crate: the shipped catalogue's names appear in no
438    /// provider-independent source file as a literal.
439    #[test]
440    fn the_reasoning_core_names_no_advisor_of_the_shipped_catalogue() {
441        let root = Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
442        let share = root.join("share");
443        let catalogue = AdvisorCatalogue::load(&share).unwrap();
444        assert!(
445            !catalogue.advisors.is_empty(),
446            "the shipped catalogue declares advisors"
447        );
448        let names = catalogue_names(&catalogue);
449        let mut scanned = 0;
450        for rel in INDEPENDENT_SOURCES {
451            for file in files_under(&root, rel, &["rs", "mjs"]) {
452                scanned += 1;
453                let text = std::fs::read_to_string(&file).unwrap();
454                assert_eq!(
455                    literals_naming(&text, &names),
456                    Vec::<String>::new(),
457                    "{} names an advisor",
458                    file.display()
459                );
460            }
461        }
462        assert!(
463            scanned >= 7,
464            "the scan reached the module ({scanned} files)"
465        );
466    }
