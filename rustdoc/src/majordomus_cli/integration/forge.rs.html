1//! The forge adapter: the one place pull-request integration talks to the network.
2//!
3//! SECURITY.md declares it: `majordomus prs refresh` and `majordomus prs drain` — and
4//! nothing else — run the GitHub CLI (`gh`), `git fetch` and `git ls-remote` against this
5//! repository's own remote. The HTTP server, the MCP tools and the Cockpit never do: they render the last
6//! [`ForgeObservation`] recorded under `.ai/local/state/integration/`, with its age, so a
7//! page load can never reach the network. The observation is plain data, so every test
8//! builds one by hand and no test needs the forge.
9//!
10//! A required check the base binds to an app is only that app's check run, and `gh pr list`
11//! does not say which app wrote a check run. So when a context is bound, the refresh reads
12//! every open pull request's checks once more, from the forge's GraphQL rollup
13//! ([`WRITERS_QUERY`]), with the app of each ([`attributed_checks()`], [`attribute()`]). A
14//! base that binds nothing asks nothing more.
15//!
16//! Who may declare that one pull request replaces another is the forge's word too, and
17//! `gh pr list` does not carry it. So whenever a pull request is open, the refresh reads, with
18//! every open one, its author's association and the pull requests that mention it
19//! ([`DECLARATIONS_QUERY`], [`declarations()`], [`declare()`]): a declaration about an open pull
20//! request is read from that pull request's own cross-references, never from a bounded search
21//! of closed ones. One whose cross-references could not be read whole says so
22//! ([`CrossReferenceRead`]), and the classifier holds it and no other. A read the forge would
23//! not answer is not a hold: it fails the observation, and none is recorded from a part.
24

114/// Origin's branches from `git ls-remote --heads` output: name → tip. The module is private,
115/// so the examples here are text; the unit tests run the same assertions.
116///
117/// ```text
118/// use majordomus_cli::integration::forge::remote_heads_of;
119/// let heads = remote_heads_of("a1\trefs/heads/master\nb2\trefs/heads/feature/x\nc3\trefs/tags/v1\n");
120/// assert_eq!(heads.get("feature/x").map(String::as_str), Some("b2"));
121/// assert_eq!(heads.len(), 2, "a tag is not a branch");
122/// ```
123pub fn remote_heads_of(ls_remote: &str) -> BTreeMap<String, String> {
124    ls_remote
125        .lines()
126        .filter_map(|l| {
127            let (sha, name) = l.split_once('\t')?;
128            let name = name.trim().strip_prefix("refs/heads/")?;
129            (!sha.is_empty() && !name.is_empty()).then(|| (name.to_string(), sha.to_string()))
130        })
131        .collect()
132}

134/// The merged pull requests whose head branch origin still serves at the head that merged,
135/// from `gh pr list --state merged --json number,state,headRefName,headRefOid,isCrossRepository,mergedAt`.
136///
137/// A pull request counts only when the forge says it `MERGED` and when: it came from this
138/// repository (a fork's branch is not a branch here, whatever it is called), its branch is not
139/// the base, and origin's tip of that branch is exactly its merged head. One branch, one entry:
140/// when several merged pull requests left the same branch at the same head, the newest
141/// number names it. In branch order.
142///
143/// ```text
144/// use majordomus_cli::integration::forge::{merged_branches_of, remote_heads_of};
145/// let heads = remote_heads_of("aa\trefs/heads/fix/a\nbb\trefs/heads/fix/b\ncc\trefs/heads/master\n");
146/// let merged = serde_json::json!([
147///   {"number": 1, "state": "MERGED", "headRefName": "fix/a", "headRefOid": "aa", "isCrossRepository": false, "mergedAt": "t1"},
148///   {"number": 2, "state": "MERGED", "headRefName": "fix/b", "headRefOid": "b0", "isCrossRepository": false, "mergedAt": "t2"},
149///   {"number": 3, "state": "OPEN",   "headRefName": "fix/a", "headRefOid": "aa", "isCrossRepository": false}
150/// ]);
151/// let left = merged_branches_of(&merged, &heads, "master");
152/// assert_eq!(left.len(), 1, "fix/b moved after its merge, and an open one is not merged");
153/// assert_eq!((left[0].branch.as_str(), left[0].pr), ("fix/a", 1));
154/// ```
155pub fn merged_branches_of(
156    merged: &Value,
157    heads: &BTreeMap<String, String>,
158    base: &str,
159) -> Vec<MergedBranch> {
160    let mut by_branch: BTreeMap<String, MergedBranch> = BTreeMap::new();
161    for p in merged.as_array().into_iter().flatten() {
162        let s = |k: &str| p.get(k).and_then(Value::as_str).unwrap_or("");
163        let (Some(pr), "MERGED") = (p.get("number").and_then(Value::as_u64), s("state")) else {
164            continue;
165        };
166        let (branch, tip, merged_at) = (s("headRefName"), s("headRefOid"), s("mergedAt"));
167        let fork = p
168            .get("isCrossRepository")
169            .and_then(Value::as_bool)
170            .unwrap_or(true);
171        if fork || branch.is_empty() || branch == base || tip.is_empty() || merged_at.is_empty() {
172            continue;
173        }
174        if heads.get(branch).map(String::as_str) != Some(tip) {
175            continue;
176        }
177        let entry = MergedBranch {
178            branch: branch.to_string(),
179            tip: tip.to_string(),
180            pr,
181            merged_at: merged_at.to_string(),
182        };
183        match by_branch.get(branch) {
184            Some(seen) if seen.pr > pr => {}
185            _ => {
186                by_branch.insert(branch.to_string(), entry);
187            }
188        }
189    }
190    by_branch.into_values().collect()
191}

193/// A pull request that is no longer open, as the forge reported it.
194#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
195pub struct ResolvedPullRequest {
196    /// Whether the forge says it was merged, rather than closed unmerged. Evidence, and the
197    /// difference between a successor that released its hold (closed) and one merged somewhere
198    /// master does not contain. Never what makes a successor landed: git does.
199    pub merged: bool,
200    /// The commit its branch pointed at when it was merged or closed.
201    pub head_sha: String,
202    /// The body, for the supersessions it declares; never rendered.
203    #[serde(default)]
204    pub body: String,
205    /// Its merge commit, as the forge named it; empty when it has none or it was not read.
206    #[serde(default)]
207    pub merge_commit: String,
208    /// Its author's login; empty when unread. Evidence only.
209    #[serde(default)]
210    pub author: String,
211    /// The forge's word for what its author is to the repository; empty when unread.
212    #[serde(default)]
213    pub author_association: String,
214    /// Whether its head lives in a fork. A record that does not say is read as a fork.
215    #[serde(default = "unread_is_a_fork")]
216    pub cross_repository: bool,
217    /// The branch it asked to merge into; empty when unread. Evidence only.
218    #[serde(default)]
219    pub base_ref: String,
220    /// How many files the forge says it changes; zero when unread. A veto, never a proof: a
221    /// pull request that changes no file brought nothing to master, wherever its head now
222    /// points, so it is never a successor that landed.
223    #[serde(default)]
224    pub changed_files: u64,
225    /// Whether the reading said both where its head lives and how many files it changes.
226    /// The two defaults above are vetoes: read as facts, "a fork's" and "no file" each say a
227    /// successor closed unmerged brought nothing and release what it held. So a reading
228    /// that left either out is not one to decide by, and a record that does not say is not
229    /// whole.
230    #[serde(default)]
231    pub whole: bool,
232}
233
234/// What a resolved pull request that does not say where its head lives is read as: a fork,
235/// which declares nothing.
236fn unread_is_a_fork() -> bool {
237    true
238}

260/// One pull request that is no longer open, from the source of a cross-reference
261/// ([`DECLARATIONS_QUERY`]) or from `gh pr view --json` output: `None` for an open one, or one
262/// without a head. What a reading does not say fails closed: no merge commit, no author, no
263/// association, a head read as a fork's and no changed file, so it declares nothing and
264/// never landed. A reading that does not say where the head lives, or how many files it
265/// changes, is not `whole`, and a successor read so is unread: it releases nothing.
266///
267/// The module is private, so the example is text; the unit tests run the same assertions.
268///
269/// ```text
270/// use majordomus_cli::integration::forge::resolved_of;
271/// let viewed = serde_json::json!({"number": 7, "state": "MERGED", "headRefOid": "h7",
272///     "mergeCommit": {"oid": "m7"}});
273/// let (number, read) = resolved_of(&viewed).unwrap();
274/// assert_eq!((number, read.merged, read.merge_commit.as_str()), (7, true, "m7"));
275/// assert!(read.cross_repository && read.author_association.is_empty(), "unread declares nothing");
276/// assert_eq!(read.changed_files, 0, "and changed nothing");
277/// assert!(!read.whole, "which is a default, not a reading: as a successor it is unread");
278/// assert!(resolved_of(&serde_json::json!({"number": 7, "state": "OPEN", "headRefOid": "h7"})).is_none());
279/// ```
280pub fn resolved_of(v: &Value) -> Option<(u64, ResolvedPullRequest)> {
281    let number = v.get("number")?.as_u64()?;
282    let merged = match v.get("state")?.as_str()? {
283        "MERGED" => true,
284        "CLOSED" => false,
285        _ => return None,
286    };
287    let head_sha = v.get("headRefOid")?.as_str()?.to_string();
288    if head_sha.is_empty() {
289        return None;
290    }
291    let text = |pointer: &str| {
292        v.pointer(pointer)
293            .and_then(Value::as_str)
294            .unwrap_or("")
295            .to_string()
296    };
297    let placed = v.get("isCrossRepository").and_then(Value::as_bool);
298    let files = v.get("changedFiles").and_then(Value::as_u64);
299    Some((
300        number,
301        ResolvedPullRequest {
302            merged,
303            head_sha,
304            body: text("/body"),
305            merge_commit: text("/mergeCommit/oid"),
306            author: text("/author/login"),
307            author_association: text("/authorAssociation"),
308            cross_repository: placed.unwrap_or(true),
309            base_ref: text("/baseRefName"),
310            changed_files: files.unwrap_or(0),
311            whole: placed.is_some() && files.is_some(),
312        },
313    ))
314}

316/// The pull requests no longer open that a supersession or a dependency involving an open one
317/// names: closed or merged ones whose body says they supersede an open one (`closed`: the
318/// sources of the open ones' cross-references, as [`declare()`] returned them), and the
319/// successors and dependencies open bodies name that are not open, each read with `view`. One
320/// `view` cannot read is left out, and the classifier says it is unread.
321pub fn resolved_for(
322    open: &[PullRequestObservation],
323    closed: &Value,
324    mut view: impl FnMut(u64) -> Option<Value>,
325) -> BTreeMap<u64, ResolvedPullRequest> {
326    use super::classify::declared_supersessions;
327    let numbers: BTreeSet<u64> = open.iter().map(|p| p.number).collect();
328    let mut resolved: BTreeMap<u64, ResolvedPullRequest> = closed
329        .as_array()
330        .into_iter()
331        .flatten()
332        .filter_map(resolved_of)
333        .filter(|(n, r)| {
334            !numbers.contains(n)
335                && declared_supersessions(&r.body)
336                    .supersedes
337                    .iter()
338                    .any(|t| t != n && numbers.contains(t))
339        })
340        .collect();
341    let named: BTreeSet<u64> = open
342        .iter()
343        .flat_map(|p| {
344            declared_supersessions(&p.body)
345                .superseded_by
346                .into_iter()
347                .chain(super::classify::declared_dependencies(&p.body))
348        })
349        .filter(|m| !numbers.contains(m) && !resolved.contains_key(m))
350        .collect();
351    for m in named {
352        if let Some((n, r)) = view(m).as_ref().and_then(resolved_of) {
353            if n == m {
354                resolved.insert(n, r);
355            }
356        }
357    }
358    resolved
359}

377/// The GitHub CLI, run in the repository.
378pub struct GhForge<'a> {
379    /// The repository root.
380    pub root: &'a Path,
381}

383fn gh(root: &Path, args: &[&str]) -> Result<(bool, String, String), ForgeError> {
384    let out = Command::new("gh")
385        .current_dir(root)
386        .args(args)
387        .output()
388        .map_err(|e| ForgeError(format!("the GitHub CLI (gh) could not run: {e}")))?;
389    Ok((
390        out.status.success(),
391        String::from_utf8_lossy(&out.stdout).into_owned(),
392        String::from_utf8_lossy(&out.stderr).into_owned(),
393    ))
394}

396/// `gh`, asked again on a transient failure ([`super::retry`]): every call here is a read,
397/// so asking twice is harmless. A refusal is returned as it came, for the caller to read.
398fn gh_retrying(root: &Path, args: &[&str]) -> Result<(bool, String, String), ForgeError> {
399    super::retry::forge(|| {
400        let (ok, out, err) = gh(root, args).map_err(|e| e.0)?;
401        if !ok && super::retry::transient(&err) {
402            Err(format!("gh {} failed: {}", args.join(" "), err.trim()))
403        } else {
404            Ok((ok, out, err))
405        }
406    })
407    .map_err(ForgeError)
408}
409
410fn gh_json(root: &Path, args: &[&str]) -> Result<Value, ForgeError> {
411    let (ok, out, err) = gh_retrying(root, args)?;
412    if !ok {
413        return Err(ForgeError(format!(
414            "gh {} failed: {}",
415            args.join(" "),
416            err.trim()
417        )));
418    }
419    serde_json::from_str(&out).map_err(|e| ForgeError(format!("gh {}: {e}", args.join(" "))))
420}

422/// The state word of one rollup entry: a check run (`status`/`conclusion`) or a commit
423/// status context (`state`).
424pub fn check_state(entry: &Value) -> CheckRunState {
425    let s = |k: &str| {
426        entry
427            .get(k)
428            .and_then(Value::as_str)
429            .unwrap_or("")
430            .to_ascii_uppercase()
431    };
432    if entry.get("__typename").and_then(Value::as_str) == Some("StatusContext") {
433        return match s("state").as_str() {
434            "SUCCESS" => CheckRunState::Passed,
435            "PENDING" | "EXPECTED" => CheckRunState::Pending,
436            _ => CheckRunState::Failed,
437        };
438    }
439    if s("status") != "COMPLETED" {
440        return CheckRunState::Pending;
441    }
442    match s("conclusion").as_str() {
443        "SUCCESS" => CheckRunState::Passed,
444        "SKIPPED" | "NEUTRAL" => CheckRunState::Skipped,
445        _ => CheckRunState::Failed,
446    }
447}

449/// One check from a rollup entry: `gh pr list --json statusCheckRollup` output, or a node of
450/// the GraphQL rollup [`WRITERS_QUERY`] reads. The two have one shape, but for the app: only
451/// the GraphQL node carries `checkSuite.app.databaseId`, so a check run read from `gh pr list`
452/// names no app, and neither does a status context, which no app writes.
453///
454/// The module is private, so the example is text; the unit tests run the same assertions.
455///
456/// ```text
457/// use majordomus_cli::integration::forge::check_of;
458/// let run = serde_json::json!({"__typename": "CheckRun", "name": "ci", "status": "COMPLETED",
459///     "conclusion": "SUCCESS", "completedAt": "2026-09-01T00:05:00Z",
460///     "checkSuite": {"app": {"databaseId": 15368}}});
461/// assert_eq!(check_of(&run).app_id, Some(15368));
462/// let listed = serde_json::json!({"__typename": "CheckRun", "name": "ci", "status": "COMPLETED"});
463/// assert_eq!(check_of(&listed).app_id, None, "gh pr list does not say who wrote it");
464/// ```
465pub fn check_of(entry: &Value) -> CheckObservation {
466    let text = |k: &str| entry.get(k).and_then(Value::as_str).unwrap_or("");
467    let kind = if text("__typename") == "StatusContext" {
468        CheckKind::StatusContext
469    } else {
470        CheckKind::CheckRun
471    };
472    CheckObservation {
473        name: entry
474            .get("name")
475            .or_else(|| entry.get("context"))
476            .and_then(Value::as_str)
477            .unwrap_or("")
478            .to_string(),
479        state: check_state(entry),
480        kind,
481        // the app of the suite the run belongs to: the one writer a forge read names
482        app_id: entry
483            .pointer("/checkSuite/app/databaseId")
484            .and_then(Value::as_u64),
485        completed_at: match text("completedAt") {
486            // a check run's zero time means it has not completed
487            "" | "0001-01-01T00:00:00Z" => text("startedAt")
488                .trim_start_matches("0001-01-01T00:00:00Z")
489                .to_string(),
490            t => t.to_string(),
491        },
492    }
493}

495/// One pull request from `gh pr list --json` output.
496pub fn pull_request_of(v: &Value) -> Option<PullRequestObservation> {
497    let s = |k: &str| v.get(k).and_then(Value::as_str).unwrap_or("").to_string();
498    Some(PullRequestObservation {
499        number: v.get("number")?.as_u64()?,
500        title: s("title"),
501        author: v
502            .pointer("/author/login")
503            .and_then(Value::as_str)
504            .unwrap_or("")
505            .to_string(),
506        head_ref: s("headRefName"),
507        head_sha: s("headRefOid"),
508        base_ref: s("baseRefName"),
509        draft: v.get("isDraft").and_then(Value::as_bool).unwrap_or(false),
510        labels: v
511            .get("labels")
512            .and_then(Value::as_array)
513            .map(|a| {
514                a.iter()
515                    .filter_map(|l| l.get("name").and_then(Value::as_str).map(str::to_string))
516                    .collect()
517            })
518            .unwrap_or_default(),
519        created_at: s("createdAt"),
520        updated_at: s("updatedAt"),
521        body: s("body"),
522        checks: v
523            .get("statusCheckRollup")
524            .and_then(Value::as_array)
525            .map(|a| a.iter().map(check_of).collect())
526            .unwrap_or_default(),
527        review_decision: s("reviewDecision"),
528        auto_merge: v
529            .get("autoMergeRequest")
530            .map(|a| !a.is_null())
531            .unwrap_or(false),
532        cross_repository: v
533            .get("isCrossRepository")
534            .and_then(Value::as_bool)
535            .unwrap_or(false),
536        // `gh pr list` reports neither: the declarations read does ([`declare()`])
537        author_association: String::new(),
538        cross_references: CrossReferenceRead::Unread,
539        latest_reviews: v
540            .get("latestReviews")
541            .and_then(Value::as_array)
542            .map(|a| {
543                a.iter()
544                    .map(|r| ReviewObservation {
545                        author: r
546                            .pointer("/author/login")

558                            .unwrap_or("")

560                    })

562            })

564        review_requests: v

582/// Union two readings of the required checks: a context required by either is required,
583/// once for each app it is bound to. Two sources that bind one context to two apps require
584/// both apps' check runs, as the forge does; an unbound entry of a context is the same
585/// requirement as a bound one and is dropped beside it. In context order, then app order.
586fn union_checks(a: Vec<RequiredCheck>, b: Vec<RequiredCheck>) -> Vec<RequiredCheck> {
587    let all: BTreeSet<RequiredCheck> = a.into_iter().chain(b).collect();
588    let bound: BTreeSet<String> = all
589        .iter()
590        .filter(|c| c.app_id.is_some())
591        .map(|c| c.context.clone())
592        .collect();
593    all.into_iter()
594        .filter(|c| c.app_id.is_some() || !bound.contains(&c.context))
595        .collect()
596}

598/// Union two review requirements: the stricter of each.
599fn union_reviews(a: ReviewPolicy, b: ReviewPolicy) -> ReviewPolicy {
600    ReviewPolicy {
601        approvals: a.approvals.max(b.approvals),
602        code_owners: a.code_owners || b.code_owners,
603        dismiss_stale: a.dismiss_stale || b.dismiss_stale,
604    }
605}

607/// The app a required check is bound to, as the forge spelled it: `Some(None)` for a check
608/// bound to none (the field absent, `null`, or `-1`, the forge's word for any app),
609/// `Some(Some(id))` for a bound one, and `None` for a spelling that is neither — a string, a
610/// fraction, another negative number.
611///
612/// Unreadable is not unbound. Read as unbound, a binding the forge spelled another way would
613/// let any app's run of that name stand for the bound app's, which is the hole the binding
614/// exists to close; so a document that carries one is a requirement that was not read.
615fn bound_app(v: Option<&Value>) -> Option<Option<u64>> {
616    match v {
617        None | Some(Value::Null) => Some(None),
618        Some(Value::Number(n)) if n.as_i64() == Some(-1) => Some(None),
619        Some(Value::Number(n)) => n.as_u64().map(Some),
620        Some(_) => None,
621    }
622}

650/// The required checks and review requirement the rulesets that apply to a branch add,
651/// from `repos/{r}/rules/branches/{base}`: its `required_status_checks` rules (a context and
652/// the `integration_id` bound to it) and its `pull_request` rules.
653pub fn rules_of(v: &Value) -> (Vec<RequiredCheck>, ReviewPolicy) {
654    let mut checks = Vec::new();
655    let mut reviews = ReviewPolicy::default();
656    for rule in v.as_array().into_iter().flatten() {
657        let params = rule.get("parameters");
658        match rule.get("type").and_then(Value::as_str) {
659            Some("required_status_checks") => {
660                for c in params
661                    .and_then(|p| p.get("required_status_checks"))
662                    .and_then(Value::as_array)
663                    .into_iter()
664                    .flatten()
665                {
666                    if let Some(context) = c.get("context").and_then(Value::as_str) {
667                        checks.push(RequiredCheck {
668                            context: context.to_string(),
669                            app_id: bound_app(c.get("integration_id")).flatten(),
670                        });
671                    }
672                }
673            }
674            Some("pull_request") => {
675                let p = |k: &str| params.and_then(|p| p.get(k));
676                reviews = union_reviews(
677                    reviews,
678                    ReviewPolicy {
679                        approvals: p("required_approving_review_count")
680                            .and_then(Value::as_u64)
681                            .unwrap_or(0),
682                        code_owners: p("require_code_owner_review")
683                            .and_then(Value::as_bool)
684                            .unwrap_or(false),
685                        dismiss_stale: p("dismiss_stale_reviews_on_push")
686                            .and_then(Value::as_bool)
687                            .unwrap_or(false),
688                    },
689                );
690            }
691            _ => {}
692        }
693    }
694    (union_checks(checks, Vec::new()), reviews)
695}

715/// The required checks and review requirement from a branch-protection document. A
716/// context the protection lists under `contexts` and again under `checks` is one check,
717/// bound to the app `checks` names for it.
718pub fn protection_of(v: &Value) -> (Vec<RequiredCheck>, ReviewPolicy) {
719    let listed: Vec<RequiredCheck> = v
720        .pointer("/required_status_checks/contexts")
721        .and_then(Value::as_array)
722        .map(|a| {
723            a.iter()
724                .filter_map(|c| c.as_str().map(RequiredCheck::from))
725                .collect()
726        })
727        .unwrap_or_default();
728    let bound: Vec<RequiredCheck> = v
729        .pointer("/required_status_checks/checks")

733                .filter_map(|c| {

738                })
739                .collect()
740        })

742    let reviews = match v.get("required_pull_request_reviews") {

803/// One page of one pull request's rollup, as a writers answer gave it.
804#[derive(Debug, Clone, PartialEq, Eq)]
805pub struct RollupPage {
806    /// The head the rollup is of.
807    pub head: String,
808    /// The checks on this page, each with the app that wrote it where the forge named one.
809    pub checks: Vec<CheckObservation>,
810    /// Whether these are all of them: `false` when the forge says more contexts follow, and
811    /// when it listed contexts without saying whether more do.
812    pub complete: bool,
813    /// Where the next page of contexts starts, when the forge named a cursor.
814    pub cursor: Option<String>,
815}

817/// One pull request node of a writers answer, read as a [`RollupPage`]. A head without a
818/// rollup (`statusCheckRollup: null`) has no check, and that is all of them.
819///
820/// The module is private, so the example is text; the unit tests run the same assertions.
821///
822/// ```text
823/// use majordomus_cli::integration::forge::rollup_page_of;
824/// let node = serde_json::json!({"number": 1, "headRefOid": "h1", "statusCheckRollup": {"contexts": {
825///     "pageInfo": {"hasNextPage": true, "endCursor": "c1"},
826///     "nodes": [{"__typename": "CheckRun", "name": "ci", "status": "COMPLETED",
827///                "conclusion": "SUCCESS", "checkSuite": {"app": {"databaseId": 15368}}}]}}});
828/// let page = rollup_page_of(&node);
829/// assert_eq!((page.head.as_str(), page.checks[0].app_id), ("h1", Some(15368)));
830/// assert_eq!((page.complete, page.cursor.as_deref()), (false, Some("c1")));
831/// assert!(rollup_page_of(&serde_json::json!({"headRefOid": "h1", "statusCheckRollup": null})).complete);
832/// ```
833pub fn rollup_page_of(node: &Value) -> RollupPage {
834    let contexts = node.pointer("/statusCheckRollup/contexts");
835    let info = |k: &str| contexts.and_then(|c| c.pointer(k));
836    RollupPage {
837        head: node
838            .get("headRefOid")
839            .and_then(Value::as_str)
840            .unwrap_or("")
841            .to_string(),
842        checks: contexts
843            .and_then(|c| c.get("nodes"))
844            .and_then(Value::as_array)
845            .map(|a| a.iter().map(check_of).collect())
846            .unwrap_or_default(),
847        complete: contexts.is_none()
848            || info("/pageInfo/hasNextPage").and_then(Value::as_bool) == Some(false),
849        cursor: info("/pageInfo/endCursor")
850            .and_then(Value::as_str)
851            .map(str::to_string),
852    }
853}

941/// One pull request's whole rollup, read a page of contexts at a time with
942/// [`WRITERS_OF_QUERY`]: `ask` answers the page after a cursor (`None` first). `None` when it
943/// cannot be read whole: the head moved between two pages, a page says more follow and names
944/// no cursor, or [`CONTEXT_PAGES`] pages were not enough. Unread is never a shorter list.
945///
946/// The module is private, so the example is text; the unit tests run the same assertions.
947///
948/// ```text
949/// use majordomus_cli::integration::forge::whole_rollup;
950/// let page = |more: bool, name: &str| serde_json::json!({"data": {"repository": {"pullRequest": {
951///     "number": 1, "headRefOid": "h1", "statusCheckRollup": {"contexts": {
952///       "pageInfo": {"hasNextPage": more, "endCursor": "c1"},
953///       "nodes": [{"__typename": "CheckRun", "name": name, "status": "COMPLETED"}]}}}}}});
954/// let whole = whole_rollup(|after| Ok(page(after.is_none(), after.unwrap_or("ci")))).unwrap();
955/// let (head, checks) = whole.expect("two pages");
956/// assert_eq!((head.as_str(), checks.len()), ("h1", 2));
957/// ```
958pub fn whole_rollup(
959    mut ask: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
960) -> Result<Option<(String, Vec<CheckObservation>)>, ForgeError> {
961    let mut read: Option<(String, Vec<CheckObservation>)> = None;
962    let mut cursor: Option<String> = None;
963    for _ in 0..CONTEXT_PAGES {
964        let answer = ask(cursor.as_deref())?;
965        let page = rollup_page_of(
966            answer
967                .pointer("/data/repository/pullRequest")
968                .unwrap_or(&Value::Null),
969        );
970        let complete = page.complete;
971        let (head, mut checks) = read
972            .take()
973            .unwrap_or_else(|| (page.head.clone(), Vec::new()));
974        // a rollup is of one head: pages of two heads are not one rollup
975        let same_head = head == page.head;
976        checks.extend(page.checks);
977        if complete || !same_head || page.cursor.is_none() {
978            return Ok((complete && same_head).then_some((head, checks)));
979        }
980        read = Some((head, checks));
981        cursor = page.cursor;
982    }
983    Ok(None)
984}

986/// The checks of the open pull requests `wanted` names, each read with the app that wrote
987/// it. `page` answers one [`WRITERS_QUERY`] page after a cursor (`None` first); pages are
988/// asked for until every wanted pull request was seen, the forge has no more, or `limit`
989/// pull requests' worth of pages were asked for. A wanted one whose contexts did not fit its
990/// page is read again whole with `rest` ([`whole_rollup()`]: its number, then the cursor). One
991/// that stays unread is absent from the answer, and [`attribute()`] leaves it unattributed. A
992/// read that fails fails the whole: a writer is read or the observation is not made.
993///
994/// The module is private, so the example is text; the unit tests run the same assertions.
995///
996/// ```text
997/// use majordomus_cli::integration::forge::attributed_checks;
998/// let page = serde_json::json!({"data": {"repository": {"pullRequests": {
999///   "pageInfo": {"hasNextPage": false}, "nodes": [
1000///     {"number": 1, "headRefOid": "h1", "statusCheckRollup": null}]}}}});
1001/// let wanted = [1].into_iter().collect();
1002/// let read = attributed_checks(&wanted, 500, |_| Ok(page.clone()), |_, _| unreachable!()).unwrap();
1003/// assert_eq!(read[&1], ("h1".to_string(), vec![]));
1004/// ```
1005pub fn attributed_checks(
1006    wanted: &BTreeSet<u64>,
1007    limit: usize,
1008    mut page: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1009    mut rest: impl FnMut(u64, Option<&str>) -> Result<Value, ForgeError>,
1010) -> Result<AttributedChecks, ForgeError> {
1011    let mut read = AttributedChecks::new();
1012    let mut truncated: BTreeSet<u64> = BTreeSet::new();
1013    let mut cursor: Option<String> = None;
1014    for _ in 0..limit.div_ceil(WRITERS_PAGE) {
1015        if wanted
1016            .iter()
1017            .all(|n| read.contains_key(n) || truncated.contains(n))
1018        {
1019            break;
1020        }
1021        let answer = page(cursor.as_deref())?;
1022        read.extend(attributed_checks_of(&answer));
1023        truncated.extend(truncated_of(&answer));
1024        cursor = next_cursor(&answer);
1025        if cursor.is_none() {
1026            break;
1027        }
1028    }
1029    for number in truncated.intersection(wanted) {
1030        let whole = whole_rollup(|after| rest(*number, after))?;
1031        read.extend(whole.map(|rollup| (*number, rollup)));
1032    }
1033    Ok(read)
1034}

1036/// Give each observed pull request the checks read with their writers. Only a read of the
1037/// head that was observed counts: a pull request the read does not name, or names at another
1038/// head (it moved between the two reads), keeps the checks `gh pr list` gave it, none of
1039/// which names an app, so a context bound to an app is `unknown` on it and never passed. A
1040/// head that was not observed is never matched, whatever the read says of it.
1041///
1042/// The module is private, so the example is text; the unit tests run the same assertions.
1043///
1044/// ```text
1045/// use majordomus_cli::integration::forge::{attribute, pull_request_of, AttributedChecks};
1046/// let mut prs = vec![pull_request_of(&serde_json::json!({"number": 1, "headRefOid": "h1"})).unwrap()];
1047/// let mut read = AttributedChecks::new();
1048/// read.insert(1, ("moved".to_string(), vec![Default::default()]));
1049/// attribute(&mut prs, &read);
1050/// assert!(prs[0].checks.is_empty(), "another head's checks are not this head's");
1051/// ```
1052pub fn attribute(prs: &mut [PullRequestObservation], read: &AttributedChecks) {

1161/// One pull request node of a declarations answer, read as a [`ReferencePage`]. A source is
1162/// kept only when the event says the mention came from this repository and the source is a
1163/// numbered pull request: a `#N` written in another repository is not this repository's #N,
1164/// an issue declares nothing, and an event that does not say where it came from is not
1165/// trusted to be from here. Unlike a rollup, a node without `timelineItems` is not complete:
1166/// what was not listed was not read.
1167///
1168/// The module is private, so the example is text; the unit tests run the same assertions.
1169///
1170/// ```text
1171/// use majordomus_cli::integration::forge::reference_page_of;
1172/// let node = serde_json::json!({"number": 1, "authorAssociation": "OWNER",
1173///     "isCrossRepository": false, "timelineItems": {
1174///     "pageInfo": {"hasNextPage": true, "endCursor": "c1"},
1175///     "nodes": [{"isCrossRepository": false, "source": {"__typename": "PullRequest", "number": 2}},
1176///               {"isCrossRepository": true, "source": {"__typename": "PullRequest", "number": 3}},
1177///               {"isCrossRepository": false, "source": {"__typename": "Issue"}}]}});
1178/// let page = reference_page_of(&node);
1179/// assert_eq!((page.association.as_str(), page.sources.len()), ("OWNER", 1));
1180/// assert_eq!(page.cross_repository, Some(false));
1181/// assert_eq!((page.complete, page.cursor.as_deref()), (false, Some("c1")));
1182/// assert!(!reference_page_of(&serde_json::json!({"number": 1})).complete, "unlisted is unread");
1183/// ```
1184pub fn reference_page_of(node: &Value) -> ReferencePage {
1185    let items = node.get("timelineItems");
1186    let info = |k: &str| items.and_then(|t| t.pointer(k));
1187    ReferencePage {
1188        association: node
1189            .get("authorAssociation")
1190            .and_then(Value::as_str)
1191            .unwrap_or("")
1192            .to_string(),
1193        cross_repository: node.get("isCrossRepository").and_then(Value::as_bool),
1194        sources: items
1195            .and_then(|t| t.get("nodes"))
1196            .and_then(Value::as_array)
1197            .into_iter()
1198            .flatten()
1199            .filter(|event| event.get("isCrossRepository").and_then(Value::as_bool) == Some(false))
1200            .filter_map(|event| event.get("source"))
1201            .filter(|source| source.get("number").and_then(Value::as_u64).is_some())
1202            .cloned()
1203            .collect(),
1204        complete: info("/pageInfo/hasNextPage").and_then(Value::as_bool) == Some(false),
1205        cursor: info("/pageInfo/endCursor")
1206            .and_then(Value::as_str)
1207            .map(str::to_string),
1208    }
1209}

1211/// What was read about one open pull request: who its author is to the repository, where its
1212/// head lives, and the pull requests that mention it.
1213#[derive(Debug, Clone, PartialEq)]
1214pub struct DeclarationRead {
1215    /// The forge's word for what its author is to the repository; empty when unread.
1216    pub association: String,
1217    /// Whether its head lives in a fork, as this read said it; `None` when it did not say.
1218    /// Only `Some(false)` lets its author declare anything ([`declare()`]).
1219    pub cross_repository: Option<bool>,
1220    /// The pull requests of this repository that mention it; empty unless `read` is whole.
1221    pub sources: Vec<Value>,
1222    /// Whether every cross-reference was read.
1223    pub read: CrossReferenceRead,
1224}

1226impl DeclarationRead {
1227    /// One page's answer as a read: whole with its sources when the page is all of them,
1228    /// truncated — with no source, since a part of the references is not the references —
1229    /// when it is not.
1230    fn of(page: ReferencePage, sources: Vec<Value>) -> Self {
1231        let (sources, read) = if page.complete {
1232            (sources, CrossReferenceRead::Whole)
1233        } else {
1234            (Vec::new(), CrossReferenceRead::Truncated)
1235        };
1236        DeclarationRead {
1237            association: page.association,
1238            cross_repository: page.cross_repository,
1239            sources,
1240            read,
1241        }
1242    }
1243}

1249/// Every numbered pull request node of one [`DECLARATIONS_QUERY`] answer, as it was read from
1250/// that page alone: whole when its cross-references fit, truncated — with no source — when
1251/// they did not.
1252fn declarations_of(page: &Value) -> Declarations {
1253    page.pointer("/data/repository/pullRequests/nodes")
1254        .and_then(Value::as_array)
1255        .into_iter()
1256        .flatten()
1257        .filter_map(|node| {

1260            let sources = std::mem::take(&mut page.sources);
1261            Some((number, DeclarationRead::of(page, sources)))
1262        })

1264}

1266/// Every cross-reference to one pull request, read a page at a time with
1267/// [`DECLARATIONS_OF_QUERY`]: `ask` answers the page after a cursor (`None` first). Whole when
1268/// the forge says no more follow; truncated, with no source, when a page says more follow and
1269/// names no cursor, or [`REFERENCE_PAGES`] pages were not enough. `None` when the forge no
1270/// longer shows the pull request: nothing was read of it, and unread is never a shorter list.
1271/// A page the forge would not answer is the error it gave: a failed read is not a hold.
1272fn whole_references(
1273    mut ask: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1274) -> Result<Option<DeclarationRead>, ForgeError> {
1275    let mut sources: Vec<Value> = Vec::new();
1276    let mut cursor: Option<String> = None;
1277    let mut last: Option<ReferencePage> = None;
1278    for _ in 0..REFERENCE_PAGES {
1279        let answer = ask(cursor.as_deref())?;
1280        let Some(node) = answer
1281            .pointer("/data/repository/pullRequest")
1282            .filter(|node| node.is_object())
1283        else {
1284            return Ok(None);
1285        };
1286        let mut page = reference_page_of(node);
1287        sources.append(&mut page.sources);
1288        cursor = page.cursor.clone();
1289        let stop = page.complete || cursor.is_none();
1290        last = Some(page);
1291        if stop {
1292            break;
1293        }
1294    }
1295    // the last page read says whether more follow: one that does not is the whole
1296    Ok(last.map(|page| DeclarationRead::of(page, sources)))
1297}

1299/// What is declared about the open pull requests `wanted` names, and by whom. `page` answers
1300/// one [`DECLARATIONS_QUERY`] page after a cursor (`None` first); pages are asked for until
1301/// every wanted pull request was seen, the forge has no more, or `limit` pull requests' worth
1302/// of pages were asked for. A wanted one that was not read whole by then — its
1303/// cross-references did not fit its page, or no page listed it — is asked for alone with
1304/// `rest` (its number, then the cursor), [`REFERENCE_PAGES`] pages at most; one that still
1305/// does not fit is truncated, with no source at all, and held.
1306///
1307/// The two ways a read falls short are kept apart. A truncated read is that pull request's:
1308/// it is held and the others are decided. A read the forge would not answer — a page, or one
1309/// pull request's references — is an error, and the caller makes no observation from it:
1310/// what was not read is never an observation with a part missing. One the forge no longer
1311/// shows when asked for alone is unread — with the association a page gave it, or else absent
1312/// from the answer, which [`declare()`] leaves unread too.
1313///
1314/// The module is private, so the example is text; the unit tests run the same assertions.
1315///
1316/// ```text
1317/// use majordomus_cli::integration::forge::declarations;
1318/// use majordomus_cli::integration::CrossReferenceRead;
1319/// let page = serde_json::json!({"data": {"repository": {"pullRequests": {
1320///   "pageInfo": {"hasNextPage": false}, "nodes": [
1321///     {"number": 1, "authorAssociation": "OWNER", "isCrossRepository": false, "timelineItems": {
1322///        "pageInfo": {"hasNextPage": false}, "nodes": []}}]}}}});
1323/// let wanted = [1, 2].into_iter().collect();
1324/// let gone = |_: u64, _: Option<&str>| Ok(serde_json::json!({"data": {"repository": {"pullRequest": null}}}));
1325/// let read = declarations(&wanted, 500, |_| Ok(page.clone()), gone).unwrap();
1326/// assert_eq!((read[&1].association.as_str(), read[&1].read), ("OWNER", CrossReferenceRead::Whole));
1327/// assert!(!read.contains_key(&2), "one the forge no longer shows is unread, and #1 still is read");
1328/// let refused = |_: u64, _: Option<&str>| Err(ForgeError("HTTP 403".into()));
1329/// assert!(declarations(&wanted, 500, |_| Ok(page.clone()), refused).is_err(), "a failed read is an error");
1330/// ```
1331pub fn declarations(
1332    wanted: &BTreeSet<u64>,
1333    limit: usize,
1334    mut page: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1335    mut rest: impl FnMut(u64, Option<&str>) -> Result<Value, ForgeError>,
1336) -> Result<Declarations, ForgeError> {
1337    let mut read = Declarations::new();
1338    let mut cursor: Option<String> = None;
1339    for _ in 0..limit.div_ceil(DECLARATIONS_PAGE) {
1340        if wanted.iter().all(|n| read.contains_key(n)) {
1341            break;
1342        }
1343        let answer = page(cursor.as_deref())?;
1344        read.extend(declarations_of(&answer));
1345        cursor = next_cursor(&answer);
1346        if cursor.is_none() {
1347            break;
1348        }
1349    }
1350    let whole = |read: &Declarations, n: &u64| {
1351        read.get(n)
1352            .is_some_and(|d| d.read == CrossReferenceRead::Whole)
1353    };
1354    let alone: Vec<u64> = wanted
1355        .iter()
1356        .copied()
1357        .filter(|n| !whole(&read, n))
1358        .collect();
1359    for number in alone {
1360        let entry = whole_references(|after| rest(number, after))?;
1361        // one the forge no longer shows is not a truncation: what a page said of its author
1362        // is kept, and its references are unread
1363        if let Some(kept) = read.get_mut(&number).filter(|_| entry.is_none()) {
1364            kept.read = CrossReferenceRead::Unread;
1365        }
1366        read.extend(entry.map(|entry| (number, entry)));
1367    }
1368    Ok(read)
1369}

1371/// Give each observed pull request what the declarations read said of it: its author's
1372/// association, and whether its cross-references were read whole. One the read does not name
1373/// keeps no association and `unread`, so it authorises nothing and is held. Returns the
1374/// sources of every named pull request's cross-references, in number order, as one array:
1375/// what [`resolved_for()`] reads the closed declarers from.
1376///
1377/// Where a head lives is half of who may declare, and `gh pr list` reads an absent answer as
1378/// "this repository". So an association is recorded only for a pull request this read placed:
1379/// one it says lives here (`Some(false)`), or one the list already calls a fork, which
1380/// declares nothing whatever its association. For one this read did not place, or placed in a
1381/// fork the list did not see, no association is recorded, and it authorises nothing. The
1382/// list's own flag is left as it was read.
1383///
1384/// A source that is open, says it supersedes the pull request, and is not among the listed
1385/// ones makes that pull request's references `unread`, whatever the read said: an open
1386/// successor the list did not carry is one nothing else would hold it for.
1387///
1388/// The module is private, so the example is text; the unit tests run the same assertions.
1389///
1390/// ```text
1391/// use majordomus_cli::integration::forge::{declare, pull_request_of, DeclarationRead, Declarations};
1392/// use majordomus_cli::integration::CrossReferenceRead;
1393/// let mut prs = vec![pull_request_of(&serde_json::json!({"number": 1})).unwrap(),
1394///                    pull_request_of(&serde_json::json!({"number": 2})).unwrap()];
1395/// let mut read = Declarations::new();
1396/// read.insert(1, DeclarationRead { association: "OWNER".into(), cross_repository: Some(false),
1397///     sources: vec![serde_json::json!({"number": 7})], read: CrossReferenceRead::Whole });
1398/// let sources = declare(&mut prs, &read);
1399/// assert_eq!(sources, serde_json::json!([{"number": 7}]));
1400/// assert_eq!(prs[0].cross_references, CrossReferenceRead::Whole);
1401/// assert_eq!(prs[1].cross_references, CrossReferenceRead::Unread, "not named: held");
1402/// ```
1403pub fn declare(prs: &mut [PullRequestObservation], read: &Declarations) -> Value {
1404    use super::classify::declared_supersessions;
1405    let listed: BTreeSet<u64> = prs.iter().map(|p| p.number).collect();
1406    let mut sources: BTreeMap<u64, &[Value]> = BTreeMap::new();
1407    for pr in prs.iter_mut() {
1408        if let Some(said) = read.get(&pr.number) {
1409            let placed = said.cross_repository == Some(false) || pr.cross_repository;
1410            pr.author_association = if placed {
1411                said.association.clone()
1412            } else {
1413                String::new()
1414            };
1415            // An open pull request that says it supersedes this one, and that the list of
1416            // open ones does not carry: the list was cut at its limit, or it was opened
1417            // between the two reads. Nothing else reads an open one's body, so its
1418            // declaration would hold nothing and this one could merge beside its successor.
1419            // What is known is that the list is not all that is open: unread, and held.
1420            let number = pr.number;
1421            let unlisted = said.sources.iter().any(|source| {
1422                source.get("state").and_then(Value::as_str) == Some("OPEN")
1423                    && source
1424                        .get("number")
1425                        .and_then(Value::as_u64)
1426                        .is_some_and(|m| m != number && !listed.contains(&m))
1427                    && source
1428                        .get("body")
1429                        .and_then(Value::as_str)
1430                        .is_none_or(|body| {
1431                            declared_supersessions(body).supersedes.contains(&number)
1432                        })
1433            });
1434            pr.cross_references = if unlisted {
1435                CrossReferenceRead::Unread
1436            } else {
1437                said.read
1438            };
1439            sources.insert(pr.number, &said.sources);
1440        }
1441    }
1442    Value::Array(sources.into_values().flatten().cloned().collect())
1443}

1548                "number,title,author,headRefName,headRefOid,baseRefName,isDraft,labels,createdAt,updatedAt,body,statusCheckRollup,reviewDecision,latestReviews,reviewRequests,autoMergeRequest,isCrossRepository",
1549            ],
1550        )?;
1551        // keyed by number, so the observation is in number order whatever the forge listed
1552        let mut pull_requests: Vec<PullRequestObservation> = list
1553            .as_array()
1554            .map(|a| {
1555                a.iter()
1556                    .filter_map(pull_request_of)
1557                    .map(|p| (p.number, p))
1558                    .collect::<std::collections::BTreeMap<_, _>>()
1559                    .into_values()
1560                    .collect()
1561            })
1562            .unwrap_or_default();
1563        // a context bound to an app is only that app's check run, and the list above does not
1564        // say who wrote one: the writers are read, or the observation is not made. A base that
1565        // binds nothing needs no writer and asks nothing more.
1566        let bound = required_checks
1567            .as_ref()
1568            .is_some_and(|r| r.iter().any(|c| c.app_id.is_some()));
1569        let (owner, name) = repository
1570            .split_once('/')
1571            .unwrap_or((repository.as_str(), ""));
1572        let wanted: BTreeSet<u64> = pull_requests.iter().map(|p| p.number).collect();
1573        let ask = |query: &str, typed: &str, after: Option<&str>| {
1574            gh_graphql(root, &writers_args(query, owner, name, typed, after))
1575        };
1576        if bound {
1577            let size = format!("n={WRITERS_PAGE}");
1578            let read = attributed_checks(
1579                &wanted,
1580                OPEN_LIMIT,
1581                |after| ask(WRITERS_QUERY, &size, after),
1582                |number, after| ask(WRITERS_OF_QUERY, &format!("number={number}"), after),
1583            )?;
1584            attribute(&mut pull_requests, &read);
1585        }
1586        // who may declare a replacement, and what was declared: every open pull request is
1587        // read with its author's association and the pull requests that mention it. A
1588        // successor that landed is no longer listed among the open ones, and it is found
1589        // among the cross-references of the one it replaces, whenever it was closed: no
1590        // bounded search of closed pull requests decides. Nothing open, nothing asked; a read
1591        // that is partial holds the pull request it was about, and no other; a read that
1592        // fails makes no observation at all.
1593        let size = format!("n={DECLARATIONS_PAGE}");
1594        let read = declarations(
1595            &wanted,
1596            OPEN_LIMIT,
1597            |after| ask(DECLARATIONS_QUERY, &size, after),
1598            |number, after| ask(DECLARATIONS_OF_QUERY, &format!("number={number}"), after),
1599        )?;
1600        let sources = declare(&mut pull_requests, &read);
1601        let mut unreadable = None;
1602        let resolved = resolved_for(&pull_requests, &sources, |m| {
1603            match gh_retrying(
1604                root,
1605                &["pr", "view", &m.to_string(), "--json", RESOLVED_FIELDS],
1606            ) {
1607                Ok((true, out, _)) => serde_json::from_str(&out).ok(),
1608                // not a pull request, or refused: the successor is unread, never landed
1609                Ok(_) => None,
1610                Err(e) => {
1611                    unreadable.get_or_insert(e);
1612                    None
1613                }
1614            }
1615        });
1616        // an outage that outlasted the retries is a failed observation, as for the open list
1617        if let Some(e) = unreadable {
1618            return Err(e);
1619        }
1620        Ok(ForgeObservation {
1621            schema: OBSERVATION_SCHEMA,
1622            repository,
1623            base,
1624            base_sha,
1625            observed_at: crate::peers::rfc3339(std::time::SystemTime::now()),
1626            required_checks,
1627            review_policy,
1628            up_to_date_required,
1629            merge_methods,
1630            pull_requests,
1631            resolved,
1632            delete_branch_on_merge,
1633        })
1634    }
1635}

1637/// The merged branches origin still serves ([`merged_branches_of`]), read now, or `None` when
1638/// origin's branches or the merged pull requests could not be read: unread is not "nothing
1639/// left behind". Asked only by `prs cleanup`, on demand — never by `prs refresh`, which the
1640/// executor runs before every decision: what the forge left behind decides no merge, so its
1641/// two reads stay off the hot path.
1642pub fn merged_branches(root: &Path, base: &str) -> Option<Vec<MergedBranch>> {
1643    let heads = origin_heads(root)?;
1644    merged_branches_given(&heads, base, || merged_list(root))
1645}

1647/// [`merged_branches`] once origin's branches are known: an origin serving only the base has
1648/// left nothing behind and the merged pull requests are never asked for; otherwise `list`
1649/// answers them, and `None` from it is unread.
1650///
1651/// ```text
1652/// use majordomus_cli::integration::forge::{merged_branches_given, remote_heads_of};
1653/// let only_base = remote_heads_of("cc\trefs/heads/master\n");
1654/// assert_eq!(merged_branches_given(&only_base, "master", || unreachable!()), Some(vec![]));
1655/// let heads = remote_heads_of("aa\trefs/heads/fix/a\ncc\trefs/heads/master\n");
1656/// assert_eq!(merged_branches_given(&heads, "master", || None), None, "unread");
1657/// ```
1658pub fn merged_branches_given(
1659    heads: &BTreeMap<String, String>,
1660    base: &str,
1661    list: impl FnOnce() -> Option<Value>,
1662) -> Option<Vec<MergedBranch>> {
1663    if heads.keys().all(|b| b == base) {
1664        return Some(Vec::new());
1665    }
1666    Some(merged_branches_of(&list()?, heads, base))
1667}

1727/// Remove the mirrors of pull requests this observation no longer names: merged, closed or
1728/// gone. Each one held the head the pull request had when it was last open, and a mirror
1729/// that is never refreshed again is only a stale answer waiting to be read. Best effort: a
1730/// mirror that cannot be removed is left, and nothing reads it.
1731fn prune_mirrors(root: &Path, obs: &ForgeObservation) {
1732    let keep: std::collections::BTreeSet<u64> = obs
1733        .pull_requests
1734        .iter()
1735        .map(|p| p.number)
1736        .chain(obs.resolved.keys().copied())
1737        .collect();
1738    // git that cannot run lists nothing, and nothing is removed
1739    let out = Command::new("git")
1740        .arg("-C")
1741        .arg(root)
1742        .args(["for-each-ref", "--format=%(refname)", PR_REF_PREFIX])
1743        .output()
1744        .map(|o| o.stdout)
1745        .unwrap_or_default();
1746    for r in String::from_utf8_lossy(&out).lines() {
1747        let stale = r
1748            .strip_prefix(PR_REF_PREFIX)
1749            .and_then(|n| n.parse::<u64>().ok())
1750            .is_some_and(|n| !keep.contains(&n));
1751        if stale {
1752            let _ = Command::new("git")
1753                .arg("-C")
1754                .arg(root)
1755                .args(["update-ref", "-d", r])
1756                .status();
1757        }
1758    }
1759}

1761/// Fetch the base and every observed head into this clone, the heads under
1762/// [`PR_REF_PREFIX`]. One `git fetch`; a head that cannot be fetched stays unknown to the
1763/// relation, which then says so.
1764pub fn fetch(root: &Path, obs: &ForgeObservation) -> Result<(), ForgeError> {
1765    let mut args: Vec<String> = vec![
1766        "-C".into(),
1767        root.display().to_string(),
1768        "fetch".into(),
1769        "--quiet".into(),
1770        "--no-tags".into(),
1771        "origin".into(),
1772        format!("+refs/heads/{0}:refs/remotes/origin/{0}", obs.base),
1773    ];
1774    for p in &obs.pull_requests {
1775        args.push(format!("+refs/pull/{0}/head:{PR_REF_PREFIX}{0}", p.number));
1776    }
1777    // a successor no longer open is fetched too, so git can say whether its head landed; the
1778    // forge keeps refs/pull/<n>/head for a closed pull request, and only one it read is here
1779    for n in obs.resolved.keys() {
1780        args.push(format!("+refs/pull/{n}/head:{PR_REF_PREFIX}{n}"));
1781    }
1782    prune_mirrors(root, obs);
1783    // a fetch is a read: a dropped connection is asked again, a refusal is not
1784    super::retry::forge(|| {
1785        let out = Command::new("git")
1786            .args(&args)
1787            .output()
1788            .map_err(|e| format!("git fetch could not run: {e}"))?;
1789        if out.status.success() {
1790            Ok(())
1791        } else {
1792            Err(format!(
1793                "git fetch failed: {}",
1794                String::from_utf8_lossy(&out.stderr).trim()
1795            ))
1796        }
1797    })
1798    .map_err(ForgeError)
1799}

3193        let listed = |number: u64| pull_request_of(&json!({"number": number})).unwrap();
3194        let references_of = |sources: Vec<Value>| {
3195            let mut open = vec![listed(1), listed(2)];
3196            let mut read = Declarations::new();
3197            for number in [1, 2] {
3198                read.insert(
3199                    number,
3200                    DeclarationRead {
3201                        association: "OWNER".into(),
3202                        cross_repository: Some(false),
3203                        sources: if number == 1 {
3204                            sources.clone()
3205                        } else {
3206                            Vec::new()
3207                        },
3208                        read: CrossReferenceRead::Whole,
3209                    },
3210                );
3211            }
3212            declare(&mut open, &read);
3213            (open[0].cross_references, open[1].cross_references)
3214        };

3217        let unlisted = source(9, "OPEN", "Supersedes #1", "OWNER", false);
