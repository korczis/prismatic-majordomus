1//! One offloaded derive, end to end: the worktree goes to a host as a git bundle, the host
2//! derives it in its turn, and the derived files come back staged.
3//!
4//! What it does, and what it never does:
5//!
6//! - The tree that travels is the worktree as it stands: HEAD when it is clean, otherwise a
7//!   `git stash create` snapshot of its tracked and staged changes. That writes a commit
8//!   object and touches neither the index, the worktree nor a hook — which matters, because
9//!   the commit hook refuses a tree that is not derived yet, so the authored change cannot
10//!   be committed before its derive. Untracked files would not travel and are refused.
11//! - A merge in progress travels as a real merge commit with both parents, written by
12//!   `git commit-tree` from the staged tree once every conflict is resolved: a stash would
13//!   drop the second parent, and the changelog — composed from the log — would be derived
14//!   without the merged commits (PR #854, 2026-10-10). No ref of the branch moves.
15//! - Nothing is pushed to origin, so no CI run starts, and no hook of this repository runs.
16//! - On the host, the derive runs in a detached worktree `offload-<tag>`, under the queue,
17//!   with a build directory of its own: a copy-on-write clone of the shared one where the
18//!   filesystem has clones, removed when the derive ends. A build into the shared directory
19//!   by a derive that never took the lock cannot replace this derive's executable (it did,
20//!   on the second MacBook, 2026-10-11 00:25Z).
21//! - Back here the files the derive changed are checked out of the derived snapshot and
22//!   staged. Nothing is committed. A derive that deletes a tracked file is refused rather
23//!   than half-applied.
24//! - The machine's own derive lock is not taken: it guards this machine, and this machine
25//!   is not deriving.
26

39/// A precondition of the worktree is not met: an untracked file, an unresolved merge.
40pub const EXIT_PRECONDITION: u8 = 2;
41/// The derive failed on the host.
42pub const EXIT_DERIVE_FAILED: u8 = 10;
43/// The derive deleted a tracked file; nothing was applied.
44pub const EXIT_DELETES: u8 = 11;
45/// No host could take the derive, or the host could not be used, or the result did not apply.
46pub const EXIT_UNAVAILABLE: u8 = 12;
47/// The queue's declared wait ran out.
48pub const EXIT_TIMED_OUT: u8 = 75;

50/// Why an offload stopped, with the exit code the command answers.
51#[derive(Debug)]
52pub struct Refusal {
53    /// The exit code.
54    pub code: u8,
55    /// What happened and what to do.
56    pub reason: String,
57}
58
59fn refuse(code: u8, reason: impl Into<String>) -> Refusal {
60    Refusal {
61        code,
62        reason: reason.into(),
63    }
64}

100fn git(dir: &Path, args: &[&str]) -> Result<String, Refusal> {
101    let out = Command::new("git")
102        .arg("-C")
103        .arg(dir)
104        .args(args)
105        .output()
106        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("git could not be run: {e}")))?;
107    if !out.status.success() {
108        return Err(refuse(
109            EXIT_UNAVAILABLE,
110            format!(
111                "git {} failed: {}",
112                args.join(" "),
113                String::from_utf8_lossy(&out.stderr).trim()
114            ),
115        ));
116    }
117    Ok(String::from_utf8_lossy(&out.stdout).into_owned())
118}

121    Command::new("git")
122        .arg("-C")
123        .arg(dir)
124        .args(args)
125        .output()
126        .map(|o| o.status.success())
127        .unwrap_or(false)

130/// The identity of the scratch commit objects this module writes (the input snapshot, the
131/// merge input): no ref names them and nobody's identity belongs on them, and a machine
132/// without a configured identity can still offload.
133const SCRATCH_NAME: &str = "user.name=offload";
134const SCRATCH_EMAIL: &str = "user.email=offload@invalid";

136/// The commit that travels, after the worktree's preconditions are checked.
137pub fn input_commit(worktree: &Path) -> Result<String, Refusal> {
138    let merge_head = git(
139        worktree,
140        &[
141            "rev-parse",
142            "--path-format=absolute",
143            "--git-path",
144            "MERGE_HEAD",
145        ],
146    )?;
147    let in_merge = Path::new(merge_head.trim()).exists();
148    if in_merge {
149        if !git(worktree, &["diff", "--name-only", "--diff-filter=U"])?
150            .trim()
151            .is_empty()
152        {
153            return Err(refuse(
154                EXIT_PRECONDITION,
155                "the merge still has unresolved paths — resolve and git add them first",
156            ));
157        }
158        if !git_ok(worktree, &["diff", "--quiet"]) {
159            return Err(refuse(EXIT_PRECONDITION, "the merge has unstaged changes — git add them first, the staged tree is what travels"));
160        }
161    }
162    let status = git(
163        worktree,
164        &["status", "--porcelain", "--untracked-files=all"],
165    )?;
166    let untracked: Vec<&str> = status
167        .lines()
168        .filter_map(|l| l.strip_prefix("?? "))
169        .collect();
170    if !untracked.is_empty() {
171        return Err(refuse(
172            EXIT_PRECONDITION,
173            format!(
174                "untracked files would not travel — git add them first (a derive sees what the commit will): {}",
175                untracked.join(", ")
176            ),
177        ));
178    }
179    if in_merge {
180        let tree = git(worktree, &["write-tree"])?;
181        let other = git(worktree, &["rev-parse", "MERGE_HEAD"])?;
182        let head = git(
183            worktree,
184            &[
185                "-c",
186                SCRATCH_NAME,
187                "-c",
188                SCRATCH_EMAIL,
189                "commit-tree",
190                tree.trim(),
191                "-p",
192                "HEAD",
193                "-p",
194                other.trim(),
195                "-m",
196                "derive-offload input: a merge in progress",
197            ],
198        )?;
199        return Ok(head.trim().to_string());
200    }
201    let stash = git(
202        worktree,
203        &[
204            "-c",
205            SCRATCH_NAME,
206            "-c",
207            SCRATCH_EMAIL,
208            "stash",
209            "create",
210            "derive-offload input",
211        ],
212    )?;
213    if !stash.trim().is_empty() {
214        return Ok(stash.trim().to_string());
215    }
216    Ok(git(worktree, &["rev-parse", "HEAD"])?.trim().to_string())
217}

219/// Prepare the host's worktree from the uploaded bundle, or — when the input is already in
220/// the base, so there was nothing to bundle — from the commit itself, which the host has
221/// once it has fetched origin. Arguments: checkout, worktrees directory, spool, tag, input.
222const PREPARE: &str = r#"set -e
223c=$1; w=$2; spool=$3; tag=$4; input=$5
224case $c in /*) ;; *) c=$HOME/$c ;; esac
225case $w in /*) ;; *) w=$HOME/$w ;; esac
226cd "$c"
227git fetch -q origin
228if [ -f "$spool/offload-$tag.bundle" ]; then
229  git fetch -q "$spool/offload-$tag.bundle" "refs/offload/in-$tag:refs/offload/$tag"
230else
231  git update-ref "refs/offload/$tag" "$input"
232fi
233d=$w/offload-$tag
234[ -d "$d" ] || git worktree add -q --detach "$d" "refs/offload/$tag"
235cd "$d"
236git checkout -q -f --detach "refs/offload/$tag"
237git clean -fdq"#;

239/// Derive on the host, under a lock the caller holds. Its own pid joins the lock's holders,
240/// so the lock stays held exactly as long as this derive runs.
241///
242/// Everything the derive compiles stays in its own run directory, `<shared>-<tag>`, which
243/// the traps remove however the derive ends. Pinning the target directory alone is not
244/// that: the checkout declares a shared `build-dir`, so cargo kept every intermediate in the
245/// host's `build-shared/majordomus`, outside what the traps remove, and on 2026-10-18 that
246/// directory held 87 GB on lundra and the volume reached 0 bytes. The build directory is
247/// pinned to the run directory too. Arguments: lock, worktrees
248/// directory, tag, shared build directory, derive command, spool, PATH prefix
249/// (colon-separated), the commit the derived files may observe (exported to the derive as
250/// `MJ_DERIVE_OBSERVE`; empty leaves it unset), then the files to touch. Prints `target <dir>`, then `nothing` or
251/// `derived <sha>`; exit 10 when the derive fails, 13 when the lock is not there.
252const DERIVE: &str = r#"l=$1; w=$2; tag=$3; shared=$4; derive=$5; spool=$6; prefix=$7; observe=$8; shift 8
253[ -d "$l" ] || { echo "the lock $l is not held" >&2; exit 13; }
254echo "pid $$"
255: > "$l/$$"
256case $w in /*) ;; *) w=$HOME/$w ;; esac
257case $shared in /*) ;; *) shared=$HOME/$shared ;; esac
258p=""; IFS=:; for e in $prefix; do case $e in /*) ;; *) e=$HOME/$e ;; esac; p="$p$e:"; done; unset IFS
259export PATH="$p$PATH"
260cd "$w/offload-$tag" || exit 13
261private="$shared-$tag"; log="$spool/offload-$tag.log"
262mkdir -p "$shared" "$spool"; rm -rf "$private"
263if [ "$(uname)" = Darwin ]; then cp -cR "$shared" "$private" 2>/dev/null || { rm -rf "$private"; mkdir -p "$private"; }
264else cp -a --reflink=always "$shared" "$private" 2>/dev/null || { rm -rf "$private"; mkdir -p "$private"; }; fi
265trap 'rm -rf "$private"' EXIT
266trap 'rm -rf "$private"; exit 129' HUP
267trap 'rm -rf "$private"; exit 130' INT
268trap 'rm -rf "$private"; exit 143' TERM
269export CARGO_TARGET_DIR="$private"
270export CARGO_BUILD_BUILD_DIR="$private"
271[ -n "$observe" ] && { MJ_DERIVE_OBSERVE=$observe; export MJ_DERIVE_OBSERVE; }
272echo "target $private"
273for f in "$@"; do [ -e "$f" ] && touch "$f"; done
274if ! env -u MAJORDOMUS_SHARE -u MAJORDOMUS_BIN "$derive" > "$log" 2>&1; then
275  echo "derive failed on $(hostname); the last lines of $log:" >&2
276  grep -vE ' INFO |^ +(Compiling|Checking)' "$log" | tail -8 >&2
277  exit 10
278fi
279grep -E '^==> derive: done' "$log" >&2 || true
280git add -A
281sha=$(git -c user.name=offload -c user.email=offload@invalid stash create derived || true)
282if [ -z "$sha" ]; then echo nothing; exit 0; fi
283git update-ref "refs/offload/$tag-derived" "$sha"
284git bundle create "$spool/offload-$tag-back.bundle" "refs/offload/$tag..refs/offload/$tag-derived" >/dev/null 2>&1
285echo "derived $sha""#;
286
287/// Upload: standard input to a spool file. Arguments: spool, file name.
288const UPLOAD: &str = r#"mkdir -p "$1" && cat > "$1/$2""#;
289
290/// Download: a spool file to standard output. Arguments: spool, file name.
291const DOWNLOAD: &str = r#"cat "$1/$2""#;
292
293/// Clean up the host: the bundles, and the worktree unless kept. Arguments: checkout,
294/// worktrees directory, spool, tag, keep (1 or 0).
295const CLEANUP: &str = r#"c=$1; w=$2; spool=$3; tag=$4; keep=$5
296case $c in /*) ;; *) c=$HOME/$c ;; esac
297case $w in /*) ;; *) w=$HOME/$w ;; esac
298rm -f "$spool/offload-$tag.bundle" "$spool/offload-$tag-back.bundle"
299[ "$keep" = 1 ] || git -C "$c" worktree remove --force "$w/offload-$tag" 2>/dev/null || true
300exit 0"#;

302/// A ref this process wrote and removes however it ends.
303struct ScratchRef<'a> {
304    dir: &'a Path,
305    name: String,
306}
307
308impl Drop for ScratchRef<'_> {
309    fn drop(&mut self) {
310        let _ = git(self.dir, &["update-ref", "-d", &self.name]);
311    }
312}

314/// The lock held on a host, released however this process ends short of a crash; after a
315/// crash the holders are dead and the next waiter reclaims it.
316struct HeldLock<'a, 'q> {
317    queue: &'q Queue<'a>,
318    pids: Vec<u32>,
319}
320
321impl Drop for HeldLock<'_, '_> {
322    fn drop(&mut self) {
323        self.queue.release(&self.pids);
324    }
325}

327fn pick_host(
328    d: &OffloadDeclaration,
329    named: Option<&str>,
330    log: &mut dyn Write,
331) -> Result<(OffloadHost, Vec<HostReport>), Refusal> {
332    if let Some(name) = named {
333        let host = d.host(name).cloned().ok_or_else(|| {
334            refuse(
335                EXIT_UNAVAILABLE,
336                format!(
337                    "'{name}' is not a declared host; declared: {}",
338                    d.hosts
339                        .iter()
340                        .map(|h| h.id.as_str())
341                        .collect::<Vec<_>>()
342                        .join(", ")
343                ),
344            )
345        })?;
346        return Ok((host, Vec::new()));
347    }
348    let reports = probe(d);
349    for r in &reports {
350        if let Some(why) = &r.reason {
351            let _ = writeln!(log, "derive-offload: {} skipped: {why}", r.id);
352        }
353    }
354    let Some(chosen) = reports.iter().find(|r| r.chosen) else {
355        return Err(refuse(
356            EXIT_UNAVAILABLE,
357            "no declared host answers with its free-space minimum free",
358        ));
359    };
360    let _ = writeln!(
361        log,
362        "derive-offload: chose {} ({} running or waiting there, {} GiB free)",
363        chosen.id,
364        chosen.load.unwrap_or(0),
365        chosen.free_gib.unwrap_or(0)
366    );
367    let host = d
368        .host(&chosen.id)
369        .cloned()
370        .expect("a chosen host is declared");
371    Ok((host, reports))
372}

374/// Offload one derive. Progress goes to `log`; the answer is the report or a refusal.
375pub fn offload(
376    d: &OffloadDeclaration,
377    request: &OffloadRequest,
378    log: &mut dyn Write,
379) -> Result<OffloadReport, Refusal> {
380    let started = Instant::now();
381    let worktree = std::fs::canonicalize(&request.worktree).map_err(|e| {
382        refuse(
383            EXIT_PRECONDITION,
384            format!("{}: {e}", request.worktree.display()),
385        )
386    })?;
387    let top = git(&worktree, &["rev-parse", "--show-toplevel"])?;
388    let worktree = PathBuf::from(top.trim());
389
390    // The commit the derived files may observe: the worktree's HEAD, never the input. The
391    // input of a dirty tree is a `git stash create` snapshot and that of a merge in progress
392    // a `commit-tree` commit; neither reaches origin, so a derived file that named one
393    // (delivery.json's observed.commit) could not be checked anywhere else, and CI's
394    // `generate-site-data --check` called it stale (#1759, I3836). Read before the input, so
395    // a branch with no commit yet is refused here, by what it lacks, before anything is
396    // snapshot.
397    let observe = git(&worktree, &["rev-parse", "--verify", "HEAD"])
398        .map_err(|_| {
399            refuse(
400                EXIT_PRECONDITION,
401                "the worktree's branch has no commit yet: a derive observes HEAD, so commit first",
402            )
403        })?
404        .trim()
405        .to_string();
406    let input = input_commit(&worktree)?;
407    let (host, probe_reports) = pick_host(d, request.host.as_deref(), log)?;
408    let remote = remote_of(&host);
409    // The base must be current here so the bundle carries only what the host lacks.
410    let _ = git(&worktree, &["fetch", "-q", "origin"]);
411
412    let tag = input[..10].to_string();
413    let in_ref = format!("refs/offload/in-{tag}");
414    git(&worktree, &["update-ref", &in_ref, &input])?;
415    let _in_guard = ScratchRef {
416        dir: &worktree,
417        name: in_ref.clone(),
418    };
419    let scratch = tempfile_dir()?;
420    let spool = d.remote.spool.as_str();
421    // An input the base already reaches has nothing to bundle (git refuses an empty bundle);
422    // the host has it once it fetches origin.
423    if !git_ok(&worktree, &["merge-base", "--is-ancestor", &input, &d.base]) {
424        let bundle = scratch.path.join("in.bundle");
425        let range = format!("{}..{in_ref}", d.base);
426        git(
427            &worktree,
428            &["bundle", "create", &bundle.to_string_lossy(), &range],
429        )?;
430        let upload = remote
431            .run(
432                UPLOAD,
433                &[spool, &format!("offload-{tag}.bundle")],
434                Some(&bundle),
435            )
436            .map_err(|e| {
437                refuse(
438                    EXIT_UNAVAILABLE,
439                    format!("{}: the bundle could not be sent: {e}", host.id),
440                )
441            })?;
442        if !upload.ok() {
443            return Err(refuse(
444                EXIT_UNAVAILABLE,
445                format!(
446                    "{}: the bundle could not be sent: {}",
447                    host.id,
448                    upload.stderr.trim()
449                ),
450            ));
451        }
452    }
453    let _ = writeln!(
454        log,
455        "derive-offload: {tag} on {} — deriving (log: {}:{spool}/offload-{tag}.log)",
456        host.id, host.address
457    );
458    let prep = remote
459        .run(
460            PREPARE,
461            &[&d.remote.checkout, &d.remote.worktrees, spool, &tag, &input],
462            None,
463        )
464        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{}: {e}", host.id)))?;
465    if !prep.ok() {
466        return Err(refuse(
467            EXIT_UNAVAILABLE,
468            format!(
469                "{}: the host could not prepare the worktree: {}",
470                host.id,
471                prep.stderr.trim()
472            ),
473        ));
474    }
475
476    let queue = Queue {
477        remote: &remote,
478        directory: &d.queue.directory,
479        lock: &d.queue.lock,
480        poll: Duration::from_secs(d.queue.poll_seconds),
481        wait_max: Duration::from_secs(d.queue.wait_max_seconds),
482    };
483    let holder = Holder::start(&remote).map_err(|e| {
484        refuse(
485            EXIT_UNAVAILABLE,
486            format!("{}: no holder process could be started: {e}", host.id),
487        )
488    })?;
489    let ticket = queue
490        .enqueue(holder.pid, &tag)
491        .map_err(|e| refuse(EXIT_UNAVAILABLE, e.to_string()))?;
492    let wait_started = Instant::now();
493    queue.wait(&ticket, holder.pid, log).map_err(|e| match e {
494        QueueError::TimedOut(_) => refuse(EXIT_TIMED_OUT, e.to_string()),
495        _ => refuse(EXIT_UNAVAILABLE, e.to_string()),
496    })?;
497    let waited = wait_started.elapsed().as_secs();
498    let mut lock = HeldLock {
499        queue: &queue,
500        pids: vec![holder.pid],
501    };
502
503    let prefix = d.remote.path.join(":");
504    let mut args: Vec<&str> = vec![
505        &d.queue.lock,
506        &d.remote.worktrees,
507        &tag,
508        &d.remote.target,
509        &d.remote.derive,
510        spool,
511        &prefix,
512        &observe,
513    ];
514    args.extend(d.remote.touch.iter().map(String::as_str));
515    let out = remote
516        .run(DERIVE, &args, None)
517        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{}: {e}", host.id)))?;
518    let _ = log.write_all(out.stderr.as_bytes());
519    // the derive's own pid file is dead now; release it with the holder's
520    lock.pids.extend(out.stdout.lines().filter_map(|l| {
521        l.strip_prefix("pid ")
522            .and_then(|p| p.trim().parse::<u32>().ok())
523    }));
524    // released now rather than at the end of the scope, so the next waiter starts while
525    // this one applies its result
526    queue.release(&lock.pids);
527    lock.pids.clear();
528    drop(holder);
529    let cleanup = |keep: bool| {
530        let _ = remote.run(
531            CLEANUP,
532            &[
533                &d.remote.checkout,
534                &d.remote.worktrees,
535                spool,
536                &tag,
537                if keep { "1" } else { "0" },
538            ],
539            None,
540        );
541    };
542    match out.code {
543        Some(0) => {}
544        Some(10) => {
545            cleanup(true);
546            return Err(refuse(
547                EXIT_DERIVE_FAILED,
548                format!(
549                    "the derive failed on {}; its worktree is kept there",
550                    host.id
551                ),
552            ));
553        }
554        other => {
555            cleanup(true);
556            return Err(refuse(
557                EXIT_UNAVAILABLE,
558                format!(
559                    "{}: the derive step ended with {other:?}: {}",
560                    host.id,
561                    out.stderr.trim()
562                ),
563            ));
564        }
565    }
566    let derived = out
567        .stdout
568        .lines()
569        .find_map(|l| l.strip_prefix("derived "))
570        .map(|s| s.trim().to_string());
571    let mut report = OffloadReport {
572        host: host.id.clone(),
573        tag: tag.clone(),
574        input: input.clone(),
575        derived: derived.clone(),
576        staged: Vec::new(),
577        waited_seconds: waited,
578        seconds: 0,
579        probe: probe_reports,
580    };
581    let Some(sha) = derived else {
582        cleanup(request.keep);
583        let _ = writeln!(
584            log,
585            "derive-offload: the derive changed nothing; {} is already current",
586            worktree.display()
587        );
588        report.seconds = started.elapsed().as_secs();
589        return Ok(report);
590    };
591
592    let back = scratch.path.join("back.bundle");
593    let dl = remote
594        .run_to_file(
595            DOWNLOAD,
596            &[spool, &format!("offload-{tag}-back.bundle")],
597            &back,
598        )
599        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{}: {e}", host.id)))?;
600    if !dl.ok() {
601        return Err(refuse(
602            EXIT_UNAVAILABLE,
603            format!(
604                "{}: the derived bundle could not be fetched: {}",
605                host.id,
606                dl.stderr.trim()
607            ),
608        ));
609    }
610    cleanup(request.keep);
611    let derived_ref = format!("refs/offload/{tag}-derived");
612    // The ref is this process's scratch and goes however the apply ends: a refused apply
613    // used to leave it behind, and the next offload of the same tree (same tag) then fetched
614    // a different derived commit (`stash create` stamps the time) onto it, which git refuses
615    // as a non-fast-forward with nothing on stderr under -q. `+` overwrites whatever an
616    // earlier process left, and the guard removes it on every path.
617    let _derived_guard = ScratchRef {
618        dir: &worktree,
619        name: derived_ref.clone(),
620    };
621    git(
622        &worktree,
623        &[
624            "fetch",
625            "-q",
626            &back.to_string_lossy(),
627            &format!("+{derived_ref}:{derived_ref}"),
628        ],
629    )?;
630    apply(&worktree, &input, &sha, &derived_ref, &mut report)?;
631    let _ = writeln!(
632        log,
633        "derive-offload: applied {} derived file(s), staged, not committed",
634        report.staged.len()
635    );
636    report.seconds = started.elapsed().as_secs();
637    Ok(report)
638}
639
640/// Check the derived snapshot against the input and stage what it changed.
641fn apply(
642    worktree: &Path,
643    input: &str,
644    sha: &str,
645    derived_ref: &str,
646    report: &mut OffloadReport,
647) -> Result<(), Refusal> {
648    apply_with(worktree, input, sha, derived_ref, report, git, git_ok)
649}
650
651/// [`apply`], running `git` and `git_ok` as given: the real ones in earnest, a test's own
652/// failures in a test.
653fn apply_with(
654    worktree: &Path,
655    input: &str,
656    sha: &str,
657    derived_ref: &str,
658    report: &mut OffloadReport,
659    git: impl Fn(&Path, &[&str]) -> Result<String, Refusal>,
660    git_ok: impl Fn(&Path, &[&str]) -> bool,
661) -> Result<(), Refusal> {
662    let status = git(worktree, &["diff", "--name-status", input, sha])?;
663    let deleted: Vec<&str> = status.lines().filter(|l| l.starts_with('D')).collect();
664    if !deleted.is_empty() {
665        return Err(refuse(
666            EXIT_DELETES,
667            format!(
668                "the derive deleted tracked files; nothing was applied — apply by hand from {derived_ref}:\n{}",
669                deleted.join("\n")
670            ),
671        ));
672    }
673    let names = git(worktree, &["diff", "--name-only", input, sha])?;
674    let files: Vec<String> = names
675        .lines()
676        .filter(|l| !l.is_empty())
677        .map(str::to_string)
678        .collect();
679    for chunk in files.chunks(200) {
680        let mut args = vec!["checkout", sha, "--"];
681        args.extend(chunk.iter().map(String::as_str));
682        git(worktree, &args)?;
683    }
684    if !git_ok(worktree, &["diff", "--quiet", sha, "--", "."]) {
685        return Err(refuse(
686            EXIT_UNAVAILABLE,
687            "the worktree does not match the derived tree after applying it",
688        ));
689    }
690    report.staged = files;
691    report.staged.sort();
692    Ok(())
693}

695/// A scratch directory removed when dropped.
696struct Scratch {
697    path: PathBuf,
698}
699
700impl Drop for Scratch {
701    fn drop(&mut self) {
702        let _ = std::fs::remove_dir_all(&self.path);
703    }
704}
705
706fn tempfile_dir() -> Result<Scratch, Refusal> {
707    let base = std::env::temp_dir();
708    for n in 0..100u32 {
709        let path = base.join(format!("derive-offload.{}.{n}", std::process::id()));
710        if std::fs::create_dir(&path).is_ok() {
711            return Ok(Scratch { path });
712        }
713    }
714    Err(refuse(
715        EXIT_UNAVAILABLE,
716        "no scratch directory could be made",
717    ))
718}
