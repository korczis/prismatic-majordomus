1//! PR Drain: every open pull request as the integration lane assessed it, narrowed by what
2//! the reader picks, with one of them inspected in full, and the issue it carries.
3//!
4//! The queue (`integration.queue`) asks the forge and takes tens of seconds, so the screen
5//! paints first and the drain arrives as a panel ([`PANEL`]), as the Dashboard's queue does.
6//! Every control is a link or a GET form: the disposition and lane tabs, the search, a row,
7//! a dependency, an overlap, a linked issue. Each leads to this screen with the choice in its
8//! query, so the screen is complete without JavaScript and every view of it has an address.
9//!
10//! An issue the reader picks (`?issue=I2417`) is read from `plan.issues`, which answers in
11//! milliseconds, so its inspector is part of the first paint.

48/// What the reader picked: which disposition or lane, what to search for, which pull
49/// request and which issue to inspect. Each is optional; an issue id that is not one is
50/// dropped rather than echoed.
51#[derive(Debug, Clone, Default, PartialEq, Eq)]
52pub struct Filter {
53    /// A disposition, in the queue's own word (`needs_refresh`).
54    pub disposition: Option<String>,
55    /// A lane, in the queue's own word (`repair`).
56    pub lane: Option<String>,
57    /// Text a title, branch, author or number must contain.
58    pub q: Option<String>,
59    /// The pull request to inspect.
60    pub pr: Option<u64>,
61    /// The plan issue to inspect (`I2417`).
62    pub issue: Option<String>,
63}
64
65impl Filter {
66    /// The filter a query names.
67    ///
68    /// ```text
69    /// let f = Filter::from_query(&[("pr".into(), "820".into()), ("issue".into(), "x".into())]);
70    /// assert_eq!((f.pr, f.issue), (Some(820), None), "an issue id that is not one is dropped");
71    /// ```
72    pub fn from_query(query: &[(String, String)]) -> Self {
73        let get = |k: &str| {
74            query
75                .iter()
76                .find(|(key, _)| key == k)
77                .map(|(_, v)| v.trim().to_string())
78                .filter(|v| !v.is_empty())
79        };
80        Filter {
81            disposition: get("disposition").filter(|w| is_word(w)),
82            lane: get("lane").filter(|w| is_word(w)),
83            q: get("q"),
84            pr: get("pr").and_then(|v| v.trim_start_matches('#').parse().ok()),
85            issue: get("issue").filter(|i| is_issue_id(i)),
86        }
87    }
88
89    /// This filter with one parameter set to `value`, or cleared when `value` is `None`, as
90    /// the query of a link to this screen or to `base`.
91    fn href(&self, base: &str, key: &str, value: Option<&str>) -> String {
92        let mut pairs: Vec<(&str, String)> = Vec::new();
93        for k in KEYS {
94            let v = if *k == key {
95                value.map(str::to_string)
96            } else {
97                self.get(k)
98            };
99            if let Some(v) = v {
100                pairs.push((k, v));
101            }
102        }
103        if pairs.is_empty() {
104            return base.to_string();
105        }
106        let query = pairs
107            .iter()
108            .map(|(k, v)| format!("{k}={}", encode(v)))
109            .collect::<Vec<_>>()
110            .join("&");
111        format!("{base}?{query}")
112    }
113
114    fn get(&self, key: &str) -> Option<String> {
115        match key {
116            "disposition" => self.disposition.clone(),
117            "lane" => self.lane.clone(),
118            "q" => self.q.clone(),
119            "pr" => self.pr.map(|n| n.to_string()),
120            "issue" => self.issue.clone(),
121            _ => None,
122        }
123    }
124
125    /// Does this assessment pass the filter? The pull request inspected is not a filter: it
126    /// is shown whatever the tabs say.
127    fn admits(&self, a: &Value) -> bool {
128        let word = |k: &str, want: &Option<String>| {
129            want.as_deref()
130                .is_none_or(|w| a.get(k).and_then(Value::as_str) == Some(w))
131        };
132        let found = self.q.as_deref().is_none_or(|q| {
133            let q = q.to_lowercase();
134            let number = number(a);
135            ["title", "head_ref", "author"]
136                .iter()
137                .any(|k| text(a, k).to_lowercase().contains(&q))
138                || number.is_some_and(|n| format!("#{n}") == q || n.to_string() == q)
139        });
140        word("disposition", &self.disposition) && word("lane", &self.lane) && found
141    }
142}

144/// A word of the queue's vocabulary: lower case, digits and underscores, nothing a link
145/// could carry anything else in.
146fn is_word(w: &str) -> bool {
147    !w.is_empty()
148        && w.len() <= 40
149        && w.bytes()
150            .all(|b| b.is_ascii_lowercase() || b == b'_' || b.is_ascii_digit())
151}
152
153/// A plan issue id: `I` and four digits.
154fn is_issue_id(i: &str) -> bool {
155    i.len() == 5 && i.starts_with('I') && i[1..].bytes().all(|b| b.is_ascii_digit())
156}
157
158/// A query value, percent-encoded except for the characters a value may carry as they are.
159fn encode(v: &str) -> String {
160    v.bytes()
161        .map(|b| match b {
162            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
163                (b as char).to_string()
164            }
165            other => format!("%{other:02X}"),
166        })
167        .collect()
168}

170/// The plan issues a pull request names, in its title or its branch, in the order they
171/// first appear: a convention of this repository's titles (`(I2417)`), read and not assumed,
172/// so a pull request that names none links to none.
173pub fn issues_named(a: &Value) -> Vec<String> {
174    let mut found: Vec<String> = Vec::new();
175    for field in [text(a, "title"), text(a, "head_ref").to_uppercase()] {
176        let bytes = field.as_bytes();
177        for (at, _) in field.match_indices('I') {
178            let id = field.get(at..at + 5).unwrap_or_default();
179            let bounded = at
180                .checked_sub(1)
181                .and_then(|p| bytes.get(p))
182                .is_none_or(|b| !b.is_ascii_alphanumeric())
183                && bytes.get(at + 5).is_none_or(|b| !b.is_ascii_alphanumeric());
184            if is_issue_id(id) && bounded && !found.iter().any(|f| f == id) {
185                found.push(id.to_string());
186            }
187        }
188    }
189    found
190}

264/// [`drain`] with the executor's trail (`integration.events`): the inspected pull request's
265/// own events, newest first, and the most recent events of the lane.
266pub fn drain_with(queue: &Answer, events: Option<&Answer>, filter: &Filter) -> El {
267    // newest first: the trail is recorded oldest first
268    let trail: Vec<Value> = events
269        .filter(|e| unanswered(e).is_none())
270        .map(|e| {
271            let mut all = list(e, "/events");
272            all.reverse();
273            all
274        })
275        .unwrap_or_default();
276    let body = unanswered(queue).unwrap_or_else(|| {
277        if queue.ok().and_then(|v| v.get("observed")) != Some(&Value::Bool(true)) {
278            return nothing(&format!(
279                "The forge was not observed in this checkout: {}",
280                queue
281                    .ok()
282                    .and_then(|v| v.get("reason"))
283                    .and_then(Value::as_str)
284                    .unwrap_or("no reason was given")
285            ));
286        }
287        let all = list(queue, "/queue/assessments");
288        let repository = queue
289            .ok()
290            .and_then(|v| v.pointer("/queue/repository"))
291            .and_then(Value::as_str)
292            .unwrap_or_default()
293            .to_string();
294        let shown: Vec<&Value> = all.iter().filter(|a| filter.admits(a)).collect();
295        let inspected = filter.pr.map(|n| {
296            all.iter().find(|a| number(a) == Some(n)).map_or_else(
297                || {
298                    nothing(&format!(
299                        "#{n} is not among the open pull requests the queue observed."
300                    ))
301                },
302                |a| {
303                    let own: Vec<Value> = trail
304                        .iter()
305                        .filter(|e| e.get("pr").and_then(Value::as_u64) == Some(n))
306                        .take(TRAIL)
307                        .cloned()
308                        .collect();
309                    inspector(a, &all, &repository, filter)
310                        .child(
311                            el("h4")
312                                .class("mj-ui-panel-title")
313                                .text("What the executor did"),
314                        )
315                        .child(if events.is_none() {
316                            nothing("The executor's trail was not asked.")
317                        } else if own.is_empty() {
318                            nothing("The executor has recorded nothing about it.")
319                        } else {
320                            trail_of(&own, filter)
321                        })
322                },
323            )
324        });
325        el("div")
326            .class("mj-ui-stack")
327            .child(tallies(queue))
328            .when(events.is_some(), |s| s.child(throughput(&trail)))
329            .child(next_up(queue, filter))
330            .child(tabs(queue, filter))
331            .child(search(filter))
332            .children(inspected)
333            .child(rows(&shown, filter))
334            .child(el("p").class("mj-ui-line").text(format!(
335                "{} of {} open pull requests shown · observed {}",
336                shown.len(),
337                all.len(),
338                queue
339                    .ok()
340                    .and_then(|v| v.pointer("/queue/observed_at"))
341                    .and_then(Value::as_str)
342                    .map(|t| t.replace('T', " "))
343                    .unwrap_or_else(|| "at an unknown time".into())
344            )))
345            .child(blockers(&all, filter))
346            .child(dependency_graph(&all, &repository, filter))
347            .when(events.is_some(), |s| {
348                let recent: Vec<Value> = trail
349                    .iter()
350                    .filter(|e| e.get("pr").and_then(Value::as_u64).is_some())
351                    .take(TRAIL)
352                    .cloned()
353                    .collect();
354                s.child(
355                    el("div")
356                        .attr("id", "activity")
357                        .child(el("h3").class("mj-ui-panel-title").text("Recent activity"))
358                        .child(if recent.is_empty() {
359                            nothing("The executor has recorded nothing about a pull request.")
360                        } else {
361                            trail_of(&recent, filter)
362                        }),
363                )
364            })
365    });
366    let mut answers = vec![queue];
367    answers.extend(events);
368    panel_to(
369        "Pull-request drain",
370        "What the integration lane would merge, refresh, repair or wait for, and why.",
371        Some(ROUTE),
372        body,
373        &answers,
374    )
375}

377/// Merges per day: how many pull requests the executor merged on each of the [`DAYS`] days
378/// that end on the day of the trail's newest event, from its `merge_succeeded` events. The
379/// window ends where the record ends, not at the reader's clock, so the chart says what was
380/// recorded and an idle week is a row of zeros the trail itself vouches for. Each day's count
381/// is written beside its bar, and a table carries the same numbers.
382pub fn merges_per_day(trail_newest_first: &[Value]) -> Vec<(String, u64)> {
383    let day_of = |e: &Value| crate::peers::parse_rfc3339(&text(e, "at")).map(|s| s / 86_400);
384    let Some(last) = trail_newest_first.iter().filter_map(day_of).max() else {
385        return Vec::new();
386    };
387    let first = last.saturating_sub(DAYS - 1);
388    let mut counts = vec![0u64; (last - first + 1) as usize];
389    for e in trail_newest_first {
390        if text(e, "action") != "merge_succeeded" {
391            continue;
392        }
393        if let Some(d) = day_of(e).filter(|d| (first..=last).contains(d)) {
394            counts[(d - first) as usize] += 1;
395        }
396    }
397    counts
398        .into_iter()
399        .enumerate()
400        .map(|(i, n)| {
401            let at =
402                std::time::UNIX_EPOCH + std::time::Duration::from_secs((first + i as u64) * 86_400);
403            (super::view::day(&crate::peers::rfc3339(at)), n)
404        })
405        .collect()
406}

408fn throughput(trail_newest_first: &[Value]) -> El {
409    let days = merges_per_day(trail_newest_first);
410    let body = if days.iter().all(|(_, n)| *n == 0) {
411        nothing("The executor recorded no merge in the days its trail covers.")
412    } else {
413        let max = days.iter().map(|(_, n)| *n).max().unwrap_or(1).max(1);
414        let w = 100 / days.len().max(1) as u64;
415        let bars: String = days
416            .iter()
417            .enumerate()
418            .map(|(i, (day, n))| {
419                let h = n * 40 / max;
420                format!(
421                    r#"<rect x="{x}" y="{y}" width="{bw}" height="{h}" rx="1" fill="currentColor"><title>{day}: {n}</title></rect><text x="{tx}" y="{ty}" font-size="4" text-anchor="middle" fill="currentColor">{n}</text>"#,
422                    x = i as u64 * w + 1,
423                    y = 45 - h,
424                    bw = w.saturating_sub(2).max(1),
425                    tx = i as u64 * w + w / 2,
426                    ty = 44 - h,
427                )
428            })
429            .collect();
430        let rows = days
431            .iter()
432            .map(|(day, n)| {
433                el("tr")
434                    .child(el("td").class("mj-ui-mono").text(day.clone()))
435                    .child(el("td").class("mj-ui-mono").text(n.to_string()))
436            })
437            .collect::<Vec<_>>();
438        el("div")
439            .child(
440                el("span")
441                    .class("mj-tone--green")
442                    .attr("role", "img")
443                    .attr(
444                        "aria-label",
445                        format!(
446                            "Merges per day from {} to {}: {}",
447                            days.first().map(|d| d.0.clone()).unwrap_or_default(),
448                            days.last().map(|d| d.0.clone()).unwrap_or_default(),
449                            days.iter().map(|(_, n)| n.to_string()).collect::<Vec<_>>().join(", ")
450                        ),
451                    )
452                    .raw(format!(
453                        r#"<svg viewBox="0 0 100 46" width="100%" height="120" preserveAspectRatio="none" focusable="false">{bars}</svg>"#
454                    )),
455            )
456            .child(table(&["Day", "Merges"], rows, "No day."))
457    };
458    el("div")
459        .attr("id", "throughput")
460        .child(el("h3").class("mj-ui-panel-title").text("Merge throughput"))
461        .child(body)
462}

464/// Events of the executor's trail as a timeline, newest first: what it did, to which pull
465/// request (a link to its inspector), why, and who did it.
466fn trail_of(events: &[Value], filter: &Filter) -> El {
467    super::view::timeline(
468        events
469            .iter()
470            .map(|e| {
471                let pr = e.get("pr").and_then(Value::as_u64);
472                let action = text(e, "action").replace('_', " ");
473                let title = match pr {
474                    Some(n) => format!("#{n} {action}"),
475                    None => action,
476                };
477                let reasons = strings(e, "reasons").join(", ");
478                let detail = el("span")
479                    .class("mj-ui-dim")
480                    .when(pr.is_some(), |d| {
481                        d.child(
482                            el("a")
483                                .class("mj-ui-more")
484                                .attr(
485                                    "href",
486                                    format!(
487                                        "{}#pr",
488                                        filter.href(
489                                            ROUTE,
490                                            "pr",
491                                            pr.map(|n| n.to_string()).as_deref()
492                                        )
493                                    ),
494                                )
495                                .text("open"),
496                        )
497                    })
498                    .child(el("span").text(format!(
499                        " {}{}{}",
500                        text(e, "detail"),
501                        if reasons.is_empty() { String::new() } else { format!(" · {reasons}") },
502                        Some(text(e, "actor"))
503                            .filter(|a| !a.is_empty())
504                            .map(|a| format!(" · {a}"))
505                            .unwrap_or_default()
506                    )));
507                (text(e, "at"), title, detail)
508            })
509            .collect(),
510    )
511}

542/// What the lane does next: the pull request it would merge, those it would refresh, those
543/// starving, and the last merge it recorded. Each pull request is a link to its inspector.
544fn next_up(queue: &Answer, filter: &Filter) -> El {
545    let v = queue.ok();
546    let links = |ns: &[u64]| -> El {
547        if ns.is_empty() {
548            return el("span").class("mj-ui-dim").text("none");
549        }
550        el("span").children(ns.iter().take(12).map(|n| {
551            el("a")
552                .class("mj-ui-chip mj-ui-chip--link")
553                .attr("href", filter.href(ROUTE, "pr", Some(&n.to_string())))
554                .child(el("code").class("mj-ui-chip-value").text(format!("#{n}")))
555        }))
556    };
557    // a list the queue did not answer is unknown; an empty one it answered is none
558    let listed = |p: &str| -> El {
559        match v
560            .and_then(|v| v.pointer(p))
561            .and_then(Value::as_array)
562            .map(|xs| xs.iter().filter_map(Value::as_u64).collect::<Vec<_>>())
563        {
564            Some(ns) => links(&ns),
565            None => el("span").class("mj-ui-dim").text("unknown"),
566        }
567    };
568    let next_merge: Vec<u64> = v
569        .and_then(|v| v.pointer("/queue/next_merge"))
570        .and_then(|m| {
571            m.as_u64()
572                .or_else(|| m.get("number").and_then(Value::as_u64))
573        })
574        .into_iter()
575        .collect();
576    let last = v.and_then(|v| v.get("last_merge")).filter(|m| !m.is_null());
577    let fact = |label: &str, value: El| {
578        el("div")
579            .class("mj-ui-fact")
580            .child(el("dt").text(label))
581            .child(el("dd").child(value))
582    };
583    el("dl")
584        .class("mj-ui-facts")
585        .attr("aria-label", "What the lane does next")
586        .child(fact("Next merge", links(&next_merge)))
587        .child(fact("Next refresh", listed("/queue/next_refresh")))
588        .child(fact("Starving", listed("/queue/starving")))

1056/// What depends on what: an edge from a pull request to each open pull request it depends
1057/// on, as the queue confirmed or inferred it. A dependency already merged is not drawn: it
1058/// no longer holds anything.
1059fn dependency_graph(all: &[Value], repository: &str, filter: &Filter) -> El {
1060    let mut edges: Vec<(String, String)> = Vec::new();
1061    let mut fallback = Vec::new();
1062    for a in all {
1063        let Some(n) = number(a) else { continue };
1064        for d in a
1065            .get("dependencies")
1066            .and_then(Value::as_array)
1067            .into_iter()
1068            .flatten()
1069        {
1070            let Some(m) = d.get("number").and_then(Value::as_u64) else {
1071                continue;
1072            };
1073            if d.get("state").and_then(Value::as_str) == Some("merged") {
1074                continue;
1075            }
1076            edges.push((n.to_string(), m.to_string()));
1077            fallback.push(
1078                el("tr")
1079                    .child(el("td").class("mj-ui-mono").text(format!("#{n}")))
1080                    .child(el("td").class("mj-ui-mono").text(format!("#{m}")))
1081                    .child(el("td").class("mj-ui-dim").text(format!(
1082                        "{} · {}",
1083                        text(d, "certainty"),
1084                        text(d, "state")
1085                    ))),
1086            );
1087        }
1088    }
1089    let mut nodes: Vec<GraphNode> = all
1090        .iter()
1091        .filter_map(|a| {
1092            let n = number(a)?;
1093            Some(GraphNode {
1094                id: n.to_string(),
1095                label: format!("#{n}"),
1096                tone: status_tone(&text(a, "disposition")),
1097                href: format!("{}#pr", filter.href(ROUTE, "pr", Some(&n.to_string()))),
1098                kind: "main",
1099            })
1100        })
1101        .collect();
1102    // a dependency the queue does not hold (it is not open against the base, or the forge
1103    // did not list it) is still drawn, as a small node leading to the forge
1104    let held: Vec<String> = nodes.iter().map(|n| n.id.clone()).collect();
1105    let mut outside: Vec<String> = edges
1106        .iter()
1107        .map(|(_, m)| m.clone())
1108        .filter(|m| !held.contains(m))
1109        .collect();
1110    crate::order::canonical_strings(&mut outside);
1111    outside.dedup();
1112    nodes.extend(outside.into_iter().map(|m| GraphNode {
1113        href: if repository.is_empty() {
1114            ROUTE.to_string()
1115        } else {
1116            format!("https://github.com/{repository}/pull/{m}")
1117        },
1118        label: format!("#{m}"),
1119        tone: "violet",
1120        kind: "leaf",
1121        id: m,
1122    }));
1123    el("div")
1124        .attr("id", "dependencies")
1125        .child(
1126            el("h3")
1127                .class("mj-ui-panel-title")
1128                .text("What depends on what"),
1129        )
1130        .child(if edges.is_empty() {
1131            nothing("No open pull request depends on another open one.")
1132        } else {
1133            graph(
1134                "Dependencies between open pull requests",
1135                Layout::Directed,
1136                &nodes,
1137                &edges,
1138                table(
1139                    &["Pull request", "Depends on", "How it is known"],
1140                    fallback,
1141                    "None.",
1142                ),
1143            )
1144        })
1145}
