1//! The crate's rustdoc tree, judged against the crate it documents.
2//!
3//! # What it is for
4//!
5//! `cargo doc` proves the documentation *compiles*: `-D warnings` refuses a broken
6//! intra-doc link and `missing_docs` refuses an undocumented export. It says nothing about
7//! the tree it writes — whether a page exists for every item the crate exports, whether a
8//! page is still there for an item that no longer exists, whether the tree was built from
9//! the commit it claims, whether its relative links resolve once it is copied under a
10//! mount, and whether it carries the machine it was built on. Those are properties of an
11//! artifact, and this module is where they become findings with a kind, a file and a remedy.
12//!
13//! # How a route is derived
14//!
15//! Nothing here lists a page. Every expected file is derived from
16//! [`super::source::Inventory`] by rustdoc's own rules, measured on rustdoc 1.98 against
17//! this crate:
18//!
19//! ```text
20//! module                 <crate>/<segments>/index.html
21//! struct enum trait fn   <crate>/<module segments>/{struct,enum,trait,fn}.<Name>.html
22//! const static type union                           {constant,static,type,union}.<Name>.html
23//! exported macro         <crate>/macro.<name>.html, at the root whatever module declares it
24//! method                 the owner's page #method.<n>; a trait's required one #tymethod.<n>
25//! field / variant        the owner's page #structfield.<n> / #variant.<N>
26//! associated type/const  the trait's page #associatedtype.<N> / #associatedconstant.<N>
27//! ```
28//!
29//! An item exported only through a `pub use` of a private module is documented where the
30//! re-export offers it, under the name it offers it by, and its original location holds a
31//! redirect stub; an item whose module chain is public is documented where it is declared.
32//! The crate segment is the library target's name, read from `Cargo.toml`. A binary target
33//! is a crate of its own in the same tree, documented with its private items, and is read
34//! by the same walk.
35//!
36//! # What the verifier judges
37//!
38//! One pass over the tree. Every file is classified — an item page, a redirect stub, a
39//! rustdoc system file, the producer's landing — and every HTML page is tokenised once for
40//! its links. The findings are typed ([`RustdocFindingKind`]) and name the file and the
41//! subject; the verdict is `clean`, `findings` or `no_tree`, and a tree that is absent is
42//! never reported clean.
43//!
44//! Two kinds of unresolved reference are rustdoc's and not this crate's, and each is a
45//! named class ([`RustdocLinkClass`]) decided by where the reference sits, counted in the
46//! report and never a finding: a link inside documentation rustdoc copied from another
47//! crate's trait, and the implementors script of a trait page when rustdoc wrote none. Any
48//! other relative reference that resolves to nothing is a `link` finding.
49//!
50//! ```text
51//! crate --syn--> Inventory --routes--> expected pages ─┐
52//!                                                       ├─> findings, counts, module routes
53//! target/web/rustdoc/ --walk--> files, pages, links ───┘
54//! ```

83/// The documented targets of a crate, as its manifest declares them.
84#[derive(Debug, Clone, PartialEq, Eq)]
85pub struct Targets {
86    /// The library's crate name: `[lib] name`, else the package name with `-` as `_`. This
87    /// is the directory rustdoc writes the library into and the first segment of every
88    /// route.
89    pub lib: String,
90    /// The library's root file, crate-relative: `[lib] path`, else `src/lib.rs`.
91    pub lib_root: String,
92    /// Every binary cargo documents: its crate name and its root file, crate-relative.
93    pub bins: Vec<(String, String)>,
94}
95
96/// Read the targets from `Cargo.toml`, with the same literal subset discipline the rest of
97/// the executable reads manifests with: `name`, `path` and `doc` in `[package]`, `[lib]`
98/// and `[[bin]]`, and nothing else. A binary cargo would not document — `doc = false`, or
99/// one named as the library — is left out, as cargo leaves it out.
100///
101/// This is the one place the library's crate name is decided; every route starts with it.
102pub fn targets(crate_dir: &Path) -> Result<Targets, Error> {
103    let manifest = crate_dir.join("Cargo.toml");
104    let text = std::fs::read_to_string(&manifest).map_err(|e| Error::InvalidSource {
105        path: "Cargo.toml".into(),
106        reason: format!("cannot be read under {}: {e}", crate_dir.display()),
107    })?;
108    #[derive(Default)]
109    struct Section {
110        name: Option<String>,
111        path: Option<String>,
112        doc: bool,
113    }
114    let mut package = Section::default();
115    let mut lib = Section::default();
116    let mut bins: Vec<Section> = Vec::new();
117    let mut current = "";
118    for line in text.lines() {
119        let line = line.split('#').next().unwrap_or("").trim();
120        if line.starts_with('[') {
121            current = match line {
122                "[package]" => "package",
123                "[lib]" => "lib",
124                "[[bin]]" => {
125                    bins.push(Section {
126                        doc: true,
127                        ..Section::default()
128                    });
129                    "bin"
130                }
131                _ => "",
132            };
133            continue;
134        }
135        let Some((key, value)) = line.split_once('=') else {
136            continue;
137        };
138        let (key, value) = (key.trim(), value.trim().trim_matches('"').to_string());
139        let section = match current {
140            "package" => &mut package,
141            "lib" => &mut lib,
142            "bin" => match bins.last_mut() {
143                Some(b) => b,
144                None => continue,
145            },
146            _ => continue,
147        };
148        match key {
149            "name" => section.name = Some(value),
150            "path" => section.path = Some(value),
151            "doc" => section.doc = value != "false",
152            _ => {}
153        }
154    }
155    let package_name = package.name.ok_or_else(|| Error::InvalidSource {
156        path: "Cargo.toml".into(),
157        reason: "declares no [package] name, so no crate name can be derived".into(),
158    })?;
159    let lib_name = lib.name.unwrap_or_else(|| package_name.replace('-', "_"));
160    let lib_root = lib.path.unwrap_or_else(|| "src/lib.rs".into());
161    let mut documented = Vec::new();
162    if bins.is_empty() && crate_dir.join("src/main.rs").is_file() {
163        // cargo's own discovery: a package with a src/main.rs and no [[bin]] builds one
164        // binary named after the package
165        let krate = package_name.replace('-', "_");
166        if krate != lib_name {
167            documented.push((krate, "src/main.rs".to_string()));
168        }
169    }
170    for bin in bins.into_iter().filter(|b| b.doc) {
171        let Some(name) = bin.name else { continue };
172        let krate = name.replace('-', "_");
173        if krate == lib_name {
174            continue;
175        }
176        let path = bin.path.unwrap_or_else(|| {
177            if name == package_name {
178                "src/main.rs".into()
179            } else {
180                format!("src/bin/{name}.rs")
181            }
182        });
183        documented.push((krate, path));
184    }
185    Ok(Targets {
186        lib: lib_name,
187        lib_root,
188        bins: documented,
189    })
190}

214/// The file prefix rustdoc gives an item that owns a page of its own, or `None` for a kind
215/// that is documented on another item's page (or, a module, as a directory).
216pub fn page_prefix(kind: ItemKind) -> Option<&'static str> {
217    Some(match kind {
218        ItemKind::Struct => "struct",
219        ItemKind::Enum => "enum",
220        ItemKind::Union => "union",
221        ItemKind::Trait => "trait",
222        ItemKind::Function => "fn",
223        ItemKind::Constant => "constant",
224        ItemKind::Static => "static",
225        ItemKind::TypeAlias => "type",
226        ItemKind::Macro => "macro",
227        ItemKind::Module
228        | ItemKind::Method
229        | ItemKind::Field
230        | ItemKind::Variant
231        | ItemKind::AssociatedType
232        | ItemKind::AssociatedConstant => return None,
233    })
234}

236/// The page of an item documented at the crate path `at`, in the tree's crate directory
237/// `krate`. The first segment of `at` is the crate as the inventory names it and is
238/// replaced by `krate`, so the derivation holds whatever the library is called.
239///
240/// `None` for a kind that owns no page: see [`member`] for those.
241pub fn page(kind: ItemKind, at: &str, krate: &str) -> Option<Route> {
242    let segments: Vec<&str> = at.split("::").skip(1).collect();
243    let file = match kind {
244        ItemKind::Module => {
245            let mut parts = vec![krate];
246            parts.extend(&segments);
247            parts.push("index.html");
248            parts.join("/")
249        }
250        // `#[macro_export]` puts a macro at the crate root, whatever module it is written in
251        ItemKind::Macro => format!("{krate}/macro.{}.html", segments.last()?),
252        other => {
253            let prefix = page_prefix(other)?;
254            let (name, module) = segments.split_last()?;
255            let mut parts = vec![krate];
256            parts.extend(module);
257            parts.push("");
258            format!("{}{prefix}.{name}.html", parts.join("/"))
259        }
260    };
261    Some(Route {
262        file,
263        fragment: None,
264    })
265}

267/// The anchor of a member on its owner's page: a method, an associated function, a field,
268/// a variant, an associated type or constant. `owner` is the kind of the item that owns
269/// it, which is what decides a trait method's anchor; `provided` is whether the trait
270/// supplies it ([`Item::provided`]).
271///
272/// `None` for a kind that is not a member, or a member of an owner that owns no page.
273pub fn member(
274    kind: ItemKind,
275    name: &str,
276    owner: ItemKind,
277    provided: bool,
278    owner_page: &Route,
279) -> Option<Route> {
280    let anchor = match (kind, owner) {
281        (ItemKind::Method | ItemKind::Function, ItemKind::Trait) if !provided => "tymethod",
282        (
283            ItemKind::Method | ItemKind::Function,
284            ItemKind::Trait | ItemKind::Struct | ItemKind::Enum | ItemKind::Union,
285        ) => "method",
286        (ItemKind::Field, ItemKind::Struct | ItemKind::Union) => "structfield",
287        (ItemKind::Variant, ItemKind::Enum) => "variant",
288        (ItemKind::AssociatedType, ItemKind::Trait) => "associatedtype",
289        (ItemKind::AssociatedConstant, ItemKind::Trait) => "associatedconstant",
290        _ => return None,
291    };
292    Some(Route {
293        file: owner_page.file.clone(),
294        fragment: Some(format!("{anchor}.{name}")),
295    })
296}

298/// The routes of one inventory: where rustdoc documents each of its items.
299///
300/// Built once over the inventory, so a route is a lookup and not a scan: the check derives
301/// a route for every item of the crate, and a search of the item list per item would be
302/// quadratic in the crate's size.
303pub struct Router<'a> {
304    krate: &'a str,
305    /// Private items are documented too: what cargo does for a binary target.
306    everything: bool,
307    modules: HashSet<&'a str>,
308    owners: HashMap<&'a str, &'a Item>,
309    /// Modules whose chain from the crate root is `pub` all the way down.
310    public_modules: HashSet<&'a str>,
311    /// A re-exported path, to every alias it is offered under, in canonical order. A glob's
312    /// alias is the re-exporting module itself: each item keeps its own remainder.
313    aliases: HashMap<&'a str, Vec<String>>,
314}
315
316impl<'a> Router<'a> {
317    /// The router of a library: exported items only, re-exports followed.
318    pub fn library(inventory: &'a Inventory, krate: &'a str) -> Self {
319        Router::new(inventory, krate, false)
320    }
321
322    /// The router of a binary: cargo documents a binary with its private items, so every
323    /// item the walk found is documented where it is declared.
324    pub fn binary(inventory: &'a Inventory, krate: &'a str) -> Self {
325        Router::new(inventory, krate, true)
326    }
327
328    fn new(inventory: &'a Inventory, krate: &'a str, everything: bool) -> Self {
329        let root = inventory
330            .items
331            .first()
332            .map(|i| i.path.as_str())
333            .unwrap_or_default();
334        let mut modules = HashSet::new();
335        let mut owners = HashMap::new();
336        for item in &inventory.items {
337            match item.kind {
338                ItemKind::Module => {
339                    modules.insert(item.path.as_str());
340                }
341                ItemKind::Struct | ItemKind::Enum | ItemKind::Union | ItemKind::Trait => {
342                    owners.entry(item.path.as_str()).or_insert(item);
343                }
344                _ => {}
345            }
346        }
347        // the walk visits a parent before its children, so one pass decides every chain
348        let mut public_modules = HashSet::new();
349        for item in inventory
350            .items
351            .iter()
352            .filter(|i| i.kind == ItemKind::Module)
353        {
354            let public = item.path == root
355                || (item.declared_pub && public_modules.contains(item.owner.as_str()));
356            if public {
357                public_modules.insert(item.path.as_str());
358            }
359        }
360        let mut aliases: HashMap<&str, Vec<String>> = HashMap::new();
361        for (alias, target) in &inventory.reexport_aliases {
362            // `m::*` offers the items of its target in `m` itself
363            let offered = alias.strip_suffix("::*").unwrap_or(alias);
364            aliases
365                .entry(target.as_str())
366                .or_default()
367                .push(offered.to_string());
368        }
369        for list in aliases.values_mut() {
370            crate::order::canonical_strings(list);
371            list.dedup();
372        }
373        Router {
374            krate,
375            everything,
376            modules,
377            owners,
378            public_modules,
379            aliases,
380        }
381    }
382
383    /// Is this item documented at all — exported by a library, or any item of a binary?
384    pub fn documented(&self, item: &Item) -> bool {
385        self.everything || item.exported
386    }
387
388    /// The crate paths an item is documented at: its own when its module chain is public
389    /// (or the target is a binary), else every alias a `pub use` offers it or one of its
390    /// ancestors under. Empty for an item that is not documented.
391    pub fn documented_at(&self, item: &Item) -> Vec<String> {
392        if !self.documented(item) {
393            return Vec::new();
394        }
395        if self.everything || self.in_public_chain(item) {
396            return vec![item.path.clone()];
397        }
398        // the nearest re-export that reaches the item: the item itself, or a module above
399        // it that a `pub use` carried out whole
400        let mut at = item.path.as_str();
401        loop {
402            if let Some(list) = self.aliases.get(at) {
403                let rest = &item.path[at.len()..];
404                return list.iter().map(|alias| format!("{alias}{rest}")).collect();
405            }
406            match at.rsplit_once("::") {
407                Some((parent, _)) => at = parent,
408                None => return Vec::new(),
409            }
410        }
411    }
412
413    fn in_public_chain(&self, item: &Item) -> bool {
414        match item.kind {
415            ItemKind::Module => self.public_modules.contains(item.path.as_str()),
416            ItemKind::Macro => true,
417            _ => item.declared_pub && self.public_modules.contains(item.owner.as_str()),
418        }
419    }
420
421    /// Does this item own a page of its own? A module does, and so does every item with a
422    /// page prefix that is declared in a module — a type, a trait, a free function, a
423    /// constant, a static, an alias — and an exported macro. A member is an anchor: an
424    /// associated function belongs to the type it is implemented on.
425    pub fn owns_page(&self, item: &Item) -> bool {
426        match item.kind {
427            ItemKind::Module | ItemKind::Macro => true,
428            kind => page_prefix(kind).is_some() && self.modules.contains(item.owner.as_str()),
429        }
430    }
431
432    /// Every route of an item: one per place it is documented. Empty when it is not
433    /// documented, or when it is a member of something this inventory does not hold (an
434    /// `impl` written for a type declared in another module).
435    pub fn routes(&self, item: &Item) -> Vec<Route> {
436        if !self.documented(item) {
437            return Vec::new();
438        }
439        if self.owns_page(item) {
440            return self
441                .documented_at(item)
442                .iter()
443                .filter_map(|at| page(item.kind, at, self.krate))
444                .collect();
445        }
446        let Some(owner) = self.owners.get(item.owner.as_str()) else {
447            return Vec::new();
448        };
449        self.routes(owner)
450            .iter()
451            .filter_map(|p| member(item.kind, &item.name, owner.kind, item.provided, p))
452            .collect()
453    }
454}

458/// What the tree was judged to be.
459#[derive(
460    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,

462#[serde(rename_all = "snake_case")]
463pub enum RustdocVerdict {
464    /// The tree holds every page the crate derives and nothing wrong was found.
465    Clean,
466    /// At least one finding stands.
467    Findings,
468    /// Nothing could be judged: the tree, or the crate that documents it, is absent. Never
469    /// a pass — the producer has not run, and a check that cannot see its subject says so.
470    NoTree,
471}

477        match self {
478            RustdocVerdict::Clean => 0,
479            RustdocVerdict::Findings => 10,
480            RustdocVerdict::NoTree => 12,
481        }
482    }

485/// What is wrong with the tree, as a stable code.
486#[derive(
487    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
488)]
489#[serde(rename_all = "kebab-case")]
490pub enum RustdocFindingKind {
491    /// An exported item that owns a page has no page at its derived route.
492    MissingPage,
493    /// An item page no item of the crate derives: a leftover, or a route rule this check
494    /// has wrong.
495    OrphanPage,
496    /// The tree declares it was built from a revision other than the repository's HEAD.
497    Stale,
498    /// The tree carries no declaration of what it is or what it was built from.
499    Undeclared,
500    /// The library's index page is absent, or does not say it is the library's.
501    CrateIndex,
502    /// A stylesheet, script, font or search file the library's index page needs is absent.
503    Asset,
504    /// A relative link of a page resolves to no file of the tree.
505    Link,
506    /// A file names a path of the machine it was built on.
507    MachinePath,
508    /// A file carries the shape of a credential.
509    Secret,
510}

512impl RustdocFindingKind {
513    /// The code as it is written everywhere: `missing-page`.
514    pub fn as_str(self) -> &'static str {
515        match self {
516            RustdocFindingKind::MissingPage => "missing-page",
517            RustdocFindingKind::OrphanPage => "orphan-page",
518            RustdocFindingKind::Stale => "stale",
519            RustdocFindingKind::Undeclared => "undeclared",
520            RustdocFindingKind::CrateIndex => "crate-index",
521            RustdocFindingKind::Asset => "asset",
522            RustdocFindingKind::Link => "link",
523            RustdocFindingKind::MachinePath => "machine-path",
524            RustdocFindingKind::Secret => "secret",
525        }
526    }
527
528    /// What to do about it, in the imperative.
529    pub fn remedy(self) -> &'static str {
530        match self {
531            RustdocFindingKind::MissingPage => {
532                "rebuild the tree with scripts/rust-check --doc; if it persists, the item's route rule in quality::rustdoc is wrong for this rustdoc and must be corrected there, not excepted"
533            }
534            RustdocFindingKind::OrphanPage => {
535                "rustdoc never deletes a page it wrote before, so an item removed from the crate keeps its page in cargo's target/doc: run cargo clean --doc in the crate and rebuild with scripts/rust-check --doc; if it persists, the route rule in quality::rustdoc is wrong for this rustdoc and must be corrected there, not excepted"
536            }
537            RustdocFindingKind::Stale | RustdocFindingKind::Undeclared => {
538                "rebuild the tree with scripts/rust-check --doc, which writes surface.json with built_from = HEAD"
539            }
540            RustdocFindingKind::CrateIndex | RustdocFindingKind::Asset => {
541                "rebuild the tree from a clean target/doc: a partial copy of rustdoc's output is not a tree"
542            }
543            RustdocFindingKind::Link => {
544                "fix the link at its source: a doc comment that links a repository file by a path relative to the source file resolves nowhere under /rustdoc; use an intra-doc link, or the forge URL of the file"
545            }
546            RustdocFindingKind::MachinePath => {
547                "find what wrote the machine's path into the documentation — a doc comment, a constant's rendered value, a test fixture in a published source page — and make it repository-relative"
548            }
549            RustdocFindingKind::Secret => {
550                "remove the value at its source and rotate it; if the match is a name rather than a value, make the pattern in apps/majordomus-cli/src/quality/leak-patterns.tsv more precise, with a test, for every reader"
551            }
552        }
553    }

558        [

560            OrphanPage,
561            Stale,
562            Undeclared,
563            CrateIndex,
564            Asset,
565            Link,
566            MachinePath,
567            Secret,
568        ]
569    }

572/// One finding about the tree.
573#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
574pub struct RustdocFinding {
575    /// What is wrong.
576    pub kind: RustdocFindingKind,
577    /// The file of the tree it is about, tree-relative; empty when it is about the tree as a
578    /// whole.
579    pub file: String,
580    /// What exactly: the item path of a missing page, the target of a broken link, the
581    /// pattern that matched.
582    pub subject: String,
583    /// One sentence, specific to this occurrence.
584    pub message: String,
585    /// What to do, from the kind.
586    pub remedy: String,
587}

589impl RustdocFinding {
590    fn new(
591        kind: RustdocFindingKind,
592        file: impl Into<String>,
593        subject: impl Into<String>,
594        message: impl Into<String>,
595    ) -> Self {
596        RustdocFinding {
597            kind,

599            subject: subject.into(),
600            message: message.into(),
601            remedy: kind.remedy().to_string(),

605    /// The finding as one line, the way the terminal shows it.
606    pub fn line_summary(&self) -> String {
607        let at = if self.file.is_empty() {
608            String::new()
609        } else {
610            format!("{}: ", self.file)
611        };
612        format!("{at}{} - {}", self.subject, self.message)
613    }
614}

622/// A kind of unresolved reference that is rustdoc's and not the crate's: decided by where
623/// it sits on its page, counted, and never a finding.

625    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,

805/// Where the tree to judge is, or why there is none.
806#[derive(Debug, Clone, PartialEq, Eq)]
807pub enum Tree {
808    /// The tree, absolute, and how a report names it.
809    At {
810        /// The directory.
811        dir: PathBuf,
812        /// Its name in the report: repository-relative when inside the repository.
813        shown: String,
814        /// The mount it is published at, when the topology says.
815        mount: Option<String>,
816    },
817    /// There is no tree to judge, and why.
818    Absent {
819        /// The tree the report names, even though it is absent.
820        shown: String,
821        /// Why.
822        reason: String,
823    },
824}

826/// What the judgement is given: the repository, its crate, the tree and HEAD.
827#[derive(Debug, Clone)]
828pub struct Subject<'a> {
829    /// The repository root, absolute: what a file of the tree must never name.
830    pub root: &'a Path,
831    /// The crate directory, when the repository has one.
832    pub crate_dir: Option<&'a Path>,
833    /// The tree.
834    pub tree: Tree,
835    /// The repository's HEAD, when git could say.
836    pub head: Option<&'a str>,
837}
838
839/// Judge a tree: read the crate, derive every page, walk the tree once, report.
840///
841/// Errors only when the crate itself cannot be read — a file that does not parse, a
842/// manifest without a name — or when the leak definition compiled into the executable does
843/// not compile, because an expectation derived from half a crate, or a scan with nothing to
844/// look for, would be the one result this must not produce. An absent tree is not an error;
845/// it is the `no_tree` verdict.
846pub fn judge(subject: &Subject<'_>) -> Result<RustdocReport, Error> {
847    let shown = match &subject.tree {
848        Tree::At { shown, .. } | Tree::Absent { shown, .. } => shown.clone(),
849    };
850    let Some(crate_dir) = subject.crate_dir else {
851        return Ok(no_tree(
852            String::new(),
853            shown,
854            None,
855            subject.head,
856            Vec::new(),
857            "the repository carries no Rust crate, so there is nothing a rustdoc tree documents"
858                .into(),
859        ));
860    };
861    let targets = targets(crate_dir)?;
862    let library = Inventory::of_target(crate_dir, &targets.lib_root, &targets.lib)?;
863    let mut binaries = Vec::new();
864    for (name, root) in &targets.bins {
865        binaries.push((name.clone(), Inventory::of_target(crate_dir, root, name)?));
866    }
867    let expected = Expected::derive(&library, &targets.lib, &binaries);
868    match &subject.tree {
869        Tree::Absent { reason, .. } => Ok(no_tree(
870            targets.lib,
871            shown,
872            None,
873            subject.head,
874            expected.modules,
875            reason.clone(),
876        )),
877        Tree::At { dir, mount, .. } if !dir.is_dir() => Ok(no_tree(
878            targets.lib,
879            shown.clone(),
880            mount.clone(),
881            subject.head,
882            expected.modules,
883            format!(
884                "{shown} does not exist: the producer, {PRODUCER}, has not run in this checkout"
885            ),
886        )),
887        Tree::At { dir, mount, .. } => {
888            let leaks = Leaks::shipped().map_err(|reason| Error::InvalidSource {
889                path: super::leaks::SOURCE.into(),
890                reason,
891            })?;
892            Ok(verify(
893                dir,
894                &shown,
895                mount.clone(),
896                &expected,
897                subject.head,
898                subject.root,
899                leaks,
900            ))
901        }
902    }
903}

928/// Everything the crate says the tree must hold.
929#[derive(Debug, Clone, Default)]
930pub struct Expected {
931    /// The library's crate directory in the tree.
932    pub lib: String,
933    /// Every crate directory the tree documents: the library, then each binary.
934    pub crates: Vec<String>,
935    /// Every page an item owns, to the item it documents (the first one, when two derive it).
936    pub pages: BTreeMap<String, String>,
937    /// Where each page-owning item was declared, for a missing page's message.
938    pub declared_at: HashMap<String, String>,
939    /// Every exported module of the library, with its route, in canonical order.
940    pub modules: Vec<RustdocModuleRoute>,
941    /// Items the library exports.
942    pub exported_items: usize,
943    /// Page-owning items of every target.
944    pub page_owning_items: usize,
945}
946
947impl Expected {
948    /// Derive the expectation from the library's inventory, its crate name and every
949    /// binary's inventory.
950    pub fn derive(library: &Inventory, lib: &str, binaries: &[(String, Inventory)]) -> Self {
951        let mut out = Expected {
952            lib: lib.to_string(),
953            crates: vec![lib.to_string()],
954            ..Expected::default()
955        };
956        out.exported_items = library.items.iter().filter(|i| i.exported).count();
957        let router = Router::library(library, lib);
958        out.add(&router, library);
959        out.modules = library_modules(&router, library);
960        for (name, inventory) in binaries {
961            out.crates.push(name.clone());
962            out.add(&Router::binary(inventory, name), inventory);
963        }
964        out
965    }
966
967    fn add(&mut self, router: &Router<'_>, inventory: &Inventory) {
968        for item in &inventory.items {
969            if !router.documented(item) || !router.owns_page(item) {
970                continue;
971            }
972            self.page_owning_items += 1;
973            for route in router.routes(item) {
974                self.declared_at
975                    .entry(route.file.clone())
976                    .or_insert_with(|| format!("{}:{}", item.file, item.line));
977                self.pages.entry(route.file).or_insert(item.path.clone());
978            }
979        }
980    }
981}

983/// Every exported module of a library with the page it is documented on, in canonical order.
984/// A module owns a page, so each route here is also one of [`Expected::pages`]: a page the
985/// judgement requires of the tree and reports as `missing-page` when it is absent.
986fn library_modules(router: &Router<'_>, library: &Inventory) -> Vec<RustdocModuleRoute> {
987    let mut modules = Vec::new();
988    for item in library
989        .items
990        .iter()
991        .filter(|i| i.kind == ItemKind::Module && i.exported)
992    {
993        for route in router.routes(item) {
994            modules.push(RustdocModuleRoute {
995                path: item.path.clone(),
996                route: route.file,
997            });
998        }
999    }
1000    canonical(&mut modules);
1001    modules.dedup();
1002    modules
1003}

1006/// [`RustdocReport::modules`], answered without a tree to judge.
1007///
1008/// This is the one module-to-page mapping a consumer that links into the published reference
1009/// reads — the site's registry dataset ([`crate::site`]) among them — so a link to a module's
1010/// page is a link to a page [`judge`] holds the tree to, and never a route of the consumer's
1011/// own making. A module rustdoc gives no page — `pub(crate)`, private, or under a private
1012/// parent with no `pub use` of it — is absent from the answer, so it is never linked.

1020        &Router::library(&library, &targets.lib),
1021        &library,
1022    ))

1027/// What a file of the tree is.
1028#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1029enum Class {
1030    /// The producer's own: the landing page and the declaration at the root.
1031    Landing,
1032    /// rustdoc's own machinery.
1033    System,
1034    /// A page of an item, under a crate directory.
1035    ItemPage,
1036    /// A page that only redirects.
1037    Redirect,
1038    /// An HTML page outside every crate directory that is not rustdoc's.
1039    Stray,
1040    /// Anything else: not a page, not rustdoc's.
1041    Other,
1042}
1043
1044/// The top-level entries rustdoc writes for the whole tree rather than for a crate.
1045const SYSTEM_DIRS: &[&str] = &[
1046    "static.files",
1047    "search.index",
1048    "search.desc",
1049    "src",
1050    "trait.impl",
1051    "type.impl",
1052    "implementors",
1053];
1054const SYSTEM_ROOT_FILES: &[&str] = &[
1055    "help.html",
1056    "settings.html",
1057    "crates.js",
1058    "src-files.js",
1059    "source-files.js",
1060    // cargo's lock of target/doc, when a tree is copied with it
1061    ".lock",
1062];
1063
1064fn classify(rel: &str, crates: &HashSet<&str>, redirect: bool) -> Class {
1065    if rel == crate::web::discover::DECLARATION_FILE || rel == "index.html" {
1066        return Class::Landing;
1067    }
1068    let (first, rest) = rel.split_once('/').unwrap_or((rel, ""));
1069    if rest.is_empty() {
1070        if SYSTEM_ROOT_FILES.contains(&first)
1071            || (first.starts_with("search-index") && first.ends_with(".js"))
1072        {
1073            return Class::System;
1074        }
1075        return if first.ends_with(".html") {
1076            Class::Stray
1077        } else {
1078            Class::Other
1079        };
1080    }
1081    if SYSTEM_DIRS.contains(&first) {
1082        return Class::System;
1083    }
1084    let html = rel.ends_with(".html");
1085    if crates.contains(first) {
1086        let name = rel.rsplit('/').next().unwrap_or(rel);
1087        if rest == "all.html" || (name.starts_with("sidebar-items") && name.ends_with(".js")) {
1088            return Class::System;
1089        }
1090        if html {
1091            return if redirect {
1092                Class::Redirect
1093            } else {
1094                Class::ItemPage
1095            };
1096        }
1097        return Class::Other;
1098    }
1099    match (html, redirect) {
1100        (true, true) => Class::Redirect,
1101        (true, false) => Class::Stray,
1102        _ => Class::Other,
1103    }
1104}

1106/// Every file under `dir`, tree-relative with `/` separators, in byte order. Symbolic links
1107/// are not followed: a tree that links out of itself is judged by what it holds.
1108fn files_of(dir: &Path) -> Vec<String> {
1109    let mut out = Vec::new();
1110    let mut stack = vec![PathBuf::new()];
1111    while let Some(rel) = stack.pop() {
1112        let Ok(entries) = std::fs::read_dir(dir.join(&rel)) else {
1113            continue;
1114        };
1115        for entry in entries.flatten() {
1116            let Ok(kind) = entry.file_type() else {
1117                continue;
1118            };
1119            let child = rel.join(entry.file_name());
1120            if kind.is_dir() {
1121                stack.push(child);
1122            } else if kind.is_file() {
1123                out.push(child.to_string_lossy().replace('\\', "/"));
1124            }
1125        }
1126    }
1127    // the canonical order, so the walk's findings come out the same on every filesystem
1128    crate::order::canonical_strings(&mut out);
1129    out
1130}

1132/// The accepted references of one class, as they are gathered.
1133#[derive(Default)]
1134struct Accepted {
1135    links: usize,
1136    pages: BTreeSet<String>,
1137    written: BTreeSet<String>,
1138}
1139
1140fn verify(
1141    dir: &Path,
1142    shown: &str,
1143    mount: Option<String>,
1144    expected: &Expected,
1145    head: Option<&str>,
1146    root: &Path,
1147    leaks: &Leaks,
1148) -> RustdocReport {
1149    let files = files_of(dir);
1150    let present: HashSet<&str> = files.iter().map(String::as_str).collect();
1151    let directories: HashSet<&str> = files
1152        .iter()
1153        .flat_map(|f| f.match_indices('/').map(move |(i, _)| &f[..i]))
1154        .collect();
1155    let crates: HashSet<&str> = expected.crates.iter().map(String::as_str).collect();
1156    // the repository's own absolute root, as literal bytes: whatever machine the tree was
1157    // judged on, a page naming it was written with this checkout's location in it
1158    let root_text = root.to_string_lossy().into_owned();
1159    let root_pattern = (root_text.len() > 1)
1160        .then(|| Regex::new(&regex::escape(&root_text)).ok())
1161        .flatten();
1162    let mut findings = Vec::new();
1163    let mut counts = RustdocCounts {
1164        exported_items: expected.exported_items,
1165        page_owning_items: expected.page_owning_items,
1166        derived_pages: expected.pages.len(),
1167        files: files.len(),
1168        ..RustdocCounts::default()
1169    };
1170    let mut hosts: BTreeMap<String, usize> = BTreeMap::new();
1171    // a broken link is reported once per target, with how many links and pages carry it and
1172    // the first page that does: one missing stylesheet is one defect, not one per page
1173    let mut broken: BTreeMap<String, (String, usize, BTreeSet<String>)> = BTreeMap::new();
1174    let mut accepted: BTreeMap<RustdocLinkClass, Accepted> = RustdocLinkClass::all()
1175        .into_iter()
1176        .map(|c| (c, Accepted::default()))
1177        .collect();
1178    let mut item_pages: HashSet<&str> = HashSet::new();
1179    let mut redirects: HashSet<&str> = HashSet::new();
1180
1181    for rel in &files {
1182        let Ok(bytes) = std::fs::read(dir.join(rel)) else {
1183            findings.push(RustdocFinding::new(
1184                RustdocFindingKind::Link,
1185                rel.clone(),
1186                rel.clone(),
1187                "the file is listed in the tree and cannot be read",
1188            ));
1189            continue;
1190        };
1191        let mut redirect = false;
1192        if rel.ends_with(".html") {
1193            let page = Page::read(&bytes);
1194            redirect = page.redirect;
1195            counts.html_files += 1;
1196            for link in &page.links {
1197                match resolve(rel, &link.value, &present, &directories) {
1198                    Resolution::Internal => counts.relative_links += 1,
1199                    Resolution::External(host) => *hosts.entry(host).or_default() += 1,
1200                    Resolution::Ignored => {}
1201                    Resolution::Broken(target) => {
1202                        counts.relative_links += 1;
1203                        match accepted_class(rel, link, &target) {
1204                            Some(class) => {
1205                                let a = accepted.entry(class).or_default();
1206                                a.links += 1;
1207                                a.pages.insert(rel.clone());
1208                                a.written.insert(link.value.clone());
1209                            }
1210                            None => {
1211                                let b = broken
1212                                    .entry(target)
1213                                    .or_insert_with(|| (rel.clone(), 0, BTreeSet::new()));
1214                                b.1 += 1;
1215                                b.2.insert(rel.clone());
1216                            }
1217                        }
1218                    }
1219                }
1220            }
1221        }
1222        match classify(rel, &crates, redirect) {
1223            Class::System => counts.system_files += 1,
1224            Class::Redirect => {
1225                counts.redirect_stubs += 1;
1226                redirects.insert(rel.as_str());
1227            }
1228            Class::ItemPage => {
1229                counts.item_pages += 1;
1230                item_pages.insert(rel.as_str());
1231                if !expected.pages.contains_key(rel.as_str()) {
1232                    findings.push(RustdocFinding::new(
1233                        RustdocFindingKind::OrphanPage,
1234                        rel.clone(),
1235                        rel.clone(),
1236                        "an item page that no item of the crate derives",
1237                    ));
1238                }
1239            }
1240            Class::Stray => findings.push(RustdocFinding::new(
1241                RustdocFindingKind::OrphanPage,
1242                rel.clone(),
1243                rel.clone(),
1244                "an HTML page outside every crate directory that is neither rustdoc's nor the producer's",
1245            )),
1246            Class::Landing | Class::Other => {}
1247        }
1248        leak_findings(rel, &bytes, root_pattern.as_ref(), leaks, &mut findings);
1249    }
1250
1251    // every page an exported item owns, at the route rustdoc gives it
1252    for (page, item) in &expected.pages {
1253        if item_pages.contains(page.as_str()) {
1254            continue;
1255        }
1256        let why = if redirects.contains(page.as_str()) {
1257            "only a redirect stub is there, so rustdoc documents the item somewhere this check did not derive"
1258        } else {
1259            "no page is there"
1260        };
1261        let at = expected
1262            .declared_at
1263            .get(page)
1264            .map(|d| format!(" (declared at {d})"))
1265            .unwrap_or_default();
1266        findings.push(RustdocFinding::new(
1267            RustdocFindingKind::MissingPage,
1268            page.clone(),
1269            item.clone(),
1270            format!("{why}{at}"),
1271        ));
1272    }
1273
1274    let built_from = declaration(dir, head, &mut findings);
1275    let missing_assets = crate_index(dir, &expected.lib, &present, &mut findings);
1276
1277    for (target, (first, links, pages)) in broken {
1278        // an asset the crate index needs is already its own, more specific finding
1279        if missing_assets.contains(&target) {
1280            continue;
1281        }
1282        findings.push(RustdocFinding::new(
1283            RustdocFindingKind::Link,
1284            first,
1285            target,
1286            format!(
1287                "resolves to no file of the tree; carried by {links} link(s) on {} page(s)",
1288                pages.len()
1289            ),
1290        ));
1291    }
1292
1293    counts.accepted_links = accepted
1294        .into_iter()
1295        .map(|(class, a)| RustdocAcceptedLinks {
1296            class,
1297            links: a.links,
1298            pages: a.pages.len(),
1299            written: a.written.into_iter().collect(),
1300        })
1301        .collect();
1302    for a in &mut counts.accepted_links {
1303        crate::order::canonical_strings(&mut a.written);
1304    }
1305    canonical(&mut counts.accepted_links);
1306    counts.external_links = hosts
1307        .into_iter()
1308        .map(|(host, links)| RustdocHostLinks { host, links })
1309        .collect();
1310    canonical(&mut counts.external_links);
1311    canonical(&mut findings);
1312    let verdict = if findings.is_empty() {
1313        RustdocVerdict::Clean
1314    } else {
1315        RustdocVerdict::Findings
1316    };
1317    RustdocReport {
1318        schema: SCHEMA.to_string(),
1319        tree: shown.to_string(),
1320        mount,
1321        krate: expected.lib.clone(),
1322        built_from,
1323        head: head.map(str::to_string),
1324        verdict,
1325        reason: None,
1326        counts,
1327        modules: expected.modules.clone(),
1328        findings,
1329    }
1330}

1332/// Is this unresolved reference one of rustdoc's own, and of which class?
1333///
1334/// Decided by where it sits and what it names, never by a list of files: a link inside a
1335/// docblock the page reader attributed to another crate's trait, or the script of the very
1336/// trait the page documents under `trait.impl/` of its own crate.

1338    if link.inherited {
1339        return Some(RustdocLinkClass::InheritedDocumentation);
1340    }

1342        return Some(RustdocLinkClass::UnwrittenImplementors);
1343    }

1347/// Is `target` the implementors script of the trait the page at `page` documents?
1348///
1349/// The page is `<crate>/<modules>/trait.<Name>.html`; rustdoc names the script after the
1350/// trait's own path, which for a re-exported trait is the module that declares it rather
1351/// than the page's, so what must agree is the crate and the trait's file name.
1352fn is_own_implementors(page: &str, target: &str) -> bool {
1353    let (krate, _) = page.split_once('/').unwrap_or(("", ""));
1354    let Some(name) = page.rsplit('/').next() else {
1355        return false;
1356    };

1358        .strip_prefix("trait.")
1359        .and_then(|n| n.strip_suffix(".html"))
1360    else {
1361        return false;
1362    };
1363    !krate.is_empty()
1364        && target.starts_with(&format!("trait.impl/{krate}/"))
1365        && target.ends_with(&format!("/trait.{trait_name}.js"))
1366}

1420/// The producer's declaration: what the tree says it is and what it was built from.
1421fn declaration(
1422    dir: &Path,
1423    head: Option<&str>,
1424    findings: &mut Vec<RustdocFinding>,
1425) -> Option<String> {
1426    use crate::web::discover::{Declaration, DECLARATION_FILE};
1427    let undeclared = |findings: &mut Vec<RustdocFinding>, why: String| {
1428        findings.push(RustdocFinding::new(
1429            RustdocFindingKind::Undeclared,
1430            DECLARATION_FILE,
1431            DECLARATION_FILE,
1432            why,
1433        ))
1434    };
1435    let Ok(text) = std::fs::read_to_string(dir.join(DECLARATION_FILE)) else {
1436        undeclared(
1437            findings,
1438            "the tree carries no declaration, so nothing says what it is or which commit it was built from"
1439                .into(),
1440        );
1441        return None;
1442    };
1443    let decl: Declaration = match serde_json::from_str(&text) {
1444        Ok(d) => d,
1445        Err(e) => {
1446            undeclared(findings, format!("does not parse as web-surface/v1: {e}"));
1447            return None;
1448        }
1449    };
1450    if decl.id != SURFACE {
1451        undeclared(
1452            findings,
1453            format!("declares the surface '{}', not '{SURFACE}'", decl.id),
1454        );
1455    }
1456    let Some(built) = decl.built_from.clone() else {
1457        undeclared(
1458            findings,
1459            "declares no built_from, so the tree cannot be proven to be of any commit".into(),
1460        );
1461        return None;
1462    };
1463    match head {
1464        Some(h) if h == built => {}
1465        Some(h) => findings.push(RustdocFinding::new(
1466            RustdocFindingKind::Stale,
1467            DECLARATION_FILE,
1468            built.clone(),
1469            format!("built from {built}, and the repository's HEAD is {h}"),
1470        )),
1471        None => findings.push(RustdocFinding::new(
1472            RustdocFindingKind::Stale,
1473            DECLARATION_FILE,
1474            built.clone(),
1475            "the repository has no HEAD to compare built_from against, so freshness is unproven",
1476        )),
1477    }
1478    Some(built)
1479}

1481/// The library's index page: present, titled as rustdoc titles a crate, and every asset it
1482/// loads present — its stylesheets and scripts, the fonts it preloads and its stylesheets
1483/// reference, the scripts its `rustdoc-vars` name, and the root of the search index.
1484///
1485/// Answers the targets it found absent, so that a missing asset is reported once, as an
1486/// asset, and not a second time as the link every page carries to it.
1487fn crate_index(
1488    dir: &Path,
1489    lib: &str,
1490    present: &HashSet<&str>,
1491    findings: &mut Vec<RustdocFinding>,
1492) -> BTreeSet<String> {
1493    let mut missing = BTreeSet::new();
1494    let index = format!("{lib}/index.html");
1495    let Ok(bytes) = std::fs::read(dir.join(&index)) else {
1496        findings.push(RustdocFinding::new(
1497            RustdocFindingKind::CrateIndex,
1498            index,
1499            lib,
1500            "the library has no index page",
1501        ));
1502        return missing;
1503    };
1504    let text = String::from_utf8_lossy(&bytes);
1505    let want = format!("{lib} - Rust");
1506    let title = text
1507        .split_once("<title>")
1508        .and_then(|(_, rest)| rest.split_once("</title>"))
1509        .map(|(t, _)| t.trim().to_string());
1510    if title.as_deref() != Some(want.as_str()) {
1511        findings.push(RustdocFinding::new(
1512            RustdocFindingKind::CrateIndex,
1513            index.clone(),
1514            lib,
1515            format!(
1516                "is titled {:?}, and rustdoc titles the library's index {want:?}",
1517                title.unwrap_or_default()
1518            ),
1519        ));
1520    }
1521    let page = Page::read(&bytes);
1522    let mut needed: BTreeSet<String> = BTreeSet::new();
1523    for asset in &page.assets {
1524        if !is_relative(asset) {
1525            continue;
1526        }
1527        if let Some(target) = internal_target(&index, asset) {
1528            needed.insert(target);
1529        }
1530    }
1531    if let Some(static_root) = page.vars.get("data-static-root-path") {
1532        for (name, value) in &page.vars {
1533            if name.ends_with("-js") && !value.is_empty() {
1534                if let Some(t) = internal_target(&index, &format!("{static_root}{value}")) {
1535                    needed.insert(t);
1536                }
1537            }
1538        }
1539        for font in &page.fonts {
1540            if let Some(t) = internal_target(&index, &format!("{static_root}{font}")) {
1541                needed.insert(t);
1542            }
1543        }
1544    }
1545    // what the stylesheets load in turn: the fonts, by url()
1546    let sheets: Vec<String> = needed
1547        .iter()
1548        .filter(|t| t.ends_with(".css"))
1549        .cloned()
1550        .collect();
1551    for sheet in sheets {
1552        let Ok(css) = std::fs::read(dir.join(&sheet)) else {
1553            continue;
1554        };
1555        for url in css_urls(&css) {
1556            if let Some(t) = internal_target(&sheet, &url) {
1557                needed.insert(t);
1558            }
1559        }
1560    }
1561    for target in needed {
1562        if !present.contains(target.as_str()) {
1563            findings.push(RustdocFinding::new(
1564                RustdocFindingKind::Asset,
1565                index.clone(),
1566                target.clone(),
1567                "the library's index page loads it and the tree does not hold it",
1568            ));
1569            missing.insert(target);
1570        }
1571    }
1572    // the search index is loaded by script, not by an attribute: rustdoc 1.98 reads
1573    // search.index/root.js from the root path, an older one search-index.js at the root
1574    let root_path = page
1575        .vars
1576        .get("data-root-path")
1577        .cloned()
1578        .unwrap_or_else(|| "../".into());
1579    let modern = internal_target(&index, &format!("{root_path}search.index/root.js"));
1580    let has_modern = modern.as_deref().is_some_and(|t| present.contains(t));
1581    let has_legacy = present
1582        .iter()
1583        .any(|f| !f.contains('/') && f.starts_with("search-index") && f.ends_with(".js"));
1584    if !has_modern && !has_legacy {
1585        findings.push(RustdocFinding::new(
1586            RustdocFindingKind::Asset,
1587            index,
1588            modern.unwrap_or_else(|| "search.index/root.js".into()),
1589            "the search index's root is absent, so the search box of every page answers nothing",
1590        ));
1591    }
1592    missing
1593}

1595/// Every `url(...)` a stylesheet names, quotes removed, data URIs left out.
1596fn css_urls(css: &[u8]) -> Vec<String> {
1597    let text = String::from_utf8_lossy(css);
1598    let mut out = Vec::new();
1599    let mut rest = text.as_ref();
1600    while let Some(at) = rest.find("url(") {
1601        rest = &rest[at + 4..];
1602        let Some(end) = rest.find(')') else { break };
1603        let url = rest[..end].trim().trim_matches(|c| c == '"' || c == '\'');
1604        if !url.is_empty() && !url.starts_with("data:") && !url.starts_with('#') {
1605            out.push(url.to_string());
1606        }
1607        rest = &rest[end..];
1608    }
1609    out
1610}

1614/// What a link of a page is.
1615#[derive(Debug, Clone, PartialEq, Eq)]
1616enum Resolution {
1617    /// A relative link to a file of the tree.
1618    Internal,
1619    /// A link to another origin, by host: counted, never fetched.
1620    External(String),
1621    /// Not a link to a file: a same-page anchor, `mailto:`, `javascript:`, `data:`.
1622    Ignored,
1623    /// A link that resolves to nothing the tree holds, or leaves the tree; the target as
1624    /// resolved (or as written, when it cannot be resolved inside the tree).
1625    Broken(String),
1626}
1627
1628/// Resolve one `href` or `src` of the page at `from`.
1629///
1630/// The fragment and the query are dropped before resolution, which is what accepts
1631/// rustdoc's `#12-40` source-line ranges and its `?search=` links. An absolute path is
1632/// broken by definition: the tree is mounted under a prefix, so `/x` leaves it; so is a
1633/// relative path that climbs above the tree's root.
1634fn resolve(
1635    from: &str,
1636    link: &str,
1637    present: &HashSet<&str>,
1638    directories: &HashSet<&str>,
1639) -> Resolution {
1640    let link = link.trim();
1641    if link.is_empty() || link.starts_with('#') {
1642        return Resolution::Ignored;
1643    }
1644    let lower = link.to_ascii_lowercase();
1645    for scheme in ["mailto:", "javascript:", "data:", "tel:"] {
1646        if lower.starts_with(scheme) {
1647            return Resolution::Ignored;
1648        }
1649    }
1650    if let Some(rest) = lower
1651        .strip_prefix("https://")
1652        .or_else(|| lower.strip_prefix("http://"))
1653        .or_else(|| lower.strip_prefix("//"))
1654    {
1655        let host = rest
1656            .split(['/', '?', '#'])
1657            .next()
1658            .unwrap_or_default()
1659            .rsplit('@')
1660            .next()
1661            .unwrap_or_default()
1662            .to_string();
1663        return Resolution::External(host);
1664    }
1665    if has_scheme(&lower) {
1666        return Resolution::Broken(link.to_string());
1667    }
1668    let Some(target) = internal_target(from, link) else {
1669        return Resolution::Broken(link.to_string());
1670    };
1671    if present.contains(target.as_str()) {
1672        return Resolution::Internal;
1673    }
1674    // a directory link is answered by its index page
1675    let dir = target.trim_end_matches('/');
1676    if (dir.is_empty() || directories.contains(dir) || target.ends_with('/'))
1677        && present.contains(format!("{}index.html", with_slash(dir)).as_str())
1678    {
1679        return Resolution::Internal;
1680    }
1681    Resolution::Broken(target)
1682}

1684/// Is this a relative reference into the tree: not an anchor, not another origin, not a
1685/// scheme of its own?
1686fn is_relative(link: &str) -> bool {
1687    let link = link.trim();
1688    !link.is_empty()
1689        && !link.starts_with('#')
1690        && !link.starts_with("//")
1691        && !has_scheme(&link.to_ascii_lowercase())
1692}

1694fn with_slash(dir: &str) -> String {
1695    if dir.is_empty() {
1696        String::new()
1697    } else {
1698        format!("{dir}/")
1699    }
1700}

1702/// Does the reference start with a URL scheme (`https:`, `mailto:`)? A colon after a slash
1703/// is a path, and `super::Subscriber` is no scheme either: a scheme is letters first.
1704fn has_scheme(link: &str) -> bool {
1705    let Some(i) = link.find(':') else {
1706        return false;
1707    };
1708    let scheme = &link[..i];
1709    scheme.starts_with(|c: char| c.is_ascii_alphabetic())
1710        && scheme
1711            .chars()
1712            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
1713        && !link[i + 1..].starts_with(':')
1714}

1716/// The tree-relative file a relative link of the page at `from` names, or `None` when it
1717/// climbs out of the tree or is written as an absolute path.
1718fn internal_target(from: &str, link: &str) -> Option<String> {
1719    let link = link.split(['#', '?']).next().unwrap_or_default();
1720    if link.starts_with('/') {
1721        return None;
1722    }
1723    let decoded = percent_decode(&link.replace("&amp;", "&"));
1724    let mut parts: Vec<&str> = from.split('/').collect();
1725    parts.pop(); // the page's own file name
1726    let trailing = decoded.ends_with('/');
1727    for seg in decoded.split('/') {
1728        match seg {
1729            "" | "." => {}
1730            ".." => {
1731                parts.pop()?;
1732            }
1733            s => parts.push(s),
1734        }
1735    }
1736    let mut out = parts.join("/");
1737    if trailing && !out.is_empty() {
1738        out.push('/');
1739    }
1740    Some(out)
1741}

1743/// `%XX` escapes decoded; anything that is not a valid escape is kept as written.
1744fn percent_decode(s: &str) -> String {
1745    if !s.contains('%') {
1746        return s.to_string();
1747    }
1748    let bytes = s.as_bytes();
1749    let mut out = Vec::with_capacity(bytes.len());
1750    let mut i = 0;
1751    while i < bytes.len() {
1752        if bytes[i] == b'%' && i + 2 < bytes.len() {
1753            let hex = std::str::from_utf8(&bytes[i + 1..i + 3]).ok();
1754            if let Some(v) = hex.and_then(|h| u8::from_str_radix(h, 16).ok()) {
1755                out.push(v);
1756                i += 3;
1757                continue;
1758            }
1759        }
1760        out.push(bytes[i]);
1761        i += 1;
1762    }
1763    String::from_utf8_lossy(&out).into_owned()
1764}

1804/// Where the reader is on a page: the nesting of `<div>`s, and which of the regions the
1805/// inherited-documentation class is decided by are open.
1806#[derive(Debug, Default)]
1807struct Position {
1808    /// `<div>`s open.
1809    divs: usize,
1810    /// The depth of the open list of foreign implementations, when inside one.
1811    foreign: Option<usize>,
1812    /// The depth of the open docblock, when inside one.
1813    docblock: Option<usize>,
1814    /// Inside the `<h3>` header of an implementation in a foreign list.
1815    impl_header: bool,
1816    /// That implementation's header links a trait page of this tree: the trait is the
1817    /// crate's own, and so is everything documented under it.
1818    impl_ours: bool,
1819}
1820
1821impl Position {
1822    fn inherited(&self) -> bool {
1823        self.foreign.is_some() && self.docblock.is_some() && !self.impl_ours
1824    }
1825
1826    fn open(&mut self, tag: &[u8], id: Option<&str>, class: Option<&str>, href: Option<&str>) {
1827        let has_class = |c: &str| class.is_some_and(|v| v.split_whitespace().any(|x| x == c));
1828        match tag {
1829            b"div" => {
1830                self.divs += 1;
1831                if id.is_some_and(|i| FOREIGN_IMPLEMENTATION_LISTS.contains(&i)) {
1832                    self.foreign = Some(self.divs);
1833                    self.impl_ours = false;
1834                }
1835                if self.docblock.is_none() && has_class("docblock") {

1841                self.impl_ours = false;
1842            }

1844                self.impl_ours = true;
1845            }
1846            _ => {}
1847        }
1848    }

1868/// The name of a tag starting at `at`: ASCII letters, digits and `-`, lower-cased.
1869fn tag_name(html: &[u8], at: usize) -> (Vec<u8>, usize) {
1870    let end = html[at..]
1871        .iter()
1872        .position(|b| !(b.is_ascii_alphanumeric() || *b == b'-'))
1873        .map_or(html.len(), |e| at + e);
1874    (html[at..end].to_ascii_lowercase(), end)
1875}

1877impl Page {
1878    /// Tokenise the page once. The scanner reads tags and their attributes and skips text,
1879    /// comments, and the bodies of `<script>` and `<style>`: a template literal like
1880    /// `href="../static.files/${f}"` inside a script is code, not a link, and text is
1881    /// escaped by rustdoc so a quoted `href=` in prose or in a source page is never a tag.
1882    fn read(html: &[u8]) -> Page {
1883        let mut page = Page::default();
1884        let mut at = Position::default();
1885        let n = html.len();
1886        let mut i = 0usize;
1887        while let Some(off) = html[i..].iter().position(|&b| b == b'<') {
1888            i += off + 1;
1889            if html[i..].starts_with(b"!--") {
1890                i = find(html, i + 3, b"-->").map_or(n, |e| e + 3);
1891                continue;
1892            }
1893            if i < n && html[i] == b'/' {
1894                let (tag, end) = tag_name(html, i + 1);
1895                at.close(&tag);
1896                i = html[end..]
1897                    .iter()
1898                    .position(|&b| b == b'>')
1899                    .map_or(n, |e| end + e + 1);
1900                continue;
1901            }
1902            if i < n && matches!(html[i], b'!' | b'?') {
1903                i = html[i..]
1904                    .iter()
1905                    .position(|&b| b == b'>')
1906                    .map_or(n, |e| i + e + 1);
1907                continue;
1908            }
1909            let (tag, name_end) = tag_name(html, i);
1910            if tag.is_empty() {
1911                continue;
1912            }
1913            i = name_end;
1914            let mut attrs: Vec<(Vec<u8>, String)> = Vec::new();
1915            // attributes, until the tag closes
1916            loop {
1917                while i < n && html[i].is_ascii_whitespace() {
1918                    i += 1;
1919                }
1920                if i >= n {
1921                    break;
1922                }
1923                if html[i] == b'>' {
1924                    i += 1;
1925                    break;
1926                }
1927                if html[i] == b'/' {
1928                    i += 1;
1929                    continue;
1930                }
1931                let a = i;
1932                while i < n
1933                    && !html[i].is_ascii_whitespace()
1934                    && !matches!(html[i], b'=' | b'>' | b'/')
1935                {
1936                    i += 1;
1937                }
1938                let attr = html[a..i].to_ascii_lowercase();
1939                while i < n && html[i].is_ascii_whitespace() {
1940                    i += 1;
1941                }
1942                let mut value = String::new();
1943                if i < n && html[i] == b'=' {
1944                    i += 1;
1945                    while i < n && html[i].is_ascii_whitespace() {
1946                        i += 1;
1947                    }
1948                    if i < n && (html[i] == b'"' || html[i] == b'\'') {
1949                        let q = html[i];
1950                        let v = i + 1;
1951                        let end = html[v..].iter().position(|&b| b == q).map_or(n, |e| v + e);
1952                        value = String::from_utf8_lossy(&html[v..end]).into_owned();
1953                        i = (end + 1).min(n);
1954                    } else {
1955                        let v = i;
1956                        while i < n && !html[i].is_ascii_whitespace() && html[i] != b'>' {
1957                            i += 1;
1958                        }
1959                        value = String::from_utf8_lossy(&html[v..i]).into_owned();
1960                    }
1961                }
1962                if matches!(
1963                    attr.as_slice(),
1964                    b"href" | b"src" | b"name" | b"http-equiv" | b"id" | b"class"
1965                ) || attr.starts_with(b"data-")
1966                {
1967                    attrs.push((attr, value));
1968                }
1969            }
1970            let get = |k: &[u8]| attrs.iter().find(|(a, _)| a == k).map(|(_, v)| v.as_str());
1971            at.open(&tag, get(b"id"), get(b"class"), get(b"href"));
1972            let script = tag == b"script";
1973            for (a, v) in &attrs {
1974                if a == b"href" || a == b"src" {
1975                    page.links.push(Link {
1976                        value: v.clone(),
1977                        script: script && a == b"src",
1978                        inherited: at.inherited(),
1979                    });
1980                }
1981            }
1982            match tag.as_slice() {
1983                b"meta" => {
1984                    if get(b"http-equiv").is_some_and(|v| v.eq_ignore_ascii_case("refresh")) {
1985                        page.redirect = true;
1986                    }
1987                    if get(b"name") == Some("rustdoc-vars") {
1988                        for (a, v) in &attrs {
1989                            if a.starts_with(b"data-") {
1990                                page.vars
1991                                    .insert(String::from_utf8_lossy(a).into_owned(), v.clone());
1992                            }
1993                        }
1994                    }
1995                }
1996                b"link" => page.assets.extend(get(b"href").map(str::to_string)),
1997                b"script" => page.assets.extend(get(b"src").map(str::to_string)),
1998                _ => {}
1999            }
2000            if tag == b"script" || tag == b"style" {
2001                let close: &[u8] = if script { b"</script" } else { b"</style" };
2002                let end = find_ascii_ci(html, i, close).unwrap_or(n);
2003                if script {
2004                    page.fonts.extend(woff2_names(&html[i..end]));
2005                }
2006                i = end;
2007            }
2008        }
2009        page
2010    }
2011}

2013fn find(hay: &[u8], from: usize, needle: &[u8]) -> Option<usize> {
2014    if from >= hay.len() {
2015        return None;
2016    }
2017    hay[from..]
2018        .windows(needle.len())
2019        .position(|w| w == needle)
2020        .map(|p| from + p)
2021}
2022
2023fn find_ascii_ci(hay: &[u8], from: usize, needle: &[u8]) -> Option<usize> {
2024    if from >= hay.len() {
2025        return None;
2026    }
2027    hay[from..]
2028        .windows(needle.len())
2029        .position(|w| w.eq_ignore_ascii_case(needle))
2030        .map(|p| from + p)
2031}

2033/// Every `name.woff2` token of a script body: how rustdoc's index page preloads its fonts.
2034fn woff2_names(script: &[u8]) -> Vec<String> {
2035    let text = String::from_utf8_lossy(script);
2036    text.split(|c: char| !(c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.')))
2037        .filter(|t| t.ends_with(".woff2") && t.len() > ".woff2".len())
2038        .map(str::to_string)
2039        .collect()
2040}
