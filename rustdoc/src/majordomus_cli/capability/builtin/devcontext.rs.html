1//! The `devcontext` module: the context compiler as a capability.
2//!
3//! Three questions, answered from the index and the composed graph this process already
4//! holds: what context does this piece of work need, why is any one thing in or out of that
5//! answer, and what are the compiler's own rules. The MCP tools, the resource, the HTTP
6//! routes, the OpenAPI operations, the command line and the Cockpit's pages are projections
7//! of these three declarations; none of them holds a selection rule of its own, and there is
8//! no Cockpit-specific implementation to hold one in.
9//!
10//! The work is [`crate::devcontext`]; this file is only the declaration. That split is
11//! deliberate: the compiler is a domain model with its own tests, and a capability module
12//! that carried the model would make the registry the place selection rules live.
13//!
14//! # Why `compile` is cached for the life of the process
15//!
16//! The answer is a pure function of the index and the request, and the executor's cache key
17//! is the canonical id, the input in canonical form and *the registry fingerprint* — which
18//! is a sha-256 over the index fingerprint, itself over every object's path and content
19//! (ADR 0004). So a cached answer is invalidated by an actual change to the repository
20//! rather than by a clock, and a time-to-live would only throw away answers that are still
21//! true. Every answer also carries the fingerprint it was compiled from, so a caller can
22//! tell two trees apart without asking a second question.

38/// The URI under which the compiler's own rules are read as an MCP resource.
39pub const DEVCONTEXT_POLICY_URI: &str = "majordomus://devcontext/policy";

41/// The input of `devcontext.compile`: what to compile a context about.
42///
43/// A newtype over [`devcontext::CompileInput`] so that the schema component the OpenAPI
44/// document carries is named for the operation rather than for the domain type, the way
45/// every other module's input is.
46#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize, JsonSchema)]
47#[serde(transparent)]
48#[schemars(rename = "DevContextInput")]
49pub struct CompileInput(pub devcontext::CompileInput);
50
51impl BenchmarkCases for CompileInput {
52    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
53        devcontext::CompileInput::benchmark_cases(ctx)
54            .into_iter()
55            .map(|c| NamedCase {
56                name: c.name,
57                input: CompileInput(c.input),
58            })
59            .collect()
60    }
61}
62
63/// The input of `devcontext.explain`: one identifier, and the request to judge it under.
64#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize, JsonSchema)]
65#[serde(transparent)]
66#[schemars(rename = "DevContextExplainInput")]
67pub struct ExplainInput(pub devcontext::ExplainInput);
68
69impl BenchmarkCases for ExplainInput {
70    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
71        devcontext::ExplainInput::benchmark_cases(ctx)
72            .into_iter()
73            .map(|c| NamedCase {
74                name: c.name,
75                input: ExplainInput(c.input),
76            })
77            .collect()
78    }
79}

81fn compile(ctx: &Context, input: CompileInput) -> Result<CompiledContext, CapabilityError> {
82    devcontext::compile(ctx, input.0)
83}
84
85fn explain(ctx: &Context, input: ExplainInput) -> Result<Explanation, CapabilityError> {
86    devcontext::explain(ctx, input.0)
87}
88
89fn policy(ctx: &Context, _: Empty) -> Result<CompilerPolicy, CapabilityError> {
90    devcontext::policy(ctx)
91}

98        description: "The context a development session should be given, derived from the repository rather than composed by hand: given an issue, a milestone, an intent or a set of paths, the canonical objects that bear on the work, why each one is in the answer, what was left out and why, what collapsed into what, and the whole cost against a budget. The selection is structured — every entry keeps its identifier, its provenance, the selector that reached it and its confidence — because rendering a prompt is a projection of the selection and not the selection itself.",
99        stability: Stability::BehaviorallyVerified,
100        capabilities: [
101            capability! {
102                id: "devcontext.compile",
103                title: "Compile the context for a piece of work",
104                description: "The objects a session working on this issue, milestone, intent or set of paths should be given: each with its canonical identifier, the index's own provenance, every discovery path that reached it with the reason and the confidence, its version, its tier and its cost in estimated tokens — followed by everything reached and not given with the reason for each, everything that collapsed into one entry, every pair that does not agree, and the per-tier spend against the budget. Deterministic for a given tree and request; the index fingerprint it was compiled from is in the answer.",
105                input: CompileInput,
106                output: CompiledContext,
107                stability: Stability::BehaviorallyVerified,
108                exposure: Exposure {
109                    mcp: mcp("majordomus_devcontext"),
110                    http: get("/api/v1/devcontext"),

120                description: "One canonical identifier judged under a request: whether it was selected, reached and excluded, folded into another identifier, held by the index and never reached, or unknown — with the entry, the exclusion or the collapse itself, and the budget the judgement was made under so that `excluded for budget` can be acted on without a second call.",

159    #[test]
160    fn the_declaration_yields_the_projections_it_claims() {
161        let m = module();
162        assert_eq!(m.id.as_str(), "devcontext");
163        let expected: &[(&str, &str, &str, &[&str])] = &[
164            (
165                "devcontext.compile",
166                "majordomus_devcontext",
167                "/api/v1/devcontext",
168                &["devcontext", "compile"],
169            ),
170            (
171                "devcontext.explain",
172                "majordomus_devcontext_explain",
173                "/api/v1/devcontext/explain",
174                &["devcontext", "explain"],
175            ),
176            (
177                "devcontext.policy",
178                "majordomus_devcontext_policy",
179                "/api/v1/devcontext/policy",
180                &["devcontext", "policy"],
181            ),
182        ];
183        let ids: Vec<&str> = m
184            .capabilities
185            .iter()
186            .map(|e| e.capability.id.as_str())
187            .collect();
188        let want: Vec<&str> = expected.iter().map(|(id, _, _, _)| *id).collect();
189        assert_eq!(ids, want);
190        for (executable, (id, tool, path, cli)) in m.capabilities.iter().zip(expected) {
191            let exposure = &executable.capability.exposure;
192            assert_eq!(
193                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
194                Some(*tool),
195                "{id} lost its MCP tool"
196            );
197            assert_eq!(
198                exposure.http.as_ref().map(|h| h.path.as_str()),
199                Some(*path),
200                "{id} lost its HTTP route"
201            );
202            assert_eq!(
203                exposure
204                    .cli
205                    .as_ref()
206                    .map(|c| c.path.iter().map(String::as_str).collect::<Vec<_>>())
207                    .as_deref(),
208                Some(*cli),
209                "{id} lost its command line"
210            );
211        }
212        // the compiler's rules are readable as a resource, because they need no input
213        let resource = m.capabilities[2]
214            .capability
215            .exposure
216            .mcp
217            .as_ref()
218            .and_then(|m| m.resource.as_ref())
219            .expect("the policy is readable as a resource");
220        assert_eq!(resource.uri, DEVCONTEXT_POLICY_URI);
221    }

223    #[test]
224    fn the_answer_is_cached_against_the_tree_and_not_against_a_clock() {
225        let m = module();
226        for e in &m.capabilities {
227            match e.capability.cache {
228                CachePolicy::Process { ttl_seconds, .. } => assert_eq!(
229                    ttl_seconds, None,
230                    "{} expires by time; the cache key already carries the index fingerprint",
231                    e.capability.id
232                ),
233                CachePolicy::Disabled => {}
234            }
235        }
236    }
