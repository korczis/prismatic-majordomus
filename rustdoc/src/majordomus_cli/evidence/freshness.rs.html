1//! Freshness: whether a recorded run still proves anything at the revision a reader is shown.
2//!
3//! A passing run is a fact about the tree it measured. Whether it is also a fact about the
4//! tree in front of a reader depends on three things the run cannot know: whether the
5//! revision presented contains the commit the run was recorded on, what changed between the
6//! two, and whether either tree was the commit it claims to be. This module decides that,
7//! once, for every surface — the evidence report, the rules report, and every later reader
8//! of the ledger — so that no two of them can disagree about what a run still proves.
9//!
10//! # The pieces
11//!
12//! * [`Presented`] — what is judged: the reader's working tree, or the checked-out commit
13//!   judged as committed ([`presented_commit`]), with the ledger committed in it
14//!   ([`ledger_at`]);
15//! * [`compare`] — the containment of the evidence commit in the presented revision, and
16//!   the paths that changed between them ([`changed_between`]);
17//! * [`freshness`] — the one truth table, pure, from a recorded run to a [`Judgement`];
18//! * [`current`] — whether a judgement is current evidence, for a reader that counts a route
19//!   as met: `proven`, or `inputs_unchanged` from a run on a tree that was its commit;
20//! * [`weakened_by`] — the monotone rule: a supplementary record may withhold `proven` and
21//!   never grant it; its first source is [`uncommitted`], the executions the working ledger
22//!   holds that the presented commit's ledger does not;
23//! * [`aggregate`] — how several states make one, with a failure outranking an absence.
24//!
25//! # Example
26//!
27//! The truth table without a repository: a recorded skip is not a failure, and only a pass
28//! the presented revision contains, with nothing but the ledger changed since, on a clean
29//! tree at both ends, is `proven`.
30//!
31//! ```
32//! use majordomus_cli::evidence::freshness::{freshness, Comparison, Recorded, TreeState};
33//! use majordomus_cli::evidence::{Execution, ProofState};
34//! use majordomus_cli::git::Containment;
35//!
36//! let run = |outcome: &str| -> Execution {
37//!     serde_json::from_value(serde_json::json!({
38//!         "test": "suite:07_scope", "runner": "suite", "source": "test/cases/07_scope.sh",
39//!         "outcome": outcome, "seconds": 1, "commit": "a".repeat(40),
40//!         "working_tree": "clean", "digest": "sha256:0", "at": "2026-09-26T00:00:00Z",
41//!         "origin": "local", "command": "bash test/run.sh 07_scope"
42//!     }))
43//!     .unwrap()
44//! };
45//! let nothing_changed = Comparison {
46//!     containment: Containment::Contains,
47//!     changed: Some(Default::default()),
48//! };
49//! let inputs = vec!["test/cases/07_scope.sh".to_string()];
50//! let judge = |e: &Execution| {
51//!     freshness(
52//!         Recorded::Ran(e),
53//!         Some(&inputs),
54//!         "test/cases/07_scope.sh",
55//!         false,
56//!         Some(&nothing_changed),
57//!         TreeState::Clean,
58//!     )
59//! };
60//! assert_eq!(judge(&run("pass")).state, ProofState::Proven);
61//! assert_eq!(judge(&run("skip")).state, ProofState::NotRun);
62//! assert_eq!(judge(&run("error")).state, ProofState::Failing);
63//! ```

81/// Whether a tree was the commit it sits on: the three words the ledger and git already
82/// use, typed.
83///
84/// `unknown` is not a softer `clean`. It is what is recorded when git could not be asked,
85/// and not knowing is not proof, so every rule here treats it as not clean.
86///
87/// ```
88/// use majordomus_cli::evidence::freshness::TreeState;
89///
90/// assert_eq!(TreeState::parse("clean"), TreeState::Clean);
91/// assert_eq!(TreeState::parse("dirty"), TreeState::Dirty);
92/// // a word nobody wrote down is not a clean tree
93/// assert_eq!(TreeState::parse("spotless"), TreeState::Unknown);
94/// assert_eq!(TreeState::Dirty.as_str(), "dirty");
95/// assert_eq!(serde_json::to_value(TreeState::Unknown).unwrap(), "unknown");
96/// ```
97#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
98#[serde(rename_all = "snake_case")]
99#[schemars(rename = "EvidenceTreeState")]
100pub enum TreeState {
101    /// The tree was the commit: nothing but the evidence ledger differed from it.
102    Clean,
103    /// Something other than the evidence ledger differed from the commit.
104    Dirty,
105    /// Git could not be asked, or the word recorded is none of the three.
106    Unknown,
107}
108
109impl TreeState {
110    /// Read the word a recorder or git wrote: `clean` and `dirty` are themselves, and
111    /// anything else is [`TreeState::Unknown`], never a guess at one of the other two.
112    ///
113    /// ```
114    /// use majordomus_cli::evidence::freshness::TreeState;
115    /// assert_eq!(TreeState::parse("clean"), TreeState::Clean);
116    /// assert_eq!(TreeState::parse("Clean"), TreeState::Unknown, "the words are exact");
117    /// assert_eq!(TreeState::parse(""), TreeState::Unknown);
118    /// ```
119    pub fn parse(word: &str) -> TreeState {
120        match word {
121            "clean" => TreeState::Clean,
122            "dirty" => TreeState::Dirty,
123            _ => TreeState::Unknown,
124        }
125    }
126
127    /// The word itself, as the ledger and the report spell it.
128    ///
129    /// ```
130    /// use majordomus_cli::evidence::freshness::TreeState;
131    /// for t in [TreeState::Clean, TreeState::Dirty, TreeState::Unknown] {
132    ///     assert_eq!(TreeState::parse(t.as_str()), t, "the word reads back as itself");
133    /// }
134    /// ```
135    pub fn as_str(self) -> &'static str {
136        match self {
137            TreeState::Clean => "clean",
138            TreeState::Dirty => "dirty",
139            TreeState::Unknown => "unknown",
140        }
141    }
142
143    /// The weaker of two statements about one tree: clean only when both say clean, dirty
144    /// when either says dirty, and unknown otherwise.
145    ///
146    /// This is how a caller's own knowledge meets a measurement. A site build that knows its
147    /// tree was dirty can weaken a clean measurement; nothing a caller says can make a
148    /// measured dirty tree clean.
149    ///
150    /// ```
151    /// use majordomus_cli::evidence::freshness::TreeState::{Clean, Dirty, Unknown};
152    /// assert_eq!(Clean.weaker(Clean), Clean);
153    /// assert_eq!(Clean.weaker(Dirty), Dirty);
154    /// assert_eq!(Dirty.weaker(Clean), Dirty, "a given clean never overrides a dirty tree");
155    /// assert_eq!(Unknown.weaker(Dirty), Dirty);
156    /// assert_eq!(Clean.weaker(Unknown), Unknown);
157    /// ```
158    pub fn weaker(self, other: TreeState) -> TreeState {
159        match (self, other) {
160            (TreeState::Clean, TreeState::Clean) => TreeState::Clean,
161            (TreeState::Dirty, _) | (_, TreeState::Dirty) => TreeState::Dirty,
162            _ => TreeState::Unknown,
163        }
164    }
165}

169/// The revision a verdict is about.
170///
171/// Every verdict is a verdict *at* something, and saying which is half of the answer. The
172/// working tree is what a person at a terminal is looking at; the checked-out commit judged
173/// as committed is the reading a site built from that commit needs.
174///
175/// ```
176/// use majordomus_cli::evidence::freshness::Presented;
177///
178/// // the default of every existing reader: the checkout, with whatever it carries
179/// let here = Presented::WorkingTree;
180/// assert_eq!(here.commit(), None);
181/// ```
182#[derive(Debug, Clone, PartialEq, Eq)]
183pub enum Presented {
184    /// The reader's checkout: HEAD plus every tracked, staged and untracked change in it.
185    WorkingTree,
186    /// The checked-out commit, judged as committed: its own ledger, its own tree. Only
187    /// [`presented_commit`] builds one, because only it checks that the commit is the one
188    /// checked out and measures the tree.
189    #[non_exhaustive]
190    Commit {
191        /// The full commit id.
192        commit: String,
193        /// Whether the checkout was that commit, ignoring the evidence ledger's working
194        /// copy, weakened by anything the caller knew.
195        tree: TreeState,
196    },
197}
198
199impl Presented {
200    /// The presented commit, when a commit rather than the working tree is presented.
201    ///
202    /// ```
203    /// use majordomus_cli::evidence::freshness::{presented_commit, Presented};
204    /// use std::process::Command;
205    ///
206    /// let dir = tempfile::tempdir().unwrap();
207    /// let git = |args: &[&str]| {
208    ///     let out = Command::new("git").arg("-C").arg(dir.path()).args(args).output().unwrap();
209    ///     String::from_utf8_lossy(&out.stdout).trim().to_string()
210    /// };
211    /// git(&["init", "-q"]);
212    /// git(&["config", "user.email", "t@example.com"]);
213    /// git(&["config", "user.name", "t"]);
214    /// git(&["commit", "-q", "--allow-empty", "-m", "one"]);
215    ///
216    /// let p = presented_commit(dir.path(), "HEAD", None).unwrap();
217    /// assert_eq!(p.commit(), Some(git(&["rev-parse", "HEAD"]).as_str()));
218    /// assert_eq!(Presented::WorkingTree.commit(), None);
219    /// ```
220    pub fn commit(&self) -> Option<&str> {
221        match self {
222            Presented::WorkingTree => None,
223            Presented::Commit { commit, .. } => Some(commit),
224        }
225    }
226
227    /// The presented tree as the truth table reads it: [`TreeState::Clean`] for the working
228    /// tree, whose changes are already in the diff, and the measured tree for a commit.
229    ///
230    /// ```
231    /// use majordomus_cli::evidence::freshness::{Presented, TreeState};
232    /// assert_eq!(Presented::WorkingTree.tree(), TreeState::Clean);
233    /// ```
234    pub fn tree(&self) -> TreeState {
235        match self {
236            Presented::WorkingTree => TreeState::Clean,
237            Presented::Commit { tree, .. } => *tree,
238        }
239    }
240}

248/// Whether git would read `rev` as an option rather than a revision. A ledger row's commit
249/// is data anyone can write, and `--output=<file>` in a revision's place makes `git diff` or
250/// `git show` write that file; no commit begins with a dash, so such a word names none and
251/// never reaches git as a revision.
252fn option_shaped(rev: &str) -> bool {
253    rev.starts_with('-')
254}

256/// The full commit a revision names, or `None` when it names none here.
257fn commit_of(root: &Path, rev: &str) -> Option<String> {
258    git::read_only(root)
259        .args([
260            "rev-parse",
261            "--verify",
262            "--quiet",
263            &format!("{rev}^{{commit}}"),
264        ])
265        .output()
266        .ok()
267        .filter(|out| out.status.success())
268        .map(|out| String::from_utf8_lossy(&out.stdout).trim().to_string())
269        .filter(|id| !id.is_empty())
270}

272/// The checked-out commit, presented as committed.
273///
274/// `rev` must name the commit that is checked out: the claims matrix, the tests' sources and
275/// the digests a verdict reads come from the checkout, so a verdict "at" some other commit
276/// would mix two trees. The tree is measured, ignoring the evidence ledger's working copy —
277/// the ledger is read from the commit, so its working copy is not part of what is judged —
278/// and `given` can only weaken what was measured.
279///
280/// ```
281/// use majordomus_cli::evidence::freshness::{presented_commit, TreeState};
282/// use std::process::Command;
283///
284/// let dir = tempfile::tempdir().unwrap();
285/// let git = |args: &[&str]| {
286///     Command::new("git").arg("-C").arg(dir.path()).args(args).output().unwrap()
287/// };
288/// git(&["init", "-q"]);
289/// git(&["config", "user.email", "t@example.com"]);
290/// git(&["config", "user.name", "t"]);
291/// git(&["commit", "-q", "--allow-empty", "-m", "one"]);
292///
293/// assert_eq!(presented_commit(dir.path(), "HEAD", None).unwrap().tree(), TreeState::Clean);
294/// // a caller that knows better can weaken it
295/// let told = presented_commit(dir.path(), "HEAD", Some(TreeState::Dirty)).unwrap();
296/// assert_eq!(told.tree(), TreeState::Dirty);
297/// // a revision that names nothing is refused, and the refusal names it
298/// let refused = presented_commit(dir.path(), "no-such-branch", None).unwrap_err();
299/// assert!(refused.contains("no-such-branch"), "{refused}");
300/// ```
301pub fn presented_commit(
302    root: &Path,
303    rev: &str,
304    given: Option<TreeState>,
305) -> std::result::Result<Presented, String> {
306    let commit = match git::resolves(root, rev) {
307        true => commit_of(root, rev),
308        false => None,
309    }
310    .ok_or_else(|| format!("`{rev}` names no commit in this repository"))?;
311    let head = commit_of(root, "HEAD").ok_or_else(|| {
312        "this repository has no checked-out commit, so no commit can be judged as committed"
313            .to_string()
314    })?;
315    if commit != head {
316        return Err(format!(
317            "{rev} is {}, not the checked-out commit {}; check it out to judge at it",
318            short12(&commit),
319            short12(&head)
320        ));
321    }
322    let measured = TreeState::parse(&git::working_tree_ignoring(root, &[LEDGER_PATH]));
323    let tree = measured.weaker(given.unwrap_or(TreeState::Clean));
324    Ok(Presented::Commit { commit, tree })
325}

327/// The ledger as `commit` holds it: `git show <commit>:.ai/repo/evidence/ledger.json`.
328///
329/// A commit that holds no ledger has an empty one, as a repository with no ledger file does.
330/// A commit this clone does not have is an error, not an empty ledger: "nothing recorded"
331/// and "nothing to read" are different answers.
332///
333/// ```
334/// use majordomus_cli::evidence::freshness::ledger_at;
335/// use majordomus_cli::evidence::Ledger;
336/// use std::process::Command;
337///
338/// let dir = tempfile::tempdir().unwrap();
339/// let git = |args: &[&str]| {
340///     let out = Command::new("git").arg("-C").arg(dir.path()).args(args).output().unwrap();
341///     String::from_utf8_lossy(&out.stdout).trim().to_string()
342/// };
343/// git(&["init", "-q"]);
344/// git(&["config", "user.email", "t@example.com"]);
345/// git(&["config", "user.name", "t"]);
346/// git(&["commit", "-q", "--allow-empty", "-m", "nothing recorded yet"]);
347/// let head = git(&["rev-parse", "HEAD"]);
348///
349/// assert_eq!(ledger_at(dir.path(), &head).unwrap(), Ledger::empty());
350/// assert!(ledger_at(dir.path(), &"0".repeat(40)).is_err());
351/// ```
352pub fn ledger_at(root: &Path, commit: &str) -> Result<Ledger> {
353    if option_shaped(commit) {
354        return Err(Error::Git {
355            reason: format!("`{commit}` is not a commit: git would read it as an option"),
356        });
357    }
358    let spec = format!("{commit}:{LEDGER_PATH}");
359    let shown = git::read_only(root)
360        .args(["show", &spec])
361        .output()
362        .ok()
363        .filter(|out| out.status.success());
364    if let Some(out) = shown {
365        return Ledger::parse(&String::from_utf8_lossy(&out.stdout));
366    }
367    if git::resolves(root, commit) {
368        // the commit is here and holds no ledger: nothing was recorded at it
369        return Ok(Ledger::empty());
370    }
371    Err(Error::Git {
372        reason: format!("`{commit}` names no commit git can read in this repository"),
373    })
374}

376/// Every execution the working copy of the ledger holds that `committed` does not.
377///
378/// The ledger keeps one execution per test, so a row that differs from the committed row of
379/// its test, or that the committed ledger has no row for, is a run recorded since. These are
380/// read only through [`weakened_by`]: they withhold `proven` and never decide a verdict.
381///
382/// An absent working copy holds none. An unreadable one is an error rather than none: a run
383/// it may hold cannot be ruled out, so a verdict that judged past it could read `proven` over
384/// a failure the checkout has.
385///
386/// ```
387/// use majordomus_cli::evidence::freshness::uncommitted;
388/// use majordomus_cli::evidence::{Ledger, LEDGER_PATH};
389///
390/// let root = tempfile::tempdir().unwrap();
391/// // nothing in the working tree: nothing uncommitted
392/// assert!(uncommitted(root.path(), &Ledger::empty()).unwrap().is_empty());
393///
394/// std::fs::create_dir_all(root.path().join(".ai/repo/evidence")).unwrap();
395/// std::fs::write(root.path().join(LEDGER_PATH), "not json").unwrap();
396/// let refused = uncommitted(root.path(), &Ledger::empty()).unwrap_err().to_string();
397/// assert!(refused.contains(LEDGER_PATH), "{refused}");
398/// ```
399pub fn uncommitted(root: &Path, committed: &Ledger) -> Result<Vec<Execution>> {
400    if !Ledger::present(root) {
401        return Ok(Vec::new());
402    }
403    let working = Ledger::load(root).map_err(|e| Error::InvalidSurface {
404        surface: "evidence".into(),
405        reason: format!(
406            "the working copy of {LEDGER_PATH} cannot be read, so a run it may hold cannot be \
407             ruled out: {e}"
408        ),
409    })?;
410    Ok(working
411        .executions
412        .into_iter()
413        .filter(|e| committed.latest(&e.test) != Some(e))
414        .collect())
415}

419/// Every path that differs between the evidence commit and the presented revision, or
420/// `None` when git cannot answer — which no caller may read as "nothing changed".
421///
422/// For the working tree this is the evidence module's own working-tree diff: tracked
423/// changes committed since, staged or edited, and untracked files. For a presented commit
424/// it is the committed difference between the two commits, and nothing in the checkout.
425///
426/// An evidence commit git would read as an option (`--output=<file>`, which makes `git
427/// diff` write that file) is `None` before git is asked: a ledger row is data, and no data
428/// may choose git's options.
429///
430/// ```
431/// use majordomus_cli::evidence::freshness::{changed_between, presented_commit, Presented};
432/// use std::process::Command;
433///
434/// let dir = tempfile::tempdir().unwrap();
435/// let git = |args: &[&str]| {
436///     let out = Command::new("git").arg("-C").arg(dir.path()).args(args).output().unwrap();
437///     String::from_utf8_lossy(&out.stdout).trim().to_string()
438/// };
439/// git(&["init", "-q"]);
440/// git(&["config", "user.email", "t@example.com"]);
441/// git(&["config", "user.name", "t"]);
442/// git(&["commit", "-q", "--allow-empty", "-m", "one"]);
443/// let e = git(&["rev-parse", "HEAD"]);
444/// std::fs::write(dir.path().join("a.md"), "a").unwrap();
445/// git(&["add", "-A"]);
446/// git(&["commit", "-q", "-m", "two"]);
447/// std::fs::write(dir.path().join("b.md"), "b").unwrap();
448///
449/// // the working tree carries the untracked file; the commit does not
450/// let here = changed_between(dir.path(), &e, &Presented::WorkingTree).unwrap();
451/// assert!(here.contains("a.md") && here.contains("b.md"));
452/// let committed = presented_commit(dir.path(), "HEAD", None).unwrap();
453/// let there = changed_between(dir.path(), &e, &committed).unwrap();
454/// assert_eq!(there.into_iter().collect::<Vec<_>>(), ["a.md"]);
455/// ```
456pub fn changed_between(
457    root: &Path,
458    evidence_commit: &str,
459    presented: &Presented,
460) -> Option<BTreeSet<String>> {
461    if option_shaped(evidence_commit) {
462        return None;
463    }
464    match presented {
465        Presented::WorkingTree => super::changed_since(root, evidence_commit),
466        Presented::Commit { commit, .. } => git::read_only(root)
467            .args([
468                "diff",
469                "--name-only",
470                evidence_commit,
471                commit.as_str(),
472                "--",
473            ])
474            .output()
475            .ok()
476            .filter(|out| out.status.success())
477            .map(|out| {
478                String::from_utf8_lossy(&out.stdout)
479                    .lines()
480                    .map(|l| l.trim().to_string())
481                    .filter(|l| !l.is_empty())
482                    .collect()
483            }),
484    }
485}

487/// What git says about one evidence commit against the presented revision: whether the
488/// revision contains it, and what changed between them.
489///
490/// ```
491/// use majordomus_cli::evidence::freshness::Comparison;
492/// use majordomus_cli::git::Containment;
493///
494/// // git could not be asked: the truth table reads this as `stale`, never as unchanged
495/// let unknown = Comparison { containment: Containment::CommitUnknown, changed: None };
496/// assert!(unknown.changed.is_none());
497/// ```
498#[derive(Debug, Clone, PartialEq, Eq)]
499pub struct Comparison {
500    /// Whether the presented revision contains the evidence commit.
501    pub containment: Containment,
502    /// The paths that differ between the two, or `None` when git could not answer.
503    pub changed: Option<BTreeSet<String>>,
504}
505
506/// Compare one evidence commit with the presented revision. For the working tree the
507/// containing commit is HEAD (an unborn HEAD contains nothing anyone can name); for a
508/// presented commit it is that commit.
509///
510/// When git cannot place the evidence commit in the presented revision — either of the two
511/// names no commit here — nothing is diffed and `changed` is `None`: the truth table reads
512/// that as `stale` already, and a word that names no commit never reaches `git diff`.
513///
514/// ```
515/// use majordomus_cli::evidence::freshness::{compare, Presented};
516/// use majordomus_cli::git::Containment;
517/// use std::process::Command;
518///
519/// let dir = tempfile::tempdir().unwrap();
520/// let git = |args: &[&str]| {
521///     let out = Command::new("git").arg("-C").arg(dir.path()).args(args).output().unwrap();
522///     String::from_utf8_lossy(&out.stdout).trim().to_string()
523/// };
524/// git(&["init", "-q"]);
525/// git(&["config", "user.email", "t@example.com"]);
526/// git(&["config", "user.name", "t"]);
527/// git(&["commit", "-q", "--allow-empty", "-m", "one"]);
528/// let e = git(&["rev-parse", "HEAD"]);
529///
530/// let c = compare(dir.path(), &e, &Presented::WorkingTree);
531/// assert_eq!(c.containment, Containment::Contains);
532/// assert_eq!(c.changed, Some(Default::default()));
533/// let gone = compare(dir.path(), &"0".repeat(40), &Presented::WorkingTree);
534/// assert_eq!(gone.containment, Containment::CommitUnknown);
535/// ```
536pub fn compare(root: &Path, evidence_commit: &str, presented: &Presented) -> Comparison {
537    let containing = presented.commit().unwrap_or("HEAD");
538    let containment = git::contains(root, containing, evidence_commit);
539    let changed = match containment {
540        Containment::CommitUnknown => None,
541        _ => changed_between(root, evidence_commit, presented),
542    };
543    Comparison {
544        containment,
545        changed,
546    }
547}

574/// use majordomus_cli::evidence::ProofState;

586    pub state: ProofState,
587    /// The declared inputs that changed since the run, in the route's own order, when they
588    /// are what made it `stale`.

590    /// Why, in a sentence, when the state alone does not say.
591    pub detail: Option<String>,
592}

612/// The one truth table: what a recorded run proves at the presented revision.
613///
614/// Pure: every fact it needs is an argument. `inputs` is what the route names — a claim's
615/// source, implementation and test source, a rule's test source and definition — or `None`
616/// for a route that declares none. `test_moved` is whether the test's source no longer
617/// hashes to the recorded digest. `presented_tree` is [`TreeState::Clean`] for the working
618/// tree, whose changes are already in `comparison`, and the measured tree for a commit.
619///
620/// Let `changed'` be the comparison's paths without the ledger's own. The first row that
621/// matches decides:
622///
623/// | # | condition | state |
624/// |---|---|---|
625/// | 1–3 | no test, unrunnable, nothing recorded | `no_test`, `unrunnable`, `not_run` |
626/// | 4 | the run was a skip | `not_run` |
627/// | 5 | it failed, timed out or errored | `failing` |
628/// | 6 | a pass git could not compare | `stale` |
629/// | 7 | a pass the presented revision does not contain | `stale` |
630/// | 8 | a pass whose test moved, no declared input changed | `stale`, naming the test |
631/// | 9 | a pass, a declared input changed | `stale`, naming those inputs |
632/// | 10 | a pass, `changed'` empty, both trees clean | `proven` |
633/// | 11 | as 10, the recorded tree not clean | `inputs_unchanged` |
634/// | 12 | as 10, the presented tree not clean | `inputs_unchanged` |
635/// | 13 | a pass, `changed'` not empty, no declared input among it | `inputs_unchanged` |
636/// | 14 | as 13, but the route declares no inputs | `stale` |
637///
638/// ```
639/// use majordomus_cli::evidence::freshness::{freshness, Comparison, Recorded, TreeState};
640/// use majordomus_cli::evidence::{Execution, ProofState};
641/// use majordomus_cli::git::Containment;
642///
643/// let e: Execution = serde_json::from_value(serde_json::json!({
644///     "test": "suite:07_scope", "runner": "suite", "source": "test/cases/07_scope.sh",
645///     "outcome": "pass", "seconds": 1, "commit": "a".repeat(40), "working_tree": "clean",
646///     "digest": "sha256:0", "at": "2026-09-26T00:00:00Z", "origin": "local",
647///     "command": "bash test/run.sh 07_scope"
648/// }))
649/// .unwrap();
650/// let inputs = vec!["docs/SCOPE.md".to_string()];
651/// let other_history = Comparison {
652///     containment: Containment::DoesNotContain,
653///     changed: Some(Default::default()),
654/// };
655/// let j = freshness(
656///     Recorded::Ran(&e),
657///     Some(&inputs),
658///     "test/cases/07_scope.sh",
659///     false,
660///     Some(&other_history),
661///     TreeState::Clean,
662/// );
663/// // a run on a commit the presented revision does not contain proves some other history
664/// assert_eq!(j.state, ProofState::Stale);
665/// assert!(j.detail.unwrap().contains("does not contain"));
666/// ```
667pub fn freshness(
668    recorded: Recorded<'_>,
669    inputs: Option<&[String]>,
670    test_source: &str,
671    test_moved: bool,
672    comparison: Option<&Comparison>,
673    presented_tree: TreeState,
674) -> Judgement {
675    let e = match recorded {
676        Recorded::NoTest => return Judgement::of(ProofState::NoTest),
677        Recorded::Unrunnable => return Judgement::of(ProofState::Unrunnable),
678        Recorded::NotRun => return Judgement::of(ProofState::NotRun),
679        Recorded::Ran(e) => e,
680    };
681    match e.outcome {
682        Outcome::Pass => {}
683        Outcome::Skip => {
684            return Judgement::because(ProofState::NotRun, "the test declined to run");
685        }
686        Outcome::Fail => return Judgement::because(ProofState::Failing, "failed"),
687        Outcome::Timeout => return Judgement::because(ProofState::Failing, "timed out"),
688        Outcome::Error => {
689            return Judgement::because(ProofState::Failing, "the harness could not run it");
690        }
691    }
692
693    let uncompared = || {
694        Judgement::because(
695            ProofState::Stale,
696            format!(
697                "git could not compare {} with the presented revision",
698                short12(&e.commit)
699            ),
700        )
701    };
702    let Some(comparison) = comparison else {
703        return uncompared();
704    };
705    let changed = match (comparison.containment, &comparison.changed) {
706        (Containment::CommitUnknown, _) | (_, None) => return uncompared(),
707        (Containment::DoesNotContain, Some(_)) => {
708            return Judgement::because(
709                ProofState::Stale,
710                format!(
711                    "recorded on {}, which the presented revision does not contain",
712                    short12(&e.commit)
713                ),
714            );
715        }
716        (Containment::Contains, Some(changed)) => changed,
717    };
718
719    let declared_changed: Vec<String> = inputs
720        .unwrap_or_default()
721        .iter()
722        .filter(|p| changed.contains(*p))
723        .cloned()
724        .collect();
725    if test_moved && declared_changed.is_empty() {
726        return Judgement {
727            state: ProofState::Stale,
728            changed: vec![test_source.to_string()],
729            detail: None,
730        };
731    }
732    if !declared_changed.is_empty() {
733        return Judgement {
734            state: ProofState::Stale,
735            changed: declared_changed,
736            detail: None,
737        };
738    }
739
740    let changed_at_all = changed.iter().any(|p| p != LEDGER_PATH);
741    if !changed_at_all {
742        // the diff says the presented revision is the commit the run sat on; the run's own
743        // tree says whether it measured that commit, and the presented tree whether what is
744        // shown is that commit. All three, or it is not proven.
745        let capped = capped_by_working_tree(ProofState::Proven, &e.working_tree);
746        if capped != ProofState::Proven {
747            let why = if TreeState::parse(&e.working_tree) == TreeState::Dirty {
748                "the run measured a tree that was not its commit"
749            } else {
750                "the run's tree was not measured, so it cannot be shown to be its commit"
751            };
752            return Judgement::because(capped, why);
753        }
754        if presented_tree != TreeState::Clean {
755            return Judgement::because(
756                ProofState::InputsUnchanged,
757                "the presented revision was built from a tree that was not its commit",
758            );
759        }
760        return Judgement::of(ProofState::Proven);
761    }
762    match inputs {
763        Some(_) => Judgement::of(ProofState::InputsUnchanged),
764        None => Judgement::because(
765            ProofState::Stale,
766            "the route names no inputs, so a change since the run cannot be ruled out",
767        ),
768    }
769}

773/// Whether a verdict of [`freshness`] is *current* evidence of what its route names: the one
774/// answer a reader that counts a route as met — an intent criterion, for one — asks for, so
775/// that no such reader draws its own line between a pass and a proof.
776///
777/// `proven` is current. `inputs_unchanged` is current only when both trees were the commits
778/// they sit on: the run measured its own commit (`recorded`), and the presented revision is
779/// its own (`presented`; [`TreeState::Clean`] for the working tree, whose changes are already
780/// in the comparison). That is the line [`freshness`] cannot draw by itself, because rows 11,
781/// 12 and 13 of its table all say `inputs_unchanged`: a clean run with nothing it names
782/// changed since is current, and a run on a dirty tree is not — its commit does not describe
783/// what ran ([`super::capped_by_working_tree`]), so no later tree can be matched against it.
784/// Currency is therefore a property of the inputs a route names and of the recorded digest,
785/// never of commit equality with HEAD: committing the ledger after a run, or any change the
786/// route does not name, leaves a current verdict current. Every other state — `stale`,
787/// `failing`, `not_run`, `unrunnable`, `no_test` — is not current.
788///
789/// ```
790/// use majordomus_cli::evidence::freshness::{current, TreeState::{Clean, Dirty, Unknown}};
791/// use majordomus_cli::evidence::ProofState;
792///
793/// assert!(current(ProofState::Proven, Clean, Clean));
794/// assert!(current(ProofState::InputsUnchanged, Clean, Clean));
795/// // a pass on a tree that was not its commit is not current, whatever changed since
796/// assert!(!current(ProofState::InputsUnchanged, Dirty, Clean));
797/// assert!(!current(ProofState::InputsUnchanged, Unknown, Clean));
798/// assert!(!current(ProofState::InputsUnchanged, Clean, Dirty));
799/// // and a stale pass is a pass of something else
800/// assert!(!current(ProofState::Stale, Clean, Clean));
801/// assert!(!current(ProofState::Failing, Clean, Clean));
802/// ```
803pub fn current(state: ProofState, recorded: TreeState, presented: TreeState) -> bool {
804    match state {
805        ProofState::Proven => true,
806        ProofState::InputsUnchanged => {
807            recorded == TreeState::Clean && presented == TreeState::Clean
808        }
809        _ => false,
810    }
811}

815/// A record read beside the verdict's own execution, which may weaken it and never
816/// strengthens it.
817///
818/// Its two containments are computed by the caller with [`git::contains`], so that
819/// [`weakened_by`] stays pure: `after_evidence` is whether the record's commit contains the
820/// evidence commit, `in_presented` whether the presented commit contains the record's.
821///
822/// ```
823/// use majordomus_cli::evidence::freshness::{Supplementary, UNCOMMITTED_RUN};
824/// use majordomus_cli::evidence::Execution;
825/// use majordomus_cli::git::Containment;
826///
827/// let e: Execution = serde_json::from_value(serde_json::json!({
828///     "test": "suite:07_scope", "runner": "suite", "source": "test/cases/07_scope.sh",
829///     "outcome": "fail", "seconds": 1, "commit": "b".repeat(40), "working_tree": "clean",
830///     "digest": "sha256:0", "at": "2026-09-26T00:00:00Z", "origin": "local",
831///     "command": "bash test/run.sh 07_scope"
832/// }))
833/// .unwrap();
834/// let record = Supplementary {
835///     execution: &e,
836///     named: UNCOMMITTED_RUN,
837///     after_evidence: Containment::Contains,
838///     in_presented: Containment::Contains,
839/// };
840/// assert_eq!(record.named, "an uncommitted run in the working ledger");
841/// ```
842#[derive(Debug, Clone, Copy)]
843pub struct Supplementary<'a> {
844    /// The record's execution.
845    pub execution: &'a Execution,
846    /// How a detail names the record, for example [`UNCOMMITTED_RUN`].
847    pub named: &'a str,
848    /// Whether the record's commit contains the evidence commit: a record from before the
849    /// evidence says nothing against it.
850    pub after_evidence: Containment,
851    /// Whether the presented commit contains the record's: a record from another history
852    /// says nothing about this one.
853    pub in_presented: Containment,
854}

856/// The monotone rule: a supplementary record may withhold `proven` and never grant it.
857///
858/// It changes a judgement only when the judgement is `proven` or `inputs_unchanged` and some
859/// record failed, timed out or errored on a clean tree, at a commit that contains the
860/// evidence commit and that the presented revision contains. Then the state becomes
861/// `stale`, naming the record. The cap is `stale` and not `failing`: a supplementary record
862/// withholds `proven` and never decides a verdict.
863///
864/// ```
865/// use majordomus_cli::evidence::freshness::{weakened_by, Judgement, Supplementary, UNCOMMITTED_RUN};
866/// use majordomus_cli::evidence::{Execution, ProofState};
867/// use majordomus_cli::git::Containment;
868///
869/// let failed: Execution = serde_json::from_value(serde_json::json!({
870///     "test": "suite:07_scope", "runner": "suite", "source": "test/cases/07_scope.sh",
871///     "outcome": "fail", "seconds": 1, "commit": "b".repeat(40), "working_tree": "clean",
872///     "digest": "sha256:0", "at": "2026-09-26T00:00:00Z", "origin": "local",
873///     "command": "bash test/run.sh 07_scope"
874/// }))
875/// .unwrap();
876/// let record = Supplementary {
877///     execution: &failed,
878///     named: UNCOMMITTED_RUN,
879///     after_evidence: Containment::Contains,
880///     in_presented: Containment::Contains,
881/// };
882/// let proven = Judgement { state: ProofState::Proven, changed: vec![], detail: None };
883/// let j = weakened_by(proven, &[record]);
884/// assert_eq!(j.state, ProofState::Stale);
885/// assert!(j.detail.unwrap().contains(&"b".repeat(12)));
886///
887/// // and never the other way
888/// let failing = Judgement { state: ProofState::Failing, changed: vec![], detail: None };
889/// assert_eq!(weakened_by(failing.clone(), &[record]), failing);
890/// ```
891pub fn weakened_by(judgement: Judgement, records: &[Supplementary<'_>]) -> Judgement {
892    if judgement.state >= ProofState::Stale {
893        return judgement;
894    }
895    let counter = records.iter().find_map(|r| {
896        let verb = match r.execution.outcome {
897            Outcome::Fail => "failed",
898            Outcome::Timeout => "timed out",
899            Outcome::Error => "errored",
900            Outcome::Pass | Outcome::Skip => return None,
901        };
902        let clean = TreeState::parse(&r.execution.working_tree) == TreeState::Clean;
903        let placed =
904            r.after_evidence == Containment::Contains && r.in_presented == Containment::Contains;
905        (clean && placed).then_some((r, verb))
906    });
907    match counter {
908        None => judgement,
909        Some((r, verb)) => Judgement {
910            state: ProofState::Stale,
911            changed: judgement.changed,
912            detail: Some(format!(
913                "{} recorded on {} {verb}, so the pass before it proves nothing at the presented \
914                 revision",
915                r.named,
916                short12(&r.execution.commit)
917            )),
918        },
919    }
920}

924/// Several states made one: a failure outranks an absence, and otherwise the weakest part
925/// that can carry proof decides.
926///
927/// Each part is a state and whether it can carry proof. The result is `failing` when any
928/// part is `failing`; otherwise the weakest (the greatest, by the enum's declared order) of
929/// the parts that can carry proof; otherwise the weakest of all parts; and `None` for none.
930///
931/// Why the first rule: the declared order ranks `failing` above `not_run`, so a plain
932/// maximum over a failing test and a test that never ran reads `not_run`, and hides the
933/// counter-evidence behind an absence. Why the second: a path no runner and no gate drives
934/// can never carry proof, so it counts only when it is all there is.
935///
936/// ```
937/// use majordomus_cli::evidence::freshness::aggregate;
938/// use majordomus_cli::evidence::ProofState::{Failing, NotRun, Proven, Unrunnable};
939///
940/// assert_eq!(aggregate(&[(Failing, true), (NotRun, true)], Failing), Some(Failing));
941/// assert_eq!(aggregate(&[(Proven, true), (NotRun, true)], Failing), Some(NotRun));
942/// assert_eq!(aggregate(&[(Proven, true), (Unrunnable, false)], Failing), Some(Proven));
943/// assert_eq!(aggregate(&[(Unrunnable, false)], Failing), Some(Unrunnable));
944/// assert_eq!(aggregate::<majordomus_cli::evidence::ProofState>(&[], Failing), None);
945/// ```
946pub fn aggregate<S: Ord + Copy>(parts: &[(S, bool)], failing: S) -> Option<S> {
947    if parts.iter().any(|(s, _)| *s == failing) {
948        return Some(failing);
949    }
950    parts
951        .iter()
952        .filter(|(_, carries)| *carries)
953        .map(|(s, _)| *s)
954        .max()
955        .or_else(|| parts.iter().map(|(s, _)| *s).max())
956}
