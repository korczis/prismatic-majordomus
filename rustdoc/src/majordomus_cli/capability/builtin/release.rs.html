1//! The `release` module: the changelog, the version, and the compatibility analysis behind
2//! it — all three derived.
3//!
4//! Three capabilities, all read-only, all answered by [`crate::release`] — the same code the
5//! command line renders, the generated document is written from, and the site page shows.
6//! Neither reads a file somebody maintains: the changelog composes the layer's own release
7//! records, decisions and the repository's commits, and the version report reads the one
8//! place the version is authored and says whether the projection the shell tool prints is
9//! current.
10//!
11//! Raising the version is not here. It writes tracked files, which makes it a repository
12//! mutation, and the exposure policy keeps repository mutations off every machine surface —
13//! so `release bump` is a command and nothing more, and this module is the read half.
14

34/// Which part of the changelog to answer with.
35#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
36#[serde(deny_unknown_fields)]
37#[schemars(rename = "ReleaseChangelogInput")]
38pub struct ChangelogInput {
39    #[serde(default, skip_serializing_if = "Option::is_none")]
40    /// One version, or `unreleased`; every section when absent.
41    pub version: Option<String>,
42}
43
44impl BenchmarkCases for ChangelogInput {
45    /// One real version from the layer's own records, so the narrow case is timed against
46    /// something that exists rather than against a filter that matches nothing. A
47    /// repository with no records has a narrow case all the same — the coverage gate counts
48    /// targets, not repositories, and the OpenAPI operation's example of `version` *is* this
49    /// case, which case 92 refuses to see missing — so it falls back to `unreleased`, the
50    /// one section a repository has whether it has released anything or not.
51    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
52        let version = ctx
53            .index
54            .objects
55            .iter()
56            .find(|o| o.kind == crate::release::changelog::RELEASE_KIND)
57            .and_then(|o| o.metadata.get("version"))
58            .and_then(|v| v.as_str())
59            .map_or_else(|| "unreleased".to_string(), str::to_string);
60        vec![
61            NamedCase::new("all", ChangelogInput::default()),
62            NamedCase::new(
63                "one-version",
64                ChangelogInput {
65                    version: Some(version),
66                },
67            ),
68        ]
69    }
70}

72/// Which release to measure this tree against.
73///
74/// ```
75/// use majordomus_cli::capability::builtin::release::AnalysisInput;
76/// // Absent means the last release the layer records, which is the canonical baseline.
77/// let input = AnalysisInput::default();
78/// assert!(input.since.is_none());
79/// let explicit = AnalysisInput { since: Some("v0.4.0".into()) };
80/// assert_eq!(explicit.since.as_deref(), Some("v0.4.0"));
81/// ```
82#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
83#[serde(deny_unknown_fields)]
84#[schemars(rename = "ReleaseAnalysisInput")]
85pub struct AnalysisInput {
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    /// A ref to compare with instead of the last release: `v0.4.0`, or any commit that
88    /// carries a committed registry.
89    pub since: Option<String>,
90}
91
92impl BenchmarkCases for AnalysisInput {
93    /// None. The capability is waived from the benchmark
94    /// ([`WaiverReason::PublishedHistory`]) because its input is a release this repository
95    /// has published, and a benchmark fixture has published nothing — so every case here
96    /// would time a refusal and be counted as coverage. The trait is still implemented
97    /// because the registry asks every input type for its cases; answering "none" is the
98    /// honest answer rather than a case that measures an error path.
99    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
100        Vec::new()
101    }
102}

187/// The capability's own id. Beside its declaration, so the two cannot drift apart without
188/// the test below noticing; the document carries it so that no page has to enumerate where
189/// this value is served.
190pub const CHANGELOG_ID: &str = "release.changelog";
191
192/// Where else the changelog can be had, read off the declarations that already say so.
193///
194/// One answer for every writer of the document: the handler that serves it and the generator
195/// that commits it both ask here, so the committed artifact and the live answer name the same
196/// surfaces. The route, the tool and the resource are the capability's own declaration; the
197/// command line is the local command that renders this capability, which `cli::LOCAL`
198/// declares once beside the reason it is local. Nothing is spelled out a second time, and a
199/// page that lists these resolves each against the datasets the registry generates rather
200/// than typing a route of its own. `None` only when the registry does not carry the
201/// capability at all, which the module's own test refuses.
202///
203/// ```
204/// use majordomus_cli::capability::builtin::release::{module, produced_by};
205/// use majordomus_cli::capability::registry::CapabilityRegistry;
206///
207/// let registry = CapabilityRegistry::builder()
208///     .with_modules(vec![module()])
209///     .build()
210///     .expect("the release module composes on its own");
211/// let by = produced_by(&registry).expect("the registry carries release.changelog");
212/// assert_eq!(by.capability, "release.changelog");
213/// assert_eq!(by.cli.as_deref(), Some("majordomus release changelog"));
214/// assert_eq!(by.http.as_deref(), Some("/api/v1/changelog"));
215/// assert_eq!(by.mcp_tool.as_deref(), Some("majordomus_changelog"));
216/// assert_eq!(by.mcp_resource.as_deref(), Some("majordomus://changelog"));
217/// ```
218pub fn produced_by(registry: &CapabilityRegistry) -> Option<ProducedBy> {
219    let c = registry.get(CHANGELOG_ID)?;
220    let cli = crate::cli::local::LOCAL
221        .iter()
222        .find(|l| l.reason.renders() == Some(CHANGELOG_ID))
223        .map(|l| format!("majordomus {}", l.command));
224    Some(ProducedBy {
225        capability: c.id.to_string(),
226        cli,
227        http: c.exposure.http.as_ref().map(|h| h.path.clone()),
228        mcp_tool: c.exposure.mcp.as_ref().and_then(|m| m.tool.clone()),
229        mcp_resource: c
230            .exposure
231            .mcp
232            .as_ref()
233            .and_then(|m| m.resource.as_ref().map(|r| r.uri.clone())),
234    })
235}

237fn changelog(ctx: &Context, input: ChangelogInput) -> Result<Changelog, CapabilityError> {
238    let root = std::path::Path::new(&ctx.index.repository.root);
239    let mut log = release::compose(root, &ctx.index.objects);
240    log.produced_by = produced_by(&ctx.registry);
241    if let Some(wanted) = input.version.as_deref() {
242        log.sections.retain(|s| {
243            s.version == wanted
244                || s.tag.as_deref() == Some(wanted)
245                || (wanted == "unreleased" && s.unreleased)
246        });
247        if log.sections.is_empty() {
248            return Err(CapabilityError::NotFound(format!(
249                "no release section carries the version '{wanted}'"
250            )));
251        }
252    }
253    Ok(log)
254}

256/// The version report, whose `next` is the contract's answer (ADR 0051).
257///
258/// [`release::version::select`] — the same selection `release bump` takes its default target
259/// from, so the two never answer differently. When the analysis cannot be made — no published
260/// baseline carries a registry, or a version is not three numbers and the analysis could only
261/// guess — the report is still answered: `next` is absent, `decided_by` is `undecided` and the
262/// reason is carried, beside the commit evidence that does not depend on it.
263fn version(ctx: &Context, _: Empty) -> Result<VersionReport, CapabilityError> {
264    let root = std::path::Path::new(&ctx.index.repository.root);
265    Ok(release::version::select(root, &ctx.registry, &ctx.index.objects).report)
266}

268/// The one analysis, behind every surface that shows a version verdict.
269///
270/// A surface that cannot be read is a refusal and never an empty diff: an empty diff means
271/// "nothing changed", and answering that when the truth is "the baseline could not be read"
272/// is the false negative this whole subsystem exists to stop.
273fn analysis(ctx: &Context, input: AnalysisInput) -> Result<VersionPlan, CapabilityError> {
274    let root = std::path::Path::new(&ctx.index.repository.root);
275    release::compat::analyze(
276        root,
277        &ctx.registry,
278        &ctx.index.objects,
279        input.since.as_deref(),
280    )
281    .map_err(|e| CapabilityError::NotFound(e.to_string()))
282}

284#[cfg(test)]
285mod tests {
286    use super::*;
287
288    /// Two capabilities, and every surface that carries them is derived from this
289    /// declaration: the MCP tools, the changelog resource, the HTTP routes, the OpenAPI
290    /// operations. A refactor that renamed a route, dropped the resource or added a third
291    /// would still compile, and the suites that exercise the changelog behind it would
292    /// still pass. This is the assertion that would not.
293    #[test]
294    fn the_declaration_yields_the_projections_it_claims() {
295        let m = module();
296        assert_eq!(m.id.as_str(), "release");
297        assert_eq!(
298            ANALYSIS_ID, "release.analysis",
299            "the id beside the declaration is the declared one"
300        );
301        let expected: &[(&str, &str, &str)] = &[
302            (
303                "release.changelog",
304                "majordomus_changelog",
305                "/api/v1/changelog",
306            ),
307            (
308                "release.version",
309                "majordomus_release_version",
310                "/api/v1/release/version",
311            ),
312            (
313                "release.analysis",
314                "majordomus_release_analysis",
315                "/api/v1/release/analysis",
316            ),
317        ];
318        let ids: Vec<&str> = m
319            .capabilities
320            .iter()
321            .map(|e| e.capability.id.as_str())
322            .collect();
323        let want: Vec<&str> = expected.iter().map(|(id, _, _)| *id).collect();
324        assert_eq!(
325            ids, want,
326            "the module declares a different set of capabilities"
327        );
328        for (executable, (id, tool, path)) in m.capabilities.iter().zip(expected) {
329            let exposure = &executable.capability.exposure;
330            assert_eq!(
331                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
332                Some(*tool),
333                "{id} lost or renamed its MCP tool"
334            );
335            assert_eq!(
336                exposure.http.as_ref().map(|h| h.path.as_str()),
337                Some(*path),
338                "{id} lost or renamed its HTTP route"
339            );
340        }
341        // the changelog is also a resource, so a client may read it without calling a tool
342        let changelog = &m.capabilities[0].capability.exposure;
343        assert_eq!(
344            changelog
345                .mcp
346                .as_ref()
347                .and_then(|m| m.resource.as_ref())
348                .map(|r| r.uri.as_str()),
349            Some(CHANGELOG_URI),
350            "the changelog stopped being an MCP resource"
351        );
352    }
