1//! The proof that the executor's non-mutating cycle moves nothing (`majordomus prs
2//! prove-dry-run`).
3//!
4//! The cycle is the four reads an operator runs before trusting the executor: observe the
5//! forge ([`super::refresh`]), build the plan ([`super::queue_of`]), drain without acting
6//! ([`super::drain::drain`] with `dry_run`) and list what cleanup would close
7//! ([`super::drain::cleanup`] without `apply`). Around it, everything the cycle could move if
8//! it were wrong is snapshotted, section by section:
9//!
10//! ```text
11//! remote  every ref origin serves                         git ls-remote origin
12//! forge   every open pull request: number, head, state and labels   gh pr list
13//! trail   the integration audit trail                     .ai/local/state/integration/events.jsonl
14//! lease   the executor's lease files                      <git-common-dir>/majordomus/locks/
15//! local   every local ref outside the two mirrored namespaces
16//! ```
17//!
18//! The two snapshots must be equal. The refresh fetches into `refs/remotes/origin/<base>`
19//! and `refs/majordomus/prs/<n>` by design, so those are not compared but checked: after the
20//! cycle each must equal what origin serves. The observation, relation and summary caches
21//! are rewritten by every read and are not part of the proof. Nothing here takes a flag
22//! that could merge, close or push: the cycle is fixed.
23

32/// One section of a snapshot: a name and its lines, in a stable order.
33#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
34pub struct SnapshotSection {
35    /// `remote`, `forge`, `trail`, `lease` or `local`.
36    pub name: String,
37    /// What the section holds, one fact per line, sorted where order carries no meaning.
38    pub lines: Vec<String>,
39}

41/// Everything the cycle could move, at one moment.
42#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
43pub struct Snapshot {
44    /// The sections, always in the order `remote`, `forge`, `trail`, `lease`, `local`.
45    pub sections: Vec<SnapshotSection>,
46}

48/// One line that differs between the two snapshots.
49#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
50pub struct Moved {

82/// The verdict: what was compared, what moved, and what the cycle observed.
83#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
84pub struct DryRunProof {
85    /// True exactly when nothing moved and every mirror equals what origin serves.
86    pub ok: bool,
87    /// The base branch the executor works on.
88    pub base: String,
89    /// The cycle, step by step.
90    pub steps: Vec<ProofStep>,
91    /// Every line that differs between the snapshots; empty when nothing moved.
92    pub moved: Vec<Moved>,
93    /// Every mirrored ref that is not what origin serves; empty when they all are.
94    pub mirrors: Vec<String>,
95    /// The snapshot before the cycle.
96    pub before: Snapshot,
97    /// The snapshot after it.
98    pub after: Snapshot,
99    /// Every open pull request as the cycle classified it, in rank order.
100    pub classification: Vec<ObservedClassification>,
101}

103fn run(root: &Path, program: &str, args: &[&str]) -> Result<String, String> {
104    let out = Command::new(program)
105        .current_dir(root)
106        .args(args)
107        .output()
108        .map_err(|e| format!("{program} could not run: {e}"))?;
109    if !out.status.success() {
110        return Err(format!(
111            "{program} {} exited {}: {}",
112            args.join(" "),
113            out.status.code().unwrap_or(-1),
114            String::from_utf8_lossy(&out.stderr).trim()
115        ));
116    }
117    Ok(String::from_utf8_lossy(&out.stdout).into_owned())
118}

134/// One `<sha> <ref>` line of `git ls-remote` or `for-each-ref` output.
135struct RefLine {
136    sha: String,
137    reference: String,
138}
139
140/// A listing is ordered by its ref names, which git keeps unique within one listing.
141impl crate::order::Ordered for RefLine {
142    fn order_key(&self) -> crate::order::OrderKey<'_> {
143        crate::order::OrderKey::plain(&self.reference, &self.reference)
144    }
145}

147/// The `<sha> <ref>` lines of `git ls-remote` or `for-each-ref` output, in canonical order.
148fn ref_lines(text: &str) -> Vec<RefLine> {
149    let mut refs: Vec<RefLine> = text
150        .lines()
151        .filter_map(|l| {
152            let mut parts = l.split_whitespace();
153            Some(RefLine {
154                sha: parts.next()?.to_string(),
155                reference: parts.next()?.to_string(),
156            })
157        })
158        .collect();
159    crate::order::canonical(&mut refs);
160    refs
161}

163/// Whether a local ref is one the refresh mirrors from origin, and so is checked rather
164/// than compared.
165fn mirrored(reference: &str, base: &str) -> bool {
166    reference.starts_with("refs/majordomus/prs/")
167        || reference == format!("refs/remotes/origin/{base}")
168}

186fn forge_lines(json: &str) -> Result<Vec<String>, String> {
187    let prs: Vec<ListedPullRequest> = serde_json::from_str(json)
188        .map_err(|e| format!("gh pr list did not answer a list of pull requests: {e}"))?;
189    // in canonical order once rendered: `#9` before `#10`, by number
190    let mut lines: Vec<String> = prs
191        .into_iter()
192        .map(|p| {
193            let mut labels: Vec<String> = p.labels.into_iter().map(|l| l.name).collect();
194            crate::order::canonical_strings(&mut labels);
195            format!(
196                "#{} {} {} [{}]",
197                p.number,
198                p.head,
199                p.state.as_deref().unwrap_or("-"),
200                labels.join(",")
201            )
202        })
203        .collect();
204    crate::order::canonical_strings(&mut lines);
205    Ok(lines)
206}

208/// The trail as the proof compares it: its acts — every line but an `observed` one — counted
209/// and hashed. A read records what it observed, and the cycle reads, so `observed` lines are
210/// expected to grow; any other line the cycle adds is an act, and the proof must see it.
211fn trail_acts(bytes: &[u8]) -> Vec<String> {
212    let text = String::from_utf8_lossy(bytes);
213    let acts: Vec<&str> = text
214        .lines()
215        .filter(|l| {
216            serde_json::from_str::<serde_json::Value>(l)
217                .ok()
218                .and_then(|v| v.get("action").and_then(|a| a.as_str()).map(str::to_string))
219                .as_deref()
220                != Some("observed")
221        })
222        .collect();
223    vec![
224        format!("acts {}", acts.len()),
225        format!(
226            "sha256 {}",
227            crate::policy::sha256_bytes_hex(acts.join("\n").as_bytes())
228        ),
229    ]
230}

232/// What origin serves and what this clone holds, read together: `git ls-remote origin` and
233/// `git for-each-ref`. One read with one failure, so a caller that cannot reach origin is
234/// refused, and nothing that reads them can be half-informed.
235fn refs(root: &Path) -> Result<(String, String), String> {
236    git(root, &["ls-remote", "origin"]).and_then(|remote| {
237        git(root, &["for-each-ref", "--format=%(objectname) %(refname)"])
238            .map(|local| (remote, local))
239    })
240}

242/// Take a snapshot of everything the cycle could move. `common` is the repository's common
243/// git directory, where its trail and its lease live.
244pub fn snapshot(root: &Path, common: &Path, base: &str) -> Result<Snapshot, String> {
245    let (remote, local) = refs(root)?;
246    let remote: Vec<String> = ref_lines(&remote)
247        .into_iter()
248        .map(|r| format!("{} {}", r.sha, r.reference))
249        .collect();
250    let forge = forge_lines(&run(
251        root,
252        "gh",
253        &[
254            "pr",
255            "list",
256            "--state",
257            "open",
258            "--limit",
259            "1000",
260            "--json",
261            "number,headRefOid,state,labels",
262        ],
263    )?)?;
264    // the repository's trail, where every worktree writes it — read where it is, never
265    // through events_path, which would move an old checkout's trail and be a write itself
266    let trail_path = common
267        .join(super::COMMON_STATE_DIR)
268        .join(super::EVENTS_FILE);
269    let trail = match std::fs::read(&trail_path) {
270        Ok(bytes) => trail_acts(&bytes),
271        // no trail is a trail of no acts: a cycle that only observes may create one
272        Err(e) if e.kind() == std::io::ErrorKind::NotFound => trail_acts(b""),
273        Err(e) => return Err(format!("{}: {e}", trail_path.display())),
274    };
275    let lock = IntegrationLease::path_for(common, base);
276    let lock_dir = lock.parent().map(Path::to_path_buf).unwrap_or_default();
277    let mut lease: Vec<String> = match std::fs::read_dir(&lock_dir) {
278        Ok(entries) => entries
279            .filter_map(|e| e.ok())
280            .filter(|e| e.file_name().to_string_lossy().starts_with("integration-"))
281            .map(|e| {
282                let digest = std::fs::read(e.path())
283                    .map(|b| crate::policy::sha256_bytes_hex(&b))
284                    .unwrap_or_else(|_| "unreadable".into());
285                format!("{} {digest}", e.file_name().to_string_lossy())
286            })
287            .collect(),
288        Err(_) => Vec::new(),
289    };
290    crate::order::canonical_strings(&mut lease);
291    if lease.is_empty() {
292        lease.push("absent".into());
293    }
294    let local: Vec<String> = ref_lines(&local)
295        .into_iter()
296        .filter(|r| !mirrored(&r.reference, base))
297        .map(|r| format!("{} {}", r.sha, r.reference))
298        .collect();
299    let section = |name: &str, lines: Vec<String>| SnapshotSection {
300        name: name.into(),
301        lines,
302    };
303    Ok(Snapshot {
304        sections: vec![
305            section("remote", remote),
306            section("forge", forge),
307            section("trail", trail),
308            section("lease", lease),
309            section("local", local),
310        ],
311    })
312}

314/// Every line in one snapshot and not the other, section by section, in section order:
315/// a line present only before is `removed`, one present only after is `added`.
316pub fn diff(before: &Snapshot, after: &Snapshot) -> Vec<Moved> {
317    let mut moved = Vec::new();
318    let names: Vec<&str> = before
319        .sections
320        .iter()
321        .chain(after.sections.iter())
322        .map(|s| s.name.as_str())
323        .fold(Vec::new(), |mut acc, n| {
324            if !acc.contains(&n) {
325                acc.push(n);
326            }
327            acc
328        });
329    let lines_of = |s: &Snapshot, name: &str| -> Vec<String> {
330        s.sections
331            .iter()
332            .find(|x| x.name == name)
333            .map(|x| x.lines.clone())
334            .unwrap_or_default()
335    };
336    for name in names {
337        let (b, a) = (lines_of(before, name), lines_of(after, name));
338        for line in b.iter().filter(|l| !a.contains(l)) {
339            moved.push(Moved {
340                section: name.into(),
341                change: "removed".into(),
342                line: line.clone(),
343            });
344        }
345        for line in a.iter().filter(|l| !b.contains(l)) {
346            moved.push(Moved {
347                section: name.into(),
348                change: "added".into(),
349                line: line.clone(),
350            });
351        }
352    }
353    moved
354}

356/// Every mirrored local ref that is not what origin serves: `refs/remotes/origin/<base>`
357/// against `refs/heads/<base>`, and each `refs/majordomus/prs/<n>` the refresh mirrored —
358/// `observed`, the open pull requests and the successors it read — against
359/// `refs/pull/<n>/head`. A mirror where origin serves nothing is as wrong as a stale one. A
360/// mirror of a pull request the refresh did not observe is no mirror of this refresh: it is
361/// not judged, because nothing the queue decides reads it.
362pub fn mirror_mismatches(
363    remote: &str,
364    local: &str,
365    base: &str,
366    observed: &std::collections::BTreeSet<u64>,
367) -> Vec<String> {
368    let served = ref_lines(remote);
369    let serves = |r: &str| {
370        served
371            .iter()
372            .find(|l| l.reference == r)
373            .map(|l| l.sha.clone())
374    };
375    let mut found = Vec::new();
376    for RefLine { sha, reference: r } in ref_lines(local) {
377        let wanted = if r == format!("refs/remotes/origin/{base}") {
378            serves(&format!("refs/heads/{base}"))
379        } else if let Some(n) = r.strip_prefix("refs/majordomus/prs/") {
380            if !n.parse::<u64>().is_ok_and(|n| observed.contains(&n)) {
381                continue;
382            }
383            Some(serves(&format!("refs/pull/{n}/head")).unwrap_or_default())
384        } else {
385            continue;
386        };
387        match wanted {
388            Some(w) if w == sha => {}
389            Some(w) if w.is_empty() => {
390                found.push(format!("{r} is {sha}, and origin serves nothing there"))
391            }
392            Some(w) => found.push(format!("{r} is {sha}, and origin serves {w}")),
393            None => {}
394        }
395    }
396    found
397}

399/// Run the cycle between two snapshots and judge it.
400pub fn prove_dry_run(root: &Path) -> Result<DryRunProof, String> {
401    // the repository's common git directory, where its trail and lease are: asked once
402    let common = common_dir(root)?;
403    // the base is the executor's: from the observation when there is one, else the forge's
404    let base = match super::load_observation(root).ok().flatten() {
405        Some(o) => o.base,
406        None => {
407            #[derive(Deserialize)]
408            struct View {
409                #[serde(rename = "defaultBranchRef")]
410                default_branch: Branch,
411            }
412            #[derive(Deserialize)]
413            struct Branch {
414                name: String,
415            }
416            let v: View = serde_json::from_str(&run(
417                root,
418                "gh",
419                &["repo", "view", "--json", "defaultBranchRef"],
420            )?)
421            .map_err(|e| format!("gh repo view did not name the default branch: {e}"))?;
422            v.default_branch.name
423        }
424    };
425    let before = snapshot(root, &common, &base)?;
426
427    let mut steps = Vec::new();
428    // the observation and the plan built from it are one read: the plan reads what the
429    // refresh recorded, so a refresh that succeeded and a plan that cannot follow are one failure
430    let (obs, queue) =
431        super::refresh(root).and_then(|obs| super::queue_of(root).map(|q| (obs, q)))?;
432    steps.push(ProofStep {
433        step: "refresh".into(),
434        summary: format!(
435            "observed {} open pull request(s); {} is {}",
436            obs.pull_requests.len(),
437            obs.base,
438            obs.base_sha
439        ),
440    });
441    steps.push(ProofStep {
442        step: "plan".into(),
443        summary: match queue.next_merge {
444            Some(n) => format!("the next merge is #{n}"),
445            None => "nothing is ready to merge".into(),
446        },
447    });
448    let mut integrator = ForgeIntegrator { root, lease: None };
449    let report = drain::drain(root, &mut integrator, 1, true, false)?;
450    steps.push(ProofStep {
451        step: "drain --dry-run".into(),
452        // the command line's own lines: one sentence per outcome, and the stop only where
453        // no step has said it
454        summary: crate::commands::prs::drain_lines(&report).join("; "),
455    });
456    let listed = drain::cleanup(root, &mut integrator, false)?;
457    steps.push(ProofStep {
458        step: "cleanup".into(),
459        summary: if listed.is_empty() {
460            "nothing to close".into()
461        } else {
462            listed
463                .iter()
464                .map(|i| format!("#{} {}", i.pr, i.action))
465                .collect::<Vec<_>>()
466                .join(", ")
467        },
468    });
469
470    let after = snapshot(root, &common, &base)?;
471    let moved = diff(&before, &after);
472    let observed: std::collections::BTreeSet<u64> = obs
473        .pull_requests
474        .iter()
475        .map(|p| p.number)
476        .chain(obs.resolved.keys().copied())
477        .collect();
478    // what origin serves now and the queue the cycle left, read together
479    let ((remote, local), queue) =
480        refs(root).and_then(|r| super::queue_of(root).map(|q| (r, q)))?;
481    let mirrors = mirror_mismatches(&remote, &local, &base, &observed);
482    let classification = queue
483        .assessments
484        .iter()
485        .map(|a| ObservedClassification {
486            number: a.number,
487            head_sha: a.evaluated_against.head_sha.clone(),
488            disposition: serde_json::to_value(a.disposition)
489                .ok()
490                .and_then(|v| v.as_str().map(str::to_string))
491                .unwrap_or_default(),
492            next_action: a.next_action.clone(),
493        })
494        .collect();
495    Ok(DryRunProof {
496        ok: moved.is_empty() && mirrors.is_empty(),
497        base,
498        steps,
499        moved,
500        mirrors,
501        before,
502        after,
503        classification,
504    })
505}

611        let more = b"{\"action\":\"observed\"}\n{\"action\":\"lease_acquired\"}\n{\"action\":\"observed\"}\n";
612        assert_eq!(
613            trail_acts(one),
614            trail_acts(more),
615            "a read records what it observed"
616        );
617        let act = b"{\"action\":\"observed\"}\n{\"action\":\"lease_acquired\"}\n{\"action\":\"merge_attempted\"}\n";
618        assert_ne!(trail_acts(one), trail_acts(act), "an act is seen");
619        assert_eq!(trail_acts(act)[0], "acts 2");
620        // a line that is not an event is not an observation either: it counts
621        assert_eq!(trail_acts(b"not json\n")[0], "acts 1");
622    }
