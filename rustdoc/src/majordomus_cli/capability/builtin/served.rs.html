1//! The `served` module: whether a deployment serves the commit it was meant to, observed
2//! from outside and recorded as evidence a later reader can re-judge.
3//!
4//! Two operations, deliberately unequal:
5//!
6//! - `served.observe` reaches the network and appends a record, so it is a command of the
7//!   trusted command line only. A browser or an MCP client that could make this process
8//!   probe an arbitrary URL would have been handed a request forger.
9//! - `served.show` is a read: it asks no network, writes nothing and re-judges what was
10//!   recorded against the commit asked about, which is what makes an old observation go
11//!   stale without anyone deciding it has (see [`crate::served`]).
12//!
13//! ```
14//! use majordomus_cli::capability::builtin::served;
15//! use majordomus_cli::capability::model::CapabilityKind;
16//!
17//! let m = served::module();
18//! let observe = m.capabilities.iter().find(|c| c.capability.id.as_str() == "served.observe").unwrap();
19//! // the operation that reaches the network is on no surface a client can call
20//! assert_eq!(observe.capability.kind, CapabilityKind::Command);
21//! assert!(observe.capability.exposure.http.is_none());
22//! assert!(observe.capability.exposure.mcp.is_none());
23//!
24//! let show = m.capabilities.iter().find(|c| c.capability.id.as_str() == "served.show").unwrap();
25//! assert_eq!(show.capability.exposure.http.as_ref().unwrap().path, "/api/v1/served");
26//! ```

53/// The input of `served.observe`: which deployment, where it is, which commit is expected
54/// and how long the probe may take. Everything is optional — the defaults are this
55/// repository's own site, its `build.json` and `HEAD`.
56///
57/// ```
58/// use majordomus_cli::capability::builtin::served::ObserveInput;
59/// let input: ObserveInput = serde_json::from_str(r#"{"commit":"HEAD"}"#).unwrap();
60/// assert_eq!(input.commit.as_deref(), Some("HEAD"));
61/// assert!(serde_json::from_str::<ObserveInput>(r#"{"shell":"rm"}"#).is_err());
62/// ```
63#[derive(Debug, Default, Serialize, Deserialize, JsonSchema)]
64#[serde(deny_unknown_fields)]
65pub struct ObserveInput {
66    /// The name the observation is recorded under; `pages` when absent.
67    #[serde(default, skip_serializing_if = "Option::is_none")]
68    pub deployment: Option<String>,
69    /// The site's base URL; `base_url` of `site/config.toml` when absent.
70    #[serde(default, skip_serializing_if = "Option::is_none")]
71    pub url: Option<String>,
72    /// The identity file under the base; `build.json` when absent.
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub identity: Option<String>,
75    /// The commit expected to be served, any revision git resolves; `HEAD` when absent.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub commit: Option<String>,
78    /// The bound on the probe in seconds; ten when absent.
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub timeout_seconds: Option<u64>,
81    /// Judge without recording. Recording is the default, because an observation nobody
82    /// kept is a measurement nobody can read.
83    #[serde(default)]
84    pub dry_run: bool,
85}

87/// What `served.observe` did: the observation, where it was recorded, and the exit status
88/// the command line reports the verdict with.
89///
90/// ```
91/// use majordomus_cli::capability::builtin::served::Observed;
92/// // the shape a caller reads: a dry run judges and records nothing
93/// let json = r#"{"observation":{"schema":"majordomus.served-observation/v1",
94///   "deployment":"pages","url":"https://majordomus.dev/build.json",
95///   "expected":"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","verdict":"unreachable",
96///   "reason":"nothing was received","at":"1970-01-01T00:00:00Z"},"exit_code":12}"#;
97/// let got: Observed = serde_json::from_str(json).unwrap();
98/// assert_eq!(got.exit_code, 12);
99/// assert!(got.recorded.is_none());
100/// ```
101#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
102pub struct Observed {
103    /// The observation, with its judgement.
104    pub observation: Observation,
105    /// Where it was recorded, repository-relative; absent on a dry run.
106    #[serde(default, skip_serializing_if = "Option::is_none")]
107    pub recorded: Option<String>,
108    /// The exit status the command line reports: 0 served, 10 a measured no, 12 unanswered.
109    pub exit_code: i32,
110}

135/// One deployment's newest record and what it proves about the commit asked. Both are
136/// carried: the recorded judgement is what was seen, `now` is what it still proves.
137///
138/// ```
139/// use majordomus_cli::capability::builtin::served::DeploymentStanding;
140/// use majordomus_cli::served::ServedVerdict;
141/// let json = r#"{"deployment":"pages","observation":{"schema":"majordomus.served-observation/v1",
142///   "deployment":"pages","url":"https://majordomus.dev/build.json",
143///   "expected":"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","verdict":"served",
144///   "reason":"the deployment serves aaaaaaaaaaaa","at":"1970-01-01T00:00:00Z"},
145///   "now":{"verdict":"stale","reason":"it does not contain bbbbbbbbbbbb"}}"#;
146/// let s: DeploymentStanding = serde_json::from_str(json).unwrap();
147/// // seen as served, and no longer proving the commit asked about
148/// assert_eq!(s.observation.judgement.verdict, ServedVerdict::Served);
149/// assert_eq!(s.now.verdict, ServedVerdict::Stale);
150/// ```
151#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
152pub struct DeploymentStanding {
153    /// The deployment name.
154    pub deployment: String,
155    /// The newest record of it.
156    pub observation: Observation,
157    /// That record re-judged against the commit asked about. This, not the recorded
158    /// judgement, is what a reader acts on.
159    pub now: Judgement,
160}

162/// Every deployment's standing against one commit: the answer `served.show` gives on the
163/// command line, over HTTP and to an MCP client, from one derivation.
164///
165/// ```
166/// use majordomus_cli::capability::builtin::served::{standing, ServedStanding};
167/// use majordomus_cli::served::{Ancestry, OBSERVATIONS_PATH};
168/// # struct A; impl Ancestry for A {
169/// #     fn has(&self, _: &str) -> Option<bool> { Some(true) }
170/// #     fn contains(&self, _: &str, _: &str) -> Option<bool> { Some(true) } }
171/// let dir = tempfile::tempdir().unwrap();
172/// let s: ServedStanding = standing(dir.path(), &"a".repeat(40), None, &A);
173/// assert_eq!(s.observations, OBSERVATIONS_PATH);
174/// assert!(s.deployments.is_empty(), "nothing observed yet");
175/// assert_eq!(s.unreadable, 0);
176/// ```
177#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
178pub struct ServedStanding {
179    /// The full commit every record was re-judged against.
180    pub expected: String,
181    /// The checkout-local file the records were read from.
182    pub observations: String,
183    /// Lines of that file that were not readable observations.
184    pub unreadable: usize,
185    /// One entry per deployment observed, in name order.
186    pub deployments: Vec<DeploymentStanding>,
187}

190    PathBuf::from(&ctx.index.repository.root)

193/// Resolves `rev` to a full commit sha, refusing what git does not resolve to a commit.
194fn resolve(root: &Path, rev: &str) -> Result<String, CapabilityError> {
195    if rev.starts_with('-') {
196        return Err(CapabilityError::InvalidInput(format!(
197            "`{rev}` is not a revision"
198        )));
199    }
200    let out = crate::git::read_only(root)
201        .args(["rev-parse", "--verify", "--quiet", "--end-of-options"])
202        .arg(format!("{rev}^{{commit}}"))
203        .output()
204        .map_err(|e| CapabilityError::Internal(format!("cannot run git: {e}")))?;
205    let sha = String::from_utf8_lossy(&out.stdout).trim().to_string();
206    if !out.status.success() || sha.len() != 40 {
207        return Err(CapabilityError::InvalidInput(format!(
208            "`{rev}` does not name a commit this clone holds"
209        )));
210    }
211    Ok(sha)
212}

214/// The `base_url` of the site configuration, when the checkout has one.
215fn configured_base(root: &Path) -> Option<String> {
216    let text = std::fs::read_to_string(root.join("site/config.toml")).ok()?;
217    text.lines().find_map(|l| {
218        let rest = l
219            .trim()
220            .strip_prefix("base_url")?
221            .trim_start()
222            .strip_prefix('=')?;
223        Some(rest.trim().trim_matches('"').to_string())
224    })
225}

227fn observe(ctx: &Context, input: ObserveInput) -> Result<Observed, CapabilityError> {
228    let root = root_of(ctx);
229    let expected = resolve(&root, input.commit.as_deref().unwrap_or("HEAD"))?;
230    let base = match input.url {
231        Some(u) => u,
232        None => configured_base(&root).ok_or_else(|| {
233            CapabilityError::InvalidInput(
234                "no `url` was given and site/config.toml names no base_url".into(),
235            )
236        })?,
237    };
238    served::check_base(&base).map_err(CapabilityError::InvalidInput)?;
239    let identity = input
240        .identity
241        .unwrap_or_else(|| served::DEFAULT_IDENTITY.into());
242    if identity.contains("..") || identity.contains('?') || identity.contains('#') {
243        return Err(CapabilityError::InvalidInput(format!(
244            "`{identity}` is not a plain path under the site"
245        )));
246    }
247    let deployment = input
248        .deployment
249        .unwrap_or_else(|| DEFAULT_DEPLOYMENT.into());
250    let fetch = Curl {
251        max_seconds: input
252            .timeout_seconds
253            .unwrap_or(served::DEFAULT_PROBE_SECONDS)
254            .max(1),
255    };
256    let obs = served::observe(
257        &deployment,
258        &base,
259        &identity,
260        &expected,
261        &fetch,
262        &GitAncestry { root: &root },
263        SystemTime::now(),
264    );
265    let recorded = if input.dry_run {
266        None
267    } else {
268        served::append(&root, &obs).map_err(|e| {
269            CapabilityError::Internal(format!("cannot record the observation: {e}"))
270        })?;
271        Some(served::OBSERVATIONS_PATH.to_string())
272    };
273    Ok(Observed {
274        exit_code: obs.judgement.verdict.exit_code(),
275        observation: obs,
276        recorded,
277    })
278}

280fn show(ctx: &Context, input: ShowInput) -> Result<ServedStanding, CapabilityError> {
281    let root = root_of(ctx);
282    let expected = resolve(&root, input.commit.as_deref().unwrap_or("HEAD"))?;
283    Ok(standing(
284        &root,
285        &expected,
286        input.deployment.as_deref(),
287        &GitAncestry { root: &root },
288    ))
289}

291/// The standing of every recorded deployment against `expected`. Public so that a consumer
292/// that judges an intent's deployment criterion reads the same derivation as this read.
293///
294/// ```
295/// use majordomus_cli::capability::builtin::served::standing;
296/// use majordomus_cli::served::{append, observe, Ancestry, Fetch, Fetched, ServedVerdict};
297/// use std::time::SystemTime;
298/// struct Linear;  // a <- b
299/// impl Ancestry for Linear {
300///     fn has(&self, c: &str) -> Option<bool> { Some(c.starts_with('a') || c.starts_with('b')) }
301///     fn contains(&self, d: &str, a: &str) -> Option<bool> { Some(d >= a) }
302/// }
303/// struct Serves(String);
304/// impl Fetch for Serves {
305///     fn fetch(&self, _: &str) -> Fetched {
306///         Fetched::Body(format!(r#"{{"commit":"{}","dirty":false}}"#, self.0).into_bytes())
307///     }
308/// }
309/// let (a, b) = ("a".repeat(40), "b".repeat(40));
310/// let dir = tempfile::tempdir().unwrap();
311/// let o = observe("pages", "https://s", "build.json", &a, &Serves(a.clone()), &Linear,
312///                 SystemTime::UNIX_EPOCH);
313/// append(dir.path(), &o).unwrap();
314///
315/// // the record still proves the commit it was taken against ...
316/// let now = standing(dir.path(), &a, None, &Linear);
317/// assert_eq!(now.deployments[0].now.verdict, ServedVerdict::Served);
318/// // ... and stops proving once the commit asked about moves past it
319/// let later = standing(dir.path(), &b, None, &Linear);
320/// assert_eq!(later.deployments[0].now.verdict, ServedVerdict::Stale);
321/// ```
322pub fn standing(
323    root: &Path,
324    expected: &str,
325    only: Option<&str>,
326    git: &dyn served::Ancestry,
327) -> ServedStanding {
328    let (all, unreadable) = served::read_all(root);
329    let mut names: Vec<&str> = all.iter().map(|o| o.deployment.as_str()).collect();
330    // the canonical order every other collection is shown in; natural_cmp is total, so
331    // equal names are adjacent and dedup still removes every repeat
332    crate::order::canonical_strings(&mut names);
333    names.dedup();
334    let deployments = names
335        .into_iter()
336        .filter(|n| only.is_none_or(|o| o == *n))
337        .filter_map(|n| served::latest(&all, n))
338        .map(|o| DeploymentStanding {
339            deployment: o.deployment.clone(),
340            now: o.rejudge(expected, git),
341            observation: o.clone(),
342        })
343        .collect();
344    ServedStanding {
345        expected: expected.into(),
346        observations: served::OBSERVATIONS_PATH.into(),
347        unreadable,
348        deployments,
349    }
350}

386        id: "served",
387        title: "Served deployments",
388        description: "Whether a deployment serves the commit it was meant to: the build identity a site serves, read from outside the way a visitor reads it, judged by commit containment and recorded so that a later reader can re-judge it. Only `served` passes; a site that could not be reached, an identity that could not be read, a build from an uncommitted tree and a served commit this clone does not hold are each an unanswered question, never a yes. Observing reaches the network and is a command of the trusted command line; reading the recorded standing is a read on every surface.",
389        stability: Stability::BehaviorallyVerified,

392                id: "served.observe",

394                title: "Observe what a deployment serves",
