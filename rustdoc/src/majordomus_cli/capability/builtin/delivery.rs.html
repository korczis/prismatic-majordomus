1//! The `delivery` module: whether each product feature exists, computed on every read.
2//!
3//! A feature exists only when it is on the trunk, deployed, publicly verified, has its
4//! required tests current, has that evidence published and is linked from the interface
5//! (ADR 0071). Every one of those is a verdict derived here from git, from the public site's
6//! own identity and — from phase 2 — from recorded evidence; none is a field anybody wrote.
7//! Unknown is never pass.
8//!
9//! Both capabilities are reads, and both may reach the network: the public identity is read
10//! from the site, and `scripts/pages` asks the site again and asks GitHub. That is why the
11//! benchmark waives them as an external dependency, and why nothing here is cached: a
12//! publication that changed a second ago is the answer a person asking wants.
13

30/// The input of `delivery.feature`: which feature.
31#[derive(Debug, Serialize, Deserialize, JsonSchema)]
32#[serde(deny_unknown_fields)]
33pub struct DeliveryFeatureInput {
34    /// The feature's id, as `product.features` gives it.
35    pub id: String,
36}

38impl BenchmarkCases for DeliveryFeatureInput {
39    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
40        let id = ctx
41            .index
42            .objects
43            .iter()
44            .find(|o| o.kind == crate::product::FEATURE)
45            .map(|o| o.identity.clone())
46            .unwrap_or_else(|| "absent".into());
47        vec![NamedCase::new("first-feature", DeliveryFeatureInput { id })]
48    }
49}

51fn report_for(ctx: &Context, only: Option<&str>) -> DeliveryReport {
52    let root = std::path::Path::new(&ctx.index.repository.root);
53    let subjects: Vec<Subject> = ctx
54        .product
55        .all()
56        .iter()
57        .filter(|r| only.is_none_or(|id| r.feature.id == id))
58        .map(|r| Subject {
59            id: r.feature.id.clone(),
60            title: r.feature.title.clone(),
61            status: r.feature.status.clone(),
62            paths: revision::implementation_paths(root, r, &ctx.index),
63        })
64        .collect();
65    delivery::assess(root, subjects, Probe::for_repository(root).observe())
66}
67
68fn delivery_report(ctx: &Context, _: Empty) -> Result<DeliveryReport, CapabilityError> {
69    Ok(report_for(ctx, None))
70}
71
72fn delivery_trends(ctx: &Context, _: Empty) -> Result<TrendsReport, CapabilityError> {
73    Ok(trends::read(std::path::Path::new(
74        &ctx.index.repository.root,
75    )))
76}
77
78fn delivery_feature(
79    ctx: &Context,
80    input: DeliveryFeatureInput,
81) -> Result<FeatureDelivery, CapabilityError> {
82    if ctx.product.feature(&input.id).is_none() {
83        return Err(CapabilityError::NotFound(format!(
84            "no feature '{}'; `product.features` names every one this repository holds",
85            input.id
86        )));
87    }
88    report_for(ctx, Some(&input.id))
89        .features
90        .into_iter()
91        .next()
92        .ok_or_else(|| CapabilityError::NotFound(format!("no feature '{}'", input.id)))
93}

265    #[test]
266    fn a_merged_feature_with_no_site_is_on_master_and_does_not_exist() {
267        let repo = repository();
268        let ctx = repo.context().expect("a context");
269        let report = delivery_report(&ctx, Empty {}).expect("a report");
270        assert_eq!(report.tallies.features, 1);
271        let f = &report.features[0];
272        assert_eq!(f.id, "shipped");
273        assert_eq!(
274            f.implementation.paths,
275            [
276                "apps/majordomus-cli/src/capability/builtin/delivery.rs",
277                "lib/start.sh"
278            ]
279        );
280        assert_eq!(f.dimensions[0].verdict, Verdict::Pass, "{f:?}");
281        // a repository with no site and no publication model is not deployed: it is unknown
282        assert_eq!(f.dimensions[1].verdict, Verdict::Unknown);
283        assert!(!f.exists);
284        assert!(matches!(
285            f.delivery,
286            DeliveryState::NotDelivered {
287                stage: DevelopmentStage::OnMaster,
288                ..
289            }
290        ));
291        let one = delivery_feature(
292            &ctx,
293            DeliveryFeatureInput {
294                id: "shipped".into(),
295            },
296        )
297        .expect("one feature");
298        assert_eq!(&one, f);
299    }
