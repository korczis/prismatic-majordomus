1//! `majordomus devcontext`: the context compiler, through the registry's own
2//! `devcontext.*` capabilities.
3//!
4//! This file renders; it selects nothing. Every answer below is the output of the capability
5//! the CLI exposure names, executed through the one executor, so the command line, the HTTP
6//! routes and the MCP tools cannot answer the same request differently — which is the
7//! property a context compiler most needs, because two surfaces that disagreed about what a
8//! session should be given would be two compilers.
9

19/// The exit code when the entries that may not be dropped already exceed the budget: the
20/// same code, for the same condition, as the shell tool's `majordomus context`.
21pub const EXIT_OVER_BUDGET: u8 = 10;
22
23/// Run `majordomus devcontext`.
24pub fn run(args: DevcontextArgs) -> Result<u8> {
25    let app = App::load(&args.repo)?;
26    let format = args.format;
27    match &args.command {
28        DevcontextCommand::Compile(req) => {
29            let v = call(&app, &["devcontext", "compile"], request_of(req))?;
30            emit(format, &v, compiled_text)?;
31            Ok(if v["budget"]["over_budget"].as_bool().unwrap_or(false) {
32                EXIT_OVER_BUDGET
33            } else {
34                0
35            })
36        }
37        DevcontextCommand::Explain { uri, request } => {
38            let mut input = request_of(request);
39            input["uri"] = Value::String(uri.clone());
40            let v = call(&app, &["devcontext", "explain"], input)?;
41            emit(format, &v, explanation_text)
42        }
43        DevcontextCommand::Policy => {
44            let v = call(&app, &["devcontext", "policy"], serde_json::json!({}))?;
45            emit(format, &v, policy_text)
46        }
47    }
48}

50/// The request flags as the capability's own input. The domain type is serialised rather
51/// than a JSON object assembled by hand, so a field added to the request is a flag here and
52/// a key of the input at once.
53fn request_of(r: &DevcontextRequest) -> Value {
54    let input = crate::devcontext::CompileInput {
55        issue: r.issue.clone(),
56        milestone: r.milestone.clone(),
57        intent: r.intent.clone(),
58        paths: r.paths.clone(),
59        uris: r.uris.clone(),
60        budget_tokens: r.budget_tokens,
61        max_depth: r.max_depth,
62        floor: r.floor,
63        all_blocking_rules: r.all_blocking_rules.then_some(true),
64    };
65    serde_json::to_value(input).unwrap_or(Value::Null)
66}

68fn call(app: &App, path: &[&str], input: Value) -> Result<Value> {
69    let ctx = &app.context;
70    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
71    let id = ctx
72        .registry
73        .by_cli(&words)
74        .map(|c| c.id.to_string())
75        .ok_or_else(|| Error::Protocol {
76            reason: format!(
77                "no capability is exposed as `majordomus {}`",
78                path.join(" ")
79            ),
80        })?;
81    ctx.execute(&id, input).map_err(map)
82}

84fn map(e: CapabilityError) -> Error {
85    match e {
86        // an issue that does not exist is a missing artifact, not an internal failure
87        CapabilityError::NotFound(reason) => Error::NotFound { reason },
88        CapabilityError::InvalidInput(reason) => Error::Protocol { reason },
89        other => Error::Protocol {
90            reason: other.to_string(),
91        },
92    }
93}

95fn emit(format: OutputFormat, v: &Value, text: impl Fn(&Value) -> String) -> Result<u8> {
96    let stdout = std::io::stdout();
97    let mut out = stdout.lock();
98    let body = match format {
99        OutputFormat::Json => serde_json::to_string_pretty(v).unwrap_or_else(|_| v.to_string()),
100        OutputFormat::Text => text(v),
101    };
102    writeln!(out, "{body}").map_err(Error::Transport)?;
103    Ok(0)
104}

110fn short(sha: &str) -> &str {
111    sha.get(..12).unwrap_or(sha)

114fn compiled_text(v: &Value) -> String {
115    let mut o = String::new();
116    let git = &v["git"];
117    o.push_str(&format!(
118        "devcontext   index {}  head {} ({}, {})\n",
119        short(s(&v["fingerprint"])),
120        short(s(&git["head"])),
121        git["branch"].as_str().unwrap_or("detached"),
122        s(&git["working_tree"])
123    ));
124    for seed in v["seeds"].as_array().into_iter().flatten() {
125        o.push_str(&format!(
126            "seed         {:<10} {}\n",
127            s(&seed["kind"]),
128            s(&seed["uri"])
129        ));
130    }
131    let b = &v["budget"];
132    o.push_str(&format!(
133        "budget       {} of {} token(s) used, {} required, {} bytes per token{}\n",
134        b["used_tokens"],
135        b["limit_tokens"],
136        b["required_tokens"],
137        b["bytes_per_token"],
138        if b["over_budget"].as_bool().unwrap_or(false) {
139            "  OVER BUDGET: what may not be dropped already exceeds the limit"
140        } else {
141            ""
142        }
143    ));
144    for t in b["tiers"].as_array().into_iter().flatten() {
145        o.push_str(&format!(
146            "  {:<11} {:>3} selected {:>3} dropped {:>7} tokens\n",
147            s(&t["tier"]),
148            t["selected"],
149            t["excluded"],
150            t["tokens"]
151        ));
152    }
153
154    let selected = v["selected"].as_array().cloned().unwrap_or_default();
155    o.push_str(&format!("\nSELECTED ({})\n", selected.len()));
156    o.push_str("TIER        REL   CONF  TOKENS  URI\n");
157    for e in &selected {
158        o.push_str(&format!(
159            "{:<11} {:<5.3} {:<5.3} {:>6}  {}{}\n",
160            s(&e["tier"]),
161            e["relevance"].as_f64().unwrap_or(0.0),
162            e["confidence"].as_f64().unwrap_or(0.0),
163            e["cost_tokens"],
164            s(&e["uri"]),
165            if e["required"].as_bool().unwrap_or(false) {
166                "  (required)"
167            } else {
168                ""
169            }
170        ));
171        for d in e["discovered_by"].as_array().into_iter().flatten() {
172            let edge = d["edge"]
173                .as_str()
174                .map(|x| format!("({x})"))
175                .unwrap_or_default();
176            o.push_str(&format!(
177                "            <- {}{} {}\n",
178                s(&d["selector"]),
179                edge,
180                s(&d["reason"])
181            ));
182        }
183    }
184
185    let excluded = v["excluded"].as_array().cloned().unwrap_or_default();
186    o.push_str(&format!("\nEXCLUDED ({})\n", excluded.len()));
187    for e in &excluded {
188        o.push_str(&format!(
189            "{:<11} {:<10} {:>6}  {}  — {}\n",
190            s(&e["tier"]),
191            s(&e["reason"]),
192            e["cost_tokens"],
193            s(&e["uri"]),
194            s(&e["detail"])
195        ));
196    }
197
198    let dedup = v["deduplicated"].as_array().cloned().unwrap_or_default();
199    o.push_str(&format!("\nDEDUPLICATED ({})\n", dedup.len()));
200    for d in &dedup {
201        o.push_str(&format!(
202            "{:<9} {}  — {}\n",
203            s(&d["key"]),
204            s(&d["kept"]),
205            s(&d["detail"])
206        ));
207    }
208
209    let conflicts = v["conflicts"].as_array().cloned().unwrap_or_default();
210    if !conflicts.is_empty() {
211        o.push_str(&format!("\nCONFLICTS ({})\n", conflicts.len()));
212        for c in &conflicts {
213            o.push_str(&format!(
214                "{:<11} {} over {}  — {}\n",
215                s(&c["kind"]),
216                s(&c["current"]),
217                s(&c["against"]),
218                s(&c["detail"])
219            ));
220        }
221    }
222
223    let commits = git["commits"].as_array().cloned().unwrap_or_default();
224    if !commits.is_empty() {
225        o.push_str(&format!("\nCOMMITS THE LAYER NAMES ({})\n", commits.len()));
226        for c in &commits {
227            o.push_str(&format!(
228                "{}  {}{}\n",
229                short(s(&c["commit"])),
230                s(&c["named_by"]),
231                c["covers"]
232                    .as_str()
233                    .map(|x| format!("  ({x})"))
234                    .unwrap_or_default()
235            ));
236        }
237    }
238
239    let diags = v["diagnostics"].as_array().cloned().unwrap_or_default();
240    if !diags.is_empty() {
241        o.push_str(&format!("\nDIAGNOSTICS ({})\n", diags.len()));
242        for d in &diags {
243            o.push_str(&format!(
244                "{:<8} {:<22} {}\n",
245                s(&d["severity"]),
246                s(&d["code"]),
247                s(&d["message"])
248            ));
249        }
250    }
251    o.trim_end().to_string()
252}

254fn explanation_text(v: &Value) -> String {
255    format!(
256        "{}  {}\n{}\nbudget {} of {} token(s) used",
257        s(&v["standing"]),
258        s(&v["uri"]),
259        s(&v["detail"]),
260        v["budget"]["used_tokens"],
261        v["budget"]["limit_tokens"]
262    )
263}

265fn policy_text(v: &Value) -> String {
266    let mut o = String::new();
267    o.push_str(&format!(
268        "graph        {}\ndefaults     budget {} tokens, depth {}, floor {}, {} bytes per token\n\nTIER  POSITION  KINDS\n",
269        s(&v["graph"]),
270        v["default_budget_tokens"],
271        v["default_max_depth"],
272        v["default_floor"],
273        v["bytes_per_token"]
274    ));
275    for t in v["tiers"].as_array().into_iter().flatten() {
276        let kinds: Vec<&str> = t["kinds"]
277            .as_array()
278            .map(|a| a.iter().filter_map(Value::as_str).collect())
279            .unwrap_or_default();
280        o.push_str(&format!(
281            "{:<11} {}  {}\n            {}\n",
282            s(&t["tier"]),
283            t["position"],
284            kinds.join(", "),
285            s(&t["meaning"])
286        ));
287    }
288    o.push_str("\nEDGE             FORWARD  REVERSE  REASON\n");
289    for e in v["edges"].as_array().into_iter().flatten() {
290        let w = |k: &str| {
291            e[k].as_f64()
292                .map(|x| format!("{x}"))
293                .unwrap_or_else(|| "-".into())
294        };
295        o.push_str(&format!(
296            "{:<16} {:<8} {:<8} {}{}\n",
297            s(&e["edge"]),
298            w("forward"),
299            w("reverse"),
300            if e["refused"].as_bool().unwrap_or(false) {
301                "REFUSED: "
302            } else {
303                ""
304            },
305            s(&e["reason"])
306        ));
307    }
308    o.push_str("\nSELECTOR       DECLARED\n");
309    for sel in v["selectors"].as_array().into_iter().flatten() {
310        o.push_str(&format!(
311            "{:<14} {}\n",
312            s(&sel["selector"]),
313            if sel["declared"].as_bool().unwrap_or(false) {
314                "yes"
315            } else {
316                "no — inferred, confidence below one"
317            }
318        ));
319    }
320    o.trim_end().to_string()
321}

328    #[test]
329    fn the_text_says_what_was_selected_excluded_and_why() {
330        let v = json!({
331            "fingerprint": "abcdef0123456789",
332            "git": {"head": "0123456789abcdef", "branch": "master", "working_tree": "clean",
333                    "commits": [{"commit": "feedfacefeedface", "named_by": "majordomus://issue/I1", "covers": "proof"}]},
334            "seeds": [{"uri": "majordomus://issue/I1", "kind": "issue", "from": "issue"}],
335            "budget": {"used_tokens": 10, "limit_tokens": 20, "required_tokens": 5,
336                       "bytes_per_token": 4, "over_budget": false,
337                       "tiers": [{"tier": "task", "selected": 1, "excluded": 0, "tokens": 10}]},
338            "selected": [{"tier": "task", "relevance": 1.0, "confidence": 1.0, "cost_tokens": 10,
339                          "uri": "majordomus://issue/I1", "required": true,
340                          "discovered_by": [{"selector": "seed", "reason": "named in the request"}]}],
341            "excluded": [{"tier": "history", "reason": "budget", "cost_tokens": 99,
342                          "uri": "majordomus://claim/x", "detail": "would not fit"}],
343            "deduplicated": [{"key": "uri", "kept": "majordomus://adr/a", "detail": "two paths"}],
344            "conflicts": [],
345            "diagnostics": []
346        });
347        let t = compiled_text(&v);
348        for want in [
349            "SELECTED (1)",
350            "majordomus://issue/I1  (required)",
351            "<- seed named in the request",
352            "EXCLUDED (1)",
353            "budget",
354            "would not fit",
355            "DEDUPLICATED (1)",
356            "COMMITS THE LAYER NAMES (1)",
357            "10 of 20 token(s) used",
358        ] {
359            assert!(t.contains(want), "missing {want:?} in\n{t}");
360        }
361        assert!(!t.contains("OVER BUDGET"));
362    }

364    #[test]
365    fn the_request_flags_become_the_capability_input() {
366        let r = DevcontextRequest {
367            issue: Some("I0001".into()),
368            milestone: None,
369            intent: None,
370            paths: vec!["lib".into()],
371            uris: vec![],
372            budget_tokens: Some(100),
373            max_depth: None,
374            floor: None,
375            all_blocking_rules: false,
376        };
377        let v = request_of(&r);
378        assert_eq!(v["issue"], "I0001");
379        assert_eq!(v["paths"][0], "lib");
380        assert_eq!(v["budget_tokens"], 100);
381        // an unset flag is absent, not null: the input type denies unknown fields and the
382        // executor's cache key is the canonical form
383        assert!(v.get("milestone").is_none());
384        assert!(v.get("all_blocking_rules").is_none());
385    }
