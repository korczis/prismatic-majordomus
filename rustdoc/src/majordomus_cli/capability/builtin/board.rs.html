1//! The `board` module: the delivery board's documents, derived by one capability (I4294).
2//!
3//! `board.snapshot` reads the plan, the roadmap, the intents, the recorded pull request
4//! queue and the release currency, and composes the documents the board's page reads
5//! through `crate::board`, with the lanes `share/board.yaml` declares. It reads and writes
6//! nothing else: the sync step that publishes the documents adds the mesh seats, live work,
7//! CI runners and blockers it observes itself, so none of them is here.

52/// The answer of `board.snapshot`: every document of the board by id, and the digest that
53/// says whether two snapshots describe the same state.
54#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
55pub(crate) struct BoardSnapshot {
56    /// `sha256:<hex>` over the canonical JSON of the documents, `meta.syncedAt` and
57    /// `meta.syncedBy` left out: equal state, equal digest.
58    pub(crate) digest: String,
59    /// The commit described, empty when the checkout has none.
60    pub(crate) master_sha: String,
61    /// Its committer time, RFC 3339.
62    pub(crate) master_committed_at: String,
63    /// The ref it was read from: `refs/remotes/origin/master`, else `HEAD`.
64    pub(crate) master_ref: String,
65    /// The bound every document is under, in bytes of compact JSON.
66    pub(crate) document_limit: usize,
67    /// Each document's compact size, in bytes, by id.
68    pub(crate) sizes: BTreeMap<String, usize>,
69    /// The documents, by id: `meta`, `milestones`, `intents`, `prs`, and `issues-<lane>`
70    /// (then `issues-<lane>-2`, ...) for each lane with open work.
71    pub(crate) documents: BTreeMap<String, Value>,
72    /// What could not be read, in words: an unobserved pull request queue, a lane naming a
73    /// milestone the plan lacks. Empty is the ordinary case.
74    #[serde(default, skip_serializing_if = "Vec::is_empty")]
75    pub(crate) diagnostics: Vec<String>,
76}

78/// The pull request queue, as the board shows it: the recorded observation, offline.
79fn queue(root: &Path) -> Queue {
80    match crate::integration::queue_of(root) {
81        Ok(q) => Queue::Observed {
82            observed_at: q.observed_at.clone(),
83            items: q
84                .assessments
85                .iter()
86                .map(|a| PrItem {
87                    n: a.number,
88                    t: a.title.clone(),
89                    b: a.head_ref.clone(),
90                    disp: board::word(&a.disposition),
91                    lane: board::word(&a.lane),
92                    why: a.reasons.iter().take(4).map(board::word).collect(),
93                    head: a.evaluated_against.head_sha.chars().take(11).collect(),
94                    train: a.head_ref.starts_with("integration/train-"),
95                    risk: board::word(&a.risk),
96                })
97                .collect(),
98        },
99        Err(reason) => Queue::Unavailable(reason),
100    }
101}

103fn board_snapshot(
104    ctx: &Context,
105    input: BoardSnapshotInput,
106) -> Result<BoardSnapshot, CapabilityError> {
107    let root = Path::new(&ctx.index.repository.root);
108    let declaration = match &ctx.index.share {
109        Some(share) => BoardDeclaration::load(share).map_err(CapabilityError::Internal)?,
110        None => BoardDeclaration::default(),
111    };
112    let plan = Plan::build(&ctx.index);
113    let roadmap: Vec<PlanMilestone> = plan
114        .roadmap()
115        .into_iter()
116        .map(|m| PlanMilestone {
117            released_in: Some(crate::plan::released_in(&ctx.index, &plan, &m.id)),
118            ..m.clone()
119        })
120        .collect();
121    let evidence = crate::intent::RepositoryEvidence::load(&ctx.index)
122        .map_err(|e| CapabilityError::Internal(e.to_string()))?;
123    let intents = crate::intent::Intents::build(&ctx.index, &plan, &evidence)
124        .intents
125        .iter()
126        .filter_map(|x| serde_json::to_value(x).ok())
127        .collect();
128    let release = serde_json::to_value(crate::release::currency::measure(
129        root,
130        &ctx.index.objects,
131        std::time::SystemTime::now(),
132    ))
133    .unwrap_or_default();
134    let master = Master::read(root);
135    let composed = board::compose(Inputs {
136        declaration: &declaration,
137        plan: &plan,
138        roadmap,
139        intents,
140        queue: queue(root),
141        release,
142        master: master.clone(),
143        synced_at: crate::peers::rfc3339(std::time::SystemTime::now()),
144        synced_by: input.by.unwrap_or_else(|| "board.snapshot".into()),
145    })
146    .map_err(CapabilityError::Refused)?;
147    let master = master.unwrap_or_default();
148    Ok(BoardSnapshot {
149        digest: board::digest(&composed.documents),
150        master_sha: master.sha,
151        master_committed_at: master.committed_at,
152        master_ref: master.reference,
153        document_limit: board::DOCUMENT_LIMIT,
154        sizes: composed
155            .documents
156            .iter()
157            .map(|(id, d)| (id.clone(), board::size(d)))
158            .collect(),
159        documents: composed.documents,
160        diagnostics: composed.diagnostics,
161    })
162}

164/// The module.
165pub fn module() -> ModuleDescriptor {
166    module! {
167        id: "board",
168        title: "The delivery board",
169        description: "The documents the delivery board reads, derived from the plan, the roadmap, the intents, the recorded pull request queue and the release currency, with the lanes share/board.yaml declares once. Every open milestone is in a lane or the snapshot refuses, naming it.",
170        stability: Stability::Experimental,
171        capabilities: [
172            capability! {
173                id: "board.snapshot",
174                title: "The delivery board's documents",
175                description: "Every document of the delivery board by id: meta (the commit, the lanes, the issue counts, the release), milestones in roadmap order, intents, the pull request queue from the last recorded observation, and the open issues of each lane as issues-<lane>, split into issues-<lane>-2 and onwards so that no document serializes past 250000 bytes. The digest is a SHA-256 over the canonical JSON of the documents with meta.syncedAt and meta.syncedBy left out, so equal state is an equal digest. A milestone that is not DONE and is in no lane of share/board.yaml is refused by name, never filed under other. An unobserved queue is said in prs.unavailable and the diagnostics, never refused. Mesh seats, live work, CI runners and blockers are the sync step's and are not here. Reads only; nothing is cached.",
176                input: BoardSnapshotInput,
177                output: BoardSnapshot,
178                stability: Stability::Experimental,
179                exposure: Exposure {
180                    mcp: mcp("majordomus_board_snapshot"),
181                    http: get("/api/v1/board/snapshot"),
182                    cli: Some(CliExposure { path: vec!["board".into(), "snapshot".into()] }),
183                },
184                tags: ["board", "plan", "delivery"],
185                cache: CachePolicy::Disabled,
186                benchmark: BenchmarkPolicy::Waived { reason: WaiverReason::RepositoryDeclaration },
187                handler: board_snapshot,
188            },
189        ],
190    }
191}

298    /// A recorded queue is listed item by item, with its head shortened, and a train's top
299    /// is told apart from a pull request of one change.

301    fn an_observed_queue_is_listed_item_by_item() {
302        let repo = crate::synthetic::SyntheticRepository::small().expect("a repository");
303        let root = repo.root();
304        git(root, &["init", "-q", "-b", "master"]);
305        git(root, &["add", "-A"]);
306        git(root, &["commit", "-qm", "fixture"]);
307        let master = git(root, &["rev-parse", "HEAD"]);
308        git(root, &["update-ref", "refs/remotes/origin/master", &master]);
309        git(root, &["commit", "-q", "--allow-empty", "-m", "a change"]);
310        let head = git(root, &["rev-parse", "HEAD"]);
311        git(root, &["reset", "-q", "--hard", &master]);
312        let pr = |number: u64, head_ref: &str| -> crate::integration::PullRequestObservation {
313            serde_json::from_value(serde_json::json!({
314                "number": number, "title": format!("pull request {number}"), "author": "a",
315                "head_ref": head_ref, "head_sha": head, "base_ref": "master", "draft": false,
316                "created_at": "t0", "updated_at": "t0",
317            }))
318            .expect("a pull request")
319        };
320        let obs = crate::integration::ForgeObservation {
321            schema: crate::integration::OBSERVATION_SCHEMA,
322            repository: "owner/repo".into(),
323            base: "master".into(),
324            base_sha: master.clone(),
325            observed_at: "t0".into(),
326            required_checks: Some(vec!["ci".into()]),
327            review_policy: Some(Default::default()),
328            up_to_date_required: Some(true),
329            merge_methods: Some(vec!["merge".into()]),
330            pull_requests: vec![pr(7, "fix/one-change"), pr(8, "integration/train-1")],
331            resolved: Default::default(),
332            delete_branch_on_merge: None,
333        };
334        crate::integration::store_observation(root, &obs).expect("the observation is stored");
335        let Queue::Observed { observed_at, items } = queue(root) else {
336            panic!("the recorded queue was not read")
337        };
338        assert_eq!(observed_at, "t0");
339        let item = |n: u64| items.iter().find(|i| i.n == n).expect("the item");
340        assert!(!item(7).train && item(8).train);
341        assert_eq!(item(7).t, "pull request 7");
342        assert_eq!(item(7).b, "fix/one-change");
343        assert_eq!(item(7).head.len(), 11, "{:?}", item(7));

346    /// The context of `repo` with its share replaced by `share`.
347    fn with_share(repo: &crate::synthetic::SyntheticRepository, share: &Path) -> Context {
348        let mut index = repo.index().expect("an index");
349        index.share = Some(share.to_path_buf());
350        let registry = crate::capability::CapabilityRegistry::builder()
351            .with_modules(crate::capability::builtin::modules())
352            .with_index(&index)
353            .build()
354            .expect("a registry");
355        Context::new(std::sync::Arc::new(index), std::sync::Arc::new(registry))
356    }

358    /// What the snapshot cannot read is the executable's own failure, never a board: a lane
359    /// declaration that does not parse and an evidence ledger that cannot be read; an issue
360    /// alone over the document bound is refused, as the board cannot carry it.
361    #[test]
362    fn a_declaration_or_a_ledger_it_cannot_read_fails_and_an_oversized_issue_is_refused() {
363        let repo = crate::synthetic::SyntheticRepository::small().expect("a repository");
364        let share = tempfile::tempdir().expect("a share");
365        std::fs::write(share.path().join(board::BOARD_FILE), "lanes: [unclosed\n").unwrap();
366        let ctx = with_share(&repo, share.path());
367        assert!(matches!(
368            board_snapshot(&ctx, BoardSnapshotInput::default()),
369            Err(CapabilityError::Internal(_))
370        ));
371
372        let ledger = repo.root().join(crate::evidence::LEDGER_PATH);
373        std::fs::create_dir_all(ledger.parent().unwrap()).unwrap();
374        std::fs::write(&ledger, r#"{"version": 99, "executions": []}"#).unwrap();
375        let ctx = repo.context().expect("a context");
376        assert!(matches!(
377            board_snapshot(&ctx, BoardSnapshotInput::default()),
378            Err(CapabilityError::Internal(m)) if m.contains("99")
379        ));
380
381        let planned = crate::synthetic::SyntheticRepository::planned();
382        let issue = planned.root().join(".ai/repo/project/issues/I0001.yaml");
383        let text = std::fs::read_to_string(&issue).unwrap();
384        let long = "x".repeat(board::DOCUMENT_LIMIT + 1);
385        std::fs::write(
386            &issue,
387            text.replace("title: The work of I0001", &format!("title: {long}")),
388        )
389        .unwrap();
390        let lanes = tempfile::tempdir().expect("a share");
391        std::fs::write(
392            lanes.path().join(board::BOARD_FILE),
393            "version: 1\nlanes:\n  - key: all\n    title: All\n    milestones: [m-first, m-second, m-loop]\n",
394        )
395        .unwrap();
396        let ctx = with_share(&planned, lanes.path());
397        match board_snapshot(&ctx, BoardSnapshotInput::default()) {
398            Err(CapabilityError::Refused(m)) => assert!(m.contains("alone is over"), "{m}"),
399            other => panic!("not refused: {other:?}"),
400        }
401    }
