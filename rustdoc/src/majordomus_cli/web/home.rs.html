1//! The page `/` answers with: this process, and everything it serves.
2//!
3//! Every entry on it is a [`crate::web::Surface`] of the resolved topology the router is serving from.
4//! There is no list of links in this file and no template with paths written into it: the
5//! sections are the categories a surface declares, the entries are the public surfaces of
6//! each, and a surface that appears in the topology tomorrow appears here with no edit.
7//! What cannot be derived — whether a static surface's producer has run — is asked of the
8//! surface at render time and shown as its state rather than as a link to a certain 404.
9//!
10//! Every link is relative to the origin, so the page is correct on whatever host, port or
11//! reverse-proxy prefix the process was reached through.
12
13use crate::web::html;
14use crate::web::model::{Category, SurfaceKind, Topology, Visibility};

16/// What the page says about the process, beside what it serves.
17///
18/// Only facts a reader can act on. The repository root is a filesystem path and a
19/// deliberate omission: this page is served to whoever can reach the socket, and where the
20/// repository sits on the host is of no use to them and of some use to somebody else.
21#[derive(Debug, Clone)]
22pub struct Identity<'a> {
23    /// The executable's version.
24    pub version: &'a str,
25    /// The one-line description of what this is.
26    pub summary: &'a str,
27    /// The repository's name, as the index read it.
28    pub repository: &'a str,
29    /// The revision the repository is at, when version control could say.
30    pub revision: Option<&'a str>,
31    /// How many capabilities the registry holds.
32    pub capabilities: usize,
33    /// Which checkout answered, as the repository model names it.
34    pub checkout: Checkout<'a>,
35}
36
37/// Which checkout a process is serving, as the repository model names it: an identity a
38/// reader can compare between two pages, never the place the checkout sits on the host.
39///
40/// Several sessions share one machine and each may bind a server of its own, so a page read
41/// in a browser may come from a server of another worktree or of another repository. Every
42/// surface that renders this says which one answered, and none of it is a path: the identities are digests
43/// (`crate::repository::identity`, `crate::repository::git_identity`), the branch and the
44/// commit are git's names.
45#[derive(Debug, Clone, Default)]
46pub struct Checkout<'a> {
47    /// The checkout's identity: one per checkout root (`crate::repository::identity`).
48    pub id: &'a str,
49    /// The git repository's identity: one for every worktree of it, `None` without git.
50    pub git_repository: Option<&'a str>,
51    /// Whether the checkout is a linked worktree rather than the primary one.
52    pub linked: bool,
53    /// The branch the served tree had checked out, `None` when detached or unknown.
54    pub branch: Option<&'a str>,
55    /// The commit the served answers were read from.
56    pub head: Option<&'a str>,
57}
58
59impl Checkout<'_> {
60    /// One line of HTML naming the checkout: what a page puts beside its content so that a
61    /// reader can tell whose answers they are reading. Every value is escaped.
62    ///
63    /// The module is the crate's own, so the example is the unit test
64    /// `the_line_names_identity_branch_and_head` rather than a doc test.
65    pub fn line(&self, repository: &str) -> String {
66        let mut parts = vec![format!(
67            "serving <strong>{}</strong>, checkout <span class=\"mono\" data-mj-checkout=\"{}\">{}</span> ({})",
68            html::escape(repository),
69            html::escape(self.id),
70            html::escape(short(self.id)),
71            if self.linked { "linked worktree" } else { "primary checkout" }
72        )];
73        if let Some(repository) = self.git_repository {
74            parts.push(format!(
75                "repository <span class=\"mono\" data-mj-repository=\"{}\">{}</span>",
76                html::escape(repository),
77                html::escape(short(repository))
78            ));
79        }
80        if let Some(branch) = self.branch {
81            parts.push(format!(
82                "branch <span class=\"mono\">{}</span>",
83                html::escape(branch)
84            ));
85        }
86        if let Some(head) = self.head {
87            parts.push(format!(
88                "HEAD <span class=\"mono\">{}</span>",
89                html::escape(short(head))
90            ));
91        }
92        format!("<p class=\"mj-checkout\">{}</p>", parts.join(" · "))
93    }
94}

96/// The first twelve characters of an identity or a commit: enough to tell two apart.
97fn short(id: &str) -> &str {
98    &id[..12.min(id.len())]
99}

101/// Whether a surface is ready to answer, decided by asking it rather than by assuming.
102pub type Readiness<'a> = &'a dyn Fn(&str) -> Availability;
103
104/// What a surface can be, from the page's point of view.
105#[derive(Debug, Clone, Copy, PartialEq, Eq)]
106pub enum Availability {
107    /// It answers now.
108    Ready,
109    /// It is part of the topology and its producer has not run.
110    NotBuilt,
111}
112
113/// Render the home page for a topology and the process serving it.
114///
115/// `ready` is asked about every static surface; a native route is always ready, because a
116/// process that serves the page serves its own routes.
117pub fn page(topology: &Topology, identity: &Identity<'_>, ready: Readiness<'_>) -> String {
118    let mut body = String::new();
119    body.push_str(&html::summary(&[
120        ("Version", identity.version.to_string()),
121        ("Capabilities", identity.capabilities.to_string()),
122        ("Surfaces served", topology.surfaces.len().to_string()),
123        (
124            "Revision",
125            identity
126                .revision
127                .map(|r| r[..12.min(r.len())].to_string())
128                .unwrap_or_else(|| "unknown".into()),
129        ),
130        ("Checkout", short(identity.checkout.id).to_string()),
131    ]));
132    body.push_str(&identity.checkout.line(identity.repository));
133
134    for category in Category::ALL {
135        let surfaces = topology.public_in(category);
136        if surfaces.is_empty() {
137            continue;
138        }
139        body.push_str(&format!("<h2>{}</h2>", html::escape(category.title())));
140        let rows: Vec<Vec<String>> = surfaces
141            .iter()
142            .map(|surface| {
143                let target = link_target(surface.mount.as_str(), surface.kind);
144                let state = match surface.kind {
145                    SurfaceKind::StaticDirectory => ready(&surface.id),
146                    _ => Availability::Ready,
147                };
148                let name = match state {
149                    Availability::Ready => format!(
150                        "<a href=\"{}\">{}</a>",
151                        html::escape(&target),
152                        html::escape(&surface.id)
153                    ),
154                    Availability::NotBuilt => html::escape(&surface.id),
155                };
156                vec![
157                    name,
158                    format!("<code>{}</code>", html::escape(&target)),
159                    html::escape(&surface.title),
160                    match state {
161                        Availability::Ready => String::from("<span class=\"pass\">ready</span>"),
162                        Availability::NotBuilt => format!(
163                            "<span class=\"fail\">not built</span> — run <code>{}</code>",
164                            html::escape(&surface.producer)
165                        ),
166                    },
167                ]
168            })
169            .collect();
170        body.push_str(&html::table(
171            &["Surface", "Path", "What it is", "State"],
172            &rows,
173        ));
174    }
175
176    body.push_str(&format!(
177        "<footer>{} — {}{}<br>This page is rendered from the resolved web topology; \
178every surface above was discovered, not listed. <code>majordomus web explain</code> says \
179where each value came from.</footer>",
180        html::escape(identity.repository),
181        html::escape(identity.summary),
182        identity
183            .revision
184            .map(|r| format!(
185                "<br>revision <span class=\"mono\">{}</span>",
186                html::escape(&r[..12.min(r.len())])
187            ))
188            .unwrap_or_default(),
189    ));
190
191    html::page(
192        "Majordomus",
193        &format!(
194            "{} — {} of {} served surface(s) are offered here; the rest are spoken to by a program",
195            html::escape(identity.repository),
196            topology
197                .surfaces
198                .iter()
199                .filter(|s| s.visibility == Visibility::Public)
200                .count(),
201            topology.surfaces.len()
202        ),
203        &body,
204    )
205}

207/// The href a mount is reached through: a directory mount keeps its trailing slash so that
208/// the page it answers with resolves its own relative links correctly.
209fn link_target(mount: &str, kind: SurfaceKind) -> String {
210    match kind {
211        SurfaceKind::StaticDirectory if !mount.ends_with('/') => format!("{mount}/"),
212        _ => mount.to_string(),
213    }
214}

222    fn identity() -> Identity<'static> {
223        Identity {
224            version: "1.2.3",
225            summary: "a control plane",
226            repository: "prismatic-majordomus",
227            revision: Some("abcdef0123456789"),
228            capabilities: 42,
229            checkout: Checkout {
230                id: "0123456789abcdef0123456789abcdef",
231                git_repository: Some("fedcba9876543210fedcba9876543210"),
232                linked: false,
233                branch: Some("master"),
234                head: Some("abcdef0123456789"),
235            },
236        }
237    }

239    #[test]
240    fn the_page_names_the_checkout_that_answered() {
241        let html = page(&topology(), &identity(), &|_| Availability::Ready);
242        assert!(
243            html.contains("data-mj-checkout=\"0123456789abcdef0123456789abcdef\""),
244            "{html}"
245        );
246        assert!(
247            html.contains("data-mj-repository=\"fedcba9876543210fedcba9876543210\""),
248            "{html}"
249        );
250        assert!(html.contains("primary checkout"), "{html}");
251        let without_git = Identity {
252            checkout: Checkout {
253                git_repository: None,
254                branch: None,
255                head: None,
256                ..identity().checkout
257            },
258            ..identity()
259        };
260        let html = page(&topology(), &without_git, &|_| Availability::Ready);
261        assert!(
262            html.contains("data-mj-checkout="),
263            "a checkout without git is still named"
264        );
265        assert!(!html.contains("data-mj-repository="), "{html}");
266    }

268    #[test]
269    fn the_line_names_identity_branch_and_head() {
270        let line = Checkout {
271            id: "0123456789abcdef0123456789abcdef",
272            branch: Some("feature/x"),
273            head: Some("89abcdef0123456789abcdef0123456789abcdef"),

283        assert!(line.contains("linked worktree"), "{line}");
284    }

286    #[test]

299    /// A repository whose only relevant fact is that it has a site.
300    fn with_a_site() -> tempfile::TempDir {
301        let tmp = tempfile::tempdir().expect("tempdir");
302        std::fs::create_dir_all(tmp.path().join("site")).expect("mkdir");
303        std::fs::write(tmp.path().join(discover::SITE_CONFIG), "base_url = \"/\"\n")
304            .expect("write");
305        tmp
306    }
307
308    fn topology() -> Topology {
309        let tmp = with_a_site();
310        let mut surfaces = discover::native(Runtime::full());
311        surfaces.extend(discover::application(tmp.path()));
312        Topology::new(surfaces).served(Runtime::full())
313    }

315    #[test]
316    fn every_public_surface_of_the_topology_is_on_the_page() {
317        let topology = topology();
318        let html = page(&topology, &identity(), &|_| Availability::Ready);
319        for surface in &topology.surfaces {
320            if surface.visibility != Visibility::Public {
