1//! The Dashboard: how the delivery stands, on one screen.
2//!
3//! Its first paint asks only sources that answer in well under a second on this
4//! repository, all at once: `repository.info`, `health.report`, `release.version`,
5//! `distribution.releases`, `plan.status`, `plan.roadmap` and `mesh.peers`. Measured one at a
6//! time on 2026-10-11, the slowest was `release.version` at 0.5-0.8 s. The pull-request queue
7//! (`integration.queue`) took 50 s for 1.2 MB, so it is a panel of its own at
8//! [`PULL_REQUESTS_PANEL`]: the script pulls it in once the screen has painted, and a reader
9//! without JavaScript follows its link.
10

36/// The sources the first paint asks, together.
37pub const FIRST_PAINT: &[Source] = &[REPOSITORY, HEALTH, VERSION, RELEASES, PLAN, ROADMAP, MESH];

39/// How many rows a panel shows before it says how many there are in all.
40pub const ROWS: usize = 8;

42/// The Dashboard screen.
43pub fn screen(ctx: &Context) -> View {
44    let answers = ask_all(ctx, FIRST_PAINT);
45    let a = |s: Source| &answers[&s];
46
47    let git = |p: &'static str| a(REPOSITORY).read(p).text();
48    let lead = match (git("/repository/git/branch"), git("/repository/git/head")) {
49        (Some(branch), Some(head)) => format!(
50            "{branch} at {} · working tree {}",
51            &head[..12.min(head.len())],
52            git("/repository/git/working_tree")
53                .as_deref()
54                .unwrap_or("unknown")
55        ),
56        _ => "The repository's version control could not be read.".into(),
57    };
58
59    let top = hero(
60        "home",
61        "blue",
62        "Dashboard",
63        "How the delivery stands, read from the repository itself.",
64        &lead,
65        vec![
66            kpi_to(kpi_state("Health", &a(HEALTH).read("/status")), "#health"),
67            kpi_to(
68                kpi("Released", &a(RELEASES).read("/latest/version")),
69                super::delivery::ROUTE,
70            ),
71            kpi_to(
72                kpi("Next release", &a(VERSION).read("/next")),
73                super::delivery::ROUTE,
74            ),
75            kpi_to(
76                kpi("Issues ready", &a(PLAN).read("/counts/by_status/READY")),
77                super::work::ROUTE,
78            ),
79            kpi_to(
80                kpi("Issues blocked", &a(PLAN).read("/counts/by_status/BLOCKED")),
81                super::roadmap::ROUTE,
82            ),
83            kpi_to(
84                kpi_count("Mesh machines", &a(MESH).read("/machines")),
85                super::mesh::ROUTE,
86            ),
87        ],
88    );
89
90    let grid = el("div")
91        .class("mj-ui-grid")
92        .child(plan_panel(a(ROADMAP), a(PLAN)))
93        .child(issues_panel(a(PLAN)))
94        .child(wide(queue_placeholder()))
95        .child(releases_panel(a(RELEASES), a(VERSION)))
96        .child(mesh_panel(a(MESH)))
97        .child(health_panel(a(HEALTH)));
98
99    View::new(
100        "Dashboard",
101        el("div").class("mj-ui-stack").child(top).child(grid),
102    )
103}

105/// The pull-request queue, as a panel: what the fragment request and the page without
106/// JavaScript both show.
107pub fn pull_requests_panel(ctx: &Context, query: &[(String, String)]) -> View {
108    let selected = query
109        .iter()
110        .find(|(k, _)| k == "pr")

118/// The milestones still open, in the plan's own order, each with the issues it has done
119/// out of those it holds.
120pub fn plan_panel(roadmap: &Answer, plan: &Answer) -> El {
121    let body = unanswered(roadmap).unwrap_or_else(|| {
122        let rows = list(roadmap, "/milestones")
123            .iter()
124            .filter(|m| m.get("status").and_then(Value::as_str) != Some("DONE"))
125            .take(ROWS)
126            .map(milestone_row)
127            .collect::<Vec<_>>();
128        el("div")
129            .child(el("p").class("mj-ui-line").text(format!(
130                "Now {} · next {}",
131                shown(&roadmap.read("/now")),
132                shown(&roadmap.read("/next"))
133            )))
134            .child(table(
135                &["Milestone", "State", "Issues done"],
136                rows,
137                "No milestone is open.",
138            ))
139    });
140    panel_to(
141        "Roadmap",
142        "The milestones still open, in the plan's own order.",
143        Some(super::roadmap::ROUTE),
144        body,
145        &[roadmap, plan],
146    )
147}

149/// One milestone as a table row: its id and title, its state and its issues done.
150pub fn milestone_row(m: &Value) -> El {
151    let total = m.pointer("/counts/total").and_then(Value::as_u64);
152    let done = m.pointer("/counts/by_status/DONE").and_then(Value::as_u64);
153    let id = text(m, "id");
154    el("tr")
155        .attr("id", format!("milestone-{id}"))

158                el("a")
159                    .class("mj-ui-row-link")

161                    .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))

163            ),
164        )
165        .child(el("td").child(state(m, "status")))
166        .child(el("td").class("mj-ui-cell-meter").child(meter(done, total)))

182/// What was published, newest first, as a timeline.
183pub fn releases_panel(releases: &Answer, version: &Answer) -> El {
184    let body = unanswered(releases).unwrap_or_else(|| {
185        let entries = list(releases, "/releases")
186            .iter()
187            .take(ROWS)
188            .map(|r| {
189                let commit = text(r, "commit");
190                (
191                    text(r, "published_at"),
192                    format!("v{}", text(r, "version")),
193                    el("span").class("mj-ui-dim").text(format!(
194                        "{} · {}",
195                        text(r, "channel"),
196                        &commit[..10.min(commit.len())]
197                    )),
198                )
199            })
200            .collect::<Vec<_>>();
201        el("div")
202            .child(el("p").class("mj-ui-line").text(format!(
203                "Declared {} · the commits since the last release imply a {} release",
204                shown(&version.read("/declared")),
205                shown(&version.read("/bump"))
206            )))
207            .child(if entries.is_empty() {
208                nothing("Nothing has been released yet.")
209            } else {
210                timeline(entries)
211            })
212    });
213    panel_to(
214        "Releases",
215        "What was published, newest first.",
216        Some(super::delivery::ROUTE),
217        body,
218        &[releases, version],
219    )

245/// One machine as a table row: its name, whether it is this one, and the liveness of each
246/// of its runtimes as the mesh runtime observed it.
247pub fn machine_row(m: &Value) -> El {
248    let local = m.get("local").and_then(Value::as_bool);
249    let words: Vec<String> = m
250        .get("runtimes")
251        .and_then(Value::as_array)
252        .map(|rs| {
253            rs.iter()
254                .filter_map(|r| r.get("liveness").and_then(Value::as_str))
255                .map(str::to_string)
256                .collect()
257        })
258        .unwrap_or_default();
259    el("tr")
260        .child(el("td").class("mj-ui-mono").text(text(m, "name")))
261        .child(el("td").text(match local {
262            Some(true) => "this machine",
263            Some(false) => "remote",
264            None => "unknown",
265        }))
266        .child(
267            el("td")
268                .class("mj-ui-badges")
269                .children(words.iter().map(|w| word_badge(w))),
270        )
271}

330/// The queue: its tallies, its throughput and the pull requests in rank order.
331pub fn queue_panel(queue: &Answer, selected: Option<u64>) -> El {
332    let body = unanswered(queue).unwrap_or_else(|| {
333        if queue.read("/observed").state != Availability::Available(Value::Bool(true)) {
334            return el("p").class("mj-ui-empty").text(format!(
335                "The forge was not observed in this checkout: {}",
336                shown(&queue.read("/reason"))
337            ));
338        }
339        let tally = |p: &'static str| queue.read(p);
340        let tallies = strip(vec![
341            kpi("Open", &tally("/queue/tallies/open")),
342            kpi(
343                "Needs refresh",
344                &tally("/queue/tallies/by_disposition/needs_refresh"),
345            ),
346            kpi(
347                "Needs repair",
348                &tally("/queue/tallies/by_disposition/needs_repair"),
349            ),
350            kpi(
351                "Conflicting",
352                &tally("/queue/tallies/by_disposition/conflicting"),
353            ),
354            kpi("Merged, 7 days", &tally("/throughput/merges")),
355            kpi("Merges per day", &tally("/throughput/merges_per_day")),
356        ]);
357        let rows = list(queue, "/queue/assessments")
358            .iter()
359            .take(ROWS * 2)
360            .map(|a| {
361                let number = a
362                    .get("number")
363                    .and_then(Value::as_u64)
364                    .map_or_else(|| "unknown".to_string(), |n| format!("#{n}"));
365                let reasons: Vec<String> = a
366                    .get("reasons")
367                    .and_then(Value::as_array)
368                    .map(|rs| {
369                        rs.iter()
370                            .filter_map(Value::as_str)
371                            .take(2)
372                            .map(str::to_string)
373                            .collect()
374                    })
375                    .unwrap_or_default();
376                let n = a.get("number").and_then(Value::as_u64);
377                el("tr")
378                    .attr_if("id", n.map(|n| format!("pr-{n}")))
379                    .child(el("td").class("mj-ui-mono").text(number))
380                    .child(
381                        el("td").child(
382                            el("a")
383                                .class("mj-ui-row-link")
384                                .attr(
385                                    "href",
386                                    n.map_or_else(
387                                        || super::prs::ROUTE.to_string(),
388                                        |n| format!("{}?pr={n}#pr", super::prs::ROUTE),
389                                    ),
390                                )
391                                .child(el("span").class("mj-ui-cell-title").text(text(a, "title"))),
392                        ),
393                    )
394                    .child(el("td").child(state(a, "disposition")))
395                    .child(el("td").text(text(a, "lane")))
396                    .child(
397                        el("td")
398                            .class("mj-ui-mono mj-ui-dim")
399                            .text(reasons.join(", ")),
400                    )
401            })
402            .collect::<Vec<_>>();
403        let observed = queue
404            .read("/queue/observed_at")
405            .text()
406            .map(|t| t.replace('T', " "))
407            .unwrap_or_else(|| "at an unknown time".into());
408        let detail = selected.map(|n| {
409            list(queue, "/queue/assessments")
410                .into_iter()
411                .find(|a| a.get("number").and_then(Value::as_u64) == Some(n))
412                .map_or_else(
413                    || {
414                        nothing(&format!(
415                            "#{n} is not among the open pull requests the queue observed."
416                        ))
417                    },
418                    |a| pull_request_detail(&a),
419                )
420        });
421        el("div")
422            .class("mj-ui-stack")
423            .children(detail)
424            .child(tallies)
425            .child(table(
426                &["PR", "Title", "Disposition", "Lane", "Why"],
427                rows,
428                "No pull request is open.",
429            ))
430            .child(
431                el("p")
432                    .class("mj-ui-line")
433                    .text(format!("Observed {observed} ({})", day(&observed))),
434            )
435    });
436    panel_to(
437        "Pull-request queue",
438        "What the integration lane would merge, repair or wait for, and why.",
439        Some(super::prs::ROUTE),
440        body,
441        &[queue],
442    )
443}

445/// One pull request as the queue assessed it: what it is, every gate with its verdict, the
446/// evidence each verdict was read from, why it stands where it stands, what would move it,
447/// and the factors of its rank. Read from the queue's own answer, so it costs nothing more.
448fn pull_request_detail(a: &Value) -> El {
449    let s = |k: &str| text(a, k);
450    // an assessment without a number is shown as unknown, never as #0
451    let number = a
452        .get("number")
453        .and_then(Value::as_u64)
454        .map_or_else(|| "unknown".to_string(), |n| n.to_string());
455    let strings = |k: &str| -> Vec<String> {
456        a.get(k)
457            .and_then(Value::as_array)
458            .map(|v| {
459                v.iter()
460                    .filter_map(Value::as_str)
461                    .map(str::to_string)
462                    .collect()
463            })
464            .unwrap_or_default()
465    };
466    let gates = el("ul").class("mj-ui-checks").children(
467        a.get("gates")
468            .and_then(Value::as_array)
469            .cloned()
470            .unwrap_or_default()
471            .iter()
472            .map(|g| {
473                let passed = g.get("passed").and_then(Value::as_bool);
474                el("li")
475                    .class("mj-ui-check")
476                    .child(
477                        el("span")
478                            .class("mj-ui-check-title mj-ui-mono")
479                            .text(text(g, "gate")),
480                    )
481                    .child(word_badge(match passed {
482                        Some(true) => "pass",
483                        Some(false) => "fail",
484                        None => "unknown",
485                    }))
486            }),
487    );
488    let evidence = a
489        .get("evidence")
490        .and_then(Value::as_array)
491        .cloned()
492        .unwrap_or_default()
493        .iter()
494        .map(|e| {
495            el("tr")
496                .child(el("td").class("mj-ui-mono").text(text(e, "kind")))
497                .child(el("td").child(state(e, "status")))
498                .child(el("td").class("mj-ui-dim").text(text(e, "detail")))
499        })
500        .collect::<Vec<_>>();
501    let factors = a
502        .get("rank_factors")
503        .and_then(Value::as_object)
504        .map(|m| {
505            m.iter()
506                .map(|(k, v)| {
507                    el("div")
508                        .class("mj-ui-fact")
509                        .child(el("dt").text(k.replace('_', " ")))
510                        .child(el("dd").class("mj-ui-mono").text(match v {
511                            Value::String(s) => s.clone(),
512                            other => other.to_string(),
513                        }))
514                })
515                .collect::<Vec<_>>()
516        })
517        .unwrap_or_default();
518    el("section")
519        .class("mj-ui-detail")
520        .attr("id", "pr")
521        .attr("aria-label", format!("Pull request #{number}"))
522        .child(
523            el("div")
524                .class("mj-ui-detail-head")
525                .child(el("p").class("mj-ui-mono mj-ui-dim").text(format!(
526                    "#{number} · {} · {} → {}",
527                    s("author"),
528                    s("head_ref"),
529                    s("base_ref")
530                )))
531                .child(el("h3").class("mj-ui-detail-title").text(s("title")))
532                .child(
533                    el("p")
534                        .class("mj-ui-badges")
535                        .child(state(a, "disposition"))
536                        .child(state(a, "lane"))
537                        .child(state(a, "risk")),
538                ),
539        )
540        .child(el("p").class("mj-ui-line").text(format!(
541            "Next: {}",
542            if s("next_action").is_empty() {
543                "nothing to do".into()
544            } else {
545                s("next_action")
546            }
547        )))
548        .child(
549            el("p")
550                .class("mj-ui-mono mj-ui-dim")
551                .text(strings("reasons").join(" · ")),
552        )
553        .child(el("h4").class("mj-ui-panel-title").text("Gates"))
554        .child(gates)
555        .child(el("h4").class("mj-ui-panel-title").text("Evidence"))
556        .child(table(
557            &["Kind", "Verdict", "What it says"],
558            evidence,
559            "No evidence was recorded.",
560        ))
561        .child(el("h4").class("mj-ui-panel-title").text("Paths it authors"))
562        .child(el("p").class("mj-ui-mono mj-ui-dim").text({
563            let paths = strings("authored_paths");
564            if paths.is_empty() {
565                "none".into()
566            } else {
567                paths.join(", ")
568            }
569        }))
570        .child(
571            el("h4")
572                .class("mj-ui-panel-title")
573                .text("Why it ranks where it does"),
574        )
575        .child(el("dl").class("mj-ui-facts").children(factors))
576}

584    /// Against a real registry every source of the first paint is a capability that
585    /// `capabilities.describe` knows, and none of them is the queue, which is the one this
586    /// screen must not wait for. Asked, not read: the test holds the rule the screen does.
587    #[test]
588    fn the_first_paint_asks_only_declared_sources_and_never_the_queue() {
589        let repo = SyntheticRepository::new(Shape::default()).expect("a synthetic repository");
590        let ctx = repo.context().expect("a context");
591        let declared = |id: &str| {
592            ctx.execute("capabilities.describe", serde_json::json!({ "id": id }))
593                .is_ok()
594        };
595        for s in FIRST_PAINT {
596            assert!(
597                declared(s.capability),
598                "{} is not a capability",
599                s.capability
600            );
601        }
602        assert!(!FIRST_PAINT.contains(&QUEUE));
603        assert!(declared(QUEUE.capability));
604        assert!(
605            !declared("no.such.capability"),
606            "the describe call must be able to say no"
607        );
608    }
