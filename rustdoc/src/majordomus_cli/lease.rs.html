1//! One shared server per checkout: the lease that decides who it is. The first
2//! `majordomus mcp` (or `serve`) to create `state/mcp/server.json` under the checkout's
3//! local half owns the server and publishes its URL there; every later process reads the
4//! file, checks that the server answers for this root, and attaches to it. A lease whose
5//! server does not answer is stale, and the next process takes it over; so is a file that
6//! is not a lease document, an empty one, or one whose owner never published a URL. A
7//! server that is slow to answer while its process is alive is busy, not stale: the
8//! election waits for it for [`BUSY_GRACE`] before it takes anything over. The
9//! file is the only thing the server writes anywhere, it lives under `.ai/local/` (never
10//! tracked, by the layer's contract), and it is removed when the server stops, or when
11//! the server dies of `SIGTERM`, `SIGINT` or `SIGHUP`.
12//!
13//! Per *checkout*, and this file said "per repository" until ADR 0044: a linked worktree
14//! is a checkout, so it has a manifest, a root, a lease and a server of its own, and on
15//! 2026-09-11 this repository had seven of them at once. What is repository-wide is the
16//! board, which `peers.list` gathers over these very leases; the election is unchanged.
17

39/// How long a lease without a URL may be (the owner is still binding) before it counts as
40/// abandoned.
41pub const BIND_GRACE: Duration = Duration::from_secs(15);
42
43/// How long the probe of a published URL waits.
44pub const PROBE_TIMEOUT: Duration = Duration::from_secs(2);
45
46/// How long a live server that does not answer its probe is waited for before its lease is
47/// taken over.
48///
49/// A probe that times out is not a dead server. On 2026-09-15 ten MCP clients attached at
50/// once to a server a shell entry had started; nine were bridged to it, and the tenth
51/// probed while the server was answering the other nine, heard nothing within
52/// [`PROBE_TIMEOUT`], took the lease over and started a second server for one checkout
53/// (test/cases/192). So a silent probe of a lease whose process is still alive is waited
54/// on, and asked again; only a server that stays silent for this long, or whose process is
55/// gone, loses its lease. The bound keeps the recovery from a hung server, and it sits
56/// well inside [`JOIN_TIMEOUT`] so that the take-over still happens within one election.
57pub const BUSY_GRACE: Duration = Duration::from_secs(10);
58
59/// How long a process keeps trying to acquire or join the lease before it gives up and
60/// says so: the bind grace with a margin for the probes. The caller then serves its
61/// client alone.
62pub const JOIN_TIMEOUT: Duration = Duration::from_secs(20);

64/// The timings above are the defaults. What a process actually judges a lease contest by is
65/// declared in `.ai/repo/policy.yaml`'s `server:` block and read once, here.
66///
67/// They were compiled constants until 2026-09-15: unchangeable without a rebuild, stated
68/// nowhere a reader would look, and invisible to every projection — while `probe_timeout` is
69/// the number that decides whether a live but slow owner keeps its lease or is taken over
70/// while it is still serving. That is a decision about how this repository is supervised, not
71/// an implementation detail, so it belongs in the model.
72///
73/// A `OnceLock`, because a process reads one policy: the repository is opened once and the
74/// answer must not change underneath a contest that is already being judged. Absent keys keep
75/// the constants, so a policy that cannot be read does not silently change behaviour.
76///
77/// ```
78/// use majordomus_cli::lease::{Timings, BIND_GRACE, BUSY_GRACE, JOIN_TIMEOUT, PROBE_TIMEOUT};
79/// // the defaults are the compiled constants, one for one
80/// let t = Timings::default();
81/// assert_eq!(t.bind_grace, BIND_GRACE);
82/// assert_eq!(t.probe_timeout, PROBE_TIMEOUT);
83/// assert_eq!(t.join_timeout, JOIN_TIMEOUT);
84/// assert_eq!(t.busy_grace, BUSY_GRACE);
85/// // a busy owner is given up on before the election that waits on it gives up itself
86/// assert!(t.busy_grace < t.join_timeout);
87/// ```
88#[derive(Debug, Clone, Copy, PartialEq, Eq)]
89pub struct Timings {
90    /// `server.bind_grace_seconds:`
91    pub bind_grace: Duration,
92    /// `server.probe_timeout_seconds:`
93    pub probe_timeout: Duration,
94    /// `server.join_timeout_seconds:`
95    pub join_timeout: Duration,
96    /// `server.busy_grace_seconds:`
97    pub busy_grace: Duration,
98}
99
100impl Default for Timings {
101    fn default() -> Self {
102        Self {
103            bind_grace: BIND_GRACE,
104            probe_timeout: PROBE_TIMEOUT,
105            join_timeout: JOIN_TIMEOUT,
106            busy_grace: BUSY_GRACE,
107        }
108    }
109}

113/// What this process judges a lease contest by. The declaration when one was read, the
114/// constants otherwise.
115///
116/// ```
117/// use majordomus_cli::lease::{timings, Timings};
118/// // this example's process declared nothing: it judges by the constants, and asking again
119/// // answers the same, because the first answer is the one the process keeps
120/// assert_eq!(timings(), Timings::default());
121/// assert_eq!(timings(), timings());
122/// ```
123pub fn timings() -> Timings {
124    *TIMINGS.get_or_init(Timings::default)
125}
126
127impl Timings {
128    /// What a declaration means, as a value rather than as a side effect.
129    ///
130    /// The mapping is the part that can be wrong; the `OnceLock` below is plumbing. Kept
131    /// separate so it can be tested directly — a test that went through the lock would
132    /// depend on which test ran first, since a process reads one policy by construction.
133    ///
134    /// ```
135    /// use majordomus_cli::lease::Timings;
136    /// use majordomus_cli::policy::ServerPolicy;
137    /// use std::time::Duration;
138    ///
139    /// // an empty declaration keeps every default
140    /// assert_eq!(Timings::from_policy(&ServerPolicy::default()), Timings::default());
141    ///
142    /// // and a declared value is the one used, rather than the constant
143    /// let declared = ServerPolicy {
144    ///     probe_timeout_seconds: Some(9),
145    ///     ..ServerPolicy::default()
146    /// };
147    /// let t = Timings::from_policy(&declared);
148    /// assert_eq!(t.probe_timeout, Duration::from_secs(9));
149    /// assert_ne!(t.probe_timeout, Timings::default().probe_timeout);
150    /// // the keys it says nothing about are untouched
151    /// assert_eq!(t.bind_grace, Timings::default().bind_grace);
152    /// ```
153    pub fn from_policy(policy: &crate::policy::ServerPolicy) -> Self {
154        let d = Self::default();
155        Self {
156            bind_grace: policy
157                .bind_grace_seconds
158                .map(Duration::from_secs)
159                .unwrap_or(d.bind_grace),
160            probe_timeout: policy
161                .probe_timeout_seconds
162                .map(Duration::from_secs)
163                .unwrap_or(d.probe_timeout),
164            join_timeout: policy
165                .join_timeout_seconds
166                .map(Duration::from_secs)
167                .unwrap_or(d.join_timeout),
168            busy_grace: policy
169                .busy_grace_seconds
170                .map(Duration::from_secs)
171                .unwrap_or(d.busy_grace),
172        }
173    }
174}

176/// Declare the timings for this process from a repository's policy. The first call decides;
177/// later ones are ignored, which is what makes [`timings`] answer the same thing all the way
178/// through one contest.
179///
180/// ```
181/// use majordomus_cli::lease::{declare_timings, timings};
182/// use majordomus_cli::policy::ServerPolicy;
183/// use std::time::Duration;
184/// // the first declaration is the one in force
185/// declare_timings(&ServerPolicy { busy_grace_seconds: Some(4), ..ServerPolicy::default() });
186/// assert_eq!(timings().busy_grace, Duration::from_secs(4));
187/// // a later one cannot move a judgement already being made
188/// declare_timings(&ServerPolicy { busy_grace_seconds: Some(9), ..ServerPolicy::default() });
189/// assert_eq!(timings().busy_grace, Duration::from_secs(4));
190/// ```
191pub fn declare_timings(policy: &crate::policy::ServerPolicy) {
192    let _ = TIMINGS.set(Timings::from_policy(policy));
193}

202/// The identity of the file an executable was started from: where it is, and the mtime
203/// and size of the file at that path. A server outlives its own binary — a rebuild replaces
204/// the file under a process that keeps serving the code it loaded hours ago — and nothing
205/// about the process itself says so. This is what makes that visible.
206///
207/// ```
208/// use majordomus_cli::lease::ExecutableIdentity;
209/// let recorded = ExecutableIdentity { path: "/opt/majordomus".into(), mtime: 1_700_000_000, size: 42 };
210/// let text = serde_json::to_string(&recorded).unwrap();
211/// assert_eq!(serde_json::from_str::<ExecutableIdentity>(&text).unwrap(), recorded);
212/// ```
213#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
214pub struct ExecutableIdentity {
215    /// The path the process was started from.
216    pub path: PathBuf,
217    /// The file's modification time, seconds since the epoch.
218    pub mtime: u64,
219    /// The file's size in bytes.
220    pub size: u64,
221}
222
223impl ExecutableIdentity {
224    /// Has the file at the recorded path been replaced or removed since it was recorded?
225    /// The reason, for a reader; `None` when the same file is still there.
226    ///
227    /// ```
228    /// use majordomus_cli::lease::ExecutableIdentity;
229    /// let gone = ExecutableIdentity { path: "/nowhere/majordomus".into(), mtime: 1, size: 1 };
230    /// assert!(gone.replaced().unwrap().contains("no longer on disk"));
231    /// ```
232    pub fn replaced(&self) -> Option<String> {
233        let Ok(meta) = fs::metadata(&self.path) else {
234            return Some(format!(
235                "the executable it was started from, {}, is no longer on disk",
236                self.path.display()
237            ));
238        };
239        let mtime = meta
240            .modified()
241            .ok()
242            .and_then(|m| m.duration_since(SystemTime::UNIX_EPOCH).ok())
243            .map(|d| d.as_secs())
244            .unwrap_or(0);
245        if mtime == self.mtime && meta.len() == self.size {
246            return None;
247        }
248        Some(format!(
249            "the executable it was started from, {}, has been replaced since (it is serving \
250             code that is no longer on disk)",
251            self.path.display()
252        ))
253    }
254}

256/// The lease file, typed: what the process serving a checkout wrote down about itself.
257/// Every reader of the file — the election, a command asking which server is running, the
258/// environment snapshot, the server's own status — reads it through [`LeaseFile::read`],
259/// so that a field added here is added once and a file that is not a lease is refused in
260/// one place. A key this version does not know is ignored, so a lease written by a newer
261/// executable still reads; a key it needs that an older lease lacks takes its default.
262///
263/// ```
264/// use majordomus_cli::lease::{LeaseDocument, SCHEMA};
265/// // a lease an older server wrote: no pid, no version, a key this version has never heard of
266/// let older = format!(r#"{{"schema":"{SCHEMA}","token":"t","root":"/r","url":"http://127.0.0.1:1","later":1}}"#);
267/// let doc: LeaseDocument = serde_json::from_str(&older).unwrap();
268/// assert_eq!(doc.pid, 0);
269/// assert!(doc.version.is_none(), "older than any executable that reads this");
270/// assert_eq!(doc.url.as_deref(), Some("http://127.0.0.1:1"));
271/// // and what this version writes reads back as itself
272/// let text = serde_json::to_string(&doc).unwrap();
273/// assert_eq!(serde_json::from_str::<LeaseDocument>(&text).unwrap(), doc);
274/// ```
275#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
276pub struct LeaseDocument {
277    /// [`SCHEMA`].
278    pub schema: String,
279    /// The process id of the server. A live pid is not a live server: only an answering
280    /// probe says that. What a live pid does decide is patience — a probe that goes
281    /// unanswered by a live owner is asked again within [`BUSY_GRACE`] before the lease
282    /// counts as stale (the election's busy wait, [`probe_patiently`]); a dead or zero pid
283    /// gets no patience at all.
284    #[serde(default)]
285    pub pid: u32,
286    /// What makes the file this process's: only the process holding this token removes it.
287    #[serde(default)]
288    pub token: String,
289    /// The checkout root the server serves.
290    #[serde(default)]
291    pub root: PathBuf,
292    /// The address, once bound; `null` while the owner is still binding.
293    pub url: Option<String>,
294    /// When the server took the lease, RFC 3339.
295    #[serde(default)]
296    pub started_at: String,
297    /// The executable the server was started from, when it could be located.
298    #[serde(default)]
299    pub executable: Option<ExecutableIdentity>,
300    /// The executable's version. Absent in a lease written before this field existed,
301    /// which is itself a fact: that server is older than any executable that reads this.
302    #[serde(default, skip_serializing_if = "Option::is_none")]
303    pub version: Option<String>,
304}

306/// What a lease file holds, read once. Every state of the file is a variant, so that a
307/// reader decides what to do about each rather than learning it from an error.
308///
309/// ```
310/// use majordomus_cli::lease::LeaseFile;
311/// let dir = tempfile::tempdir().unwrap();
312/// let path = dir.path().join("server.json");
313/// assert_eq!(LeaseFile::read(&path), LeaseFile::Absent);
314/// assert!(LeaseFile::read(&path).document().is_none(), "no document to hand back");
315/// ```
316#[derive(Debug, Clone, PartialEq, Eq)]
317pub enum LeaseFile {
318    /// There is no file.
319    Absent,
320    /// A file with nothing in it: its owner created it and has not written it yet.
321    Empty,
322    /// A file that is not a lease document; the reason, for the log.
323    Corrupt(String),
324    /// A lease document.
325    Document(LeaseDocument),
326}
327
328impl LeaseFile {
329    /// Read and type the file at `path`. Never fails: every state of the file is one of the
330    /// variants, and what to do about each is the reader's decision.
331    ///
332    /// ```
333    /// use majordomus_cli::lease::{LeaseFile, SCHEMA};
334    /// let dir = tempfile::tempdir().unwrap();
335    /// let path = dir.path().join("server.json");
336    /// assert_eq!(LeaseFile::read(&path), LeaseFile::Absent);
337    /// std::fs::write(&path, "").unwrap();
338    /// assert_eq!(LeaseFile::read(&path), LeaseFile::Empty);
339    /// std::fs::write(&path, "{not json").unwrap();
340    /// assert!(matches!(LeaseFile::read(&path), LeaseFile::Corrupt(r) if r.starts_with("not JSON")));
341    /// std::fs::write(&path, r#"{"schema":"other/v1"}"#).unwrap();
342    /// assert!(matches!(LeaseFile::read(&path), LeaseFile::Corrupt(r) if r.contains(SCHEMA)));
343    /// std::fs::write(&path, format!(r#"{{"schema":"{SCHEMA}","token":"x","root":"/r","url":null}}"#)).unwrap();
344    /// match LeaseFile::read(&path) {
345    ///     LeaseFile::Document(d) => { assert_eq!(d.token, "x"); assert!(d.url.is_none()); assert!(d.version.is_none()); }
346    ///     other => panic!("{other:?}"),
347    /// }
348    /// ```
349    pub fn read(path: &Path) -> LeaseFile {
350        let text = match fs::read_to_string(path) {
351            Ok(text) => text,
352            Err(e) if e.kind() == std::io::ErrorKind::NotFound => return LeaseFile::Absent,
353            Err(e) => return LeaseFile::Corrupt(format!("cannot be read ({e})")),
354        };
355        if text.trim().is_empty() {
356            return LeaseFile::Empty;
357        }
358        let value: Value = match serde_json::from_str(&text) {
359            Ok(value) => value,
360            Err(e) => return LeaseFile::Corrupt(format!("not JSON ({e})")),
361        };
362        if !value.is_object() || value["schema"] != SCHEMA {
363            return LeaseFile::Corrupt(format!("not a {SCHEMA} document"));
364        }
365        match serde_json::from_value::<LeaseDocument>(value) {
366            Ok(document) => LeaseFile::Document(document),
367            Err(e) => LeaseFile::Corrupt(format!("not a {SCHEMA} document ({e})")),
368        }
369    }
370
371    /// The document, when the file holds one; `None` for an absent, empty or corrupt file,
372    /// for a reader that has no different answer for those three.
373    ///
374    /// ```
375    /// use majordomus_cli::lease::{LeaseDocument, LeaseFile, SCHEMA};
376    /// let doc = LeaseDocument {
377    ///     schema: SCHEMA.into(), pid: 7, token: "t".into(), root: "/r".into(), url: None,
378    ///     started_at: "2026-09-10T00:00:00Z".into(), executable: None, version: None,
379    /// };
380    /// assert_eq!(LeaseFile::Document(doc.clone()).document(), Some(&doc));
381    /// assert_eq!(LeaseFile::Corrupt("not JSON".into()).document(), None);
382    /// ```
383    pub fn document(&self) -> Option<&LeaseDocument> {
384        match self {
385            LeaseFile::Document(d) => Some(d),
386            _ => None,
387        }
388    }
389}

400/// The lease this process holds and has published: `None` in a process that serves nothing.
401///
402/// ```
403/// // a process that has published no lease holds none

406pub fn held() -> Option<LeaseDocument> {

408}

410/// Is `doc` the lease this very process holds, published and not lost?
411///
412/// A server asked where its own checkout stands must answer from memory. Probing the
413/// address its lease names is a request to itself, served by the same small pool of HTTP
414/// workers the question arrived on: with every worker answering such a question at once,
415/// none is left to answer the probes, each waits out [`PROBE_TIMEOUT`], the server calls
416/// itself stale — and a client electing in that window times out on the same queue and
417/// takes a live lease over. Measured on 2026-09-15: ten MCP clients calling
418/// `majordomus_peers` at once, and one of them started a second server of the checkout.
419///
420/// ```
421/// use majordomus_cli::lease::{is_own, LeaseDocument};
422/// let doc: LeaseDocument = serde_json::from_str(
423///     r#"{"schema":"lease/v1","pid":1,"token":"t","url":"http://127.0.0.1:1"}"#,
424/// )
425/// .unwrap();
426/// // a process that holds no lease owns no document
427/// assert!(!is_own(&doc));
428/// ```
429pub fn is_own(doc: &LeaseDocument) -> bool {
430    !was_lost()
431        && held().is_some_and(|h| h.token == doc.token && h.pid == doc.pid && h.url == doc.url)
432}

434/// The lease this process holds. Dropping it removes the file (when the file is still
435/// this process's), so a failed start never leaves a stale lease behind.
436#[derive(Debug)]
437pub struct Lease {
438    path: PathBuf,
439    token: String,
440    root: PathBuf,
441    released: bool,
442    /// When this process took the lease, RFC 3339. Fixed here rather than at publish time
443    /// because it is the identity of this server's generation.
444    started_at: String,
445    /// `true` from the election until [`Lease::publish`]: the window in which
446    /// [`Lease::keep_alive`] rewrites the file so that it never looks abandoned, and the
447    /// lock under which it and `publish` write, so that a touch can never land on top of
448    /// the URL.
449    binding: Arc<Mutex<bool>>,
450}

486/// Which server is serving this repository right now, if any.
487///
488/// Read-only, unlike [`elect`]: it takes no lease, creates no file and waits for nobody, so
489/// a command that only wants to *ask* the running server something cannot accidentally
490/// become it. `None` means no lease, no URL in it, or a lease whose server does not answer
491/// for this root.
492pub fn serving(repo: &Repository) -> Option<String> {
493    let url = LeaseFile::read(&lease_path(repo)).document()?.url.clone()?;
494    probe(&url, repo.root()).then_some(url)
495}

497/// Decide whether this process serves the repository or attaches to the process that does.
498///
499/// An existing file is read on every attempt and classified: the lease of a live server
500/// (it answers for this root) is attached to; a stale one (its server does not answer),
501/// a corrupt one (not a lease document), an empty one, or an abandoned one (no URL after
502/// [`BIND_GRACE`]) is taken over; a fresh lease without a URL means its owner is still
503/// binding, and this process waits for it. Nothing a client leaves behind can lock the
504/// others out; when the file can neither be created nor removed within [`JOIN_TIMEOUT`],
505/// the error names the path and the caller serves its client alone.
506pub fn elect(repo: &Repository) -> Result<Role> {
507    let path = lease_path(repo);
508    let root = repo.root().to_path_buf();
509    if let Some(dir) = path.parent() {
510        fs::create_dir_all(dir).map_err(|e| Error::io(dir, e))?;
511    }
512    let waited_since = Instant::now();
513    // when a live server first failed to answer this election; cleared whenever it answers,
514    // or the file stops naming a busy server
515    let mut busy_since: Option<Instant> = None;
516    loop {
517        match fs::OpenOptions::new()
518            .write(true)
519            .create_new(true)
520            .open(&path)
521        {
522            Ok(mut file) => {
523                let token = format!(
524                    "{}-{:x}",
525                    std::process::id(),
526                    SystemTime::now()
527                        .duration_since(SystemTime::UNIX_EPOCH)
528                        .map(|d| d.as_nanos())
529                        .unwrap_or(0)
530                );
531                let lease = Lease {
532                    path: path.clone(),
533                    token,
534                    root,
535                    released: false,
536                    started_at: crate::peers::rfc3339(SystemTime::now()),
537                    binding: Arc::new(Mutex::new(true)),
538                };
539                let text =
540                    serde_json::to_string(&lease.document(None)).map_err(|e| Error::Lease {
541                        reason: format!("cannot render the lease: {e}"),
542                    })?;
543                file.write_all(text.as_bytes())
544                    .map_err(|e| Error::io(&path, e))?;
545                signals::hold(&path);
546                return Ok(Role::Server(lease));
547            }
548            Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
549                let (found, seen) = inspect(&path, &root);
550                if !matches!(found, Found::Busy(_)) {
551                    busy_since = None;
552                }
553                match found {
554                    Found::Live(url) => return Ok(Role::Peer { url }),
555                    Found::Busy(url) => {
556                        let since = *busy_since.get_or_insert_with(Instant::now);
557                        if since.elapsed() < timings().busy_grace {
558                            std::thread::sleep(Duration::from_millis(200));
559                        } else {
560                            tracing::warn!(
561                                lease = %path.display(),
562                                "the server at {url} is alive and has not answered for {} seconds; taking it over",
563                                timings().busy_grace.as_secs()
564                            );
565                            take_over(&path, &seen)?;
566                            busy_since = None;
567                        }
568                    }
569                    Found::Stale(reason) => {
570                        tracing::warn!(lease = %path.display(), "{reason}; taking it over");
571                        take_over(&path, &seen)?;
572                    }
573                    // an owner that is still binding keeps its file young (`keep_alive`), so
574                    // a file that is old enough to be abandoned is classified so by `inspect`
575                    // and this arm only ever waits; the whole attempt is bounded below
576                    Found::Binding => std::thread::sleep(Duration::from_millis(100)),
577                }
578            }
579            Err(e) => return Err(Error::io(&path, e)),
580        }
581        if waited_since.elapsed() > timings().join_timeout {
582            return Err(Error::Lease {
583                reason: format!(
584                    "could not acquire or join the lease at {} within {} seconds",
585                    path.display(),
586                    timings().join_timeout.as_secs()
587                ),
588            });
589        }
590    }
591}

593/// What a lease file that already exists says.
594enum Found {
595    /// A server answers at this URL for this root.
596    Live(String),
597    /// The server at this URL did not answer in time, and the process that holds the lease
598    /// is alive: busy rather than gone, and waited on for [`BUSY_GRACE`].
599    Busy(String),
600    /// The file is not a usable lease; the reason says why, for the log.
601    Stale(String),
602    /// A lease without a URL, young enough that its owner may still be binding.
603    Binding,
604}

606/// What the running executable is, for the lease to record: the path it was started from
607/// and the identity of the file at that path. A server outlives its own binary — a rebuild
608/// replaces the file under a process that keeps serving the code it loaded hours ago — and
609/// nothing about the process itself says so. This is what makes that visible.
610///
611/// `None` when the executable cannot be located or stat'ed; the lease then carries no
612/// claim, and the election's `superseded` check makes none either.
613///
614/// ```
615/// use majordomus_cli::lease::executable_identity;
616/// // the process running this example can locate itself, and the file it names is there
617/// let mine = executable_identity().expect("a test binary knows where it is");
618/// assert!(mine.path.is_file());
619/// assert!(mine.replaced().is_none(), "the file that is running has not been replaced");
620/// ```
621pub fn executable_identity() -> Option<ExecutableIdentity> {
622    let path = std::env::current_exe().ok()?;
623    let meta = fs::metadata(&path).ok()?;
624    let mtime = meta
625        .modified()
626        .ok()?
627        .duration_since(SystemTime::UNIX_EPOCH)
628        .ok()?
629        .as_secs();
630    Some(ExecutableIdentity {
631        path,
632        mtime,
633        size: meta.len(),
634    })
635}
636
637/// Has the executable behind a lease been replaced since that server started?
638///
639/// Only a lease naming *this process's own* executable path can answer: same path, a file
640/// that is now a different file, and the server on the other end is provably running code
641/// that no longer exists on disk. A lease naming some other path — a release install next
642/// to a debug build — makes no claim either way and is left alone, so two legitimate
643/// binaries never fight over the lease.
644fn superseded(doc: &LeaseDocument) -> Option<String> {
645    let recorded = doc.executable.as_ref()?;
646    let mine = executable_identity()?;
647    if recorded.path != mine.path {
648        return None;
649    }
650    if recorded.mtime == mine.mtime && recorded.size == mine.size {
651        return None;
652    }
653    Some(format!(
654        "superseded lease: the server was started from {} and that file has been replaced since \
655         (it is serving code that is no longer on disk)",
656        mine.path.display()
657    ))
658}

660/// Read and classify an existing lease file. What was read comes back beside the verdict,
661/// so that a take-over can insist on removing the file it judged and not one that arrived
662/// in the meantime.
663fn inspect(path: &Path, root: &Path) -> (Found, LeaseFile) {
664    let age = file_age(path);
665    let seen = LeaseFile::read(path);
666    let doc = match &seen {
667        // gone between the failed create and this read: the next attempt creates it
668        LeaseFile::Absent => return (Found::Binding, seen),
669        LeaseFile::Empty if age > timings().bind_grace => {
670            return (
671                Found::Stale("empty lease: its owner never wrote it".into()),
672                seen,
673            )
674        }
675        LeaseFile::Empty => return (Found::Binding, seen),
676        LeaseFile::Corrupt(reason) => {
677            return (Found::Stale(format!("corrupt lease: {reason}")), seen)
678        }
679        LeaseFile::Document(doc) => doc.clone(),
680    };
681    // before asking whether it answers: a server that answers from a binary that has been
682    // replaced answers with yesterday's code, which is the harder failure to see
683    if let Some(reason) = superseded(&doc) {
684        return (Found::Stale(reason), seen);
685    }
686    let found = match doc.url.as_deref() {
687        Some(url) => match ask(url, root, timings().probe_timeout) {
688            Answer::Ours(_) => Found::Live(url.to_string()),
689            // a live owner that is silent is busy and waited on — unless the process that
690            // started this one already waited out that patience on this very lease
691            Answer::Silent if alive(doc.pid) && !judged_stale(&doc) => Found::Busy(url.to_string()),
692            Answer::Silent | Answer::NotOurs => Found::Stale(format!(
693                "stale lease: the server it names at {url} does not answer for this repository"
694            )),
695        },
696        None if age > timings().bind_grace => {
697            Found::Stale("abandoned lease: its owner never published a URL".into())
698        }
699        None => Found::Binding,
700    };
701    (found, seen)
702}

704/// How long ago the file was last written; zero when that cannot be read, so that a file
705/// whose age is unknown is never taken for an old one.
706///
707/// ```
708/// use majordomus_cli::lease::file_age;
709/// use std::time::Duration;
710/// let dir = tempfile::tempdir().unwrap();
711/// let path = dir.path().join("server.json");
712/// assert_eq!(file_age(&path), Duration::ZERO, "no file: no age");
713/// std::fs::write(&path, "").unwrap();
714/// assert!(file_age(&path) < Duration::from_secs(60), "just written");
715/// ```
716pub fn file_age(path: &Path) -> Duration {
717    fs::metadata(path)
718        .and_then(|m| m.modified())
719        .ok()
720        .and_then(|m| m.elapsed().ok())
721        .unwrap_or(Duration::ZERO)
722}

724/// Remove a lease that cannot be used, so that the next attempt creates a fresh one — but
725/// only the file that was judged: when another process has replaced it since, the file on
726/// disk is somebody's fresh lease, and the next round of the election reads that one.
727fn take_over(path: &Path, seen: &LeaseFile) -> Result<()> {
728    if LeaseFile::read(path) != *seen {
729        tracing::debug!(lease = %path.display(), "the lease changed under the take-over; reading it again");
730        return Ok(());
731    }
732    match fs::remove_file(path) {
733        Ok(()) => Ok(()),
734        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
735        Err(e) => Err(Error::Lease {
736            reason: format!(
737                "cannot remove the unusable lease at {}: {e}",
738                path.display()
739            ),
740        }),
741    }
742}

744/// Set by [`lost`], never cleared: a lease this process was given and no longer has.
745///
746/// Distinct from `held().is_none()`, which is also true of a process that never took a
747/// lease at all — a test, a one-off command, a bridged client. Only a process that *had*
748/// the lease and lost it must change how it behaves, so only that one is recorded here.
749static LOST: AtomicBool = AtomicBool::new(false);
750
751/// The key the index answers [`was_lost`]'s converse under: whether the server answering
752/// still holds the lease of the checkout it serves. Named once, read by [`probe`] and
753/// written by the index route, so the two cannot drift apart.
754pub const LEASEHOLDER_KEY: &str = "leaseholder";
755
756/// This process's lease was taken over by another process: from now on a signal must not
757/// unlink the file, which is somebody else's, and this process stops claiming to be the
758/// checkout's server — [`held`] answers `None` and [`was_lost`] answers `true` from here
759/// on. It serves the peers it has and ends with them; what it must not do is take on new
760/// ones, or answer a stranger's probe as though it were still the one.
761///
762/// ```
763/// // a process that holds no lease has nothing to lose; saying so twice changes nothing
764/// majordomus_cli::lease::lost();
765/// majordomus_cli::lease::lost();
766/// assert!(majordomus_cli::lease::held().is_none());
767/// assert!(majordomus_cli::lease::was_lost(), "it was told the lease is gone");
768/// ```
769pub fn lost() {
770    LOST.store(true, Ordering::SeqCst);
771    if let Ok(mut published) = published().lock() {
772        *published = None;
773    }
774    signals::release();
775}

777/// Did this process hold the checkout's lease and lose it?
778///
779/// The one question a server asks before taking on work that outlives the request. A
780/// process that never held a lease answers `false` and goes on serving whoever asked it —
781/// a test, a bridged client, a one-off command. One that was superseded answers `true`,
782/// and from then on refuses to become anybody's server again. It never resets: a lease
783/// that was taken over is not given back.
784///
785/// ```
786/// // nothing has taken a lease away from the process running this example; `lost()` in
787/// // the doctest above runs in a process of its own
788/// assert!(!majordomus_cli::lease::was_lost());
789/// ```
790pub fn was_lost() -> bool {
791    LOST.load(Ordering::SeqCst)
792}

794/// Does a Majordomus server answer at `url` for the repository at `root`, *as* that
795/// repository's server?
796///
797/// Three questions, and the third is the one a remembered address needs: it is a
798/// Majordomus server, it serves this root, and it still holds this checkout's lease.
799/// Without the third, a process whose lease was taken over answers every probe exactly as
800/// the current server does — same name, same `repository_id`, its own peer board, its own
801/// generation of the layer — and a client holding an address from before the takeover is
802/// served plausible answers by a server nobody else is talking to. That was measured on
803/// 2026-09-10: two servers of one checkout, one lease, and the older one indistinguishable
804/// from the current one on this endpoint.
805///
806/// A server too old to answer the question is accepted. It cannot be told from a current
807/// one here, and refusing it would be the worse failure: a live server taken for dead is
808/// taken over, which is how one checkout comes to have two.
809pub fn probe(url: &str, root: &Path) -> bool {
810    matches!(ask(url, root, timings().probe_timeout), Answer::Ours(_))
811}
812
813/// [`probe`], keeping what the server answered: its index document when it answers as the
814/// leaseholder of this checkout, `None` otherwise. One decision, so a caller that also wants
815/// the surfaces the server lists makes neither a second request nor a second judgement.
816///
817/// ```
818/// use majordomus_cli::lease::probe_reply;
819/// use std::time::Duration;
820/// // nothing listens on port 1 of the loopback address
821/// let root = std::path::Path::new("/r");
822/// assert!(probe_reply("http://127.0.0.1:1", root, Duration::from_millis(50)).is_none());
823/// ```
824pub fn probe_reply(url: &str, root: &Path, timeout: Duration) -> Option<Value> {
825    match ask(url, root, timeout) {
826        Answer::Ours(index) => Some(index),
827        Answer::NotOurs | Answer::Silent => None,
828    }
829}

831/// What a probe of a published URL heard.
832enum Answer {
833    /// A Majordomus server that serves this root and holds its lease, with the index
834    /// document it answered.
835    Ours(Value),
836    /// Something answered that is not this checkout's server, or nothing is listening.
837    NotOurs,
838    /// The connection was made and no answer came within the timeout: a server that is
839    /// busy looks exactly like this, and so does a hung one.
840    Silent,
841}
842
843/// Ask a published URL the probe's three questions within `timeout`, telling a server that
844/// said no apart from one that said nothing in time.
845fn ask(url: &str, root: &Path, timeout: Duration) -> Answer {
846    match bridge::request(url, "GET", "/", &[], None, timeout) {
847        Ok(reply) if reply.status == 200 => {
848            let v: Value = serde_json::from_str(&reply.body).unwrap_or(Value::Null);
849            // the identity and not the path: the index names the repository it serves
850            // without telling every caller where the checkout sits
851            let ours = v["name"] == "majordomus"
852                && v["repository_id"].as_str() == Some(crate::repository::identity(root).as_str())
853                && v[LEASEHOLDER_KEY] != Value::Bool(false);
854            if ours {
855                Answer::Ours(v)
856            } else {
857                Answer::NotOurs
858            }
859        }
860        Ok(_) => Answer::NotOurs,
861        Err(e)
862            if matches!(
863                e.kind(),
864                std::io::ErrorKind::TimedOut | std::io::ErrorKind::WouldBlock
865            ) =>
866        {
867            Answer::Silent
868        }
869        Err(_) => Answer::NotOurs,
870    }
871}

873/// Is the process with this id alive on this machine? A lease is a file of this checkout,
874/// so its pid is a pid of this host. `kill(pid, 0)` delivers nothing and asks only whether
875/// the process exists; a process owned by somebody else answers `EPERM`, which is alive.
876/// Whether a process with this pid exists (signal 0: checked, nothing sent).
877pub(crate) fn alive(pid: u32) -> bool {
878    if pid == 0 {
879        return false;
880    }
881    let Ok(pid) = libc::pid_t::try_from(pid) else {
882        return false;
883    };
884    // SAFETY: signal 0 performs the existence and permission check and sends nothing.
885    let rc = unsafe { libc::kill(pid, 0) };
886    rc == 0 || std::io::Error::last_os_error().raw_os_error() == Some(libc::EPERM)
887}

889/// [`probe`], with patience for an owner that is alive but slow.
890///
891/// A probe that times out says nothing about whether the server is gone: a server loading
892/// a large layer, or one on a machine under load (an instrumented test run, a loaded CI
893/// runner), answers late. Judging it gone from one silent probe makes `serve ensure` start
894/// a second server beside it and report the old one's address as though it had started it.
895/// So when the probe fails and the lease's `pid` is a live process on this machine, the
896/// probe is repeated every second for up to `patience` — the caller's share of
897/// [`Timings::busy_grace`], the same wait the election gives a busy owner. The answer is
898/// `false` only when every attempt fails.
899///
900/// The patience is bounded on purpose. A live pid can be a wedged server that never answers
901/// again, and a probe that waited on it forever would make every client hang on a ghost. A
902/// dead pid, or pid 0, gets no patience at all, so a killed server is recovered at once,
903/// and a `patience` of zero is a single probe.
904///
905/// ```
906/// use majordomus_cli::lease::probe_patiently;
907/// use std::path::Path;
908/// use std::time::{Duration, Instant};
909/// // nothing listens on port 1 and pid 0 names no process: one refused probe, no patience
910/// let t0 = Instant::now();
911/// let patience = Duration::from_secs(10);
912/// assert!(!probe_patiently("http://127.0.0.1:1", Path::new("/nowhere"), 0, patience));
913/// assert!(t0.elapsed() < Duration::from_secs(5));
914/// ```
915pub fn probe_patiently(url: &str, root: &Path, pid: u32, patience: Duration) -> bool {
916    if probe(url, root) {
917        return true;
918    }
919    let deadline = Instant::now() + patience;
920    while alive(pid) && Instant::now() < deadline {
921        std::thread::sleep(Duration::from_secs(1));
922        if probe(url, root) {
923            return true;
924        }
925    }
926    false
927}

929/// Was this lease already waited on, and judged stale, by the process that started this
930/// one? Its token is then in [`JUDGED_STALE_ENV`], and the election takes exactly that
931/// lease over at once instead of spending the same patience on it a second time: waiting
932/// twice would only double the delay before a wedged owner is replaced. Any other lease —
933/// a new owner, or one that changed in between — still gets the full wait.
934fn judged_stale(doc: &LeaseDocument) -> bool {
935    !doc.token.is_empty() && std::env::var(JUDGED_STALE_ENV).is_ok_and(|token| token == doc.token)
936}

938impl Lease {
939    /// The file.
940    pub fn path(&self) -> &Path {
941        &self.path
942    }
943
944    /// The repository root the lease is for.
945    pub fn root(&self) -> &Path {
946        &self.root
947    }
948
949    /// When this process took the lease, RFC 3339: the identity of this server's
950    /// generation, the same value the lease file carries as `started_at`.
951    pub fn started_at(&self) -> &str {
952        &self.started_at
953    }
954
955    /// What makes the file this process's. For the server's own reader, which checks from
956    /// another thread whether the lease is still its own.
957    pub(crate) fn token(&self) -> &str {
958        &self.token
959    }
960
961    /// Keep the lease young while the layer loads. A peer that finds a lease without a URL
962    /// waits for it only as long as the file is younger than [`BIND_GRACE`]; a cold start
963    /// that takes longer than that — a large layer, a machine under load — would otherwise
964    /// be taken for an abandoned one and taken over while its owner is still binding. So
965    /// until [`Lease::publish`], a thread rewrites the same document every third of the
966    /// grace, under the lock `publish` also takes, so that a touch can never land after the
967    /// URL. It writes only while the file is still this process's: a lease that was taken
968    /// over anyway is somebody else's to write.
969    ///
970    /// ```
971    /// use majordomus_cli::lease::{elect, held, LeaseFile, Role};
972    /// use majordomus_cli::Repository;
973    /// let dir = tempfile::tempdir().unwrap();
974    /// std::fs::create_dir_all(dir.path().join(".ai/repo")).unwrap();
975    /// std::fs::write(dir.path().join(".ai/manifest.yaml"),
976    ///     "schema: ai-repository/v1\nrepo:\n  path: repo\nlocal:\n  path: local\n  tracked: false\n  implicit_context: false\nsections:\n  policy: repo/policy.yaml\n").unwrap();
977    /// let repo = Repository::discover(dir.path()).unwrap();
978    /// let Role::Server(lease) = elect(&repo).unwrap() else { panic!("nobody else serves a fresh directory") };
979    /// lease.keep_alive();                         // returns at once; the touching runs beside the load
980    /// assert!(held().is_none(), "nothing is published yet");
981    /// lease.publish("http://127.0.0.1:1").unwrap();  // stops the touching, under the same lock
982    /// assert_eq!(held().unwrap().url.as_deref(), Some("http://127.0.0.1:1"));
983    /// assert!(matches!(LeaseFile::read(lease.path()), LeaseFile::Document(d) if d.url.is_some()));
984    /// lease.release();
985    /// assert_eq!(LeaseFile::read(&dir.path().join(".ai/local/state/mcp/server.json")), LeaseFile::Absent);
986    /// ```
987    pub fn keep_alive(&self) {
988        let path = self.path.clone();
989        let token = self.token.clone();
990        let binding = Arc::clone(&self.binding);
991        let document = serde_json::to_string(&self.document(None)).unwrap_or_default();
992        let tick = timings().bind_grace / 3;
993        let _ = std::thread::Builder::new()
994            .name("majordomus-lease-keep-alive".into())
995            .spawn(move || loop {
996                std::thread::sleep(tick);
997                let Ok(guard) = binding.lock() else { return };
998                if !*guard {
999                    return;
1000                }
1001                let mine = LeaseFile::read(&path)
1002                    .document()
1003                    .is_some_and(|d| d.token == token);
1004                if !mine {
1005                    return;
1006                }
1007                let tmp = path.with_extension("json.tmp");
1008                if fs::write(&tmp, &document).is_ok() {
1009                    let _ = fs::rename(&tmp, &path);
1010                }
1011            });
1012    }
1013
1014    fn document(&self, url: Option<&str>) -> LeaseDocument {
1015        LeaseDocument {
1016            schema: SCHEMA.into(),
1017            pid: std::process::id(),
1018            token: self.token.clone(),
1019            root: self.root.clone(),
1020            url: url.map(str::to_string),
1021            started_at: self.started_at.clone(),
1022            executable: executable_identity(),
1023            version: Some(crate::VERSION.into()),
1024        }
1025    }
1026
1027    /// Record the URL the server is listening on, atomically, so that a peer reading the
1028    /// file sees either no URL or the whole one. Refused when the file is no longer this
1029    /// process's: a lease taken over while its owner was binding belongs to whoever took it,
1030    /// and writing over it would leave two servers claiming one address.
1031    pub fn publish(&self, url: &str) -> Result<()> {
1032        let mut binding = self.binding.lock().map_err(|_| Error::Lease {
1033            reason: "the lease's binding lock is poisoned".into(),
1034        })?;
1035        *binding = false;
1036        if !self.is_mine() {
1037            return Err(Error::Lease {
1038                reason: format!(
1039                    "the lease at {} was taken over while this server was starting; another process serves this checkout",
1040                    self.path.display()
1041                ),
1042            });
1043        }
1044        let tmp = self.path.with_extension("json.tmp");
1045        let document = self.document(Some(url));
1046        let text = serde_json::to_string(&document).map_err(|e| Error::Lease {
1047            reason: format!("cannot render the lease: {e}"),
1048        })?;
1049        fs::write(&tmp, text).map_err(|e| Error::io(&tmp, e))?;
1050        fs::rename(&tmp, &self.path).map_err(|e| Error::io(&self.path, e))?;
1051        if let Ok(mut slot) = published().lock() {
1052            *slot = Some(document);
1053        }
1054        Ok(())
1055    }
1056
1057    /// Is the file on disk still this process's lease?
1058    fn is_mine(&self) -> bool {
1059        LeaseFile::read(&self.path)
1060            .document()
1061            .is_some_and(|d| d.token == self.token)
1062    }
1063
1064    /// Remove the lease: the server has stopped.
1065    pub fn release(mut self) {
1066        self.release_now();
1067    }
1068
1069    fn release_now(&mut self) {
1070        if let Ok(mut binding) = self.binding.lock() {
1071            *binding = false;
1072        }
1073        if !self.released && self.is_mine() {
1074            let _ = fs::remove_file(&self.path);
1075        }
1076        self.released = true;
1077        if let Ok(mut slot) = published().lock() {
1078            if slot.as_ref().is_some_and(|d| d.token == self.token) {
1079                *slot = None;
1080            }
1081        }
1082        signals::release();
1083    }
1084}

1092/// The lease is removed when the process dies of `SIGTERM`, `SIGINT` or `SIGHUP`: a client
1093/// killing its server, a person pressing Ctrl-C, a terminal closing. The handler does only
1094/// what is safe inside a signal handler: `unlink` the path recorded when the lease was
1095/// taken, restore the default disposition, and raise the signal again so that the exit
1096/// status still says which signal it was. The path is recorded once per process (a
1097/// repository's lease never moves) and is never freed, because a handler may be reading
1098/// it. A `kill -9` cannot be caught: the next process finds the stale lease and takes it
1099/// over.
1100#[cfg(unix)]
1101mod signals {
1102    use std::ffi::CString;
1103    use std::os::unix::ffi::OsStrExt;
1104    use std::path::Path;
1105    use std::sync::atomic::{AtomicBool, AtomicPtr, Ordering};
1106    use std::sync::Once;
1107
1108    static HELD: AtomicBool = AtomicBool::new(false);
1109    static PATH: AtomicPtr<libc::c_char> = AtomicPtr::new(std::ptr::null_mut());
1110    static INSTALL: Once = Once::new();
1111
1112    /// This process now holds the lease at `path`; install the handlers the first time.
1113    pub fn hold(path: &Path) {
1114        let Ok(c) = CString::new(path.as_os_str().as_bytes()) else {
1115            return;
1116        };
1117        let raw = c.into_raw();
1118        if PATH
1119            .compare_exchange(
1120                std::ptr::null_mut(),
1121                raw,
1122                Ordering::SeqCst,
1123                Ordering::SeqCst,
1124            )
1125            .is_err()
1126        {
1127            // the path is already recorded (the same repository, taken over again): this
1128            // copy is not needed, and the recorded one stays reachable for the handler
1129            drop(unsafe { CString::from_raw(raw) });
1130        }
1131        HELD.store(true, Ordering::SeqCst);
1132        INSTALL.call_once(|| {
1133            let handler = on_signal as extern "C" fn(libc::c_int) as libc::sighandler_t;
1134            for signal in [libc::SIGTERM, libc::SIGINT, libc::SIGHUP] {
1135                // SAFETY: installing a handler that only calls async-signal-safe functions
1136                unsafe { libc::signal(signal, handler) };
1137            }
1138        });
1139    }
1140
1141    /// The lease is released (or was never this process's any more): stop removing it.
1142    pub fn release() {
1143        HELD.store(false, Ordering::SeqCst);
1144    }
1145
1146    extern "C" fn on_signal(signal: libc::c_int) {
1147        if HELD.load(Ordering::SeqCst) {
1148            let path = PATH.load(Ordering::SeqCst);
1149            if !path.is_null() {
1150                // SAFETY: a valid NUL-terminated path that is never freed; unlink is
1151                // async-signal-safe
1152                unsafe { libc::unlink(path) };
1153            }
1154        }
1155        // SAFETY: restoring the default disposition and re-raising are async-signal-safe
1156        unsafe {
1157            libc::signal(signal, libc::SIG_DFL);
1158            libc::raise(signal);
1159        }
1160    }
1161}
1162
1163#[cfg(not(unix))]
1164mod signals {
1165    use std::path::Path;
1166
1167    pub fn hold(_: &Path) {}
1168
1169    pub fn release() {}
1170}
