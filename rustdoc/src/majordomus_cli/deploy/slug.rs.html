1//! The tree a provider receives, assembled from the deployment object at one commit (ADR
2//! 0219, decisions 3 and 4).
3//!
4//! What is pushed to a provider is never the repository: not its history, not the
5//! checkout-local half, not a file the object does not name. It is built from three things
6//! only, all read at one commit:
7//!
8//! - the object's `build.inputs`, except the package directory the executable is compiled
9//!   from (`render::runtime_inputs`), read from git rather than from the working tree, so a
10//!   file edited and not committed never reaches a provider;
11//! - the provider's generated files (`deploy/<provider>/`), moved to the slug's root where
12//!   the provider reads them, each held to what the object renders at that commit and the
13//!   hooks made executable;
14//! - the executable built at that same commit, installed at [`render::slug_executable`], and
15//!   the provider's buildpack line ([`render::buildpack`]).
16//!
17//! Beside them the slug carries one record, [`SOURCE_FILE`], naming the deployment, the
18//! commit, the executable's digest and the provider, so a person reading the slug can tell
19//! what it was assembled from without asking anyone.
20//!
21//! The assembler writes only into a directory outside the checkout, absent or empty, and
22//! never into the repository: the only process it starts is git, read-only
23//! ([`crate::git::read_only`]). Pushing the result is a workflow's work
24//! (`.github/workflows/provider.yml`), with a credential this module never sees.
25//!
26//! ```
27//! use majordomus_cli::deploy::slug::{BUILDPACKS_FILE, SOURCE_FILE, SOURCE_SCHEMA};
28//!
29//! // the record and the buildpack list sit beside the provider files, at fixed places
30//! assert_eq!(SOURCE_FILE, ".majordomus/SOURCE.json");
31//! assert_eq!(BUILDPACKS_FILE, ".buildpacks");
32//! assert!(SOURCE_SCHEMA.ends_with("/v1"));
33//! ```

55/// Why a slug was not assembled.
56///
57/// ```
58/// use majordomus_cli::deploy::slug::SlugError;
59/// let e = SlugError::Refused("an out directory inside the checkout".into());
60/// assert_eq!(e.to_string(), "an out directory inside the checkout");
61/// ```
62#[derive(Debug, Clone, PartialEq, Eq)]
63pub enum SlugError {
64    /// The request is well-formed and declined: an out directory inside the checkout or not
65    /// empty, a deployment that names no provider, a provider file the object does not
66    /// render, two files that would land on one path.
67    Refused(String),
68    /// Something the request names does not exist: the commit, the object at that commit,
69    /// the executable.
70    Missing(String),
71    /// git or the filesystem failed under a request that was otherwise sound.
72    Failed(String),
73}
74
75impl std::fmt::Display for SlugError {
76    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
77        match self {
78            SlugError::Refused(s) | SlugError::Missing(s) | SlugError::Failed(s) => f.write_str(s),
79        }
80    }
81}

127/// What was assembled: where, from what, and the provider facts a publisher needs to push
128/// it and to ask the deployment what it serves.
129///
130/// ```
131/// use majordomus_cli::deploy::slug::Slug;
132/// let json = r#"{"deployment":"majordomus","commit":"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa",
133///   "out":"/tmp/slug","entries":[".buildpacks","Procfile"],"files":2,"bytes":10,
134///   "executable":".majordomus/bin/majordomus","executable_sha256":"00",
135///   "buildpack":"https://example.org/bp#0","liveness":"/api/v1/live",
136///   "provider":{"name":"gigalixir","app":"a-b","stack":"gigalixir-24",
137///               "url":"https://a-b.gigalixirapp.com"}}"#;
138/// let s: Slug = serde_json::from_str(json).unwrap();
139/// assert_eq!(s.provider.app, "a-b");
140/// assert_eq!(s.entries.len(), 2);
141/// ```
142#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
143pub struct Slug {
144    /// The deployment's `id`.
145    pub deployment: String,
146    /// The full commit every file was read at.
147    pub commit: String,
148    /// The directory the slug was written into, absolute.
149    pub out: String,
150    /// Every top-level entry of the slug, sorted: the whole of what a provider receives at
151    /// its root.
152    pub entries: Vec<String>,
153    /// How many files (and links) were written.
154    pub files: usize,
155    /// Their total size in bytes.
156    pub bytes: u64,
157    /// Where the executable sits, relative to the slug's root.
158    pub executable: String,
159    /// Its hex SHA-256.
160    pub executable_sha256: String,
161    /// The buildpack line written to [`BUILDPACKS_FILE`].
162    pub buildpack: String,
163    /// The route the deployment answers its commit on (`health.liveness`).
164    pub liveness: String,
165    /// The provider block of the object, as read at the commit.
166    pub provider: Provider,
167}

189/// One path of the slug: its bytes, its mode, and what put it there (for a collision's
190/// message).
191#[derive(Debug, Clone)]
192struct Planned {
193    content: Content,
194    mode: Mode,
195    origin: String,
196}

198/// Every path of the slug before any is written, so that a collision refuses the whole slug
199/// rather than leaving half of one.
200#[derive(Debug, Default)]
201struct Plan(BTreeMap<String, Planned>);
202
203impl Plan {
204    fn add(
205        &mut self,
206        path: &str,
207        content: Content,
208        mode: Mode,
209        origin: &str,
210    ) -> Result<(), SlugError> {
211        let safe = !path.is_empty()
212            && Path::new(path)
213                .components()
214                .all(|c| matches!(c, Component::Normal(_)));
215        if !safe {
216            return Err(SlugError::Failed(format!(
217                "{origin} would write `{path}`, which is not a relative path inside the slug"
218            )));
219        }
220        if let Some(there) = self.0.get(path) {
221            return Err(SlugError::Refused(format!(
222                "{origin} and {} would both write `{path}` in the slug; a provider file must not \
223                 shadow a build input — rename one of them in the deployment object",
224                there.origin
225            )));
226        }
227        self.0.insert(
228            path.to_string(),
229            Planned {
230                content,
231                mode,
232                origin: origin.to_string(),
233            },
234        );
235        Ok(())
236    }
237
238    /// No file may stand where another needs a directory.
239    fn check_prefixes(&self) -> Result<(), SlugError> {
240        for path in self.0.keys() {
241            let mut ancestor = Path::new(path).parent();
242            while let Some(dir) = ancestor.filter(|d| !d.as_os_str().is_empty()) {
243                if let Some(file) = self.0.get(dir.to_string_lossy().as_ref()) {
244                    return Err(SlugError::Refused(format!(
245                        "{} writes the file `{}`, where `{path}` needs a directory",
246                        file.origin,
247                        dir.display()
248                    )));
249                }
250                ancestor = dir.parent();
251            }
252        }
253        Ok(())
254    }
255
256    /// The sorted top-level names.
257    fn entries(&self) -> Vec<String> {
258        let top: std::collections::BTreeSet<&str> = self
259            .0
260            .keys()
261            .map(|p| p.split('/').next().unwrap_or(p))
262            .collect();
263        top.into_iter().map(String::from).collect()
264    }
265}

267/// Assemble the slug a request names and write it into its out directory.
268///
269/// Refused, with nothing written, when the out directory is inside the checkout or holds
270/// anything, when the object at the commit names no provider, when a build input is not
271/// tracked at the commit, when a provider file at the commit is not what the object renders,
272/// and when two files would land on one path.
273pub(crate) fn assemble(req: &Request<'_>) -> Result<Slug, SlugError> {
274    let commit = resolve(req.root, req.commit)?;
275    let short = &commit[..12];
276    let d = object_at(req.root, &commit, req.file)?;
277    if d.id != req.id {
278        return Err(SlugError::Refused(format!(
279            "{} at {short} is the deployment `{}`, not `{}`",
280            req.file, d.id, req.id
281        )));
282    }
283    let provider = d.provider.clone().ok_or_else(|| {
284        SlugError::Refused(format!(
285            "the deployment `{}` names no provider at {short} ({}), so nothing hosts it and \
286             there is no tree to assemble; add a `provider` block (ADR 0219)",
287            d.id, req.file
288        ))
289    })?;
290    let out = outside(req.root, req.out)?;
291    let executable = std::fs::read(req.executable).map_err(|e| {
292        SlugError::Missing(format!(
293            "the executable {} cannot be read: {e}",
294            req.executable.display()
295        ))
296    })?;
297    let executable_sha256 = crate::policy::sha256_bytes_hex(&executable);
298
299    let mut plan = Plan::default();
300
301    // --- the build inputs the running process reads, at the commit
302    let inputs: Vec<&str> = render::runtime_inputs(&d)
303        .into_iter()
304        .map(|p| p.trim_end_matches('/'))
305        .collect();
306    let tracked = ls_tree(req.root, &commit, &inputs)?;
307    for input in &inputs {
308        if !tracked.iter().any(|t| under(&t.path, input)) {
309            return Err(SlugError::Refused(format!(
310                "build.inputs names `{input}`, which {short} does not track; a slug carries \
311                 what the commit holds"
312            )));
313        }
314    }
315    for t in tracked
316        .iter()
317        .filter(|t| inputs.iter().any(|i| under(&t.path, i)))
318    {
319        let mode = match (t.kind.as_str(), t.mode.as_str()) {
320            ("blob", "100755") => Mode::Executable,
321            ("blob", "120000") => Mode::Link,
322            ("blob", _) => Mode::File,
323            _ => {
324                return Err(SlugError::Refused(format!(
325                    "`{}` is a {} at {short}, not a file; a slug carries files, never another \
326                     repository",
327                    t.path, t.kind
328                )))
329            }
330        };
331        plan.add(
332            &t.path,
333            Content::Blob(t.oid.clone()),
334            mode,
335            &format!("build.inputs `{}`", t.path),
336        )?;
337    }
338
339    // --- the provider's files, held to what the object renders, moved to the root
340    let dir = render::provider_dir(provider.name);
341    let rendered = render::provider_artifacts(&d, req.file);
342    let paths: Vec<&str> = rendered.iter().map(|a| a.path.as_str()).collect();
343    let committed: BTreeMap<String, Tracked> = ls_tree(req.root, &commit, &paths)?
344        .into_iter()
345        .map(|t| (t.path.clone(), t))
346        .collect();
347    for artifact in &rendered {
348        let at = committed.get(&artifact.path).ok_or_else(|| {
349            SlugError::Refused(format!(
350                "{} is not committed at {short}; run `majordomus generate deployment` and \
351                 commit what it writes",
352                artifact.path
353            ))
354        })?;
355        let bytes = blob(req.root, &at.oid)?;
356        if bytes != artifact.content.as_bytes() {
357            return Err(SlugError::Refused(format!(
358                "{} at {short} is not what {} renders; run `majordomus generate deployment` \
359                 and commit what it writes",
360                artifact.path, req.file
361            )));
362        }
363        let rel = artifact
364            .path
365            .strip_prefix(dir)
366            .and_then(|p| p.strip_prefix('/'))
367            .unwrap_or(&artifact.path);
368        // a script names its interpreter on its first line; every other file is data
369        let mode = if artifact.content.starts_with("#!") {
370            Mode::Executable
371        } else {
372            Mode::File
373        };
374        plan.add(rel, Content::Bytes(bytes), mode, &artifact.path)?;
375    }
376
377    // --- the buildpack line, the executable and the record
378    let buildpack = render::buildpack(provider.name).to_string();
379    plan.add(
380        BUILDPACKS_FILE,
381        Content::Bytes(format!("{buildpack}\n").into_bytes()),
382        Mode::File,
383        "the provider's buildpack line",
384    )?;
385    let installed = render::slug_executable(&d);
386    plan.add(
387        &installed,
388        Content::Bytes(executable),
389        Mode::Executable,
390        &format!("the executable {}", req.executable.display()),
391    )?;
392    let source = Source {
393        schema: SOURCE_SCHEMA.into(),
394        deployment: d.id.clone(),
395        commit: commit.clone(),
396        executable_sha256: executable_sha256.clone(),
397        provider: provider.name,
398    };
399    // strings and a unit variant: a record that cannot fail to serialize
400    let record = serde_json::to_string_pretty(&source).expect("the slug's record serializes");
401    plan.add(
402        SOURCE_FILE,
403        Content::Bytes(format!("{record}\n").into_bytes()),
404        Mode::File,
405        "the slug's record",
406    )?;
407    plan.check_prefixes()?;
408
409    let (files, bytes) = write(req.root, &out, &plan)?;
410    Ok(Slug {
411        deployment: d.id.clone(),
412        commit,
413        out: out.display().to_string(),
414        entries: plan.entries(),
415        files,
416        bytes,
417        executable: installed,
418        executable_sha256,
419        buildpack,
420        liveness: d.health.liveness.as_str().to_string(),
421        provider,
422    })
423}

425/// Is `path` the input itself or a file under it?
426fn under(path: &str, input: &str) -> bool {
427    path == input
428        || path
429            .strip_prefix(input)
430            .is_some_and(|rest| rest.starts_with('/'))
431}

433/// The full commit a revision names.
434fn resolve(root: &Path, rev: &str) -> Result<String, SlugError> {
435    let out = crate::git::read_only(root)
436        .args([
437            "rev-parse",
438            "--verify",
439            "--quiet",
440            &format!("{rev}^{{commit}}"),
441        ])
442        .output()
443        .map_err(|e| SlugError::Failed(format!("cannot run git: {e}")))?;
444    let sha = String::from_utf8_lossy(&out.stdout).trim().to_string();
445    let hex = sha.chars().all(|c| c.is_ascii_hexdigit());
446    if !out.status.success() || !hex || !(sha.len() == 40 || sha.len() == 64) {
447        return Err(SlugError::Missing(format!(
448            "`{rev}` does not name a commit this clone has"
449        )));
450    }
451    Ok(sha)
452}

454/// The deployment object as the commit holds it.
455fn object_at(root: &Path, commit: &str, file: &str) -> Result<Deployment, SlugError> {
456    let out = crate::git::read_only(root)
457        .args(["cat-file", "blob", &format!("{commit}:{file}")])
458        .output()
459        .map_err(|e| SlugError::Failed(format!("cannot run git: {e}")))?;
460    if !out.status.success() {
461        return Err(SlugError::Missing(format!(
462            "{file} does not exist at {}; the slug is assembled from the object the commit holds",
463            &commit[..12]
464        )));
465    }
466    let text = String::from_utf8(out.stdout)
467        .map_err(|_| SlugError::Refused(format!("{file} at {} is not UTF-8", &commit[..12])))?;
468    let map = crate::metadata::yaml::parse_mapping(&text)
469        .map_err(|e| SlugError::Refused(format!("{file} at {}: {e}", &commit[..12])))?;
470    Deployment::from_metadata(file, serde_json::Value::Object(map))
471        .map_err(|r| SlugError::Refused(r.to_string()))
472}

474/// The out directory, absolute and checked: outside the checkout, and absent or empty.
475fn outside(root: &Path, out: &Path) -> Result<PathBuf, SlugError> {
476    if out.components().any(|c| matches!(c, Component::ParentDir)) {
477        return Err(SlugError::Refused(format!(
478            "the out directory {} climbs with `..`; name it directly",
479            out.display()
480        )));
481    }
482    let absolute = std::path::absolute(out).map_err(|e| {
483        SlugError::Refused(format!(
484            "the out directory `{}` cannot be made absolute: {e}; name a directory",
485            out.display()
486        ))
487    })?;
488    // the nearest ancestor that resolves, and the part below it that does not exist yet; the
489    // filesystem's root always resolves, so the fallback is the path as named
490    let (existing, mut resolved) = absolute
491        .ancestors()
492        .find_map(|a| a.canonicalize().ok().map(|c| (a, c)))
493        .unwrap_or((absolute.as_path(), absolute.clone()));
494    for name in absolute
495        .strip_prefix(existing)
496        .unwrap_or(Path::new(""))
497        .components()
498    {
499        resolved.push(name);
500    }
501    let root = root
502        .canonicalize()
503        .map_err(|e| SlugError::Failed(format!("{} cannot be resolved: {e}", root.display())))?;
504    if resolved.starts_with(&root) {
505        return Err(SlugError::Refused(format!(
506            "the out directory {} is inside the checkout {}; a slug is assembled outside the \
507             repository and never in it",
508            resolved.display(),
509            root.display()
510        )));
511    }
512    if resolved.exists() {
513        let mut entries = std::fs::read_dir(&resolved).map_err(|e| {
514            SlugError::Refused(format!(
515                "the out directory {} is not a readable directory: {e}",
516                resolved.display()
517            ))
518        })?;
519        if entries.next().is_some() {
520            return Err(SlugError::Refused(format!(
521                "the out directory {} is not empty; a slug holds what it was assembled from and \
522                 nothing else",
523                resolved.display()
524            )));
525        }
526    }
527    Ok(resolved)
528}

530/// One entry of `git ls-tree -r`.
531#[derive(Debug, Clone)]
532struct Tracked {
533    mode: String,
534    kind: String,
535    oid: String,
536    path: String,
537}
538
539/// Every file the commit tracks under the paths, recursively. No path is no file: an empty
540/// pathspec would list the whole tree.
541fn ls_tree(root: &Path, commit: &str, paths: &[&str]) -> Result<Vec<Tracked>, SlugError> {
542    if paths.is_empty() {
543        return Ok(Vec::new());
544    }
545    let out = crate::git::read_only(root)
546        .args(["ls-tree", "-r", "-z", "--full-tree", commit, "--"])
547        .args(paths)
548        .output()
549        .map_err(|e| SlugError::Failed(format!("cannot run git: {e}")))?;
550    if !out.status.success() {
551        return Err(SlugError::Failed(format!(
552            "git ls-tree {commit}: {}",
553            String::from_utf8_lossy(&out.stderr).trim()
554        )));
555    }
556    parse_ls_tree(&out.stdout)
557}

559/// The entries of what `git ls-tree -r -z` printed.
560fn parse_ls_tree(stdout: &[u8]) -> Result<Vec<Tracked>, SlugError> {
561    let mut tracked = Vec::new();
562    for raw in stdout.split(|b| *b == 0).filter(|r| !r.is_empty()) {
563        let line = std::str::from_utf8(raw).map_err(|_| {
564            SlugError::Refused(format!(
565                "a tracked path is not UTF-8: {}",
566                String::from_utf8_lossy(raw)
567            ))
568        })?;
569        let (meta, path) = line
570            .split_once('\t')
571            .ok_or_else(|| SlugError::Failed(format!("git ls-tree said `{line}`")))?;
572        let mut parts = meta.split(' ');
573        let (Some(mode), Some(kind), Some(oid)) = (parts.next(), parts.next(), parts.next()) else {
574            return Err(SlugError::Failed(format!("git ls-tree said `{line}`")));
575        };
576        tracked.push(Tracked {
577            mode: mode.into(),
578            kind: kind.into(),
579            oid: oid.into(),
580            path: path.into(),
581        });
582    }
583    Ok(tracked)
584}

586/// One blob's bytes.
587fn blob(root: &Path, oid: &str) -> Result<Vec<u8>, SlugError> {
588    let out = crate::git::read_only(root)
589        .args(["cat-file", "blob", oid])
590        .output()
591        .map_err(|e| SlugError::Failed(format!("cannot run git: {e}")))?;
592    if !out.status.success() {
593        return Err(SlugError::Failed(format!(
594            "git cat-file blob {oid}: {}",
595            String::from_utf8_lossy(&out.stderr).trim()
596        )));
597    }
598    Ok(out.stdout)
599}

601/// Write the plan: every blob streamed through one `git cat-file --batch`, every other file
602/// from the bytes in hand. Answers how many files were written and their total size.
603fn write(root: &Path, out: &Path, plan: &Plan) -> Result<(usize, u64), SlugError> {
604    let io = |what: &Path, e: std::io::Error| {
605        SlugError::Failed(format!("{} cannot be written: {e}", what.display()))
606    };
607    std::fs::create_dir_all(out).map_err(|e| io(out, e))?;
608    let mut bytes = 0u64;
609    let blobs: Vec<(&String, &Planned, &String)> = plan
610        .0
611        .iter()
612        .filter_map(|(path, p)| match &p.content {
613            Content::Blob(oid) => Some((path, p, oid)),
614            Content::Bytes(_) => None,
615        })
616        .collect();
617    if !blobs.is_empty() {
618        let oids: Vec<String> = blobs.iter().map(|(_, _, oid)| (*oid).clone()).collect();
619        let mut child = crate::git::read_only(root)
620            .args(["cat-file", "--batch"])
621            .stdin(Stdio::piped())
622            .stdout(Stdio::piped())
623            .stderr(Stdio::null())
624            .spawn()
625            .map_err(|e| SlugError::Failed(format!("cannot run git: {e}")))?;
626        let mut stdin = child.stdin.take().expect("piped stdin");
627        // the requests go in on a thread of their own, so a full pipe on either side cannot
628        // hold the other: git answers while it is still being asked
629        let asking = std::thread::spawn(move || -> std::io::Result<()> {
630            for oid in oids {
631                writeln!(stdin, "{oid}")?;
632            }
633            Ok(())
634        });
635        let mut reader = BufReader::new(child.stdout.take().expect("piped stdout"));
636        let streamed = stream(&mut reader, out, &blobs);
637        drop(reader);
638        if streamed.is_err() {
639            let _ = child.kill();
640        }
641        let _ = asking.join();
642        let _ = child.wait();
643        bytes += streamed?;
644    }
645    for (path, planned) in &plan.0 {
646        if let Content::Bytes(content) = &planned.content {
647            place(out, path, content, planned.mode)?;
648            bytes += content.len() as u64;
649        }
650    }
651    Ok((plan.0.len(), bytes))
652}

654/// Read what `git cat-file --batch` answers for each planned blob, in the order they were
655/// asked for, and place each. Answers the bytes placed.
656fn stream(
657    reader: &mut impl BufRead,
658    out: &Path,
659    blobs: &[(&String, &Planned, &String)],
660) -> Result<u64, SlugError> {
661    let mut total = 0u64;
662    for (path, planned, oid) in blobs {
663        let mut header = String::new();
664        reader
665            .read_line(&mut header)
666            .map_err(|e| SlugError::Failed(format!("git cat-file --batch: {e}")))?;
667        let fields: Vec<&str> = header.trim_end().split(' ').collect();
668        let size = match fields.as_slice() {
669            [o, "blob", size] if *o == oid.as_str() => size.parse::<usize>().ok(),
670            _ => None,
671        }
672        .ok_or_else(|| {
673            SlugError::Failed(format!(
674                "git cat-file --batch answered `{}` for {oid}",
675                header.trim_end()
676            ))
677        })?;
678        let mut content = vec![0u8; size + 1];
679        reader
680            .read_exact(&mut content)
681            .map_err(|e| SlugError::Failed(format!("git cat-file --batch: {e}")))?;
682        content.truncate(size);
683        place(out, path, &content, planned.mode)?;
684        total += size as u64;
685    }
686    Ok(total)
687}

689/// Write one file of the slug with its mode.
690fn place(out: &Path, rel: &str, content: &[u8], mode: Mode) -> Result<(), SlugError> {
691    let path = out.join(rel);
692    let io =
693        |e: std::io::Error| SlugError::Failed(format!("{} cannot be written: {e}", path.display()));
694    // `rel` is never empty (`Plan::add`), so the path always has a parent
695    std::fs::create_dir_all(path.parent().unwrap_or(out)).map_err(io)?;
696    match mode {
697        Mode::Link => link(content, &path).map_err(io),
698        Mode::File | Mode::Executable => {
699            std::fs::write(&path, content).map_err(io)?;
700            permit(&path, mode == Mode::Executable).map_err(io)
701        }
702    }
703}

708    std::os::unix::fs::symlink(std::ffi::OsStr::from_bytes(target), path)
