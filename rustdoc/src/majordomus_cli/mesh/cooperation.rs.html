1//! Cooperation: the runtime that turns discovered, trusted runtimes of one repository into
2//! linked peers and keeps their journal converged. It owns the one peer-link table, the
3//! one journal of this runtime, the link handlers the HTTP surface answers, the dialers,
4//! the heartbeat and the expiry — and the operations (sessions, claims, handovers,
5//! reviews) every surface performs through it.
6//!
7//! The sequence one link goes through, and where each step is decided:
8//!
9//! ```text
10//! candidate  a trusted registry record of the same repository, or a declared seed
11//!    ↓
12//! dial       hello → welcome  (link::)  signature, freshness, nonce, protocol,
13//!                                        repository, self, trust — both directions
14//!    ↓
15//! connected  sync every heartbeat: marks + missing events, both ways, signed
16//!    ↓        a failed round: degraded → unreachable (backoff, capped at the expiry)
17//! expired    no exchange within the expiry: the link is dropped, the runtime's
18//!    ↓        streams stop beating everywhere, its claims stop excluding
19//! reconnect  the dialer says hello again; an unknown link id or a new instance is a
20//!             re-handshake, never a second peer — the table is keyed by runtime
21//! ```
22//!
23//! StreamLiveness has two layers and they are deliberately separate. A *link* is this runtime's
24//! connection to one peer (connected, degraded, unreachable, expired). A *stream* is
25//! alive while its beat keeps rising anywhere in the mesh — a runtime linked only through
26//! a relay is still alive, and a runtime that crashed stops beating for everyone at once.
27//! Claims follow streams, not links.
28//!
29//! The whole of it, two runtimes in one process. Only the socket is replaced: the
30//! handshake, the signatures, the admission and the fold are the ones a LAN takes.
31//!
32//! ```
33//! use std::collections::BTreeMap;
34//! use std::sync::{Arc, Mutex, Weak};
35//!
36//! use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
37//! use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
38//! use majordomus_cli::mesh::identity::NodeIdentity;
39//! use majordomus_cli::mesh::journal::{ClaimMode, SessionInfo};
40//! use majordomus_cli::mesh::link::{LinkReply, LinkTransport, Signed, HELLO_PATH};
41//! use majordomus_cli::mesh::registry::MeshRegistry;
42//! use majordomus_cli::mesh::repository::{of_root_commits, runtime_id};
43//! use majordomus_cli::mesh::trust::TrustPolicy;
44//!
45//! #[derive(Default)]
46//! struct Net(Mutex<BTreeMap<String, Weak<Cooperation>>>);
47//!
48//! impl LinkTransport for Net {
49//!     fn post(&self, endpoint: &str, path: &str, m: &Signed) -> Result<LinkReply, String> {
50//!         let peer = (self.0.lock().unwrap().get(endpoint).and_then(Weak::upgrade))
51//!             .ok_or_else(|| format!("{endpoint}: nobody there"))?;
52//!         Ok(if path == HELLO_PATH { peer.accept_hello(m) } else { peer.accept_sync(m) })
53//!     }
54//! }
55//!
56//! let net = Arc::new(Net::default());
57//! let runtime = |endpoint: &str| {
58//!     let c = Cooperation::new(CooperationSetup {
59//!         identity: Arc::new(NodeIdentity::ephemeral().unwrap()),
60//!         runtime: runtime_id(endpoint),
61//!         repository: of_root_commits(&["root".into()]),
62//!         endpoints: vec![endpoint.into()],
63//!         version: "doc".into(),
64//!         config: CooperationConfig::default(),
65//!         trust: TrustConfig { policy: TrustPolicy::Tofu, allow: vec![] },
66//!         journal_path: None,
67//!         registry: Arc::new(MeshRegistry::new()),
68//!         transport: Arc::clone(&net) as Arc<dyn LinkTransport>,
69//!         board: None,
70//!         checkout: CheckoutFacts::default(),
71//!     })
72//!     .unwrap();
73//!     net.0.lock().unwrap().insert(endpoint.into(), Arc::downgrade(&c));
74//!     c
75//! };
76//!
77//! let (a, b) = (runtime("a:1"), runtime("b:1"));
78//! let key = a.dial(&["b:1".into()]).expect("B welcomes A");
79//! assert_eq!(key, b.runtime_key(), "the link is keyed by the peer's runtime");
80//!
81//! a.claim(&SessionInfo::named("s1", "cli"), vec!["apps".into()], None,
82//!     ClaimMode::Exclusive, None).unwrap();
83//! for _ in 0..2 {
84//!     a.journal().beat_own();
85//!     a.sync_with(&key).unwrap();
86//! }
87//! assert_eq!(b.state().claims.len(), 1, "the claim crossed the link");
88//! assert_eq!(a.state().digest, b.state().digest, "one state, two runtimes");
89//! ```

193/// What the projected sessions say about the checkout they work in. Computed once at
194/// start — branch and head are refreshed when a board session changes, never on a
195/// heartbeat, so liveness never rescans the repository.
196///
197/// Both fields are optional because a runtime need not be in a checkout at all, and a
198/// runtime that is says so once: the root is kept rather than the branch, so that a
199/// session announced after a checkout moved carries where it is now and not where it was

202/// ```
203/// use majordomus_cli::mesh::cooperation::CheckoutFacts;

205/// let unknown = CheckoutFacts::default();

207///
208/// let here = CheckoutFacts { id: Some("c0ffee".into()), root: Some(".".into()) };
209/// assert_eq!(here.id.as_deref(), Some("c0ffee"));
210/// ```

213    /// The checkout id (a digest).
214    pub id: Option<String>,
215    /// The checkout root, for reading the branch and the head when a session changes.
216    pub root: Option<PathBuf>,
217}

241/// A link's standing, from this runtime's side. It is a statement about a connection and
242/// never about the peer: a runtime reachable only through a third party is `Expired` here
243/// while its streams beat on everywhere, which is why claims follow streams and not this.
244///
245/// The variants are ordered by how far the link has fallen, so the worst link in a table
246/// is the maximum and a reader never has to rank them by hand.
247///
248/// ```
249/// use majordomus_cli::mesh::cooperation::LinkState;
250///
251/// let table = [LinkState::Connected, LinkState::Unreachable, LinkState::Degraded];
252/// assert_eq!(table.iter().max(), Some(&LinkState::Unreachable), "the worst link shows");
253/// assert_eq!(serde_json::to_value(LinkState::Connected).unwrap(), "connected");
254/// ```
255#[derive(
256    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
257)]
258#[serde(rename_all = "snake_case")]
259pub enum LinkState {
260    /// Handshaking, or never yet exchanged.
261    Connecting,
262    /// An exchange within two heartbeats.
263    Connected,
264    /// Rounds are failing; the last exchange is under half the expiry old.
265    Degraded,
266    /// The last exchange is over half the expiry old; the link is about to expire.
267    Unreachable,
268    /// No exchange within the expiry: the link is dropped and will be re-dialed.
269    Expired,
270}

489/// The link and replication counters since start: what this runtime did, not what it
490/// holds. They are monotonic and never reset, so two readings a minute apart are a rate —
491/// which is the only way to tell a mesh that is working from one that is retrying, since
492/// both show the same peers in the same states.
493///
494/// The counts are split by direction throughout (`handshakes_out` against
495/// `handshakes_in`, `refused_in` against `refused_out`) because a runtime that refuses
496/// everyone and one that everyone refuses need opposite remedies.
497///
498/// ```
499/// # use std::collections::BTreeMap;
500/// # use std::sync::{Arc, Mutex, Weak};
501/// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
502/// # use majordomus_cli::mesh::cooperation::*;
503/// # use majordomus_cli::mesh::identity::NodeIdentity;
504/// # use majordomus_cli::mesh::link::{LinkReply, LinkTransport, Signed, HELLO_PATH};
505/// # use majordomus_cli::mesh::registry::MeshRegistry;
506/// # use majordomus_cli::mesh::repository::{of_root_commits, runtime_id};
507/// # use majordomus_cli::mesh::trust::TrustPolicy;
508/// # #[derive(Default)] struct Net(Mutex<BTreeMap<String, Weak<Cooperation>>>);
509/// # impl LinkTransport for Net {
510/// #   fn post(&self, e: &str, p: &str, m: &Signed) -> Result<LinkReply, String> {
511/// #     let peer = (self.0.lock().unwrap().get(e).and_then(Weak::upgrade))
512/// #         .ok_or_else(|| format!("{e}: nobody there"))?;
513/// #     Ok(if p == HELLO_PATH { peer.accept_hello(m) } else { peer.accept_sync(m) }) } }
514/// # let net = Arc::new(Net::default());
515/// # let runtime = |endpoint: &str| { let c = Cooperation::new(CooperationSetup {
516/// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: runtime_id(endpoint),
517/// #     repository: of_root_commits(&["root".into()]), endpoints: vec![endpoint.into()],
518/// #     version: "doc".into(), config: CooperationConfig::default(),
519/// #     trust: TrustConfig { policy: TrustPolicy::Tofu, allow: vec![] }, journal_path: None,
520/// #     registry: Arc::new(MeshRegistry::new()),
521/// #     transport: Arc::clone(&net) as Arc<dyn LinkTransport>, board: None,
522/// #     checkout: CheckoutFacts::default() }).unwrap();
523/// #   net.0.lock().unwrap().insert(endpoint.into(), Arc::downgrade(&c)); c };
524/// # let (a, b) = (runtime("a:1"), runtime("b:1"));
525/// let key = a.dial(&["b:1".into()]).unwrap();
526/// a.sync_with(&key).unwrap();
527///

680/// Why a cooperation operation did not happen. Every variant is a decision rather than a
681/// fault: nothing here is a transport failure or a bug, so a caller that sees one has been
682/// told something true about the mesh and should say it to its user rather than retry.
683///
684/// `Conflict` carries the claims it met instead of a message about them, because the
685/// worker being refused needs the other side's scope, session and runtime to go and talk
686/// to whoever holds it.
687///
688/// ```
689/// # use std::sync::Arc;
690/// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
691/// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
692/// # use majordomus_cli::mesh::identity::NodeIdentity;
693/// # use majordomus_cli::mesh::link::HttpTransport;
694/// # use majordomus_cli::mesh::registry::MeshRegistry;
695/// # use majordomus_cli::mesh::repository::of_root_commits;
696/// # let cooperation = Cooperation::new(CooperationSetup {
697/// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
698/// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
699/// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
700/// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
701/// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
702/// # }).unwrap();
703/// use majordomus_cli::mesh::cooperation::CooperationError;
704/// use majordomus_cli::mesh::journal::{ClaimMode, SessionInfo};
705///
706/// let held = SessionInfo::named("s1", "cli");
707/// cooperation.claim(&held, vec!["apps".into()], None, ClaimMode::Exclusive, None).unwrap();
708///
709/// let latecomer = SessionInfo::named("s2", "cli");
710/// let refused = cooperation
711///     .claim(&latecomer, vec!["apps/cli".into()], None, ClaimMode::Exclusive, None)
712///     .unwrap_err();
713/// match &refused {
714///     CooperationError::Conflict(claims) => assert_eq!(claims[0].scope, ["apps"]),
715///     other => panic!("expected a conflict, got {other}"),
716/// }
717/// assert!(refused.to_string().contains("apps"), "the message names what it met");
718/// ```
719#[derive(Debug, Clone, thiserror::Error)]
720pub enum CooperationError {
721    /// An exclusive claim meets live exclusive claims of other sessions.
722    #[error("the scope is claimed: {}", .0.iter().map(|c| format!("{} ({})", c.key, c.scope.join(", "))).collect::<Vec<_>>().join("; "))]
723    Conflict(Vec<ClaimView>),
724    /// The operation needs something this runtime wrote, and this was written elsewhere.
725    #[error("{0}")]
726    NotOwn(String),
727    /// Nothing by that name.
728    #[error("{0}")]
729    NotFound(String),
730    /// The peer lacks a feature the operation needs.
731    #[error("{0}")]
732    FeatureUnsupported(String),
733    /// Out of bounds.
734    #[error("{0}")]
735    Invalid(String),
736}

785/// What one sync round, run now rather than remembered, said about one peer. Verification
786/// does not report the link's standing and stop there: it exchanges with the peer and says
787/// whether the exchange worked and whether both sides ended holding the same marks, which
788/// is the difference between a peer that answers and a peer that agrees.
789///
790/// `round_trip` and `converged` are optional because a peer this runtime does not dial is
791/// not asked — an absent answer is not a failed one, and reporting `false` there would
792/// make an inbound-only link look broken every time.
793///
794/// ```
795/// # use std::collections::BTreeMap;
796/// # use std::sync::{Arc, Mutex, Weak};
797/// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
798/// # use majordomus_cli::mesh::cooperation::*;
799/// # use majordomus_cli::mesh::identity::NodeIdentity;
800/// # use majordomus_cli::mesh::link::{LinkReply, LinkTransport, Signed, HELLO_PATH};
801/// # use majordomus_cli::mesh::registry::MeshRegistry;
802/// # use majordomus_cli::mesh::repository::{of_root_commits, runtime_id};
803/// # use majordomus_cli::mesh::trust::TrustPolicy;
804/// # #[derive(Default)] struct Net(Mutex<BTreeMap<String, Weak<Cooperation>>>);
805/// # impl LinkTransport for Net {
806/// #   fn post(&self, e: &str, p: &str, m: &Signed) -> Result<LinkReply, String> {
807/// #     let peer = (self.0.lock().unwrap().get(e).and_then(Weak::upgrade))
808/// #         .ok_or_else(|| format!("{e}: nobody there"))?;
809/// #     Ok(if p == HELLO_PATH { peer.accept_hello(m) } else { peer.accept_sync(m) }) } }
810/// # let net = Arc::new(Net::default());
811/// # let runtime = |endpoint: &str| { let c = Cooperation::new(CooperationSetup {
812/// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: runtime_id(endpoint),
813/// #     repository: of_root_commits(&["root".into()]), endpoints: vec![endpoint.into()],
814/// #     version: "doc".into(), config: CooperationConfig::default(),
815/// #     trust: TrustConfig { policy: TrustPolicy::Tofu, allow: vec![] }, journal_path: None,
816/// #     registry: Arc::new(MeshRegistry::new()),
817/// #     transport: Arc::clone(&net) as Arc<dyn LinkTransport>, board: None,
818/// #     checkout: CheckoutFacts::default() }).unwrap();
819/// #   net.0.lock().unwrap().insert(endpoint.into(), Arc::downgrade(&c)); c };
820/// # let (a, b) = (runtime("a:1"), runtime("b:1"));
821/// a.dial(&["b:1".into()]).unwrap();
822///
823/// let dialer: &PeerVerdict = &a.verify().peers[0];
824/// assert_eq!(dialer.round_trip, Some(true), "asked, and answered");
825///
826/// let answerer = &b.verify().peers[0];
827/// assert!(!answerer.dialed, "B does not dial A, so B asks nothing of it");
828/// assert_eq!(answerer.round_trip, None, "and reports no verdict it did not earn");
829/// ```
830#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
831pub struct PeerVerdict {
832    /// The peer's runtime key.
833    pub runtime: String,
834    /// The link's standing after the round.
835    pub state: LinkState,
836    /// Whether this runtime dials the peer (and so ran a round with it now).
837    pub dialed: bool,
838    /// Whether the round succeeded; `None` when the peer dials this runtime instead.
839    #[serde(default, skip_serializing_if = "Option::is_none")]
840    pub round_trip: Option<bool>,
841    /// The round trip, milliseconds.
842    #[serde(default, skip_serializing_if = "Option::is_none")]
843    pub rtt_ms: Option<u64>,
844    /// Whether both sides hold the same high-water mark of every stream after the round.
845    #[serde(default, skip_serializing_if = "Option::is_none")]
846    pub converged: Option<bool>,
847    /// What happened.
848    pub detail: String,
849}

929/// What an operation wrote, as its answer to the caller. Every mesh operation is an event
930/// appended to this runtime's journal, and the caller is handed the three facts it cannot
931/// work out for itself: the mesh-wide key of the thing it created, the event that says so,

935/// acquisition returned, never by the local name it was given — and the stamp is what a
936/// caller waiting for a peer to catch up compares against.

939/// # use std::sync::Arc;

941/// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
942/// # use majordomus_cli::mesh::identity::NodeIdentity;

945/// # use majordomus_cli::mesh::repository::of_root_commits;
946/// # let cooperation = Cooperation::new(CooperationSetup {
947/// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
948/// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
949/// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
950/// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
951/// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
952/// # }).unwrap();
953/// use majordomus_cli::mesh::cooperation::Written;
954/// use majordomus_cli::mesh::journal::{ClaimMode, SessionInfo};
955///
956/// let session = SessionInfo::named("s1", "cli");
957/// let written: Written = cooperation
958///     .claim(&session, vec!["docs".into()], None, ClaimMode::Advisory, None)
959///     .unwrap();
960/// assert!(written.lamport > 0, "placed against every other runtime's events");
961///
962/// // and the key it returned is how the claim is addressed from here on
963/// cooperation.release(&written.key).unwrap();

967    /// `<stream>/<local id>` of what was created or changed.
968    pub key: String,
969    /// `<stream>/<seq>` of the event that says so.
970    pub event: String,
971    /// The event's Lamport stamp.

975/// What a dial or a sync round did not do. The three are kept apart because they are acted
976/// on differently: an unreachable peer is retried with a backoff, a refusal is a decision
977/// that will be made the same way next time and is recorded rather than retried, and the
978/// absence of a link is this runtime's own state and not the peer's.
979///
980/// ```
981/// # use std::sync::Arc;
982/// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
983/// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
984/// # use majordomus_cli::mesh::identity::NodeIdentity;
985/// # use majordomus_cli::mesh::link::HttpTransport;
986/// # use majordomus_cli::mesh::registry::MeshRegistry;
987/// # use majordomus_cli::mesh::repository::of_root_commits;
988/// # let cooperation = Cooperation::new(CooperationSetup {
989/// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
990/// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
991/// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
992/// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
993/// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
994/// # }).unwrap();
995/// use majordomus_cli::mesh::cooperation::RoundError;
996///
997/// // nobody answers at the discard port: a peer to back off from, not a decision
998/// let nobody = cooperation.dial(&["127.0.0.1:9".into()]).unwrap_err();
999/// assert!(matches!(nobody, RoundError::Unreachable(_)), "{nobody}");
1000/// assert!(nobody.to_string().contains("127.0.0.1:9"), "a failure names where it was going");
1001///
1002/// // and syncing with a peer this runtime never linked to is its own state, not the peer's
1003/// let never = cooperation.sync_with("somebody-0000000000000002").unwrap_err();
1004/// assert!(matches!(never, RoundError::NoLink), "{never}");
1005/// ```
1006#[derive(Debug, Clone)]
1007pub enum RoundError {
1008    /// No endpoint answered.
1009    Unreachable(String),
1010    /// The peer, or this runtime's check of the peer, refused.
1011    Refused(LinkRefusal),
1012    /// There is no outbound link to sync.
1013    NoLink,
1014}

1032/// A registry record this runtime may dial, under the runtime key it would be dialed by.
1033///
1034/// Its order is the order [`MAX_TARGETS_PER_NODE`] cuts a node's runtimes in: by node, then
1035/// the runtimes answering now before the ones that stopped, then by runtime. Presence comes
1036/// before the runtime because the registry keeps a stopped runtime's record for its retention
1037/// and every runtime of a machine orders the records alike: cut by runtime alone, stopped
1038/// servers that sort first fill the cap on every live runtime at once, and no live runtime of
1039/// the node dials a live sibling until the stopped records expire.
1040struct DialCandidate<'a> {
1041    record: &'a NodeRecord,
1042    key: String,
1043}
1044
1045impl crate::order::Ordered for DialCandidate<'_> {
1046    fn order_key(&self) -> crate::order::OrderKey<'_> {
1047        let rank = match self.record.presence {
1048            Presence::Present => 0,
1049            Presence::Absent => 1,
1050        };
1051        crate::order::OrderKey::grouped(
1052            self.record.node_id.as_str(),
1053            &self.record.runtime,
1054            &self.key,
1055        )
1056        .ranked(rank)
1057    }
1058}

1060/// The registry records this runtime dials: every record of `repository` that names a
1061/// runtime slot and an endpoint, whose key `trusted` accepts, other than `own_key` — at most
1062/// [`MAX_TARGETS_PER_NODE`] of one node, taken in [`DialCandidate`] order, so a node's present
1063/// runtimes are dialed before its stopped ones. The same records in any order choose the
1064/// same targets.
1065fn dial_targets(
1066    records: &[NodeRecord],
1067    repository: &str,
1068    own_key: &str,
1069    trusted: impl Fn(&str) -> bool,
1070) -> Vec<Target> {
1071    let mut candidates: Vec<DialCandidate<'_>> = records
1072        .iter()
1073        .filter(|record| {
1074            !record.runtime.is_empty()
1075                && !record.endpoints.is_empty()
1076                && record.repositories.iter().any(|r| r == repository)
1077        })
1078        .map(|record| DialCandidate {
1079            key: format!("{}-{}", record.node_id, record.runtime),
1080            record,
1081        })
1082        .filter(|candidate| candidate.key != own_key && trusted(&candidate.record.public_key))
1083        .collect();
1084    crate::order::canonical(&mut candidates);
1085    // At most MAX_TARGETS_PER_NODE runtimes of one node are dialed: runtime slots are free
1086    // to invent, and a key must not turn this runtime into a dialer of arbitrary endpoints.
1087    let mut per_node: BTreeMap<&str, usize> = BTreeMap::new();
1088    let mut targets = Vec::new();
1089    for candidate in candidates {
1090        let dialed = per_node
1091            .entry(candidate.record.node_id.as_str())
1092            .or_default();
1093        if *dialed >= MAX_TARGETS_PER_NODE {
1094            continue;
1095        }
1096        *dialed += 1;
1097        targets.push(Target {
1098            id: candidate.key.clone(),
1099            key: Some(candidate.key),
1100            endpoints: candidate.record.endpoints.clone(),
1101        });
1102    }
1103    targets
1104}

1106/// One runtime's whole part in the mesh, behind one handle. It holds the single journal

1172/// `text` cut to at most `max` bytes on a character boundary, with control characters
1173/// removed: what the journal's single-line bounds accept, so a projection never fails
1174/// validation on a long or multi-byte title and silently retries forever.
1175fn clip(text: &str, max: usize) -> String {
1176    let mut out = String::new();
1177    for c in text.chars().filter(|c| !c.is_control()) {
1178        if out.len() + c.len_utf8() > max {
1179            break;
1180        }
1181        out.push(c);
1182    }
1183    out
1184}

1186fn remember_refusal(table: &mut Table, key: String, refused: Refused) {
1187    if !table.refused.contains_key(&key) && table.refused.len() >= MAX_REFUSED {
1188        if let Some(oldest) = table
1189            .refused
1190            .iter()
1191            .min_by(|a, b| a.1.at.cmp(&b.1.at))
1192            .map(|(k, _)| k.clone())
1193        {
1194            table.refused.remove(&oldest);
1195        }
1196    }
1197    table.refused.insert(key, refused);
1198}

1285    /// Project the peer board into the journal: an attached session that is new opens a
1286    /// mesh session, a changed announcement releases its previous advisory claim and
1287    /// takes the new one, a withdrawn announcement is released, and a session that left
1288    /// the board is closed. One-way and idempotent — the board stays the local truth, the
1289    /// journal carries it to every linked runtime. Nothing happens when nothing changed.
1290    ///
1291    /// This is the whole bridge between the board a person reads on one machine and the
1292    /// mesh every other machine reads: a worker announces once, to its own server, and is
1293    /// visible everywhere without knowing that anything else exists.
1294    ///
1295    /// ```
1296    /// # use std::sync::Arc;
1297    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
1298    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
1299    /// # use majordomus_cli::mesh::identity::NodeIdentity;
1300    /// # use majordomus_cli::mesh::link::HttpTransport;
1301    /// # use majordomus_cli::mesh::registry::MeshRegistry;
1302    /// # use majordomus_cli::mesh::repository::of_root_commits;
1303    /// use majordomus_cli::peers::{ClientInfo, PeerBoard, Transport};
1304    ///
1305    /// let board = Arc::new(PeerBoard::new());
1306    /// # let cooperation = Cooperation::new(CooperationSetup {
1307    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
1308    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
1309    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
1310    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
1311    /// #     transport: Arc::new(HttpTransport), board: Some(Arc::clone(&board)),
1312    /// #     checkout: CheckoutFacts::default() }).unwrap();
1313    /// let worker = board.attach(Transport::Http);
1314    /// board.identify(&worker, ClientInfo { name: "claude-code".into(), version: "1".into(),
1315    ///     title: None });
1316    /// board.announce(&worker, "documenting the mesh", vec!["apps/majordomus-cli".into()]);
1317    ///
1318    /// cooperation.project_board();
1319    /// let state = cooperation.state();
1320    /// assert_eq!(state.claims[0].scope, ["apps/majordomus-cli"], "the announcement crossed");
1321    /// assert_eq!(state.sessions[0].info.client, "claude-code");
1322    ///
1323    /// // idempotent: projecting an unchanged board writes nothing
1324    /// cooperation.project_board();
1325    /// assert_eq!(cooperation.state().digest, state.digest);
1326    ///
1327    /// // and a worker that leaves takes its claim with it
1328    /// board.detach(&worker);
1329    /// cooperation.project_board();
1330    /// assert!(!cooperation.state().claims[0].state.is_live());
1331    /// ```
1332    pub fn project_board(&self) {
1333        let Some(board) = &self.board else { return };
1334        let peers = board.list();
1335        let mut projected = self.projected.lock().expect("cooperation projection");
1336        let mut seen = Vec::new();
1337        for peer in peers.iter().filter(|p| p.attached) {
1338            let session = format!("board-{}", peer.id);
1339            seen.push(session.clone());
1340            let intent = peer.claims.first().map(|c| c.intent.clone());
1341            let entry = projected.entry(session.clone()).or_default();
1342            let fresh = entry.info.session.is_empty();
1343            if fresh || entry.info.intent != intent {
1344                let info = SessionInfo {
1345                    session: session.clone(),
1346                    client: clip(&peer.client.name, 64),
1347                    worker: peer
1348                        .client
1349                        .title
1350                        .clone()
1351                        .filter(|t| !t.is_empty())
1352                        .map(|t| clip(&t, 128)),
1353                    intent: intent.clone().map(|i| clip(&i, 512)),
1354                    checkout: self.checkout.id.clone(),
1355                    branch: self.checkout.git(&["symbolic-ref", "--short", "HEAD"]),
1356                    head: self.checkout.git(&["rev-parse", "--verify", "HEAD"]),
1357                    ..SessionInfo::default()
1358                };
1359                if self.open_session(info.clone()).is_ok() {
1360                    entry.info = info;
1361                }
1362            }
1363            let mut wanted: BTreeMap<String, (Vec<String>, String)> = BTreeMap::new();
1364            for claim in &peer.claims {
1365                let scope: Vec<String> = claim
1366                    .scope
1367                    .iter()
1368                    .map(|p| p.trim().trim_start_matches("./").to_string())
1369                    .filter(|p| !p.is_empty())
1370                    .collect();
1371                if !scope.is_empty() {
1372                    wanted.insert(
1373                        claim.name.clone().unwrap_or_default(),
1374                        (scope, claim.intent.clone()),
1375                    );
1376                }
1377            }
1378            let stale: Vec<String> = entry
1379                .claims
1380                .iter()
1381                .filter(|(name, (_, scope, intent))| {
1382                    wanted
1383                        .get(*name)
1384                        .is_none_or(|(s, i)| s != scope || i != intent)
1385                })
1386                .map(|(name, _)| name.clone())
1387                .collect();
1388            for name in stale {
1389                if let Some((key, _, _)) = entry.claims.remove(&name) {
1390                    let _ = self.release(&key);
1391                }
1392            }
1393            for (name, (scope, claim_intent)) in wanted {
1394                if entry.claims.contains_key(&name) {
1395                    continue;
1396                }
1397                match self.claim(
1398                    &entry.info,
1399                    scope.clone(),
1400                    Some(clip(&claim_intent, 512)).filter(|i| !i.is_empty()),
1401                    ClaimMode::Advisory,
1402                    None,
1403                ) {
1404                    Ok(written) => {
1405                        entry
1406                            .claims
1407                            .insert(name, (written.key, scope, claim_intent));
1408                    }
1409                    Err(e) => {
1410                        tracing::debug!(runtime_id = %self.own_key, session_id = %session, error = %e, "a board announcement was not projected")
1411                    }
1412                }
1413            }
1414        }
1415        let gone: Vec<String> = projected
1416            .keys()
1417            .filter(|s| !seen.contains(s))
1418            .cloned()
1419            .collect();
1420        for session in gone {
1421            if let Some(entry) = projected.remove(&session) {
1422                for (key, _, _) in entry.claims.values() {
1423                    let _ = self.release(key);
1424                }
1425            }
1426            let _ = self.close_session(&session);
1427        }
1428    }

1454    /// Start the supervisor: the heartbeat, the dialers and the expiry. Never blocks.
1455    ///
1456    /// It returns before anything has linked, because linking depends on other machines
1457    /// and a server must answer while it is still finding them. A disabled declaration
1458    /// starts nothing at all and says so through [`Cooperation::status`], rather than
1459    /// starting a supervisor that would refuse every round.
1460    ///
1461    /// ```
1462    /// # use std::sync::Arc;
1463    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
1464    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
1465    /// # use majordomus_cli::mesh::identity::NodeIdentity;
1466    /// # use majordomus_cli::mesh::link::HttpTransport;
1467    /// # use majordomus_cli::mesh::registry::MeshRegistry;
1468    /// # use majordomus_cli::mesh::repository::of_root_commits;
1469    /// # let cooperation = Cooperation::new(CooperationSetup {
1470    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
1471    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
1472    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
1473    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
1474    /// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
1475    /// # }).unwrap();
1476    /// cooperation.start();
1477    /// assert!(!cooperation.stopped(), "the supervisor is up and the caller was not held");
1478    ///
1479    /// cooperation.stop();
1480    /// assert!(cooperation.state().digest.len() == 32, "and the journal is still readable");
1481    /// ```
1482    pub fn start(self: &Arc<Self>) {
1483        if !self.config.enabled {
1484            return;
1485        }
1486        let me = Arc::clone(self);
1487        me.counters.threads.fetch_add(1, Ordering::SeqCst);
1488        let _ = std::thread::Builder::new()
1489            .name("majordomus-mesh-cooperation".into())
1490            .spawn(move || {
1491                tracing::info!(runtime_id = %me.own_key, repository_id = %me.repository.id, "mesh cooperation started");
1492                let mut tick: u64 = 0;
1493                while !me.stop.load(Ordering::SeqCst) {
1494                    me.journal.beat_own();
1495                    me.project_board();
1496                    me.reconcile_workers();
1497                    me.expire_links();
1498                    if tick % 60 == 59 {
1499                        me.journal.compact(me.expiry(), PEER_RETENTION, false);
1500                    }
1501                    tick += 1;
1502                    me.sleep(me.heartbeat());
1503                }
1504                me.counters.threads.fetch_sub(1, Ordering::SeqCst);
1505            });
1506    }

1534    /// cooperation.claim(&session, vec!["apps".into()], None, ClaimMode::Exclusive, None).unwrap();
1535    /// cooperation.stop();
1536    ///
1537    /// assert!(cooperation.stopped());
1538    /// assert_eq!(cooperation.state().claims.len(), 1, "the record survives the runtime");
1539    /// ```
1540    ///
1541    /// [`begin_stop`](Self::begin_stop) is the half of this that takes no lock, for a caller
1542    /// that must not be delayed.
1543    pub fn stop(&self) {
1544        self.begin_stop();
1545        let mut table = self.table.lock().expect("cooperation table");
1546        for flag in table.workers.values() {
1547            flag.store(true, Ordering::SeqCst);
1548        }
1549        table.workers.clear();
1550        table.inbound_links.clear();
1551        for peer in table.peers.values_mut() {
1552            peer.outbound = None;
1553            peer.inbound = None;
1554        }
1555    }

1557    /// Tell every part of this runtime to stop, and return at once.
1558    ///
1559    /// [`stop`](Self::stop) also empties the link table, and to do that it must take the
1560    /// table's lock — which a worker can be holding across a dial that waits out the link
1561    /// timeout. A server shutting down must not wait for that: it has a lease to release,
1562    /// and a lease released late reads to the next process as a server that is still there.
1563    /// So the flag, which every loop and every handler checks, is set without any lock, and

1631    /// The trust verdict for a key: this machine's own key is trusted as itself; any
1632    /// other by the declared policy against what discovery knows of the node.
1633    ///
1634    /// It is asked twice for every event, once when a link is admitted and again when an
1635    /// event is ingested, because a link admitted an hour ago is not a warrant for what
1636    /// arrives over it now — an event carries the key that signed it, and a key that has
1637    /// since been withdrawn from the allowlist stops being believed without dropping
1638    /// anything.
1639    ///
1640    /// ```
1641    /// # use std::sync::Arc;
1642    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
1643    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
1644    /// # use majordomus_cli::mesh::identity::NodeIdentity;
1645    /// # use majordomus_cli::mesh::link::HttpTransport;
1646    /// # use majordomus_cli::mesh::registry::MeshRegistry;
1647    /// # use majordomus_cli::mesh::repository::of_root_commits;
1648    /// # let identity = Arc::new(NodeIdentity::ephemeral().unwrap());
1649    /// # let own_key = identity.public.public_key.clone();
1650    /// # let cooperation = Cooperation::new(CooperationSetup {
1651    /// #     identity, runtime: "0000000000000001".into(),
1652    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
1653    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
1654    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
1655    /// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
1656    /// # }).unwrap();
1657    /// let stranger = NodeIdentity::ephemeral().unwrap().public.public_key.clone();
1658    ///
1659    /// // two runtimes of one machine share a key, and neither has to allow the other
1660    /// assert!(cooperation.trust_of_key(&own_key).is_trusted(), "this machine, twice over");
1661    ///
1662    /// // and under the default policy an unheard-of key is not trusted by being reachable
1663    /// assert!(!cooperation.trust_of_key(&stranger).is_trusted());
1664    /// assert!(!cooperation.trust_of_key("not a key at all").is_trusted());
1665    /// ```
1666    pub fn trust_of_key(&self, public_key: &str) -> TrustState {
1667        if public_key == self.identity.public.public_key {
1668            return TrustState::Trusted("this machine's own key".into());
1669        }
1670        let Some(node) = node_id_of_key(public_key) else {
1671            return TrustState::Rejected("not a key".into());
1672        };
1673        trust::evaluate(
1674            &self.trust.policy,
1675            &self.trust.allow,
1676            public_key,
1677            self.registry.known_key(&node).as_deref(),
1678            self.registry.trust_of(&node).as_ref(),
1679        )
1680    }

1684            Ok(())
1685        } else {
1686            Err(Rejection::Untrusted)
1687        }

1719    /// Answer a hello: the whole admission, in order — shape, signature, freshness,
1720    /// replay, protocol, self, repository, trust, capacity — and a signed welcome.
1721    ///
1722    /// The order is the point. Everything cheap and unauthenticated is decided before
1723    /// anything expensive, and nothing a stranger says is believed before its signature is
1724    /// checked, so an unsigned flood costs a parse and a refusal. It answers a refusal
1725    /// rather than dropping the message, because a peer that is being refused for its
1726    /// clock, its repository or its key can only fix that if it is told.
1727    ///
1728    /// ```
1729    /// # use std::collections::BTreeMap;
1730    /// # use std::sync::{Arc, Mutex, Weak};
1731    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
1732    /// # use majordomus_cli::mesh::cooperation::*;
1733    /// # use majordomus_cli::mesh::identity::NodeIdentity;
1734    /// # use majordomus_cli::mesh::link::{LinkReply, LinkTransport, RefusalCode, Signed, HELLO_PATH};
1735    /// # use majordomus_cli::mesh::registry::MeshRegistry;
1736    /// # use majordomus_cli::mesh::repository::{of_root_commits, runtime_id};
1737    /// # use majordomus_cli::mesh::trust::TrustPolicy;
1738    /// # #[derive(Default)] struct Net(Mutex<BTreeMap<String, Weak<Cooperation>>>);
1739    /// # impl LinkTransport for Net {
1740    /// #   fn post(&self, e: &str, p: &str, m: &Signed) -> Result<LinkReply, String> {
1741    /// #     let peer = (self.0.lock().unwrap().get(e).and_then(Weak::upgrade))
1742    /// #         .ok_or_else(|| format!("{e}: nobody there"))?;
1743    /// #     Ok(if p == HELLO_PATH { peer.accept_hello(m) } else { peer.accept_sync(m) }) } }
1744    /// # let net = Arc::new(Net::default());
1745    /// # let runtime = |endpoint: &str| { let c = Cooperation::new(CooperationSetup {
1746    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: runtime_id(endpoint),
1747    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec![endpoint.into()],
1748    /// #     version: "doc".into(), config: CooperationConfig::default(),
1749    /// #     trust: TrustConfig { policy: TrustPolicy::Tofu, allow: vec![] }, journal_path: None,
1750    /// #     registry: Arc::new(MeshRegistry::new()),
1751    /// #     transport: Arc::clone(&net) as Arc<dyn LinkTransport>, board: None,
1752    /// #     checkout: CheckoutFacts::default() }).unwrap();
1753    /// #   net.0.lock().unwrap().insert(endpoint.into(), Arc::downgrade(&c)); c };
1754    /// # let (a, b) = (runtime("a:1"), runtime("b:1"));
1755    /// // A dials; the transport hands B's accept_hello the signed hello, and B welcomes it
1756    /// a.dial(&["b:1".into()]).unwrap();
1757    /// assert_eq!(b.peers()[0].runtime, a.runtime_key(), "admitted, both ways");
1758    ///
1759    /// // anything that is not a signed hello of this protocol is refused, and named
1760    /// let nonsense = Signed { body: serde_json::json!({"hello": true}), sig: "00".into() };
1761    /// let refusal = b.accept_hello(&nonsense).refusal.unwrap();
1762    /// assert_eq!(refusal.code, RefusalCode::Malformed);
1763    /// ```
1764    pub fn accept_hello(&self, message: &Signed) -> LinkReply {
1765        if self.stopped() {
1766            return LinkReply::refused(RefusalCode::NotActive, "cooperation is stopped");
1767        }
1768        if serde_json::to_vec(message)
1769            .map(|b| b.len())
1770            .unwrap_or(usize::MAX)
1771            > MAX_LINK_MESSAGE
1772        {
1773            return self.refuse_in(
1774                RefusalCode::Oversized,
1775                "the hello exceeds the message bound".into(),
1776                None,
1777            );
1778        }
1779        let hello: Hello = match serde_json::from_value(message.body.clone()) {
1780            Ok(h) => h,
1781            Err(e) => {
1782                return self.refuse_in(RefusalCode::Malformed, format!("not a hello: {e}"), None)
1783            }
1784        };
1785        if let Err(e) = hello.card.check() {
1786            return self.refuse_in(RefusalCode::Malformed, e, None);
1787        }
1788        let runtime = hello.card.runtime_key();
1789        if !verify_signed(&hello.card.pk, Domain::Hello, message) {
1790            return self.refuse_in(
1791                RefusalCode::Signature,
1792                "the hello's signature does not verify".into(),
1793                runtime,
1794            );
1795        }
1796        if hello.ts.abs_diff(super::protocol::now()) > MAX_LINK_SKEW_SECONDS {
1797            return self.refuse_in(
1798                RefusalCode::Stale,
1799                "the hello's clock is outside the skew window".into(),
1800                runtime,
1801            );
1802        }
1803        if hello.nonce.len() != 32 {
1804            return self.refuse_in(
1805                RefusalCode::Malformed,
1806                "the nonce is not 32 hex characters".into(),
1807                runtime,
1808            );
1809        }
1810        let Some(proto) = negotiate(hello.proto_min, hello.proto_max) else {
1811            return self.refuse_in(
1812                RefusalCode::ProtocolUnsupported,
1813                format!(
1814                    "this runtime speaks link protocol {LINK_PROTOCOL_MIN}..{LINK_PROTOCOL_MAX}; the peer speaks {}..{}",
1815                    hello.proto_min, hello.proto_max
1816                ),
1817                runtime,
1818            );
1819        };
1820        let runtime_key = runtime.clone().unwrap_or_default();
1821        if runtime_key == self.own_key {
1822            return self.refuse_in(
1823                RefusalCode::SelfLink,
1824                "that is this runtime".into(),
1825                runtime,
1826            );
1827        }
1828        if hello.card.repo != self.repository.id {
1829            return self.refuse_in(
1830                RefusalCode::RepositoryMismatch,
1831                format!(
1832                    "this runtime serves repository {}; the peer serves {}",
1833                    self.repository.id, hello.card.repo
1834                ),
1835                runtime,
1836            );
1837        }
1838        let verdict = self.trust_of_key(&hello.card.pk);
1839        if !verdict.is_trusted() {
1840            return self.refuse_in(
1841                RefusalCode::Untrusted,
1842                format!("the peer's key is not trusted here ({verdict:?}): discovery is not trust"),
1843                runtime,
1844            );
1845        }
1846        // The replay cache holds only nonces of trusted keys of this repository: a stranger's
1847        // hellos, however many, can neither fill it nor evict a real peer's nonce.
1848        {
1849            let mut table = self.table.lock().expect("cooperation table");
1850            let now = super::protocol::now();
1851            table
1852                .nonces
1853                .retain(|_, ts| now.saturating_sub(*ts) <= 2 * MAX_LINK_SKEW_SECONDS);
1854            let key = (hello.card.pk.clone(), hello.nonce.clone());
1855            if table.nonces.contains_key(&key) {
1856                drop(table);
1857                return self.refuse_in(
1858                    RefusalCode::Replay,
1859                    "this hello was already answered".into(),
1860                    runtime,
1861                );
1862            }
1863            if table.nonces.len() >= NONCE_CACHE {
1864                if let Some(oldest) = table
1865                    .nonces
1866                    .iter()
1867                    .min_by_key(|(_, ts)| **ts)
1868                    .map(|(k, _)| k.clone())
1869                {
1870                    table.nonces.remove(&oldest);
1871                }
1872            }
1873            table.nonces.insert(key, now);
1874        }
1875        let link = fresh_token();
1876        {
1877            let mut table = self.table.lock().expect("cooperation table");
1878            if !table.peers.contains_key(&runtime_key) && table.peers.len() >= MAX_PEERS {
1879                drop(table);
1880                return self.refuse_in(
1881                    RefusalCode::Capacity,
1882                    "the peer table is full".into(),
1883                    runtime,
1884                );
1885            }
1886            let restarts = &self.counters.restarts;
1887            let reconnects = &self.counters.reconnects;
1888            let peer = table
1889                .peers
1890                .entry(runtime_key.clone())
1891                .or_insert_with(|| PeerLink {
1892                    card: hello.card.clone(),
1893                    trust: verdict.clone(),
1894                    proto,
1895                    outbound: None,
1896                    inbound: None,
1897                    last_exchange: None,
1898                    linked_at: rfc3339_now(),
1899                    rtt: None,
1900                    failures: 0,
1901                    last_error: None,
1902                    handshakes: 0,
1903                    reconnects: 0,
1904                    restarts: 0,
1905                    expired: false,
1906                });
1907            if peer.card.instance != hello.card.instance {
1908                // A new run of the same runtime: its old links are dead on both sides.
1909                peer.restarts += 1;
1910                restarts.fetch_add(1, Ordering::Relaxed);
1911                peer.outbound = None;
1912                peer.linked_at = rfc3339_now();
1913            } else if peer.expired || (peer.handshakes > 0 && peer.inbound.is_some()) {
1914                peer.reconnects += 1;
1915                reconnects.fetch_add(1, Ordering::Relaxed);
1916            }
1917            if let Some(old) = peer.inbound.take() {
1918                table.inbound_links.remove(&old.link);
1919            }
1920            let peer = table.peers.get_mut(&runtime_key).expect("inserted above");
1921            peer.card = hello.card.clone();
1922            peer.trust = verdict;
1923            peer.proto = proto;
1924            peer.inbound = Some(Inbound {
1925                link: link.clone(),
1926                counter: 0,
1927            });
1928            peer.last_exchange = Some(Instant::now());
1929            peer.handshakes += 1;
1930            peer.expired = false;
1931            table
1932                .inbound_links
1933                .insert(link.clone(), runtime_key.clone());
1934            table.refused.remove(&format!("inbound:{runtime_key}"));
1935        }
1936        self.counters.handshakes_in.fetch_add(1, Ordering::Relaxed);
1937        tracing::info!(runtime_id = %self.own_key, peer = %runtime_key, link_id = %link, "mesh link accepted");
1938        let welcome = Welcome {
1939            proto,
1940            link,
1941            card: self.card.clone(),
1942            nonce: hello.nonce,
1943            ts: super::protocol::now(),
1944            marks: self.journal.marks(),
1945            heartbeat_seconds: self.config.heartbeat_seconds,
1946            expiry_seconds: self.config.expiry_seconds,
1947        };
1948        match serde_json::to_value(&welcome) {
1949            Ok(body) => LinkReply::accepted(sign(&self.identity, Domain::Welcome, body)),
1950            Err(e) => LinkReply::refused(RefusalCode::Malformed, e.to_string()),
1951        }
1952    }

1954    /// Answer a sync round: the link must be known and its peer's instance current, the
1955    /// signature must be the peer's and the counter must rise; then ingest, merge, and
1956    /// answer with what the peer lacks.
1957    ///
1958    /// The rising counter is what makes a captured round worthless: a recorded request
1959    /// replayed later carries a counter that has already been used, and is refused without
1960    /// the events in it being looked at. Answering with what the peer lacks in the same
1961    /// round is what makes the exchange symmetrical — one call, both directions — so
1962    /// convergence does not depend on which side dials.
1963    ///
1964    /// ```
1965    /// # use std::collections::BTreeMap;
1966    /// # use std::sync::{Arc, Mutex, Weak};
1967    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
1968    /// # use majordomus_cli::mesh::cooperation::*;
1969    /// # use majordomus_cli::mesh::identity::NodeIdentity;
1970    /// # use majordomus_cli::mesh::journal::{ClaimMode, SessionInfo};
1971    /// # use majordomus_cli::mesh::link::{LinkReply, LinkTransport, RefusalCode, Signed, HELLO_PATH};
1972    /// # use majordomus_cli::mesh::registry::MeshRegistry;
1973    /// # use majordomus_cli::mesh::repository::{of_root_commits, runtime_id};
1974    /// # use majordomus_cli::mesh::trust::TrustPolicy;
1975    /// # #[derive(Default)] struct Net(Mutex<BTreeMap<String, Weak<Cooperation>>>);
1976    /// # impl LinkTransport for Net {
1977    /// #   fn post(&self, e: &str, p: &str, m: &Signed) -> Result<LinkReply, String> {
1978    /// #     let peer = (self.0.lock().unwrap().get(e).and_then(Weak::upgrade))
1979    /// #         .ok_or_else(|| format!("{e}: nobody there"))?;
1980    /// #     Ok(if p == HELLO_PATH { peer.accept_hello(m) } else { peer.accept_sync(m) }) } }
1981    /// # let net = Arc::new(Net::default());
1982    /// # let runtime = |endpoint: &str| { let c = Cooperation::new(CooperationSetup {
1983    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: runtime_id(endpoint),
1984    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec![endpoint.into()],
1985    /// #     version: "doc".into(), config: CooperationConfig::default(),
1986    /// #     trust: TrustConfig { policy: TrustPolicy::Tofu, allow: vec![] }, journal_path: None,
1987    /// #     registry: Arc::new(MeshRegistry::new()),
1988    /// #     transport: Arc::clone(&net) as Arc<dyn LinkTransport>, board: None,
1989    /// #     checkout: CheckoutFacts::default() }).unwrap();
1990    /// #   net.0.lock().unwrap().insert(endpoint.into(), Arc::downgrade(&c)); c };
1991    /// # let (a, b) = (runtime("a:1"), runtime("b:1"));
1992    /// let key = a.dial(&["b:1".into()]).unwrap();
1993    /// b.claim(&SessionInfo::named("s1", "cli"), vec!["docs".into()], None,
1994    ///     ClaimMode::Advisory, None).unwrap();
1995    ///
1996    /// // A asks; B's answer carries B's own events back, so one round settles both sides
1997    /// a.sync_with(&key).unwrap();
1998    /// assert_eq!(a.state().claims.len(), 1, "B answered with what A lacked");
1999    /// assert_eq!(b.status().counters.syncs_in, 1);
2000    ///
2001    /// // a round under no link this runtime issued is refused before its events are read
2002    /// let stranger = Signed { body: serde_json::json!({"link": "ff", "counter": 1}), sig: "".into() };
2003    /// assert_eq!(b.accept_sync(&stranger).refusal.unwrap().code, RefusalCode::Malformed);
2004    /// ```
2005    pub fn accept_sync(&self, message: &Signed) -> LinkReply {
2006        if self.stopped() {
2007            return LinkReply::refused(RefusalCode::NotActive, "cooperation is stopped");
2008        }
2009        if serde_json::to_vec(message)
2010            .map(|b| b.len())
2011            .unwrap_or(usize::MAX)
2012            > MAX_LINK_MESSAGE
2013        {
2014            return self.refuse_in(
2015                RefusalCode::Oversized,
2016                "the sync exceeds the message bound".into(),
2017                None,
2018            );
2019        }
2020        let request: SyncRequest = match serde_json::from_value(message.body.clone()) {
2021            Ok(r) => r,
2022            Err(e) => {
2023                return self.refuse_in(
2024                    RefusalCode::Malformed,
2025                    format!("not a sync request: {e}"),
2026                    None,
2027                )
2028            }
2029        };

2035                return LinkReply::refused(
2036                    RefusalCode::UnknownLink,
2037                    "no such link here: say hello",
2038                );
2039            };

2041                .peers
2042                .get_mut(&key)

2047                return LinkReply::refused(
2048                    RefusalCode::UnknownLink,
2049                    "the link belongs to a previous instance: say hello",
2050                );
2051            }

2058                );
2059            }

2067                );
2068            }

2073        };
2074        let report = self
2075            .journal
2076            .ingest(&request.events, &|e| self.origin_accept(e));
2077        self.journal.merge_marks(
2078            &request.marks,
2079            &|pk| self.trust_of_key(pk).is_trusted(),
2080            self.expiry(),
2081        );
2082        let events = self.journal.missing_for(&request.marks, SYNC_EVENT_BUDGET);
2083        self.counters.syncs_in.fetch_add(1, Ordering::Relaxed);
2084        self.counters
2085            .events_served
2086            .fetch_add(events.len() as u64, Ordering::Relaxed);
2087        if report.accepted > 0 || report.rejected_total() > 0 {
2088            tracing::debug!(runtime_id = %self.own_key, peer = %runtime_key, accepted = report.accepted, duplicate = report.duplicate, rejected = report.rejected_total(), "mesh sync ingested");
2089        }
2090        let answer = SyncAnswer {
2091            link: request.link,
2092            counter: request.counter,
2093            marks: self.journal.marks(),
2094            events,
2095            report,
2096        };
2097        match serde_json::to_value(&answer) {
2098            Ok(body) => LinkReply::accepted(sign(&self.identity, Domain::SyncAnswer, body)),
2099            Err(e) => LinkReply::refused(RefusalCode::Malformed, e.to_string()),
2100        }
2101    }

2127    /// welcome as strictly as a hello is verified. Returns the peer's runtime key.

2326    /// One sync round with a peer this runtime dials: push what the peer lacks, take what
2327    /// this runtime lacks, in one exchange.
2328    ///
2329    /// Only the dialing side runs rounds, so a link has one driver and one counter however
2330    /// many threads hold it; an inbound-only peer is synced by its own side. A round is
2331    /// bounded by an event budget rather than by time, so a runtime that has been away for
2332    /// a day catches up over several rounds instead of one that never finishes.
2333    ///
2334    /// ```
2335    /// # use std::collections::BTreeMap;
2336    /// # use std::sync::{Arc, Mutex, Weak};
2337    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
2338    /// # use majordomus_cli::mesh::cooperation::*;
2339    /// # use majordomus_cli::mesh::identity::NodeIdentity;
2340    /// # use majordomus_cli::mesh::journal::{ClaimMode, SessionInfo};
2341    /// # use majordomus_cli::mesh::link::{LinkReply, LinkTransport, Signed, HELLO_PATH};
2342    /// # use majordomus_cli::mesh::registry::MeshRegistry;
2343    /// # use majordomus_cli::mesh::repository::{of_root_commits, runtime_id};
2344    /// # use majordomus_cli::mesh::trust::TrustPolicy;
2345    /// # #[derive(Default)] struct Net(Mutex<BTreeMap<String, Weak<Cooperation>>>);
2346    /// # impl LinkTransport for Net {
2347    /// #   fn post(&self, e: &str, p: &str, m: &Signed) -> Result<LinkReply, String> {
2348    /// #     let peer = (self.0.lock().unwrap().get(e).and_then(Weak::upgrade))
2349    /// #         .ok_or_else(|| format!("{e}: nobody there"))?;
2350    /// #     Ok(if p == HELLO_PATH { peer.accept_hello(m) } else { peer.accept_sync(m) }) } }
2351    /// # let net = Arc::new(Net::default());
2352    /// # let runtime = |endpoint: &str| { let c = Cooperation::new(CooperationSetup {
2353    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: runtime_id(endpoint),
2354    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec![endpoint.into()],
2355    /// #     version: "doc".into(), config: CooperationConfig::default(),
2356    /// #     trust: TrustConfig { policy: TrustPolicy::Tofu, allow: vec![] }, journal_path: None,
2357    /// #     registry: Arc::new(MeshRegistry::new()),
2358    /// #     transport: Arc::clone(&net) as Arc<dyn LinkTransport>, board: None,
2359    /// #     checkout: CheckoutFacts::default() }).unwrap();
2360    /// #   net.0.lock().unwrap().insert(endpoint.into(), Arc::downgrade(&c)); c };
2361    /// # let (a, b) = (runtime("a:1"), runtime("b:1"));
2362    /// let key = a.dial(&["b:1".into()]).unwrap();
2363    /// a.claim(&SessionInfo::named("s1", "cli"), vec!["apps".into()], None,
2364    ///     ClaimMode::Exclusive, None).unwrap();
2365    ///
2366    /// for _ in 0..2 {
2367    ///     a.journal().beat_own();
2368    ///     a.sync_with(&key).unwrap();
2369    /// }
2370    /// assert_eq!(a.state().digest, b.state().digest, "the two now hold one state");
2371    ///
2372    /// // B's own claims come back over the same rounds, without B dialing anybody
2373    /// b.claim(&SessionInfo::named("s2", "cli"), vec!["docs".into()], None,
2374    ///     ClaimMode::Exclusive, None).unwrap();
2375    /// b.journal().beat_own();
2376    /// a.sync_with(&key).unwrap();
2377    /// assert_eq!(a.state().claims.len(), 2);
2378    /// ```
2379    pub fn sync_with(&self, runtime_key: &str) -> Result<(), RoundError> {
2380        // One round per link at a time: a verification and a worker (or two workers for one
2381        // peer) racing on the counter would refuse each other's rounds as replays.
2382        let round = {
2383            let mut rounds = self.rounds.lock().expect("sync rounds");
2384            Arc::clone(rounds.entry(runtime_key.to_string()).or_default())
2385        };
2386        let _round = round.lock().expect("sync round");
2387        let (out, pk) = {
2388            let mut table = self.table.lock().expect("cooperation table");
2389            let Some(peer) = table.peers.get_mut(runtime_key) else {
2390                return Err(RoundError::NoLink);
2391            };
2392            let Some(out) = peer.outbound.as_mut() else {
2393                return Err(RoundError::NoLink);
2394            };
2395            out.counter += 1;
2396            (out.clone(), peer.card.pk.clone())
2397        };
2398        let request = SyncRequest {
2399            link: out.link.clone(),
2400            counter: out.counter,
2401            instance: self.card.instance.clone(),
2402            marks: self.journal.marks(),
2403            events: self
2404                .journal
2405                .missing_for(&out.remote_marks, SYNC_EVENT_BUDGET),
2406        };
2407        let sent = request.events.len() as u64;
2408        let body =
2409            serde_json::to_value(&request).map_err(|e| RoundError::Unreachable(e.to_string()))?;
2410        let message = sign(&self.identity, Domain::SyncRequest, body);
2411        // The push is counted as the request leaves, not when its answer returns. The peer
2412        // holds these events from the moment it ingests them, before its answer is written,
2413        // so a count taken on the answer trails every reading of the peer: in between, the
2414        // events are held there and not yet sent here, and the count that lands a moment
2415        // later reads as an event that travelled twice. The peer counts what it serves the
2416        // same way, as it forms the answer. A round that fails after this has still put its
2417        // events on the wire, and a round that sends them again counts them again.
2418        self.counters.events_sent.fetch_add(sent, Ordering::Relaxed);
2419        let started = Instant::now();
2420        let failed = |why: String| {
2421            self.counters.syncs_failed.fetch_add(1, Ordering::Relaxed);
2422            let mut table = self.table.lock().expect("cooperation table");
2423            if let Some(peer) = table.peers.get_mut(runtime_key) {
2424                peer.failures += 1;
2425                peer.last_error = Some(why.clone());
2426            }
2427            why
2428        };
2429        let reply = match self.transport.post(&out.endpoint, SYNC_PATH, &message) {
2430            Ok(reply) => reply,
2431            Err(e) => return Err(RoundError::Unreachable(failed(e))),
2432        };
2433        if let Some(refusal) = reply.refusal {
2434            failed(refusal.to_string());
2435            if refusal.code == RefusalCode::UnknownLink {
2436                let mut table = self.table.lock().expect("cooperation table");
2437                if let Some(peer) = table.peers.get_mut(runtime_key) {
2438                    peer.outbound = None;
2439                }
2440            }
2441            Self::count_refusal(&self.counters.refused_out, refusal.code);
2442            return Err(RoundError::Refused(refusal));
2443        }
2444        let Some(signed) = reply.signed else {
2445            return Err(RoundError::Unreachable(failed(
2446                "an empty sync reply".into(),
2447            )));
2448        };
2449        let answer: SyncAnswer = match serde_json::from_value(signed.body.clone()) {
2450            Ok(a) => a,
2451            Err(e) => {
2452                return Err(RoundError::Unreachable(failed(format!(
2453                    "not a sync answer: {e}"
2454                ))))
2455            }
2456        };
2457        if !verify_signed(&pk, Domain::SyncAnswer, &signed)
2458            || answer.link != out.link
2459            || answer.counter != out.counter
2460        {
2461            return Err(RoundError::Refused(LinkRefusal::new(
2462                RefusalCode::Signature,
2463                failed("the sync answer is not the linked peer's answer to this round".into()),
2464            )));
2465        }
2466        self.journal
2467            .ingest(&answer.events, &|e| self.origin_accept(e));
2468        self.journal.merge_marks(
2469            &answer.marks,
2470            &|pk| self.trust_of_key(pk).is_trusted(),
2471            self.expiry(),
2472        );
2473        {
2474            let mut table = self.table.lock().expect("cooperation table");
2475            if let Some(peer) = table.peers.get_mut(runtime_key) {
2476                if let Some(o) = peer.outbound.as_mut() {
2477                    o.remote_marks = answer.marks;
2478                }
2479                peer.last_exchange = Some(Instant::now());
2480                peer.rtt = Some(started.elapsed());
2481                peer.failures = 0;
2482                peer.last_error = None;
2483                peer.expired = false;
2484            }
2485        }
2486        self.counters.syncs_out.fetch_add(1, Ordering::Relaxed);
2487        Ok(())
2488    }

2540    /// Whether this runtime should dial a known peer: the lower runtime key dials, and
2541    /// the higher one dials only when nothing has arrived from the peer lately — so a
2542    /// pair keeps one link when both can reach each other, and two when only one can.
2543    fn should_dial(&self, key: &str) -> bool {
2544        if self.own_key.as_str() < key {
2545            return true;
2546        }
2547        let table = self.table.lock().expect("cooperation table");

2559    fn work(&self, target: Target, flag: Arc<AtomicBool>) {
2560        let mut learned = target.key.clone();
2561        let mut failures: u32 = 0;
2562        while !self.stopped() && !flag.load(Ordering::SeqCst) {
2563            let linked = learned
2564                .as_deref()
2565                .filter(|k| self.outbound_key(k))
2566                .map(str::to_string);
2567            let outcome = match linked {
2568                Some(key) => self.sync_with(&key),
2569                None => {
2570                    let may = target.key.as_deref().is_none_or(|k| self.should_dial(k));
2571                    if !may {
2572                        self.sleep(self.heartbeat());
2573                        continue;
2574                    }
2575                    match self.dial(&target.endpoints) {
2576                        Ok(key) => {
2577                            learned = Some(key.clone());
2578                            // A fresh link syncs at once: the peer should see this
2579                            // runtime's state now, not a heartbeat from now.
2580                            self.sync_with(&key)
2581                        }
2582                        Err(e) => Err(e),
2583                    }
2584                }
2585            };
2586            let delay = match outcome {
2587                Ok(()) => {
2588                    failures = 0;
2589                    self.heartbeat()
2590                }
2591                Err(RoundError::Refused(r)) if r.code == RefusalCode::UnknownLink => {
2592                    failures = 0;
2593                    Duration::ZERO
2594                }
2595                Err(RoundError::Refused(_)) => {
2596                    failures = failures.saturating_add(3).min(6);
2597                    self.backoff(failures)
2598                }
2599                Err(_) => {
2600                    failures = failures.saturating_add(1).min(6);
2601                    self.backoff(failures)
2602                }
2603            };
2604            self.sleep(delay);
2605        }
2606    }

2608    /// Heartbeat × 2^failures, never beyond the expiry: an unreachable peer is asked less
2609    /// and less, and still within one expiry of coming back.
2610    fn backoff(&self, failures: u32) -> Duration {
2611        let factor = 1u32 << failures.min(6);
2612        (self.heartbeat() * factor).min(self.expiry().max(self.heartbeat()))
2613    }

2615    fn expire_links(&self) {
2616        let expiry = self.expiry();
2617        let mut table = self.table.lock().expect("cooperation table");
2618        let mut dropped_links = Vec::new();
2619        let mut gone = Vec::new();
2620        for (key, peer) in table.peers.iter_mut() {
2621            let age = peer.last_exchange.map(|t| t.elapsed());
2622            if age.is_some_and(|a| a > expiry) && !peer.expired {
2623                peer.expired = true;
2624                if let Some(inbound) = peer.inbound.take() {
2625                    dropped_links.push(inbound.link);
2626                }
2627                peer.outbound = None;
2628                self.counters.peers_expired.fetch_add(1, Ordering::Relaxed);
2629                tracing::warn!(runtime_id = %self.own_key, peer = %key, "mesh link expired: no exchange within the expiry");
2630            }
2631            if age.is_some_and(|a| a > expiry + PEER_RETENTION) {
2632                gone.push(key.clone());
2633            }
2634        }
2635        for link in dropped_links {
2636            table.inbound_links.remove(&link);
2637        }
2638        for key in gone {
2639            table.peers.remove(&key);
2640        }
2641    }

2643    fn state_of(&self, peer: &PeerLink) -> LinkState {
2644        let Some(age) = peer.last_exchange.map(|t| t.elapsed()) else {
2645            return LinkState::Connecting;
2646        };
2647        let expiry = self.expiry();

2649            LinkState::Expired

2651            LinkState::Connected
2652        } else if age <= expiry / 2 {
2653            LinkState::Degraded
2654        } else {
2655            LinkState::Unreachable
2656        }

2828    /// hold the same events and agree about who is alive fold the same state, which is what

3492    /// reviewer is checked here rather than left to fail silently later: a request
3493    /// addressed to a runtime that is not linked, or to one whose executable does not do
3494    /// reviews, would otherwise sit unanswered and look like indifference.
3495    ///
3496    /// ```
3497    /// # use std::sync::Arc;
3498    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
3499    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
3500    /// # use majordomus_cli::mesh::identity::NodeIdentity;
3501    /// # use majordomus_cli::mesh::link::HttpTransport;
3502    /// # use majordomus_cli::mesh::registry::MeshRegistry;
3503    /// # use majordomus_cli::mesh::repository::of_root_commits;
3504    /// # let cooperation = Cooperation::new(CooperationSetup {
3505    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
3506    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
3507    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
3508    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
3509    /// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
3510    /// # }).unwrap();
3511    /// use majordomus_cli::mesh::state::ReviewState;
3512    ///
3513    /// cooperation.request_review("s1", "feature/mesh-cooperation".into(),
3514    ///     vec!["apps/majordomus-cli/src/mesh".into()], Some("#184".into()), None).unwrap();
3515    ///
3516    /// let review = &cooperation.state().reviews[0];
3517    /// assert_eq!(review.subject, "feature/mesh-cooperation");
3518    /// assert_eq!(review.state, ReviewState::Open, "asked of anybody, answered by nobody yet");
3519    ///
3520    /// // a reviewer nobody is linked to is refused now, not left waiting
3521    /// cooperation.request_review("s1", "feature/x".into(), vec![], None,
3522    ///     Some("somebody-0000000000000002".into())).expect_err("no such peer");
3523    /// ```
3524    pub fn request_review(
3525        &self,
3526        session: &str,
3527        subject: String,
3528        scope: Vec<String>,
3529        issue: Option<String>,
3530        reviewer: Option<String>,
3531    ) -> Result<Written, CooperationError> {
3532        if let Some(target) = &reviewer {
3533            if *target != self.own_key {
3534                let table = self.table.lock().expect("cooperation table");
3535                match table.peers.get(target) {
3536                    None => {
3537                        return Err(CooperationError::NotFound(format!(
3538                            "no linked peer {target}"
3539                        )))
3540                    }
3541                    Some(peer) if !peer.card.supports("reviews") => {
3542                        return Err(CooperationError::FeatureUnsupported(format!(
3543                            "{target} does not carry the reviews feature (it carries: {})",
3544                            peer.card.features.join(", ")
3545                        )))
3546                    }
3547                    Some(_) => {}
3548                }
3549            }
3550        }
3551        let review = format!("r-{}", &fresh_token()[..12]);
3552        self.write(
3553            EventBody::ReviewRequested {
3554                review: review.clone(),
3555                session: session.into(),
3556                subject,
3557                scope,
3558                issue,
3559                reviewer,
3560            },
3561            self.own(&review),
3562        )
3563    }

3566    ///
3567    /// The one refusal here is self-review, because a verdict from the session under review
3568    /// is not a second pair of eyes and a record that allowed it would make every review in
3569    /// the mesh worth less. Several sessions may answer one request and their verdicts all
3570    /// stand: the disagreement is the information.
3571    ///
3572    /// ```
3573    /// # use std::sync::Arc;
3574    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
3575    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
3576    /// # use majordomus_cli::mesh::identity::NodeIdentity;
3577    /// # use majordomus_cli::mesh::link::HttpTransport;
3578    /// # use majordomus_cli::mesh::registry::MeshRegistry;
3579    /// # use majordomus_cli::mesh::repository::of_root_commits;
3580    /// # let cooperation = Cooperation::new(CooperationSetup {
3581    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
3582    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
3583    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
3584    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
3585    /// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
3586    /// # }).unwrap();
3587    /// use majordomus_cli::mesh::state::ReviewState;
3588    ///
3589    /// let asked = cooperation
3590    ///     .request_review("s1", "feature/mesh-cooperation".into(), vec![], None, None)
3591    ///     .unwrap();
3592    ///
3593    /// cooperation.answer_review(&asked.key, "s2", "approved".into(),
3594    ///     Some("the examples run".into())).unwrap();
3595    /// let review = &cooperation.state().reviews[0];
3596    /// assert_eq!(review.state, ReviewState::Answered);
3597    /// assert_eq!(review.answers[0].verdict, "approved");
3598    ///
3599    /// // the session under review does not get a vote on itself
3600    /// cooperation.answer_review(&asked.key, "s1", "approved".into(), None)
3601    ///     .expect_err("a review is another session's verdict");
3602    /// ```
3603    pub fn answer_review(
3604        &self,
3605        request: &str,
3606        session: &str,
3607        verdict: String,
3608        note: Option<String>,
3609    ) -> Result<Written, CooperationError> {
3610        let state = self.state();
3611        let Some(review) = state.reviews.iter().find(|r| r.key == request) else {
3612            return Err(CooperationError::NotFound(format!(
3613                "no review request {request}"
3614            )));
3615        };
3616        if review.session == self.own(session) {
3617            return Err(CooperationError::Invalid(format!(
3618                "session {session} requested review {request} and cannot answer it: a review is another session's verdict"
3619            )));
3620        }
3621        self.write(
3622            EventBody::ReviewAnswered {
3623                request: request.into(),
3624                session: session.into(),
3625                verdict,
3626                note,
3627            },
3628            request.into(),
3629        )
3630    }
