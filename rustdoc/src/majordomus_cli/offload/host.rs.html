1//! Which declared host a derive goes to: of the hosts that answer and have at least their
2//! free-space minimum, the one with the fewest derives running or waiting; a tie goes to the
3//! host declared first.
4
5use serde::Serialize;
6
7use super::declaration::{OffloadDeclaration, OffloadHost};
8use super::queue::{Observation, Queue};
9use super::remote::Remote;
10
11/// Where a host stands for the next derive.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
13#[serde(rename_all = "snake_case")]
14pub enum Standing {
15    /// It answers and has the space: it may be chosen.
16    Eligible,
17    /// It did not answer, or answered something the queue cannot read.
18    Unreachable,
19    /// It has less free space than its minimum.
20    UnderMinimum,
21    /// It answered without a free-space figure, and unknown is never enough.
22    UnknownSpace,
23}
24
25/// One host's measurement and verdict.
26#[derive(Debug, Clone, Serialize)]
27pub struct HostReport {
28    /// The declared id.
29    pub id: String,
30    /// `user@host`.
31    pub address: String,
32    /// The minimum that applies to it, GiB.
33    pub min_free_gib: u64,
34    /// GiB free, when measured.
35    #[serde(skip_serializing_if = "Option::is_none")]
36    pub free_gib: Option<u64>,
37    /// Derives running or waiting, when measured.
38    #[serde(skip_serializing_if = "Option::is_none")]
39    pub load: Option<usize>,
40    /// The verdict.
41    pub standing: Standing,
42    /// Why, in words, when it is not eligible.
43    #[serde(skip_serializing_if = "Option::is_none")]
44    pub reason: Option<String>,
45    /// Whether this is the host a derive would take.
46    pub chosen: bool,
47}
48
49/// The remote a declared host is reached by.
50pub fn remote_of(host: &OffloadHost) -> Remote {
51    Remote::ssh(&host.address, host.identity_file.as_deref(), &host.id)
52}
53
54/// Judge every host from its observation (in declaration order) and mark the one chosen.
55pub fn choose(
56    d: &OffloadDeclaration,
57    observations: Vec<Result<Observation, String>>,
58) -> Vec<HostReport> {
59    let mut reports: Vec<HostReport> = d
60        .hosts
61        .iter()
62        .zip(observations)
63        .map(|(h, o)| {
64            let min = h.min_free_gib(d);
65            let mut r = HostReport {
66                id: h.id.clone(),
67                address: h.address.clone(),
68                min_free_gib: min,
69                free_gib: None,
70                load: None,
71                standing: Standing::Eligible,
72                reason: None,
73                chosen: false,
74            };
75            match o {
76                Err(e) => {
77                    r.standing = Standing::Unreachable;
78                    r.reason = Some(e);
79                }
80                Ok(o) => {
81                    r.free_gib = o.free_gib();
82                    r.load = Some(o.load());
83                    match r.free_gib {
84                        None => {
85                            r.standing = Standing::UnknownSpace;
86                            r.reason = Some("df reported no free space figure".into());
87                        }
88                        Some(f) if f < min => {
89                            r.standing = Standing::UnderMinimum;
90                            r.reason = Some(format!("{f} GiB free, under {min}"));
91                        }
92                        Some(_) => {}
93                    }
94                }
95            }
96            r
97        })
98        .collect();
99    let best = reports
100        .iter()
101        .enumerate()
102        .filter(|(_, r)| r.standing == Standing::Eligible)
103        .min_by_key(|(i, r)| (r.load.unwrap_or(usize::MAX), *i))
104        .map(|(i, _)| i);
105    if let Some(i) = best {
106        reports[i].chosen = true;
107    }
108    reports
109}

111/// Observe every declared host, one after another, and judge them.
112pub fn probe(d: &OffloadDeclaration) -> Vec<HostReport> {
113    let observations = d
114        .hosts
115        .iter()
116        .map(|h| {
117            let remote = remote_of(h);
118            Queue {
119                remote: &remote,
120                directory: &d.queue.directory,
121                lock: &d.queue.lock,
122                poll: std::time::Duration::from_secs(d.queue.poll_seconds),
123                wait_max: std::time::Duration::from_secs(d.queue.wait_max_seconds),
124            }
125            .observe()
126            .map_err(|e| e.to_string())
127        })
128        .collect();
129    choose(d, observations)
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    fn declaration(n: usize) -> OffloadDeclaration {
137        let hosts: Vec<_> = (0..n)
138            .map(|i| serde_json::json!({ "id": format!("h{i}"), "address": format!("u@h{i}") }))
139            .collect();
140        serde_json::from_value(serde_json::json!({
141            "schema": "derive-offload/v1", "kind": "derive-offload", "id": "d", "hosts": hosts
142        }))
143        .unwrap()
144    }
145
146    fn o(text: &str) -> Result<Observation, String> {
147        Ok(Observation::parse(text).unwrap())
148    }
149
150    const ROOMY: &str = "free 104857600\n";
151
152    #[test]
153    fn the_least_loaded_eligible_host_is_chosen() {
154        let reports = choose(
155            &declaration(3),
156            vec![
157                o(&format!("{ROOMY}ticket 1-1-a alive\nticket 2-2-b alive\n")),
158                o(&format!("{ROOMY}lock\nholder 3 alive\n")),
159                o(&format!("{ROOMY}ticket 1-1-a alive\nlock\n")),
160            ],
161        );
162        let chosen: Vec<_> = reports
163            .iter()
164            .filter(|r| r.chosen)
165            .map(|r| r.id.as_str())
166            .collect();
167        assert_eq!(chosen, ["h1"]);
168        assert_eq!(reports[0].load, Some(2));
169        assert_eq!(reports[2].load, Some(2));
170    }
171
172    #[test]
173    fn a_tie_goes_to_the_host_declared_first() {
174        let reports = choose(&declaration(2), vec![o(ROOMY), o(ROOMY)]);
175        assert!(reports[0].chosen && !reports[1].chosen);
176    }
177
178    #[test]
179    fn hosts_that_cannot_take_a_derive_are_skipped_and_say_why() {
180        let reports = choose(
181            &declaration(4),
182            vec![
183                Err("ssh: connect to host timed out".into()),
184                o("free 9437184\n"), // 9 GiB, under the default 10
185                o("free unknown\n"),
186                o(&format!(
187                    "{ROOMY}ticket 1-1-a alive\nticket 2-2-b alive\nticket 3-3-c alive\n"
188                )),
189            ],
190        );
191        assert_eq!(reports[0].standing, Standing::Unreachable);
192        assert_eq!(reports[1].standing, Standing::UnderMinimum);
193        assert!(reports[1]
194            .reason
195            .as_deref()
196            .unwrap()
197            .contains("9 GiB free, under 10"));
198        assert_eq!(reports[2].standing, Standing::UnknownSpace);
199        assert!(
200            reports[3].chosen,
201            "the busiest host is still better than none"
202        );
203    }
204
205    #[test]
206    fn a_host_may_carry_its_own_minimum() {
207        let mut d = declaration(2);
208        d.hosts[0].own_min_free_gib = Some(200);
209        let reports = choose(&d, vec![o(ROOMY), o(ROOMY)]);
210        assert_eq!(reports[0].standing, Standing::UnderMinimum);
211        assert!(reports[1].chosen);
212    }
213
214    #[test]
215    fn no_eligible_host_chooses_nothing() {
216        let reports = choose(&declaration(2), vec![Err("down".into()), o("free 0\n")]);
217        assert!(reports.iter().all(|r| !r.chosen));
218    }
