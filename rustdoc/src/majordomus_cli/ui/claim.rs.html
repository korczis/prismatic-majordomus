1//! One claim in full, for the Evidence screen's inspector (`/ui/evidence?claim=<id>`): what it
2//! says and where, the state its proof is in and what that state means, the run behind it,
3//! what changed since, how to see it again, every run the ledger holds for it, and the other
4//! claims the same test proves, each a link to its own inspector.
5//!
6//! Read from `evidence.claim`, the one capability that joins a claim to its execution; a claim
7//! the matrix does not declare is its refusal, said as such, never an empty proof.
8

18/// A claim id as the matrix spells them (`a-finding-names-how-to-see-it-again`): letters,
19/// digits, `-`, `_` and `.`, and nothing a link could carry anything else in.
20pub fn is_claim_id(id: &str) -> bool {
21    !id.is_empty()
22        && id.len() <= 128
23        && id
24            .bytes()
25            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.'))
26}

28/// The claim a query names, when it names one.
29pub fn picked(query: &[(String, String)]) -> Option<String> {
30    query
31        .iter()
32        .find(|(k, _)| k == "claim")
33        .map(|(_, v)| v.trim().to_string())
34        .filter(|v| is_claim_id(v))
35}

37/// Where a claim's inspector is.
38pub fn href(id: &str) -> String {
39    format!("{}?claim={id}#claim", super::evidence::ROUTE)
40}

42/// Ask `evidence.claim` for `id`.
43pub fn ask(ctx: &Context, id: &str) -> Answer {
44    ask_input(ctx, "evidence.claim", serde_json::json!({ "claim": id }))
45}

47fn strings(v: &Value, k: &str) -> Vec<String> {
48    v.get(k)
49        .and_then(Value::as_array)
50        .map(|a| {
51            a.iter()
52                .filter_map(Value::as_str)
53                .map(str::to_string)
54                .collect()
55        })
56        .unwrap_or_default()
57}

59fn mono(t: String) -> El {
60    el("span").class("mj-ui-mono").text(t)

63/// The inspector of the claim `answer` is about.
64pub fn inspector(answer: &Answer, id: &str) -> El {
65    let body = unanswered(answer).unwrap_or_else(|| {
66        // `unanswered` has spoken for an answer that failed; this one answered
67        let c = answer.ok().cloned().unwrap_or(Value::Null);
68        let fact = |label: &str, value: El| {
69            el("div")
70                .class("mj-ui-fact")
71                .child(el("dt").text(label))
72                .child(el("dd").child(value))
73        };
74        let run = c.get("execution").cloned().unwrap_or(Value::Null);
75        let last = if run.is_null() {
76            el("span").class("mj-ui-dim").text("never run")
77        } else {
78            el("span")
79                .child(word_badge(&text(&run, "outcome")))
80                .child(mono(format!(
81                    " {} s · {} · {}",
82                    run.get("seconds")
83                        .map(Value::to_string)
84                        .unwrap_or_else(|| "?".into()),
85                    text(&run, "commit").get(..10).unwrap_or_default(),
86                    text(&run, "working_tree")
87                )))
88        };
89        let history = c
90            .get("executions")
91            .and_then(Value::as_array)
92            .cloned()
93            .unwrap_or_default()
94            .iter()
95            .rev()
96            .take(10)
97            .map(|e| {
98                el("tr")
99                    .child(el("td").child(word_badge(&text(e, "outcome"))))
100                    .child(
101                        el("td")
102                            .class("mj-ui-mono")
103                            .text(e.get("seconds").map(Value::to_string).unwrap_or_default()),
104                    )
105                    .child(
106                        el("td")
107                            .class("mj-ui-mono")
108                            .text(text(e, "commit").get(..10).unwrap_or_default().to_string()),
109                    )
110                    .child(
111                        el("td")
112                            .class("mj-ui-mono mj-ui-dim")
113                            .text(text(e, "working_tree")),
114                    )
115                    .child(
116                        el("td")
117                            .class("mj-ui-mono mj-ui-dim")
118                            .text(text(e, "at").replace('T', " ")),
119                    )
120            })
121            .collect::<Vec<_>>();
122        let changed = strings(&c, "changed");
123        let also = strings(&c, "also_proves");
124        let reproduce = text(&c, "reproduce");
125        el("section")
126            .class("mj-ui-detail")
127            .attr("id", "claim")
128            .attr("aria-label", format!("Claim {id}"))
129            .child(
130                el("div")
131                    .class("mj-ui-detail-head")
132                    .child(el("p").class("mj-ui-mono mj-ui-dim").text(id.to_string()))
133                    .child(el("h3").class("mj-ui-detail-title").text(text(&c, "claim")))
134                    .child(
135                        el("p")
136                            .class("mj-ui-badges")
137                            .child(state(&c, "state"))
138                            .child(state(&c, "status"))
139                            .child(
140                                el("a")
141                                    .class("mj-ui-more")
142                                    .attr("href", super::evidence::ROUTE)
143                                    .text("Close"),
144                            ),
145                    ),
146            )
147            .child(el("p").class("mj-ui-lead").text(text(&c, "meaning")))
148            .child(
149                el("dl")
150                    .class("mj-ui-facts")
151                    .child(fact("Said in", mono(text(&c, "source"))))
152                    .child(fact("Implemented by", mono(text(&c, "implementation"))))
153                    .child(fact("Settled by", mono(text(&c, "test_path"))))
154                    .child(fact("Last run", last))
155                    .child(fact(
156                        "Reproduce",
157                        if reproduce.is_empty() {
158                            el("span").class("mj-ui-dim").text("none")
159                        } else {
160                            el("code").class("mj-ui-mono").text(reproduce)
161                        },
162                    )),
163            )
164            .child(
165                el("h4")
166                    .class("mj-ui-panel-title")
167                    .text("Changed since it was proven"),
168            )
169            .child(if changed.is_empty() {
170                nothing("Nothing it names has changed since its proof.")
171            } else {
172                el("ul")
173                    .class("mj-ui-list")
174                    .children(changed.into_iter().map(|p| el("li").child(mono(p))))
175            })
176            .child(
177                el("h4")
178                    .class("mj-ui-panel-title")
179                    .text("Also proved by the same test"),
180            )
181            .child(if also.is_empty() {
182                nothing("The test proves this claim alone.")
183            } else {
184                el("p").class("mj-ui-badges").children(also.iter().map(|a| {
185                    el("a")
186                        .class("mj-ui-chip mj-ui-chip--link")
187                        .attr("href", href(a))
188                        .child(el("code").class("mj-ui-chip-value").text(a.clone()))
189                }))
190            })
191            .child(
192                el("h4")
193                    .class("mj-ui-panel-title")
194                    .text("Runs the ledger holds"),
195            )
196            .child(table(
197                &["Outcome", "Seconds", "Commit", "Tree", "At"],
198                history,
199                "The ledger holds no run of its test.",
200            ))
201    });
202    panel(
203        "Claim",
204        "One claim of the product, with what proves it and how to see it again.",
205        body,
206        &[answer],
207    )
208}

216    fn answer(value: Result<Value, String>) -> Answer {

222            took: Duration::from_millis(1),

226    /// A claim that never ran, names no reproduction, changed nothing since and shares its
227    /// test with no other claim says each of those, rather than drawing an empty field.
228    #[test]
229    fn a_claim_never_run_says_what_it_lacks() {
230        let html = inspector(
231            &answer(Ok(
232                serde_json::json!({ "id": "c-2", "claim": "Alone", "state": "not_run" }),
233            )),
234            "c-2",
235        )
236        .render();
237        for want in [
238            "never run",
239            ">none<",
240            "Nothing it names has changed since its proof.",
241            "The test proves this claim alone.",
242        ] {
243            assert!(html.contains(want), "no {want}: {html}");
244        }
245        let html = inspector(
246            &answer(Ok(
247                serde_json::json!({ "id": "c-3", "claim": "Timed nowhere",
248                "execution": { "outcome": "pass", "commit": "0123456789abcdef" } }),
249            )),
250            "c-3",
251        )
252        .render();
253        assert!(html.contains("0123456789"), "{html}");
254        assert!(
255            html.contains("?"),
256            "a run without its seconds says so: {html}"
257        );
258    }

260    #[test]
261    fn a_claim_shows_its_proof_what_changed_and_the_claims_its_test_also_proves() {
262        let a = answer(Ok(serde_json::json!({
263            "id": "c-1", "claim": "A finding names how to see it again", "status": "guaranteed",
264            "source": "docs/DOCTRINE.md", "implementation": "scripts/x", "test_path": "test/cases/138.sh",
265            "state": "stale", "meaning": "The proof is older than what it is about.",
266            "execution": { "outcome": "pass", "seconds": 6, "commit": "a04b65c9110a74dc", "working_tree": "dirty" },
267            "changed": ["docs/DOCTRINE.md"], "reproduce": "bash test/run.sh 138",
268            "executions": [{ "outcome": "fail", "seconds": 5, "commit": "1111111111aa", "working_tree": "clean", "at": "2026-10-01T00:00:00Z" },
269                           { "outcome": "pass", "seconds": 6, "commit": "a04b65c9110a", "working_tree": "dirty", "at": "2026-10-02T00:00:00Z" }],
270            "also_proves": ["authored-files-are-english"]
271        })));
272        let html = inspector(&a, "c-1").render();
273        for want in [
274            "id=\"claim\"",
275            "The proof is older than what it is about.",
276            "bash test/run.sh 138",
277            "docs/DOCTRINE.md",
278            "/ui/evidence?claim=authored-files-are-english#claim",
279            "a04b65c911",
280        ] {
281            assert!(html.contains(want), "no {want}: {html}");
282        }
283        match (html.find("2026-10-02"), html.find("2026-10-01")) {
284            (Some(newer), Some(older)) => assert!(newer < older, "the newest run first"),
285            other => panic!("both runs are listed: {other:?}"),
286        }
287    }

289    #[test]
290    fn a_claim_the_matrix_does_not_declare_is_said_as_its_refusal_and_a_hostile_id_never_asked() {
291        let refused = inspector(&answer(Err("not found: no claim nope".into())), "nope").render();
292        assert!(refused.contains("not found: no claim nope"), "{refused}");
293        assert!(!refused.contains("id=\"claim\""));
294        let q = |v: &str| vec![("claim".to_string(), v.to_string())];
295        assert_eq!(
296            picked(&q("a-finding.names_it")),
297            Some("a-finding.names_it".into())
298        );
299        assert_eq!(picked(&q("x\" onclick=\"y")), None);
300    }
