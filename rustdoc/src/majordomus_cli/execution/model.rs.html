12/// The identity of one execution: `x-<UTC timestamp>-<8 hex>`.
13///
14/// The shape is the layer's own — a task is `t-20260905034523-a9f1` — so that an id read
15/// in a log, a URL or a browser tab is recognisable as this repository's without a
16/// dependency on a UUID crate. The timestamp orders ids by creation for a reader; the
17/// suffix, not the timestamp, is what makes two ids created in the same second differ.
18#[derive(
19    Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
20)]
21#[serde(transparent)]
22pub struct ExecutionId(String);
23
24impl ExecutionId {
25    /// A fresh id for now.
26    ///
27    /// ```
28    /// use majordomus_cli::execution::ExecutionId;
29    /// let a = ExecutionId::fresh();
30    /// let b = ExecutionId::fresh();
31    /// assert_ne!(a, b, "two ids from the same second still differ");
32    /// assert!(ExecutionId::parse(a.as_str()).is_some());
33    /// ```
34    pub fn fresh() -> Self {
35        static COUNTER: AtomicU64 = AtomicU64::new(0);
36        let now = SystemTime::now()
37            .duration_since(UNIX_EPOCH)
38            .unwrap_or_default();
39        let stamp = compact_utc(now.as_secs());
40        // the suffix is not a secret and does not need to be unguessable: the server
41        // refuses a state-changing request from another origin, and an id is not a
42        // credential. It needs to be unique within a process, which a counter guarantees,
43        // and unlikely to repeat across two processes of one repository, which the
44        // sub-second part gives it.
45        let seq = COUNTER.fetch_add(1, Ordering::Relaxed);
46        let mix = (now.subsec_nanos() as u64) ^ (seq << 20) ^ (seq.wrapping_mul(0x9e37_79b9));
47        ExecutionId(format!("x-{stamp}-{:08x}", mix & 0xffff_ffff))
48    }
49
50    /// Parse an id a client sent. `None` when it is not one this executable would mint,
51    /// which is what keeps a path, a traversal or a wildcard out of a store lookup.
52    ///
53    /// ```
54    /// use majordomus_cli::execution::ExecutionId;
55    /// assert!(ExecutionId::parse("x-20260908T010203Z-0a1b2c3d").is_some());
56    /// assert!(ExecutionId::parse("../../etc/passwd").is_none());
57    /// assert!(ExecutionId::parse("x-20260908T010203Z-0a1b2c3").is_none(), "eight hex digits");
58    /// ```
59    pub fn parse(text: &str) -> Option<Self> {
60        let rest = text.strip_prefix("x-")?;
61        let (stamp, suffix) = rest.split_once('-')?;
62        let stamp_ok = stamp.len() == 16
63            && stamp.ends_with('Z')
64            && stamp.as_bytes()[8] == b'T'
65            && stamp[..8].bytes().all(|b| b.is_ascii_digit())
66            && stamp[9..15].bytes().all(|b| b.is_ascii_digit());
67        let suffix_ok = suffix.len() == 8 && suffix.bytes().all(|b| b.is_ascii_hexdigit());
68        (stamp_ok && suffix_ok).then(|| ExecutionId(text.to_string()))
69    }
70
71    /// The id as text.
72    pub fn as_str(&self) -> &str {
73        &self.0
74    }
75}
76
77impl std::fmt::Display for ExecutionId {
78    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
79        f.write_str(&self.0)
80    }
81}

84pub const MAX_IDEMPOTENCY_KEY: usize = 128;

86/// The key a client sends with a start so that sending it again — a double click, a retry
87/// after a timeout whose answer never arrived — answers the execution the first send
88/// started instead of starting a second one (I4000).
89///
90/// One to [`MAX_IDEMPOTENCY_KEY`] characters of `A-Z a-z 0-9 . _ : -`: enough for a UUID,
91/// a ULID or `issue:I4000:implement`, and nothing a log line, a URL or a shell has to
92/// escape. A key names a request, not a secret, and is shown in the snapshot it started.
93///
94/// ```
95/// use majordomus_cli::execution::IdempotencyKey;
96/// assert_eq!(IdempotencyKey::parse("issue:I4000:implement").unwrap().as_str(), "issue:I4000:implement");
97/// assert!(IdempotencyKey::parse("").is_err(), "empty");
98/// assert!(IdempotencyKey::parse("two words").is_err(), "a space");
99/// assert!(IdempotencyKey::parse(&"k".repeat(129)).is_err(), "too long");
100/// assert!(IdempotencyKey::parse(&"k".repeat(128)).is_ok());
101/// ```
102#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
103pub struct IdempotencyKey(String);
104
105impl IdempotencyKey {
106    /// Parse a key a client sent, or say why it is not one. Nothing is trimmed or folded:
107    /// a key is compared byte for byte, so `Click-1` and `click-1` are two requests.
108    ///
109    /// ```
110    /// use majordomus_cli::execution::IdempotencyKey;
111    /// assert_ne!(IdempotencyKey::parse("Click-1"), IdempotencyKey::parse("click-1"));
112    /// let refused = IdempotencyKey::parse("a/b").unwrap_err();
113    /// assert!(refused.contains("'/'"), "the refusal names the character: {refused}");
114    /// ```
115    pub fn parse(text: &str) -> Result<Self, String> {
116        let length = text.chars().count();
117        if length == 0 || length > MAX_IDEMPOTENCY_KEY {
118            return Err(format!(
119                "an idempotency key is 1 to {MAX_IDEMPOTENCY_KEY} characters; this one has {length}"
120            ));
121        }
122        if let Some(bad) = text
123            .chars()
124            .find(|c| !(c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | ':' | '-')))
125        {
126            return Err(format!(
127                "an idempotency key is made of A-Z a-z 0-9 . _ : - only, and {bad:?} is not"
128            ));
129        }
130        Ok(IdempotencyKey(text.to_string()))
131    }
132
133    /// The key exactly as the client sent it, for the snapshot and for a refusal that
134    /// quotes it back.
135    ///
136    /// ```
137    /// use majordomus_cli::execution::IdempotencyKey;
138    /// assert_eq!(IdempotencyKey::parse("retry.7").unwrap().as_str(), "retry.7");
139    /// ```
140    pub fn as_str(&self) -> &str {
141        &self.0
142    }
143}
144
145impl std::fmt::Display for IdempotencyKey {
146    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
147        f.write_str(&self.0)
148    }
149}

151/// The longest issue id a start accepts, in characters.
152pub const MAX_ISSUE_ID: usize = 64;
153
154/// Parse the issue a start names (I4004): `I4004`, `#184`. One to [`MAX_ISSUE_ID`]
155/// characters of `A-Z a-z 0-9 # . _ -`, compared byte for byte; anything else is refused
156/// with the character named, because the id becomes part of a mesh scope (`issue/<id>`)
157/// and a `/` or a space would make that scope mean something else.
158///
159/// ```
160/// use majordomus_cli::execution::model::parse_issue;
161/// assert_eq!(parse_issue("I4004").unwrap(), "I4004");
162/// assert_eq!(parse_issue("#184").unwrap(), "#184");
163/// assert!(parse_issue("").is_err());
164/// assert!(parse_issue("I1/../x").unwrap_err().contains("'/'"));
165/// ```
166pub fn parse_issue(text: &str) -> Result<String, String> {
167    let length = text.chars().count();
168    if length == 0 || length > MAX_ISSUE_ID {
169        return Err(format!(
170            "an issue id is 1 to {MAX_ISSUE_ID} characters; this one has {length}"
171        ));
172    }
173    if let Some(bad) = text
174        .chars()
175        .find(|c| !(c.is_ascii_alphanumeric() || matches!(c, '#' | '.' | '_' | '-')))
176    {
177        return Err(format!(
178            "an issue id is made of A-Z a-z 0-9 # . _ - only, and {bad:?} is not"
179        ));
180    }
181    Ok(text.to_string())
182}

184/// Where the mesh claim of an execution started for an issue stands (I4004).
185///
186/// ```text
187///   wanted ──► held ──► released            the run reached a final state
188///                  └──► kept                its outcome is not known: the claim stays
189///          ──► unclaimed                    no mesh runs here: it ran without a claim
190///          ──► refused                      another session holds the scope: nothing ran
191/// ```
192///
193/// ```
194/// use majordomus_cli::execution::ClaimStanding;
195/// let word = serde_json::to_value(ClaimStanding::Kept).unwrap();
196/// assert_eq!(word, "kept");
197/// let back: ClaimStanding = serde_json::from_value("unclaimed".into()).unwrap();
198/// assert_eq!(back, ClaimStanding::Unclaimed);
199/// ```
200#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
201#[serde(rename_all = "snake_case")]
202pub enum ClaimStanding {
203    /// Started for an issue; the claim is taken when a worker picks it up.
204    Wanted,
205    /// The exclusive claim is held, and the handler runs under it.
206    Held,
207    /// The claim was given back when the execution reached its final state.
208    Released,
209    /// The outcome is not known, so the claim is kept: whether the work happened is still
210    /// to be read, and the scope is not free until it has been.
211    Kept,
212    /// No mesh cooperation runs in this process, so it ran without a claim. Said, never a
213    /// failure.
214    Unclaimed,
215    /// Another session holds an exclusive claim on the scope; the execution was refused and
216    /// nothing ran.
217    Refused,
218}
219
220impl ClaimStanding {
221    /// The word serde writes for it, for a line a terminal prints without serialising: the
222    /// same word on the wire, in the snapshot and in `majordomus run --follow`.
223    ///
224    /// ```
225    /// use majordomus_cli::execution::ClaimStanding;
226    /// assert_eq!(ClaimStanding::Unclaimed.as_str(), "unclaimed");
227    /// assert_eq!(serde_json::to_value(ClaimStanding::Held).unwrap(), ClaimStanding::Held.as_str());
228    /// ```
229    pub fn as_str(self) -> &'static str {
230        match self {
231            ClaimStanding::Wanted => "wanted",
232            ClaimStanding::Held => "held",
233            ClaimStanding::Released => "released",
234            ClaimStanding::Kept => "kept",
235            ClaimStanding::Unclaimed => "unclaimed",
236            ClaimStanding::Refused => "refused",
237        }
238    }
239}

241/// The issue an execution works for and the mesh claim it holds for it (I4004): the issue,
242/// the scope claimed, the claim's key while there is one, and where it stands.
243///
244/// ```
245/// use majordomus_cli::execution::{ClaimStanding, IssueClaim};
246/// let wanted = IssueClaim::wanted("I4004");
247/// assert_eq!(wanted.standing, ClaimStanding::Wanted);
248/// assert!(wanted.scope.is_empty() && wanted.key.is_none());
249/// let json = serde_json::to_value(&wanted).unwrap();
250/// assert_eq!(json, serde_json::json!({ "issue": "I4004", "standing": "wanted" }));
251/// ```
252#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
253pub struct IssueClaim {
254    /// The issue, as the start named it.
255    pub issue: String,
256    /// Where the claim stands.
257    pub standing: ClaimStanding,
258    /// The repository paths claimed: the issue record's `scope`, or `issue/<id>` when the
259    /// record declares none or is not in this checkout.
260    #[serde(default, skip_serializing_if = "Vec::is_empty")]
261    pub scope: Vec<String>,
262    /// The mesh claim's key, `<stream>/<claim>`, once one was taken.
263    #[serde(default, skip_serializing_if = "Option::is_none")]
264    pub key: Option<String>,
265    /// The mesh session the claim belongs to, once one was opened: `execution-<id>`.
266    #[serde(default, skip_serializing_if = "Option::is_none")]
267    pub session: Option<String>,
268    /// What there is to say: why it ran unclaimed, who holds the scope it was refused, why
269    /// the claim is kept, or that a release did not go through.
270    #[serde(default, skip_serializing_if = "Option::is_none")]
271    pub detail: Option<String>,
272}

296/// `YYYYMMDDTHHMMSSZ` in UTC from a Unix second, the compact form of the layer's own
297/// timestamps.
298///
299/// ```
300/// use majordomus_cli::execution::model::compact_utc;
301/// assert_eq!(compact_utc(0), "19700101T000000Z");
302/// assert_eq!(compact_utc(1_788_000_000), "20260829T104000Z");
303/// ```
304pub fn compact_utc(secs: u64) -> String {
305    let iso = crate::peers::rfc3339(UNIX_EPOCH + std::time::Duration::from_secs(secs));
306    iso.chars().filter(|c| !matches!(c, '-' | ':')).collect()
307}

309/// Where an execution is in its life.
310///
311/// The transitions are the whole contract, and [`ExecutionState::EDGES`] is the one place
312/// they are written down: [`ExecutionState::may_move_to`] reads it, the engine asks before
313/// every change, the store refuses a move it does not allow, and a client that reads a
314/// final state never sees it move again.
315///
316/// ```text
317///   authorizing     ──► queued | refused | failed | cancelled
318///   queued          ──► running | claimed | refused | failed | cancelled
319///   claimed         ──► running | queued | refused | failed | cancelling | unknown_outcome
320///   running         ──► verifying | succeeded | failed | cancelling | unknown_outcome
321///   verifying       ──► succeeded | failed | cancelling | unknown_outcome
322///   cancelling      ──► cancelled | succeeded | failed | unknown_outcome
323///   unknown_outcome ──► verifying | succeeded | failed | cancelled        (never running)
324///   succeeded, failed, cancelled, refused: final
325/// ```
326///
327/// The in-process engine moves `queued ──► running` directly: it claims an execution and
328/// enters its handler in the same instant, so `claimed` would be a state no client could
329/// observe. `authorizing`, `claimed`, `verifying`, `refused` and `unknown_outcome` are
330/// entered by a worker that decides, holds, checks or loses an execution in its own time
331/// (Mission Control B3): a policy decision before the queue, a lease on a queued ticket, a
332/// check of what the handler produced, a refusal, and a run whose outcome cannot be read.
333///
334/// `unknown_outcome` is the one state a retry may not leave by running again. Whether the
335/// work happened is not known, so the next move is a reconciliation that reads evidence
336/// — to `verifying`, or straight to the outcome it found — and never `running`.
337#[derive(
338    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
339)]
340#[serde(rename_all = "snake_case")]
341pub enum ExecutionState {
342    /// Accepted, and a policy decision on whether it may run has not been taken yet.
343    Authorizing,
344    /// Accepted, and no worker has picked it up.
345    Queued,
346    /// A worker holds it and has not entered the handler yet.
347    Claimed,
348    /// The handler is running.
349    Running,
350    /// The handler returned, and what it produced is being checked.
351    Verifying,
352    /// Cancellation was asked for and the handler has not stopped yet.
353    Cancelling,
354    /// Whether the work happened is not known: the worker that ran it went silent or its
355    /// answer could not be read. Only a reconciliation moves it on, never a rerun.
356    UnknownOutcome,
357    /// The handler returned an output.
358    Succeeded,
359    /// The handler returned an error, or the worker could not run it.
360    Failed,
361    /// It stopped because it was asked to.
362    Cancelled,
363    /// A policy refused it before it ran, at the request or when a worker claimed it.
364    Refused,
365}
366
367impl ExecutionState {
368    /// Every state, in the order of an execution's life.
369    ///
370    /// ```
371    /// use majordomus_cli::execution::ExecutionState;
372    /// assert_eq!(ExecutionState::ALL.len(), 11);
373    /// assert_eq!(ExecutionState::ALL[0], ExecutionState::Authorizing);
374    /// ```
375    pub const ALL: [ExecutionState; 11] = [
376        ExecutionState::Authorizing,
377        ExecutionState::Queued,
378        ExecutionState::Claimed,
379        ExecutionState::Running,
380        ExecutionState::Verifying,
381        ExecutionState::Cancelling,
382        ExecutionState::UnknownOutcome,
383        ExecutionState::Succeeded,
384        ExecutionState::Failed,
385        ExecutionState::Cancelled,
386        ExecutionState::Refused,
387    ];
388
389    /// The transition table: every move an execution may make, and no other. A pair that
390    /// is not here is refused.
391    ///
392    /// ```
393    /// use majordomus_cli::execution::ExecutionState::{self, *};
394    /// assert!(ExecutionState::EDGES.contains(&(Queued, Running)));
395    /// assert!(!ExecutionState::EDGES.contains(&(UnknownOutcome, Running)));
396    /// ```
397    pub const EDGES: &'static [(ExecutionState, ExecutionState)] = {
398        use ExecutionState::*;
399        &[
400            // a policy decision before the queue
401            (Authorizing, Queued),
402            (Authorizing, Refused),
403            (Authorizing, Failed),
404            (Authorizing, Cancelled),
405            // the queue: the in-process engine enters the handler at once; a worker claims
406            (Queued, Running),
407            (Queued, Claimed),
408            (Queued, Refused),
409            (Queued, Failed),
410            (Queued, Cancelled),
411            // a claim: entered, released before anything ran, refused when the worker
412            // authorises the ticket it consumes, or lost with its holder
413            (Claimed, Running),
414            (Claimed, Queued),
415            (Claimed, Refused),
416            (Claimed, Failed),
417            (Claimed, Cancelling),
418            (Claimed, UnknownOutcome),
419            // the handler
420            (Running, Verifying),
421            (Running, Succeeded),
422            (Running, Failed),
423            (Running, Cancelling),
424            (Running, UnknownOutcome),
425            // the check of what it produced
426            (Verifying, Succeeded),
427            (Verifying, Failed),
428            (Verifying, Cancelling),
429            (Verifying, UnknownOutcome),
430            // a stop asked for
431            (Cancelling, Cancelled),
432            (Cancelling, Succeeded),
433            (Cancelling, Failed),
434            (Cancelling, UnknownOutcome),
435            // a reconciliation, which reads evidence and never runs the work again
436            (UnknownOutcome, Verifying),
437            (UnknownOutcome, Succeeded),
438            (UnknownOutcome, Failed),
439            (UnknownOutcome, Cancelled),
440        ]
441    };
442
443    /// The word as serialised.
444    pub fn as_str(self) -> &'static str {
445        match self {
446            ExecutionState::Authorizing => "authorizing",
447            ExecutionState::Queued => "queued",
448            ExecutionState::Claimed => "claimed",
449            ExecutionState::Running => "running",
450            ExecutionState::Verifying => "verifying",
451            ExecutionState::Cancelling => "cancelling",
452            ExecutionState::UnknownOutcome => "unknown_outcome",
453            ExecutionState::Succeeded => "succeeded",
454            ExecutionState::Failed => "failed",
455            ExecutionState::Cancelled => "cancelled",
456            ExecutionState::Refused => "refused",
457        }
458    }
459
460    /// Is this the last thing that will be said about the execution?
461    ///
462    /// ```
463    /// use majordomus_cli::execution::ExecutionState;
464    /// assert!(ExecutionState::Succeeded.is_final());
465    /// assert!(ExecutionState::Refused.is_final());
466    /// assert!(!ExecutionState::Cancelling.is_final());
467    /// assert!(!ExecutionState::UnknownOutcome.is_final(), "it waits for a reconciliation");
468    /// ```
469    pub fn is_final(self) -> bool {
470        matches!(
471            self,
472            ExecutionState::Succeeded
473                | ExecutionState::Failed
474                | ExecutionState::Cancelled
475                | ExecutionState::Refused
476        )
477    }
478
479    /// Is the execution not final yet: doing work, about to, or waiting to be reconciled?
480    pub fn is_active(self) -> bool {
481        !self.is_final()
482    }
483
484    /// May the execution move from here to `next`? Exactly the pairs of
485    /// [`ExecutionState::EDGES`].
486    ///
487    /// ```
488    /// use majordomus_cli::execution::ExecutionState::*;
489    /// assert!(Queued.may_move_to(Running));
490    /// assert!(Running.may_move_to(Cancelling));
491    /// assert!(Queued.may_move_to(Cancelled), "a queued execution is cancelled without running");
492    /// assert!(!Succeeded.may_move_to(Running), "a final state is final");
493    /// assert!(!Running.may_move_to(Queued), "nothing that ran goes back");
494    /// assert!(Authorizing.may_move_to(Refused));
495    /// assert!(!UnknownOutcome.may_move_to(Running), "reconciled first, never rerun");
496    /// ```
497    pub fn may_move_to(self, next: ExecutionState) -> bool {
498        Self::EDGES.contains(&(self, next))
499    }
500
501    /// The states this one may move to, in the order of [`ExecutionState::ALL`].
502    ///
503    /// ```
504    /// use majordomus_cli::execution::ExecutionState::*;
505    /// assert_eq!(UnknownOutcome.successors(), vec![Verifying, Succeeded, Failed, Cancelled]);
506    /// assert!(Refused.successors().is_empty());
507    /// ```
508    pub fn successors(self) -> Vec<ExecutionState> {
509        Self::ALL
510            .into_iter()
511            .filter(|next| self.may_move_to(*next))
512            .collect()
513    }
514}

560impl ProgressView {
561    /// The fraction done, 0..=100, when a total is known.
562    ///
563    /// ```
564    /// use majordomus_cli::execution::ProgressView;
565    /// let p = ProgressView { current: 3, total: Some(4), message: None };
566    /// assert_eq!(p.percent(), Some(75));
567    /// let open = ProgressView { current: 3, total: None, message: None };
568    /// assert_eq!(open.percent(), None);
569    /// ```
570    pub fn percent(&self) -> Option<u64> {
571        match self.total {
572            Some(total) if total > 0 => Some((self.current.min(total) * 100) / total),
573            _ => None,
574        }
575    }
576}

578/// Why an execution ended badly, in the repository's diagnostic vocabulary rather than in
579/// Rust's: a code a program branches on, a sentence a person reads, and where to look.
580///
581/// A panic message and a backtrace are never in here. The engine catches a panicking
582/// handler and reports `internal` with the execution's correlation id; the panic itself is
583/// on the process's own error stream, where the operator running the server can read it.
584#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
585pub struct ExecutionError {
586    /// `invalid_input`, `not_found`, `refused`, `forbidden`, `conflict`, `internal`, `cancelled`,
587    /// `unavailable`.
588    pub code: String,
589    /// What went wrong, for a person.
590    pub message: String,
591    /// What to do about it, when there is something.
592    #[serde(skip_serializing_if = "Option::is_none")]
593    pub suggestion: Option<String>,
594    /// The execution this failure belongs to, so a log line and a UI agree.
595    pub correlation_id: String,
596}
597
598/// A structured finding an execution reported on its way, distinct from its outcome.
599#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
600pub struct ExecutionDiagnostic {
601    /// How bad.
602    pub severity: crate::model::Severity,
603    /// A stable machine-readable code.
604    pub code: String,
605    /// One line, for a person.
606    pub summary: String,
607    /// The rest, when there is more.
608    #[serde(skip_serializing_if = "Option::is_none")]
609    pub detail: Option<String>,
610    /// What to do about it, when there is something.
611    #[serde(skip_serializing_if = "Option::is_none")]
612    pub suggestion: Option<String>,
613}

615/// Which stream a log line came from.
616#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
617#[serde(rename_all = "snake_case")]
618pub enum LogStream {
619    /// The standard output of a child process the handler ran.
620    Stdout,
621    /// Its standard error.
622    Stderr,
623    /// The handler itself, saying something it wants a reader to see.
624    Handler,
625}
626
627/// Who asked for the execution. Not an authorisation decision — this server authenticates
628/// nobody — but a fact worth carrying into the audit line and the UI.
629#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
630#[serde(rename_all = "snake_case")]
631pub enum ActorKind {
632    /// An HTTP client, which includes the Cockpit.
633    Http,
634    /// An MCP client attached to this process.
635    Mcp,
636    /// A command line: this process's own, or `majordomus run` in another process of the
637    /// same checkout that reported its execution to this server, with its pid.
638    Cli,
639    /// This executable's own tests and benchmarks.
640    Internal,
641}
642
643/// Who asked, and which peer they are when this process knows.
644#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
645pub struct Actor {
646    /// Where the request came from.
647    pub kind: ActorKind,
648    /// The peer id, for a call that arrived through an MCP session.
649    #[serde(skip_serializing_if = "Option::is_none")]
650    pub peer: Option<String>,
651    /// The process that ran it, for an execution another process ran and reported to this
652    /// one: the work happened there, and this process holds what it was told about it.
653    #[serde(default, skip_serializing_if = "Option::is_none")]
654    pub pid: Option<u32>,
655}
656
657impl Actor {
658    /// An actor of a kind, with no peer.
659    pub fn of(kind: ActorKind) -> Self {
660        Actor {
661            kind,
662            peer: None,
663            pid: None,
664        }
665    }
666
667    /// `majordomus run` in another process, which ran the execution and reported it here.
668    ///
669    /// ```
670    /// use majordomus_cli::execution::{Actor, ActorKind};
671    /// let a = Actor::reported(4242);
672    /// assert_eq!((a.kind, a.pid), (ActorKind::Cli, Some(4242)));
673    /// ```
674    pub fn reported(pid: u32) -> Self {
675        Actor {
676            kind: ActorKind::Cli,
677            peer: None,
678            pid: Some(pid),
679        }
680    }
681}

683/// Which repository, and which checkout of it, an execution ran against.
684///
685/// An execution is never run against "wherever this process happens to be": the engine
686/// stamps the repository the index was read from, and a request that names a different one
687/// is refused. The path itself is not here — this value is served to whoever can reach the
688/// socket, and where the checkout sits on the host is of no use to them.
689#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
690pub struct RepositoryRef {
691    /// The repository's name: the last component of its root.
692    pub name: String,
693    /// The stable identity two processes over one checkout both compute.
694    pub id: String,
695    /// The branch checked out, when git can say.
696    #[serde(skip_serializing_if = "Option::is_none")]
697    pub branch: Option<String>,
698}
699
700/// One execution, as every transport answers it.
701///
702/// This is the whole of what a client may know. It is produced under the store's lock, so
703/// a reader never sees a state and an output that disagree: an execution that says
704/// `succeeded` carries its output in the same snapshot, and one that says `failed` carries
705/// its error.
706#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
707pub struct Execution {
708    /// The identity.
709    pub id: ExecutionId,
710    /// The capability that ran, by its canonical id.
711    pub capability: String,
712    /// That capability's title, so a list needs no second lookup.
713    pub title: String,
714    /// Where it is.
715    pub state: ExecutionState,
716    /// When it was accepted, RFC 3339 in UTC.
717    pub created_at: String,
718    /// When the handler was entered.
719    #[serde(skip_serializing_if = "Option::is_none")]
720    pub started_at: Option<String>,
721    /// When it reached a final state.
722    #[serde(skip_serializing_if = "Option::is_none")]
723    pub finished_at: Option<String>,
724    /// How long it ran, in milliseconds, once it has finished.
725    #[serde(skip_serializing_if = "Option::is_none")]
726    pub duration_ms: Option<u64>,
727    /// The input it was given, with every value the input schema marks sensitive replaced.
728    pub input: Value,
729    /// What the handler returned, for a succeeded execution and never before.
730    #[serde(skip_serializing_if = "Option::is_none")]
731    pub output: Option<Value>,
732    /// Why it failed, for a failed execution and never before; why it was refused, for a
733    /// refused one; and why its outcome is not known, for one whose outcome is not.
734    #[serde(skip_serializing_if = "Option::is_none")]
735    pub error: Option<ExecutionError>,
736    /// The last progress the handler reported.
737    #[serde(skip_serializing_if = "Option::is_none")]
738    pub progress: Option<ProgressView>,
739    /// The steps it entered, in order.
740    #[serde(default, skip_serializing_if = "Vec::is_empty")]
741    pub steps: Vec<StepView>,
742    /// The findings it reported.
743    #[serde(default, skip_serializing_if = "Vec::is_empty")]
744    pub diagnostics: Vec<ExecutionDiagnostic>,
745    /// The sequence number of the last event: the cursor a client subscribes from.
746    pub last_sequence: u64,
747    /// How many events it has produced, including the ones no longer retained.
748    pub event_count: u64,
749    /// Whether the retained event history has lost its oldest entries to the bound.
750    pub events_truncated: bool,
751    /// Whether asking to cancel this execution will do anything: the capability's policy.
752    pub cancellable: bool,
753    /// Who asked for it.
754    pub actor: Actor,
755    /// Which repository it ran against.
756    pub repository: RepositoryRef,
757    /// The id used in this process's logs and tracing spans for this execution; the
758    /// execution id itself, carried under the name a reader of a log expects.
759    pub correlation_id: String,
760    /// The idempotency key the start was sent with, when it was sent with one: a repeated
761    /// start with this key answers this execution instead of starting another (I4000).
762    #[serde(default, skip_serializing_if = "Option::is_none")]
763    pub idempotency_key: Option<String>,
764    /// The issue it was started for, and the mesh claim it takes for it, when it was
765    /// started for one (I4004).
766    #[serde(default, skip_serializing_if = "Option::is_none")]
767    pub issue_claim: Option<IssueClaim>,
768}
769
770impl Execution {
771    /// The percentage a progress bar shows, when the handler said enough to compute one.
772    pub fn percent(&self) -> Option<u64> {
773        self.progress.as_ref().and_then(ProgressView::percent)
774    }
775}

918        use ExecutionState::*;
919        let all = ExecutionState::ALL;
920        for state in all {
921            if state.is_final() {
922                assert!(
923                    all.iter().all(|next| !state.may_move_to(*next)),
924                    "{state:?} moved"

926            } else {
927                assert!(
928                    all.iter().any(|next| state.may_move_to(*next)),
929                    "{state:?} is a dead end"
930                );
931            }
932            assert_eq!(state.is_active(), !state.is_final());
933        }
934        // every state is reachable from a start: authorizing, or queued for a start that
935        // needs no decision

938            let before = reached.len();
939            let next: Vec<_> = reached.iter().flat_map(|s| s.successors()).collect();

943            }

946            all.iter()
