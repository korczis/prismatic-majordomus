1//! The `mcp` module: the MCP projection of this registry, described as one typed value.
2//!
3//! MCP is a projection and owns nothing: every tool is a capability, every resource a
4//! capability or an object of the layer. What a client can find out *about* the projection
5//! was nevertheless scattered — the protocol versions in `initialize`, the tools one page
6//! at a time in `tools/list`, the effect of each in an annotation, the client
7//! configurations in `share/providers.yaml`, whether each one starts this repository's
8//! server in a shell function. `mcp.projection` is the same facts as one answer, read by
9//! the Cockpit's MCP page, the HTTP route, the tool and `majordomus mcp --inspect` alike,
10//! so that none of them keeps an inventory of its own.
11//!
12//! Nothing here is a second list. The tools are the registry's, filtered by the exposure
13//! they declare; the effect and the hints are the capability's classification
14//! ([`ExecutionPolicy::hints`](crate::capability::ExecutionPolicy::hints)); the protocol
15//! facts are the constants the server answers `initialize` with; the clients are the
16//! providers the distribution declares.
17//!
18//! ```
19//! use majordomus_cli::capability::builtin::mcp;
20//! use majordomus_cli::capability::model::CapabilityKind;
21//!
22//! let m = mcp::module();
23//! let c = &m.capabilities[0].capability;
24//! assert_eq!(c.id.as_str(), "mcp.projection");
25//! // describing the projection reads the registry and changes nothing
26//! assert_eq!(c.kind, CapabilityKind::Query);
27//! assert_eq!(c.exposure.http.as_ref().unwrap().path, "/api/v1/mcp");
28//! ```

54/// The input of `mcp.projection`: which tools to list. Everything else in the answer is
55/// about the projection as a whole and is always given.
56///
57/// ```
58/// use majordomus_cli::capability::builtin::mcp::McpProjectionInput;
59/// use majordomus_cli::capability::Effect;
60/// let all: McpProjectionInput = serde_json::from_str("{}").unwrap();
61/// assert!(all.effect.is_none());
62/// let writers: McpProjectionInput =
63///     serde_json::from_str(r#"{"effect":"repository_mutation"}"#).unwrap();
64/// assert_eq!(writers.effect, Some(Effect::RepositoryMutation));
65/// ```
66#[derive(Debug, Default, Clone, Serialize, Deserialize, JsonSchema)]
67#[serde(deny_unknown_fields)]
68pub struct McpProjectionInput {
69    /// List only the tools with this effect: `read`, `process_state` or
70    /// `repository_mutation`. Absent lists every tool.
71    #[serde(default, skip_serializing_if = "Option::is_none")]
72    pub effect: Option<Effect>,
73}
74

84            ),

129/// What the server serves of the protocol, and what it does not.
130///
131/// ```
132/// use majordomus_cli::capability::builtin::mcp::McpServing;
133/// let s: McpServing = serde_json::from_value(serde_json::json!({
134///     "methods": ["initialize", "tools/call"], "tools": true, "resources": true,
135///     "prompts": false, "notifications": false,
136/// })).unwrap();
137/// // what is not served is stated, so a client does not learn it from a refusal
138/// assert!(s.tools && !s.prompts && !s.notifications);
139/// assert!(!s.methods.iter().any(|m| m.starts_with("prompts/")));
140/// ```
141#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
142pub struct McpServing {
143    /// The methods a request may name. Anything else is answered `-32601`.
144    pub methods: Vec<String>,
145    /// Tools: every capability that declares one.
146    pub tools: bool,
147    /// Resources: every capability that declares one, and every object of the layer.
148    pub resources: bool,
149    /// Prompts. Not served: the repository's prompt assets are resources, because a prompt
150    /// served unrendered would read as finished.
151    pub prompts: bool,
152    /// Subscriptions and list-change notifications. Not served: this server sends nothing
153    /// unasked.
154    pub notifications: bool,
155}

224/// Where a client's project-scoped configuration stands in this repository.
225///
226/// ```
227/// use majordomus_cli::capability::builtin::mcp::McpClientStanding;
228/// // the words a caller reads, as they are written on every surface
229/// assert_eq!(serde_json::to_value(McpClientStanding::Wired).unwrap(), "wired");
230/// assert_eq!(serde_json::to_value(McpClientStanding::Foreign).unwrap(), "foreign");
231/// assert_eq!(serde_json::to_value(McpClientStanding::Absent).unwrap(), "absent");
232/// ```
233#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
234#[serde(rename_all = "snake_case")]
235pub enum McpClientStanding {
236    /// The file exists and names the launcher: this client starts this repository's server.
237    Wired,
238    /// The file exists and does not name the launcher: the client reads it and starts
239    /// something else, or nothing.
240    Foreign,
241    /// The file does not exist: this client starts no server here until it is written.
242    Absent,
243}

245/// One client the distribution declares a project-scoped configuration for.
246///
247/// ```
248/// use majordomus_cli::capability::builtin::mcp::{McpClient, McpClientStanding};
249/// let mut clients: Vec<McpClient> = serde_json::from_value(serde_json::json!([
250///     { "id": "gemini", "title": "Gemini CLI", "config": ".gemini/settings.json", "standing": "absent" },
251///     { "id": "claude-code", "title": "Claude Code", "config": ".mcp.json", "standing": "wired" },
252/// ])).unwrap();
253/// // ordered by the provider's id, whatever order the distribution declared them in
254/// majordomus_cli::order::canonical(&mut clients);
255/// assert_eq!(clients[0].id, "claude-code");
256/// assert_eq!(clients[0].standing, McpClientStanding::Wired);
257/// ```
258#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
259pub struct McpClient {
260    /// The provider id.
261    pub id: String,
262    /// The name a person knows it by.
263    pub title: String,
264    /// The file it reads, repository-relative.
265    pub config: String,
266    /// Where that file stands here.
267    pub standing: McpClientStanding,
268}

300/// The MCP projection of this registry in this repository.
301///
302/// ```
303/// use majordomus_cli::capability::builtin::mcp::McpProjection;
304/// let p: McpProjection = serde_json::from_value(serde_json::json!({
305///     "server": { "name": "majordomus", "title": "Majordomus", "version": "0.14.0" },
306///     "protocol_versions": ["2025-06-18"],
307///     "transports": [{ "id": "stdio", "reached_by": "majordomus mcp", "attached": 0 }],
308///     "serving": { "methods": ["initialize"], "tools": true, "resources": true, "prompts": false, "notifications": false },
309///     "tool_count": 2,
310///     "tools": [],
311///     "effects": [{ "effect": "read", "tools": 1 }, { "effect": "repository_mutation", "tools": 1 }],
312///     "writers": ["majordomus_plan_transition"],
313///     "resources": { "builtin": 1, "objects": 0, "template": "majordomus://<kind>/<identity>" },
314///     "launcher_in_repository": false,
315///     "clients": [],
316///     "findings": [],
317/// })).unwrap();
318/// // the count is of every tool, whatever the `effect` filter narrowed `tools` to
319/// assert_eq!(p.tool_count, p.effects.iter().map(|e| e.tools).sum::<usize>());
320/// assert!(p.tools.is_empty() && p.writers.len() == 1);
321/// ```
322#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
323pub struct McpProjection {
324    /// Who answers `initialize`.
325    pub server: McpServerIdentity,
326    /// The protocol versions accepted, newest first.
327    pub protocol_versions: Vec<String>,
328    /// How a client reaches the server.
329    pub transports: Vec<McpTransportEntry>,
330    /// What of the protocol is served.
331    pub serving: McpServing,
332    /// How many tools the registry projects, whatever the filter asked for.
333    pub tool_count: usize,
334    /// The tools asked for, in registry order.
335    pub tools: Vec<McpToolEntry>,
336    /// The tools counted by effect, every effect present.
337    pub effects: Vec<McpEffectCount>,
338    /// The tools that write the repository, by name. The `initialize` instructions name the
339    /// same ones.
340    pub writers: Vec<String>,
341    /// The resources.
342    pub resources: McpResourceSummary,
343    /// Whether this repository carries the launcher at `bin/majordomus-mcp`. Where it does
344    /// not, the installed command of the same name is what a configuration names.
345    pub launcher_in_repository: bool,
346    /// The clients the distribution declares a configuration for, and where each stands.
347    pub clients: Vec<McpClient>,
348    /// What a person should act on. Empty is nothing to do.
349    pub findings: Vec<McpFinding>,
350}

352/// Every tool the registry projects, in registry order.
353///
354/// ```
355/// use majordomus_cli::capability::builtin::{self, mcp};
356/// use majordomus_cli::capability::{CapabilityRegistry, Effect};
357/// let registry = CapabilityRegistry::builder().with_builtin(builtin::all()).build().unwrap();
358/// let tools = mcp::tools(&registry);
359/// let transition = tools.iter().find(|t| t.name == "majordomus_plan_transition").unwrap();
360/// assert_eq!(transition.effect, Effect::RepositoryMutation);
361/// assert!(transition.hints.destructive && !transition.hints.read_only);
362/// ```
363pub fn tools(registry: &CapabilityRegistry) -> Vec<McpToolEntry> {
364    registry
365        .iter()
366        .filter_map(|c| {
367            let name = c.exposure.mcp.as_ref()?.tool.clone()?;
368            Some(McpToolEntry {
369                name,
370                capability: c.id.to_string(),
371                module: c.module.to_string(),
372                title: c.title.clone(),
373                effect: c.execution.effect,
374                hints: c.execution.hints(),
375            })
376        })
377        .collect()
378}

380/// Where a client configuration stands: read from the file, never from a list of who is
381/// expected to be wired.
382///
383/// ```
384/// use majordomus_cli::capability::builtin::mcp::{standing, McpClientStanding};
385/// let dir = std::env::temp_dir().join(format!("mj-mcp-standing-{}", std::process::id()));
386/// std::fs::create_dir_all(&dir).unwrap();
387/// assert_eq!(standing(&dir, ".mcp.json"), McpClientStanding::Absent);
388/// std::fs::write(dir.join(".mcp.json"), r#"{"mcpServers":{"other":{"command":"x"}}}"#).unwrap();
389/// assert_eq!(standing(&dir, ".mcp.json"), McpClientStanding::Foreign);
390/// std::fs::write(dir.join(".mcp.json"), r#"{"mcpServers":{"majordomus":{"command":"bin/majordomus-mcp"}}}"#).unwrap();
391/// assert_eq!(standing(&dir, ".mcp.json"), McpClientStanding::Wired);
392/// std::fs::remove_dir_all(&dir).unwrap();
393/// ```
394pub fn standing(root: &Path, config: &str) -> McpClientStanding {
395    match std::fs::read_to_string(root.join(config)) {
396        Err(_) => McpClientStanding::Absent,
397        Ok(text) if text.contains(LAUNCHER) => McpClientStanding::Wired,
398        Ok(_) => McpClientStanding::Foreign,
399    }
400}

402/// How many tools carry each effect, in the order the effects are declared; an effect no tool
403/// has is left out rather than counted as zero.
404fn effect_counts(all: &[McpToolEntry]) -> Vec<McpEffectCount> {
405    [
406        Effect::Read,
407        Effect::ProcessState,
408        Effect::RepositoryMutation,
409    ]
410    .into_iter()
411    .map(|effect| McpEffectCount {
412        effect,
413        tools: all.iter().filter(|t| t.effect == effect).count(),
414    })
415    .filter(|count| count.tools > 0)
416    .collect()
417}

419/// What the client configurations of this repository leave to be done: one finding for
420/// each configuration that exists and does not start this repository's server, and one
421/// when no declared client has a configuration at all.
422fn client_findings(clients: &[McpClient], launcher_in_repository: bool) -> Vec<McpFinding> {
423    let command = if launcher_in_repository {
424        "bin/majordomus-mcp"
425    } else {
426        LAUNCHER
427    };
428    let mut findings: Vec<McpFinding> = clients
429        .iter()
430        .filter(|c| c.standing == McpClientStanding::Foreign)
431        .map(|c| McpFinding {
432            code: "mcp_client_config_foreign".into(),
433            message: format!(
434                "{} exists and does not name {LAUNCHER}, so {} does not start this repository's server from it",
435                c.config, c.title
436            ),
437            remedy: format!(
438                "add a `majordomus` server to {} whose command is {command}",
439                c.config
440            ),
441        })
442        .collect();
443    if !clients.is_empty()
444        && clients
445            .iter()
446            .all(|c| c.standing == McpClientStanding::Absent)
447    {
448        findings.push(McpFinding {
449            code: "mcp_no_client_configured".into(),
450            message: "no declared client has a configuration in this repository, so none of them starts its server".into(),
451            remedy: format!(
452                "write one of {} naming the command {LAUNCHER}; docs/MCP.md shows each form",
453                clients
454                    .iter()
455                    .map(|c| c.config.as_str())
456                    .collect::<Vec<_>>()
457                    .join(", ")
458            ),
459        });
460    }
461    findings
462}

468/// The projection as the capability answers it, for a caller inside this program that
469/// already holds the context: `mcp --inspect` prints the same value the tool, the resource
470/// and the route are asked for, and reading it cannot fail, because nothing in it is
471/// fetched — the registry, the peer board and the layer's index are in the context, and a
472/// client configuration that cannot be read is one that is absent.
473///
474/// ```
475/// use majordomus_cli::capability::builtin::mcp::{describe, McpProjectionInput};
476/// use majordomus_cli::capability::{builtin, CapabilityRegistry, Context, Effect};
477/// use majordomus_cli::git::GitState;
478/// use majordomus_cli::index::{Index, RepositoryInfo, State};
479/// use std::sync::Arc;
480/// // an index with no objects and no declared provider: the registry alone answers
481/// let index = Index {
482///     repository: RepositoryInfo {
483///         root: "/tmp/doc".into(), layer_schema: "ai-repository/v1".into(),
484///         sections: Default::default(), git: GitState::Unavailable { reason: "doc".into() },
485///         discovery: "filesystem".into(), source_classes: vec![], kind_sources: vec![],
486///         scope_origin: majordomus_cli::scope::Origin::Distribution, scope_path: String::new(),
487///         observed: Default::default(),
488///     },
489///     objects: vec![], diagnostics: vec![], state: State::Ok, fingerprint: String::new(),
490///     scoped: Default::default(), distribution: None, providers: Default::default(),
491///     share: None,
492/// };
493/// let registry = CapabilityRegistry::builder().with_builtin(builtin::all()).with_index(&index).build().unwrap();
494/// let ctx = Context::new(Arc::new(index), Arc::new(registry));
495///
496/// let all = describe(&ctx, McpProjectionInput::default());
497/// assert_eq!(all.tools.len(), all.tool_count);
498/// assert_eq!(all.effects.iter().map(|e| e.tools).sum::<usize>(), all.tool_count);
499/// // no provider is declared, so there is no client and nothing to be done about one
500/// assert!(all.clients.is_empty() && all.findings.is_empty());
501///
502/// // one effect narrows the tools listed and nothing else
503/// let writers = describe(&ctx, McpProjectionInput { effect: Some(Effect::RepositoryMutation) });
504/// assert_eq!(writers.tool_count, all.tool_count);
505/// assert_eq!(writers.tools.len(), all.writers.len());
506/// assert!(writers.tools.iter().all(|t| all.writers.contains(&t.name)));
507/// ```
508pub fn describe(ctx: &Context, input: McpProjectionInput) -> McpProjection {
509    use crate::mcp::protocol;
510    use crate::peers::Transport;
511
512    let all = tools(&ctx.registry);
513    let effects = effect_counts(&all);
514    let writers = all
515        .iter()
516        .filter(|t| t.effect == Effect::RepositoryMutation)
517        .map(|t| t.name.clone())
518        .collect();
519
520    let peers = ctx.peers.list();
521    let attached = |t: Transport| {
522        peers
523            .iter()
524            .filter(|p| p.attached && p.transport == t)
525            .count()
526    };
527
528    let root = Path::new(&ctx.index.repository.root);
529    let launcher_in_repository = root.join("bin").join(LAUNCHER).is_file();
530    let mut clients: Vec<McpClient> = ctx
531        .index
532        .providers
533        .providers
534        .iter()
535        .filter_map(|d| {
536            let config = d.client_config.clone()?;
537            Some(McpClient {
538                id: d.id.clone(),
539                title: d.title.clone(),
540                standing: standing(root, &config),
541                config,
542            })
543        })
544        .collect();
545    crate::order::canonical(&mut clients);
546
547    let findings = client_findings(&clients, launcher_in_repository);
548
549    let builtin = ctx
550        .registry
551        .iter()

608/// The `mcp` module: the one capability that describes the MCP projection, composed into
609/// the application like every other module, so that describing the projection is itself a
610/// tool, a resource and a route of it.
611///
612/// ```
613/// use majordomus_cli::capability::builtin::mcp;
614/// let m = mcp::module();
615/// assert_eq!(m.id.as_str(), "mcp");
616/// let c = &m.capabilities[0].capability;
617/// assert_eq!(c.exposure.mcp.as_ref().unwrap().tool.as_deref(), Some("majordomus_mcp"));
618/// assert_eq!(c.exposure.mcp.as_ref().unwrap().resource.as_ref().unwrap().uri, mcp::MCP_URI);
619/// ```
620pub fn module() -> ModuleDescriptor {
621    module! {
622        id: "mcp",
623        title: "MCP projection",
624        description: "The MCP projection of this registry, described as one typed value: who answers `initialize` and with which protocol versions, the transports and how many sessions each holds, what of the protocol is served and what is not, every tool with the effect and the hints its annotations are derived from, the tools that write the repository, the resources, and where each declared client's configuration stands in this repository. Nothing in it is a list of its own: the tools are the registry's, the effect is the capability's classification, the clients are the distribution's providers.",
625        stability: Stability::Experimental,
626        capabilities: [
627            capability! {
628                id: "mcp.projection",
629                title: "The MCP projection of this registry",
630                description: "Who answers `initialize` and with which protocol versions; the transports and the sessions attached over each; the methods served, and that prompts and notifications are not; every tool (or only those with one effect) with its capability, effect and the hints its annotations are derived from; the tools that write the repository; the resources; and whether each client the distribution declares has a configuration here that starts this repository's server. Reads the registry and one file per declared client; writes nothing.",
631                input: McpProjectionInput,
632                output: McpProjection,
633                stability: Stability::Experimental,
634                exposure: Exposure {
635                    mcp: Some(McpExposure {
636                        tool: Some("majordomus_mcp".into()),
637                        resource: Some(McpResource { uri: MCP_URI.into(), name: "mcp".into() }),
638                    }),
639                    http: get("/api/v1/mcp"),
640                    cli: None,
641                },
642                tags: ["mcp", "introspection", "projections"],
643                // the sessions attached are read from the board at the moment of asking
644                cache: CachePolicy::Disabled,
645                handler: projection,
646            },
647        ],
648    }
649}

663    /// The listing is the registry's and nothing else's: a capability with a tool exposure

675            declared

678        for t in &listed {

680            assert_eq!(c.id.as_str(), t.capability);
681            assert_eq!(c.execution.effect, t.effect);
682            assert_eq!(c.execution.hints(), t.hints);
683        }
