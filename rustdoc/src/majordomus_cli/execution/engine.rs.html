1//! The engine: what turns a capability call into an execution anybody can watch.
2//!
3//! It owns no behaviour of its own. Everything it runs is a capability of the registry,
4//! reached through [`crate::capability::Context::execute`] — the same executor MCP, the
5//! HTTP routes, the command line and the benchmarks call — so a browser, an agent and a
6//! terminal reach one implementation and there is no second copy of anything to drift.
7//!
8//! ```text
9//!   submit ── validate ── create ── queue ──► worker thread
10//!                                               │
11//!                          Context.progress ────┤ ctx.execute(id, input)
12//!                                               │        │
13//!                                               │        └─► the capability's own handler
14//!                                               ▼
15//!                                       ExecutionStore ──► subscribers ──► WebSocket
16//! ```
17//!
18//! It is deliberately small: an in-memory store, a bounded number of worker threads, and
19//! a cooperative cancellation flag. There is no durable queue, no retry and no scheduler,
20//! because nothing here is a background job system — it is a way to watch a call that
21//! takes longer than a request should be held open for.
22

40/// Why an execution was not accepted. Refused before anything is created, so a client that
41/// is told no knows nothing started.
42#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
43pub enum SubmitError {
44    /// No capability of that id.
45    #[error("unknown capability: {0}")]
46    UnknownCapability(String),
47    /// A resource is read, not executed; a planned or unsupported capability is listed and
48    /// never run.
49    #[error("not executable: {0}")]
50    NotExecutable(String),
51    /// The input does not satisfy the capability's input schema.
52    #[error("invalid input: {0}")]
53    InvalidInput(String),
54    /// The engine is stopping and accepts nothing further.
55    #[error("unavailable: {0}")]
56    Unavailable(String),
57    /// The idempotency key already started an execution of this capability with a
58    /// different input (I4000).
59    #[error("conflict: {0}")]
60    Conflict(KeyConflict),
61}
62
63impl SubmitError {
64    /// The stable code a transport maps to its own vocabulary.
65    pub fn code(&self) -> &'static str {
66        match self {
67            SubmitError::UnknownCapability(_) => "unknown_capability",
68            SubmitError::NotExecutable(_) => "action_unavailable",
69            SubmitError::InvalidInput(_) => "validation_error",
70            SubmitError::Unavailable(_) => "unavailable",
71            SubmitError::Conflict(_) => "conflict",
72        }
73    }
74}

76/// What a start was answered with: the execution, and whether it is one an earlier start
77/// with the same idempotency key began rather than one this start did.
78///
79/// ```
80/// use majordomus_cli::execution::{Actor, ActorKind, IdempotencyKey, Submitted};
81/// use majordomus_cli::synthetic::SyntheticRepository;
82/// let repo = SyntheticRepository::small().unwrap();
83/// let ctx = repo.context().unwrap();
84/// let start = || ctx.executions.submit_once(
85///     &ctx,
86///     "executions.demonstrate",
87///     serde_json::json!({ "steps": 1, "delay_ms": 0 }),
88///     Actor::of(ActorKind::Internal),
89///     Some(IdempotencyKey::parse("click-1").unwrap()),
90/// );
91/// let first: Submitted = start().unwrap();
92/// let again: Submitted = start().unwrap();
93/// assert!(!first.deduplicated && again.deduplicated);
94/// assert_eq!(first.execution.id, again.execution.id, "one execution");
95/// ```
96#[derive(Debug, Clone)]
97pub struct Submitted {
98    /// The execution, as accepted, or as it stands when the key had already started it.
99    pub execution: Execution,
100    /// `true` when nothing was started: the key had already started this execution with
101    /// the same input, and this is that execution.
102    pub deduplicated: bool,
103}

192    /// Check an input against a capability's schema, with the compiled validator kept.
193    ///
194    /// The handler validates again when it deserialises, and that is what is authoritative;
195    /// this exists so that a client learns its input is wrong from the request that sent
196    /// it, with the constraint named, rather than from an execution that fails a moment
197    /// later.
198    fn validate(&self, c: &Capability, input: &Value) -> Result<(), SubmitError> {
199        let validator = {
200            let mut cache = self.validators.lock().unwrap_or_else(|e| e.into_inner());
201            match cache.get(c.id.as_str()) {
202                Some(v) => Arc::clone(v),
203                None => match jsonschema::validator_for(&c.input.schema) {
204                    Ok(v) => {
205                        let v = Arc::new(v);
206                        cache.insert(c.id.to_string(), Arc::clone(&v));
207                        v
208                    }
209                    // a schema this crate derived that will not compile is a defect of the
210                    // crate, not of the caller: say so and let the handler decide
211                    Err(e) => {
212                        tracing::warn!(capability_id = %c.id, "the input schema does not compile: {e}");
213                        return Ok(());
214                    }
215                },
216            }
217        };
218        let problems: Vec<String> = validator
219            .iter_errors(input)
220            .take(5)
221            .map(|e| {
222                let at = e.instance_path().to_string();
223                if at.is_empty() {
224                    e.to_string()
225                } else {
226                    format!("{at}: {e}")
227                }
228            })
229            .collect();
230        if problems.is_empty() {
231            Ok(())
232        } else {
233            Err(SubmitError::InvalidInput(problems.join("; ")))
234        }
235    }

237    /// Accept a capability call as an execution and answer with its first snapshot.
238    ///
239    /// Fast on purpose: the input is checked, the record is created, the work is queued and
240    /// this returns. Nothing waits for the handler.
241    ///
242    /// The answer is the execution *as accepted* — `queued`, sequence 2, the `execution.queued`
243    /// event already published — and never whatever a worker has since made of it. That is
244    /// what `docs/EXECUTIONS.md` promises a client of `executions.start`, and it is not a
245    /// courtesy: reading the record again after the work became dispatchable would answer
246    /// `queued`, `running` or even `succeeded` depending on how the machine was loaded that
247    /// millisecond, and the same window would let a worker publish `execution.started` before
248    /// `execution.queued`, whereupon the state machine refuses the queued event and the
249    /// stream loses it. Acceptance is therefore recorded and read while the queue lock is
250    /// held, before `Self::dispatch` can hand the pending to anybody.
251    pub fn submit(
252        &self,
253        ctx: &Context,
254        capability: &str,
255        input: Value,
256        actor: Actor,
257    ) -> Result<Execution, SubmitError> {
258        self.submit_once(ctx, capability, input, actor, None)
259            .map(|s| s.execution)
260    }

262    /// [`Self::submit`], held to an idempotency key when the client sent one (I4000).
263    ///
264    /// The key is scoped to this repository and this capability, and is matched with the
265    /// digest of the input in canonical JSON: the same key with the same input is answered
266    /// the execution it started, `deduplicated`, and nothing is queued; the same key with
267    /// another input is refused with [`SubmitError::Conflict`], naming that execution. The
268    /// lookup and the record are one step of the store, so two starts sent at once with
269    /// one key start one execution. The store holds a key for as long as it holds the
270    /// execution the key started, and is in memory: a key does not outlive the server.
271    ///
272    /// ```
273    /// use majordomus_cli::execution::{Actor, ActorKind, IdempotencyKey};
274    /// use majordomus_cli::synthetic::SyntheticRepository;
275    /// let repo = SyntheticRepository::small().unwrap();
276    /// let ctx = repo.context().unwrap();
277    /// let start = || ctx.executions.submit_once(
278    ///     &ctx,
279    ///     "executions.demonstrate",
280    ///     serde_json::json!({ "steps": 1, "delay_ms": 0 }),
281    ///     Actor::of(ActorKind::Internal),
282    ///     Some(IdempotencyKey::parse("click-1").unwrap()),
283    /// );
284    /// let first = start().unwrap();
285    /// let again = start().unwrap();
286    /// assert!(!first.deduplicated && again.deduplicated);
287    /// assert_eq!(first.execution.id, again.execution.id, "one execution");
288    /// ```
289    pub fn submit_once(
290        &self,
291        ctx: &Context,
292        capability: &str,
293        input: Value,
294        actor: Actor,
295        idempotency_key: Option<IdempotencyKey>,
296    ) -> Result<Submitted, SubmitError> {
297        self.submit_with(
298            ctx,
299            capability,
300            input,
301            actor,
302            StartOptions {
303                idempotency_key,
304                issue: None,
305            },
306        )
307    }

309    /// [`Self::submit_once`], for an issue when the start names one (I4004).
310    ///
311    /// An execution started for an issue holds that issue's exclusive mesh claim while it
312    /// runs ([`super::claim`]): the worker that picks it up takes the claim before entering
313    /// the handler and moves it `queued ──► claimed ──► running`; a claim another session
314    /// holds refuses it, with the holder named, and nothing runs; the claim is given back at
315    /// the final state, and kept for an outcome that is not known. With no mesh cooperation
316    /// in this process it runs unclaimed, and its `execution.issue_claim` event says so.
317    /// The issue is part of what an idempotency key is matched against.
318    ///
319    /// ```
320    /// use majordomus_cli::execution::{Actor, ActorKind, ClaimStanding, StartOptions};
321    /// use majordomus_cli::synthetic::SyntheticRepository;
322    /// let repo = SyntheticRepository::small().unwrap();
323    /// let ctx = repo.context().unwrap();
324    /// let started = ctx.executions.submit_with(
325    ///     &ctx,
326    ///     "executions.demonstrate",
327    ///     serde_json::json!({ "steps": 1, "delay_ms": 0 }),
328    ///     Actor::of(ActorKind::Internal),
329    ///     StartOptions { issue: Some("I4004".into()), ..StartOptions::default() },
330    /// ).unwrap();
331    /// let claim = started.execution.issue_claim.unwrap();
332    /// assert_eq!((claim.issue.as_str(), claim.standing), ("I4004", ClaimStanding::Wanted));
333    /// ```
334    pub fn submit_with(
335        &self,
336        ctx: &Context,
337        capability: &str,
338        input: Value,
339        actor: Actor,
340        options: StartOptions,
341    ) -> Result<Submitted, SubmitError> {
342        let StartOptions {
343            idempotency_key,
344            issue,
345        } = options;
346        let issue = issue
347            .as_deref()
348            .map(super::model::parse_issue)
349            .transpose()
350            .map_err(SubmitError::InvalidInput)?;
351        if self.stopping.load(Ordering::SeqCst) {
352            return Err(SubmitError::Unavailable(
353                "this server is shutting down and is accepting no further executions".into(),
354            ));
355        }
356        let c = ctx
357            .registry
358            .get(capability)
359            .ok_or_else(|| SubmitError::UnknownCapability(capability.to_string()))?;
360        if !c.kind.is_executable() {
361            return Err(SubmitError::NotExecutable(format!(
362                "'{}' is a resource: it is read, never executed. Read it with objects.get",
363                c.id
364            )));
365        }
366        if !c.stability.executable() {
367            return Err(SubmitError::NotExecutable(format!(
368                "'{}' is {:?} and no projection executes it",
369                c.id, c.stability
370            )));
371        }
372        let input = if input.is_null() {
373            Value::Object(serde_json::Map::new())
374        } else {
375            input
376        };
377        self.validate(c, &input)?;
378
379        let id = ExecutionId::fresh();
380        let stored = redact(&c.input.schema, &input);
381        // the issue is part of the request a key names: one key, one input, two issues is
382        // two requests. Without an issue the digest is the input's alone, as it always was.
383        let digested = match &issue {
384            Some(issue) => serde_json::json!({ "input": &input, "issue": issue }),
385            None => input.clone(),
386        };
387        let idempotency = idempotency_key.map(|key| Idempotency {
388            key,
389            digest: crate::policy::sha256_hex(&crate::capability::executor::canonical_json(
390                &digested,
391            )),
392        });
393        let seed = Seed {
394            capability: c.id.as_str(),
395            title: &c.title,
396            input: stored,
397            cancellable: c.execution.cancellable,
398            actor,
399            repository: repository_of(ctx),
400        };
401        match self.store.create_once(id.clone(), seed, idempotency) {
402            Ok(Admission::Created(_)) => {}
403            Ok(Admission::Existing(execution)) => {
404                return Ok(Submitted {
405                    execution: *execution,
406                    deduplicated: true,
407                })
408            }
409            Err(conflict) => return Err(SubmitError::Conflict(conflict)),
410        }
411        if let Some(issue) = &issue {
412            self.store
413                .publish(&id, EventPayload::IssueClaim(IssueClaim::wanted(issue)));
414        }
415        let pending = Pending {
416            id: id.clone(),
417            capability: c.id.to_string(),
418            title: c.title.to_string(),
419            input,
420            ctx: ctx.clone(),
421            issue,
422        };
423        // The queue lock is held across the queued event and the snapshot: a worker
424        // finishing another execution calls `dispatch` too, and it needs this lock to take
425        // the pending, so nothing can start what has not yet been announced as accepted.
426        // The nesting is one way — this is the only place that takes the store's lock while
427        // holding the engine's, the store knows nothing of the engine, and the store's
428        // fan-out never blocks — so it cannot deadlock.
429        let accepted = {
430            let mut inner = self.lock();
431            inner.queue.push_back(pending);
432            let ahead = inner.queue.len() - 1;
433            self.store.publish(&id, EventPayload::Queued { ahead });
434            self.store.get(&id)
435        };
436        self.dispatch();
437        // the store evicts only finished executions, so the snapshot is there unless a cancel
438        // and a full store forgot it in between; the refusal for that is made either way
439        let forgotten =
440            SubmitError::Unavailable("the execution was forgotten as it was created".into());
441        accepted
442            .map(|execution| Submitted {
443                execution,
444                deduplicated: false,
445            })
446            .ok_or(forgotten)
447    }

449    /// Start whatever may start now: the bound on concurrent handlers, and the serial
450    /// policy of a capability that must not overlap with itself.
451    fn dispatch(&self) {
452        loop {
453            let next = {
454                let mut inner = self.lock();
455                if inner.running >= self.store.limits().max_running {
456                    return;
457                }
458                let position = inner.queue.iter().position(|p| {
459                    let serial = self
460                        .concurrency_of(&p.ctx, &p.capability)
461                        .is_some_and(|c| c == Concurrency::Serial);
462                    !serial || !inner.busy.contains_key(&p.capability)
463                });
464                let Some(position) = position else { return };
465                let Some(pending) = inner.queue.remove(position) else {
466                    return;
467                };
468                inner.running += 1;
469                *inner.busy.entry(pending.capability.clone()).or_insert(0) += 1;
470                inner.workers.retain(|w| !w.is_finished());
471                pending
472            };
473            let store = Arc::clone(&self.store);
474            let id = next.id.clone();
475            let capability = next.capability.clone();
476            let done_with = capability.clone();
477            let engine = Arc::clone(&next.ctx.executions);
478            let worker = std::thread::Builder::new()
479                .name(format!(
480                    "execution-{}",
481                    &id.as_str()[..12.min(id.as_str().len())]
482                ))
483                .spawn(move || {
484                    run(&store, next);
485                    let mut inner = engine.lock();
486                    inner.running = inner.running.saturating_sub(1);
487                    if let Some(count) = inner.busy.get_mut(&done_with) {
488                        *count -= 1;
489                        if *count == 0 {
490                            inner.busy.remove(&done_with);
491                        }
492                    }
493                    drop(inner);
494                    engine.dispatch();
495                });
496            match worker {
497                Ok(handle) => self.lock().workers.push(handle),
498                Err(e) => {
499                    // the thread could not be spawned: the execution fails now rather than
500                    // sitting queued for a worker that will never come
501                    let mut inner = self.lock();
502                    inner.running = inner.running.saturating_sub(1);
503                    inner.busy.remove(&capability);
504                    drop(inner);
505                    self.store.publish(
506                        &id,
507                        EventPayload::Failed {
508                            error: ExecutionError {
509                                code: "internal".into(),
510                                message: format!("no worker could be started: {e}"),
511                                suggestion: None,
512                                correlation_id: id.to_string(),
513                            },
514                        },
515                    );
516                }
517            }
518        }
519    }

532    /// Stop accepting executions, ask every one that is running to stop, and wait for the
533    /// workers, up to `grace`.
534    ///
535    /// A handler that does not look at its cancellation flag is not killed: this process
536    /// is about to end, and interrupting a read half way leaves nothing behind that
537    /// matters. What the wait buys is that a client watching gets the final event.
538    pub fn shutdown(&self, grace: std::time::Duration) {
539        self.stopping.store(true, Ordering::SeqCst);
540        for execution in self.store.list(None, None, usize::MAX) {
541            if execution.state.is_active() {
542                self.store.request_cancel(&execution.id, "server");
543            }
544        }
545        let deadline = std::time::Instant::now() + grace;
546        loop {
547            let done = {
548                let mut inner = self.lock();
549                inner.workers.retain(|w| !w.is_finished());
550                inner.workers.is_empty() && inner.queue.is_empty()
551            };
552            if done || std::time::Instant::now() >= deadline {
553                break;
554            }
555            std::thread::sleep(std::time::Duration::from_millis(10));
556        }
557        let stranded: Vec<ExecutionId> = self
558            .store
559            .list(None, None, usize::MAX)
560            .into_iter()
561            .filter(|e| e.state.is_active())
562            .map(|e| e.id)
563            .collect();
564        for id in stranded {
565            self.store.publish(&id, EventPayload::Cancelled);
566        }
567    }

585/// [`run`], with `between` called after the issue's claim is taken and before the start is
586/// published: the instant a cancel can land in, which a test holds open with it.
587fn run_with(store: &Arc<ExecutionStore>, pending: Pending, between: impl FnOnce()) {
588    let Pending {
589        id,
590        capability,
591        title,
592        input,
593        ctx,
594        issue,
595    } = pending;
596    let Some(cancel) = store.token(&id) else {
597        return;
598    };
599    // an execution for an issue holds the issue's claim before anything runs; `held` lives
600    // to the end of this function, so a worker that unwinds gives the claim back on its way
601    let held = match &issue {
602        Some(issue) => match claim_issue(store, &ctx, &id, &title, issue) {
603            Some(held) => held,
604            None => return,
605        },
606        None => None,
607    };
608    between();
609    // it may have been cancelled while it waited, up to this very instant: the start is
610    // refused from a final state under the lock a cancel takes, and a refused start runs
611    // nothing, where a look at the state first would leave that instant open
612    if store.publish(&id, EventPayload::Started).is_none() {
613        // claimed, and asked to stop before the handler was entered: nothing ran
614        let state = store.get(&id).map(|e| e.state);
615        if state == Some(ExecutionState::Cancelling) {
616            finish(store, &id, held, EventPayload::Cancelled);
617        } else if let Some(held) = held {
618            store.publish(&id, EventPayload::IssueClaim(held.settle(state)));
619        }
620        return;
621    }
622    let progress =
623        super::sink::Progress::reporting(Arc::clone(store), id.clone(), Arc::clone(&cancel));
624    let ctx = ctx.reporting(progress);
625    let span = tracing::info_span!("execution", execution_id = %id, capability_id = %capability);
626    let entered = span.enter();
627    let outcome = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
628        ctx.execute_observed(&capability, input)
629    }));
630    drop(entered);
631    let cancelled = cancel.load(Ordering::SeqCst);
632    let payload = match outcome {
633        Ok(Ok(output)) => EventPayload::Completed { output },
634        Ok(Err(error)) if cancelled => {
635            tracing::debug!(execution_id = %id, "the handler stopped after cancellation: {error}");
636            EventPayload::Cancelled
637        }
638        Ok(Err(error)) => EventPayload::Failed {
639            error: failure(&id, &error),
640        },
641        Err(panic) => {
642            let what = panic
643                .downcast_ref::<&str>()
644                .map(|s| (*s).to_string())
645                .or_else(|| panic.downcast_ref::<String>().cloned())
646                .unwrap_or_else(|| "a handler panicked".to_string());
647            // the message goes to this process's log, where an operator can read it; the
648            // client is told which execution it was and nothing about this crate's insides
649            tracing::error!(execution_id = %id, capability_id = %capability, "a handler panicked: {what}");
650            EventPayload::Failed {
651                error: ExecutionError {
652                    code: "internal".into(),
653                    message: "the handler failed unexpectedly; the server's log carries the detail"
654                        .into(),
655                    suggestion: Some(format!("search the server's output for execution_id={id}")),
656                    correlation_id: id.to_string(),
657                },
658            }
659        }
660    };
661    finish(store, &id, held, payload);
662}

664/// Take the issue's claim for an execution a worker picked up (I4004). `Some(held)` to go on
665/// — `held` is `None` when it runs unclaimed — and `None` when nothing may run: the claim was
666/// refused, or the execution was stopped while the claim was taken.
667fn claim_issue(
668    store: &Arc<ExecutionStore>,
669    ctx: &Context,
670    id: &ExecutionId,
671    title: &str,
672    issue: &str,
673) -> Option<Option<HeldClaim>> {
674    // stopped while it waited: nothing to claim for
675    if store.get(id).is_none_or(|e| e.state.is_final()) {
676        return None;
677    }
678    match super::claim::take(ctx, id, title, issue) {
679        Taken::Held(held) => {
680            store.publish(id, EventPayload::IssueClaim(held.claim().clone()));
681            let holder = held.claim().session.clone().unwrap_or_default();
682            if store
683                .publish(id, EventPayload::Claimed { holder })
684                .is_none()
685            {
686                // cancelled between the look above and the claim: give it back at once
687                let state = store.get(id).map(|e| e.state);
688                store.publish(id, EventPayload::IssueClaim(held.settle(state)));
689                return None;
690            }
691            Some(Some(held))
692        }
693        Taken::Unclaimed(claim) => {
694            store.publish(id, EventPayload::IssueClaim(claim));
695            let holder = "this process's execution engine, without a mesh claim".to_string();
696            store.publish(id, EventPayload::Claimed { holder })?;
697            Some(None)
698        }
699        Taken::Refused {
700            claim,
701            conflict,
702            message,
703        } => {
704            store.publish(id, EventPayload::IssueClaim(claim));
705            let error = ExecutionError {
706                code: if conflict { "refused" } else { "unavailable" }.into(),
707                message,
708                suggestion: Some(if conflict {
709                    "wait for the holder to finish, or ask it with majordomus_mesh_peers; the claim ends when its execution does".into()
710                } else {
711                    "read majordomus mesh doctor; nothing ran".into()
712                }),
713                correlation_id: id.to_string(),
714            };
715            let payload = if conflict {
716                EventPayload::Refused {
717                    error,
718                    policy: Some("mesh.claim".into()),
719                }
720            } else {
721                EventPayload::Failed { error }
722            };
723            store.publish(id, payload);
724            None
725        }
726    }
727}

729/// Publish an execution's last event, settling its issue claim first: released for a final
730/// state, kept for an outcome that is not known, so that the claim's event is said before the
731/// event a subscriber stops reading at.
732fn finish(
733    store: &Arc<ExecutionStore>,
734    id: &ExecutionId,
735    held: Option<HeldClaim>,
736    payload: EventPayload,
737) {
738    if let Some(held) = held {
739        let settled = held.settle(payload.implies_state());
740        store.publish(id, EventPayload::IssueClaim(settled));
741    }
742    store.publish(id, payload);
743}

745/// A capability error as an execution reports it.
746fn failure(id: &ExecutionId, error: &CapabilityError) -> ExecutionError {
747    let (code, suggestion) = match error {
748        CapabilityError::InvalidInput(_) => (
749            "validation_error",
750            Some("check the input against the capability's input schema".to_string()),
751        ),
752        CapabilityError::NotFound(_) => ("not_found", None),
753        CapabilityError::Refused(_) => ("refused", None),
754        CapabilityError::Forbidden(_) => ("forbidden", None),
755        CapabilityError::Conflict(_) => ("conflict", None),
756        CapabilityError::Internal(_) => (
757            "internal",
758            Some(format!("search the server's output for execution_id={id}")),
759        ),
760    };
761    ExecutionError {
762        code: code.into(),
763        message: match error {
764            CapabilityError::InvalidInput(m)
765            | CapabilityError::NotFound(m)
766            | CapabilityError::Refused(m)
767            | CapabilityError::Internal(m)
768            | CapabilityError::Forbidden(m)
769            | CapabilityError::Conflict(m) => m.clone(),
770        },
771        suggestion,
772        correlation_id: id.to_string(),
773    }
774}

776/// Which repository this process runs executions against. Never the request's idea of it:
777/// a process serves the one repository it read at start-up.
778pub(crate) fn repository_of(ctx: &Context) -> RepositoryRef {
779    let root = std::path::Path::new(&ctx.index.repository.root);
780    RepositoryRef {
781        name: root
782            .file_name()
783            .map(|n| n.to_string_lossy().into_owned())
784            .unwrap_or_else(|| "repository".into()),
785        id: crate::repository::identity(root),
786        branch: match &ctx.index.repository.git {
787            crate::git::GitState::Available(info) => info.branch.clone(),
788            crate::git::GitState::Unavailable { .. } => None,
789        },
790    }
791}

793/// Is this capability one an execution may run at all? The same question `submit` asks,
794/// answered without submitting, for a projection that lists what a client may start.
795///
796/// ```
797/// use majordomus_cli::capability::{CapabilityKind, Stability};
798/// use majordomus_cli::execution::engine::runnable;
799/// assert!(runnable(CapabilityKind::Query, Stability::Implemented).is_none());
800/// assert!(runnable(CapabilityKind::Resource, Stability::Implemented).is_some());
801/// assert!(runnable(CapabilityKind::Query, Stability::Planned).is_some());
802/// ```
803pub fn runnable(kind: CapabilityKind, stability: crate::capability::Stability) -> Option<String> {
804    if !kind.is_executable() {
805        return Some("it is a resource: read it rather than running it".into());
806    }
807    if !stability.executable() {
808        return Some(format!(
809            "it is {}, and no projection executes it",
810            format!("{stability:?}").to_lowercase()
811        ));
812    }
813    None
814}
