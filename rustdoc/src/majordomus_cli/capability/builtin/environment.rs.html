1//! The `environment` module: what this checkout is, as a capability.
2//!
3//! One capability, one handler, and every surface a projection of it — the MCP tool and
4//! resource, the HTTP route, the OpenAPI operation, the Cockpit page and the command line
5//! all answer with the value [`crate::environment::resolve()`] built. The direnv banner is
6//! the same value rendered for a terminal.
7//!
8//! The handler runs inside a process that already holds the index, so it resolves in full:
9//! nothing here pays the index build, because whoever called it had already paid it. The
10//! fast, cached resolution belongs to the command line, which may be starting cold on a
11//! shell prompt.

35/// What a served preflight may cost.
36///
37/// ```
38/// use majordomus_cli::capability::builtin::environment::PreflightInput;
39/// let input: PreflightInput = serde_json::from_str("{}").unwrap();
40/// assert!(!input.probe, "a served request asks no other process by default");
41/// ```
42#[derive(Debug, Default, Serialize, Deserialize, JsonSchema)]
43#[serde(deny_unknown_fields)]
44pub struct PreflightInput {
45    /// Ask the loopback address a server of this checkout published, and its peer board.
46    /// Off by default: a served request answers from what this process can see, and when
47    /// this process holds the checkout's lease that is itself the evidence.
48    #[serde(default)]
49    pub probe: bool,
50}
51
52impl BenchmarkCases for PreflightInput {
53    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
54        vec![NamedCase::new("sealed", PreflightInput { probe: false })]
55    }
56}

58fn preflight(ctx: &Context, input: PreflightInput) -> Result<Preflight, CapabilityError> {
59    let root = std::path::Path::new(&ctx.index.repository.root);
60    let repository = Repository::open(root).map_err(|e| {
61        CapabilityError::Internal(format!("the repository could not be re-read: {e}"))
62    })?;
63    // Located here rather than carried on the context, so that the provider projections are
64    // compared against their templates exactly as the command line compares them; without
65    // it they would read `unknown` on this surface and `verified` on that one.
66    let share = crate::share::Share::locate(None, root).ok();
67    let environment = resolve(
68        &Inputs {
69            repository: &repository,
70            share: share.as_ref(),
71            index: Some(&ctx.index),
72            registry: Some(&ctx.registry),
73            policy: None,
74        },
75        &EnvironmentQuery {
76            resolution: Resolution::Full,
77            probe_services: false,
78            use_cache: false,
79            write_cache: false,
80        },
81    );
82    let ledger =
83        crate::evidence::Ledger::load(root).unwrap_or_else(|_| crate::evidence::Ledger::empty());
84    let tally = RulesTally::of(&crate::rules::report(&ctx.index, &ledger));
85    let policy = crate::policy::LoadedPolicy::load(&repository).map_err(|e| e.to_string());
86    let mut observations = preflight::observe(
87        root,
88        &environment,
89        policy.as_ref().map_err(Clone::clone),
90        Some(tally),
91        // never the cache: a served request writes nothing to the checkout it serves
92        Probe {
93            cache: false,
94            ..if input.probe {
95                Probe::asked()
96            } else {
97                Probe::sealed()
98            }
99        },
100    );
101    // The process that holds the lease holds the board: read it here rather than asking
102    // this process over its own socket, and the command line — which asks over the socket —
103    // reads the same peers.
104    if observations.server.this_process {
105        observations.peers = Ok(preflight::PeersObservation::of(&ctx.peers.list()));
106    }
107    Ok(preflight::derive(&observations))
108}

110/// What to include in the snapshot.
111#[derive(Debug, Default, Serialize, Deserialize, JsonSchema)]
112#[serde(deny_unknown_fields)]
113pub struct EnvironmentInput {
114    /// Contact the local address a running server published, to say whether it answers.

156/// Where the values of one snapshot came from.
157#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
158pub struct EnvironmentProvenance {
159    /// How the snapshot behind these entries was resolved.
160    pub resolution: Resolution,
161    /// One entry per field, in the order the resolver produced them.
162    pub fields: Vec<FieldSource>,
163}
164
165fn snapshot(
166    ctx: &Context,
167    input: EnvironmentInput,
168) -> Result<RepositoryEnvironment, CapabilityError> {
169    let root = std::path::Path::new(&ctx.index.repository.root);
170    let repository = Repository::open(root).map_err(|e| {
171        CapabilityError::Internal(format!("the repository could not be re-read: {e}"))
172    })?;
173    // The share directory is not carried on the context; without it the provider
174    // projections are still listed from the policy, and their state is `unknown` rather
175    // than guessed. `env status` on the command line has it and reports the state.
176    Ok(resolve(
177        &Inputs {
178            repository: &repository,
179            share: None,
180            index: Some(&ctx.index),
181            registry: Some(&ctx.registry),
182            policy: None,
183        },
184        &EnvironmentQuery {
185            resolution: Resolution::Full,
186            probe_services: input.probe_services,
187            use_cache: false,
188            // A served request writes nothing: the cache belongs to the checkout's own
189            // command line, and a request that wrote it would be a request with a side
190            // effect on the repository.
191            write_cache: false,
192        },
193    ))
194}
195
196fn explain(ctx: &Context, input: ExplainInput) -> Result<EnvironmentProvenance, CapabilityError> {
197    let environment = snapshot(
198        ctx,
199        EnvironmentInput {
200            probe_services: false,
201        },
202    )?;
203    let fields = match &input.field {
204        None => environment.provenance.clone(),
205        Some(field) => {
206            let found: Vec<FieldSource> = environment
207                .provenance
208                .iter()
209                .filter(|p| p.field == *field || p.field.starts_with(&format!("{field}.")))
210                .cloned()
211                .collect();
212            if found.is_empty() {
213                return Err(CapabilityError::NotFound(format!(
214                    "no field named '{field}'; `environment.explain` with no field lists every one"
215                )));
216            }
217            found
218        }
219    };
220    Ok(EnvironmentProvenance {
221        resolution: environment.resolution,
222        fields,
223    })
224}

226/// The module.

228    module! {
229        id: "environment",
230        title: "Repository environment",
231        description: "What this checkout is right now: the project and its version, the repository and its layer, version control, the toolchains it declares, what the layer holds, the workflows a person can run, the provider projections and the local services — one typed snapshot, with a provenance entry for every value in it. The direnv banner, the Cockpit's overview and this route are renderings of the same value.",
232        stability: Stability::BehaviorallyVerified,
233        capabilities: [
234            capability! {
235                id: "environment.status",
236                title: "The repository environment",
237                description: "One snapshot of this checkout: project identity, repository identity, version control, declared toolchains, what the layer holds counted per kind, the workflows the runner describes, the provider projections against the policy that renders them, and the local services with the address a running server published. Every value carries where it came from.",
238                input: EnvironmentInput,
239                output: RepositoryEnvironment,
240                stability: Stability::BehaviorallyVerified,
241                exposure: Exposure {
242                    mcp: Some(McpExposure {
243                        tool: Some("majordomus_environment".into()),
244                        resource: Some(McpResource { uri: ENVIRONMENT_URI.into(), name: "environment".into() }),
245                    }),
246                    http: get("/api/v1/environment"),
247                    cli: None,
248                },
249                tags: ["environment", "repository", "introspection"],
250                // Not cached, and deliberately so. This answer is a snapshot of *now*:
251                // it runs `git status`, reads the lease, and stamps the moment it was
252                // taken into `generated_at`, so two calls a second apart legitimately
253                // differ. The executor holds that a cached capability answers the same
254                // value from the handler, from a cold cache and from a warm one
255                // (`tests/executor.rs`, `docs/claims/execution-cache-equivalence.md`), and
256                // a cache that may change the answer is a cache that lies. For a value
257                // whose whole product is its freshness the honest declaration is no cache
258                // — and the window bought nothing this module was not already paying:
259                // `environment.explain` runs the same full resolution uncached, and
260                // nothing polls the route on a timer. `project.cache-is-invisible` is the
261                // rule: a cache key is a content hash of what it derives from and never a
262                // timestamp, and a cache a caller can observe is a defect. This value
263                // cannot be made invisible while it carries when it was taken.
264                cache: CachePolicy::Disabled,
265                handler: snapshot,
266            },
267            capability! {
268                id: "environment.explain",
269                title: "Where an environment value came from",
270                description: "The provenance of the snapshot: for each field, what decided it — a compile-time constant, a file, a command, or the cache — which resolver read it, and how far it can be trusted. Narrow it to one field, or to a prefix, by name.",

277                    cli: None,
278                },
279                tags: ["environment", "provenance", "introspection"],
280                handler: explain,
281            },
282            capability! {
283                id: "environment.preflight",
284                title: "Whether Majordomus is in force here, and what proves it",
285                description: "One verdict per claim about this checkout — git; the episode, its briefing, the task and the handover; the policy, the rule corpus and the ADRs; the shared server and the MCP, API and Cockpit surfaces it serves; the peer board; recorded test runs, rule enforcement, provider projections, generated documentation and the deployment — each `verified`, `active`, `fresh`, `stale`, `degraded`, `unavailable`, `failed`, `unknown` or `not_applicable`, with the evidence it rests on. A verdict that asserts something is in force cannot be produced without evidence. The same value the command line prints and the entry banner summarises.",
286                input: PreflightInput,
287                output: Preflight,
288                stability: Stability::BehaviorallyVerified,
289                exposure: Exposure {
290                    mcp: Some(McpExposure {
291                        tool: Some("majordomus_preflight".into()),

293                    }),
294                    http: get("/api/v1/environment/preflight"),

297                tags: ["environment", "governance", "verification", "evidence"],

311    /// The declaration is the only place these names exist, and every projection — the MCP
312    /// tool, the resource, the HTTP route, the OpenAPI operation, the benchmark target —
313    /// is derived from it. A refactor that dropped an exposure would still compile.
314    #[test]
315    fn the_declaration_yields_the_projections_it_claims() {
316        let m = module();
317        assert_eq!(m.id.as_str(), "environment");
318        let expected: &[(&str, &str, &str)] = &[
319            (
320                "environment.status",
321                "majordomus_environment",
322                "/api/v1/environment",
323            ),
324            (
325                "environment.explain",
326                "majordomus_environment_explain",
327                "/api/v1/environment/explain",
328            ),
329            (
330                "environment.preflight",
331                "majordomus_preflight",
332                "/api/v1/environment/preflight",
333            ),
334        ];
335        let ids: Vec<&str> = m
336            .capabilities
337            .iter()
338            .map(|e| e.capability.id.as_str())
339            .collect();
340        let want: Vec<&str> = expected.iter().map(|(id, _, _)| *id).collect();
341        assert_eq!(
342            ids, want,
343            "the module declares a different set of capabilities"
344        );
345        for (executable, (id, tool, path)) in m.capabilities.iter().zip(expected) {
346            let exposure = &executable.capability.exposure;
347            assert_eq!(
348                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
349                Some(*tool),
350                "{id} lost or renamed its MCP tool"
351            );
352            assert_eq!(
353                exposure.http.as_ref().map(|h| h.path.as_str()),
354                Some(*path),
355                "{id} lost or renamed its HTTP route"
356            );
357        }
358        let resource = m.capabilities[0]
359            .capability
360            .exposure
361            .mcp
362            .as_ref()
363            .and_then(|m| m.resource.as_ref())
364            .expect("the snapshot is readable as a resource");
365        assert_eq!(resource.uri, ENVIRONMENT_URI);
366    }

368    /// A snapshot of *now* must not be served from a cache. The value stamps the moment
369    /// it was taken into `generated_at`, and the executor holds that a cached capability
370    /// answers the same value from the handler, from a cold cache and from a warm one; the
371    /// two together made `tests/executor.rs` fail whenever a pair of calls straddled a
372    /// second boundary — a test that failed on the calendar. The declaration was the
373    /// defect, not the invariant.
374    #[test]
375    fn the_snapshot_is_not_cached_because_it_carries_the_moment_it_was_taken() {
376        let m = module();
377        let status = m
378            .capabilities
379            .iter()
380            .find(|e| e.capability.id.as_str() == "environment.status")
381            .expect("the module declares the snapshot");
382        assert!(
383            !status.capability.cache.is_enabled(),
384            "environment.status answers with a wall clock in it, so it cannot be cached"
385        );
386    }

388    /// A served request must not write to the checkout it is serving. The whole cache is
389    /// the command line's, and a handler that wrote it would give an HTTP GET a side
390    /// effect on the repository.
391    #[test]
392    fn a_served_snapshot_writes_nothing() {
393        let repo = crate::synthetic::SyntheticRepository::small().expect("a repository");
394        let ctx = repo.context().expect("a context");
395        let cache = crate::environment::Cache::path(repo.root(), ".ai/local");
396        let environment = snapshot(
397            &ctx,
398            EnvironmentInput {
399                probe_services: false,
400            },
401        )
402        .expect("a snapshot");
403        assert_eq!(environment.resolution, Resolution::Full);
404        assert!(
405            !cache.exists(),
406            "a served request wrote the checkout's cache"
407        );
408    }
