1//! Pull-request integration: the next provably safe change, onto the current master, one
2//! at a time (ADR 0101).
3//!
4//! # The pipeline
5//!
6//! ```text
7//! forge (gh) ──► ForgeObservation ──► relation to master (git merge-tree, cached by SHA pair)
8//!                (recorded locally)            │
9//!                                              ▼
10//!                               classify ──► PullRequestAssessment (one per PR, with evidence)
11//!                                              │
12//!                                              ▼
13//!                                 plan ──► IntegrationQueue (ranked, deterministic)
14//!                                              │
15//!           CLI · HTTP · MCP · Cockpit ◄───────┤
16//!                                              ▼
17//!                           drain: lease ► re-observe ► re-decide ► act ► verify ► re-plan
18//! ```text
19//!
20//! The observation is the only network product and it is recorded with its moment; every
21//! read surface renders it offline. A decision names the master and head it was taken
22//! against ([`EvaluatedAgainst`]), and the executor acts only when a fresh observation
23//! still says the same — a merge invalidates every earlier plan by construction, because
24//! the next step starts from a new observation of a new master.
25//!
26//! # Why this module is not public
27//!
28//! It is `pub(crate)`. Integration is an operation of this executable, reached through its
29//! command line (`majordomus prs`) and its capability registry (`integration.*` on HTTP, MCP
30//! and OpenAPI), and not a library other crates link against, so it is not part of the
31//! crate's exported API and the exported-surface gate (`quality report`) does not cover it.
32//! What covers it instead: the unit tests beside it (`tests.rs`: every disposition, the
33//! stale-decision refusal, the re-plan after every merge, the cleanup threshold, relations
34//! computed by real git, and two properties), and
35//! `test/cases/720_integration_follows_the_current_master.sh`, which drives the real
36//! command line against a scripted forge.
37//!
38//! # Why the forge's `mergeable` is not read
39//!
40//! This repository resolves derived files with a per-clone merge driver the forge cannot
41//! run, so the forge calls nearly every pull request conflicting. The relation to master is
42//! decided here, by git, with the drivers ([`relation`]).

106/// The repository's audit trail. A checkout that still carries a trail of its own from
107/// before the trail was the repository's (`.ai/local/state/integration/events.jsonl`) has it
108/// appended to the repository's the first time this is asked while the repository has none;
109/// a marker keeps that from ever happening twice. A second checkout's old trail is left
110/// where it is: appended after another's, its lines would fold out of order.
111pub fn events_path(root: &Path) -> Result<PathBuf, String> {
112    let path = common_state_path(root, EVENTS_FILE)?;
113    let old = state_path(root, EVENTS_FILE);
114    if path.exists() || !old.is_file() {
115        return Ok(path);
116    }
117    move_trail(&old, &path).map(|()| path)
118}
119
120/// Append `old` to the repository's trail at `path`, once. The marker is taken with
121/// `create_new` before anything is appended, so of two processes that both find no trail,
122/// one appends; and a trail removed afterwards is not refilled.
123fn move_trail(old: &Path, path: &Path) -> Result<(), String> {
124    let dir = path.parent().unwrap_or(path);
125    let marker = dir.join(TRAIL_MOVED_MARKER);
126    // a directory that cannot be made is said by the marker's open below, which then fails
127    let _ = std::fs::create_dir_all(dir);
128    match std::fs::OpenOptions::new()
129        .write(true)
130        .create_new(true)
131        .open(&marker)
132    {
133        Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => Ok(()),
134        taken => taken
135            .and_then(|mut m| writeln!(m, "{}", old.display()))
136            .and_then(|()| std::fs::read_to_string(old))
137            .and_then(|text| {
138                std::fs::OpenOptions::new()
139                    .create(true)
140                    .append(true)
141                    .open(path)
142                    .and_then(|mut trail| writeln!(trail, "{}", text.trim_end_matches('\n')))
143            })
144            .map_err(at(path)),
145    }
146}

148/// The last recorded observation, if any.
149pub fn load_observation(root: &Path) -> Result<Option<ForgeObservation>, String> {
150    let path = state_path(root, OBSERVATION_FILE);
151    let text = match std::fs::read_to_string(&path) {
152        Ok(t) => t,
153        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(None),
154        Err(e) => return Err(format!("{}: {e}", path.display())),
155    };
156    let obs: ForgeObservation =
157        serde_json::from_str(&text).map_err(|e| format!("{}: {e}", path.display()))?;
158    if obs.schema != OBSERVATION_SCHEMA {
159        return Err(format!(
160            "{}: schema {}, this reads {OBSERVATION_SCHEMA}",
161            path.display(),
162            obs.schema
163        ));
164    }
165    Ok(Some(obs))
166}

168fn write_atomic(path: &Path, text: &str) -> Result<(), String> {
169    if let Some(dir) = path.parent() {
170        std::fs::create_dir_all(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
171    }
172    let tmp = path.with_extension(format!("tmp.{}", std::process::id()));
173    std::fs::write(&tmp, text).map_err(|e| format!("{}: {e}", tmp.display()))?;
174    std::fs::rename(&tmp, path).map_err(|e| format!("{}: {e}", path.display()))
175}

183/// The relation cache: `master..head` → relation, and beside it under the same key what the
184/// merge changes by kind ([`ChangeShape`]). Both SHAs are immutable, and both answers read
185/// derived attributes from master's own `.gitattributes`, so an entry is true forever and a
186/// change of either commit stales both together; the file is bounded by dropping entries
187/// whose master is not the current.
188#[derive(Debug, Default, Serialize, Deserialize)]
189struct RelationCache {
190    entries: BTreeMap<String, RelationToMaster>,
191    #[serde(default)]
192    shapes: BTreeMap<String, ChangeShape>,
193}
194
195fn relation_cached(
196    root: &Path,
197    cache: &mut RelationCache,
198    master: &str,
199    head: &str,
200) -> RelationToMaster {
201    // a ref or an abbreviation may name another commit tomorrow: only a pair of commit ids
202    // is decided here, so that every key of the cache is a fact forever
203    if !is_object_id(master) || !is_object_id(head) {
204        return RelationToMaster::Unknown {
205            reason: format!("{master}..{head} is not a pair of full commit ids"),
206        };
207    }
208    let key = format!("{master}..{head}");
209    if let Some(r) = cache.entries.get(&key) {
210        return r.clone();
211    }
212    let r = relation::relation_to_master(root, master, head);
213    // an unknown is not a fact about the SHAs; it is not cached
214    if !matches!(r, RelationToMaster::Unknown { .. }) {
215        cache.entries.insert(key, r.clone());
216    }
217    r
218}

220/// [`relation::change_shape`] through the cache, under exactly the relation's key: only a pair
221/// of full commit ids is decided, and an unanswered shape is not cached.
222fn shape_cached(
223    root: &Path,
224    cache: &mut RelationCache,
225    master: &str,
226    head: &str,
227) -> Option<ChangeShape> {
228    let key = format!("{master}..{head}");
229    (is_object_id(master) && is_object_id(head))
230        .then(|| {
231            cache.shapes.get(&key).cloned().or_else(|| {
232                relation::change_shape(root, master, head)
233                    .inspect(|s| drop(cache.shapes.insert(key.clone(), s.clone())))
234            })
235        })
236        .flatten()
237}

239/// Whether `s` is a full commit id, SHA-1 or SHA-256, as git prints one.
240fn is_object_id(s: &str) -> bool {
241    matches!(s.len(), 40 | 64) && s.bytes().all(|b| matches!(b, b'0'..=b'9' | b'a'..=b'f'))
242}

244/// The master commit this clone has for `base`, as fetched.
245pub fn local_master(root: &Path, base: &str) -> Option<String> {
246    let out = std::process::Command::new("git")
247        .arg("-C")
248        .arg(root)
249        .args([
250            "rev-parse",
251            "--verify",
252            "--quiet",
253            &format!("refs/remotes/origin/{base}^{{commit}}"),
254        ])
255        .output()
256        .ok()?;
257    out.status
258        .success()
259        .then(|| String::from_utf8_lossy(&out.stdout).trim().to_string())
260        .filter(|s| !s.is_empty())
261}

313/// The policy of an observation: required checks and reviews as the forge reported them,
314/// the label policy declared once in [`LABEL_POLICY`], and the merge method — a merge commit
315/// when the repository's settings allow one (`allow_merge_commit`, which the observation
316/// carries as `merge`), and none otherwise. Never a squash or a rebase: the derived-file
317/// driver resolves merges, and either would replay commits it never saw. With none, every
318/// pull request not already on master is held (`merge_commit_not_allowed`); one that is stays
319/// `redundant`, since closing it needs no merge.
320pub fn policy_of(obs: &ForgeObservation) -> IntegrationPolicy {
321    let merge_method = obs
322        .merge_methods
323        .iter()
324        .any(|m| m == "merge")
325        .then(|| "merge".to_string());
326    IntegrationPolicy {
327        base: obs.base.clone(),
328        required_checks: obs.required_checks.clone(),
329        review_policy: obs.review_policy,
330        skipped_permitted: Vec::new(),
331        labels: LABEL_POLICY.to_vec(),
332        merge_method,
333        up_to_date_required: obs.up_to_date_required,
334    }
335}

337/// The rank key: lane, then disposition, then risk, then how many other *ready or
338/// refreshable* pull requests it overlaps (fewer first: landing it invalidates less), then how many declared dependents wait on it (more first: landing it unblocks
339/// them), then how many authored paths it changes (fewer first), then age (older first, so
340/// easy new work cannot starve old work), then number. Every component is a value of the
341/// assessment or of the queue around it; the order is total and input-order free.
342fn rank_factors(
343    a: &PullRequestAssessment,
344    contenders: &BTreeSet<u64>,
345    dependents: &BTreeMap<u64, usize>,
346) -> RankFactors {
347    RankFactors {
348        lane: a.lane,
349        disposition: a.disposition,
350        risk: a.risk,
351        contention: a
352            .overlaps
353            .iter()
354            .filter(|o| contenders.contains(&o.number))
355            .count(),
356        dependents: dependents.get(&a.number).copied().unwrap_or(0),
357        authored_paths: a.authored_paths.len(),
358        created_at: a.created_at.clone(),
359        number: a.number,
360    }
361}
362
363fn rank_key(
364    f: &RankFactors,
365) -> (
366    u8,
367    u8,
368    u8,
369    usize,
370    std::cmp::Reverse<usize>,
371    usize,
372    String,
373    u64,
374) {
375    (
376        f.lane as u8,
377        f.disposition as u8,
378        f.risk as u8,
379        f.contention,
380        std::cmp::Reverse(f.dependents),
381        f.authored_paths,
382        f.created_at.clone(),
383        f.number,
384    )
385}

387/// Rank assessments. Deterministic whatever order they arrive in; each carries the factors
388/// it was ranked by.
389pub fn rank(assessments: Vec<PullRequestAssessment>) -> Vec<PullRequestAssessment> {
390    let contenders: BTreeSet<u64> = assessments
391        .iter()
392        .filter(|a| {
393            matches!(
394                a.disposition,
395                PullRequestDisposition::Ready | PullRequestDisposition::NeedsRefresh
396            )
397        })
398        .map(|a| a.number)
399        .collect();
400    // declared, unsatisfied edges into each pull request: what landing it would unblock
401    let mut dependents: BTreeMap<u64, usize> = BTreeMap::new();
402    for a in &assessments {
403        for d in &a.dependencies {
404            if d.certainty == DependencyCertainty::Confirmed && !d.satisfied {
405                *dependents.entry(d.number).or_default() += 1;
406            }
407        }
408    }
409    let mut ranked: Vec<PullRequestAssessment> = assessments
410        .into_iter()
411        .map(|mut a| {
412            a.rank_factors = Some(rank_factors(&a, &contenders, &dependents));
413            a
414        })
415        .collect();
416    ranked.sort_by_cached_key(|a| rank_key(a.rank_factors.as_ref().expect("set above")));
417    ranked
418}

430/// Build the queue from an observation, against `master_sha`. `relation` answers what a
431/// head is to master; the command line passes git, a test passes a table. It is asked about
432/// every open pull request, and about every declared successor that is no longer open (a
433/// pull request built from what the forge reported of it): such a successor landed exactly
434/// when its head is `contained`. `shape` answers what each open pull request's merge changes
435/// by kind ([`ChangeShape`]): its authored paths decide overlaps and risk, and two that raise
436/// the version or change a release overlap as such. A pull request it does not answer is
437/// assessed on its relation's paths alone.
438pub fn build_queue_shaped(
439    obs: &ForgeObservation,
440    master_sha: &str,
441    mut relation: impl FnMut(&PullRequestObservation) -> RelationToMaster,
442    shape: impl FnMut(&PullRequestObservation) -> Option<ChangeShape>,
443) -> IntegrationQueue {
444    let policy = policy_of(obs);
445    let mut diagnostics = Vec::new();
446    if !obs.base_sha.is_empty() && obs.base_sha != master_sha {
447        diagnostics.push(format!(
448            "stale observation: the forge reported {} at {} but this clone's master is {}; majordomus prs refresh",
449            short(&obs.base_sha),
450            obs.observed_at,
451            short(master_sha)
452        ));
453    }
454    match &policy.required_checks {
455        None => diagnostics.push(
456            "the branch protection or rulesets could not be read: no pull request can be ready"
457                .into(),
458        ),
459        Some(r) if r.is_empty() => diagnostics.push(format!(
460            "{} requires no check, in its protection or any ruleset: a head can prove nothing \
461             to it, so no pull request can be ready (owner decision D5)",
462            policy.base
463        )),
464        Some(_) => {}
465    }
466    if policy.merge_method.is_none() {
467        diagnostics.push(format!(
468            "{} allows no merge commit (it allows: {}): the executor never squashes or rebases, \
469             so no pull request can be ready",
470            obs.repository,
471            if obs.merge_methods.is_empty() {
472                "nothing".to_string()
473            } else {
474                obs.merge_methods.join(", ")
475            }
476        ));
477    }
478    // in number order, whatever order the forge listed them in
479    let armed: BTreeSet<u64> = obs
480        .pull_requests
481        .iter()
482        .filter(|p| p.auto_merge)
483        .map(|p| p.number)
484        .collect();
485    if !armed.is_empty() {
486        diagnostics.push(format!(
487            "auto-merge is armed on {}: the forge would merge them on its own, outside the \
488             executor, so they are unsafe until it is disarmed (gh pr merge <n> --disable-auto)",
489            armed
490                .iter()
491                .map(|n| format!("#{n}"))
492                .collect::<Vec<_>>()
493                .join(" ")
494        ));
495    }
496    let relations: Vec<RelationToMaster> = obs.pull_requests.iter().map(&mut relation).collect();
497    let mut queue = QueueContext {
498        open: obs.pull_requests.iter().map(|p| p.number).collect(),
499        // a fork's branch is not a branch of this repository, whatever it is called: a fork
500        // named `master` would otherwise stack every pull request onto itself
501        heads: obs
502            .pull_requests
503            .iter()
504            .filter(|p| !p.cross_repository)
505            .map(|p| (p.head_ref.clone(), p.number))
506            .collect(),
507        authored: BTreeMap::new(),
508        shapes: obs
509            .pull_requests
510            .iter()
511            .map(shape)
512            .zip(&obs.pull_requests)
513            .filter_map(|(s, p)| s.map(|s| (p.number, s)))
514            .collect(),
515        superseded_by: BTreeMap::new(),
516        dependency_states: obs
517            .resolved
518            .iter()
519            .map(|(n, r)| {
520                let state = if r.merged {
521                    DependencyState::Merged
522                } else {
523                    DependencyState::ClosedUnmerged
524                };
525                (*n, state)
526            })
527            .collect(),
528        cycles: BTreeMap::new(),
529    };
530    queue.cycles = dependency_cycles(obs, &queue.heads);
531    if obs.pull_requests.len() >= forge::OPEN_LIMIT {
532        diagnostics.push(format!(
533            "the forge listed {} open pull requests, its limit: more may be open, and a dependency on one of those reads as unread",
534            obs.pull_requests.len()
535        ));
536    }
537    queue.superseded_by = successors(obs, &queue.open, &mut relation);
538    for (p, r) in obs.pull_requests.iter().zip(&relations) {
539        let authored = match (queue.shapes.get(&p.number), r) {
540            (Some(s), _) => s.authored.clone(),
541            (
542                None,
543                RelationToMaster::UpToDate { authored } | RelationToMaster::Behind { authored, .. },
544            ) => authored.clone(),
545            (None, RelationToMaster::Conflicting { paths }) => paths.clone(),
546            _ => Vec::new(),
547        };
548        queue.authored.insert(p.number, authored);
549    }
550    let assessments: Vec<PullRequestAssessment> = obs
551        .pull_requests
552        .iter()
553        .zip(&relations)
554        .map(|(p, r)| classify(p, r, master_sha, &obs.observed_at, &policy, &queue))
555        .collect();
556    let mut queue = IntegrationQueue {
557        repository: obs.repository.clone(),
558        base: obs.base.clone(),
559        master_sha: master_sha.to_string(),
560        observed_at: obs.observed_at.clone(),
561        observed_base_sha: obs.base_sha.clone(),
562        next_merge: None,
563        next_refresh: Vec::new(),
564        policy,
565        assessments: rank(assessments),
566        tallies: QueueTallies::default(),
567        diagnostics,
568        starving: Vec::new(),
569    };
570    derive_heads(&mut queue);
571    queue
572}

574/// The open pull requests caught in a cycle of confirmed dependencies
575/// ([`classify::confirmed_dependencies`], between open pull requests only), each with the
576/// others it can reach and that reach it. A pull request on such a cycle waits for itself,
577/// so nothing about the others can let it land first. The module is private, so the
578/// example is text; `dependency_queue` runs the same assertions.
579///
580/// ```text
581/// use majordomus_cli::integration::{forge, ForgeObservation, OBSERVATION_SCHEMA};
582/// let pr = |n: u64, body: &str| forge::pull_request_of(&serde_json::json!({
583///     "number": n, "title": "t", "author": {"login": "a"}, "headRefName": format!("f/{n}"),
584///     "headRefOid": format!("{n:040}"), "baseRefName": "master", "isDraft": false,
585///     "labels": [], "createdAt": "t", "updatedAt": "t", "body": body,
586///     "statusCheckRollup": [], "reviewDecision": "", "autoMergeRequest": null,
587///     "isCrossRepository": false})).unwrap();
588/// let mut obs: ForgeObservation = serde_json::from_value(serde_json::json!({
589///     "schema": OBSERVATION_SCHEMA, "repository": "o/r", "base": "master", "base_sha": "m",
590///     "observed_at": "t", "required_checks": null, "review_policy": null,
591///     "merge_methods": ["merge"], "pull_requests": []})).unwrap();
592/// obs.pull_requests = vec![pr(1, "Depends on #2"), pr(2, "Depends on #1"), pr(3, "Depends on #1")];
593/// let cycles = majordomus_cli::integration::dependency_cycles(&obs, &Default::default());
594/// assert_eq!(cycles.get(&1), Some(&vec![2]));
595/// assert!(cycles.get(&3).is_none(), "waiting on a cycle is not being in one");
596/// ```
597pub fn dependency_cycles(
598    obs: &ForgeObservation,
599    heads: &BTreeMap<String, u64>,
600) -> BTreeMap<u64, Vec<u64>> {
601    let open: BTreeSet<u64> = obs.pull_requests.iter().map(|p| p.number).collect();
602    let edges: BTreeMap<u64, Vec<u64>> = obs
603        .pull_requests
604        .iter()
605        .map(|p| {
606            let (declared, stacked) = classify::confirmed_dependencies(p, &obs.base, heads);
607            let to: Vec<u64> = declared
608                .into_iter()
609                .chain(stacked)
610                .filter(|n| open.contains(n))
611                .collect();
612            (p.number, to)
613        })
614        .collect();
615    // everything each pull request reaches through its edges
616    let reach = |from: u64| -> BTreeSet<u64> {
617        let mut seen = BTreeSet::new();
618        let mut stack = vec![from];
619        while let Some(n) = stack.pop() {
620            for m in edges.get(&n).into_iter().flatten() {
621                if seen.insert(*m) {
622                    stack.push(*m);
623                }
624            }
625        }
626        seen
627    };
628    let reaches: BTreeMap<u64, BTreeSet<u64>> = open.iter().map(|n| (*n, reach(*n))).collect();
629    open.iter()
630        .filter(|n| reaches[n].contains(n))
631        .map(|n| {
632            let others: Vec<u64> = reaches[n]
633                .iter()
634                .filter(|m| *m != n && reaches[*m].contains(n))
635                .copied()
636                .collect();
637            (*n, others)
638        })
639        .collect()
640}

642/// Add the dependencies git implies: a pull request whose observed head contains another's
643/// observed head carries that one's commits, so landing it lands both. Inferred, and evidence
644/// only — a block is only ever declared ([`DependencyCertainty::Confirmed`]), never guessed,
645/// and nothing here changes a disposition or a rank. `containing` answers, for a head, the
646/// pull requests whose mirrored head contains it and the commit each mirror names; a mirror
647/// that is not the head the queue decided on implies nothing.
648pub fn infer_dependencies(
649    queue: &mut IntegrationQueue,
650    mut containing: impl FnMut(&str) -> Vec<(u64, String)>,
651) {
652    let heads: BTreeMap<u64, String> = queue
653        .assessments
654        .iter()
655        .map(|a| (a.number, a.evaluated_against.head_sha.clone()))
656        .collect();
657    let mut implied: BTreeMap<u64, BTreeSet<u64>> = BTreeMap::new();
658    for (n, head) in &heads {
659        for (m, tip) in containing(head) {
660            if m != *n && heads.get(&m) == Some(&tip) {
661                implied.entry(m).or_default().insert(*n);
662            }
663        }
664    }
665    for a in &mut queue.assessments {
666        for n in implied.get(&a.number).into_iter().flatten() {
667            if a.dependencies.iter().any(|d| d.number == *n) {
668                continue;
669            }
670            a.dependencies.push(PullRequestDependency {
671                number: *n,
672                certainty: DependencyCertainty::Inferred,
673                satisfied: false,
674                state: DependencyState::Open,
675            });
676            a.evidence.push(IntegrationEvidence {
677                kind: EvidenceKind::Dependency,
678                status: "inferred".into(),
679                detail: format!("#{n}'s head is in this head: landing it lands #{n} too"),
680                source: Some(EvidenceSource::Git {
681                    master_sha: a.evaluated_against.master_sha.clone(),
682                    head_sha: a.evaluated_against.head_sha.clone(),
683                }),
684            });
685        }
686    }
687}

689/// What the queue's ranked assessments imply: the next merge, the refresh candidates and the
690/// counts. Derived again whenever a disposition changes after the queue was built.
691fn derive_heads(queue: &mut IntegrationQueue) {
692    let mut tallies = QueueTallies {
693        open: queue.assessments.len(),
694        ..Default::default()
695    };
696    for a in &queue.assessments {
697        *tallies
698            .by_disposition
699            .entry(a.disposition.as_str().to_string())
700            .or_default() += 1;
701        *tallies.by_lane.entry(classify::word(&a.lane)).or_default() += 1;
702    }
703    queue.tallies = tallies;
704    queue.next_merge = queue
705        .assessments
706        .iter()
707        .find(|a| a.disposition == PullRequestDisposition::Ready)
708        .map(|a| a.number);
709    queue.next_refresh = queue
710        .assessments
711        .iter()
712        .filter(|a| a.disposition == PullRequestDisposition::NeedsRefresh)
713        .map(|a| a.number)
714        .collect();
715}

717/// The executor's own refusals, folded from the trail: a candidate the forge refused to
718/// merge, or whose refresh failed, at the head and master it still has, is held back as
719/// `needs_repair` instead of being chosen again, so one refused change does not block the
720/// queue. A new head or a new master clears it, because the key no longer matches; a
721/// transient failure never holds anything back, because it says nothing about the change.
722/// No plan survives: the trail is the only memory, and every queue is folded from it again.
723pub fn executor_feedback(queue: &mut IntegrationQueue, trail: &[drain::IntegrationEvent]) {
724    use drain::{FailureClass, IntegrationAction};
725    let refused = |action: IntegrationAction| -> BTreeSet<(u64, String, String)> {
726        trail
727            .iter()
728            .filter(|e| e.action == action && e.class != Some(FailureClass::Transient))
729            .filter_map(|e| Some((e.pr?, e.head_sha.clone()?, e.master_before.clone()?)))
730            .collect()
731    };
732    let merges = refused(IntegrationAction::MergeFailed);
733    let refreshes = refused(IntegrationAction::RefreshFailed);
734    let master = queue.master_sha.clone();
735    let mut changed = false;
736    for a in &mut queue.assessments {
737        let key = (
738            a.number,
739            a.evaluated_against.head_sha.clone(),
740            master.clone(),
741        );
742        let held = match a.disposition {
743            PullRequestDisposition::Ready if merges.contains(&key) => {
744                Some(ReasonCode::ExecutorMergeRefused {
745                    head: key.1.clone(),
746                })
747            }
748            PullRequestDisposition::NeedsRefresh if refreshes.contains(&key) => {
749                Some(ReasonCode::ExecutorRefreshFailed {
750                    master: master.clone(),
751                })
752            }
753            _ => None,
754        };
755        if let Some(reason) = held {
756            a.disposition = PullRequestDisposition::NeedsRepair;
757            a.lane = a.disposition.lane();
758            a.reasons.insert(0, reason);
759            changed = true;
760        }
761    }
762    if changed {
763        queue.assessments = rank(std::mem::take(&mut queue.assessments));
764        derive_heads(queue);
765    }
766}

768/// The declared successors of every open pull request, each with what became of it: open; or
769/// no longer open and landed when `relation` finds its head contained in master; or not landed;
770/// or unread, when the forge did not report it or git could not answer.
771fn successors(
772    obs: &ForgeObservation,
773    open: &BTreeSet<u64>,
774    relation: &mut impl FnMut(&PullRequestObservation) -> RelationToMaster,
775) -> BTreeMap<u64, Vec<classify::Successor>> {
776    use classify::{declared_supersessions, Successor, SuccessorState};
777    // replaced → successor → whose body said so (the first to)
778    let mut declared: BTreeMap<u64, BTreeMap<u64, u64>> = BTreeMap::new();
779    let mut declare = |replaced: u64, successor: u64, by: u64| {
780        if replaced != successor && open.contains(&replaced) {
781            declared
782                .entry(replaced)
783                .or_default()
784                .entry(successor)
785                .or_insert(by);
786        }
787    };
788    for p in &obs.pull_requests {
789        let s = declared_supersessions(&p.body);
790        for m in s.superseded_by {
791            declare(p.number, m, p.number);
792        }
793        for n in s.supersedes {
794            declare(n, p.number, p.number);
795        }
796    }
797    for (m, r) in &obs.resolved {
798        for n in declared_supersessions(&r.body).supersedes {
799            declare(n, *m, *m);
800        }
801    }
802    let mut state_of = |m: u64| -> SuccessorState {
803        if open.contains(&m) {
804            return SuccessorState::Open;
805        }
806        let Some(r) = obs.resolved.get(&m) else {
807            return SuccessorState::Unread;
808        };
809        let head = PullRequestObservation {
810            number: m,
811            title: String::new(),
812            author: String::new(),
813            head_ref: String::new(),
814            head_sha: r.head_sha.clone(),
815            base_ref: obs.base.clone(),
816            draft: false,
817            labels: Vec::new(),
818            created_at: String::new(),
819            updated_at: String::new(),
820            body: r.body.clone(),
821            checks: Vec::new(),
822            review_decision: String::new(),
823            auto_merge: false,
824            cross_repository: false,
825            latest_reviews: Vec::new(),
826            review_requests: Vec::new(),
827        };
828        match relation(&head) {
829            RelationToMaster::Contained => SuccessorState::Landed {
830                head_sha: r.head_sha.clone(),
831                merged: r.merged,
832            },
833            RelationToMaster::Unknown { .. } => SuccessorState::Unread,
834            _ => SuccessorState::NotLanded { merged: r.merged },
835        }
836    };
837    let mut states: BTreeMap<u64, SuccessorState> = BTreeMap::new();
838    declared
839        .into_iter()
840        .map(|(n, by)| {
841            let list = by
842                .into_iter()
843                .map(|(m, declared_in)| Successor {
844                    number: m,
845                    declared_in,
846                    state: states.entry(m).or_insert_with(|| state_of(m)).clone(),
847                })
848                .collect();
849            (n, list)
850        })
851        .collect()
852}

854/// The queue of this checkout, from the last recorded observation, with relations decided
855/// by git against this clone's fetched master. Offline: it reads the network's last answer
856/// and never asks it again.
857pub fn queue_of(root: &Path) -> Result<IntegrationQueue, String> {
858    computed(root, std::time::SystemTime::now()).map(|(queue, _)| queue)
859}

861/// [`queue_of`], and what it learnt kept for the next reader: the relations decided (the
862/// cache, by commit pair) and the summary a briefing reads without deciding one. Only the
863/// paths that just observed the forge write — `prs refresh` and the executor — so every
864/// read, from the command line, the capability or the Cockpit, leaves the checkout as it was.
865pub fn queue_and_record(root: &Path) -> Result<IntegrationQueue, String> {
866    let (queue, cache) = computed(root, std::time::SystemTime::now())?;
867    // both are caches a later read rebuilds: a write that fails costs that rebuild, never this
868    // answer, so neither failure is the caller's
869    let _ = serde_json::to_string(&cache)
870        .ok()
871        .map(|text| write_atomic(&state_path(root, RELATIONS_FILE), &text));
872    // the summary a briefing reads without deciding a single relation (QueueSummary), the
873    // repository's like the trail it sits beside
874    let _ = common_state_path(root, SUMMARY_FILE)
875        .ok()
876        .zip(serde_json::to_string_pretty(&QueueSummary::of(&queue)).ok())
877        .map(|(path, text)| write_atomic(&path, &(text + "\n")));
878    Ok(queue)
879}

881/// An observation older than this describes a forge that has moved on: the queue built from
882/// it is said to be stale, and nothing reading it may take it for the present.
883pub const OBSERVATION_STALE_AFTER: std::time::Duration = std::time::Duration::from_secs(3600);
884
885/// The queue of the recorded observation, and the relation cache as it stands after deciding
886/// it. Writes nothing.
887/// Name each assessment's issue and milestone: the issue its head branch names among
888/// `issues` (id → the milestone its record declares), by
889/// [`crate::worktree::state::issue_of`]. Neither decides a disposition or a rank: they say
890/// what the change is part of.
891pub fn link_issues(queue: &mut IntegrationQueue, issues: &BTreeMap<String, Option<String>>) {
892    let ids: Vec<String> = issues.keys().cloned().collect();
893    for a in &mut queue.assessments {
894        a.issue = crate::worktree::state::issue_of(&a.head_ref, &ids);
895        a.milestone = a
896            .issue
897            .as_ref()
898            .and_then(|i| issues.get(i).cloned().flatten());
899    }
900}

902/// The issues this checkout's project model declares, each with the milestone its record
903/// names: [`crate::worktree::state::issue_ids`], each record read with the layer's own YAML
904/// reader. A record that does not parse names no milestone; the issue still links.
905pub fn issue_milestones(root: &Path) -> BTreeMap<String, Option<String>> {
906    crate::worktree::state::issue_ids(root)
907        .into_iter()
908        .map(|id| {
909            let milestone = std::fs::read_to_string(
910                root.join(".ai/repo/project/issues")
911                    .join(format!("{id}.yaml")),
912            )
913            .ok()
914            .and_then(|t| crate::metadata::yaml::parse_mapping(&t).ok())
915            .and_then(|m| {
916                m.get("milestone")
917                    .and_then(crate::metadata::yaml::scalar_string)
918            })
919            .filter(|m| !m.is_empty());
920            (id, milestone)
921        })
922        .collect()
923}

925fn computed(
926    root: &Path,
927    now: std::time::SystemTime,
928) -> Result<(IntegrationQueue, RelationCache), String> {
929    let obs = load_observation(root)?.ok_or_else(|| {
930        "no forge observation is recorded in this checkout; majordomus prs refresh".to_string()
931    })?;
932    let master = local_master(root, &obs.base).ok_or_else(|| {
933        format!(
934            "this clone has no refs/remotes/origin/{}; majordomus prs refresh fetches it",
935            obs.base
936        )
937    })?;
938    let cache_path = state_path(root, RELATIONS_FILE);
939    let mut cache: RelationCache = std::fs::read_to_string(&cache_path)
940        .ok()
941        .and_then(|t| serde_json::from_str(&t).ok())
942        .unwrap_or_default();
943    // bounded: only the current master's entries are worth keeping, and only a pair of
944    // commit ids is a fact — a key written before refs were refused names a ref, which may
945    // hold another commit tomorrow, and is dropped even while master stands still
946    let current = |k: &String| {
947        k.split_once("..")
948            .is_some_and(|(m, h)| m == master && is_object_id(m) && is_object_id(h))
949    };
950    cache.entries.retain(|k, _| current(k));
951    cache.shapes.retain(|k, _| current(k));
952    // the relation and the shape share the cache, and each is asked from its own closure
953    let cell = std::cell::RefCell::new(cache);
954    let mut queue = build_queue_shaped(
955        &obs,
956        &master,
957        |p| {
958            // decided on exactly the head the forge reported, which the assessment names as
959            // evaluated: a head that moved during the refresh is not in this clone, and what
960            // the fetched ref holds now is another head nobody observed
961            if !relation::has_commit(root, &p.head_sha) {
962                return RelationToMaster::Unknown {
963                reason: format!(
964                    "the observed head {} is not fetched; the pull request moved during the refresh — majordomus prs refresh",
965                    p.head_sha
966                ),
967            };
968            }
969            relation_cached(root, &mut cell.borrow_mut(), &master, &p.head_sha)
970        },
971        |p| shape_cached(root, &mut cell.borrow_mut(), &master, &p.head_sha),
972    );
973    let cache = cell.into_inner();
974    infer_dependencies(&mut queue, |head| relation::containing(root, head));
975    link_issues(&mut queue, &issue_milestones(root));
976    let now_secs = now
977        .duration_since(std::time::UNIX_EPOCH)
978        .map_or(0, |d| i64::try_from(d.as_secs()).unwrap_or(i64::MAX));
979    if let Some(age) = crate::peers::epoch_seconds(&obs.observed_at)
980        .map(|then| now_secs.saturating_sub(then))
981        .filter(|age| *age > i64::try_from(OBSERVATION_STALE_AFTER.as_secs()).unwrap_or(i64::MAX))
982    {
983        queue.diagnostics.push(format!(
984            "the observation is {} min old (observed {}): the forge has moved on since; majordomus prs refresh",
985            age / 60,
986            obs.observed_at
987        ));
988    }
989    let trail = drain::events(root);
990    executor_feedback(&mut queue, &trail);
991    wait::annotate(&mut queue, &trail);
992    Ok((queue, cache))
993}

1045/// Observe the forge, fetch what it names, and record the observation — in the checkout's
1046/// state, and as an `observed` line of the repository's trail. The network step; the only
1047/// one outside the executor.
1048pub fn refresh(root: &Path) -> Result<ForgeObservation, String> {
1049    use forge::Forge;
1050    let obs = forge::GhForge { root }.observe().map_err(|e| e.0)?;
1051    forge::fetch(root, &obs).map_err(|e| e.0)?;
1052    store_observation(root, &obs)?;
1053    drain::record(
1054        root,
1055        drain::IntegrationEvent {
1056            master_before: Some(obs.base_sha.clone()),
1057            detail: format!(
1058                "{} open pull request(s) of {} at {}",
1059                obs.pull_requests.len(),
1060                obs.repository,
1061                obs.observed_at
1062            ),
1063            ..drain::IntegrationEvent::of(drain::IntegrationAction::Observed)
1064        },
1065    )?;
1066    Ok(obs)
1067}

1361    fn containment_is_evidence_and_never_a_block() {
1362        let mut q = build_queue(
1363            &obs(vec![pr(1, ""), pr(2, ""), pr(3, "Depends on #1")]),
1364            "m",
1365            unknown,

1372        infer_dependencies(&mut q, |h| {
1373            if h == head(1) {
1374                vec![(2, head(2)), (3, head(3)), (1, head(1))]
