1//! The `integration` module: the pull-request integration queue, projected (ADR 0101).
2//!
3//! Four questions, all read-only and all offline but the dry-run proof. Each renders the queue
4//! [`crate::integration::queue_of`] builds from the last recorded forge observation and
5//! this clone's fetched master — the value `majordomus prs status` prints — so the HTTP
6//! route, the MCP tool, the Cockpit and the command line cannot disagree. None reaches the
7//! network: observing the forge is `majordomus prs refresh`, and merging is
8//! `majordomus prs drain`, both command-line operations of the same module, because a
9//! capability that writes the repository or a remote must be declared as one and the
10//! shared server does not take that on.

34/// The queue, or why there is none.
35#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
36pub struct IntegrationStatus {
37    /// Whether a forge observation is recorded in this checkout.
38    pub observed: bool,
39    /// Why there is no queue, when there is none.
40    pub reason: Option<String>,
41    /// The queue.
42    pub queue: Option<IntegrationQueue>,
43    /// The base branch's integration lease as an observer reads it: `None` when nobody
44    /// holds it (or nothing is observed, so no base is known).
45    #[serde(default)]
46    pub lease: Option<IntegrationLeaseState>,
47    /// The last merge the executor recorded, from the audit trail.
48    #[serde(default)]
49    pub last_merge: Option<IntegrationEvent>,
50    /// How fast the executor has turned work into master over the last
51    /// [`THROUGHPUT_WINDOW_DAYS`] days, folded from the same trail.
52    pub throughput: IntegrationThroughput,
53}

55/// What cleanup would do, decided offline: the pull requests it would close or leave for a
56/// person, from the recorded observation, and the last branch report `prs cleanup` recorded.
57#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
58pub struct IntegrationCleanup {
59    /// Whether a forge observation is recorded in this checkout.
60    pub observed: bool,
61    /// Why there is no plan, when there is none.
62    pub reason: Option<String>,
63    /// When the forge was observed, when it was.
64    pub observed_at: Option<String>,
65    /// The plan, in rank order: `would_close` for what is provably on master or superseded by
66    /// a successor that landed, `left_for_a_person` for weak evidence and for what a person
67    /// marked obsolete. Nothing here closes anything.
68    pub items: Vec<CleanupItem>,
69    /// The branches merged pull requests left on origin, as `prs cleanup` last read them;
70    /// `None` when it never ran in this checkout. Never read from the network here.
71    pub branches: Option<LeftBranchReport>,
72    /// How old that report is, in seconds, when its time reads.
73    pub branches_age_seconds: Option<u64>,
74}
75
76/// [`IntegrationCleanup`] for the checkout at `root`, at `now` (seconds since the epoch):
77/// [`crate::integration::drain::cleanup_plan`] over [`crate::integration::queue_of`], which
78/// writes nothing, and the recorded branch report with its age. No network, no write.
79pub fn cleanup_status(root: &Path, now: u64) -> IntegrationCleanup {
80    let branches = crate::integration::drain::recorded_left_branches(root);
81    let branches_age_seconds = branches
82        .as_ref()
83        .and_then(|b| crate::peers::epoch_seconds(&b.read_at))
84        .and_then(|at| u64::try_from(at).ok())
85        .map(|at| now.saturating_sub(at));
86    match crate::integration::queue_of(root) {
87        Ok(q) => IntegrationCleanup {
88            observed: true,
89            reason: None,
90            observed_at: Some(q.observed_at.clone()),
91            items: crate::integration::drain::cleanup_plan(&q),
92            branches,
93            branches_age_seconds,
94        },
95        Err(reason) => IntegrationCleanup {
96            observed: false,
97            reason: Some(reason),
98            observed_at: None,
99            items: Vec::new(),
100            branches,
101            branches_age_seconds,
102        },
103    }
104}

150fn root(ctx: &Context) -> &Path {
151    Path::new(&ctx.index.repository.root)
152}

154fn integration_queue(ctx: &Context, _: Empty) -> Result<IntegrationStatus, CapabilityError> {
155    let root = root(ctx);
156    let trail = crate::integration::drain::events(root);
157    let last_merge = trail
158        .iter()
159        .rev()
160        .find(|e| e.action == crate::integration::drain::IntegrationAction::MergeSucceeded)
161        .cloned();
162    let now = std::time::SystemTime::now()
163        .duration_since(std::time::UNIX_EPOCH)
164        .map(|d| d.as_secs())
165        .unwrap_or(0);
166    let throughput = metrics::throughput(&trail, now, THROUGHPUT_WINDOW_DAYS);
167    Ok(match crate::integration::queue_of(root) {
168        Ok(q) => IntegrationStatus {
169            observed: true,
170            reason: None,
171            lease: IntegrationLease::read(root, &q.base).ok().flatten(),
172            queue: Some(q),
173            last_merge,
174            throughput,
175        },
176        Err(reason) => IntegrationStatus {
177            observed: false,
178            reason: Some(reason),
179            queue: None,
180            lease: None,
181            last_merge,
182            throughput,
183        },
184    })
185}
186
187fn integration_explain(
188    ctx: &Context,
189    input: PullRequestExplainInput,
190) -> Result<IntegrationExplanation, CapabilityError> {
191    let q = match crate::integration::queue_of(root(ctx)) {
192        Ok(q) => q,
193        Err(reason) => {
194            return Ok(IntegrationExplanation {
195                found: false,
196                reason: Some(reason),
197                rank: None,
198                of: 0,
199                observed_at: None,
200                assessment: None,
201            })
202        }
203    };
204    let rank = q
205        .assessments
206        .iter()
207        .position(|a| a.number == input.number)
208        .map(|i| i + 1);
209    Ok(IntegrationExplanation {
210        found: rank.is_some(),
211        reason: rank.is_none().then(|| {
212            format!(
213                "#{} is not among the {} open pull request(s) observed",
214                input.number, q.tallies.open
215            )
216        }),
217        rank,
218        of: q.assessments.len(),
219        observed_at: Some(q.observed_at.clone()),
220        assessment: q.get(input.number).cloned(),
221    })
222}
223
224fn integration_events(ctx: &Context, _: Empty) -> Result<IntegrationEvents, CapabilityError> {
225    Ok(IntegrationEvents {
226        events: crate::integration::drain::events(root(ctx)),
227    })
228}
229
230fn integration_cleanup(ctx: &Context, _: Empty) -> Result<IntegrationCleanup, CapabilityError> {
231    let now = std::time::SystemTime::now()
232        .duration_since(std::time::UNIX_EPOCH)
233        .map(|d| d.as_secs())
234        .unwrap_or(0);
235    Ok(cleanup_status(root(ctx), now))
236}
