1//! Token economics, measured rather than claimed.
2//!
3//! The question is the one `.ai/repo/benchmarks/economics/methodology.yaml` states: for the
4//! same task, from the same repository state, with the same model and the same acceptance
5//! tests, how many tokens does a coding session consume with Majordomus installed and
6//! without it? This module is the one calculator that answers it. The CLI, the HTTP API,
7//! MCP, the Cockpit, the generated report and the public site are projections of
8//! [`summarize`]; none of them computes a number of its own.
9//!
10//! The data flow:
11//!
12//! ```text
13//! methodology.yaml, suites/*.yaml, tasks/*/task.yaml    declared
14//!          │
15//!   runner (live sessions) ─► runs/<suite>/<run>.json   raw facts: provider usage, gates
16//!   context (no model)     ─► runs/context/<rev>.json   raw facts: counted tokens
17//!          │
18//!   summarize ─► pairs ─► per-pair reductions ─► distributions, intervals ─► verdict
19//! ```
20//!
21//! Nothing derived is stored. A pair, a metric and a verdict are recomputed from the raw
22//! records on every read, so the records never need rewriting when the derivation changes,
23//! and a changed derivation is a methodology version, not an edit to history.
24//!
25//! What the verdict may say is decided by the methodology's publication rule, not by the
26//! caller: below it, the statement is that no verified total-token claim exists, with the
27//! preliminary observation labelled as such. The rule is evaluated for the primary metric
28//! over all the evidence and for nothing else, so only that metric, unnarrowed, can ever
29//! be `verified`.
30//!
31//! What a generated artifact publishes comes from [`summarize_tracked`], which reads only
32//! the records git tracks, so that two checkouts of one commit generate the same bytes.
33//!
34//! ```
35//! use majordomus_cli::economics::{summarize, model::EconomicsQuery};
36//! let dir = tempfile::tempdir().unwrap();
37//! let s = summarize(dir.path(), &EconomicsQuery::default());
38//! assert!(!s.present);
39//! assert!(!s.verdict.publishable);
40//! assert!(s.verdict.statement.starts_with("No verified total-token-savings claim"));
41//! ```

70/// Every declaration of the economics subsystem, read from [`DIR`].
71///
72/// Only [`load`] builds one, and only when every file it names was read: a value of this
73/// type is a complete benchmark, never one with a task silently missing. Suites and tasks

75///
76/// ```

85/// #     "classes: []", "variants: []", "success: []",
86/// #     "pairing:", "  key: []", "  comparable: []", "  valid: v",
87/// #     "statistics:", "  per_pair: p", "  location: median", "  interval: bootstrap",

114fn read_yaml<T: serde::de::DeserializeOwned>(root: &Path, path: &Path) -> Result<T, String> {
115    let shown = path
116        .strip_prefix(root)
117        .unwrap_or(path)
118        .display()
119        .to_string();
120    let text = std::fs::read_to_string(path).map_err(|e| format!("{shown}: {e}"))?;
121    yaml::parse_into(&text).map_err(|e| format!("{shown}: {e}"))
122}

124/// A directory's entries in canonical order, by name.
125fn sorted_entries(dir: &Path) -> Vec<PathBuf> {
126    let mut out: Vec<Entry> = std::fs::read_dir(dir)
127        .into_iter()
128        .flatten()
129        .flatten()
130        .map(|e| Entry {
131            name: e.file_name().to_string_lossy().into_owned(),
132            path: e.path(),
133        })
134        .collect();
135    crate::order::canonical(&mut out);
136    out.into_iter().map(|e| e.path).collect()
137}

153/// Read every declaration under `root`. `Ok(None)` when no methodology is declared, which
154/// is the ordinary state of a repository that does not benchmark itself; an error names
155/// every file that could not be read, because a benchmark with a silently missing task is
156/// a different benchmark.
157///
158/// ```
159/// use majordomus_cli::economics::{load, DIR};
160/// let dir = tempfile::tempdir().unwrap();
161/// assert!(load(dir.path()).unwrap().is_none(), "no methodology: nothing is benchmarked");
162///
163/// std::fs::create_dir_all(dir.path().join(DIR)).unwrap();
164/// std::fs::write(dir.path().join(DIR).join("methodology.yaml"), "schema: x\n").unwrap();
165/// let errors = load(dir.path()).unwrap_err();
166/// assert_eq!(errors.len(), 1);
167/// assert!(errors[0].starts_with(&format!("{DIR}/methodology.yaml: ")), "{}", errors[0]);
168/// ```
169pub fn load(root: &Path) -> Result<Option<Declarations>, Vec<String>> {
170    let dir = root.join(DIR);
171    let methodology_path = dir.join("methodology.yaml");
172    if !methodology_path.is_file() {
173        return Ok(None);
174    }
175    let mut errors = Vec::new();
176    let methodology = match read_yaml::<EconomicsMethodology>(root, &methodology_path) {
177        Ok(m) => Some(m),
178        Err(e) => {
179            errors.push(e);
180            None
181        }
182    };
183    let mut suites = BTreeMap::new();
184    for p in sorted_entries(&dir.join("suites")) {
185        if p.extension().and_then(|e| e.to_str()) != Some("yaml") {
186            continue;
187        }
188        match read_yaml::<EconomicsSuite>(root, &p) {
189            Ok(s) => {
190                suites.insert(s.id.clone(), s);
191            }
192            Err(e) => errors.push(e),
193        }
194    }
195    let mut tasks = BTreeMap::new();
196    for p in sorted_entries(&dir.join("tasks")) {
197        let file = p.join("task.yaml");
198        if !file.is_file() {
199            continue;
200        }
201        match read_yaml::<EconomicsTask>(root, &file) {
202            Ok(t) => {
203                tasks.insert(t.id.clone(), t);
204            }
205            Err(e) => errors.push(e),
206        }
207    }
208    for s in suites.values() {
209        for t in &s.tasks {
210            if !tasks.contains_key(t) {
211                errors.push(format!(
212                    "suite {} names task {t}, which is not declared",
213                    s.id
214                ));
215            }
216        }
217    }
218    match (methodology, errors.is_empty()) {
219        (Some(methodology), true) => Ok(Some(Declarations {
220            methodology,
221            suites,
222            tasks,
223        })),
224        _ => Err(errors),
225    }
226}

228/// A digest of the tracked files under `pathspecs`: each file's path and the SHA-256 of its
229/// content as it is on disk now, in git's index order. Recorded with every run, and
230/// recomputed on every read, it is what makes evidence current or stale.
231///
232/// ```
233/// use majordomus_cli::economics::inputs_digest;
234/// use std::process::Command;
235/// let dir = tempfile::tempdir().unwrap();
236/// let git = |a: &[&str]| Command::new("git").arg("-C").arg(dir.path()).args(a).output().unwrap();
237/// git(&["init", "-q"]);
238/// std::fs::write(dir.path().join("a.txt"), "one").unwrap();
239/// git(&["add", "-A"]);
240/// let before = inputs_digest(dir.path(), &["a.txt".to_string()]).unwrap();
241/// std::fs::write(dir.path().join("a.txt"), "two").unwrap();
242/// assert_ne!(before, inputs_digest(dir.path(), &["a.txt".to_string()]).unwrap());
243/// ```
244pub fn inputs_digest(root: &Path, pathspecs: &[String]) -> Result<String, String> {
245    let specs: Vec<&str> = pathspecs.iter().map(String::as_str).collect();
246    let files = crate::git::ls_files_any(root, &specs).map_err(|e| e.to_string())?;
247    let mut h = Sha256::new();
248    for f in &files {
249        let bytes = std::fs::read(root.join(f)).unwrap_or_default();
250        h.update(f.as_bytes());
251        h.update([0]);
252        h.update(crate::policy::sha256_bytes_hex(&bytes).as_bytes());
253        h.update(b"\n");
254    }
255    Ok(format!("{:x}", h.finalize()))
256}

258/// The revision evidence is recorded at: HEAD, and whether the working tree differs from it
259/// anywhere outside the evidence directory. The records a suite writes are changes to the
260/// working tree too, and a measurement must not call its own output a dirty tree.
261///
262/// ```
263/// use majordomus_cli::economics::{revision, DIR};
264/// let dir = tempfile::tempdir().unwrap();
265/// # let git = |a: &[&str]| {
266/// #     let out = std::process::Command::new("git")
267/// #         .env_remove("GIT_DIR").env_remove("GIT_WORK_TREE").env_remove("GIT_INDEX_FILE")
268/// #         .arg("-C").arg(dir.path()).args(a).output().unwrap();
269/// #     assert!(out.status.success(), "git {a:?}");
270/// #     String::from_utf8(out.stdout).unwrap()
271/// # };
272/// git(&["init", "-q"]);
273/// assert!(revision(dir.path()).is_err(), "no commit yet, so no HEAD to record against");
274/// std::fs::write(dir.path().join("a.txt"), "one").unwrap();
275/// git(&["add", "-A"]);
276/// git(&["-c", "user.name=t", "-c", "user.email=t@example.com", "-c", "commit.gpgsign=false",
277///     "commit", "-qm", "one"]);
278/// let clean = revision(dir.path()).unwrap();
279/// assert_eq!((clean.commit.as_str(), clean.dirty), (git(&["rev-parse", "HEAD"]).trim(), false));
280///
281/// std::fs::create_dir_all(dir.path().join(DIR).join("runs")).unwrap();
282/// std::fs::write(dir.path().join(DIR).join("runs/r1.json"), "{}").unwrap();
283/// assert!(!revision(dir.path()).unwrap().dirty, "its own records do not dirty the tree");
284/// std::fs::write(dir.path().join("a.txt"), "two").unwrap();
285/// assert!(revision(dir.path()).unwrap().dirty);
286/// ```
287pub fn revision(root: &Path) -> Result<EconomicsRevision, String> {
288    let head = crate::git::read_only(root)
289        .args(["rev-parse", "--verify", "-q", "HEAD"])
290        .output()
291        .map_err(|e| format!("git: {e}"))?;
292    if !head.status.success() {
293        return Err("git: no HEAD to record evidence against".into());
294    }
295    let status = crate::git::read_only(root)
296        .args(["status", "--porcelain", "--untracked-files=all", "--", "."])
297        .arg(format!(":(exclude){DIR}/runs"))
298        .output()
299        .map_err(|e| format!("git: {e}"))?;
300    Ok(EconomicsRevision {
301        commit: String::from_utf8_lossy(&head.stdout).trim().to_string(),
302        dirty: !String::from_utf8_lossy(&status.stdout).trim().is_empty(),
303    })
304}

332/// Where run records are read from. A person asking for a summary wants every record on
333/// disk, the ones not yet committed included; a generated artifact must be the same on
334/// every checkout of one commit, so it reads only the records git tracks.
335#[derive(Debug, Clone, Copy, PartialEq, Eq)]
336enum Source {
337    /// Every file in the suite's run directory.
338    Directory,
339    /// Only the files of that directory git tracks, read from the working tree.
340    Tracked,
341}

343/// The files of `runs/<suite>/` the source admits, in canonical order. Only the directory's
344/// own entries, never a nested one, in both sources, so the two differ only in what git
345/// tracks.
346fn record_files(root: &Path, suite: &str, source: Source) -> Result<Vec<PathBuf>, String> {
347    let dir = root.join(DIR).join("runs").join(suite);
348    match source {
349        Source::Directory => Ok(sorted_entries(&dir)),
350        Source::Tracked => {
351            let rel = format!("{DIR}/runs/{suite}");
352            let tracked = crate::git::ls_files_any(root, &[rel.as_str()])
353                .map_err(|e| format!("{rel}: the tracked records could not be listed: {e}"))?;
354            let mut own: Vec<String> = tracked
355                .into_iter()
356                .filter(|f| Path::new(f.as_str()).parent() == Some(Path::new(&rel)))
357                .collect();
358            crate::order::canonical(&mut own);
359            Ok(own.iter().map(|f| root.join(f)).collect())
360        }
361    }
362}

364fn load_runs_from(
365    root: &Path,
366    suite: &str,
367    source: Source,
368) -> (Vec<(String, EconomicsRun)>, Vec<String>) {
369    let mut runs = Vec::new();
370    let mut errors = Vec::new();
371    let files = match record_files(root, suite, source) {
372        Ok(files) => files,
373        Err(e) => return (runs, vec![e]),
374    };
375    for p in files {
376        if p.extension().and_then(|e| e.to_str()) != Some("json") {
377            continue;
378        }
379        let rel = p
380            .strip_prefix(root)
381            .unwrap_or(&p)
382            .to_string_lossy()
383            .to_string();
384        match std::fs::read_to_string(&p)
385            .map_err(|e| e.to_string())
386            .and_then(|t| serde_json::from_str::<EconomicsRun>(&t).map_err(|e| e.to_string()))
387        {
388            Ok(r) if r.schema == RUN_SCHEMA => runs.push((rel, r)),
389            Ok(r) => errors.push(format!("{rel}: schema {} is not {RUN_SCHEMA}", r.schema)),
390            Err(e) => errors.push(format!("{rel}: {e}")),
391        }
392    }
393    crate::order::canonical(&mut runs);
394    let mut seen = BTreeSet::new();
395    for (rel, r) in &runs {
396        if !seen.insert(r.id.clone()) {
397            errors.push(format!("{rel}: run id {} is recorded twice", r.id));
398        }
399    }
400    (runs, errors)
401}

481/// Judge two runs as the methodology's pairing rule does, the outlier rule aside: a missing
482/// side, then comparability, then the success gates, then usage.
483fn judge(
484    control: Option<&EconomicsRun>,
485    treatment: Option<&EconomicsRun>,
486) -> (EconomicsPairStatus, Vec<String>) {
487    let (Some(c), Some(t)) = (control, treatment) else {
488        let side = if control.is_none() {
489            "control"
490        } else {
491            "treatment"
492        };
493        return (
494            EconomicsPairStatus::Missing,
495            vec![format!("the {side} run is not recorded")],
496        );
497    };
498    let mut differ = Vec::new();
499    let mut check = |what: &str, a: &str, b: &str| {
500        if a != b {
501            differ.push(format!("{what} differs: {a} against {b}"));
502        }
503    };
504    check(
505        "suite version",
506        &c.suite_version.to_string(),
507        &t.suite_version.to_string(),
508    );
509    check("requested model", &c.model_requested, &t.model_requested);
510    // the model asked for is not always the model that answered: a harness may route part
511    // of a session elsewhere, and two runs answered by different models are not one task
512    check(
513        "reported models",
514        &c.models_reported.join(", "),
515        &t.models_reported.join(", "),
516    );
517    check("fixture digest", &c.fixture_digest, &t.fixture_digest);
518    check("harness version", &c.harness.version, &t.harness.version);
519    check(
520        "configuration digest",
521        &c.configuration_digest,
522        &t.configuration_digest,
523    );
524    check(
525        "repository revision",
526        &c.repository.commit,
527        &t.repository.commit,
528    );
529    check(
530        "session count",
531        &c.sessions.len().to_string(),
532        &t.sessions.len().to_string(),
533    );
534    if !differ.is_empty() {
535        return (EconomicsPairStatus::Incomparable, differ);
536    }
537    let failed = |r: &EconomicsRun| -> Vec<String> {
538        r.outcome
539            .checks
540            .iter()
541            .filter(|k| !k.passed)
542            .map(|k| format!("{} {}: {}", r.variant, k.id, k.detail))
543            .collect()
544    };
545    match (c.outcome.completed, t.outcome.completed) {
546        (false, false) => {
547            let mut r = failed(c);
548            r.extend(failed(t));
549            return (EconomicsPairStatus::BothFailed, r);
550        }
551        (false, true) => return (EconomicsPairStatus::ControlFailed, failed(c)),
552        (true, false) => return (EconomicsPairStatus::TreatmentFailed, failed(t)),
553        (true, true) => {}
554    }
555    let mut missing = Vec::new();
556    for r in [c, t] {
557        if usage::totals(r).is_none() {
558            missing.push(format!("{} reported no usage for a session", r.id));
559        }
560    }
561    if !missing.is_empty() {
562        return (EconomicsPairStatus::UsageUnavailable, missing);
563    }
564    (EconomicsPairStatus::Valid, Vec::new())
565}

567/// The pair's status, its reasons, and the status it would have without the outlier rule.
568/// A pair with both sides recorded and a side the methodology excludes is `Excluded`, with
569/// the exclusion's reasons first and whatever else was wrong with it after them.
570fn pair_status(
571    methodology: &EconomicsMethodology,
572    control: Option<&EconomicsRun>,
573    treatment: Option<&EconomicsRun>,
574) -> (EconomicsPairStatus, Vec<String>, EconomicsPairStatus) {
575    let (underlying, reasons) = judge(control, treatment);
576    if underlying == EconomicsPairStatus::Missing {
577        return (underlying, reasons, underlying);
578    }
579    let ids: Vec<&str> = [control, treatment]
580        .into_iter()
581        .flatten()
582        .map(|r| r.id.as_str())
583        .collect();
584    let mut excluded: Vec<String> = methodology
585        .outliers
586        .excluded
587        .iter()
588        .filter(|x| ids.contains(&x.run.as_str()))
589        .map(|x| format!("{} excluded: {}", x.run, x.reason))
590        .collect();
591    if excluded.is_empty() {
592        return (underlying, reasons, underlying);
593    }
594    excluded.extend(reasons);
595    (EconomicsPairStatus::Excluded, excluded, underlying)
596}

615/// Form every pair a live suite declares (each task, each repetition) from the runs
616/// recorded for it, and judge each. A declared pair nobody ran is `missing`, not absent.
617///
618/// Runs recorded under another methodology version or for another suite are ignored, and a
619/// suite that does not name both a control and a treatment arm forms no pair at all.
620///
621/// ```
622/// use majordomus_cli::economics::{load, pairs};
623/// use majordomus_cli::economics::model::EconomicsPairStatus;
624/// let dir = tempfile::tempdir().unwrap();
625/// # use majordomus_cli::economics::DIR;
626/// # let decl_dir = dir.path().join(DIR);
627/// # std::fs::create_dir_all(decl_dir.join("suites")).unwrap();
628/// # std::fs::write(decl_dir.join("methodology.yaml"), [
629/// #     "schema: economics-methodology/v1", "version: 1", "title: t", "question: q",
630/// #     "unit: tokens", "primary_metric: effective_token_reduction",
631/// #     "classes: []", "variants: []", "success: []",
632/// #     "pairing:", "  key: []", "  comparable: []", "  valid: v",
633/// #     "statistics:", "  per_pair: p", "  location: median", "  interval: bootstrap",
634/// #     "  confidence_bp: 9500", "  resamples: 100", "  seed: 1", "  min_pairs_for_interval: 5",
635/// #     "publication:", "  min_valid_pairs: 30", "  min_categories: 4",
636/// #     "  min_pairs_per_category: 5", "  min_repetitions: 3", "  min_valid_pair_rate_bp: 8000",
637/// #     "  max_interval_width_bp: 2000", "  require_current: true",
638/// #     "outliers:", "  rule: none", "",
639/// # ].join("\n")).unwrap();
640/// # std::fs::write(decl_dir.join("suites/pilot.yaml"), [
641/// #     "schema: economics-suite/v1", "id: pilot", "version: 1", "kind: live", "title: Pilot",
642/// #     "control: baseline", "treatment: majordomus", "freshness_inputs: [methodology.yaml]", "",
643/// # ].join("\n")).unwrap();
644/// // a methodology and one live suite `pilot` with a control and a treatment arm
645/// let decl = load(dir.path()).unwrap().unwrap();
646/// let mut suite = decl.suites["pilot"].clone();
647/// suite.tasks = vec!["vat-rounding".into()];
648/// suite.repetitions = Some(2);
649/// let declared = pairs(&decl, &suite, &[]);
650/// assert_eq!(declared.len(), 2, "each task times each repetition, run or not");
651/// assert!(declared.iter().all(|p| p.status == EconomicsPairStatus::Missing));
652/// suite.control = None;
653/// assert!(pairs(&decl, &suite, &[]).is_empty(), "without both arms nothing is paired");
654/// ```
655pub fn pairs(
656    decl: &Declarations,
657    suite: &EconomicsSuite,
658    runs: &[EconomicsRun],
659) -> Vec<EconomicsPair> {
660    judged_pairs(decl, suite, runs)
661        .into_iter()
662        .map(|j| j.pair)
663        .collect()
664}
665
666fn judged_pairs(decl: &Declarations, suite: &EconomicsSuite, runs: &[EconomicsRun]) -> Vec<Judged> {
667    let (Some(control), Some(treatment)) = (&suite.control, &suite.treatment) else {
668        return Vec::new();
669    };
670    let current: Vec<&EconomicsRun> = runs
671        .iter()
672        .filter(|r| r.methodology == decl.methodology.version && r.suite == suite.id)
673        .collect();
674    let mut keys: BTreeSet<(String, u32)> = BTreeSet::new();
675    for t in &suite.tasks {
676        for rep in 1..=suite.repetitions.unwrap_or(1) {
677            keys.insert((t.clone(), rep));
678        }
679    }
680    for r in &current {
681        keys.insert((r.task.clone(), r.repetition));
682    }
683    let find = |task: &str, variant: &str, rep: u32| {
684        current
685            .iter()
686            .copied()
687            .find(|r| r.task == task && r.variant == variant && r.repetition == rep)
688    };
689    keys.into_iter()
690        .map(|(task, rep)| {
691            let c = find(&task, control, rep);
692            let t = find(&task, treatment, rep);
693            let (status, reasons, underlying) = pair_status(&decl.methodology, c, t);
694            let declared = decl.tasks.get(&task);
695            let cu = c.and_then(usage::totals);
696            let tu = t.and_then(usage::totals);
697            // an excluded pair that would otherwise be valid keeps its reductions, so the
698            // exclusion can be shown both ways; every metric but the including-excluded one
699            // reads valid pairs only
700            let computable = underlying == EconomicsPairStatus::Valid;
701            let both = |f: &dyn Fn(&EconomicsUsage) -> Option<u64>| -> Option<f64> {
702                if !computable {
703                    return None;
704                }
705                reduction(f(cu.as_ref()?)?, f(tu.as_ref()?)?)
706            };
707            let multi = declared.map(|d| d.sessions.len()).unwrap_or(1) > 1;
708            let continuation = if computable && multi {
709                c.zip(t)
710                    .and_then(|(c, t)| reduction(last_session_tokens(c)?, last_session_tokens(t)?))
711            } else {
712                None
713            };
714            let model = c
715                .or(t)
716                .map(|r| r.model_requested.clone())
717                .or_else(|| suite.model.clone())
718                .unwrap_or_default();
719            let pair = EconomicsPair {
720                suite: suite.id.clone(),
721                task: task.clone(),
722                category: declared
723                    .map(|d| d.category.clone())
724                    .unwrap_or_else(|| "undeclared".into()),
725                sessions: declared.map(|d| d.sessions.len()).unwrap_or(1),
726                repetition: rep,
727                control: c.map(|r| r.id.clone()),
728                treatment: t.map(|r| r.id.clone()),
729                status,
730                reasons,
731                token_reduction: both(&|u| Some(u.total)),
732                cost_reduction: both(&|u| u.cost_microusd),
733                tool_call_reduction: both(&|u| Some(u.tool_calls)),
734                continuation_reduction: continuation,
735                first_request_overhead: if computable {
736                    let a = c
737                        .and_then(|r| r.sessions.first())
738                        .and_then(usage::first_request_input);
739                    let b = t
740                        .and_then(|r| r.sessions.first())
741                        .and_then(usage::first_request_input);
742                    a.zip(b).map(|(a, b)| b as i64 - a as i64)
743                } else {
744                    None
745                },
746                control_usage: cu,
747                treatment_usage: tu,
748            };
749            Judged {
750                pair,
751                underlying,
752                model,
753            }
754        })
755        .collect()
756}

760fn freshness(
761    root: &Path,
762    decl: &Declarations,
763    suite: &EconomicsSuite,
764    recorded: &[(u32, String)],
765) -> (EconomicsFreshness, Option<String>) {
766    if recorded.is_empty() {
767        return (EconomicsFreshness::NoEvidence, None);
768    }
769    let other: Vec<u32> = recorded
770        .iter()
771        .map(|(m, _)| *m)
772        .filter(|m| *m != decl.methodology.version)
773        .collect::<BTreeSet<_>>()
774        .into_iter()
775        .collect();
776    if other.len() == recorded.len() || recorded.iter().all(|(m, _)| *m != decl.methodology.version)
777    {
778        return (
779            EconomicsFreshness::Incompatible,
780            Some(format!(
781                "recorded under methodology {:?}; the current methodology is {}",
782                other, decl.methodology.version
783            )),
784        );
785    }
786    let now = match inputs_digest(root, &suite.freshness_inputs) {
787        Ok(d) => d,
788        Err(e) => {
789            return (
790                EconomicsFreshness::Stale,
791                Some(format!("the inputs could not be read: {e}")),
792            )
793        }
794    };
795    let stale = recorded
796        .iter()
797        .filter(|(m, d)| *m == decl.methodology.version && *d != now)
798        .count();
799    if stale > 0 {
800        return (
801            EconomicsFreshness::Stale,
802            Some(format!(
803                "{stale} record(s) were measured against inputs that have changed since: {}",
804                suite.freshness_inputs.join(", ")
805            )),
806        );
807    }
808    (EconomicsFreshness::Current, None)
809}

886/// The cluster-bootstrap interval of the median over samples, clustered by task.
887fn task_interval(
888    samples: &[Sample],
889    policy: &EconomicsStatisticsPolicy,
890) -> Option<EconomicsInterval> {
891    let mut clusters: BTreeMap<&str, Vec<f64>> = BTreeMap::new();
892    for s in samples {
893        clusters
894            .entry(s.cluster.as_str())
895            .or_default()
896            .push(s.value);
897    }
898    let clusters: Vec<Vec<f64>> = clusters.into_values().collect();
899    bootstrap_median_clustered(
900        &clusters,
901        policy.confidence_bp,
902        policy.resamples,
903        policy.seed,
904        policy.min_pairs_for_interval,
905    )
906}

908/// Why an interval over `n` values of `tasks` distinct tasks is absent, in the words a
909/// statement uses.
910fn no_interval(n: usize, policy: &EconomicsStatisticsPolicy) -> &'static str {
911    if n < policy.min_pairs_for_interval.max(2) {
912        "no interval: too few pairs"
913    } else {
914        "no interval: too few tasks, and the repetitions of one task are not independent"
915    }
916}

918/// A metric's value, distribution and interval from its samples, with its status: none
919/// without a value, `verified` only when the caller says the publication rule was evaluated
920/// for exactly this metric and met, `preliminary` otherwise.
921fn sampled(
922    mut m: EconomicsMetric,
923    samples: &[Sample],
924    policy: &EconomicsStatisticsPolicy,
925    resampling: Resampling,
926    verified: bool,
927) -> EconomicsMetric {
928    let v: Vec<f64> = samples.iter().map(|s| s.value).collect();
929    m.n = v.len();
930    m.evidence = samples.iter().map(|s| s.key.clone()).collect();
931    m.distribution = distribution(&v);
932    m.value = m.distribution.as_ref().map(|d| d.median);
933    m.interval = match resampling {
934        Resampling::Tasks => task_interval(samples, policy),
935        Resampling::Independent => bootstrap_median(
936            &v,
937            policy.confidence_bp,
938            policy.resamples,
939            policy.seed,
940            policy.min_pairs_for_interval,
941        ),
942    };
943    m.status = match (m.n, verified) {
944        (0, _) => EconomicsMetricStatus::NotMeasured,
945        (_, true) => EconomicsMetricStatus::Verified,
946        _ => EconomicsMetricStatus::Preliminary,
947    };
948    if m.n > 0 && m.interval.is_none() {
949        let tasks = samples
950            .iter()
951            .map(|s| s.cluster.as_str())
952            .collect::<BTreeSet<_>>()
953            .len();
954        if resampling == Resampling::Tasks && m.n >= policy.min_pairs_for_interval.max(2) {
955            m.warnings.push(format!(
956                "{} value(s) from {tasks} task(s): too few tasks for an interval, because the repetitions of one task are not independent",
957                m.n
958            ));
959        } else {
960            m.warnings.push(format!(
961                "{} value(s): too few for an interval (the methodology computes one from {})",
962                m.n, policy.min_pairs_for_interval
963            ));
964        }
965    }
966    m
967}

991/// How the median moved, in a sentence's words. The verb follows the sign as it prints,
992/// so a median that rounds to zero is `unchanged`, never `reduced by 0.0%`.

998        format!("increased median total token consumption by {x}")

1001    }

1005/// median increase is followed by an interval of increases, never by negative reductions.

1033/// Every threshold of the publication rule the live evidence does not meet. Empty means a
1034/// quantitative total-token claim may be made.
1035fn unmet(
1036    policy: &EconomicsPublicationPolicy,
1037    valid: &[&EconomicsPair],
1038    attempted: usize,
1039    floor: Option<&RepetitionFloor>,
1040    interval: Option<&EconomicsInterval>,
1041    all_current: bool,
1042    dirty: usize,
1043) -> Vec<String> {
1044    let mut out = Vec::new();
1045    if valid.len() < policy.min_valid_pairs {
1046        out.push(format!(
1047            "{} valid pair(s); the rule asks for {}",
1048            valid.len(),
1049            policy.min_valid_pairs
1050        ));
1051    }
1052    let mut per_category: BTreeMap<&str, usize> = BTreeMap::new();
1053    for p in valid {
1054        *per_category.entry(p.category.as_str()).or_insert(0) += 1;
1055    }
1056    let categories = per_category
1057        .values()
1058        .filter(|n| **n >= policy.min_pairs_per_category)
1059        .count();
1060    if categories < policy.min_categories {
1061        out.push(format!(
1062            "{categories} categor(ies) with at least {} valid pairs; the rule asks for {}",
1063            policy.min_pairs_per_category, policy.min_categories
1064        ));
1065    }
1066    match floor {
1067        None => out.push(format!(
1068            "no declared task and variant to repeat; the rule asks for {} repetition(s) of each",
1069            policy.min_repetitions
1070        )),
1071        Some(f) if f.reps < policy.min_repetitions => out.push(format!(
1072            "{} repetition(s) of {}, the least repeated; the rule asks for {} of every task and variant",
1073            f.reps, f.of, policy.min_repetitions
1074        )),
1075        Some(_) => {}
1076    }
1077    if let Some(rate) = (valid.len() * 10_000).checked_div(attempted) {
1078        if rate < policy.min_valid_pair_rate_bp as usize {
1079            out.push(format!(
1080                "{} of {attempted} attempted pairs are valid; the rule asks for {} in 10000",
1081                valid.len(),
1082                policy.min_valid_pair_rate_bp
1083            ));
1084        }
1085    }
1086    match interval {
1087        None => out.push("no interval: too few valid pairs, or too few tasks among them".into()),
1088        Some(i) if (i.high - i.low) * 10_000.0 > policy.max_interval_width_bp as f64 => {
1089            out.push(format!(
1090                "the interval is {} wide; the rule allows {} in 10000",
1091                width_text(i.high - i.low),
1092                policy.max_interval_width_bp
1093            ))
1094        }
1095        _ => {}
1096    }
1097    if policy.require_current && !all_current {
1098        out.push(
1099            "the evidence is not current: a measured mechanism changed since it was recorded"
1100                .into(),
1101        );
1102    }
1103    if dirty > 0 {
1104        out.push(format!(
1105            "{dirty} run(s) were recorded from a working tree with uncommitted changes"
1106        ));
1107    }
1108    out
1109}

1164/// The string a value serialises to, as every JSON surface spells it:
1165/// `EconomicsMetricStatus::NotMeasured` is `not_measured`. A renderer that prints an enum
1166/// uses this rather than `Debug`, so that a page and the API never spell one state two
1167/// ways. A value that does not serialise to a string is given as its JSON text.
1168///
1169/// ```
1170/// use majordomus_cli::economics::wire;
1171/// use majordomus_cli::economics::model::{EconomicsFreshness, EconomicsMetricStatus};
1172/// assert_eq!(wire(&EconomicsMetricStatus::NotMeasured), "not_measured");
1173/// assert_eq!(wire(&EconomicsFreshness::NoEvidence), "no_evidence");
1174/// assert_eq!(wire(&3), "3");
1175/// ```
1176pub fn wire<T: Serialize>(value: &T) -> String {
1177    match serde_json::to_value(value) {
1178        Ok(serde_json::Value::String(s)) => s,
1179        Ok(other) => other.to_string(),
1180        Err(e) => format!("(not serialisable: {e})"),
1181    }
1182}

1191/// The one calculator. Reads the declarations and every recorded run under `root`, forms
1192/// and judges the pairs, derives every metric with its class, distribution and interval,
1193/// and states the strongest thing the publication rule allows. Never fails: what cannot be
1194/// read is a diagnostic, and a repository with no methodology gets `present: false`.
1195///
1196/// Every record file on disk is read, the ones not yet committed included, so that a
1197/// person sees what they just recorded; [`summarize_tracked`] is the same calculation over
1198/// the committed records only.
1199///
1200/// ```
1201/// use majordomus_cli::economics::{summarize, model::EconomicsQuery, DIR, NO_CLAIM};
1202/// let dir = tempfile::tempdir().unwrap();
1203/// let none = summarize(dir.path(), &EconomicsQuery::default());
1204/// assert!(!none.present && none.diagnostics.is_empty(), "not benchmarking is not a fault");
1205///
1206/// std::fs::create_dir_all(dir.path().join(DIR)).unwrap();
1207/// std::fs::write(dir.path().join(DIR).join("methodology.yaml"), "schema: x\n").unwrap();
1208/// let broken = summarize(dir.path(), &EconomicsQuery::default());
1209/// assert!(!broken.present && !broken.verdict.publishable);
1210/// assert!(broken.verdict.statement.starts_with(NO_CLAIM));
1211/// assert_eq!(broken.diagnostics.len(), 1, "what cannot be read is reported, not raised");
1212/// ```
1213pub fn summarize(root: &Path, query: &EconomicsQuery) -> EconomicsSummary {
1214    summarize_from(root, query, Source::Directory)
1215}

1217/// [`summarize`] over the run records git tracks and nothing else: what `majordomus
1218/// generate` publishes. A generated artifact must be the same on every checkout of one
1219/// commit, and a record someone ran locally and never committed would make it depend on
1220/// whose checkout generated it. Tracked records are read from the working tree; a record
1221/// that is on disk and untracked is not read at all, not even to report it.

1224/// use majordomus_cli::economics::{summarize, summarize_tracked, DIR};

1232/// #     "pairing:", "  key: []", "  comparable: []", "  valid: v",
1233/// #     "statistics:", "  per_pair: p", "  location: median", "  interval: bootstrap",

1237/// #     "  max_interval_width_bp: 2000", "  require_current: true",

1246/// #     .arg("-C").arg(dir.path()).args(["init", "-q"]).status().unwrap();

1248/// // a git repository declaring one live suite `pilot`; one record, never committed
1249/// let recorded = dir.path().join(DIR).join("runs/pilot");

1251/// std::fs::write(recorded.join("r1.json"), "{").unwrap();
1252/// let q = Default::default();
1253/// assert_eq!(summarize(dir.path(), &q).diagnostics.len(), 1, "a person sees the local record");

1260fn summarize_from(root: &Path, query: &EconomicsQuery, source: Source) -> EconomicsSummary {
1261    let decl = match load(root) {
1262        Ok(Some(d)) => d,
1263        Ok(None) => {
1264            return empty(
1265                &format!("{NO_CLAIM}: no benchmark methodology is declared in this repository."),
1266                Vec::new(),
1267            )
1268        }
1269        Err(errors) => {
1270            return empty(
1271                &format!("{NO_CLAIM}: the benchmark declarations could not be read."),
1272                errors,
1273            )
1274        }
1275    };
1276    let m = &decl.methodology;
1277    let mut diagnostics = Vec::new();
1278    let mut suite_views = Vec::new();
1279    let mut all_pairs: Vec<Judged> = Vec::new();
1280    let mut live_runs: Vec<EconomicsRun> = Vec::new();
1281    let mut context_runs: Vec<ContextEvidence> = Vec::new();
1282    let mut all_current = true;
1283    let mut floor: Option<RepetitionFloor> = None;
1284    let mut history = Vec::new();
1285
1286    for suite in decl.suites.values() {
1287        match suite.kind.as_str() {
1288            "live" => {
1289                let (runs, errors) = load_runs_from(root, &suite.id, source);
1290                diagnostics.extend(errors);
1291                let runs: Vec<EconomicsRun> = runs.into_iter().map(|(_, r)| r).collect();
1292                let recorded: Vec<(u32, String)> = runs
1293                    .iter()
1294                    .map(|r| (r.methodology, r.inputs_digest.clone()))
1295                    .collect();
1296                let (fresh, detail) = freshness(root, &decl, suite, &recorded);
1297                if !matches!(
1298                    fresh,
1299                    EconomicsFreshness::Current | EconomicsFreshness::NoEvidence
1300                ) {
1301                    all_current = false;
1302                }
1303                let judged = judged_pairs(&decl, suite, &runs);
1304                let mut counts = EconomicsPairCounts {
1305                    declared: suite.tasks.len() * suite.repetitions.unwrap_or(1) as usize,
1306                    ..Default::default()
1307                };
1308                for p in judged.iter().map(|j| &j.pair) {
1309                    if p.control.is_some() || p.treatment.is_some() {
1310                        counts.attempted += 1;
1311                    }
1312                    match p.status {
1313                        EconomicsPairStatus::Valid => counts.valid += 1,
1314                        EconomicsPairStatus::ControlFailed => counts.control_failed += 1,
1315                        EconomicsPairStatus::TreatmentFailed => counts.treatment_failed += 1,
1316                        EconomicsPairStatus::BothFailed => counts.both_failed += 1,
1317                        _ if p.control.is_some() || p.treatment.is_some() => counts.other += 1,
1318                        _ => {}
1319                    }
1320                }
1321                // the rule asks for repetitions of every task and variant: count, for each,
1322                // the distinct repetitions that ran under this methodology (a failed run
1323                // ran), and keep the least; tasks in id order, so that a tie names the same
1324                // task however the suite lists them
1325                for task in suite.tasks.iter().collect::<BTreeSet<_>>() {
1326                    for variant in [&suite.control, &suite.treatment].into_iter().flatten() {
1327                        let reps = runs
1328                            .iter()
1329                            .filter(|r| r.methodology == m.version && r.suite == suite.id)
1330                            .filter(|r| &r.task == task && &r.variant == variant)
1331                            .map(|r| r.repetition)
1332                            .collect::<BTreeSet<_>>()
1333                            .len() as u32;
1334                        if floor.as_ref().is_none_or(|f| reps < f.reps) {
1335                            floor = Some(RepetitionFloor {
1336                                reps,
1337                                of: format!("{task} ({variant}) in suite {}", suite.id),
1338                            });
1339                        }
1340                    }
1341                }
1342                // one history point per revision the suite's valid pairs were recorded at
1343                let mut by_rev: BTreeMap<String, Vec<f64>> = BTreeMap::new();
1344                for p in judged
1345                    .iter()
1346                    .map(|j| &j.pair)
1347                    .filter(|p| p.status == EconomicsPairStatus::Valid)
1348                {
1349                    if let Some(r) = runs.iter().find(|r| Some(&r.id) == p.control.as_ref()) {
1350                        if let Some(x) = p.token_reduction {
1351                            by_rev
1352                                .entry(r.repository.commit.clone())
1353                                .or_default()
1354                                .push(x);
1355                        }
1356                    }
1357                }
1358                for (rev, v) in by_rev {
1359                    let at = runs
1360                        .iter()
1361                        .filter(|r| r.repository.commit == rev)
1362                        .map(|r| r.finished_at.clone())
1363                        .max()
1364                        .unwrap_or_default();
1365                    if let Some(d) = distribution(&v) {
1366                        history.push(EconomicsHistoryPoint {
1367                            suite: suite.id.clone(),
1368                            methodology: m.version,
1369                            revision: rev,
1370                            at,
1371                            metric: EFFECTIVE_TOKEN_REDUCTION.into(),
1372                            value: d.median,
1373                            n: d.n,
1374                        });
1375                    }
1376                }
1377                suite_views.push(EconomicsSuiteView {
1378                    id: suite.id.clone(),
1379                    kind: suite.kind.clone(),
1380                    version: suite.version,
1381                    title: suite.title.clone(),
1382                    model: suite.model.clone(),
1383                    freshness: fresh,
1384                    freshness_detail: detail,
1385                    runs: runs.len(),
1386                    pairs: counts,
1387                    revisions: runs
1388                        .iter()
1389                        .map(|r| r.repository.commit.clone())
1390                        .collect::<BTreeSet<_>>()
1391                        .into_iter()
1392                        .collect(),
1393                    harnesses: runs
1394                        .iter()
1395                        .map(|r| format!("{} {}", r.harness.name, r.harness.version))
1396                        .collect::<BTreeSet<_>>()
1397                        .into_iter()
1398                        .collect(),
1399                    models_reported: runs
1400                        .iter()
1401                        .flat_map(|r| r.models_reported.clone())
1402                        .collect::<BTreeSet<_>>()
1403                        .into_iter()
1404                        .collect(),
1405                });
1406                all_pairs.extend(judged);
1407                live_runs.extend(runs);
1408            }
1409            "context" => {
1410                let (runs, errors) = load_context_runs_from(root, &suite.id, source);
1411                diagnostics.extend(errors);
1412                let recorded: Vec<(u32, String)> = runs
1413                    .iter()
1414                    .map(|r| (r.methodology, r.inputs_digest.clone()))
1415                    .collect();
1416                let latest_only: Vec<(u32, String)> =
1417                    recorded.last().cloned().into_iter().collect();
1418                let (fresh, detail) = freshness(root, &decl, suite, &latest_only);
1419                for r in &runs {
1420                    let v: Vec<f64> = r
1421                        .seeds
1422                        .iter()
1423                        .filter_map(|s| reduction(s.candidate_tokens, s.selected_tokens))
1424                        .collect();
1425                    if let Some(d) = distribution(&v) {
1426                        history.push(EconomicsHistoryPoint {
1427                            suite: suite.id.clone(),
1428                            methodology: r.methodology,
1429                            revision: r.repository.commit.clone(),
1430                            at: r.measured_at.clone(),
1431                            metric: CONTEXT_REDUCTION.into(),
1432                            value: d.median,
1433                            n: d.n,
1434                        });
1435                    }
1436                }
1437                suite_views.push(EconomicsSuiteView {
1438                    id: suite.id.clone(),
1439                    kind: suite.kind.clone(),
1440                    version: suite.version,
1441                    title: suite.title.clone(),
1442                    model: None,
1443                    freshness: fresh,
1444                    freshness_detail: detail,
1445                    runs: runs.len(),
1446                    pairs: EconomicsPairCounts::default(),
1447                    revisions: runs
1448                        .iter()
1449                        .map(|r| r.repository.commit.clone())
1450                        .collect::<BTreeSet<_>>()
1451                        .into_iter()
1452                        .collect(),
1453                    harnesses: Vec::new(),
1454                    models_reported: Vec::new(),
1455                });
1456                context_runs.push(ContextEvidence {
1457                    suite,
1458                    runs,
1459                    freshness: fresh,
1460                });
1461            }
1462            other => diagnostics.push(format!("suite {}: unknown kind {other:?}", suite.id)),
1463        }
1464    }
1465
1466    // a run is found by its suite and id: ids are unique within a suite, not across them
1467    let mut run_of: BTreeMap<(&str, &str), &EconomicsRun> = BTreeMap::new();
1468    for r in &live_runs {
1469        run_of.entry((r.suite.as_str(), r.id.as_str())).or_insert(r);
1470    }
1471    let token_samples = |ps: &[&EconomicsPair]| -> Vec<Sample> {
1472        ps.iter()
1473            .filter_map(|p| {
1474                p.token_reduction.map(|x| Sample {
1475                    key: pair_key(p),
1476                    cluster: cluster_of(p),
1477                    value: x,
1478                })
1479            })
1480            .collect()
1481    };
1482
1483    // ------------------------------------------------ the verdict, over all live evidence
1484    let valid_all: Vec<&EconomicsPair> = all_pairs
1485        .iter()
1486        .map(|j| &j.pair)
1487        .filter(|p| p.status == EconomicsPairStatus::Valid)
1488        .collect();
1489    let attempted = all_pairs
1490        .iter()
1491        .filter(|j| j.pair.control.is_some() || j.pair.treatment.is_some())
1492        .count();
1493    let samples_all = token_samples(&valid_all);
1494    let reductions_all: Vec<f64> = samples_all.iter().map(|s| s.value).collect();
1495    let interval_all = task_interval(&samples_all, &m.statistics);
1496    let dirty = live_runs.iter().filter(|r| r.repository.dirty).count();
1497    let unmet_list = unmet(
1498        &m.publication,
1499        &valid_all,
1500        attempted,
1501        floor.as_ref(),
1502        interval_all.as_ref(),
1503        all_current,
1504        dirty,
1505    );
1506    let publishable = unmet_list.is_empty() && !valid_all.is_empty();
1507    // what the statement names is what the figure rests on: the runs of valid pairs
1508    let valid_runs: Vec<&EconomicsRun> = valid_all
1509        .iter()
1510        .flat_map(|p| {
1511            [&p.control, &p.treatment]
1512                .into_iter()
1513                .flatten()
1514                .map(move |id| (p.suite.as_str(), id.as_str()))
1515        })
1516        .filter_map(|k| run_of.get(&k).copied())
1517        .collect();
1518    let models: Vec<String> = valid_runs
1519        .iter()
1520        .flat_map(|r| r.models_reported.clone())
1521        .collect::<BTreeSet<_>>()
1522        .into_iter()
1523        .collect();
1524    let models_text = if models.is_empty() {
1525        "a model the harness did not name".to_string()
1526    } else {
1527        models.join(", ")
1528    };
1529    let revisions: BTreeSet<&str> = valid_runs
1530        .iter()
1531        .map(|r| r.repository.commit.as_str())
1532        .collect();
1533    let tasks: BTreeSet<(&str, &str)> = valid_all
1534        .iter()
1535        .map(|p| (p.suite.as_str(), p.task.as_str()))
1536        .collect();
1537    let categories: BTreeSet<&str> = valid_all.iter().map(|p| p.category.as_str()).collect();
1538    let none = no_interval(reductions_all.len(), &m.statistics);
1539    let mut statement = match (distribution(&reductions_all), publishable) {
1540        (None, _) => format!(
1541            "{NO_CLAIM}: no valid matched pair of live runs is recorded under methodology {}.",
1542            m.version
1543        ),
1544        (Some(d), true) => format!(
1545            "Across methodology {}, {} valid matched pairs across {} tasks in {} categories, at revision(s) {}, using {}, Majordomus {}, with every run passing the same acceptance tests.",
1546            m.version,
1547            d.n,
1548            tasks.len(),
1549            categories.len(),
1550            revisions.iter().map(|r| &r[..r.len().min(12)]).collect::<Vec<_>>().join(", "),
1551            models_text,
1552            change_with_interval(d.median, interval_all.as_ref(), none),
1553        ),
1554        (Some(d), false) => format!(
1555            "{NO_CLAIM}. Preliminary observation, below the publication rule: over {} valid matched pair(s) across {} task(s) in {} categor(ies), using {}, Majordomus {}.",
1556            d.n,
1557            tasks.len(),
1558            categories.len(),
1559            models_text,
1560            change_with_interval(d.median, interval_all.as_ref(), none),
1561        ),
1562    };
1563    let excluded_ids: BTreeSet<&str> = m.outliers.excluded.iter().map(|x| x.run.as_str()).collect();
1564    let excluded_runs: BTreeSet<(&str, &str)> = all_pairs
1565        .iter()
1566        .flat_map(|j| {
1567            [&j.pair.control, &j.pair.treatment]
1568                .into_iter()
1569                .flatten()
1570                .map(move |id| (j.pair.suite.as_str(), id.as_str()))
1571        })
1572        .filter(|(_, id)| excluded_ids.contains(id))
1573        .collect();
1574    if !excluded_runs.is_empty() {
1575        statement.push_str(&format!(
1576            " {} run(s) excluded by the methodology's outlier rule are left out",
1577            excluded_runs.len()
1578        ));
1579        let with_excluded: Vec<&EconomicsPair> = all_pairs
1580            .iter()
1581            .filter(|j| j.underlying == EconomicsPairStatus::Valid)
1582            .map(|j| &j.pair)
1583            .collect();
1584        let v: Vec<f64> = token_samples(&with_excluded)
1585            .into_iter()
1586            .map(|s| s.value)
1587            .collect();
1588        match distribution(&v) {
1589            Some(d) => statement.push_str(&format!(
1590                "; with them included, over {} pair(s), Majordomus {} (preliminary).",
1591                d.n,
1592                change_words(d.median)
1593            )),
1594            None => statement.push('.'),
1595        }
1596    }
1597
1598    // ------------------------------------------------ narrowed pairs and metrics
1599    let narrowed = query.suite.is_some()
1600        || query.category.is_some()
1601        || query.task.is_some()
1602        || query.model.is_some();
1603    let pairs_q: Vec<&Judged> = all_pairs
1604        .iter()
1605        .filter(|j| matches(query, &j.pair, &j.model))
1606        .collect();
1607    let valid: Vec<&EconomicsPair> = pairs_q
1608        .iter()
1609        .map(|j| &j.pair)
1610        .filter(|p| p.status == EconomicsPairStatus::Valid)
1611        .collect();
1612    let values = |f: &dyn Fn(&EconomicsPair) -> Option<f64>| -> Vec<Sample> {
1613        valid
1614            .iter()
1615            .filter_map(|p| {
1616                f(p).map(|x| Sample {
1617                    key: pair_key(p),
1618                    cluster: cluster_of(p),
1619                    value: x,
1620                })
1621            })
1622            .collect()
1623    };
1624    let live_suite = decl
1625        .suites
1626        .values()
1627        .filter(|s| s.kind == "live")
1628        .map(|s| s.id.clone())
1629        .collect::<Vec<_>>()
1630        .join(",");
1631    let mut metrics = Vec::new();
1632    // the publication rule is evaluated for this metric over all the evidence, and for
1633    // nothing else: it is verified only unnarrowed and only when the rule is met
1634    let mut primary = sampled(
1635        metric(
1636            MetricSpec {
1637                id: EFFECTIVE_TOKEN_REDUCTION,
1638                title: "Effective total-token reduction",
1639                class: EconomicsClass::Derived,
1640                inputs: Some(EconomicsClass::Observed),
1641                unit: "ratio",
1642                formula: "median over valid pairs of 1 - treatment_total_tokens / control_total_tokens, where total tokens are all input (uncached, cache write, cache read) and all output the provider reported for every session of the run",
1643                not: Some("not context reduction: this is everything the provider reported for the whole task, not what one mechanism selected"),
1644            },
1645            &live_suite,
1646        ),
1647        &values(&|p| p.token_reduction),
1648        &m.statistics,
1649        Resampling::Tasks,
1650        publishable && !narrowed,
1651    );
1652    if primary.n > 0 && !publishable {
1653        primary
1654            .warnings
1655            .push("preliminary: the evidence is below the publication rule".into());
1656    }
1657    if narrowed {
1658        primary.warnings.push(NARROWED.into());
1659    }
1660    if !valid.is_empty() {
1661        primary.warnings.push(
1662            "cache reads are counted in total tokens at full weight; the cost reduction weighs them by price".into(),
1663        );
1664    }
1665    metrics.push(primary);
1666    if !m.outliers.excluded.is_empty() {
1667        let put_back: Vec<&EconomicsPair> = pairs_q
1668            .iter()
1669            .filter(|j| j.underlying == EconomicsPairStatus::Valid)
1670            .map(|j| &j.pair)
1671            .collect();
1672        let excluded_pairs = put_back
1673            .iter()
1674            .filter(|p| p.status == EconomicsPairStatus::Excluded)
1675            .count();
1676        let mut including = secondary(sampled(
1677            metric(
1678                MetricSpec {
1679                    id: EFFECTIVE_TOKEN_REDUCTION_INCLUDING_EXCLUDED,
1680                    title: "Effective total-token reduction, excluded runs included",
1681                    class: EconomicsClass::Derived,
1682                    inputs: Some(EconomicsClass::Observed),
1683                    unit: "ratio",
1684                    formula: "the formula of effective_token_reduction, over the valid pairs and the pairs the outlier rule excluded that would otherwise be valid",
1685                    not: Some("not the primary metric: the methodology excludes these runs, and this figure is reported beside the primary one so that the exclusion's effect is visible"),
1686                },
1687                &live_suite,
1688            ),

2108/// exclusion it rests on, with the commands that reproduce it. `Err` names the metrics that
