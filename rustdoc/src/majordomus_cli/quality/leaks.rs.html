1//! The shapes of text a published artifact must never carry, read from the one file that
2//! defines them for every check in the repository.
3//!
4//! `leak-patterns.tsv` beside this module is the definition. `scripts/ci/no-machine-paths`
5//! applies it to every tracked file with `git grep`, `scripts/site-check` to the built site
6//! with `grep`, and `majordomus quality rustdoc` to the crate's documentation tree through
7//! this module. Before the file existed the site check, the tracked-file gate and this crate
8//! each held a copy of the expressions, and copies of a pattern drift the way every other
9//! repeated definition does. The file is compiled into the executable, so the check a
10//! released binary runs is the definition it was built from, whatever tree it is run in.
11//!
12//! The expressions are POSIX extended regular expressions in the subset `grep -E` and the
13//! [`regex`] crate read the same way, and they are applied the way `grep` applies them: one
14//! line at a time, case-sensitively, with ASCII semantics. A credential match is dropped
15//! when a `not-credential` expression finds something in the matched text itself, which is
16//! what the shell's `grep -o … | grep -v` does.
17//!
18//! Nothing here reports the matched text of a credential: a report that repeated the value it
19//! found would be the leak it exists to catch. A finding names the pattern and the line.
20
21use std::sync::OnceLock;
22
23use regex::bytes::{Regex, RegexBuilder};
24
25/// The definition, as committed.
26const DEFINITION: &str = include_str!("leak-patterns.tsv");
27
28/// Where the definition lives, repository-relative: what a finding tells a reader to open.
29pub(crate) const SOURCE: &str = "apps/majordomus-cli/src/quality/leak-patterns.tsv";

31/// What a pattern is about. The words are the first column of the definition.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
33pub(crate) enum LeakClass {
34    /// A path of the machine that wrote the file.
35    MachinePath,
36    /// The home of a hosted CI runner.
37    RunnerPath,
38    /// A host name only a private network resolves.
39    PrivateHost,
40    /// A long opaque value assigned to a key, a secret, a token or a password.
41    Credential,
42    /// Applied to a credential match: a hit says the match is a name, not a value.
43    NotCredential,
44}
45
46impl LeakClass {
47    fn parse(word: &str) -> Option<Self> {
48        Some(match word {
49            "machine-path" => LeakClass::MachinePath,
50            "runner-path" => LeakClass::RunnerPath,
51            "private-host" => LeakClass::PrivateHost,
52            "credential" => LeakClass::Credential,
53            "not-credential" => LeakClass::NotCredential,
54            _ => return None,
55        })
56    }
57}
58
59/// One line of the definition, compiled.
60#[derive(Debug)]
61pub(crate) struct Pattern {
62    /// What it is about.
63    pub class: LeakClass,
64    /// Its name in the definition, which is what a finding reports.
65    pub name: String,
66    /// The expression as written, compiled.
67    regex: Regex,
68}
69
70/// One place a pattern matched.
71#[derive(Debug, Clone, PartialEq, Eq)]
72pub(crate) struct Leak {
73    /// The class of the pattern that matched.
74    pub class: LeakClass,
75    /// The pattern's name.
76    pub name: String,
77    /// The line, 1-based.
78    pub line: usize,
79    /// What matched, for every class but a credential, whose value is never repeated.
80    pub excerpt: Option<String>,
81}
82
83/// The compiled definition.
84#[derive(Debug)]
85pub(crate) struct Leaks {
86    patterns: Vec<Pattern>,
87}
88
89impl Leaks {
90    /// The definition this executable was built with, compiled once per process.
91    ///
92    /// An expression that does not compile is an error the caller reports, never an empty
93    /// set of patterns: a check that could not read its patterns and then found nothing
94    /// would report a clean tree it never looked at.
95    pub(crate) fn shipped() -> Result<&'static Leaks, String> {
96        static SHIPPED: OnceLock<Result<Leaks, String>> = OnceLock::new();
97        SHIPPED
98            .get_or_init(|| Leaks::parse(DEFINITION))
99            .as_ref()
100            .map_err(Clone::clone)
101    }
102
103    /// Compile a definition in the committed format. Refuses a line it cannot read, a class
104    /// it does not know, an expression that does not compile, and a definition without a
105    /// single machine-path or credential pattern.
106    pub(crate) fn parse(text: &str) -> Result<Leaks, String> {
107        let mut patterns = Vec::new();
108        for (n, line) in text.lines().enumerate() {
109            if line.trim().is_empty() || line.starts_with('#') {
110                continue;
111            }
112            let fields: Vec<&str> = line.split('\t').collect();
113            let [class, name, expression] = fields[..] else {
114                return Err(format!(
115                    "{SOURCE}:{}: expected <class> TAB <name> TAB <expression>, found {} field(s)",
116                    n + 1,
117                    fields.len()
118                ));
119            };
120            let class = LeakClass::parse(class)
121                .ok_or_else(|| format!("{SOURCE}:{}: unknown class '{class}'", n + 1))?;
122            // ASCII semantics, as grep applies the expression in the C locale: `\b` is an ASCII
123            // word boundary and a class never widens to Unicode letters
124            let regex = RegexBuilder::new(expression)
125                .unicode(false)
126                .build()
127                .map_err(|e| format!("{SOURCE}:{}: '{name}' does not compile: {e}", n + 1))?;
128            patterns.push(Pattern {
129                class,
130                name: name.to_string(),
131                regex,
132            });
133        }
134        for needed in [LeakClass::MachinePath, LeakClass::Credential] {
135            if !patterns.iter().any(|p| p.class == needed) {
136                return Err(format!(
137                    "{SOURCE}: no {needed:?} pattern; a check with nothing to look for reports clean"
138                ));
139            }
140        }
141        Ok(Leaks { patterns })
142    }
143
144    /// Every pattern of the definition, in its order.
145    #[cfg(test)]
146    pub(crate) fn patterns(&self) -> &[Pattern] {
147        &self.patterns
148    }
149
150    /// Every leak of the given classes in `text`, in pattern order and then line order.
151    ///
152    /// Each pattern is first tried against the whole text, which is one pass and almost
153    /// always a miss; only a text it matches is read again line by line, which is where
154    /// the line number and grep's one-line semantics come from. A match inside a line is
155    /// a match inside the text, so the first pass cannot hide one.
156    pub(crate) fn scan(&self, text: &[u8], classes: &[LeakClass]) -> Vec<Leak> {
157        let mut out = Vec::new();
158        for p in self.patterns.iter().filter(|p| classes.contains(&p.class)) {
159            if !p.regex.is_match(text) {
160                continue;
161            }
162            for (n, line) in text.split(|b| *b == b'\n').enumerate() {
163                let hit = p.regex.find_iter(line).find(|m| {
164                    p.class != LeakClass::Credential || !self.not_credential(m.as_bytes())
165                });
166                if let Some(m) = hit {
167                    out.push(Leak {
168                        class: p.class,
169                        name: p.name.clone(),
170                        line: n + 1,
171                        excerpt: (p.class != LeakClass::Credential)
172                            .then(|| String::from_utf8_lossy(m.as_bytes()).into_owned()),
173                    });
174                }
175            }
176        }
177        out
178    }
179
180    /// Does a `not-credential` pattern find something in this credential match?
181    fn not_credential(&self, matched: &[u8]) -> bool {
182        self.patterns
183            .iter()
184            .filter(|p| p.class == LeakClass::NotCredential)
185            .any(|p| p.regex.is_match(matched))
186    }
187}

189#[cfg(test)]

193    // Every fixture below is assembled at run time. This file is committed and its source is
194    // published as a page of the documentation tree, and both are read by the checks this
195    // definition drives: a literal home directory or a literal credential here would be a
196    // finding of the gate about the gate.
197    fn home(user: &str) -> String {
198        format!("{}{user}/", "/Users/")
199    }
200
201    fn assignment(key: &str, value: &str) -> String {
202        format!("{key}{}{value}", ": ")
203    }

205    #[test]
206    fn the_shipped_definition_compiles_and_names_every_class_a_reader_asks_for() {
207        let leaks = Leaks::shipped().expect("the committed definition compiles");
208        for class in [
209            LeakClass::MachinePath,
210            LeakClass::RunnerPath,
211            LeakClass::PrivateHost,
212            LeakClass::Credential,
213            LeakClass::NotCredential,
214        ] {
215            assert!(
216                leaks.patterns().iter().any(|p| p.class == class),
217                "no {class:?} pattern in {SOURCE}"
218            );
219        }
220        // and every expression is the one written, not a translation of it
221        assert!(leaks
222            .patterns()
223            .iter()
224            .all(|p| DEFINITION.contains(&format!("\t{}\t{}", p.name, p.regex.as_str()))));
225    }
226
227    #[test]
228    fn a_home_directory_is_found_on_its_line_and_named() {
229        let leaks = Leaks::shipped().unwrap();
230        let text = format!("first\nthe checkout is at {}dev/app\n", home("someone"));
231        let found = leaks.scan(text.as_bytes(), &[LeakClass::MachinePath]);
232        assert_eq!(found.len(), 1, "{found:?}");
233        assert_eq!(found[0].line, 2);
234        assert_eq!(found[0].name, "macos-home");
235        assert_eq!(found[0].excerpt.as_deref(), Some(home("someone").as_str()));
236        // a class nobody asked for is not reported
237        assert!(leaks
238            .scan(text.as_bytes(), &[LeakClass::Credential])
239            .is_empty());
240    }

242    #[test]
243    fn a_home_without_a_checkout_is_documentation_and_not_a_leak() {
244        // docs/CLI.md's sample output: the reason the Linux pattern is narrow
245        let leaks = Leaks::shipped().unwrap();
246        let sample = format!("repository   {}dev/app", "/home/");
247        assert!(leaks
248            .scan(sample.as_bytes(), &[LeakClass::MachinePath])
249            .is_empty());
250        let checkout = format!("{}someone/dev/app", "/home/");
251        assert_eq!(
252            leaks
253                .scan(checkout.as_bytes(), &[LeakClass::MachinePath])
254                .len(),
255            1
256        );
257    }

272    #[test]
273    fn a_credential_is_found_and_its_value_is_never_repeated() {
274        let leaks = Leaks::shipped().unwrap();
275        let value = format!("{}{}", "abcdef", "0123456789");
276        let text = assignment("api_key", &value);
277        let found = leaks.scan(text.as_bytes(), &[LeakClass::Credential]);
278        assert_eq!(found.len(), 1, "{found:?}");
279        assert_eq!(found[0].name, "assignment");
280        assert_eq!(found[0].excerpt, None, "the value is not repeated");
281    }

283    #[test]
284    fn a_constant_name_assigned_to_a_token_field_is_code_and_not_a_credential() {
285        // the false positive rustdoc's source pages produced: `bytes_per_token:
286        // BYTES_PER_TOKEN` is a struct literal naming a constant
287        let leaks = Leaks::shipped().unwrap();
288        let code = assignment("bytes_per_token", &["BYTES", "PER", "TOKEN"].join("_"));
289        assert!(leaks
290            .scan(code.as_bytes(), &[LeakClass::Credential])
291            .is_empty());
292        // and the refinement is as narrow as the case: the same shape with a digit or a
293        // lower-case letter in the value is a value, and is still found
294        for value in [
295            ["BYTES", "PER", "TOKEN9"].join("_"),
296            ["Bytes", "per", "token"].join("_"),
297            format!("{}{}", "ABCDEFGHIJ", "KLMNOP"),
298        ] {
299            let text = assignment("token", &value);
300            assert_eq!(
301                leaks.scan(text.as_bytes(), &[LeakClass::Credential]).len(),
302                1,
303                "{text} is a credential"
304            );
305        }
306    }

324    #[test]
325    fn a_match_never_spans_two_lines() {
326        // grep reads one line at a time; so does this, or `token:` at the end of one line
327        // and an identifier at the start of the next would be a credential

335    #[test]
336    fn a_definition_that_cannot_be_read_is_refused_and_never_read_as_empty() {
337        assert!(Leaks::parse("machine-path\tonly-two-fields\n").is_err());
338        assert!(Leaks::parse("elsewhere\tx\tabc\n").is_err());
339        assert!(Leaks::parse("machine-path\tx\t(unclosed\n").is_err());
