1//! The one execution path.
2//!
3//! Every call of an executable capability — from the stdio MCP session, from `/mcp`, from
4//! an HTTP route, from the command line, from the Cockpit or from a benchmark runner —
5//! goes through [`CapabilityExecutor::execute`]: registry lookup, counters, the cache the
6//! capability's own policy asks for, then the handler. Transport adapters convert protocol
7//! to JSON and back and own nothing else, so instrumentation and caching apply to every
8//! transport at once and cannot drift apart. A second execution path is the defect this
9//! module exists to make impossible.
10//!
11//! # The cache key
12//!
13//! Process memory, bounded per capability by its [`CachePolicy`], keyed
14//! by three things: the canonical id, the input **normalised** (object keys sorted at every
15//! level, so `{"a":1,"b":2}` and `{"b":2,"a":1}` are one entry), and the registry
16//! fingerprint — which hashes every descriptor and every declarative object's content. Two
17//! processes over different repository states therefore never share an entry, and a changed
18//! layer never answers from an old one.
19//!
20//! Errors are never cached. Commands are never cached, and the registry refuses a
21//! descriptor that asks for it, so that is a build-time refusal rather than a runtime one.
22//!
23//! Two policies narrow the key further, for answers the registry fingerprint cannot see
24//! move. [`CachePolicy::Generation`] adds the context's [`Context::reading`]: an answer
25//! read off the index, including the git state it was read at, is kept for as long as that
26//! reading is the one asked. [`CachePolicy::Checkout`] adds that and the fingerprint of the
27//! checkout, taken again on every call (`capability::checkout`): HEAD, the staging index,
28//! the refs, and every changed or untracked path of the work tree with its size and
29//! modification time. A checkout git cannot fingerprint is not cached.
30//!
31//! # One handler per key at a time
32//!
33//! Concurrent calls that miss on the same key do not each run the handler: the first runs
34//! it, the others wait for it and are answered from what it stored. A screen that asks the
35//! same source from two panels at once, or two browsers that open one page together, pay
36//! for one computation. A handler that fails stores nothing, so a caller that waited on it
37//! runs the handler itself.
38//!
39//! # Switching it off
40//!
41//! `MAJORDOMUS_CAPABILITY_CACHE=off` in a process's environment makes every call run its
42//! handler and store nothing, whatever the capability's policy: the way to settle a doubt
43//! about a cached answer by asking without the cache (`project.cache-is-invisible`).
44//!
45//! # Invisibility
46//!
47//! A cached answer is byte-identical to a computed one; nothing in a response says which it
48//! was. `project.cache-is-invisible` requires it, and [`canonical_json`] is what makes the
49//! key well-defined enough for it to be true.
50//!
51//! ```
52//! use majordomus_cli::capability::executor::canonical_json;
53//!
54//! // the same input written two ways is one cache entry, at every level of nesting
55//! let one = serde_json::json!({ "b": 2, "a": { "y": 1, "x": 0 } });
56//! let other = serde_json::json!({ "a": { "x": 0, "y": 1 }, "b": 2 });
57//! assert_eq!(canonical_json(&one), canonical_json(&other));
58//!
59//! // and a different input is a different entry
60//! assert_ne!(canonical_json(&one), canonical_json(&serde_json::json!({ "b": 3 })));
61//! ```

75/// The executor of one registry.
76#[derive(Debug, Default)]
77pub struct CapabilityExecutor {
78    cache: Mutex<Cache>,
79    /// The handlers this executor ran: cache misses and uncached calls. A measurement of
80    /// this executor reads it; the process-wide counters also count every other executor
81    /// in the process, which a test running beside another reads as its own.
82    invocations: AtomicU64,
83    /// How many calls of a capability that writes the repository have succeeded here.
84    writes: AtomicU64,
85    /// The latest reading of every source a page has asked for, kept current off the path
86    /// of the request (`capability::readings`). Beside the cache and not in it: the cache
87    /// holds answers, and a reading says how old it is.
88    pub(crate) readings: super::readings::Readings,
89}

91#[derive(Debug, Default)]
92struct Cache {
93    entries: HashMap<CacheKey, CacheEntry>,
94    /// Insertion order per capability, for eviction.
95    order: HashMap<String, VecDeque<CacheKey>>,
96    /// The keys whose handler is running now, each with the lock its waiters queue on.
97    flights: HashMap<CacheKey, Arc<Mutex<()>>>,
98}

100#[derive(Debug, Clone, PartialEq, Eq, Hash)]
101struct CacheKey {
102    id: String,
103    input: String,
104    fingerprint: String,
105    /// What the policy adds to the registry fingerprint: empty for
106    /// [`CachePolicy::Process`], the context's reading for [`CachePolicy::Generation`], and
107    /// the reading with the checkout's fingerprint for [`CachePolicy::Checkout`].
108    state: String,
109}

114/// Whether this process was told to run every capability without its cache: `SWITCH`
115/// set to `off`.
116///
117/// ```
118/// use majordomus_cli::capability::executor::{bypassed, SWITCH};
119/// assert_eq!(bypassed(), std::env::var_os(SWITCH).is_some_and(|v| v == "off"));

122    std::env::var_os(SWITCH).is_some_and(|v| v == "off")

137    /// Execute a capability by id with a JSON input. The one place a handler runs.
138    pub fn execute(&self, ctx: &Context, id: &str, input: Value) -> Result<Value, CapabilityError> {
139        let policy = ctx.registry.get(id).map(|c| c.cache);
140        self.run(ctx, id, input, policy)
141    }

143    /// The same call with the cache stepped over: the handler runs, whatever is stored.
144    ///
145    /// One caller: the execution engine. A cached answer is indistinguishable from a fresh
146    /// one — that is the point of the cache and the rule `project.cache-is-invisible` says
147    /// so — but an *execution* is not an answer. It is a record of work happening, watched
148    /// while it happens, and an execution answered from a cache would report no steps, no
149    /// progress and no output of its own while claiming to have succeeded: a story of work
150    /// that did not occur. So an execution always runs the handler, and nothing it produces
151    /// is put into the cache either, because the cache holds what calls returned and this
152    /// call is a different kind of thing.
153    pub fn execute_uncached(
154        &self,
155        ctx: &Context,
156        id: &str,
157        input: Value,
158    ) -> Result<Value, CapabilityError> {
159        self.run(ctx, id, input, Some(CachePolicy::Disabled))
160    }
161
162    fn run(
163        &self,
164        ctx: &Context,
165        id: &str,
166        input: Value,
167        policy: Option<CachePolicy>,
168    ) -> Result<Value, CapabilityError> {
169        Counters::bump(&COUNTERS.executions);
170        let policy = policy.ok_or_else(|| CapabilityError::NotFound(format!("capability {id}")))?;
171        let policy = if bypassed() {
172            CachePolicy::Disabled
173        } else {
174            policy
175        };
176        let Some(key) = key_of(ctx, id, &input, policy) else {
177            return self.invoke(ctx, id, input);
178        };
179        let ttl = policy.ttl_seconds();
180        if let Some(value) = self.lookup(&key, ttl) {
181            return Ok(value);
182        }
183        // a miss: take the key's flight, so that one handler runs for it at a time, and look
184        // again once it is ours — a caller that waited is answered by the one it waited for
185        let flight = {
186            let mut cache = lock(&self.cache);
187            Arc::clone(cache.flights.entry(key.clone()).or_default())
188        };
189        let _flight = flight.lock().unwrap_or_else(|e| e.into_inner());
190        if let Some(value) = self.peek(&key, ttl) {
191            return Ok(value);
192        }
193        let value = self.invoke(ctx, id, input);
194        let mut cache = lock(&self.cache);
195        if cache
196            .flights
197            .get(&key)
198            .is_some_and(|f| Arc::ptr_eq(f, &flight))
199        {
200            cache.flights.remove(&key);
201        }
202        let value = value?;
203        let max_entries = policy.max_entries().unwrap_or(1);
204        let Cache { entries, order, .. } = &mut *cache;
205        let order = order.entry(key.id.clone()).or_default();
206        if !entries.contains_key(&key) {
207            order.push_back(key.clone());
208            while order.len() > max_entries {
209                if let Some(old) = order.pop_front() {
210                    entries.remove(&old);
211                    Counters::bump(&COUNTERS.cache_evictions);
212                }
213            }
214        }
215        entries.insert(
216            key,
217            CacheEntry {
218                value: value.clone(),
219                stored: Instant::now(),
220            },
221        );
222        Ok(value)
223    }

225    /// The stored answer for `key` when it is still fresh, counted as a hit or a miss.
226    fn lookup(&self, key: &CacheKey, ttl: Option<u64>) -> Option<Value> {
227        let _guard = perf::phase(Phase::CacheLookup);
228        let mut cache = lock(&self.cache);
229        let fresh = cache
230            .entries
231            .get(key)
232            .map(|e| ttl.is_none_or(|ttl| e.stored.elapsed() < Duration::from_secs(ttl)));
233        match fresh {
234            Some(true) => {
235                Counters::bump(&COUNTERS.cache_hits);
236                return cache.entries.get(key).map(|e| e.value.clone());
237            }
238            Some(false) => {
239                cache.entries.remove(key);
240                Counters::bump(&COUNTERS.cache_evictions);
241                Counters::bump(&COUNTERS.cache_misses);
242            }
243            None => Counters::bump(&COUNTERS.cache_misses),
244        }
245        None
246    }
247
248    /// The stored answer for `key` once its flight is ours: what the handler a caller waited
249    /// for stored. Counted as a hit, since no handler runs for it.
250    fn peek(&self, key: &CacheKey, ttl: Option<u64>) -> Option<Value> {
251        let cache = lock(&self.cache);
252        let value = cache
253            .entries
254            .get(key)
255            .filter(|e| ttl.is_none_or(|ttl| e.stored.elapsed() < Duration::from_secs(ttl)))
256            .map(|e| e.value.clone());
257        if value.is_some() {
258            Counters::bump(&COUNTERS.cache_hits);
259        }
260        value
261    }

263    /// Run the handler: counted, timed, and its writes noted.
264    fn invoke(&self, ctx: &Context, id: &str, input: Value) -> Result<Value, CapabilityError> {
265        Counters::bump(&COUNTERS.handler_invocations);
266        self.invocations.fetch_add(1, Ordering::Relaxed);
267        let value = {
268            let _guard = perf::phase(Phase::HandlerExecution);
269            ctx.registry.dispatch(ctx, id, input)?
270        };
271        // counted on the one path every transport and every execution shares, so a write
272        // made from a page, a tool, a route or an execution is seen by `crate::live` alike
273        if ctx
274            .registry
275            .get(id)
276            .is_some_and(|c| c.execution.effect == crate::capability::Effect::RepositoryMutation)
277        {
278            self.writes.fetch_add(1, Ordering::SeqCst);
279        }
280        Ok(value)
281    }

283    /// Count one repository mutation, as a successful mutating call does: what a test of a
284    /// reader of [`Executor::writes`] needs without a layer that a mutation can succeed on.
285    #[cfg(test)]
286    pub(crate) fn count_write(&self) {
287        self.writes.fetch_add(1, Ordering::SeqCst);
288    }

290    /// How many calls of a capability whose effect is a repository mutation have succeeded
291    /// through this executor.
292    ///
293    /// A write to a tracked file moves no git control file, so a long-lived process that
294    /// follows the repository by those files would go on answering from the picture it had
295    /// before its own write. The executor outlives a reload, so this number is what
296    /// `crate::live` compares: when it has moved, the process wrote, and the next reading
297    /// is a new generation.
298    ///
299    /// ```
300    /// use majordomus_cli::capability::executor::CapabilityExecutor;
301    /// let executor = CapabilityExecutor::new();
302    /// assert_eq!(executor.writes(), 0, "an executor that has run nothing has written nothing");
303    /// ```
304    pub fn writes(&self) -> u64 {
305        self.writes.load(Ordering::SeqCst)
306    }
307
308    /// How many entries the cache holds, all capabilities together.
309    pub fn cached_entries(&self) -> usize {
310        lock(&self.cache).entries.len()
311    }
312
313    /// How many handlers this executor has run — its cache misses and uncached calls, never
314    /// another executor's. Dropping the cache does not reset it.
315    ///
316    /// ```
317    /// use majordomus_cli::capability::executor::CapabilityExecutor;
318    /// let e = CapabilityExecutor::default();
319    /// assert_eq!(e.handler_invocations(), 0);
320    /// ```
321    pub fn handler_invocations(&self) -> u64 {
322        self.invocations.load(Ordering::Relaxed)
323    }

325    /// The answer this executor holds for `id` asked with `input` in the repository `ctx`
326    /// reads, when its cache has one that is still fresh, and `None` otherwise. No handler
327    /// runs, nothing is stored, and a miss is not counted: it is how the latest reading of a
328    /// source learns that the answer for a moved repository is already at hand
329    /// (`capability::readings`).
330    ///
331    /// ```
332    /// use majordomus_cli::synthetic::SyntheticRepository;
333    /// use serde_json::json;
334    ///
335    /// let repo = SyntheticRepository::small().unwrap();
336    /// let ctx = repo.context().unwrap();
337    /// let describe = json!({ "id": "plan.status" });
338    /// // nothing is held before the capability was asked, and asking this way runs nothing
339    /// assert_eq!(ctx.executor.cached(&ctx, "capabilities.describe", &describe), None);
340    /// assert_eq!(ctx.executor.handler_invocations(), 0);
341    /// // once it was asked, what it answered is at hand
342    /// let answer = ctx.execute("capabilities.describe", describe.clone()).unwrap();
343    /// assert_eq!(ctx.executor.cached(&ctx, "capabilities.describe", &describe), Some(answer));
344    /// // a capability that does not exist holds nothing
345    /// assert_eq!(ctx.executor.cached(&ctx, "no.such.capability", &json!({})), None);
346    /// ```
347    pub fn cached(&self, ctx: &Context, id: &str, input: &Value) -> Option<Value> {
348        if bypassed() {
349            return None;
350        }
351        let policy = ctx.registry.get(id).map(|c| c.cache)?;
352        let key = key_of(ctx, id, input, policy)?;
353        self.peek(&key, policy.ttl_seconds())
354    }
355
356    /// How many sources this executor remembers the latest reading of
357    /// (`capability::readings`).
358    ///
359    /// ```
360    /// use majordomus_cli::capability::executor::CapabilityExecutor;
361    /// assert_eq!(CapabilityExecutor::new().readings_remembered(), 0);
362    /// ```
363    pub fn readings_remembered(&self) -> usize {
364        self.readings.remembered()
365    }
366
367    /// Drop every cached entry.
368    pub fn clear(&self) {
369        let mut cache = lock(&self.cache);
370        cache.entries.clear();
371        cache.order.clear();
372    }
373}

375/// The key a call is kept under, or `None` when its policy keeps nothing — or keeps it per
376/// checkout and the checkout has no fingerprint.
377fn key_of(ctx: &Context, id: &str, input: &Value, policy: CachePolicy) -> Option<CacheKey> {
378    let state = match policy {
379        CachePolicy::Disabled => return None,
380        CachePolicy::Process { .. } => String::new(),
381        CachePolicy::Generation { .. } => format!("reading {}", ctx.reading),
382        CachePolicy::Checkout { .. } => {
383            let root = std::path::Path::new(&ctx.index.repository.root);
384            let checkout = super::checkout::fingerprint(root)?;
385            format!("reading {} checkout {checkout}", ctx.reading)
386        }
387    };
388    Some(CacheKey {
389        id: id.to_string(),
390        input: canonical_json(input),
391        fingerprint: ctx.registry.fingerprint().to_string(),
392        state,
393    })
394}

396fn lock(m: &Mutex<Cache>) -> std::sync::MutexGuard<'_, Cache> {
397    m.lock().unwrap_or_else(|e| e.into_inner())
398}

400/// The input as one canonical string: object keys sorted at every level, no whitespace,
401/// so that two inputs equal as data are equal as keys whatever order a client wrote them.
402///
403/// ```
404/// use majordomus_cli::capability::executor::canonical_json;
405/// use serde_json::json;
406/// assert_eq!(canonical_json(&json!({"b": 2, "a": {"d": 1, "c": [3, {"z": 0, "y": 1}]}})),
407///            canonical_json(&json!({"a": {"c": [3, {"y": 1, "z": 0}], "d": 1}, "b": 2})));
408/// assert_ne!(canonical_json(&json!([1, 2])), canonical_json(&json!([2, 1])), "arrays are ordered");
409/// ```
410pub fn canonical_json(v: &Value) -> String {
411    fn write(v: &Value, out: &mut String) {
412        match v {
413            Value::Object(m) => {
414                let mut keys: Vec<&String> = m.keys().collect();
415                keys.sort();
416                out.push('{');
417                for (i, k) in keys.iter().enumerate() {
418                    if i > 0 {
419                        out.push(',');
420                    }
421                    out.push_str(&serde_json::to_string(k).unwrap_or_default());
422                    out.push(':');
423                    write(&m[*k], out);
424                }
425                out.push('}');
426            }
427            Value::Array(a) => {
428                out.push('[');
429                for (i, x) in a.iter().enumerate() {
430                    if i > 0 {
431                        out.push(',');
432                    }
433                    write(x, out);
434                }
435                out.push(']');
436            }
437            other => out.push_str(&other.to_string()),
438        }
439    }
440    let mut out = String::new();
441    write(v, &mut out);
442    out
