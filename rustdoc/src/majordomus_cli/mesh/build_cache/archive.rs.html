1//! The archive of one cached directory: written with the system's `tar` and gzip, read back
2//! only after every entry has been judged safe (I3952, ADR 0157 decision 9).
3//!
4//! **Extracted beside, never over.** An archive is unpacked into a fresh directory next to its
5//! target, on the same filesystem, so that putting it in place is a rename: a reader of the
6//! target sees the old content or the new, never half of one.
7//!
8//! **Judged twice.** The names `tar` lists are refused when one is absolute or climbs with
9//! `..`, before anything is written; then the unpacked tree is walked without following a
10//! link, and a link that resolves outside the tree, or a file that is neither a regular file,
11//! a directory nor a link, refuses the whole archive. A refusal is a miss: the caller runs
12//! the install itself, exactly as it would have without the cache.
13//!
14//! **Bounded.** Every `tar` this module runs carries [`TAR_BUDGET`]; one that does not finish
15//! within it is killed and counted as a failure (rule `project.every-wait-is-bounded`).
16

28/// The longest one `tar` may run: a cargo build directory of three gigabytes is written in
29/// about a minute on the machines this was measured on, so this is a stall, not a slow disk.
30pub(crate) const TAR_BUDGET: Duration = Duration::from_secs(1800);
31
32/// The lock file cargo holds on a build directory's profile from its first compile to its
33/// last, and the one a target directory of an older cargo has.
34pub(crate) const CARGO_LOCKS: [&str; 2] = [".cargo-build-lock", ".cargo-lock"];

36/// The SHA-256 of a file, streamed, as lowercase hex.
37pub(crate) fn sha256_file(path: &Path) -> io::Result<String> {
38    let mut file = File::open(path)?;
39    let mut hasher = Sha256::new();
40    let mut buf = vec![0u8; 1 << 20];
41    loop {
42        let n = file.read(&mut buf)?;
43        if n == 0 {
44            break;
45        }
46        hasher.update(&buf[..n]);
47    }
48    Ok(hex(&hasher.finalize()))
49}

51/// A directory that is removed when this value is dropped, whatever happened in between.
52#[derive(Debug)]
53pub(crate) struct Scratch {
54    pub(crate) path: PathBuf,
55}
56
57impl Scratch {
58    /// A new, empty directory beside `target` (same parent, so on the same filesystem), named
59    /// after it, this process and the moment.
60    pub(crate) fn beside(target: &Path, purpose: &str) -> io::Result<Scratch> {
61        let parent = target.parent().ok_or_else(|| {
62            io::Error::new(io::ErrorKind::InvalidInput, "a target needs a parent")
63        })?;
64        fs::create_dir_all(parent)?;
65        let name = target
66            .file_name()
67            .map(|n| n.to_string_lossy().into_owned())
68            .unwrap_or_else(|| "target".into());
69        let nanos = std::time::SystemTime::now()
70            .duration_since(std::time::UNIX_EPOCH)
71            .map(|d| d.as_nanos())
72            .unwrap_or(0);
73        let path = parent.join(format!(
74            ".{name}.mj-{purpose}-{}-{nanos}",
75            std::process::id()
76        ));
77        fs::create_dir(&path)?;
78        Ok(Scratch { path })
79    }
80}

82impl Drop for Scratch {
83    fn drop(&mut self) {
84        let _ = fs::remove_dir_all(&self.path);
85    }

88fn run_tar(args: &[&std::ffi::OsStr]) -> Result<Vec<u8>, String> {
89    run_tar_within(args, TAR_BUDGET)
90}

92/// [`run_tar`] with the budget as a parameter, so that a stall is testable in milliseconds.
93fn run_tar_within(args: &[&std::ffi::OsStr], budget: Duration) -> Result<Vec<u8>, String> {
94    let mut command = Command::new("tar");
95    command.args(args);
96    // macOS tar otherwise adds an AppleDouble `._name` entry for every file with extended
97    // attributes, which another platform would unpack as files of its own
98    command.env("COPYFILE_DISABLE", "1");
99    let output = bounded_output(&mut command, budget)
100        .ok_or_else(|| "tar did not start, or did not finish within its budget".to_string())?;
101    if output.status.success() {
102        Ok(output.stdout)
103    } else {
104        let err = String::from_utf8_lossy(&output.stderr);
105        Err(format!(
106            "tar failed: {}",
107            err.lines().next().unwrap_or("no message").trim()
108        ))
109    }
110}

112/// Write `source`'s content (not `source` itself) into the gzip archive `dest`, leaving out
113/// every entry whose name is one of `exclude`.
114pub(crate) fn create(source: &Path, dest: &Path, exclude: &[&str]) -> Result<(), String> {
115    let excludes: Vec<std::ffi::OsString> = exclude
116        .iter()
117        .map(|e| std::ffi::OsString::from(format!("--exclude={e}")))
118        .collect();
119    let mut args: Vec<&std::ffi::OsStr> = vec!["-czf".as_ref(), dest.as_os_str()];
120    args.extend(excludes.iter().map(|e| e.as_os_str()));
121    args.extend(["-C".as_ref(), source.as_os_str(), ".".as_ref()]);
122    run_tar(&args).map(|_| ())
123}

125/// Why one listed name may not be unpacked, or `None` when it may.
126pub(crate) fn refuse_name(name: &str) -> Option<String> {
127    if name.starts_with('/') {
128        return Some(format!("{name}: an absolute name"));
129    }
130    if Path::new(name)
131        .components()
132        .any(|c| matches!(c, Component::ParentDir))
133    {
134        return Some(format!("{name}: climbs out with .."));
135    }
136    None
137}

139/// Every name the archive lists, judged before anything is written.
140pub(crate) fn check_listing(archive: &Path) -> Result<usize, String> {
141    let listing = run_tar(&["-tzf".as_ref(), archive.as_os_str()])?;
142    let text = String::from_utf8_lossy(&listing);
143    let mut n = 0;
144    for name in text.lines().filter(|l| !l.is_empty()) {
145        if let Some(why) = refuse_name(name) {
146            return Err(format!("the archive was refused: {why}"));
147        }
148        n += 1;
149    }
150    Ok(n)
151}

153/// Where a link at `link` (relative to the tree's root) with the relative target `to` lands,
154/// lexically; `None` when it leaves the tree. An absolute target starts with a root (or a
155/// prefix), and that component alone refuses it.
156pub(crate) fn resolve_inside(link: &Path, to: &Path) -> Option<PathBuf> {
157    let mut at: Vec<std::ffi::OsString> = link
158        .parent()
159        .map(|p| {
160            p.components()
161                .filter_map(|c| match c {
162                    Component::Normal(s) => Some(s.to_os_string()),
163                    _ => None,
164                })
165                .collect()
166        })
167        .unwrap_or_default();
168    for c in to.components() {
169        match c {
170            Component::Normal(s) => at.push(s.to_os_string()),
171            Component::CurDir => {}
172            Component::ParentDir => {
173                at.pop()?;
174            }
175            Component::RootDir | Component::Prefix(_) => return None,
176        }
177    }
178    Some(at.iter().collect())
179}

181/// Walk an unpacked tree without following a link: every entry is a directory, a regular
182/// file or a link that stays inside the tree.
183pub(crate) fn check_tree(root: &Path) -> Result<(), String> {
184    let mut stack = vec![PathBuf::new()];
185    while let Some(rel) = stack.pop() {
186        let dir = root.join(&rel);
187        let entries = listed(&dir).map_err(|e| format!("{}: {e}", rel.display()))?;
188        for (entry, kind) in entries {
189            let rel_entry = rel.join(entry.file_name());
190            if kind.is_dir() {
191                stack.push(rel_entry);
192            } else if kind.is_symlink() {
193                if let Some(why) = link_refusal(&entry.path(), &rel_entry) {
194                    return Err(why);
195                }
196            } else if !kind.is_file() {
197                return Err(format!(
198                    "the archive was refused: {} is neither a file, a directory nor a link",
199                    rel_entry.display()
200                ));
201            }
202        }
203    }
204    Ok(())
205}

207/// Why the link at `path` (at `rel` in the tree) is refused: it leaves the tree, or it cannot
208/// be read; `None` when it stays inside.
209fn link_refusal(path: &Path, rel: &Path) -> Option<String> {
210    match fs::read_link(path) {
211        Ok(to) if resolve_inside(rel, &to).is_some() => None,
212        Ok(to) => Some(format!(
213            "the archive was refused: {} links to {}, outside its target",
214            rel.display(),
215            to.display()
216        )),
217        Err(e) => Some(e.to_string()),
218    }
219}

221/// Unpack `archive` into a fresh directory beside `target`, judged before and after. The
222/// returned directory is removed when dropped unless its content was moved out first.
223pub(crate) fn unpack_beside(archive: &Path, target: &Path) -> Result<Scratch, String> {
224    check_listing(archive)?;
225    let scratch = Scratch::beside(target, "restore").map_err(|e| e.to_string())?;
226    run_tar(&[
227        "-xzf".as_ref(),
228        archive.as_os_str(),
229        "-C".as_ref(),
230        scratch.path.as_os_str(),
231    ])?;
232    check_tree(&scratch.path)?;
233    Ok(scratch)
234}

236/// Put an unpacked tree in place of `target`: the old target, if any, is moved aside, the
237/// new one renamed in, and the old one removed. A failed rename puts the old one back.
238pub(crate) fn replace(scratch: Scratch, target: &Path) -> Result<(), String> {
239    let aside = if target.symlink_metadata().is_ok() {
240        let trash = Scratch::beside(target, "old").map_err(|e| e.to_string())?;
241        let into = trash.path.join("old");
242        fs::rename(target, &into)
243            .map_err(|e| format!("{} cannot be moved aside: {e}", target.display()))?;
244        Some((trash, into))
245    } else {
246        None
247    };
248    if let Err(e) = fs::rename(&scratch.path, target) {
249        if let Some((_, into)) = &aside {
250            let _ = fs::rename(into, target);
251        }
252        return Err(format!("{} cannot be put in place: {e}", target.display()));
253    }
254    // the scratch path is now the target: nothing is left for its guard to remove
255    std::mem::forget(scratch);
256    Ok(())
257}

259/// Is a build directory cold: absent, or holding nothing but directories, cargo's lock files
260/// and its `CACHEDIR.TAG`?
261pub(crate) fn is_cold(dir: &Path) -> bool {
262    let mut stack = vec![dir.to_path_buf()];
263    while let Some(d) = stack.pop() {
264        let Ok(entries) = fs::read_dir(&d) else {
265            // absent is cold; a directory that cannot be read is not known to be
266            return d == dir && !dir.exists();
267        };
268        for entry in entries.flatten() {
269            let name = entry.file_name();
270            let name = name.to_string_lossy();
271            match entry.file_type() {
272                Ok(t) if t.is_dir() => stack.push(entry.path()),
273                Ok(_) if CARGO_LOCKS.contains(&name.as_ref()) || name == "CACHEDIR.TAG" => {}
274                _ => return false,
275            }
276        }
277    }
278    true
279}

281/// How many times a lock is tried, 20 ms apart, before it counts as held.
282const LOCK_TRIES: u32 = 10;
283
284/// An exclusive flock(2) held on one file until dropped.
285#[derive(Debug)]
286pub(crate) struct Held {
287    _file: File,
288}
289
290/// Take cargo's lock on `profile` (its `.cargo-build-lock`, created when absent) without
291/// waiting: `None` when some build holds it.
292pub(crate) fn lock_profile(profile: &Path, create: bool) -> Result<Option<Held>, String> {
293    use std::os::fd::AsRawFd;
294    let path = profile.join(CARGO_LOCKS[0]);
295    if !create && !path.exists() {
296        return Ok(Some(Held {
297            _file: File::open(profile).map_err(|e| e.to_string())?,
298        }));
299    }
300    fs::create_dir_all(profile).map_err(|e| e.to_string())?;
301    let file = fs::OpenOptions::new()
302        .read(true)
303        .append(true)
304        .create(true)
305        .open(&path)
306        .map_err(|e| format!("{}: {e}", path.display()))?;
307    // A few tries a moment apart, never a wait on a build: cargo holds its lock for the whole
308    // build, while the copy of a released descriptor that a child forked in another thread
309    // keeps until it execs is gone within milliseconds.
310    for attempt in 0..LOCK_TRIES {
311        // SAFETY: the descriptor is this open file's own, and flock only reads it.
312        let rc = unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX | libc::LOCK_NB) };
313        if rc == 0 {
314            return Ok(Some(Held { _file: file }));
315        }
316        if attempt + 1 < LOCK_TRIES {
317            std::thread::sleep(Duration::from_millis(20));
318        }
319    }
320    Ok(None)
321}

323/// Hold cargo's lock on every profile directory directly under `build_dir` that `names`
324/// lists; `Err` naming the first one some build holds.
325pub(crate) fn lock_profiles(
326    build_dir: &Path,
327    names: &[String],
328    create: bool,
329) -> Result<Vec<Held>, String> {
330    let mut held = Vec::new();
331    for name in names {
332        match lock_profile(&build_dir.join(name), create)? {
333            Some(h) => held.push(h),
334            None => {
335                return Err(format!(
336                    "a build holds cargo's lock on {}",
337                    build_dir.join(name).display()
338                ))
339            }
340        }
341    }
342    Ok(held)
343}

345/// The directories directly under `dir`, by name.
346pub(crate) fn subdirectories(dir: &Path) -> Vec<String> {
347    let names: std::collections::BTreeSet<String> = fs::read_dir(dir)
348        .map(|it| {
349            it.flatten()
350                .filter(|e| e.file_type().map(|t| t.is_dir()).unwrap_or(false))
351                .map(|e| e.file_name().to_string_lossy().into_owned())
352                .collect()
353        })
354        .unwrap_or_default();
355    names.into_iter().collect()
356}

358/// Every entry of `dir` with its own type (a link is not followed), read in full: an entry
359/// or a type that cannot be read fails the listing as `dir` itself failing to open does.
360fn listed(dir: &Path) -> io::Result<Vec<(fs::DirEntry, fs::FileType)>> {
361    fs::read_dir(dir)?
362        .map(|entry| entry.and_then(|e| e.file_type().map(|kind| (e, kind))))
363        .collect()
364}

366/// Move every entry of `from` into `into`, renaming whole subtrees where `into` has none and
367/// descending where both have a directory; an entry `into` already has is kept as it is.
368pub(crate) fn merge_into(from: &Path, into: &Path) -> io::Result<()> {
369    fs::create_dir_all(into)?;
370    for (entry, kind) in listed(from)? {
371        let dest = into.join(entry.file_name());
372        match dest.symlink_metadata() {
373            Err(_) => fs::rename(entry.path(), &dest)?,
374            Ok(meta) if meta.is_dir() && kind.is_dir() => merge_into(&entry.path(), &dest)?,
375            Ok(_) => {}
376        }
377    }
378    Ok(())
379}

381/// Put an unpacked build directory in place of a cold one, under cargo's own lock on each of
382/// its profiles: the build directory is judged cold again while the locks are held, so a build
383/// that started meanwhile is never written under.
384pub(crate) fn place_build_dir(scratch: Scratch, build_dir: &Path) -> Result<(), String> {
385    let profiles = subdirectories(&scratch.path);
386    let _held = lock_profiles(build_dir, &profiles, true)?;
387    if !is_cold(build_dir) {
388        return Err(format!(
389            "{} is no longer cold: a build wrote to it meanwhile",
390            build_dir.display()
391        ));
392    }
393    merge_into(&scratch.path, build_dir).map_err(|e| {
394        format!(
395            "{} could not be filled from the archive: {e}",
396            build_dir.display()
397        )
398    })
399}

401/// Total bytes of the regular files under `dir`, links not followed.
402pub(crate) fn tree_bytes(dir: &Path) -> u64 {
403    let mut total = 0;
404    let mut stack = vec![dir.to_path_buf()];
405    while let Some(d) = stack.pop() {
406        let Ok(entries) = fs::read_dir(&d) else {
407            continue;
408        };
409        for entry in entries.flatten() {
410            match entry.file_type() {
411                Ok(t) if t.is_dir() => stack.push(entry.path()),
412                Ok(t) if t.is_file() => total += entry.metadata().map(|m| m.len()).unwrap_or(0),
413                _ => {}
414            }
415        }
416    }
417    total
418}

562            fs::read_to_string(b.join("debug/deps/a.rlib")).unwrap(),
563            "a"
564        );
565        assert!(b.join("debug/.cargo-build-lock").exists());
566        assert!(b.join("CACHEDIR.TAG").exists());
567        assert!(tree_bytes(&b) >= 2);
568
569        // now warm: a second restore is refused
570        let scratch = Scratch::beside(&b, "restore").unwrap();
571        tree(&scratch.path, &[("debug/deps/b.rlib", "b")]);
572        assert!(place_build_dir(scratch, &b)
573            .unwrap_err()
574            .contains("no longer cold"));
575        assert!(!b.join("debug/deps/b.rlib").exists());
576
577        // a held lock refuses, without waiting
578        let c = dir.path().join("cold");
579        let holder = lock_profile(&c.join("debug"), true)
580            .unwrap()
581            .expect("first lock");
582        let scratch = Scratch::beside(&c, "restore").unwrap();
583        tree(&scratch.path, &[("debug/deps/c.rlib", "c")]);
584        assert!(place_build_dir(scratch, &c)
585            .unwrap_err()
586            .contains("holds cargo's lock"));
587        drop(holder);
588        assert!(
589            lock_profiles(&c, &["debug".into()], false).is_ok(),
590            "released, it is taken again"
591        );
592        assert_eq!(subdirectories(&c), vec!["debug".to_string()]);
593    }

605    /// A gzip tar archive written by hand, so that it can carry what `tar -c` never writes:
606    /// `(name, typeflag, link name, content)`.
607    fn forged(entries: &[(&str, u8, &str, &str)]) -> Vec<u8> {
608        let mut raw = Vec::new();
609        for (name, flag, link, body) in entries {
610            let mut h = [0u8; 512];
611            h[..name.len()].copy_from_slice(name.as_bytes());
612            h[100..107].copy_from_slice(b"0000644");
613            h[108..115].copy_from_slice(b"0000000");
614            h[116..123].copy_from_slice(b"0000000");
615            h[124..135].copy_from_slice(format!("{:011o}", body.len()).as_bytes());
616            h[136..147].copy_from_slice(b"00000000000");
617            h[156] = *flag;
618            h[157..157 + link.len()].copy_from_slice(link.as_bytes());
619            h[257..263].copy_from_slice(b"ustar\0");
620            h[263..265].copy_from_slice(b"00");
621            h[148..156].copy_from_slice(b"        ");
622            let sum: u32 = h.iter().map(|b| u32::from(*b)).sum();
623            h[148..156].copy_from_slice(format!("{sum:06o}\0 ").as_bytes());
624            raw.extend_from_slice(&h);
625            raw.extend_from_slice(body.as_bytes());
626            raw.resize(raw.len().div_ceil(512) * 512, 0);
627        }
628        raw.extend_from_slice(&[0u8; 1024]);
629        let mut gzip = Command::new("gzip")
630            .arg("-c")
631            .stdin(std::process::Stdio::piped())
632            .stdout(std::process::Stdio::piped())
633            .spawn()
634            .unwrap();
635        let mut stdin = gzip.stdin.take().unwrap();
636        let writer = std::thread::spawn(move || {
637            use std::io::Write;
638            stdin.write_all(&raw).unwrap();
639        });
640        let out = gzip.wait_with_output().unwrap();
641        writer.join().unwrap();
642        out.stdout
643    }
