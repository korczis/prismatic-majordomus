1//! How fast the integrator turns work into master, folded from its trail.
2//!
3//! Like [`super::wait`], everything here is a fold over the audit trail
4//! ([`super::drain::events`]); nothing is stored beside it, so the numbers cannot drift from
5//! the record they describe. A window bounds the fold: only events at or after `now - window`
6//! count, so a quiet week reads as a quiet week rather than as the average of the whole
7//! history.
8//!
9//! Five facts are folded:
10//!
11//! - **merges** in the window, and merges per day;
12//! - **actionable → merged**: per merged pull request, from the last time it became actionable
13//!   to its merge, as a median — how long ready work waits for the executor;
14//! - **CI rounds per merge**: the `refreshed` events since the previous merge, per merge, as a
15//!   median. Every refresh pushes a new head that the required checks must run on again, and
16//!   that is where the time between merges goes;
17//! - **cycle**: from a selection to its outcome (merged, failed, refused as stale), as a
18//!   median;
19//! - **failures** by kind: stale decisions, failed merges, failed verifications.
20//!
21//! A median with nothing to take the median of is `None`, never zero: zero would claim that
22//! work moved instantly.
23

32/// The integrator's throughput over a window of the trail.
33#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
34pub struct IntegrationThroughput {
35    /// The window, in days, ending at the time the fold was asked for.
36    pub window_days: u64,
37    /// Pull requests merged in the window.
38    pub merges: u64,
39    /// `merges / window_days`.
40    pub merges_per_day: f64,
41    /// Median seconds from becoming actionable to being merged; `None` when no merge in the
42    /// window had a recorded `became_actionable` before it.
43    pub median_actionable_to_merged_secs: Option<u64>,
44    /// Median `refreshed` events between one merge and the next; `None` without a merge.
45    pub ci_rounds_per_merge: Option<u64>,
46    /// Median seconds from a selection to its outcome; `None` when no selection in the window
47    /// reached one.
48    pub median_cycle_secs: Option<u64>,
49    /// Decisions refused because master or the head moved after they were taken.
50    pub stale_decisions: u64,
51    /// Merge attempts that failed.
52    pub merge_failures: u64,
53    /// Merges whose resulting master failed verification.
54    pub verification_failures: u64,
55    /// Events in the trail whose time could not be read; they count toward nothing.
56    pub unreadable_events: u64,
57}
58
59/// What the fold needs to know about one event. The trail's own action words are mapped here
60/// and nowhere else.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62enum Kind {
63    BecameActionable,
64    Selected,
65    Refreshed,
66    MergeSucceeded,
67    MergeFailed,
68    VerificationFailed,
69    StaleDecision,
70    Other,
71}
72
73fn kind_of(e: &IntegrationEvent) -> Kind {
74    use super::drain::IntegrationAction as A;
75    match e.action {
76        A::BecameActionable => Kind::BecameActionable,
77        A::Selected | A::RefreshSelected => Kind::Selected,
78        // a repaired head is a CI round exactly like a refreshed one
79        A::Refreshed | A::Repaired => Kind::Refreshed,
80        A::MergeSucceeded => Kind::MergeSucceeded,
81        A::MergeFailed => Kind::MergeFailed,
82        A::VerificationFailed => Kind::VerificationFailed,
83        A::StaleDecision => Kind::StaleDecision,
84        _ => Kind::Other,
85    }
86}

88/// The median of `v`, lower of the two middle values for an even count, so it is always a
89/// value that was observed. `None` for an empty slice.
90fn median(mut v: Vec<u64>) -> Option<u64> {
91    if v.is_empty() {
92        return None;
93    }
94    // the middle element alone, selected: a median needs no ordered collection
95    let middle = (v.len() - 1) / 2;
96    Some(*v.select_nth_unstable(middle).1)
97}

99/// Fold `trail` (oldest first, as [`super::drain::events`] returns it) over the `window_days`
100/// ending at `now` (seconds since the epoch). Deterministic for a given trail and `now`; the
101/// unit tests below hold each fact.
102pub fn throughput(trail: &[IntegrationEvent], now: u64, window_days: u64) -> IntegrationThroughput {
103    let since = now.saturating_sub(window_days.saturating_mul(86_400));
104    let mut out = IntegrationThroughput {
105        window_days,
106        merges: 0,
107        merges_per_day: 0.0,
108        median_actionable_to_merged_secs: None,
109        ci_rounds_per_merge: None,
110        median_cycle_secs: None,
111        stale_decisions: 0,
112        merge_failures: 0,
113        verification_failures: 0,
114        unreadable_events: 0,
115    };
116    // Facts that open before the window still close inside it: a pull request that became
117    // actionable last week and merged today waited across the boundary, so state is folded
118    // over the whole trail and only outcomes are filtered by the window.
119    let mut actionable_since: BTreeMap<u64, u64> = BTreeMap::new();
120    let mut selected_at: BTreeMap<u64, u64> = BTreeMap::new();
121    let mut refreshes_since_merge: u64 = 0;
122    let (mut waits, mut rounds, mut cycles) = (Vec::new(), Vec::new(), Vec::new());
123
124    for e in trail {
125        let Some(at) = parse_rfc3339(&e.at) else {
126            out.unreadable_events += 1;
127            continue;
128        };
129        let inside = at >= since && at <= now;
130        let kind = kind_of(e);
131        match (kind, e.pr) {
132            (Kind::BecameActionable, Some(n)) => {
133                actionable_since.insert(n, at);
134            }
135            (Kind::Selected, Some(n)) => {
136                selected_at.insert(n, at);
137            }
138            (Kind::Refreshed, _) => refreshes_since_merge += 1,
139            (Kind::MergeSucceeded, pr) => {
140                if inside {
141                    out.merges += 1;
142                    rounds.push(refreshes_since_merge);
143                    if let Some(from) = pr.and_then(|n| actionable_since.get(&n)) {
144                        waits.push(at.saturating_sub(*from));
145                    }
146                }
147                refreshes_since_merge = 0;
148            }
149            _ => {}
150        }
151        // a selection's outcome closes its cycle
152        if matches!(
153            kind,
154            Kind::MergeSucceeded
155                | Kind::MergeFailed
156                | Kind::VerificationFailed
157                | Kind::StaleDecision
158        ) {
159            if let Some(n) = e.pr {
160                if let Some(from) = selected_at.remove(&n) {
161                    if inside {
162                        cycles.push(at.saturating_sub(from));
163                    }
164                }
165                if kind == Kind::MergeSucceeded {
166                    actionable_since.remove(&n);
167                }
168            }
169            if inside {
170                match kind {
171                    Kind::MergeFailed => out.merge_failures += 1,
172                    Kind::VerificationFailed => out.verification_failures += 1,
173                    Kind::StaleDecision => out.stale_decisions += 1,
174                    _ => {}
175                }
176            }
177        }
178    }
179
180    if window_days > 0 {
181        out.merges_per_day = out.merges as f64 / window_days as f64;
182    }
183    out.median_actionable_to_merged_secs = median(waits);
184    out.ci_rounds_per_merge = median(rounds);
185    out.median_cycle_secs = median(cycles);
186    out
187}

189#[cfg(test)]
190mod tests {
191    use super::*;
192    use crate::peers::rfc3339;
193    use std::time::{Duration, UNIX_EPOCH};
194
195    const T0: u64 = 1_790_000_000;
196    const DAY: u64 = 86_400;
197
198    fn ev(at: u64, action: &str, pr: Option<u64>) -> IntegrationEvent {
199        // the trail's wire word, read the way a trail line is read
200        let action = serde_json::from_value(serde_json::Value::String(action.into()))
201            .unwrap_or_else(|e| panic!("{action} is not an integration action: {e}"));
202        IntegrationEvent {
203            at: rfc3339(UNIX_EPOCH + Duration::from_secs(at)),
204            actor: "test".into(),
205            pr,
206            ..IntegrationEvent::of(action)
207        }
208    }
209
210    #[test]
211    fn an_empty_trail_moved_nothing_and_claims_no_speed() {
212        let t = throughput(&[], T0, 7);
213        assert_eq!(t.merges, 0);
214        assert_eq!(t.merges_per_day, 0.0);
215        assert_eq!(t.median_actionable_to_merged_secs, None);
216        assert_eq!(t.ci_rounds_per_merge, None);
217        assert_eq!(t.median_cycle_secs, None);
218    }
219
220    #[test]
221    fn a_merge_is_timed_from_becoming_actionable_and_from_its_selection() {
222        let trail = [
223            ev(T0, "became_actionable", Some(1)),
224            ev(T0 + 600, "selected", Some(1)),
225            ev(T0 + 660, "merge_attempted", Some(1)),
226            ev(T0 + 900, "merge_succeeded", Some(1)),
227        ];
228        let t = throughput(&trail, T0 + DAY, 7);
229        assert_eq!(t.merges, 1);
230        assert_eq!(t.median_actionable_to_merged_secs, Some(900));
231        assert_eq!(t.median_cycle_secs, Some(300));
232        assert_eq!(t.ci_rounds_per_merge, Some(0));
233    }
234
235    #[test]
236    fn a_repaired_head_is_a_ci_round_like_a_refreshed_one() {
237        let trail = [
238            ev(T0, "repaired", Some(1)),
239            ev(T0 + 10, "refreshed", Some(1)),
240            ev(T0 + 20, "merge_succeeded", Some(1)),
241        ];
242        let t = throughput(&trail, T0 + DAY, 7);
243        assert_eq!(t.merges, 1);
244        assert_eq!(t.ci_rounds_per_merge, Some(2));
245    }
246
247    #[test]
248    fn ci_rounds_are_the_refreshes_between_one_merge_and_the_next() {
249        let trail = [
250            ev(T0, "refreshed", Some(2)),
251            ev(T0 + 10, "refreshed", Some(2)),
252            ev(T0 + 20, "merge_succeeded", Some(1)),
253            ev(T0 + 30, "refreshed", Some(2)),
254            ev(T0 + 40, "merge_succeeded", Some(2)),
255            ev(T0 + 50, "merge_succeeded", Some(3)),
256        ];
257        let t = throughput(&trail, T0 + DAY, 7);
258        assert_eq!(t.merges, 3);
259        // rounds per merge: 2, 1, 0 -> median 1
260        assert_eq!(t.ci_rounds_per_merge, Some(1));
261    }
262
263    #[test]
264    fn the_median_is_an_observed_value() {
265        assert_eq!(median(vec![]), None);
266        assert_eq!(median(vec![5]), Some(5));
267        assert_eq!(median(vec![9, 1, 5]), Some(5));
268        // even count: the lower middle, a value that happened
269        assert_eq!(median(vec![1, 2, 3, 100]), Some(2));
270    }
271
272    #[test]
273    fn only_outcomes_inside_the_window_count_but_waits_may_start_before_it() {
274        let trail = [
275            ev(T0, "became_actionable", Some(7)),
276            ev(T0 + 10, "merge_succeeded", Some(6)),
277            ev(T0 + 10 * DAY, "merge_succeeded", Some(7)),
278        ];
279        let t = throughput(&trail, T0 + 10 * DAY, 7);
280        assert_eq!(
281            t.merges, 1,
282            "the merge ten days ago is outside a 7-day window"
283        );
284        assert_eq!(t.median_actionable_to_merged_secs, Some(10 * DAY));
285        assert!((t.merges_per_day - 1.0 / 7.0).abs() < 1e-9);
286    }
287
288    #[test]
289    fn failures_are_counted_by_kind_and_close_their_cycle() {
290        let trail = [
291            ev(T0, "selected", Some(1)),
292            ev(T0 + 5, "stale_decision", Some(1)),
293            ev(T0 + 10, "selected", Some(2)),
294            ev(T0 + 70, "merge_failed", Some(2)),
295            ev(T0 + 80, "verification_failed", Some(3)),
296        ];
297        let t = throughput(&trail, T0 + DAY, 7);
298        assert_eq!(
299            (t.stale_decisions, t.merge_failures, t.verification_failures),
300            (1, 1, 1)
301        );
302        assert_eq!(t.merges, 0);
303        // cycles 5 and 60 -> lower middle 5
304        assert_eq!(t.median_cycle_secs, Some(5));
305    }

307    #[test]
308    fn an_unreadable_time_counts_toward_nothing_and_is_reported() {
309        let mut bad = ev(T0, "merge_succeeded", Some(1));
310        bad.at = "yesterday".into();
311        let t = throughput(&[bad], T0 + DAY, 7);
312        assert_eq!(t.merges, 0);
313        assert_eq!(t.unreadable_events, 1);

331mod window_branches {
332    //! Outcomes outside the window, outcomes without a pull request, and a window of no days.
333
334    use super::*;
335    use crate::integration::drain::{IntegrationAction, IntegrationEvent};
336
337    fn at(action: IntegrationAction, pr: Option<u64>, when: &str) -> IntegrationEvent {
338        IntegrationEvent {
339            at: when.into(),
340            pr,
341            ..IntegrationEvent::of(action)
342        }
343    }
344
345    #[test]
346    fn a_cycle_closed_before_the_window_counts_nothing_and_no_days_divide_nothing() {
347        let trail = vec![
348            at(IntegrationAction::Selected, Some(1), "2026-01-01T00:00:00Z"),
349            at(
350                IntegrationAction::MergeFailed,
351                Some(1),
352                "2026-01-01T00:01:00Z",
353            ),
354            at(IntegrationAction::MergeFailed, None, "2026-01-01T00:02:00Z"),
355        ];
356        let now = parse_rfc3339("2026-10-01T00:00:00Z").unwrap();
357        let t = throughput(&trail, now, 7);
358        assert_eq!(t.merge_failures, 0, "outside the window");
359        let t = throughput(&trail, now, 0);
360        assert_eq!(t.merges_per_day, 0.0);
361    }
