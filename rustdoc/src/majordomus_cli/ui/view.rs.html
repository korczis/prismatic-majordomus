1//! The UI's components: the hero band of a screen, its KPI group, a panel, a donut, a
2//! timeline and a meter. Each is a function of readings (`super::source`), never of a value
3//! of its own, and each says where it read from.
4//!
5//! The source of a value is shown in the UI itself — a tile's title, a panel's "Source"
6//! line. The UI is its own product; the Cockpit is its low-level reference, so each source
7//! carries one quiet "technical view" link to the Cockpit's page for that capability, for the
8//! reader who wants the registry behind a value. An object a screen names and the UI has no
9//! screen of its own for — a document, a test, a declaring file, an ADR — leads to the
10//! Cockpit's page for that object, its technical view, and only when the index holds it.
11//! Nothing else leads there (ADR 0135: the diagnostic view of the capability, object or graph
12//! a panel read).
13//!
14//! Every entity a screen names is a link: to the UI's own screen or inspector for it when
15//! there is one, to its technical view when there is not, and to the address its source gives
16//! when it lives elsewhere. A value a source does not have is said as `none`, never left blank,
17//! and a list cut short says how much is left out and where the rest is.
18//!
19//! Charts are drawn here as SVG, on the server: they are complete without JavaScript, they
20//! carry their numbers as text beside them, and they draw only what a capability answered —
21//! a series nobody measured is not drawn at all.
22

88/// A KPI tile whose value is the length of the list or map at the reading's pointer: the
89/// one count a screen may take of an answer, because it is the answer's own size.
90pub fn kpi_count(label: &str, reading: &Reading) -> El {
91    let value = match &reading.state {
92        Availability::Available(Value::Array(items)) => items.len().to_string(),

120/// What a reading shows: its value, or which way it is missing. A number is shown as the
121/// source gave it, rounded to two places when it is a fraction.
122pub fn shown(r: &Reading) -> String {
123    match &r.state {
124        Availability::Available(Value::String(s)) => s.clone(),
125        Availability::Available(Value::Number(n)) => number(n),
126        Availability::Available(Value::Bool(b)) => if *b { "yes" } else { "no" }.into(),
127        Availability::Available(v) => v.to_string(),
128        Availability::Absent => "none".into(),
129        Availability::Failed(_) => "not answered".into(),
130    }
131}
132
133fn number(n: &serde_json::Number) -> String {
134    // an integer is written as the source wrote it; only a fraction is rounded
135    match n.as_f64() {
136        Some(f) if n.is_f64() => {
137            let s = format!("{f:.2}");
138            s.trim_end_matches('0').trim_end_matches('.').to_string()
139        }
140        _ => n.to_string(),
141    }
142}

200/// The attribute a source line carries while its reading is not of the repository as it is
201/// now. The page's script reads the screen again shortly when it finds one.
202pub const CATCHING_UP: &str = "data-catching-up";

204/// The sources of a panel: for each, the capability and how long it took, then every other
205/// way to ask it the same question — the API route a browser opens, the command a person
206/// types, the MCP tool an agent calls — and its technical reference in the Cockpit. Each is
207/// the capability's own declaration, as `capabilities.describe` answered it.
208pub fn sources(answers: &[&Answer]) -> El {
209    let mut box_ = el("div").class("mj-ui-sources");
210    for a in answers {
211        let mut line = el("p")
212            .class("mj-ui-source")
213            .child(
214                el("span")
215                    .class("mj-ui-source-name")
216                    .text(a.source.capability),
217            )
218            .child(
219                el("span")
220                    .class("mj-ui-dim")
221                    .text(format!("{} ms", a.took.as_millis())),
222            );
223        // a reading that is not simply current says so, and one a newer reading is about to
224        // replace marks its line, which is what the page's script reads the screen again for
225        if let Some(note) = a.standing.note() {
226            line = line.child(el("span").class("mj-ui-dim").text(note));
227        }
228        if a.standing.catching_up() {
229            line = line.attr(CATCHING_UP, "");
230        }
231        if let Some(path) = &a.exposure.get {
232            line = line.child(chip_link("API", path, "open the same answer as JSON"));
233        }
234        if let Some(cli) = &a.exposure.cli {
235            line = line.child(chip("CLI", cli));
236        }
237        if let Some(tool) = &a.exposure.mcp {
238            line = line.child(chip("MCP", tool));
239        }
240        line = line.child(
241            el("a")
242                .class("mj-ui-technical")
243                .attr("href", technical(a.source.capability))
244                .attr("title", "the capability behind this value, in the Cockpit")
245                .text("technical view"),
246        );
247        box_ = box_.child(line);
248    }
249    box_
250}

297/// [`object`] when the index holds `uri`, the label as text when it does not, so no link
298/// leads to a page that answers 404, and `none` when there is no label at all. Whether the
299/// index holds it is asked of `objects.get`, the capability every surface resolves a URI with
300/// (ADR 0012), never read from the index beside it. Asked as a reading, like every source of
301/// a screen (`project.no-screen-waits-for-a-source`): an object whose first reading has not
302/// arrived is its label as text until it has.
303pub fn object_if(ctx: &Context, uri: &str, label: &str) -> El {
304    if label.is_empty() {
305        return none();
306    }
307    match ctx
308        .read("objects.get", serde_json::json!({ "uri": uri }))
309        .value
310    {
311        Ok(_) => object(uri, label),
312        Err(_) => el("span").class("mj-ui-mono").text(label),
313    }
314}

320        (_, true) => none(),

355/// What a list cut short leaves out: "the first `shown` of `total`", and where the whole list
356/// is when there is a place that shows it. Nothing when nothing was left out.
357pub fn more(shown: usize, total: usize, whole: Option<&str>) -> Option<El> {
358    (total > shown).then(|| {
359        let line = el("p").class("mj-ui-line mj-ui-dim").text(format!(
360            "The first {shown} of {total}; {} more not shown here. ",
361            total - shown
362        ));
363        match whole {
364            Some(href) => line.child(link(href, "Open the whole list")),
365            None => line,
366        }
367    })
368}

370/// An identifier as a fragment a page can carry: letters, digits, `-`, `_` and `.` kept,
371/// anything else a `-`, so an id a source writes never breaks out of the attribute.
372pub fn anchor(id: &str) -> String {
373    id.chars()
374        .map(|c| {
375            if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.') {
376                c
377            } else {
378                '-'
379            }
380        })
381        .collect()
382}

479/// A node of a relationship graph: an entity the source named, its label, the tone its
480/// state is drawn in, and where clicking it leads.
481pub struct GraphNode {
482    /// The entity's id, as the source gave it.
483    pub id: String,
484    /// The label drawn on the node.
485    pub label: String,
486    /// The tone of the design system it is drawn in.
487    pub tone: &'static str,
488    /// Where it leads.
489    pub href: String,
490    /// `leaf` for a node drawn small and labelled only when selected; anything else is a
491    /// node of the graph's main kind.
492    pub kind: &'static str,
493}
494
495/// How a graph is laid out: `directed` for a dependency order, `force` for a topology.
496#[derive(Debug, Clone, Copy, PartialEq, Eq)]
497pub enum Layout {
498    /// Breadth-first along the edges' direction.
499    Directed,
500    /// Force-directed, for a topology with no order.
501    Force,
502}
503
504/// A relationship graph: the nodes and the typed edges a source answered, drawn by the
505/// vendored Cytoscape once the screen has painted. Without JavaScript the `fallback` — a
506/// table of the same edges — is what the reader gets, and with it the graph is a second view
507/// of the same data, never a different one. An edge whose end is not a node is dropped
508/// rather than drawn to nothing.
509///
510/// Only nodes that take part in an edge are drawn: a node with no relation says nothing in a
511/// graph that its row in the fallback does not say better.
512pub fn graph(
513    label: &str,
514    layout: Layout,
515    nodes: &[GraphNode],
516    edges: &[(String, String)],
517    fallback: El,
518) -> El {
519    let ids: std::collections::BTreeSet<&str> = nodes.iter().map(|n| n.id.as_str()).collect();
520    let related: std::collections::BTreeSet<&str> = edges
521        .iter()
522        .filter(|(a, b)| ids.contains(a.as_str()) && ids.contains(b.as_str()))
523        .flat_map(|(a, b)| [a.as_str(), b.as_str()])
524        .collect();
525    let data = serde_json::json!({
526        "layout": match layout { Layout::Directed => "directed", Layout::Force => "force" },
527        "nodes": nodes.iter().filter(|n| related.contains(n.id.as_str())).map(|n| serde_json::json!({
528            "id": n.id, "label": n.label, "tone": n.tone, "href": n.href, "kind": n.kind,
529        })).collect::<Vec<_>>(),
530        "edges": edges.iter()
531            .filter(|(a, b)| ids.contains(a.as_str()) && ids.contains(b.as_str()))
532            .map(|(a, b)| serde_json::json!({ "source": a, "target": b }))
533            .collect::<Vec<_>>(),
534    });
535    el("div")
536        .class("mj-ui-graph")
537        .attr("data-graph", data.to_string())
538        .child(
539            el("div")
540                .class("mj-ui-graph-canvas")
541                .attr("role", "img")
542                .attr("aria-label", label)
543                .flag("hidden"),
544        )
545        .child(el("div").class("mj-ui-graph-fallback").child(fallback))
546}

563/// A donut of named counts, each in a tone of the design system, with the counts beside it
564/// as a legend: a reader without the picture reads the same numbers. Segments that are
565/// zero are listed and not drawn; a donut of nothing says so.
566pub fn donut(center: &str, segments: &[(String, u64)]) -> El {
567    let total: u64 = segments.iter().map(|(_, n)| n).sum();
568    if total == 0 {
569        return nothing("Nothing has been counted yet.");
570    }
571    // a circle of circumference 100, so a segment's dash is its share in percent
572    let radius = 15.915_494_309;
573    let mut offset = 25.0_f64;
574    let mut svg = el("svg")
575        .class("mj-ui-donut-svg")
576        .attr("viewBox", "0 0 42 42")
577        .attr("aria-hidden", "true")
578        .child(
579            el("circle")
580                .class("mj-ui-donut-track")
581                .attr("cx", "21")
582                .attr("cy", "21")
583                .attr("r", format!("{radius}")),
584        );
585    for (i, (_, n)) in segments.iter().enumerate() {
586        if *n == 0 {
587            continue;
588        }
589        let share = *n as f64 * 100.0 / total as f64;
590        svg = svg.child(
591            el("circle")
592                .class(format!(
593                    "mj-ui-donut-segment mj-tone--{}",
594                    TONES[i % TONES.len()]
595                ))
596                .attr("cx", "21")
597                .attr("cy", "21")
598                .attr("r", format!("{radius}"))
599                .attr(
600                    "stroke-dasharray",
601                    format!("{share:.3} {:.3}", 100.0 - share),
602                )
603                .attr("stroke-dashoffset", format!("{offset:.3}")),
604        );
605        offset -= share;
606    }
607    let legend = el("ul")
608        .class("mj-ui-legend")
609        .children(segments.iter().enumerate().map(|(i, (label, n))| {
610            el("li")
611                .class(format!(
612                    "mj-ui-legend-item mj-tone--{}",
613                    TONES[i % TONES.len()]
614                ))
615                .child(
616                    el("span")
617                        .class("mj-ui-legend-dot")
618                        .attr("aria-hidden", "true"),
619                )
620                .child(el("span").class("mj-ui-legend-name").text(label))
621                .child(el("span").class("mj-ui-legend-value").text(n.to_string()))
622        }));
623    el("div")
624        .class("mj-ui-donut")
625        .child(
626            el("div").class("mj-ui-donut-figure").child(svg).child(
627                el("span")
628                    .class("mj-ui-donut-center")
629                    .child(el("strong").text(total.to_string()))
630                    .child(el("span").text(center)),
631            ),
632        )
633        .child(legend)
634}

636/// The counts of an object at `pointer`, as donut segments in the order the source gave.
637pub fn counts(answer: &Answer, pointer: &str) -> Vec<(String, u64)> {
638    answer
639        .ok()
640        .and_then(|v| v.pointer(pointer))
641        .and_then(Value::as_object)
642        .map(|m| {
643            m.iter()
644                .filter_map(|(k, v)| v.as_u64().map(|n| (k.to_lowercase(), n)))
645                .collect()
646        })
647        // ui-integrity: an absent object is no segments, never a zero; the donut then says nothing was counted
648        .unwrap_or_default()
649}
