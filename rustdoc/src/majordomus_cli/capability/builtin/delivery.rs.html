1//! The `delivery` module: whether each product feature exists, computed on every read.
2//!
3//! A feature exists only when it is on the trunk, deployed, publicly verified, has its
4//! required tests current, has that evidence published and is linked from the interface
5//! (ADR 0071). Every one of those is a verdict derived here from git, from the public site's
6//! own identity and — from phase 2 — from recorded evidence; none is a field anybody wrote.
7//! Unknown is never pass.
8//!
9//! Both capabilities are reads, and both may reach the network: the public identity is read
10//! from the site, and `scripts/pages` asks the site again and asks GitHub. That is why the
11//! benchmark waives them as an external dependency, and why nothing here is cached: a
12//! publication that changed a second ago is the answer a person asking wants.
13

29/// The input of `delivery.feature`: which feature.
30#[derive(Debug, Serialize, Deserialize, JsonSchema)]
31#[serde(deny_unknown_fields)]
32pub struct DeliveryFeatureInput {
33    /// The feature's id, as `product.features` gives it.
34    pub id: String,
35}

37impl BenchmarkCases for DeliveryFeatureInput {
38    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
39        let id = ctx
40            .index
41            .objects
42            .iter()
43            .find(|o| o.kind == crate::product::FEATURE)
44            .map(|o| o.identity.clone())
45            .unwrap_or_else(|| "absent".into());
46        vec![NamedCase::new("first-feature", DeliveryFeatureInput { id })]
47    }
48}

50fn report_for(ctx: &Context, only: Option<&str>) -> DeliveryReport {
51    let root = std::path::Path::new(&ctx.index.repository.root);
52    let subjects: Vec<Subject> = ctx
53        .product
54        .all()
55        .iter()
56        .filter(|r| only.is_none_or(|id| r.feature.id == id))
57        .map(|r| Subject {
58            id: r.feature.id.clone(),
59            title: r.feature.title.clone(),
60            status: r.feature.status.clone(),
61            paths: revision::implementation_paths(root, r, &ctx.index),
62        })
63        .collect();
64    delivery::assess(root, subjects, Probe::for_repository(root).observe())
65}
66
67fn delivery_report(ctx: &Context, _: Empty) -> Result<DeliveryReport, CapabilityError> {
68    Ok(report_for(ctx, None))
69}
70
71fn delivery_feature(
72    ctx: &Context,
73    input: DeliveryFeatureInput,
74) -> Result<FeatureDelivery, CapabilityError> {
75    if ctx.product.feature(&input.id).is_none() {
76        return Err(CapabilityError::NotFound(format!(
77            "no feature '{}'; `product.features` names every one this repository holds",
78            input.id
79        )));
80    }
81    report_for(ctx, Some(&input.id))
82        .features
83        .into_iter()
84        .next()
85        .ok_or_else(|| CapabilityError::NotFound(format!("no feature '{}'", input.id)))
86}

143    fn git(root: &std::path::Path, args: &[&str]) {
144        let ok = std::process::Command::new("git")
145            .arg("-C")
146            .arg(root)
147            .args(args)
148            .env_remove("GIT_DIR")
149            .env_remove("GIT_WORK_TREE")
150            .output()
151            .unwrap()
152            .status
153            .success();
154        assert!(ok, "git {args:?}");
155    }

165        std::fs::create_dir_all(root.join(".ai/repo/features")).unwrap();

206                    c.exposure.mcp.as_ref().and_then(|m| m.tool.clone()),
207                    c.exposure.http.as_ref().map(|h| h.path.clone()),
208                    c.exposure
209                        .cli
210                        .as_ref()
211                        .map(|c| c.path.clone())

236    #[test]
237    fn a_merged_feature_with_no_site_is_on_master_and_does_not_exist() {
238        let repo = repository();
239        let ctx = repo.context().expect("a context");
240        let report = delivery_report(&ctx, Empty {}).expect("a report");
241        assert_eq!(report.tallies.features, 1);
242        let f = &report.features[0];
243        assert_eq!(f.id, "shipped");
244        assert_eq!(
245            f.implementation.paths,
246            [
247                "apps/majordomus-cli/src/capability/builtin/delivery.rs",
248                "lib/start.sh"
249            ]
250        );
251        assert_eq!(f.dimensions[0].verdict, Verdict::Pass, "{f:?}");
252        // a repository with no site and no publication model is not deployed: it is unknown
253        assert_eq!(f.dimensions[1].verdict, Verdict::Unknown);
254        assert!(!f.exists);
255        assert!(matches!(
256            f.delivery,
257            DeliveryState::NotDelivered {
258                stage: DevelopmentStage::OnMaster,
259                ..
260            }
261        ));
262        let one = delivery_feature(
263            &ctx,
264            DeliveryFeatureInput {
265                id: "shipped".into(),
266            },
267        )
268        .expect("one feature");
269        assert_eq!(&one, f);
270    }
