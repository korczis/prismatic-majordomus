1//! `majordomus mcp`: serve one MCP client over stdio. By default the process joins the
2//! repository's one shared server: it starts it when nobody has (loopback HTTP with
3//! Swagger UI, OpenAPI and MCP over HTTP beside its own stdio session) and bridges its
4//! stdio to it when someone has. `--standalone` serves the client alone, as the first
5//! version of this executable did; `--inspect` prints what would be served and stops.

28/// Run `majordomus mcp`.
29pub fn run(args: McpArgs) -> Result<u8> {
30    if args.inspect {
31        let app = App::load(&args.repo)?;
32        return inspect(&Surface::new(app.context.clone()), args.format);
33    }
34    match args.transport {
35        Transport::Stdio => {}
36    }
37    let stdin = std::io::stdin();
38    let stdout = std::io::stdout();
39    if args.standalone {
40        let mut server = standalone(&args.repo)?;
41        tracing::info!("standalone: no shared server, no HTTP, no peers");
42        stdio::serve(stdin.lock(), stdout.lock(), |m| server.handle(m))?;
43        return Ok(0);
44    }
45    let repo = Repository::discover(&start_dir(&args.repo)?)?;
46    let mut session = Session::open(args, repo)?;
47    stdio::serve(stdin.lock(), stdout.lock(), |m| session.handle(m))?;
48    session.finish();
49    Ok(0)
50}

52/// A server answering one stdio session alone: no shared server, no HTTP, no peers.
53fn standalone(repo: &RepoArgs) -> Result<Server> {
54    let app = App::load(repo)?;
55    // a stdio session outlives commits like any other: it follows the repository too
56    let live = Arc::new(Live::watching(repo.clone(), app.context.clone()));
57    let peer = live.current().peers.attach(PeerTransport::Stdio);
58    Ok(Server::new(
59        Surface::new(Arc::clone(&live)).for_peer(peer),
60        crate::VERSION,
61    ))
62}
63
64fn start_dir(args: &RepoArgs) -> Result<std::path::PathBuf> {
65    match &args.repo {
66        Some(p) => Ok(p.clone()),
67        None => std::env::current_dir().map_err(|e| Error::io(".", e)),
68    }
69}

71/// The stdio session of this process, answered locally by the shared server this
72/// process runs, or forwarded to the one another process runs.
73struct Session {
74    args: McpArgs,
75    repo: Repository,
76    backend: Backend,
77}
78
79enum Backend {
80    Local(Box<Local>),
81    Remote {
82        bridge: Arc<Mutex<Bridge>>,
83        heartbeat: Heartbeat,
84    },
85    /// The shared server cannot be used (the lease cannot be written or replaced, the
86    /// server cannot start): this client is served alone, as `--standalone` would, and
87    /// the log said why.
88    Alone(Box<Server>),
89}
90
91/// This process is the shared server; its own stdio is answered here.
92struct Local {
93    server: Server,
94    peer: PeerId,
95    shared: SharedServer,
96}
97
98struct Heartbeat {
99    stop: Arc<AtomicBool>,
100    thread: Option<JoinHandle<()>>,
101}
102
103impl Heartbeat {
104    fn start(bridge: Arc<Mutex<Bridge>>) -> Self {
105        let stop = Arc::new(AtomicBool::new(false));
106        let flag = Arc::clone(&stop);
107        let thread = std::thread::Builder::new()
108            .name("mcp-heartbeat".into())
109            .spawn(move || {
110                let tick = Duration::from_millis(500);
111                let mut waited = Duration::ZERO;
112                while !flag.load(Ordering::SeqCst) {
113                    std::thread::sleep(tick);
114                    waited += tick;
115                    if waited >= HEARTBEAT {
116                        waited = Duration::ZERO;
117                        if let Err(e) = lock(&bridge).heartbeat() {
118                            tracing::debug!("heartbeat: {e}");
119                        }
120                    }
121                }
122            })
123            .ok();
124        Heartbeat { stop, thread }
125    }
126
127    fn stop(mut self) {
128        self.stop.store(true, Ordering::SeqCst);
129        if let Some(t) = self.thread.take() {
130            let _ = t.join();
131        }
132    }
133}

135impl Session {
136    fn open(args: McpArgs, repo: Repository) -> Result<Self> {
137        let backend = match lease::elect(&repo) {
138            Ok(Role::Server(lease)) => match Self::serve(&args, lease, None) {
139                Ok(backend) => backend,
140                Err(e) => Self::alone(&args, None, &format!("starting it failed: {e}"))?,
141            },
142            Ok(Role::Peer { url }) => Self::attach(url),
143            Err(e) => Self::alone(&args, None, &e.to_string())?,
144        };
145        Ok(Session {
146            args,
147            repo,
148            backend,
149        })
150    }
151
152    /// The shared server cannot be used: serve this client alone and say why. The layer
153    /// itself still has to load; when it does not, that error is the answer, not this.
154    fn alone(args: &McpArgs, resume: Option<ClientInfo>, why: &str) -> Result<Backend> {
155        let mut server = standalone(&args.repo)?;
156        tracing::warn!(
157            "cannot use the shared server: {why}; serving this client alone (no HTTP, no Swagger UI, no peers), as --standalone would"
158        );
159        if let Some(client) = resume {
160            server.resume(client);
161        }
162        Ok(Backend::Alone(Box::new(server)))
163    }
164
165    /// Become the shared server: load the layer, bind, and answer this stdio locally.
166    fn serve(
167        args: &McpArgs,
168        lease: Lease,
169        resume: Option<crate::peers::ClientInfo>,
170    ) -> Result<Backend> {
171        // the lease is kept young while the layer loads, so that a peer waiting on it
172        // never mistakes a slow start for an abandoned one
173        lease.keep_alive();
174        let app = App::load(&args.repo)?;
175        let live = Arc::new(Live::watching(args.repo.clone(), app.context.clone()));
176        let ctx = live.current();
177        let shared = SharedServer::start(
178            Arc::clone(&live),
179            crate::VERSION,
180            &args.http_host,
181            args.http_port,
182            true,
183            None,
184            lease,
185            Some(app.share.dir()),
186        )?;
187        let peer = ctx.peers.attach(PeerTransport::Stdio);
188        let mut server = Server::new(Surface::new(live).for_peer(peer.clone()), crate::VERSION)
189            .with_endpoint(Some(shared.url()));
190        if let Some(client) = resume {
191            server.resume(client);
192        }
193        Ok(Backend::Local(Box::new(Local {
194            server,
195            peer,
196            shared,
197        })))
198    }
199
200    /// Attach to the shared server another process runs.
201    fn attach(url: String) -> Backend {
202        tracing::info!(
203            url = %url,
204            "a shared server for this repository is already running at {url} (swagger ui {url}{swagger}); bridging this stdio session to it",
205            swagger = crate::http::swagger::SWAGGER_PATH
206        );
207        let bridge = Arc::new(Mutex::new(Bridge::new(url)));
208        let heartbeat = Heartbeat::start(Arc::clone(&bridge));
209        Backend::Remote { bridge, heartbeat }
210    }
211
212    fn handle(&mut self, message: Value) -> Option<Reply> {
213        match &mut self.backend {
214            Backend::Local(local) => local.server.handle(message),
215            Backend::Alone(server) => server.handle(message),
216            Backend::Remote { bridge, .. } => {
217                let answer = lock(bridge).handle(&message);
218                match answer {
219                    Ok(v) => v.map(Reply::Value),
220                    Err(e) => self.failover(message, e),
221                }
222            }
223        }
224    }
225
226    /// The server this session was bridged to is gone: elect again, and either become
227    /// the server (carrying the client's session over) or attach to whoever did.
228    fn failover(&mut self, message: Value, cause: BridgeError) -> Option<Reply> {
229        tracing::warn!("{cause}; electing again");
230        let (client, announcements) = match &self.backend {
231            Backend::Remote { bridge, .. } => {
232                let b = lock(bridge);
233                (b.client().cloned(), b.announcements().cloned().collect())
234            }
235            Backend::Local(_) | Backend::Alone(_) => (None, Vec::new()),
236        };
237        match lease::elect(&self.repo) {
238            Ok(Role::Server(lease)) => {
239                match Self::serve(&self.args, lease, client.clone()) {
240                    Ok(backend) => {
241                        self.replace(backend);
242                        tracing::info!(
243                            "took over as the shared server; this session continues locally"
244                        );
245                        // what the client said it was working on, said again on the board it
246                        // now serves itself: an announcement outlives the server it was made
247                        // to. Every claim it holds, not the newest — a client that carried
248                        // one of three onto its own board would be understating itself to
249                        // every peer that arrives after the takeover.
250                        if let Backend::Local(local) = &self.backend {
251                            for a in &announcements {
252                                let intent = a["intent"].as_str().unwrap_or_default().trim();
253                                if intent.is_empty() {
254                                    continue;
255                                }
256                                let scope: Vec<String> = a["scope"]
257                                    .as_array()
258                                    .map(|s| {
259                                        s.iter()
260                                            .filter_map(|v| v.as_str())
261                                            .map(|v| v.trim().to_string())
262                                            .filter(|v| !v.is_empty())
263                                            .collect()
264                                    })
265                                    .unwrap_or_default();
266                                let claim =
267                                    a["claim"].as_str().map(str::trim).filter(|c| !c.is_empty());
268                                local.server.surface().context().peers.announce_claim(
269                                    &local.peer,
270                                    claim,
271                                    intent,
272                                    scope,
273                                );
274                            }
275                            if !announcements.is_empty() {
276                                tracing::info!(
277                                    claims = announcements.len(),
278                                    "the client's announcement was carried onto this server's board"
279                                );
280                            }
281                        }
282                        self.handle(message)
283                    }
284                    Err(e) => {
285                        tracing::error!("cannot take over as the shared server: {e}");
286                        self.settle_alone(
287                            message,
288                            client,
289                            &format!("{cause}; and taking over failed: {e}"),
290                        )
291                    }
292                }
293            }
294            Ok(Role::Peer { url }) => {
295                let Backend::Remote { bridge, .. } = &self.backend else {
296                    return unavailable(&message, &cause.to_string());
297                };
298                let mut b = lock(bridge);
299                b.move_to(url.clone());
300                let retry = b.reinitialize().and_then(|()| b.handle(&message));
301                match retry {
302                    Ok(v) => {
303                        tracing::info!(url = %url, "re-attached to the shared server");
304                        v.map(Reply::Value)
305                    }
306                    Err(e) => unavailable(&message, &e.to_string()),
307                }
308            }
309            Err(e) => self.settle_alone(
310                message,
311                client,
312                &format!("{cause}; electing again failed: {e}"),
313            ),
314        }
315    }
316
317    /// Replace the backend, stopping the heartbeat of a bridge that is left behind.
318    fn replace(&mut self, backend: Backend) {
319        let old = std::mem::replace(&mut self.backend, backend);
320        if let Backend::Remote { heartbeat, .. } = old {
321            heartbeat.stop();
322        }
323    }
324
325    /// Neither serving nor attaching worked: serve this client alone when the layer
326    /// loads, and answer the request with an error naming every failure when it does not.
327    fn settle_alone(
328        &mut self,
329        message: Value,
330        client: Option<ClientInfo>,
331        reason: &str,
332    ) -> Option<Reply> {
333        match Self::alone(&self.args, client, reason) {
334            Ok(backend) => {
335                self.replace(backend);
336                self.handle(message)
337            }
338            Err(e) => {
339                tracing::error!("cannot serve this client alone either: {e}");
340                unavailable(&message, reason)
341            }
342        }
343    }
344
345    /// The client has gone. A server lingers while other peers are attached; a bridge
346    /// tells the server its session is over.
347    fn finish(self) {
348        match self.backend {
349            Backend::Local(local) => {
350                let Local {
351                    server,
352                    peer,
353                    shared,
354                } = *local;
355                server.surface().context().peers.detach(&peer);
356                drop(server);
357                shared.wait_until_peers_leave();
358                shared.stop();
359            }
360            Backend::Remote { bridge, heartbeat } => {
361                heartbeat.stop();
362                lock(&bridge).close();
363                tracing::info!("bridge closed");
364            }
365            Backend::Alone(_) => {}
366        }
367    }
368}

370/// The JSON-RPC answer for a request that no server could take; nothing for a notification.
371fn unavailable(message: &Value, reason: &str) -> Option<Reply> {
372    let id = message.get("id").cloned().filter(|i| !i.is_null())?;
373    Some(Reply::Value(json!({
374        "jsonrpc": "2.0",
375        "id": id,
376        "error": { "code": -32603, "message": format!("shared server unavailable: {reason}") }
377    })))
378}

380fn lock<T>(m: &Mutex<T>) -> std::sync::MutexGuard<'_, T> {
381    m.lock().unwrap_or_else(|e| e.into_inner())
382}

384fn inspect(surface: &Surface, format: OutputFormat) -> Result<u8> {
385    let stdout = std::io::stdout();
386    let mut out = stdout.lock();
387    let index = surface.index();
388    match format {
389        OutputFormat::Json => {
390            let v = json!({
391                "repository": surface.repository_info().map_err(|e| Error::Protocol { reason: e.to_string() })?,
392                "resources": &*surface.resources(),
393                "tools": &*surface.tools(),
394            });
395            writeln!(
396                out,
397                "{}",
398                serde_json::to_string_pretty(&v).map_err(|e| Error::Protocol {
399                    reason: e.to_string()
400                })?
401            )
402            .map_err(Error::Transport)?;
403        }
404        OutputFormat::Text => {
405            let w = |out: &mut std::io::StdoutLock<'_>, line: String| {
406                writeln!(out, "{line}").map_err(Error::Transport)
407            };
408            w(&mut out, format!("repository  {}", index.repository.root))?;
409            w(
410                &mut out,
411                format!("discovery   {}", index.repository.discovery),
412            )?;
413            w(
414                &mut out,
415                format!(
416                    "state       {}",
417                    match index.state {
418                        crate::index::State::Ok => "ok",
419                        crate::index::State::Degraded => "degraded",
420                    }
421                ),
422            )?;
423            for (kind, n) in index.kinds() {
424                w(&mut out, format!("kind        {kind:<12} {n}"))?;
425            }
426            let summary = surface.registry().summary();
427            w(
428                &mut out,
429                format!(
430                    "capabilities {} ({} builtin, {} declarative)",
431                    summary.total, summary.builtin, summary.declarative
432                ),
433            )?;
434            for r in surface.resources().iter() {
435                w(&mut out, format!("resource    {}", r.uri))?;
436            }
437            for t in surface.tools().iter() {
438                w(&mut out, format!("tool        {}", t.name))?;
439            }
440            for d in &index.diagnostics {
441                let level = match d.severity {
442                    Severity::Error => "FAIL",
443                    Severity::Warning => "WARN",
444                    Severity::Info => "INFO",
445                };
446                w(
447                    &mut out,
448                    format!(
449                        "{level:<4} {:<22} {} — {}",
450                        d.code,
451                        d.path.as_deref().unwrap_or("-"),
452                        d.message
453                    ),
454                )?;
455            }
456        }
457    }
458    Ok(if index.errors() > 0 { 10 } else { 0 })
