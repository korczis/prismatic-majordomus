1//! The shell every Cockpit page is rendered into, and the small vocabulary of components
2//! the pages are built from.
3//!
4//! The pages carry no raw utility classes. Every class here is a semantic one the
5//! Cockpit's stylesheet defines (`mj-card`, `mj-table`, `mj-badge--fail`), so the design
6//! lives in one file a designer can read and the Rust says what a thing *is* rather than
7//! how it looks. That is also what makes the stylesheet deterministic: it is a function of
8//! its own source and nothing else, and the drift check over it is exact.
9//!
10//! Every `x-` attribute here names a *top-level* property or method of the one Alpine
11//! component and never an expression — not even a dotted path, which the CSP build reads
12//! as an expression and does not evaluate. A palette bound to `palette.open` looks right,
13//! renders right, and leaves a full-page modal backdrop over every click. That is what the content-security policy requires: the Cockpit
14//! ships Alpine's CSP build, which evaluates no strings, so the policy needs neither
15//! `unsafe-eval` nor `unsafe-inline`. A handler that would take an argument gets its own
16//! method in `cockpit.js` instead.
17

28const NO_ASSETS: &str = "This distribution has no share/cockpit/ directory, so the Cockpit is serving unstyled markup. Everything works; nothing is styled. Run `just cockpit-assets` to build it.";

30/// What a page needs from the process in order to render its shell.
31pub struct Shell<'a> {
32    /// The page title, before the suffix.
33    pub title: &'a str,
34    /// One line under the heading: what this page shows.
35    pub subtitle: Option<String>,
36    /// The area of the navigation this page belongs to.
37    pub area: Area,
38    /// The trail from the Cockpit root to this page, label and href; the last has no href.
39    pub breadcrumbs: Vec<(String, Option<String>)>,
40    /// The navigation, derived from the registry.
41    pub navigation: &'a Navigation,
42    /// The stylesheet URL, with its digest.
43    pub stylesheet: String,
44    /// The Cockpit's own script modules, in load order, with their digests.
45    pub scripts: Vec<String>,
46    /// Whether the distribution has an asset directory at all.
47    pub assets_present: bool,
48    /// The executable's version, shown in the footer.
49    pub version: &'a str,
50}

52/// Render one page into the shell.
53pub fn page(shell: &Shell<'_>, main: El) -> String {
54    let head = vec![
55        el("link")
56            .attr("rel", "stylesheet")
57            .attr("href", shell.stylesheet.clone()),
58        el("link")
59            .attr("rel", "icon")
60            .attr("href", "/cockpit/assets/favicon.svg"),
61        // the theme is applied before first paint, so a dark-mode reader never sees a
62        // white flash; this is the one inline script the Cockpit ships, and the
63        // content-security policy names its digest rather than allowing inline scripts
64        el("script").raw(theme_bootstrap()),
65    ]
66    .into_iter()
67    .chain(shell.scripts.iter().map(|src| {
68        el("script")
69            .attr("type", "module")
70            .attr("src", src.clone())
71            .flag("defer")
72    }))
73    .collect();
74
75    // the theme contract and the design fingerprint, from the declaration compiled into
76    // this executable: the script reads the key and the class here rather than naming
77    // them, and compares the fingerprint with the one the stylesheet carries
78    let design = DesignSystem::compiled().ok();
79    let body = el("body")
80        .class("mj-body")
81        .attr("x-data", "cockpit")
82        .attr_if(
83            "data-theme-key",
84            design.map(|d| d.theme.storage_key.clone()),
85        )
86        .attr_if("data-theme-class", design.map(|d| d.theme.class.clone()))
87        .attr_if("data-design", design.map(|d| d.short_fingerprint()))
88        .child(skip_link())
89        .child(header(shell))
90        .child(
91            el("div").class("mj-layout").child(sidebar(shell)).child(
92                el("main")
93                    .attr("id", "main")
94                    .class("mj-main")
95                    .when(!shell.assets_present, |m| m.child(alert("warn", NO_ASSETS)))
96                    .child(breadcrumbs(&shell.breadcrumbs))
97                    .child(
98                        el("header")
99                            .class("mj-page-head")
100                            .child(el("h1").class("mj-page-title").text(shell.title))
101                            .node(match &shell.subtitle {
102                                Some(s) => Node::Element(el("p").class("mj-page-subtitle").text(s)),
103                                None => empty(),
104                            }),
105                    )
106                    .child(main),
107            ),
108        )
109        .child(palette())
110        .child(footer(shell));
111
112    super::html::document(&format!("{} · Majordomus Cockpit", shell.title), head, body)
113}

115/// Read the stored theme before the page paints: the one statement the design declaration
116/// generates for every surface, so the site and the Cockpit honour one storage key and one
117/// class. One statement, so that its digest is stable and the content-security policy can
118/// name it. An executable whose compiled declaration is invalid ships no statement and
119/// follows the system through the stylesheet alone; the design capability says why.
120pub fn theme_bootstrap() -> &'static str {
121    static BOOTSTRAP: LazyLock<String> = LazyLock::new(|| {
122        DesignSystem::compiled()
123            .map(|d| d.theme_bootstrap())
124            .unwrap_or_default()
125    });
126    BOOTSTRAP.as_str()
127}

185fn sidebar(shell: &Shell<'_>) -> El {
186    let mut nav = el("nav")
187        .class("mj-sidebar")
188        .attr("aria-label", "Cockpit sections");
189    for section in shell.navigation.sections() {
190        let mut list = el("ul").class("mj-nav-list");
191        // A grouped section shows its headings as the group changes. The items arrive in
192        // the canonical order, which puts a group's members together and the ungrouped
193        // ones last, so one pass is enough and nothing here decides the sequence.
194        let mut group: Option<&str> = None;
195        for item in &section.items {
196            if item.group.as_deref() != group {
197                group = item.group.as_deref();
198                if let Some(heading) = group {
199                    list = list.child(
200                        el("li")
201                            .class("mj-nav-group")
202                            .attr("role", "presentation")
203                            .text(heading),
204                    );
205                }
206            }
207            let current = item.area == shell.area && item.current;
208            list = list.child(
209                el("li").child(
210                    el("a")
211                        .class(if current {
212                            "mj-nav-link mj-nav-link--current"
213                        } else {
214                            "mj-nav-link"
215                        })
216                        .attr("href", item.href.clone())
217                        .attr_if("aria-current", current.then_some("page"))
218                        .child(el("span").class("mj-nav-label").text(&item.label))
219                        .node(match item.count {
220                            Some(Count::Known(n)) => {
221                                Node::Element(el("span").class("mj-nav-count").text(n.to_string()))
222                            }
223                            Some(Count::Unknown) => Node::Element(
224                                el("span")
225                                    .class("mj-nav-count")
226                                    .attr(
227                                        "title",
228                                        "the capability behind this count did not answer",
229                                    )
230                                    .text("unknown"),
231                            ),
232                            None => empty(),
233                        }),
234                ),
235            );
236        }
237        // a catalogue long enough to bury the sections under it folds away, and unfolds
238        // itself when what the reader is looking at is inside it
239        let holds_current = section
240            .items
241            .iter()
242            .any(|i| i.area == shell.area && i.current);
243        nav = nav.child(if section.items.len() > FOLD_ABOVE {
244            el("details")
245                .class("mj-nav-section mj-nav-section--foldable")
246                .when(holds_current, |d| d.flag("open"))
247                .child(
248                    el("summary")
249                        .class("mj-nav-heading mj-nav-summary")
250                        .child(el("span").text(&section.title))
251                        .child(
252                            el("span")
253                                .class("mj-nav-count")
254                                .text(section.items.len().to_string()),
255                        ),
256                )
257                .child(list)
258        } else {
259            el("div")
260                .class("mj-nav-section")
261                .child(el("h2").class("mj-nav-heading").text(&section.title))
262                .child(list)
263        });
264    }
265    nav
266}

284fn breadcrumbs(trail: &[(String, Option<String>)]) -> El {
285    if trail.is_empty() {
286        return el("div").class("mj-hidden");
287    }
288    let mut list = el("ol").class("mj-crumbs");
289    for (label, href) in trail {
290        list = list.child(el("li").child(match href {
291            Some(h) => el("a").attr("href", h.clone()).text(label),
292            None => el("span").attr("aria-current", "page").text(label),
293        }));
294    }
295    el("nav").attr("aria-label", "Breadcrumb").child(list)
296}

429/// A badge for a status word that arrived as *data* — an object's own `status` field, a
430/// graph node's — rather than one this code chose. Filed under a status by the design
431/// declaration, it is a badge and carries that status's colour; filed under nothing, it is
432/// a plain tag, so that an arbitrary word never wears a meaning nobody gave it and the
433/// browser probe's vocabulary check stays exact. The words this code chooses itself go
434/// through [`badge`], where an unfiled word is a defect the probe reports.
435pub fn word_badge(word: &str) -> El {
436    let filed = DesignSystem::compiled()
437        .ok()
438        .is_some_and(|d| d.role_of_state(&css_word(word)).is_some());
439    if filed {
440        badge(word, word)
441    } else {
442        tag(word)
443    }
444}

518/// When the state behind what is shown was read, and — past its window — that it is stale.
519///
520/// The freshness contract rendered: the observation's own time, its age at `now`, and a
521/// `stale` badge once `now` is past `stale_after`. The attributes carry the machine values,
522/// so a probe reads the time the answer carried rather than parsing prose.
523pub fn as_of(o: &crate::index::AnswerObservation, now: std::time::SystemTime) -> El {
524    if o.observed_at.is_empty() {
525        return el("span")
526            .class("mj-note")
527            .attr("data-observed-at", "")
528            .text(format!("{}: time not recorded", o.source));
529    }
530    let age = observed_age(o, now);
531    let stale = o.is_stale_at(now);
532    el("span")
533        .class("mj-note")
534        .attr("data-observed-at", &o.observed_at)
535        .attr(
536            "data-stale-after",
537            o.stale_after.clone().unwrap_or_default(),
538        )
539        .attr("data-stale", if stale { "true" } else { "false" })
540        .text(format!("{} as of {}{}", o.source, o.observed_at, age))
541        .when(stale, |e| e.text(" ").child(badge("warn", "stale")))
542}

544/// " (N s ago)", from the observation's time; empty when the time does not parse.
545fn observed_age(o: &crate::index::AnswerObservation, now: std::time::SystemTime) -> String {
546    let Some(at) = parse_rfc3339(&o.observed_at) else {
547        return String::new();
548    };
549    let secs = now
550        .duration_since(std::time::UNIX_EPOCH)
551        .map(|d| d.as_secs())
552        .unwrap_or(0)
553        .saturating_sub(at);
554    match secs {
555        0..=119 => format!(" ({secs} s ago)"),
556        120..=7199 => format!(" ({} min ago)", secs / 60),
557        _ => format!(" ({} h ago)", secs / 3600),
558    }
559}

561/// Seconds since the epoch of `YYYY-MM-DDTHH:MM:SSZ`, the one shape [`crate::peers::rfc3339`]
562/// writes.
563fn parse_rfc3339(s: &str) -> Option<u64> {
564    let b = s.as_bytes();
565    if b.len() != 20 || b[19] != b'Z' {
566        return None;
567    }
568    let n = |r: std::ops::Range<usize>| s.get(r)?.parse::<i64>().ok();
569    let (y, mo, d) = (n(0..4)?, n(5..7)?, n(8..10)?);
570    let (h, mi, se) = (n(11..13)?, n(14..16)?, n(17..19)?);
571    // days-from-civil, Howard Hinnant's algorithm: the inverse of the one peers::rfc3339 uses
572    let y = if mo <= 2 { y - 1 } else { y };
573    let era = y.div_euclid(400);
574    let yoe = y - era * 400;
575    let doy = (153 * (if mo > 2 { mo - 3 } else { mo + 9 }) + 2) / 5 + d - 1;
576    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
577    let days = era * 146_097 + doe - 719_468;
578    u64::try_from(days * 86_400 + h * 3600 + mi * 60 + se).ok()
579}

601/// How many rows a listing shows at once before it is paged. A control plane is read a

606/// The window a listing is showing of its result set. The page number is a view
607/// parameter and lives in the URL, so a window is a deep link like every other state
608/// the Cockpit holds.
609#[derive(Debug, Clone, Copy, PartialEq, Eq)]
610pub struct Window {
611    /// The page being shown, 1-based and never past the last one.
612    pub page: usize,
613    /// How many rows a page holds.
614    pub per_page: usize,
615    /// How many rows matched, before the window was taken.
616    pub total: usize,
617}
618
619impl Window {
620    /// The window a request asks for, clamped to what the result set has: page 0 and page
621    /// 900 of a 3-page listing are both a page that exists.
622    pub fn new(page: usize, per_page: usize, total: usize) -> Self {
623        let per_page = per_page.max(1);
624        let pages = total.div_ceil(per_page).max(1);
625        Self {
626            page: page.clamp(1, pages),
627            per_page,
628            total,
629        }
630    }
631
632    /// How many pages the result set has; never zero, because an empty listing still has
633    /// a page to show its empty state on.
634    pub fn pages(&self) -> usize {
635        self.total.div_ceil(self.per_page).max(1)
636    }
637
638    /// The half-open range of rows this window covers, for slicing the matches.
639    pub fn range(&self) -> std::ops::Range<usize> {
640        let start = (self.page - 1) * self.per_page;
641        start..(start + self.per_page).min(self.total)
642    }
643}
644
645/// The page numbers a pagination control offers: the first, the last, the neighbours of
646/// the current one, and a gap marker for what is skipped. Nine hundred pages must not
647/// render nine hundred links.
648fn page_numbers(current: usize, pages: usize) -> Vec<Option<usize>> {
649    let mut wanted: Vec<usize> = [1, pages]
650        .into_iter()
651        .chain(current.saturating_sub(1).max(1)..=(current + 1).min(pages))
652        .filter(|n| *n >= 1 && *n <= pages)
653        .collect();
654    wanted.sort_unstable();
655    wanted.dedup();
656
657    let mut out = Vec::with_capacity(wanted.len() + 2);
658    let mut previous = 0usize;
659    for n in wanted {
660        if previous != 0 && n > previous + 1 {
661            out.push(None);
662        }
663        out.push(Some(n));
664        previous = n;
665    }
666    out
667}

669/// Pagination over a listing: what is being shown, and the way to the rest of it.
670///
671/// `href` builds the URL of one page — the caller owns the listing's filters and puts
672/// them back into every link, so paging never drops a filter and a link can be sent to
673/// somebody as it is.
674pub fn pagination(w: Window, href: impl Fn(usize) -> String) -> El {
675    if w.total == 0 {
676        return el("div").class("mj-hidden");
677    }
678    let range = w.range();
679    let summary = el("p").class("mj-pagination-summary").text(format!(
680        "{}–{} of {}",
681        range.start + 1,
682        range.end,
683        w.total
684    ));
685    if w.pages() == 1 {
686        return el("nav").class("mj-pagination").child(summary);
687    }
688
689    let step = |label: &str, to: Option<usize>| {
690        let li = el("li");
691        match to {
692            Some(n) => li.child(
693                el("a")
694                    .class("mj-page-link")
695                    .attr("href", href(n))
696                    .attr("rel", if n < w.page { "prev" } else { "next" })
697                    .text(label),
698            ),
699            None => li.child(
700                el("span")
701                    .class("mj-page-link mj-page-link--disabled")
702                    .attr("aria-disabled", "true")
703                    .text(label),
704            ),
705        }
706    };
707
708    let mut list = el("ul")
709        .class("mj-pagination-list")
710        .child(step("Previous", (w.page > 1).then(|| w.page - 1)));
711    for slot in page_numbers(w.page, w.pages()) {
712        list = list.child(match slot {
713            Some(n) if n == w.page => el("li").child(
714                el("span")
715                    .class("mj-page-link mj-page-link--current")
716                    .attr("aria-current", "page")
717                    .text(n.to_string()),
718            ),
719            Some(n) => el("li").child(
720                el("a")
721                    .class("mj-page-link")
722                    .attr("href", href(n))
723                    .attr("aria-label", format!("Page {n}"))
724                    .text(n.to_string()),
725            ),
726            None => el("li").child(
727                el("span")
728                    .class("mj-page-gap")
729                    .attr("aria-hidden", "true")
730                    .text("…"),
731            ),
732        });
733    }
734    list = list.child(step("Next", (w.page < w.pages()).then(|| w.page + 1)));
735
736    el("nav")
737        .class("mj-pagination")
738        .attr("aria-label", "Pagination")
739        .child(summary)
740        .child(list)
741}

743/// A row of filters a reader browses by: a label, where it goes, how many are behind it,
744/// and whether it is the one in force. It is the same catalogue the sidebar shows, put
745/// where the listing is, so a set of nine hundred rows is entered by its parts rather
746/// than scrolled.
747pub fn chips(items: Vec<(String, String, usize, bool)>) -> El {
748    if items.is_empty() {
749        return el("div").class("mj-hidden");
750    }
751    let mut list = el("ul").class("mj-chips");
752    for (label, href, count, current) in items {
753        list = list.child(
754            el("li").child(
755                el("a")
756                    .class(if current {
757                        "mj-chip mj-chip--current"
758                    } else {
759                        "mj-chip"
760                    })
761                    .attr("href", href)
762                    .attr_if("aria-current", current.then_some("true"))
763                    .child(el("span").class("mj-chip-label").text(label))
764                    .child(el("span").class("mj-chip-count").text(count.to_string())),
765            ),
766        );
767    }
768    list
769}

771/// A cell holding an identifier: monospaced, on one line, and truncated with the whole
772/// of it in the tooltip. An identifier broken across three lines mid-word is what makes
773/// a table of them unreadable.
774pub fn id_cell(href: impl Into<String>, id: impl Into<String>) -> El {
775    let id = id.into();
776    el("td").class("mj-cell-id").child(
777        el("a")
778            .class("mj-link mj-mono mj-id")
779            .attr("href", href)
780            .attr("title", id.clone())
781            .text(id),
782    )
783}

785/// A class-safe form of a status word: lowercase ASCII letters, digits and dashes, so a
786/// word from data can never build a class name that is not one.
787fn css_word(word: &str) -> String {
788    let cleaned: String = word
789        .chars()
790        .map(|c| {
791            if c.is_ascii_alphanumeric() {
792                c.to_ascii_lowercase()
793            } else {
794                '-'
795            }
796        })
797        .collect();
798    if cleaned.is_empty() {
799        "unknown".into()
800    } else {
801        cleaned
802    }
803}
