1//! The version: the one place it is authored, what is derived from that place, what the
2//! contract requires next and what the commits imply as evidence, and the one writer that
3//! raises it.
4//!
5//! The product version is authored in exactly one place, the `[package] version` line of
6//! the crate manifest ([`MANIFEST`]). Every other statement of it is derived from that line
7//! and checked, or is a record of a version that was released:
8//!
9//! ```text
10//!   apps/majordomus-cli/Cargo.toml   authored    the one place; `release bump` writes it
11//!   crate::VERSION                   compiled    cargo's CARGO_PKG_VERSION
12//!   share/version.txt                projected   `majordomus generate`; bin/majordomus reads it
13//!   apps/majordomus-cli/Cargo.lock   derived     cargo's record; the writer keeps it in step
14//!   generator stamps, the changelog  generated   `majordomus generate`
15//!   .ai/repo/releases/*.yaml         records     the release pipeline, after publication
16//! ```
17//!
18//! The shell tool used to state the version a second time, by hand, because an installed
19//! tree has no `Cargo.toml` and the tool cannot read one at run time. It reads
20//! [`PROJECTION`] instead: a typed generated artifact that ships in every archive beside the
21//! tool, that `generate --check` refuses when it falls behind the manifest, and that nobody
22//! edits. [`diagnose`] is what refuses a version anybody writes down by hand where the
23//! tool's own files live.
24//!
25//! One reader of the authority per language: [`declared`] here, `scripts/release-version`
26//! in shell. Nothing else parses the manifest for its version.
27//!
28//! ```
29//! use majordomus_cli::release::version::{declared_in, parse_projection, render_projection};
30//! // the authority, and the one line the shell tool reads back from its projection of it
31//! let authority = declared_in("[package]\nversion = \"0.9.0\"\n").unwrap();
32//! assert_eq!(parse_projection(&render_projection(&authority)).as_deref(), Some("0.9.0"));
33//! ```

83/// How much a set of changes raises a version.
84#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
85pub enum Bump {
86    /// Nothing since the last release.
87    None,
88    /// A fix, or anything that is not a feature.
89    Patch,
90    /// At least one feature.
91    Minor,
92    /// At least one breaking change.
93    Major,
94}
95
96impl Bump {
97    /// The word the reports use.
98    pub fn as_str(self) -> &'static str {
99        match self {
100            Bump::None => "none",
101            Bump::Patch => "patch",
102            Bump::Minor => "minor",
103            Bump::Major => "major",
104        }
105    }
106
107    /// The bump named by a word, for `release bump --minor` and friends.
108    pub fn parse(word: &str) -> Option<Bump> {
109        match word {
110            "major" => Some(Bump::Major),
111            "minor" => Some(Bump::Minor),
112            "patch" => Some(Bump::Patch),
113            "none" => Some(Bump::None),
114            _ => None,
115        }
116    }
117}

119/// What a set of conventional commits *says about itself*.
120///
121/// The rule is the conventional-commit rule and nothing more: a breaking change is major, a
122/// feature is minor, anything else is patch, and no commits at all is none.
123///
124/// # This is evidence, not authority
125///
126/// It used to decide what `release bump` wrote, and that was the defect this subsystem
127/// exists to remove. A commit message is a label a person chose, and a capability deleted
128/// under a `refactor:` heading was a patch to this function and a broken caller to everyone
129/// else. The compatibility level is now measured from the public contract by
130/// [`crate::release::compat::analyze`]; this answer is carried beside it as
131/// [`crate::release::compat::CommitEvidence`], and when the two disagree the plan says which
132/// one understated the change. Nothing decides a version from this alone.
133pub fn bump_of(changes: &[Change]) -> Bump {
134    if changes.is_empty() {
135        return Bump::None;
136    }
137    if changes.iter().any(|c| c.breaking) {
138        return Bump::Major;
139    }
140    if changes.iter().any(|c| c.kind == ChangeKind::Feat) {
141        return Bump::Minor;
142    }
143    Bump::Patch
144}

146/// A semantic version, only as much as this needs.
147///
148/// Ordered by the three numbers, which is what lets the writer refuse a version smaller than
149/// the one the contract requires rather than only a *bump* smaller than required.
150#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
151pub struct Version {
152    /// Major.
153    pub major: u64,
154    /// Minor.
155    pub minor: u64,
156    /// Patch.
157    pub patch: u64,
158}
159
160impl Version {
161    /// Read `1.2.3`; anything else is `None`.
162    pub fn parse(text: &str) -> Option<Version> {
163        let mut parts = text.trim().split('.');
164        let major = parts.next()?.parse().ok()?;
165        let minor = parts.next()?.parse().ok()?;
166        let patch = parts.next()?.parse().ok()?;
167        if parts.next().is_some() {
168            return None;
169        }
170        Some(Version {
171            major,
172            minor,
173            patch,
174        })
175    }
176
177    /// This version raised by `bump`.
178    ///
179    /// Arithmetic and nothing else: zero-major is not special-cased here, because whether a
180    /// breaking change below 1.0 costs a major or a minor is a *policy* question and this
181    /// repository answers it in exactly one place, [`crate::release::compat::Policy`]. A
182    /// second answer here is the shape this subsystem was built to remove.
183    pub fn raised(self, bump: Bump) -> Version {
184        match bump {
185            Bump::None => self,
186            Bump::Patch => Version {
187                patch: self.patch + 1,
188                ..self
189            },
190            Bump::Minor => Version {
191                minor: self.minor + 1,
192                patch: 0,
193                ..self
194            },
195            Bump::Major => Version {
196                major: self.major + 1,
197                minor: 0,
198                patch: 0,
199            },
200        }
201    }
202}

204impl Version {
205    /// This version raised by a measured compatibility impact.
206    ///
207    /// The bridge between the engine's verdict and the arithmetic: the policy has already
208    /// decided what a `0.x` breaking change costs, so this only has to apply it.
209    ///
210    /// ```
211    /// use majordomus_cli::release::compat::Impact;
212    /// use majordomus_cli::release::version::Version;
213    /// let v = Version::parse("0.5.0").unwrap();
214    /// assert_eq!(v.raised_to(Impact::Minor).to_string(), "0.6.0");
215    /// assert_eq!(v.raised_to(Impact::Major).to_string(), "1.0.0");
216    /// assert_eq!(v.raised_to(Impact::None).to_string(), "0.5.0");
217    /// ```
218    pub fn raised_to(self, impact: crate::release::compat::Impact) -> Version {
219        use crate::release::compat::Impact;
220        self.raised(match impact {
221            Impact::None => Bump::None,
222            Impact::Patch => Bump::Patch,
223            Impact::Minor => Bump::Minor,
224            Impact::Major => Bump::Major,
225        })
226    }
227}

253/// The `[package] version` of a manifest's text.
254///
255/// The first `version = "..."` inside `[package]`, which is where the manifest puts it: a
256/// dependency pinned at a version further down is a different package, a key that merely
257/// begins with `version` is a different key, and a trailing comment is not part of the value.
258/// Read without a TOML parser, because the value wanted is one quoted scalar in a table whose
259/// header is unambiguous, and a dependency to read it would be one this crate argues against.
260///
261/// ```
262/// use majordomus_cli::release::version::declared_in;
263/// let manifest = "[package]\nname = \"x\"\nversion = \"1.2.3\"   # bumped\n\n\
264///                 [dependencies]\nserde = { version = \"1.2.4\" }\n";
265/// assert_eq!(declared_in(manifest).as_deref(), Some("1.2.3"));
266/// // outside [package] there is no package version at all
267/// assert_eq!(declared_in("[dependencies]\nversion = \"1.0.0\"\n"), None);
268/// // a key that only starts with the word is another key
269/// assert_eq!(declared_in("[package]\nversioned = \"1.0.0\"\n"), None);
270/// ```
271pub fn declared_in(manifest: &str) -> Option<String> {
272    let mut in_package = false;
273    for line in manifest.lines() {
274        let line = line.trim();
275        if line.starts_with('[') {
276            in_package = line == "[package]";
277            continue;
278        }
279        if !in_package {
280            continue;
281        }
282        let Some(rest) = line.strip_prefix("version") else {
283            continue;
284        };
285        let Some(rest) = rest.trim_start().strip_prefix('=') else {
286            continue;
287        };
288        let rest = rest.trim();
289        let value = match rest.strip_prefix('"') {
290            Some(quoted) => quoted.split('"').next().unwrap_or_default(),
291            None => rest.split('#').next().unwrap_or_default().trim(),
292        };
293        return (!value.is_empty()).then(|| value.to_string());
294    }
295    None
296}

298/// The version the lock file records for this crate.
299///
300/// Only the entry whose `name` is this crate's: a dependency that happens to be at the same
301/// version is a different package and is never read or written here.
302///
303/// ```
304/// use majordomus_cli::release::version::locked;
305/// let dir = tempfile::tempdir().unwrap();
306/// // No lock file at all is `None`, never a guess.
307/// assert_eq!(locked(dir.path()), None);
308/// ```
309pub fn locked(root: &Path) -> Option<String> {
310    let text = std::fs::read_to_string(root.join(LOCK)).ok()?;
311    let mut in_package = false;
312    for line in text.lines() {
313        let line = line.trim();
314        if line == "[[package]]" {
315            in_package = false;
316            continue;
317        }
318        if line == "name = \"majordomus-cli\"" {
319            in_package = true;
320            continue;
321        }
322        if in_package {
323            if let Some(rest) = line.strip_prefix("version") {
324                return rest.split('"').nth(1).map(|v| v.to_string());
325            }
326        }
327    }
328    None
329}

343/// The version [`PROJECTION`] states: its first `version=` line, past the `#` banner.
344///
345/// Exactly what `bin/majordomus` reads at start-up, so the executable and the shell tool
346/// cannot disagree about what the file says.
347///
348/// ```
349/// use majordomus_cli::release::version::parse_projection;
350/// let file = "# GENERATED FILE — DO NOT EDIT DIRECTLY\n# Generator: majordomus-cli 0.9.0\nversion=0.9.0\n";
351/// assert_eq!(parse_projection(file).as_deref(), Some("0.9.0"));
352/// // a file that states nothing states no version, and an empty value is not one
353/// assert_eq!(parse_projection("# nothing here\n"), None);
354/// assert_eq!(parse_projection("version=\n"), None);
355/// ```
356pub fn parse_projection(text: &str) -> Option<String> {
357    text.lines()
358        .filter(|l| !l.trim_start().starts_with('#'))
359        .find_map(|l| l.strip_prefix("version="))
360        .map(|v| v.trim().to_string())
361        .filter(|v| !v.is_empty())
362}

378/// How [`PROJECTION`] stands against the authority.
379///
380/// Three answers that lead to three different actions: nothing to do, run the derivation,
381/// or find out who wrote a version nobody authored.
382///
383/// ```
384/// use majordomus_cli::release::version::ProjectionState;
385/// let stale = ProjectionState::Stale { projected: Some("0.8.0".into()), declared: "0.9.0".into() };
386/// // the state every bump leaves behind is not the state nothing could have produced
387/// assert_ne!(stale, ProjectionState::Disagrees { projected: "0.8.0".into(), declared: Some("0.9.0".into()) });
388/// ```
389#[derive(Debug, Clone, PartialEq, Eq)]
390pub enum ProjectionState {
391    /// Neither the manifest nor the projection is here: this repository does not carry the
392    /// product, so there is nothing to project and nothing to compare.
393    NotHere,
394    /// The projection states the version the manifest declares.
395    Current,
396    /// The projection is missing, or states an older version than the manifest: the state
397    /// between `release bump` and the derivation that follows it. `generate` fixes it.
398    Stale {
399        /// What the projection states, when it states anything.
400        projected: Option<String>,
401        /// What the manifest declares.
402        declared: String,
403    },
404    /// The projection states a version newer than the manifest's, or one that cannot be
405    /// compared with it, or there is a projection and no manifest. No derivation writes
406    /// that, so somebody did.
407    Disagrees {
408        /// What the projection states.
409        projected: String,
410        /// What the manifest declares, when there is one.
411        declared: Option<String>,
412    },
413}

415/// How [`PROJECTION`] stands against [`MANIFEST`]: compared as versions, so that a projection
416/// behind the manifest — what a bump leaves until the derivation runs — is told apart from
417/// one ahead of it or unrelated to it, which no derivation ever writes.
418///
419/// ```
420/// use majordomus_cli::release::version::{projection_state, ProjectionState, MANIFEST, PROJECTION};
421/// let dir = tempfile::tempdir().unwrap();
422/// let root = dir.path();
423/// assert_eq!(projection_state(root), ProjectionState::NotHere);
424/// std::fs::create_dir_all(root.join("apps/majordomus-cli")).unwrap();
425/// std::fs::create_dir_all(root.join("share")).unwrap();
426/// std::fs::write(root.join(MANIFEST), "[package]\nversion = \"0.9.0\"\n").unwrap();
427/// // raised and not yet derived: stale, which a derivation fixes
428/// std::fs::write(root.join(PROJECTION), "version=0.8.0\n").unwrap();
429/// assert!(matches!(projection_state(root), ProjectionState::Stale { .. }));
430/// // a version the manifest never declared: nothing derives that
431/// std::fs::write(root.join(PROJECTION), "version=1.0.0\n").unwrap();
432/// assert!(matches!(projection_state(root), ProjectionState::Disagrees { .. }));
433/// std::fs::write(root.join(PROJECTION), "version=0.9.0\n").unwrap();
434/// assert_eq!(projection_state(root), ProjectionState::Current);
435/// ```
436pub fn projection_state(root: &Path) -> ProjectionState {
437    match (declared(root), tool(root)) {
438        (None, None) => ProjectionState::NotHere,
439        (None, Some(projected)) => ProjectionState::Disagrees {
440            projected,
441            declared: None,
442        },
443        (Some(declared), None) => ProjectionState::Stale {
444            projected: None,
445            declared,
446        },
447        (Some(declared), Some(projected)) if declared == projected => ProjectionState::Current,
448        (Some(declared), Some(projected)) => {
449            match (Version::parse(&projected), Version::parse(&declared)) {
450                (Some(p), Some(d)) if p < d => ProjectionState::Stale {
451                    projected: Some(projected),
452                    declared,
453                },
454                _ => ProjectionState::Disagrees {
455                    projected,
456                    declared: Some(declared),
457                },
458            }
459        }
460    }
461}

502/// The shape a line states a version in, when it states one.
503///
504/// Three shapes, each a place a version is *declared* rather than mentioned:
505///
506/// - an assignment to a name ending in `version`, in any case and in the languages the
507///   tool's files are written in: shell `MJ_VERSION="0.8.0"` and `export FOO_VERSION=1.2.3`,
508///   TOML `version = "1.2.3"`, JavaScript `const VERSION = '1.2.3'`, Python
509///   `__version__ = "1.2.3"`;
510/// - a `version` member of YAML or JSON at the start of a line: `version: 1.2.3`,
511///   `"version": "1.2.3"`;
512/// - the product's own answer to `version`: `majordomus 0.8.0`.
513///
514/// A version is three numbers, optionally after a `v`; a schema's `version: 1` is not one.
515///
516/// ```
517/// use majordomus_cli::release::version::carrier_shape;
518/// assert!(carrier_shape("MJ_VERSION=\"0.8.0\"").is_some());
519/// assert!(carrier_shape("  export FOO_VERSION=v1.2.3").is_some());
520/// assert!(carrier_shape("declare -r MJ_VERSION=\"1.2.3\"").is_some());
521/// assert!(carrier_shape("version = \"1.2.3\"").is_some());
522/// assert!(carrier_shape("export const VERSION = '1.2.3';").is_some());
523/// assert!(carrier_shape("let appVersion = `1.2.3`;").is_some());
524/// assert!(carrier_shape("__version__ = \"1.2.3\"").is_some());
525/// assert!(carrier_shape("version: 1.2.3").is_some());
526/// assert!(carrier_shape("  \"version\": \"1.2.3\",").is_some());
527/// assert!(carrier_shape("echo \"majordomus 0.8.0\"").is_some());
528/// // read, not stated: a variable, a schema version, a generator stamp, a sentinel, a
529/// // comparison, a property of something else
530/// assert!(carrier_shape("MJ_VERSION=\"${line#version=}\"").is_none());
531/// assert!(carrier_shape("version: 1").is_none());
532/// assert!(carrier_shape("# Generator: majordomus-cli 0.8.0").is_none());
533/// assert!(carrier_shape("MJ_VERSION=\"site\"").is_none());
534/// assert!(carrier_shape("version == \"1.2.3\"").is_none());
535/// assert!(carrier_shape("pkg.version = \"1.2.3\"").is_none());
536/// assert!(carrier_shape("versioned = \"1.2.3\"").is_none());
537/// ```
538pub fn carrier_shape(line: &str) -> Option<&'static str> {
539    let t = line.trim_start();
540
541    // an assignment to a name ending in `version`, once the words that declare it are set aside
542    if let Some((name, value)) = declared_name(t).split_once('=') {
543        let name = name.trim_end();
544        let is_version_name = !name.is_empty()
545            && name
546                .chars()
547                .all(|c| c.is_ascii_alphanumeric() || c == '_' || c == '$')
548            && name
549                .trim_matches('_')
550                .to_ascii_lowercase()
551                .ends_with("version");
552        let value = value.trim_start().trim_start_matches(['"', '\'', '`']);
553        if is_version_name && leads_with_version(value) {
554            return Some("an assignment to a version name");
555        }
556    }
557
558    // a `version` member of YAML or JSON, at the start of a line or of a list item
559    let member = t.strip_prefix("- ").unwrap_or(t);
560    let after_key = member
561        .strip_prefix("\"version\"")
562        .or_else(|| member.strip_prefix("'version'"))
563        .or_else(|| member.strip_prefix("version"));
564    if let Some(rest) = after_key {
565        if let Some(value) = rest.trim_start().strip_prefix(':') {
566            if leads_with_version(value.trim_start().trim_start_matches(['"', '\''])) {
567                return Some("a version member");
568            }
569        }
570    }
571
572    // the product's own answer to `majordomus version`
573    const PRODUCT: &str = "majordomus ";
574    for (at, _) in t.match_indices(PRODUCT) {
575        let before = t[..at].chars().next_back();
576        let joined = before.is_some_and(|c| c.is_ascii_alphanumeric() || "-_./".contains(c));
577        if !joined && leads_with_version(&t[at + PRODUCT.len()..]) {
578            return Some("the product's version line");
579        }
580    }
581    None
582}

584/// A line with the words that declare a name set aside — `export`, `readonly`, `local`,
585/// `declare` and `typeset` with their flags, `const`, `let`, `var` — so that what is left
586/// begins with the name being assigned.
587fn declared_name(mut t: &str) -> &str {
588    const DECLARING: &[&str] = &[
589        "export", "readonly", "local", "declare", "typeset", "const", "let", "var",
590    ];
591    const TAKES_FLAGS: &[&str] = &["export", "readonly", "local", "declare", "typeset"];
592    while let Some((word, rest)) = t.split_once(char::is_whitespace) {
593        if !DECLARING.contains(&word) {
594            break;
595        }
596        t = rest.trim_start();
597        if TAKES_FLAGS.contains(&word) {
598            while let Some((flag, rest)) = t.split_once(char::is_whitespace) {
599                if flag.len() > 1 && flag.starts_with('-') && !flag.contains('=') {
600                    t = rest.trim_start();
601                } else {
602                    break;
603                }
604            }
605        }
606    }
607    t
608}

610/// Whether `text` begins with a version: three dot-separated numbers, optionally after a
611/// `v`, and not followed by a fourth.
612fn leads_with_version(text: &str) -> bool {
613    let bytes = text.strip_prefix('v').unwrap_or(text).as_bytes();
614    let mut at = 0;
615    for group in 0..3 {
616        let start = at;
617        while at < bytes.len() && bytes[at].is_ascii_digit() {
618            at += 1;
619        }
620        if at == start {
621            return false;
622        }
623        if group < 2 {
624            if bytes.get(at) != Some(&b'.') {
625                return false;
626            }
627            at += 1;
628        }
629    }
630    !matches!(bytes.get(at), Some(b) if *b == b'.' || b.is_ascii_digit())
631}

633/// Every version written down by hand under [`CARRIER_ROOTS`].
634///
635/// The files git would commit there — tracked, or new and not ignored, so a file is refused
636/// before it is staged rather than after — or, where this is not a git checkout, every file
637/// under those directories, so that a tree nobody committed is still read rather than
638/// passed. [`PROJECTION`] is not a statement by hand, and neither is any path the generated
639/// artifact manifest names: `majordomus generate` wrote those, and `generate --check` is
640/// what holds them.
641///
642/// ```
643/// use majordomus_cli::release::version::carriers;
644/// let dir = tempfile::tempdir().unwrap();
645/// std::fs::create_dir_all(dir.path().join("lib")).unwrap();
646/// std::fs::write(dir.path().join("lib/x.sh"), "FOO_VERSION=\"1.2.3\"\n").unwrap();
647/// let found = carriers(dir.path());
648/// assert_eq!(found.len(), 1);
649/// assert_eq!((found[0].path.as_str(), found[0].line), ("lib/x.sh", 1));
650/// ```
651pub fn carriers(root: &Path) -> Vec<Carrier> {
652    let files = committable(root).unwrap_or_else(|| walk(root));
653    let generated = generated_paths(root);
654    let mut out = Vec::new();
655    for path in files {
656        if path == PROJECTION || generated.contains(&path) {
657            continue;
658        }
659        let full = root.join(&path);
660        let small =
661            std::fs::symlink_metadata(&full).is_ok_and(|m| m.is_file() && m.len() <= LARGEST_READ);
662        if !small {
663            continue;
664        }
665        let Ok(text) = std::fs::read_to_string(&full) else {
666            continue;
667        };
668        for (i, line) in text.lines().enumerate() {
669            if let Some(shape) = carrier_shape(line) {
670                out.push(Carrier {
671                    path: path.clone(),
672                    line: i + 1,
673                    shape,
674                    text: line.trim().to_string(),
675                });
676            }
677        }
678    }
679    out
680}

682/// The files under [`CARRIER_ROOTS`] git would commit: tracked, or untracked and not
683/// ignored. `None` when git cannot answer, which is not the same as "no files".
684fn committable(root: &Path) -> Option<Vec<String>> {
685    let out = crate::git::read_only(root)
686        .args([
687            "ls-files",
688            "-z",
689            "--cached",
690            "--others",
691            "--exclude-standard",
692            "--",
693        ])
694        .args(CARRIER_ROOTS)
695        .output()
696        .ok()?;
697    if !out.status.success() {
698        return None;
699    }
700    let mut files: Vec<String> = out
701        .stdout
702        .split(|b| *b == 0)
703        .filter(|s| !s.is_empty())
704        .map(|s| String::from_utf8_lossy(s).into_owned())
705        .collect();
706    crate::order::canonical_strings(&mut files);
707    files.dedup();
708    Some(files)
709}

711/// Every file under [`CARRIER_ROOTS`], repository-relative, in a stable order: the answer
712/// when git cannot list them.
713fn walk(root: &Path) -> Vec<String> {
714    fn visit(dir: &Path, root: &Path, out: &mut Vec<String>) {
715        let Ok(entries) = std::fs::read_dir(dir) else {
716            return;
717        };
718        for entry in entries.flatten() {
719            let path = entry.path();
720            match entry.file_type() {
721                Ok(t) if t.is_dir() => visit(&path, root, out),
722                Ok(t) if t.is_file() => {
723                    if let Ok(rel) = path.strip_prefix(root) {
724                        out.push(rel.to_string_lossy().replace('\\', "/"));
725                    }
726                }
727                _ => {}
728            }
729        }
730    }
731    let mut out = Vec::new();
732    for dir in CARRIER_ROOTS {
733        visit(&root.join(dir), root, &mut out);
734    }
735    crate::order::canonical_strings(&mut out);
736    out
737}

751/// Everything wrong with where the version is stated: the projection against the
752/// authority, and every version written down by hand where the tool's own files live.
753///
754/// A stale projection is a [`Severity::Warning`] — it is the state every bump leaves until
755/// `scripts/derive` runs, `generate --check` is what refuses it, and a writer that refused
756/// its own un-derived state could not correct a bump it had just made. A projection nothing
757/// could have derived, and a version written by hand, are [`Severity::Error`]s. Asked only
758/// where the product lives: a repository without the crate has its own `scripts/` and its
759/// own versions, and they are none of this module's business.
760///
761/// ```
762/// use majordomus_cli::release::compat::Severity;
763/// use majordomus_cli::release::version::{diagnose, MANIFEST, PROJECTION};
764/// let dir = tempfile::tempdir().unwrap();
765/// let root = dir.path();
766/// std::fs::create_dir_all(root.join("apps/majordomus-cli")).unwrap();
767/// std::fs::create_dir_all(root.join("share")).unwrap();
768/// std::fs::create_dir_all(root.join("lib")).unwrap();
769/// std::fs::write(root.join(MANIFEST), "[package]\nversion = \"0.9.0\"\n").unwrap();
770/// std::fs::write(root.join(PROJECTION), "version=0.9.0\n").unwrap();
771/// assert!(diagnose(root).is_empty(), "authored once, projected once");
772///
773/// std::fs::write(root.join("lib/x.sh"), "FOO_VERSION=\"1.2.3\"\n").unwrap();
774/// let found = diagnose(root);
775/// assert_eq!(found.len(), 1);
776/// assert_eq!(found[0].id, "version-stated-by-hand");
777/// assert_eq!(found[0].severity, Severity::Error);
778/// ```
779pub fn diagnose(root: &Path) -> Vec<Diagnostic> {
780    let mut out = Vec::new();
781    match projection_state(root) {
782        ProjectionState::NotHere | ProjectionState::Current => {}
783        ProjectionState::Stale {
784            projected,
785            declared,
786        } => out.push(Diagnostic {
787            id: PROJECTION_STALE.into(),
788            severity: Severity::Warning,
789            message: format!(
790                "{PROJECTION} states '{}' and {MANIFEST} declares '{declared}': the projection \
791                 is stale — run scripts/derive, which builds and then generates it; never edit \
792                 it or bin/majordomus",
793                projected.as_deref().unwrap_or("nothing")
794            ),
795        }),
796        ProjectionState::Disagrees {
797            projected,
798            declared,
799        } => out.push(Diagnostic {
800            id: WRITERS_DISAGREE.into(),
801            severity: Severity::Error,
802            message: format!(
803                "{PROJECTION} states '{projected}' and {MANIFEST} declares '{}': no derivation \
804                 writes a version the manifest does not declare, so it was written by hand — \
805                 run scripts/derive, and raise the version only with `majordomus release bump`",
806                declared.as_deref().unwrap_or("nothing")
807            ),
808        }),
809    }
810    if root.join(MANIFEST).is_file() {
811        for c in carriers(root) {
812            out.push(Diagnostic {
813                id: STATED_BY_HAND.into(),
814                severity: Severity::Error,
815                message: format!(
816                    "{}:{} states a version by hand, as {}: `{}`. The version is authored \
817                     only in {MANIFEST}; read it with `scripts/release-version` in shell, \
818                     `release::version::declared` in Rust, or `bin/majordomus version`",
819                    c.path, c.line, c.shape, c.text
820                ),
821            });
822        }
823    }
824    out
825}

827/// Write `to` into the one place the version is authored, keep the lock's record of it in
828/// step, and say which files changed.
829///
830/// Byte-exact and narrow: the manifest's `version` line inside `[package]`, and the `version`
831/// line of the lock's own `majordomus-cli` entry — the one line cargo would rewrite. Nothing
832/// else in either file is read or rewritten, so a dependency at the same version is
833/// untouched. [`PROJECTION`], the generator stamps and the changelog are not written here:
834/// they are derived, and `scripts/derive` derives them.
835///
836/// ```
837/// use majordomus_cli::release::version::{declared, write, MANIFEST};
838/// let dir = tempfile::tempdir().unwrap();
839/// std::fs::create_dir_all(dir.path().join("apps/majordomus-cli")).unwrap();
840/// std::fs::write(dir.path().join(MANIFEST), "[package]\nversion = \"0.8.0\"\n").unwrap();
841/// assert_eq!(write(dir.path(), "0.9.0").unwrap(), vec![MANIFEST.to_string()]);
842/// assert_eq!(declared(dir.path()).as_deref(), Some("0.9.0"));
843/// ```
844pub fn write(root: &Path, to: &str) -> std::io::Result<Vec<String>> {
845    let mut written = Vec::new();
846
847    let manifest = root.join(MANIFEST);
848    let text = std::fs::read_to_string(&manifest)?;
849    let mut out = String::with_capacity(text.len());
850    let mut in_package = false;
851    let mut done = false;
852    for line in text.lines() {
853        let trimmed = line.trim();
854        if trimmed.starts_with('[') {
855            in_package = trimmed == "[package]";
856        }
857        if in_package && !done && trimmed.starts_with("version") && trimmed.contains('"') {
858            out.push_str(&format!("version = \"{to}\"\n"));
859            done = true;
860            continue;
861        }
862        out.push_str(line);
863        out.push('\n');
864    }
865    if out != text {
866        std::fs::write(&manifest, out)?;
867        written.push(MANIFEST.to_string());
868    }
869
870    // The lock's own entry, and only it: the `version` line that follows
871    // `name = "majordomus-cli"`. Every other `version` in the file belongs to a dependency.
872    let lock = root.join(LOCK);
873    if let Ok(text) = std::fs::read_to_string(&lock) {
874        let mut out = String::with_capacity(text.len());
875        let mut here = false;
876        let mut done = false;
877        for line in text.lines() {
878            let trimmed = line.trim();
879            if trimmed == "[[package]]" {
880                here = false;
881            } else if trimmed == "name = \"majordomus-cli\"" {
882                here = true;
883            } else if here && !done && trimmed.starts_with("version") && trimmed.contains('"') {
884                out.push_str(&format!("version = \"{to}\"\n"));
885                here = false;
886                done = true;
887                continue;
888            }
889            out.push_str(line);
890            out.push('\n');
891        }
892        if out != text {
893            std::fs::write(&lock, out)?;
894            written.push(LOCK.to_string());
895        }
896    }
897
898    Ok(written)
899}

901/// The version the commit subjects imply: the last release raised by their bump — from the
902/// declared version only when nothing was released. Evidence, never the answer on its own.
903///
904/// It raises the last release and not the declared version, because a declared version
905/// already raised for the window must not be raised again by the same commits.
906///
907/// ```
908/// use majordomus_cli::release::version::{commits_imply, Bump};
909/// // released 0.10.0, already declared 0.11.0, only features since: 0.11.0, not 0.12.0
910/// assert_eq!(commits_imply(Some("0.10.0"), "0.11.0", Bump::Minor).as_deref(), Some("0.11.0"));
911/// // nothing released: the declared version is the only base there is
912/// assert_eq!(commits_imply(None, "0.3.0", Bump::Patch).as_deref(), Some("0.3.1"));
913/// ```
914pub fn commits_imply(last: Option<&str>, declared: &str, bump: Bump) -> Option<String> {
915    last.and_then(Version::parse)
916        .or_else(|| Version::parse(declared))
917        .map(|v| v.raised(bump).to_string())
918}

920/// Who answers `next`, and with what: the contract when it could be measured, and no one
921/// when it could not.
922///
923/// `contract` is the analysis' measured version ([`crate::release::compat::VersionPlan::
924/// measured_version`]), or why none could be measured. ADR 0051 refuses an unmeasurable
925/// baseline rather than guessing one, so an `Err` leaves `next` absent, [`DecidedBy::
926/// Undecided`] and the reason carried — the commit inference stays evidence beside it and
927/// never stands in for the measurement. The contract's answer is a minimum: when it is no
928/// greater than `last` — only behaviour behind the public boundary changed, so the contract
929/// requires no release — it is the release already made, never the next one. Then the
930/// commits decide whether anything is released at all, and the contract decides how much:
931/// the smallest release above the last, a patch, labelled [`DecidedBy::ContractAndCommits`].
932/// The commits never raise `next` beyond what the contract requires.
933///
934/// ```
935/// use majordomus_cli::release::model::DecidedBy;
936/// use majordomus_cli::release::version::{decide, Bump};
937/// // the commits say BREAKING, the contract only grew: below 1.0 that owes a minor
938/// let (next, by, why) = decide(Some("0.10.0"), Bump::Major, Ok("0.11.0".into()));
939/// assert_eq!((next.as_deref(), by, why), (Some("0.11.0"), DecidedBy::Contract, None));
940/// // only fixes, the surface unchanged: the contract's minimum is the release already made
941/// let (next, by, _) = decide(Some("0.10.0"), Bump::Patch, Ok("0.10.0".into()));
942/// assert_eq!((next.as_deref(), by), (Some("0.10.1"), DecidedBy::ContractAndCommits));
943/// // no published baseline: no one decides, and the reason is carried
944/// let (next, by, why) = decide(None, Bump::Minor, Err("no release".into()));
945/// assert_eq!((next, by), (None, DecidedBy::Undecided));
946/// assert_eq!(why.as_deref(), Some("no release"));
947/// ```
948pub fn decide(
949    last: Option<&str>,
950    bump: Bump,
951    contract: Result<String, String>,
952) -> (Option<String>, DecidedBy, Option<String>) {
953    let required = match contract {
954        Ok(required) => required,
955        Err(why) => return (None, DecidedBy::Undecided, Some(why)),
956    };
957    match (Version::parse(&required), last.and_then(Version::parse)) {
958        (Some(r), Some(l)) if r <= l => match bump {
959            Bump::None => (None, DecidedBy::Contract, None),
960            _ => (
961                Some(l.raised(Bump::Patch).to_string()),
962                DecidedBy::ContractAndCommits,
963                None,
964            ),
965        },
966        _ => (Some(required), DecidedBy::Contract, None),
967    }
968}

970/// The whole version answer: what is declared, whether the projection the shell tool reads
971/// states it, what the public contract requires next, and what the commits since the last
972/// release imply — as evidence.
973///
974/// `contract` is the analysis' answer ([`decide`] says how it is used); [`select`] is the
975/// caller that asks it, and the one every surface goes through. The last release is
976/// [`crate::release::compat::last_release`] — the same baseline `release analyze` measures
977/// from, the highest version the layer records — so the contract and the commit evidence
978/// always measure one window. The commit inference raises that release ([`commits_imply`]).
979pub fn report(root: &Path, objects: &[Object], contract: Result<String, String>) -> VersionReport {
980    let declared = declared(root).unwrap_or_else(|| "unknown".into());
981    let tool_version = tool(root).unwrap_or_else(|| "unknown".into());
982
983    let last = super::compat::last_release(root, objects);
984    let changes = match &last {
985        Some(l) => super::commits::in_range(root, &format!("{}..HEAD", l.read_at), objects),
986        None => super::commits::in_range(root, "HEAD", objects),
987    };
988    let bump = bump_of(&changes);
989    let last_version = last.map(|l| l.version);
990    let commits_imply = commits_imply(last_version.as_deref(), &declared, bump);
991    let (next, decided_by, contract_unreadable) = decide(last_version.as_deref(), bump, contract);
992
993    VersionReport {
994        agree: projection_current(root),
995        declared,
996        tool: tool_version,
997        last_release: last_version,
998        bump: bump.as_str().to_string(),
999        commits_imply,
1000        next,
1001        decided_by,
1002        contract_unreadable,
1003        changes,
1004    }
1005}

1007/// The release selection: the contract's analysis against the last release, and the version
1008/// report decided from it — made once, by one function.
1009///
1010/// `release version` (the capability, and every surface that renders it) and `release bump`
1011/// without an explicit target both take their answer from here, so the report and the writer
1012/// cannot disagree about what the next version is: the report's `next` is the version the
1013/// writer raises to ([`default_target`]).
1014///
1015/// ```
1016/// use majordomus_cli::capability::registry::CapabilityRegistry;
1017/// use majordomus_cli::release::model::DecidedBy;
1018/// use majordomus_cli::release::version::{select, Selection};
1019/// let dir = tempfile::tempdir().unwrap();
1020/// let registry = CapabilityRegistry::builder().build().unwrap();
1021/// let selection: Selection = select(dir.path(), &registry, &[]);
1022/// // nothing published: no plan, and so a report that decides nothing — with the same reason
1023/// let why = selection.plan.as_ref().unwrap_err();
1024/// assert_eq!(selection.report.decided_by, DecidedBy::Undecided);
1025/// assert_eq!(selection.report.contract_unreadable.as_ref(), Some(why));
1026/// assert_eq!(selection.report.next, None);
1027/// ```
1028#[derive(Debug)]
1029pub struct Selection {
1030    /// The analysis `release analyze` makes against the last release, or why it could not be
1031    /// made at all (no release to compare with, a baseline whose registry is not committed).
1032    pub plan: Result<super::compat::VersionPlan, String>,
1033    /// The version report decided from that analysis.
1034    pub report: VersionReport,
1035}

1037/// Make the release selection ([`Selection`]): analyse the contract against the last
1038/// release, refuse a guess ([`crate::release::compat::VersionPlan::measured_version`]), and
1039/// decide the report from what is left ([`report`]).
1040///
1041/// ```
1042/// use majordomus_cli::capability::registry::CapabilityRegistry;
1043/// use majordomus_cli::release::version::{default_target, select, Version};
1044/// let dir = tempfile::tempdir().unwrap();
1045/// let registry = CapabilityRegistry::builder().build().unwrap();
1046/// let selection = select(dir.path(), &registry, &[]);
1047/// // the writer's default target is read from this report, so it refuses with its reason
1048/// let current = Version::parse("0.1.0").unwrap();
1049/// assert_eq!(
1050///     default_target(&selection.report, current).unwrap_err(),
1051///     selection.report.contract_unreadable.clone().unwrap()
1052/// );
1053/// ```
1054pub fn select(
1055    root: &Path,
1056    registry: &crate::capability::registry::CapabilityRegistry,
1057    objects: &[Object],
1058) -> Selection {
1059    let plan = super::compat::analyze(root, registry, objects, None).map_err(|e| e.to_string());
1060    let contract = match &plan {
1061        Ok(p) => p.measured_version(),
1062        Err(why) => Err(why.clone()),
1063    };
1064    Selection {
1065        report: report(root, objects, contract),
1066        plan,
1067    }
1068}

1070/// The version `release bump` raises to when no target is named: the report's `next`, the
1071/// declared version when nothing would be released — and a refusal, with the report's
1072/// reason, when the contract could not be measured and so no one decided.
1073///
1074/// ```
1075/// use majordomus_cli::release::model::{DecidedBy, VersionReport};
1076/// use majordomus_cli::release::version::{default_target, Version};
1077/// let report = |next: Option<&str>, decided_by, why: Option<&str>| VersionReport {
1078///     declared: "0.11.0".into(), tool: "0.11.0".into(), agree: true,
1079///     last_release: Some("0.11.0".into()), bump: "patch".into(),
1080///     commits_imply: Some("0.11.1".into()), next: next.map(Into::into), decided_by,
1081///     contract_unreadable: why.map(Into::into), changes: Vec::new() };
1082/// let current = Version::parse("0.11.0").unwrap();
1083/// let fix = report(Some("0.11.1"), DecidedBy::ContractAndCommits, None);
1084/// assert_eq!(default_target(&fix, current).unwrap().to_string(), "0.11.1");
1085/// let nothing = report(None, DecidedBy::Contract, None);
1086/// assert_eq!(default_target(&nothing, current), Ok(current));
1087/// let unread = report(None, DecidedBy::Undecided, Some("no release"));
1088/// assert_eq!(default_target(&unread, current), Err("no release".into()));
1089/// ```
1090pub fn default_target(report: &VersionReport, current: Version) -> Result<Version, String> {
1091    match (&report.next, report.decided_by) {
1092        (_, DecidedBy::Undecided) => Err(report
1093            .contract_unreadable
1094            .clone()
1095            .unwrap_or_else(|| "the public contract could not be measured".into())),
1096        (Some(next), _) => Version::parse(next)
1097            .ok_or_else(|| format!("the selected version '{next}' is not three numbers")),
1098        (None, _) => Ok(current),
1099    }
1100}
