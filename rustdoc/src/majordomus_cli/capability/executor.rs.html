1//! The one execution path.
2//!
3//! Every call of an executable capability — from the stdio MCP session, from `/mcp`, from
4//! an HTTP route, from the command line, from the Cockpit or from a benchmark runner —
5//! goes through [`CapabilityExecutor::execute`]: registry lookup, counters, the cache the
6//! capability's own policy asks for, then the handler. Transport adapters convert protocol
7//! to JSON and back and own nothing else, so instrumentation and caching apply to every
8//! transport at once and cannot drift apart. A second execution path is the defect this
9//! module exists to make impossible.
10//!
11//! # The cache key
12//!
13//! Process memory, bounded per capability by its [`CachePolicy`], keyed
14//! by three things: the canonical id, the input **normalised** (object keys sorted at every
15//! level, so `{"a":1,"b":2}` and `{"b":2,"a":1}` are one entry), and the registry
16//! fingerprint — which hashes every descriptor and every declarative object's content. Two
17//! processes over different repository states therefore never share an entry, and a changed
18//! layer never answers from an old one.
19//!
20//! Errors are never cached. Commands are never cached, and the registry refuses a
21//! descriptor that asks for it, so that is a build-time refusal rather than a runtime one.
22//!
23//! # Invisibility
24//!
25//! A cached answer is byte-identical to a computed one; nothing in a response says which it
26//! was. `project.cache-is-invisible` requires it, and [`canonical_json`] is what makes the
27//! key well-defined enough for it to be true.
28//!
29//! ```
30//! use majordomus_cli::capability::executor::canonical_json;
31//!
32//! // the same input written two ways is one cache entry, at every level of nesting
33//! let one = serde_json::json!({ "b": 2, "a": { "y": 1, "x": 0 } });
34//! let other = serde_json::json!({ "a": { "x": 0, "y": 1 }, "b": 2 });
35//! assert_eq!(canonical_json(&one), canonical_json(&other));
36//!
37//! // and a different input is a different entry
38//! assert_ne!(canonical_json(&one), canonical_json(&serde_json::json!({ "b": 3 })));
39//! ```

52/// The executor of one registry.
53#[derive(Debug, Default)]
54pub struct CapabilityExecutor {
55    cache: Mutex<Cache>,
56}
57
58#[derive(Debug, Default)]
59struct Cache {
60    entries: HashMap<CacheKey, CacheEntry>,
61    /// Insertion order per capability, for eviction.
62    order: HashMap<String, VecDeque<CacheKey>>,
63}
64
65#[derive(Debug, Clone, PartialEq, Eq, Hash)]
66struct CacheKey {
67    id: String,
68    input: String,
69    fingerprint: String,
70}
71
72#[derive(Debug)]
73struct CacheEntry {
74    value: Value,
75    stored: Instant,
76}

78impl CapabilityExecutor {
79    /// An executor with an empty cache.

84    /// Execute a capability by id with a JSON input. The one place a handler runs.
85    pub fn execute(&self, ctx: &Context, id: &str, input: Value) -> Result<Value, CapabilityError> {
86        let policy = ctx.registry.get(id).map(|c| c.cache);
87        self.run(ctx, id, input, policy)
88    }

90    /// The same call with the cache stepped over: the handler runs, whatever is stored.
91    ///
92    /// One caller: the execution engine. A cached answer is indistinguishable from a fresh
93    /// one — that is the point of the cache and the rule `project.cache-is-invisible` says
94    /// so — but an *execution* is not an answer. It is a record of work happening, watched
95    /// while it happens, and an execution answered from a cache would report no steps, no
96    /// progress and no output of its own while claiming to have succeeded: a story of work
97    /// that did not occur. So an execution always runs the handler, and nothing it produces
98    /// is put into the cache either, because the cache holds what calls returned and this
99    /// call is a different kind of thing.
100    pub fn execute_uncached(
101        &self,
102        ctx: &Context,
103        id: &str,
104        input: Value,
105    ) -> Result<Value, CapabilityError> {
106        self.run(ctx, id, input, Some(CachePolicy::Disabled))
107    }
108
109    fn run(
110        &self,
111        ctx: &Context,
112        id: &str,
113        input: Value,
114        policy: Option<CachePolicy>,
115    ) -> Result<Value, CapabilityError> {
116        Counters::bump(&COUNTERS.executions);
117        let policy = policy.ok_or_else(|| CapabilityError::NotFound(format!("capability {id}")))?;
118        let key = match policy {
119            CachePolicy::Disabled => None,
120            CachePolicy::Process { .. } => Some(CacheKey {
121                id: id.to_string(),
122                input: canonical_json(&input),
123                fingerprint: ctx.registry.fingerprint().to_string(),
124            }),
125        };
126        if let (Some(key), CachePolicy::Process { ttl_seconds, .. }) = (&key, policy) {
127            let _guard = perf::phase(Phase::CacheLookup);
128            let mut cache = lock(&self.cache);
129            let fresh = cache.entries.get(key).map(|e| {
130                ttl_seconds.is_none_or(|ttl| e.stored.elapsed() < Duration::from_secs(ttl))
131            });
132            match fresh {
133                Some(true) => {
134                    Counters::bump(&COUNTERS.cache_hits);
135                    return Ok(cache.entries[key].value.clone());
136                }
137                Some(false) => {
138                    cache.entries.remove(key);
139                    Counters::bump(&COUNTERS.cache_evictions);
140                    Counters::bump(&COUNTERS.cache_misses);
141                }
142                None => Counters::bump(&COUNTERS.cache_misses),
143            }
144        }
145        Counters::bump(&COUNTERS.handler_invocations);
146        let value = {
147            let _guard = perf::phase(Phase::HandlerExecution);
148            ctx.registry.dispatch(ctx, id, input)?
149        };
150        if let (Some(key), CachePolicy::Process { max_entries, .. }) = (key, policy) {
151            let mut cache = lock(&self.cache);
152            let Cache { entries, order } = &mut *cache;
153            let order = order.entry(key.id.clone()).or_default();
154            if !entries.contains_key(&key) {
155                order.push_back(key.clone());
156                while order.len() > max_entries {
157                    if let Some(old) = order.pop_front() {
158                        entries.remove(&old);
159                        Counters::bump(&COUNTERS.cache_evictions);
160                    }
161                }
162            }
163            entries.insert(
164                key,
165                CacheEntry {
166                    value: value.clone(),
167                    stored: Instant::now(),
168                },
169            );
170        }
171        Ok(value)
172    }
173
174    /// How many entries the cache holds, all capabilities together.
175    pub fn cached_entries(&self) -> usize {
176        lock(&self.cache).entries.len()
177    }
178
179    /// Drop every cached entry.
180    pub fn clear(&self) {
181        let mut cache = lock(&self.cache);
182        cache.entries.clear();
183        cache.order.clear();
184    }
185}
186
187fn lock(m: &Mutex<Cache>) -> std::sync::MutexGuard<'_, Cache> {
188    m.lock().unwrap_or_else(|e| e.into_inner())
189}

191/// The input as one canonical string: object keys sorted at every level, no whitespace,
192/// so that two inputs equal as data are equal as keys whatever order a client wrote them.
193///
194/// ```
195/// use majordomus_cli::capability::executor::canonical_json;
196/// use serde_json::json;
197/// assert_eq!(canonical_json(&json!({"b": 2, "a": {"d": 1, "c": [3, {"z": 0, "y": 1}]}})),
198///            canonical_json(&json!({"a": {"c": [3, {"y": 1, "z": 0}], "d": 1}, "b": 2})));
199/// assert_ne!(canonical_json(&json!([1, 2])), canonical_json(&json!([2, 1])), "arrays are ordered");
200/// ```
201pub fn canonical_json(v: &Value) -> String {
202    fn write(v: &Value, out: &mut String) {
203        match v {
204            Value::Object(m) => {
205                let mut keys: Vec<&String> = m.keys().collect();
206                keys.sort();
207                out.push('{');
208                for (i, k) in keys.iter().enumerate() {
209                    if i > 0 {
210                        out.push(',');
211                    }
212                    out.push_str(&serde_json::to_string(k).unwrap_or_default());
213                    out.push(':');
214                    write(&m[*k], out);
215                }
216                out.push('}');
217            }
218            Value::Array(a) => {
219                out.push('[');
220                for (i, x) in a.iter().enumerate() {
221                    if i > 0 {
222                        out.push(',');
223                    }
224                    write(x, out);
225                }
226                out.push(']');
227            }
228            other => out.push_str(&other.to_string()),
229        }
230    }
231    let mut out = String::new();
232    write(v, &mut out);
233    out
234}
