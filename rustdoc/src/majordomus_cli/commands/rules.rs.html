1//! `majordomus rules`: the rule corpus against the proof there is for it.
2//!
3//! Every subcommand runs a capability of the `rules` module through the one executor, so
4//! the answer a person reads here and the answer a client reads over MCP or HTTP are the
5//! same answer rendered twice, not two derivations that happen to agree today.
6//!
7//! The text rendering is blunt about the distinctions the subsystem exists for. `proven`
8//! and `inputs unchanged` print differently, and so do `gated` and `proven`: a gate refuses
9//! violations, which is a mechanism, and a recorded pass is a verdict, and a column that
10//! showed both as a tick would be the badge whose derivation cannot be inspected. `reviewed`
11//! prints its reason, because the reason is the whole declaration.
12

22/// The exit code when `--check` finds a rule whose class the proof does not support.
23pub const EXIT_UNSUPPORTED: u8 = 10;
24
25/// Run `majordomus rules`.
26pub fn run(args: RulesArgs) -> Result<u8> {
27    let app = App::load(&args.repo)?;
28    let ctx = &app.context;
29    let stdout = std::io::stdout();
30    let mut out = stdout.lock();
31
32    match args.command {
33        RulesCommand::Report {
34            state,
35            class,
36            namespace,
37            findings,
38            check,
39        } => {
40            let mut input = json!({ "findings_only": findings });
41            for (key, value) in [
42                ("state", &state),
43                ("class", &class),
44                ("namespace", &namespace),
45            ] {
46                if let Some(v) = value {
47                    input[key] = json!(v);
48                }
49            }
50            let v = execute(ctx, &["rules", "report"], input)?;
51            match args.format {
52                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
53                OutputFormat::Text => report_text(&mut out, &v)?,
54            }
55            if check && !v["findings"].as_array().map(Vec::is_empty).unwrap_or(true) {
56                return Ok(EXIT_UNSUPPORTED);
57            }
58            Ok(0)
59        }
60
61        RulesCommand::Show { id } => {
62            let v = execute(ctx, &["rules", "show"], json!({ "rule": id }))?;
63            match args.format {
64                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
65                OutputFormat::Text => show_text(&mut out, &v)?,
66            }
67            Ok(0)
68        }
69
70        RulesCommand::Proves { id } => {
71            let v = execute(ctx, &["rules", "proves"], json!({ "test": id }))?;
72            match args.format {
73                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
74                OutputFormat::Text => proves_text(&mut out, &v)?,
75            }
76            Ok(0)
77        }
78    }
79}

83fn w(out: &mut std::io::StdoutLock<'_>, s: String) -> Result<()> {
84    writeln!(out, "{s}").map_err(Error::Transport)
85}
86
87fn s<'a>(v: &'a Value, key: &str) -> &'a str {
88    v[key].as_str().unwrap_or("?")
89}

91fn report_text(out: &mut std::io::StdoutLock<'_>, v: &Value) -> Result<()> {
92    // The verdict comes first, with its sentence: a reader who stops at the first line must
93    // not leave believing that a corpus with no findings is a corpus that was proven.
94    let verdict = s(v, "verdict");
95    let meaning = serde_json::from_value::<crate::rules::RulesVerdict>(v["verdict"].clone())
96        .map(|x| x.meaning())
97        .unwrap_or("the report carries no verdict this executable can read");
98    w(out, format!("verdict      {verdict} — {meaning}"))?;
99    w(
100        out,
101        format!(
102            "head         {} ({})",
103            short(v["head"].as_str().unwrap_or("unknown")),
104            s(v, "working_tree")
105        ),
106    )?;
107    let c = &v["coverage"];
108    w(
109        out,
110        format!(
111            "rules        {} — {} blocking, {} advisory",
112            c["rules"], c["blocking"], c["advisory"]
113        ),
114    )?;
115    // Every number below is counted apart on purpose. A single "governance is N% complete"
116    // would be the vanity metric this subsystem exists instead of.
117    for (label, key) in [
118        ("names a proof", "named_proof"),
119        ("proof in the tree", "artifacts_present"),
120        ("a runner drives", "runnable"),
121        ("a run recorded", "recorded"),
122        ("a run that passed", "passing"),
123        ("a gate runs", "gated"),
124        ("gate, no verdict", "mechanism_only"),
125        ("review-enforced", "review_only"),
126    ] {
127        w(out, format!("{label:<18} {} / {}", c[key], c["rules"]))?;
128    }
129    w(out, String::new())?;
130
131    if let Some(t) = v["states"].as_object() {
132        for (state, count) in t {
133            w(out, format!("{state:<18} {count}"))?;
134        }
135        w(out, String::new())?;
136    }
137
138    for p in v["rules"].as_array().into_iter().flatten() {
139        let r = &p["rule"];
140        w(
141            out,
142            format!(
143                "{:<18} {:<9} {:<10} {}",
144                s(p, "state"),
145                s(r, "class"),
146                r["enforcement"]["mode"].as_str().unwrap_or("?"),
147                s(r, "id")
148            ),
149        )?;
150        for t in p["tests"].as_array().into_iter().flatten() {
151            w(
152                out,
153                format!(
154                    "                   {} {}{}",
155                    t["kind"].as_str().unwrap_or("?"),
156                    s(t, "path"),
157                    if t["present"].as_bool().unwrap_or(false) {
158                        String::new()
159                    } else {
160                        "  (not in this checkout)".to_string()
161                    }
162                ),
163            )?;
164        }
165        if let Some(why) = r["enforcement"]["reviewed_because"].as_str() {
166            w(out, format!("                   because {why}"))?;
167        }
168    }
169
170    let findings = v["findings"].as_array().cloned().unwrap_or_default();
171    if !findings.is_empty() {
172        w(out, String::new())?;
173        w(
174            out,
175            format!(
176                "{} rule(s) declare a class the proof does not support:",
177                findings.len()
178            ),
179        )?;
180        for f in &findings {
181            w(out, format!("  {:<44} {}", s(f, "rule"), s(f, "reason")))?;
182            w(
183                out,
184                format!("  {:<44} reproduce: {}", "", s(f, "reproduce")),
185            )?;
186        }
187    }
188    Ok(())
189}

191fn show_text(out: &mut std::io::StdoutLock<'_>, v: &Value) -> Result<()> {
192    let p = &v["proof"];
193    let r = &p["rule"];
194    w(out, format!("rule         {}", s(r, "id")))?;
195    w(out, format!("             {}", s(r, "title")))?;
196    if let Some(st) = r["statement"].as_str() {
197        w(out, format!("             {st}"))?;
198    }
199    w(out, format!("class        {}", s(r, "class")))?;
200    w(out, format!("status       {}", s(r, "status")))?;
201    w(out, format!("source       {}", s(r, "path")))?;
202    w(
203        out,
204        format!(
205            "mode         {}",
206            r["enforcement"]["mode"].as_str().unwrap_or("?")
207        ),
208    )?;
209    w(out, format!("state        {}", s(p, "state")))?;
210    w(out, format!("             {}", s(p, "meaning")))?;
211
212    if let Some(val) = p["validator"].as_object() {
213        w(
214            out,
215            format!(
216                "validator    {}{}",
217                val["function"].as_str().unwrap_or("?"),
218                match val["defined_in"].as_str() {
219                    Some(f) => format!("  in {f}"),
220                    None => "  (nothing in lib/ defines it)".to_string(),
221                }
222            ),
223        )?;
224    }
225    if let Some(why) = r["enforcement"]["reviewed_because"].as_str() {
226        w(out, format!("because      {why}"))?;
227    }
228
229    let tests = p["tests"].as_array().cloned().unwrap_or_default();
230    if !tests.is_empty() {
231        w(out, String::new())?;
232        w(out, format!("proved by {} artifact(s):", tests.len()))?;
233        for t in &tests {
234            w(
235                out,
236                format!(
237                    "  {:<8} {:<44} {}",
238                    t["kind"].as_str().unwrap_or("?"),
239                    s(t, "path"),
240                    if t["present"].as_bool().unwrap_or(false) {
241                        s(t, "state")
242                    } else {
243                        "not in this checkout"
244                    }
245                ),
246            )?;
247            if let Some(e) = t["execution"].as_object() {
248                w(
249                    out,
250                    format!(
251                        "  {:<8} {:<44} {} at {} · {}s · {}",
252                        "",
253                        "",
254                        e["outcome"].as_str().unwrap_or("?"),
255                        short(e["commit"].as_str().unwrap_or("?")),
256                        e["seconds"],
257                        e["at"].as_str().unwrap_or("?")
258                    ),
259                )?;
260            }
261            if let Some(rp) = t["reproduce"].as_str() {
262                w(out, format!("  {:<8} {:<44} reproduce: {rp}", "", ""))?;
263            }
264        }
265    }
266
267    let gates: Vec<&str> = p["gates"]
268        .as_array()
269        .into_iter()
270        .flatten()
271        .filter_map(Value::as_str)
272        .collect();
273    if !gates.is_empty() {
274        w(out, String::new())?;
275        w(out, format!("gates        {}", gates.join(", ")))?;
276    }
277
278    for (label, key) in [("depends on", "depends_on"), ("required by", "required_by")] {
279        let list = v[key].as_array().cloned().unwrap_or_default();
280        if list.is_empty() {
281            continue;
282        }
283        w(out, String::new())?;
284        w(out, format!("{label} {} rule(s):", list.len()))?;
285        for d in &list {
286            w(out, format!("  {:<18} {}", s(d, "state"), s(d, "id")))?;
287        }
288    }
289
290    if let Some(m) = v["missing"].as_str() {
291        w(out, String::new())?;
292        w(out, format!("missing      {m}"))?;
293    }
294    Ok(())
295}

297fn proves_text(out: &mut std::io::StdoutLock<'_>, v: &Value) -> Result<()> {
298    w(
299        out,
300        format!(
301            "test         {}",
302            v["test"].as_str().unwrap_or("(no runner)")
303        ),
304    )?;
305    w(
306        out,
307        format!(
308            "source       {}{}",
309            s(v, "path"),
310            if v["present"].as_bool().unwrap_or(false) {
311                ""
312            } else {
313                "  (not in this checkout)"
314            }
315        ),
316    )?;
317    if let Some(r) = v["reproduce"].as_str() {
318        w(out, format!("reproduce    {r}"))?;
319    }
320
321    let proves = v["proves"].as_array().cloned().unwrap_or_default();
322    w(out, String::new())?;
323    w(out, format!("proves {} rule(s):", proves.len()))?;
324    for r in &proves {
325        w(
326            out,
327            format!(
328                "  {:<18} {:<9} {}",
329                s(r, "state"),
330                s(r, "class"),
331                s(r, "id")
332            ),
333        )?;
334    }
335
336    let sole: Vec<&str> = v["sole_proof_of"]
337        .as_array()
338        .into_iter()
339        .flatten()
340        .filter_map(Value::as_str)
341        .collect();
342    if !sole.is_empty() {
343        w(out, String::new())?;
344        w(
345            out,
346            format!(
347                "deleting it would leave {} rule(s) with no proof at all:",
348                sole.len()
349            ),
350        )?;
351        for r in sole {
352            w(out, format!("  {r}"))?;
353        }
354    }
355    Ok(())
356}

360/// The first eight characters of a commit, for a line a person reads. Never used as an
361/// identity: the full object name is what the payload carries.
362fn short(commit: &str) -> String {
363    commit.chars().take(8).collect()
364}
365
366fn pretty(v: &Value) -> String {
367    serde_json::to_string_pretty(v).unwrap_or_else(|_| v.to_string())
368}

370/// Run one of this module's capabilities by the command line it declares, so that the
371/// command line cannot reach a capability it does not claim.
372fn execute(ctx: &crate::capability::Context, path: &[&str], input: Value) -> Result<Value> {
373    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
374    let id = ctx
375        .registry
376        .by_cli(&words)
377        .map(|c| c.id.as_str())
378        .ok_or_else(|| Error::Protocol {
379            reason: format!(
380                "no capability is exposed as `majordomus {}`",
381                path.join(" ")
382            ),
383        })?;
384    ctx.execute(id, input).map_err(map)
385}
386
387fn map(e: CapabilityError) -> Error {
388    match e {
389        // A rule the repository does not declare is a missing thing, not a malformed
390        // request, and the exit code has to say which: a typo that read as "this rule has no
391        // proof" is the one answer this command must never give.
392        CapabilityError::NotFound(reason) => Error::NotFound { reason },
393        CapabilityError::InvalidInput(reason) | CapabilityError::Refused(reason) => {
394            Error::Protocol { reason }
395        }
396        other => Error::Protocol {
397            reason: other.to_string(),
398        },
399    }
