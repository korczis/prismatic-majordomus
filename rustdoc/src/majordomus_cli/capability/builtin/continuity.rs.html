1//! The `continuity` module: what this checkout's lifecycle is holding right now — the open
2//! episode, the record the next worker would resume from, the newest progress note, what is
3//! blocking acceptance, and whether the provider's events actually reach the tool.
4//!
5//! Everything else in this executable reads the shared half of the layer: objects that are
6//! tracked, identical in every clone, and safe to project onto a public site. This module
7//! reads the other half, `.ai/local/state/`, and the difference decides the whole design.
8//!
9//! **It is read, never written.** The lifecycle is the shell tool's; this process is one of
10//! its readers. A second writer for the same records would be a second account of events the
11//! ledger already holds, which is the thing that design refuses.
12//!
13//! **It is served, never published.** These records name this machine — a worktree path, the
14//! checkpoint that happened to be newest on this disk — and a fact about a disk is not a
15//! fact about the repository (ADR 0014). So the exposure is the loopback server and MCP,
16//! both of which serve the worker sitting in front of this checkout, and there is no
17//! projection into `docs/generated/` or `site/`. The static site documents the mechanism and
18//! carries none of its content. A generator that published a session's briefing would
19//! publish the one thing in the layer that no other clone can reproduce.
20//!
21//! **Absence is an answer.** Every field here can be legitimately empty: a fresh clone has
22//! no ledger, a worker outside a task has no task, and a branch with no prior episode has no
23//! handover. Each of those is reported as itself rather than as a failure, and the resolver
24//! below never widens its search to make one go away — a record from another worktree
25//! silently becoming your context is worse than having none, because you cannot tell it is
26//! wrong until you have acted on it.

53/// How far a record's recorded commit is from the commit this checkout is on.
54///
55/// The four words are the shell tool's, deliberately. A reader that met `advanced` from one
56/// surface and `stale` from another would have to learn the same four facts twice.
57#[derive(
58    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
59)]
60#[serde(rename_all = "snake_case")]
61pub enum Divergence {
62    /// Written at this commit. Trust it.
63    Exact,
64    /// Git has moved forward since. Trust it, and expect some of it to be done.
65    Advanced,
66    /// The recorded commit is not an ancestor: history was rewritten. Trust git, not this.
67    Diverged,
68    /// Another branch or another worktree. This record is not about your work.
69    DifferentContext,
70    /// Git could not answer, so this process says so rather than guessing `exact`.
71    Unknown,
72}

74impl Divergence {
75    /// The word as serialised.
76    ///
77    /// ```
78    /// use majordomus_cli::capability::builtin::continuity::Divergence;
79    /// assert_eq!(Divergence::Advanced.as_str(), "advanced");
80    /// ```
81    pub fn as_str(self) -> &'static str {
82        match self {
83            Divergence::Exact => "exact",
84            Divergence::Advanced => "advanced",
85            Divergence::Diverged => "diverged",
86            Divergence::DifferentContext => "different_context",
87            Divergence::Unknown => "unknown",
88        }
89    }
90
91    /// Whether a record carrying this label may be read as current knowledge. `diverged`
92    /// and `different_context` may not: the first describes a history that no longer
93    /// exists, the second describes somebody else's work.
94    ///
95    /// ```
96    /// use majordomus_cli::capability::builtin::continuity::Divergence;
97    /// assert!(Divergence::Advanced.trustworthy());
98    /// assert!(!Divergence::Diverged.trustworthy());
99    /// ```
100    pub fn trustworthy(self) -> bool {
101        matches!(self, Divergence::Exact | Divergence::Advanced)
102    }
103}

105/// How old a record is, judged against `session.freshness` in the policy.
106///
107/// This is the second of two independent judgements every record carries, and it exists
108/// because the first one cannot answer this question. [`Divergence`] says where a record's
109/// commit sits relative to HEAD; `advanced` means that commit is an ancestor of this one,
110/// which sounds like agreement and is identically true on the day the record is written and
111/// a month later. Between 2026-09-05 and 2026-09-11 this repository served a handover that
112/// was `advanced` and six days dead, and every surface that read this model — MCP, the HTTP
113/// API, the Cockpit — presented its `Next Action` as the thing to do (ADR 0052).
114///
115/// The thresholds are policy and are read, never restated here: a constant in this file
116/// would be a second source of truth for a number the policy owns, and a repository whose
117/// policy predates the key is reported as [`Freshness::Unknown`] naming the missing key
118/// rather than judged against a default nobody declared.
119///
120/// The five words divide into two groups, and the division is the one a caller acts on:
121/// `fresh` and `aging` are current and may be presented as the instruction, `stale` and
122/// `invalid` must be shown as history, and `unknown` is neither — a record nobody can date
123/// is not thereby an old record, and treating it as one would quietly discard a briefing
124/// that is perfectly good. [`Freshness::history`] is that division, and it is the only
125/// place it is decided.
126///
127/// ```
128/// use majordomus_cli::capability::builtin::continuity::Freshness;
129///
130/// // the two renderings are one value: the serialised word is the word `as_str` gives,
131/// // so a reader who met `stale` from MCP and from the command line met one fact
132/// let stale = serde_json::to_value(Freshness::Stale).unwrap();
133/// assert_eq!(stale, serde_json::json!("stale"));
134/// assert_eq!(stale.as_str(), Some(Freshness::Stale.as_str()));
135///
136/// // and the division a caller acts on: absence of a judgement is not a bad judgement
137/// assert!(Freshness::Stale.history() && Freshness::Invalid.history());
138/// assert!(!Freshness::Unknown.history(), "undated is not old");
139/// assert!(!Freshness::Fresh.history() && !Freshness::Aging.history());
140/// ```
141#[derive(
142    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
143)]
144#[serde(rename_all = "snake_case")]
145pub enum Freshness {
146    /// Inside `session.freshness.fresh_minutes`. Current; act on it.
147    Fresh,
148    /// Between the two thresholds. Still current, but its age is worth stating.
149    Aging,
150    /// At or beyond `session.freshness.stale_minutes`. History, never current.
151    Stale,
152    /// Nothing to judge it by: no timestamp, or no thresholds declared.
153    Unknown,
154    /// A timestamp that does not parse, or one in the future.
155    Invalid,
156}

195/// Seconds since the Unix epoch for an RFC 3339 instant in UTC (`2026-09-05T12:34:56Z`), or
196/// `None` when the string is not one. The inverse of [`crate::peers::rfc3339`], and written
197/// beside it for the same reason: no date crate, and the layer's timestamps are one shape.
198///
199/// ```
200/// use majordomus_cli::capability::builtin::continuity::epoch_seconds;
201/// assert_eq!(epoch_seconds("1970-01-01T00:00:00Z"), Some(0));
202/// assert_eq!(epoch_seconds("2026-08-29T10:40:00Z"), Some(1_788_000_000));
203/// assert_eq!(epoch_seconds("not a timestamp"), None);
204/// ```
205pub fn epoch_seconds(ts: &str) -> Option<i64> {
206    let b = ts.as_bytes();
207    if b.len() != 20 || b[4] != b'-' || b[7] != b'-' || b[10] != b'T' || b[19] != b'Z' {
208        return None;
209    }
210    let n = |a: usize, z: usize| ts.get(a..z)?.parse::<i64>().ok();
211    let (y, mth, d) = (n(0, 4)?, n(5, 7)?, n(8, 10)?);
212    let (h, mi, sec) = (n(11, 13)?, n(14, 16)?, n(17, 19)?);
213    if !(1..=12).contains(&mth) || !(1..=31).contains(&d) || h > 23 || mi > 59 || sec > 60 {
214        return None;
215    }
216    // days-from-civil, Howard Hinnant's algorithm: the inverse of the one in peers.rs.
217    let y = if mth <= 2 { y - 1 } else { y };
218    let era = y.div_euclid(400);
219    let yoe = y - era * 400;
220    let mp = if mth > 2 { mth - 3 } else { mth + 9 };
221    let doy = (153 * mp + 2) / 5 + d - 1;
222    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
223    let days = era * 146_097 + doe - 719_468;
224    Some(days * 86_400 + h * 3600 + mi * 60 + sec)
225}

227/// The thresholds a record is judged against, read from `session.freshness:` of the policy
228/// and never defaulted here.
229///
230/// Both fields are `Option` because the policy is allowed to be silent, and the silence is
231/// carried rather than filled in: a constant in this file would be a second source of truth
232/// for a number the policy owns. So the default value of this type is not "judge nothing
233/// old" — it is "this repository has not said", and [`Thresholds::judge`] answers
234/// [`Freshness::Unknown`] naming the missing key, which is a fact a reader can act on by
235/// declaring it.
236///
237/// ```
238/// use majordomus_cli::capability::builtin::continuity::{epoch_seconds, Freshness, Thresholds};
239///
240/// let then = epoch_seconds("2026-09-05T12:00:00Z").unwrap();
241///
242/// // undeclared: the age is still reported, and the verdict withholds itself
243/// let (verdict, age, why) = Thresholds::default().judge("2026-09-05T12:00:00Z", then + 3600);
244/// assert_eq!(verdict, Freshness::Unknown);
245/// assert_eq!(age, Some(60), "the age is measurable without a threshold to judge it by");
246/// assert!(why.contains("session.freshness"), "it names the key to declare: {why}");
247///
248/// // declared: the same record, now placed in a band
249/// let policy = Thresholds { fresh_minutes: Some(30), stale_minutes: Some(240) };
250/// let (verdict, _, _) = policy.judge("2026-09-05T12:00:00Z", then + 3600);
251/// assert_eq!(verdict, Freshness::Aging);
252/// ```
253#[derive(Debug, Clone, Copy, Default)]
254pub struct Thresholds {
255    /// `session.freshness.fresh_minutes`.
256    pub fresh_minutes: Option<i64>,
257    /// `session.freshness.stale_minutes`.
258    pub stale_minutes: Option<i64>,
259}

261impl Thresholds {
262    /// Judge one record's `created_at` against these thresholds, at `now` (Unix seconds).
263    /// Returns the verdict, the age in whole minutes when there is one, and the reason —
264    /// which is never empty, because a verdict a reader cannot act on is the failure this
265    /// whole subsystem is being corrected for.
266    ///
267    /// Both bands are closed at the bottom: a record exactly `fresh_minutes` old is already
268    /// `aging`, and one exactly `stale_minutes` old is already `stale`. `now` is a parameter
269    /// rather than a clock read inside, which is what makes the judgement a pure function of
270    /// two recorded facts and lets a test pin the instant.
271    ///
272    /// The three ways there is nothing to judge are kept apart, because they call for
273    /// different actions: no timestamp and no declared threshold are `unknown` — go and
274    /// declare one — while a timestamp that does not parse, or one in the future, is
275    /// `invalid`, which means a writer produced a record this reader should not trust.
276    ///
277    /// ```
278    /// use majordomus_cli::capability::builtin::continuity::{epoch_seconds, Freshness, Thresholds};
279    ///
280    /// let policy = Thresholds { fresh_minutes: Some(30), stale_minutes: Some(240) };
281    /// let at = "2026-09-05T12:00:00Z";
282    /// let then = epoch_seconds(at).unwrap();
283    /// let verdict = |minutes: i64| policy.judge(at, then + minutes * 60).0;
284    ///
285    /// // the bands, and the boundaries, which are closed at the bottom
286    /// assert_eq!(verdict(29), Freshness::Fresh);
287    /// assert_eq!(verdict(30), Freshness::Aging, "at the threshold it has already aged");
288    /// assert_eq!(verdict(239), Freshness::Aging);
289    /// assert_eq!(verdict(240), Freshness::Stale);
290    ///
291    /// // the age comes back measured, and the reason says it in the tool's own words
292    /// let (_, age, why) = policy.judge(at, then + 1500 * 60);
293    /// assert_eq!(age, Some(1500));
294    /// assert!(why.contains("25h"), "spans read as the shell tool writes them: {why}");
295    ///
296    /// // a record from the future is not old, it is wrong, and it says so
297    /// let (verdict, _, why) = policy.judge(at, then - 600);
298    /// assert_eq!(verdict, Freshness::Invalid);
299    /// assert!(why.contains("future"), "{why}");
300    /// assert_eq!(policy.judge("the day before", then).0, Freshness::Invalid);
301    ///
302    /// // and nothing to judge at all is `unknown`, which is not a complaint about the record
303    /// assert_eq!(policy.judge("", then).0, Freshness::Unknown);
304    /// ```
305    pub fn judge(self, created_at: &str, now: i64) -> (Freshness, Option<i64>, String) {
306        if created_at.is_empty() {
307            return (Freshness::Unknown, None, "no timestamp".into());
308        }
309        let Some(then) = epoch_seconds(created_at) else {
310            return (
311                Freshness::Invalid,
312                None,
313                format!("timestamp does not parse: {created_at}"),
314            );
315        };
316        let minutes = (now - then) / 60;
317        if minutes < 0 {
318            return (
319                Freshness::Invalid,
320                Some(minutes),
321                format!("timestamp is {} minute(s) in the future", -minutes),
322            );
323        }
324        let (Some(fresh), Some(stale)) = (self.fresh_minutes, self.stale_minutes) else {
325            return (
326                Freshness::Unknown,
327                Some(minutes),
328                "the policy declares no session.freshness thresholds".into(),
329            );
330        };
331        if minutes >= stale {
332            (
333                Freshness::Stale,
334                Some(minutes),
335                format!(
336                    "{} old, past the {} this repository calls stale",
337                    span(minutes),
338                    span(stale)
339                ),
340            )
341        } else if minutes >= fresh {
342            (
343                Freshness::Aging,
344                Some(minutes),
345                format!(
346                    "{} old, past the {} this repository calls fresh",
347                    span(minutes),
348                    span(fresh)
349                ),
350            )
351        } else {
352            (
353                Freshness::Fresh,
354                Some(minutes),
355                format!("{} old", span(minutes)),
356            )
357        }
358    }
359}

361/// Whole minutes as the shell tool renders a span: `18m`, `25h`, `6d`.
362///
363/// ```
364/// use majordomus_cli::capability::builtin::continuity::span;
365/// assert_eq!(span(18), "18m");
366/// assert_eq!(span(1500), "25h");
367/// assert_eq!(span(8640), "6d");
368/// ```
369pub fn span(minutes: i64) -> String {
370    if minutes < 90 {
371        format!("{minutes}m")
372    } else if minutes < 2880 {
373        format!("{}h", minutes / 60)
374    } else {
375        format!("{}d", minutes / 1440)
376    }
377}

503/// What the lifecycle of this checkout is holding.

508    ///

512    pub present: bool,

514    pub worktree: String,

516    pub branch: String,
517    /// The commit this checkout is on.
518    #[serde(default, skip_serializing_if = "String::is_empty")]
519    pub head: String,
520    /// `clean` or `dirty`.
521    #[serde(default, skip_serializing_if = "String::is_empty")]
522    pub working_tree: String,
523    /// The open episode, or `None`.
524    #[serde(default, skip_serializing_if = "Option::is_none")]
525    pub session: Option<OpenSession>,
526    /// The active task, or `None`.
527    #[serde(default, skip_serializing_if = "Option::is_none")]
528    pub task: Option<ActiveTask>,
529    /// The record the next worker would resume from, or `None` when nothing resolves here.
530    #[serde(default, skip_serializing_if = "Option::is_none")]
531    pub handover: Option<Record>,
532    /// The newest progress note for this worktree and branch, or `None`.
533    #[serde(default, skip_serializing_if = "Option::is_none")]
534    pub checkpoint: Option<Record>,
535    /// Unresolved questions on this branch. Every one refuses `finish --outcome completed`,
536    /// whichever task opened it.

577/// One level-one section of a Markdown record, body only, trimmed. Used to lift the part of
578/// a handover a resuming worker acts on without copying the document into every answer.
579fn section(path: &Path, want: &str) -> String {
580    let Ok(text) = std::fs::read_to_string(path) else {
581        return String::new();
582    };
583    let mut out: Vec<&str> = Vec::new();
584    let mut on = false;
585    for line in text.lines() {
586        if let Some(head) = line.strip_prefix("# ") {
587            on = head.trim_end() == want;
588            continue;
589        }
590        if on {
591            out.push(line);
592        }
593    }
594    while out.first().is_some_and(|l| l.trim().is_empty()) {
595        out.remove(0);
596    }
597    while out.last().is_some_and(|l| l.trim().is_empty()) {
598        out.pop();
599    }
600    out.join("\n")
601}

603/// Compare a record's commit with this checkout's, the same way the shell tool does: by
604/// ancestry, not by equality of timestamps. `git` being unavailable yields `Unknown`, which
605/// is the honest answer and not `Exact`.
606pub(crate) fn divergence(root: &Path, theirs: &str, ours: Option<&str>) -> Divergence {
607    let Some(ours) = ours else {
608        return Divergence::Unknown;
609    };
610    if theirs == ours {
611        return Divergence::Exact;
612    }
613    if theirs.is_empty() {
614        return Divergence::Unknown;
615    }
616    match git::is_ancestor(root, theirs, ours) {
617        Some(true) => Divergence::Advanced,
618        Some(false) => Divergence::Diverged,
619        None => Divergence::Unknown,
620    }
621}

623/// The most relevant record in `dir` for this worktree and branch.
624///
625/// Two tiers and no third, exactly as `mj_resolve_latest` has them: same worktree and
626/// branch, then same branch in another worktree of the same repository, then nothing. The
627/// ordering inside a tier is by the timestamp the record asserts — never by filesystem
628/// modification time, which does not survive a clone and is not the time the record claims.
629///
630/// Returns the record and the number of files that were skipped because they could not be
631/// read as records.
632///
633/// Shared with the preflight (`environment::preflight`), which reports the same record's
634/// freshness at entry: one resolution rule, so the briefing and the entry banner cannot
635/// name two different handovers.
636pub(crate) fn resolve_record(
637    root: &Path,
638    dir: &Path,
639    branch: &str,
640    head: Option<&str>,
641    thresholds: Thresholds,
642    now: i64,
643) -> (Option<Record>, usize) {
644    resolve(root, dir, branch, head, thresholds, now)
645}

647fn resolve(
648    root: &Path,
649    dir: &Path,
650    branch: &str,
651    head: Option<&str>,
652    thresholds: Thresholds,
653    now: i64,
654) -> (Option<Record>, usize) {
655    let Ok(entries) = std::fs::read_dir(dir) else {
656        return (None, 0);
657    };
658    let worktree = root.to_string_lossy().to_string();
659    let mut best: Option<(u8, String, Record)> = None;
660    let mut skipped = 0usize;
661
662    let mut paths: Vec<PathBuf> = entries
663        .filter_map(|e| e.ok().map(|e| e.path()))
664        .filter(|p| p.extension().is_some_and(|x| x == "md"))
665        .collect();
666    // A deterministic walk, so two runs over one directory agree about which of two
667    // records written in the same second is newer.
668    paths.sort();
669
670    for path in paths {
671        let Some(f) = front(&path) else {
672            skipped += 1;
673            continue;
674        };
675        let (Some(created), Some(rhead)) = (f.get("created_at"), f.get("head")) else {
676            skipped += 1;
677            continue;
678        };
679        if f.get("schema_version").map(String::as_str) != Some("1") {
680            skipped += 1;
681            continue;
682        }
683        let rbranch = f.get("branch").cloned().unwrap_or_default();
684        let rworktree = f.get("worktree").cloned().unwrap_or_default();
685        let tier = if rworktree == worktree && rbranch == branch {
686            0u8
687        } else if branch != "DETACHED" && rbranch == branch {
688            1u8
689        } else {
690            continue;
691        };
692        let rel = path
693            .strip_prefix(root)
694            .unwrap_or(&path)
695            .to_string_lossy()
696            .to_string();
697        let record = Record {
698            path: rel,
699            created_at: created.clone(),
700            task_id: f.get("task_id").cloned().unwrap_or_else(|| "none".into()),
701            branch: rbranch,
702            head: rhead.clone(),
703            working_tree: f.get("working_tree").cloned().unwrap_or_default(),
704            matched: if tier == 0 {
705                Match::SameWorktreeSameBranch
706            } else {
707                Match::SameBranch
708            },
709            divergence: divergence(root, rhead, head),
710            freshness: Freshness::Unknown,
711            age_minutes: None,
712            freshness_reason: String::new(),
713            next_action: section(&path, "Next Action"),
714            next_action_withheld: String::new(),
715        };
716        // Judged here rather than by each surface: MCP, the HTTP API and the Cockpit all
717        // read this one value, which is the point. A record past the stale threshold keeps
718        // its path — knowing what the last worker was doing is worth having — and loses the
719        // section a reader would otherwise act on, with the reason in its place.
720        let mut record = record;
721        let (f, age, why) = thresholds.judge(&record.created_at, now);
722        record.freshness = f;
723        record.age_minutes = age;
724        record.freshness_reason = why;
725        if f.history() && !record.next_action.is_empty() {
726            record.next_action_withheld = format!(
727                "the record is {}: {}. It is history, not an instruction; read it at the path above.",
728                f.as_str(),
729                record.freshness_reason
730            );
731            record.next_action.clear();
732        }
733        let key = created.clone();
734        let better = match &best {
735            None => true,
736            Some((btier, bkey, _)) => tier < *btier || (tier == *btier && key > *bkey),
737        };
738        if better {
739            best = Some((tier, key, record));
740        }
741    }
742    (best.map(|(_, _, r)| r), skipped)
743}

756/// The unresolved entries of the blocker store. The line format is machine-written precisely
757/// so a reader like this one can rely on it; a line that does not match is not silently
758/// dropped, it is simply not unresolved, and `doctor` is what reports a store that does not
759/// parse.
760///
761/// The HTML comment block is skipped, exactly as the shell reader skips it. The store's
762/// template documents its own line format inside a comment, so a reader that does not skip
763/// it reports a blocker that does not exist in every fresh checkout — and a phantom blocker
764/// is worse than a missed one, because it refuses work nobody can unblock.
765fn blockers(path: &Path) -> Vec<String> {
766    let Ok(text) = std::fs::read_to_string(path) else {
767        return Vec::new();
768    };
769    let mut out = Vec::new();
770    let mut in_comment = false;
771    for line in text.lines() {
772        if line.contains("<!--") {
773            in_comment = true;
774        }
775        if in_comment {
776            if line.contains("-->") {
777                in_comment = false;
778            }
779            continue;
780        }
781        if let Some(rest) = line.strip_prefix("- [unresolved] ") {
782            out.push(rest.trim().to_string());
783        }
784    }
785    out
786}

790/// The freshness thresholds this repository declares, read from the policy the manifest
791/// names.
792///
793/// Read here rather than carried on the [`Context`] because nothing else in this process
794/// needs them, and read from the policy rather than written down because a constant in this
795/// file would be the second copy of a number the policy owns — the drift `session.freshness`
796/// exists in one place to prevent. A repository whose policy predates the key, or whose
797/// policy cannot be read at all, yields no thresholds; every record is then reported as
798/// `unknown` naming the missing key, which is the honest answer and not a default.
799fn thresholds_of(ctx: &Context) -> Thresholds {
800    let root = PathBuf::from(&ctx.index.repository.root);
801    let Ok(repo) = crate::repository::Repository::open(&root) else {
802        return Thresholds::default();
803    };
804    let Ok(loaded) = crate::policy::LoadedPolicy::load(&repo) else {
805        return Thresholds::default();
806    };
807    Thresholds {
808        fresh_minutes: loaded.policy.session.freshness.fresh_minutes,
809        stale_minutes: loaded.policy.session.freshness.stale_minutes,
810    }

813fn state(ctx: &Context, _: Empty) -> Result<Continuity, CapabilityError> {
814    let root = PathBuf::from(&ctx.index.repository.root);
815    let dir = root.join(STATE_DIR);
816
817    let (branch, head, working_tree) = match &ctx.index.repository.git {
818        GitState::Available(info) => (
819            info.branch.clone().unwrap_or_else(|| "DETACHED".into()),
820            info.head.clone(),
821            info.working_tree.clone(),
822        ),
823        GitState::Unavailable { .. } => ("DETACHED".into(), None, String::new()),
824    };
825
826    let mut findings = Vec::new();
827    let mut skipped = 0usize;
828
829    // --- the open episode
830    let session = match document(&dir.join("session-current.yaml")) {
831        Some(f) => {
832            let sworktree = f.get("worktree").cloned().unwrap_or_default();
833            let foreign = !sworktree.is_empty() && sworktree != root.to_string_lossy();
834            if foreign {
835                findings.push(format!(
836                    "the open session record here belongs to {sworktree}, not this checkout; nothing about it is about your work"
837                ));
838            }
839            f.get("session_id").map(|id| OpenSession {
840                session_id: id.clone(),
841                started_at: f.get("started_at").cloned().unwrap_or_default(),
842                owner: f.get("owner").cloned().unwrap_or_default(),
843                worker: f.get("worker").cloned().unwrap_or_default(),
844                provider: f.get("provider").cloned().unwrap_or_default(),
845                provider_session: f.get("provider_session").cloned().unwrap_or_default(),
846                branch: f.get("branch").cloned().unwrap_or_default(),
847                start_head: f.get("start_head").cloned().unwrap_or_default(),
848                foreign,
849            })
850        }
851        None => None,
852    };
853
854    // --- the active task. Its `scope` is a list, so it is read from the document rather
855    // than from the scalar flattening the other fields use.
856    let task = read_task(&dir.join("current.yaml"));
857
858    // --- the two resolved records
859    // One reading of the policy's thresholds and one reading of the clock for both records,
860    // so that two records resolved in the same call cannot be judged against different
861    // instants — and so that the numbers come from the policy rather than from this file.
862    let thresholds = thresholds_of(ctx);
863    let now = std::time::SystemTime::now()
864        .duration_since(std::time::UNIX_EPOCH)
865        .map(|d| d.as_secs() as i64)
866        .unwrap_or(0);
867    let (handover, s1) = resolve(
868        &root,
869        &dir.join("handovers"),
870        &branch,
871        head.as_deref(),
872        thresholds,
873        now,
874    );
875    let (checkpoint, s2) = resolve(
876        &root,
877        &dir.join("checkpoints"),
878        &branch,
879        head.as_deref(),
880        thresholds,
881        now,
882    );
883    skipped += s1 + s2;
884    if skipped > 0 {
885        findings.push(format!(
886            "{skipped} file(s) under {STATE_DIR} could not be read as records and were skipped; run `majordomus doctor`"
887        ));
888    }
889    for (what, r) in [("handover", &handover), ("checkpoint", &checkpoint)] {
890        if let Some(r) = r {
891            if !r.divergence.trustworthy() {
892                findings.push(format!(
893                    "the resolved {what} is {}: it was written at {} and that commit is not in this history; trust git over it",
894                    r.divergence.as_str(),
895                    &r.head[..7.min(r.head.len())]
896                ));
897            }
898            // Reported separately, because it is a separate fact. A record is routinely
899            // `advanced` — its commit an ancestor of HEAD, which reads as agreement — and
900            // long dead at the same time, and it was exactly that pair, unreported, that
901            // handed six days of workers a finished instruction (ADR 0052).
902            if r.freshness.history() {
903                findings.push(format!(
904                    "the resolved {what} is {}: {}. It is context, not an instruction; what it said to do next is not offered as current.",
905                    r.freshness.as_str(),
906                    r.freshness_reason
907                ));
908            }
909        }
910    }
911
912    let blockers = blockers(&dir.join("open-questions.md"));
913    if !blockers.is_empty() {
914        findings.push(format!(
915            "{} unresolved question(s) on this branch; every one refuses `majordomus finish --outcome completed`",
916            blockers.len()
917        ));
918    }
919
920    let mut tallies = BTreeMap::new();
921    tallies.insert("handovers".to_string(), count(&dir.join("handovers")));
922    tallies.insert("checkpoints".to_string(), count(&dir.join("checkpoints")));
923    // No `sessions` tally. A closed episode is an object of the shared layer, so the index
924    // already counts it and `objects.list --kind session` already answers it; a second
925    // count here, over a directory the closed records no longer live in, is the drift this
926    // repository forbids rather than a convenience.
927    tallies.insert("blockers".to_string(), blockers.len());
928    tallies.insert(
929        "ledger_lines".to_string(),
930        std::fs::read_to_string(dir.join("ledger.jsonl"))
931            .map(|t| t.lines().filter(|l| !l.trim().is_empty()).count())
932            .unwrap_or(0),
933    );
934
935    let present = task.is_some() || session.is_some() || tallies.values().any(|n| *n > 0);
936
937    Ok(Continuity {
938        present,
939        worktree: root.to_string_lossy().to_string(),
940        branch,
941        head: head.unwrap_or_default(),
942        working_tree,
943        session,
944        task,
945        handover,
946        checkpoint,
947        blockers,
948        tallies,
949        findings,
950    })
951}

953/// The active task record. Read whole rather than through the scalar flattening, because
954/// `scope` and `requires` are lists, and a task without its scope is a task whose claim
955/// nobody can check.
956///
957/// Shared with [`super::obligations`], which expands `requires` against the evidence in the
958/// ledger. One reader for one file: a second parse of the same record is how two surfaces
959/// come to disagree about what the task said.
960pub(crate) fn read_task(path: &Path) -> Option<ActiveTask> {
961    let text = std::fs::read_to_string(path).ok()?;
962    let map = yaml::parse_mapping(&text).ok()?;
963    let s = |k: &str| map.get(k).and_then(yaml::scalar_string).unwrap_or_default();
964    let id = s("id");
965    if id.is_empty() {
966        return None;
967    }
968    let list = |k: &str| -> Vec<String> {
969        map.get(k)
970            .and_then(|v| v.as_array())
971            .map(|a| a.iter().filter_map(yaml::scalar_string).collect())
972            .unwrap_or_default()
973    };
974    let scope = list("scope");
975    let requires = list("requires");
976    Some(ActiveTask {
977        id,
978        task: s("task"),
979        profile: s("profile"),
980        outcome: s("outcome"),
981        scope,
982        requires,
983        started_at: s("started_at"),
984        head: s("head"),
985    })
986}
