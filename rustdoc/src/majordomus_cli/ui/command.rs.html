1//! Command: everything a person or an agent can run, and what this process is running.
2//! Read from `commands.list` (the one catalogue of the executable's and the shell tool's
3//! commands, with each one's effect) and `executions.list`.
4//!
5//! The catalogue is filtered by the capability's own input (`?origin=`), never by this
6//! screen: the tabs are links to the same route with the filter the capability declares.
7
8use crate::capability::Context;
9use crate::cockpit::html::el;
10
11use super::dashboard::ROWS;
12use super::shell::View;
13use super::source::{ask_all, ask_input, Source};
14use super::view::{hero, kpi, list, panel, state, table, text, unanswered, wide};
15
16/// The route of the screen.
17pub const ROUTE: &str = "/ui/command";
18
19const EXECUTIONS: Source = Source::of("executions.list");
20
21/// The origins `commands.list` filters by, as its input declares them.
22const ORIGINS: &[(&str, &str)] = &[
23    ("", "All"),
24    ("executable", "Executable"),
25    ("shell", "Shell tool"),
26];
27
28/// The Command screen; `query` may carry the catalogue's `origin` filter.
29pub fn screen(ctx: &Context, query: &[(String, String)]) -> View {
30    let origin = query
31        .iter()
32        .find(|(k, _)| k == "origin")
33        .map(|(_, v)| v.as_str())
34        .filter(|o| ORIGINS.iter().any(|(id, _)| id == o && !id.is_empty()));
35    let input = match origin {
36        Some(o) => serde_json::json!({ "origin": o }),
37        None => serde_json::json!({}),
38    };
39    let commands = ask_input(ctx, "commands.list", input);
40    let answers = ask_all(ctx, &[EXECUTIONS]);
41    let executions = &answers[&EXECUTIONS];
42
43    let top = hero(
44        "terminal",
45        "violet",
46        "Command",
47        "Everything that can be run, and what is running.",
48        "Every command of the executable and the shell tool, with what it changes, and the executions this process is running or remembers.",
49        vec![
50            kpi("Commands", &commands.read("/total")),
51            kpi("Executions", &executions.read("/count")),
52            kpi("Active", &executions.read("/active")),
53            kpi("Queued", &executions.read("/queued")),
54            kpi("Live channels", &executions.read("/live_channels")),
55        ],
56    );
57
58    let tabs = el("nav")
59        .class("mj-ui-tabs")
60        .attr("aria-label", "Which commands")
61        .children(ORIGINS.iter().map(|(id, label)| {
62            let current = origin.unwrap_or("") == *id;
63            let href = if id.is_empty() {
64                ROUTE.to_string()
65            } else {
66                format!("{ROUTE}?origin={id}")
67            };
68            el("a")
69                .class("mj-ui-tab")
70                .attr("href", href)
71                .attr_if("aria-current", current.then_some("page"))
72                .text(*label)
73        }));
74
75    let catalogue = unanswered(&commands).unwrap_or_else(|| {
76        let rows = list(&commands, "/commands")
77            .iter()
78            .map(|c| {
79                el("tr")
80                    .child(
81                        el("td").child(el("code").class("mj-ui-mono").text(text(c, "invocation"))),
82                    )
83                    .child(el("td").class("mj-ui-dim").text(text(c, "summary")))
84                    .child(el("td").child(state(c, "effect")))
85            })
86            .collect::<Vec<_>>();
87        table(
88            &["Command", "What it does", "Effect"],
89            rows,
90            "No command matches.",
91        )
92    });
93
94    let running = unanswered(executions).unwrap_or_else(|| {
95        let rows = list(executions, "/executions")
96            .iter()
97            .take(ROWS * 2)
98            .map(|e| {
99                el("tr")
100                    .child(el("td").class("mj-ui-mono").text(text(e, "id")))
101                    .child(el("td").class("mj-ui-mono").text(text(e, "capability")))
102                    .child(el("td").child(state(e, "state")))
103            })
104            .collect::<Vec<_>>();
105        table(
106            &["Execution", "Capability", "State"],
107            rows,
108            "This process is running no execution and remembers none.",
109        )
110    });
111
112    let grid = el("div")
113        .class("mj-ui-grid")
114        .child(wide(panel(
115            "Commands",
116            "Each with the line a person types and what it changes when it runs.",
117            el("div").class("mj-ui-stack").child(tabs).child(catalogue),
118            &[&commands],
119        )))
120        .child(wide(panel(
121            "Executions",
122            "What this process is running, or ran and remembers, through its execution engine.",
123            running,
124            &[executions],
125        )));
126
127    View::new(
128        "Command",
129        el("div").class("mj-ui-stack").child(top).child(grid),
130    )
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136    use crate::synthetic::SyntheticRepository;
137
138    /// The origin filter is the capability's own input: the tab it names is current, and an
139    /// origin the capability does not declare is ignored rather than passed on.
140    #[test]
141    fn the_origin_filter_marks_its_tab_and_an_unknown_one_is_ignored() {
142        let repo = SyntheticRepository::small().expect("a synthetic repository");
143        let ctx = repo.context().expect("a context");
144        let html = screen(&ctx, &[("origin".into(), "executable".into())])
145            .main
146            .render();
147        assert!(
148            html.contains(&format!(
149                "href=\"{ROUTE}?origin=executable\" aria-current=\"page\""
150            )),
151            "{html}"
152        );
153        let html = screen(&ctx, &[("origin".into(), "nonsense".into())])
154            .main
155            .render();
156        assert!(
157            html.contains(&format!("href=\"{ROUTE}\" aria-current=\"page\"")),
158            "{html}"
159        );
160    }
161
162    /// An execution this process started is listed with its capability and its state.
163    #[test]
164    fn an_execution_this_process_started_is_listed() {
165        let repo = SyntheticRepository::small().expect("a synthetic repository");
166        let ctx = repo.context().expect("a context");
167        let started = ctx
168            .execute(
169                "executions.start",
170                serde_json::json!({
171                    "capability": "executions.demonstrate",
172                    "input": { "steps": 1, "delay_ms": 0 }
173                }),
174            )
175            .expect("an execution starts");
176        let id = started["id"].as_str().expect("its id").to_string();
177        let html = screen(&ctx, &[]).main.render();
178        assert!(html.contains(&id), "{html}");
179        assert!(html.contains("executions.demonstrate"), "{html}");
180        assert!(
181            !html.contains("This process is running no execution"),
182            "{html}"
183        );
184    }
