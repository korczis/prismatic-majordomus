1//! One intent in full, for the Roadmap's inspector (`/ui/roadmap?intent=intent-bound-work`):
2//! what it says, the invariants it holds the work to, the milestones and issues that serve
3//! it, and each criterion of its satisfaction with the evidence that would prove it and
4//! whether it is met.
5//!
6//! Read from `intents.list` and `plan.issues`, both answering in milliseconds. Every milestone
7//! leads to its inspector and every issue to its own: from what the work is for to the work.
8

17/// An intent id, as the repository writes them (`intent-bound-work`): letters, digits, `-`
18/// and `_`, and nothing a link could carry anything else in.
19pub fn is_intent_id(id: &str) -> bool {
20    !id.is_empty()
21        && id.len() <= 96
22        && id
23            .bytes()
24            .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_')
25}
26
27/// The intent a `serves` entry names: an entry is the intent (`intent-planning`) or one of
28/// its criteria (`intent-planning#criteria-are-covered`).
29pub fn intent_of(serves: &str) -> &str {
30    serves.split('#').next().unwrap_or(serves)
31}

33/// The intent a query names, when it names one.
34pub fn picked(query: &[(String, String)]) -> Option<String> {
35    query
36        .iter()
37        .find(|(k, _)| k == "intent")
38        .map(|(_, v)| v.trim().to_string())
39        .filter(|v| is_intent_id(v))
40}

42/// Where an intent's inspector is.
43pub fn href(id: &str) -> String {
44    format!("{}?intent={id}#intent", super::roadmap::ROUTE)
45}

47fn chip(href: String, label: String) -> El {
48    el("a")
49        .class("mj-ui-chip mj-ui-chip--link")
50        .attr("href", href)
51        .child(el("code").class("mj-ui-chip-value").text(label))
52}
53
54fn strings(v: &Value, k: &str) -> Vec<String> {
55    v.get(k)
56        .and_then(Value::as_array)
57        .map(|a| {
58            a.iter()
59                .filter_map(Value::as_str)
60                .map(str::to_string)
61                .collect()
62        })
63        .unwrap_or_default()
64}

66/// The verdict of an intent as its badge: `intents.list` answers the verdict as an object
67/// whose `state` is the word, which a reader of a plain string would draw as `unknown`.
68pub fn verdict(i: &Value) -> El {
69    match i.get("verdict") {
70        Some(v @ Value::Object(_)) => state(v, "state"),
71        _ => state(i, "verdict"),
72    }
73}

75/// The inspector of intent `id`, closed to the Roadmap.
76pub fn inspector(intents: &Answer, issues: &Answer, id: &str) -> El {
77    inspector_closing_to(intents, issues, id, super::roadmap::ROUTE)
78}

80/// The inspector of intent `id`, whose "Close" leads to `close`: the screen it is open on.
81pub fn inspector_closing_to(intents: &Answer, issues: &Answer, id: &str, close: &str) -> El {
82    let body = unanswered(intents)
83        .or_else(|| unanswered(issues))
84        .unwrap_or_else(|| {
85            let Some(i) = list(intents, "/intents")
86                .into_iter()
87                .find(|i| text(i, "id") == id)
88            else {
89                return nothing(&format!("{id} is not an intent of this repository."));
90            };
91            let milestones = i
92                .get("milestones")
93                .and_then(Value::as_array)
94                .cloned()
95                .unwrap_or_default();
96            let serving: Vec<Value> = list(issues, "/issues")
97                .into_iter()
98                .filter(|x| strings(x, "serves").iter().any(|s| intent_of(s) == id))
99                .collect();
100            let criteria = i
101                .get("satisfaction")
102                .and_then(Value::as_array)
103                .cloned()
104                .unwrap_or_default();
105            let met = criteria
106                .iter()
107                .filter(|c| c.get("met").and_then(Value::as_bool) == Some(true))
108                .count();
109            let rows = criteria
110                .iter()
111                .map(|c| {
112                    let reproduce = text(c, "reproduce");
113                    el("tr")
114                        .child(
115                            el("td")
116                                .child(el("span").class("mj-ui-mono mj-ui-dim").text(text(c, "id")))
117                                .child(
118                                    el("span")
119                                        .class("mj-ui-cell-title")
120                                        .text(text(c, "criterion")),
121                                ),
122                        )
123                        .child(el("td").class("mj-ui-mono").text(text(c, "evidence")))
124                        .child(el("td").child(state(c, "state")))
125                        .child(el("td").child(word_badge(
126                            if c.get("met").and_then(Value::as_bool) == Some(true) {
127                                "met"
128                            } else {
129                                "unmet"
130                            },
131                        )))
132                        .child(el("td").child(if reproduce.is_empty() {
133                            el("span").class("mj-ui-dim").text("none")
134                        } else {
135                            el("code").class("mj-ui-mono").text(reproduce)
136                        }))
137                })
138                .collect::<Vec<_>>();
139            el("section")
140                .class("mj-ui-detail")
141                .attr("id", "intent")
142                .attr("aria-label", format!("Intent {id}"))
143                .child(
144                    el("div")
145                        .class("mj-ui-detail-head")
146                        .child(el("p").class("mj-ui-mono mj-ui-dim").text(id.to_string()))
147                        .child(el("h3").class("mj-ui-detail-title").text(text(&i, "title")))
148                        .child(
149                            el("p")
150                                .class("mj-ui-badges")
151                                .child(state(&i, "stage"))
152                                .child(verdict(&i))
153                                .child(
154                                    el("span")
155                                        .class("mj-ui-mono mj-ui-dim")
156                                        .text(format!("{met} of {} criteria met", criteria.len())),
157                                )
158                                .child(
159                                    el("a")
160                                        .class("mj-ui-more")
161                                        .attr("href", close)
162                                        .text("Close"),
163                                ),
164                        ),
165                )
166                .child(el("p").class("mj-ui-lead").text(text(&i, "statement")))
167                .child(
168                    el("h4")
169                        .class("mj-ui-panel-title")
170                        .text("What it holds the work to"),
171                )
172                .child({
173                    let inv = strings(&i, "invariants");
174                    if inv.is_empty() {
175                        nothing("It declares no invariant.")
176                    } else {
177                        el("ul")
178                            .class("mj-ui-prose-list")
179                            .children(inv.into_iter().map(|t| el("li").text(t)))
180                    }
181                })
182                .child(
183                    el("h4")
184                        .class("mj-ui-panel-title")
185                        .text("Milestones that serve it"),
186                )
187                .child(if milestones.is_empty() {
188                    nothing("No milestone serves it yet.")
189                } else {
190                    el("p")
191                        .class("mj-ui-badges")
192                        .children(milestones.iter().map(|m| {
193                            let mid = text(m, "id");
194                            el("span")
195                                .child(chip(super::milestone::href(&mid), mid))
196                                .child(state(m, "status"))
197                        }))
198                })
199                .child(
200                    el("h4")
201                        .class("mj-ui-panel-title")
202                        .text("Issues that serve it"),
203                )
204                .child(if serving.is_empty() {
205                    nothing("No issue of the plan names it.")
206                } else {
207                    el("p")
208                        .class("mj-ui-badges")
209                        .children(serving.iter().map(|x| {
210                            let iid = text(x, "id");
211                            chip(format!("{}?issue={iid}#issue", super::prs::ROUTE), iid)
212                        }))
213                })
214                .child(
215                    el("h4")
216                        .class("mj-ui-panel-title")
217                        .text("When it is satisfied"),
218                )
219                .child(table(
220                    &["Criterion", "Evidence", "State", "Met", "Reproduce"],
221                    rows,
222                    "It declares no criterion.",
223                ))
224        });
225    panel(
226        "Intent",
227        "What the work is for, what it holds the work to, and whether the evidence says it is met.",
228        body,
229        &[intents, issues],
230    )
231}

239    fn answer(capability: &'static str, value: Value) -> Answer {
240        Answer {
241            source: Source::of(capability),
242            route: None,
243            exposure: Exposure::default(),
244            value: Ok(value.into()),
245            took: Duration::from_millis(1),
246            standing: crate::ui::source::Standing::NOW,
247        }
248    }
249
250    fn data() -> (Answer, Answer) {
251        (
252            answer(
253                "intents.list",
254                serde_json::json!({ "intents": [{
255                    "id": "intent-a", "title": "A is done", "statement": "Work is bound.", "stage": "planned",
256                    "verdict": "unmet", "invariants": ["one rule"],
257                    "milestones": [{ "id": "M7", "status": "BLOCKED" }],
258                    "satisfaction": [
259                        { "id": "c1", "criterion": "it binds", "evidence": "test", "state": "not_run", "met": false, "reproduce": "cargo test --test x" },
260                        { "id": "c2", "criterion": "it says why", "evidence": "doc", "state": "passed", "met": true, "reproduce": "" }
261                    ]
262                }]}),
263            ),
264            answer(
265                "plan.issues",
266                serde_json::json!({ "issues": [
267                { "id": "I0009", "serves": ["intent-a#c1"] }, { "id": "I0010", "serves": ["other"] } ]}),
268            ),
269        )
270    }
271
272    #[test]
273    fn an_intent_shows_its_criteria_and_leads_to_what_serves_it() {
274        let (intents, issues) = data();
275        let html = inspector(&intents, &issues, "intent-a").render();
276        for want in [
277            "id=\"intent\"",
278            "1 of 2 criteria met",
279            "cargo test --test x",
280            "/ui/roadmap?milestone=M7#milestone",
281            "/ui/prs?issue=I0009#issue",
282            "one rule",
283        ] {
284            assert!(html.contains(want), "no {want}: {html}");
285        }
286        assert!(
287            !html.contains("I0010"),
288            "an issue serving another intent is not its own"
289        );
290    }
291
292    /// An intent that declares no invariant and that no milestone or issue serves yet says
293    /// each of those.
294    #[test]
295    fn an_intent_nothing_serves_yet_says_so() {
296        let intents = answer(
297            "intents.list",
298            serde_json::json!({ "intents": [{ "id": "intent-b", "title": "B", "statement": "B holds." }] }),
299        );
300        let issues = answer("plan.issues", serde_json::json!({ "issues": [] }));
301        let html = inspector(&intents, &issues, "intent-b").render();
302        for want in [
303            "It declares no invariant.",
304            "No milestone serves it yet.",
305            "No issue of the plan names it.",
306            "It declares no criterion.",
307        ] {
308            assert!(html.contains(want), "no {want}: {html}");
309        }
310    }
311
312    #[test]
313    fn an_unknown_intent_says_so_and_a_hostile_one_is_never_picked() {
314        let (intents, issues) = data();
315        assert!(inspector(&intents, &issues, "nope")
316            .render()
317            .contains("nope is not an intent"));
318        let q = |v: &str| vec![("intent".to_string(), v.to_string())];
319        assert_eq!(picked(&q("intent-a")), Some("intent-a".into()));
320        assert_eq!(picked(&q("a\" onclick=\"x")), None);
321    }
