1//! The shell every Cockpit page is rendered into, and the small vocabulary of components
2//! the pages are built from.
3//!
4//! The pages carry no raw utility classes. Every class here is a semantic one the
5//! Cockpit's stylesheet defines (`mj-card`, `mj-table`, `mj-badge--fail`), so the design
6//! lives in one file a designer can read and the Rust says what a thing *is* rather than
7//! how it looks. That is also what makes the stylesheet deterministic: it is a function of
8//! its own source and nothing else, and the drift check over it is exact.
9//!
10//! Every `x-` attribute here names a *top-level* property or method of the one Alpine
11//! component and never an expression — not even a dotted path, which the CSP build reads
12//! as an expression and does not evaluate. A palette bound to `palette.open` looks right,
13//! renders right, and leaves a full-page modal backdrop over every click. That is what the content-security policy requires: the Cockpit
14//! ships Alpine's CSP build, which evaluates no strings, so the policy needs neither
15//! `unsafe-eval` nor `unsafe-inline`. A handler that would take an argument gets its own
16//! method in `cockpit.js` instead.
17

28const NO_ASSETS: &str = "This distribution has no share/cockpit/ directory, so the Cockpit is serving unstyled markup. Everything works; nothing is styled. Run `just cockpit-assets` to build it.";

30/// What a page needs from the process in order to render its shell.
31pub struct Shell<'a> {
32    /// The page title, before the suffix.
33    pub title: &'a str,
34    /// One line under the heading: what this page shows.
35    pub subtitle: Option<String>,
36    /// The area of the navigation this page belongs to.
37    pub area: Area,
38    /// The trail from the Cockpit root to this page, label and href; the last has no href.
39    pub breadcrumbs: Vec<(String, Option<String>)>,
40    /// The navigation, derived from the registry.
41    pub navigation: &'a Navigation,
42    /// The stylesheet URL, with its digest.
43    pub stylesheet: String,
44    /// The Cockpit's own script modules, in load order, with their digests.
45    pub scripts: Vec<String>,
46    /// Whether the distribution has an asset directory at all.
47    pub assets_present: bool,
48    /// The executable's version, shown in the footer.
49    pub version: &'a str,
50}

52/// Render one page into the shell.
53pub fn page(shell: &Shell<'_>, main: El) -> String {
54    let head = vec![
55        el("link")
56            .attr("rel", "stylesheet")
57            .attr("href", shell.stylesheet.clone()),
58        el("link")
59            .attr("rel", "icon")
60            .attr("href", "/cockpit/assets/favicon.svg"),
61        // the theme is applied before first paint, so a dark-mode reader never sees a
62        // white flash; this is the one inline script the Cockpit ships, and the
63        // content-security policy names its digest rather than allowing inline scripts
64        el("script").raw(theme_bootstrap()),
65    ]
66    .into_iter()
67    .chain(shell.scripts.iter().map(|src| {
68        el("script")
69            .attr("type", "module")
70            .attr("src", src.clone())
71            .flag("defer")
72    }))
73    .collect();
74
75    // the theme contract and the design fingerprint, from the declaration compiled into
76    // this executable: the script reads the key and the class here rather than naming
77    // them, and compares the fingerprint with the one the stylesheet carries
78    let design = DesignSystem::compiled().ok();
79    let body = el("body")
80        .class("mj-body")
81        .attr("x-data", "cockpit")
82        .attr_if(
83            "data-theme-key",
84            design.map(|d| d.theme.storage_key.clone()),
85        )
86        .attr_if("data-theme-class", design.map(|d| d.theme.class.clone()))
87        .attr_if("data-design", design.map(|d| d.short_fingerprint()))
88        .child(skip_link())
89        .child(header(shell))
90        .child(
91            el("div").class("mj-layout").child(sidebar(shell)).child(
92                el("main")
93                    .attr("id", "main")
94                    .class("mj-main")
95                    .when(!shell.assets_present, |m| m.child(alert("warn", NO_ASSETS)))
96                    .child(breadcrumbs(&shell.breadcrumbs))
97                    .child(
98                        el("header")
99                            .class("mj-page-head")
100                            .child(el("h1").class("mj-page-title").text(shell.title))
101                            .node(match &shell.subtitle {
102                                Some(s) => Node::Element(el("p").class("mj-page-subtitle").text(s)),
103                                None => empty(),
104                            }),
105                    )
106                    .child(main),
107            ),
108        )
109        .child(palette())
110        .child(footer(shell));
111
112    super::html::document(&format!("{} · Majordomus Cockpit", shell.title), head, body)
113}

115/// Read the stored theme before the page paints: the one statement the design declaration
116/// generates for every surface, so the site and the Cockpit honour one storage key and one
117/// class. One statement, so that its digest is stable and the content-security policy can
118/// name it. An executable whose compiled declaration is invalid ships no statement and
119/// follows the system through the stylesheet alone; the design capability says why.
120pub fn theme_bootstrap() -> &'static str {
121    static BOOTSTRAP: LazyLock<String> = LazyLock::new(|| {
122        DesignSystem::compiled()
123            .map(|d| d.theme_bootstrap())
124            .unwrap_or_default()
125    });
126    BOOTSTRAP.as_str()
127}

185fn sidebar(shell: &Shell<'_>) -> El {
186    let mut nav = el("nav")
187        .class("mj-sidebar")
188        .attr("aria-label", "Cockpit sections");
189    for section in shell.navigation.sections() {
190        let mut list = el("ul").class("mj-nav-list");
191        // A grouped section shows its headings as the group changes. The items arrive in
192        // the canonical order, which puts a group's members together and the ungrouped
193        // ones last, so one pass is enough and nothing here decides the sequence.
194        let mut group: Option<&str> = None;
195        for item in &section.items {
196            if item.group.as_deref() != group {
197                group = item.group.as_deref();
198                if let Some(heading) = group {
199                    list = list.child(
200                        el("li")
201                            .class("mj-nav-group")
202                            .attr("role", "presentation")
203                            .text(heading),
204                    );
205                }
206            }
207            let current = item.area == shell.area && item.current;
208            list = list.child(
209                el("li").child(
210                    el("a")
211                        .class(if current {
212                            "mj-nav-link mj-nav-link--current"
213                        } else {
214                            "mj-nav-link"
215                        })
216                        .attr("href", item.href.clone())
217                        .attr_if("aria-current", current.then_some("page"))
218                        .child(el("span").class("mj-nav-label").text(&item.label))
219                        .node(match item.count {
220                            Some(n) => {
221                                Node::Element(el("span").class("mj-nav-count").text(n.to_string()))
222                            }
223                            None => empty(),
224                        }),
225                ),
226            );
227        }
228        // a catalogue long enough to bury the sections under it folds away, and unfolds
229        // itself when what the reader is looking at is inside it
230        let holds_current = section
231            .items
232            .iter()
233            .any(|i| i.area == shell.area && i.current);
234        nav = nav.child(if section.items.len() > FOLD_ABOVE {
235            el("details")
236                .class("mj-nav-section mj-nav-section--foldable")
237                .when(holds_current, |d| d.flag("open"))
238                .child(
239                    el("summary")
240                        .class("mj-nav-heading mj-nav-summary")
241                        .child(el("span").text(&section.title))
242                        .child(
243                            el("span")
244                                .class("mj-nav-count")
245                                .text(section.items.len().to_string()),
246                        ),
247                )
248                .child(list)
249        } else {
250            el("div")
251                .class("mj-nav-section")
252                .child(el("h2").class("mj-nav-heading").text(&section.title))
253                .child(list)
254        });
255    }
256    nav
257}

275fn breadcrumbs(trail: &[(String, Option<String>)]) -> El {
276    if trail.is_empty() {
277        return el("div").class("mj-hidden");
278    }
279    let mut list = el("ol").class("mj-crumbs");
280    for (label, href) in trail {
281        list = list.child(el("li").child(match href {
282            Some(h) => el("a").attr("href", h.clone()).text(label),
283            None => el("span").attr("aria-current", "page").text(label),
284        }));
285    }
286    el("nav").attr("aria-label", "Breadcrumb").child(list)
287}

420/// A badge for a status word that arrived as *data* — an object's own `status` field, a
421/// graph node's — rather than one this code chose. Filed under a status by the design
422/// declaration, it is a badge and carries that status's colour; filed under nothing, it is
423/// a plain tag, so that an arbitrary word never wears a meaning nobody gave it and the
424/// browser probe's vocabulary check stays exact. The words this code chooses itself go
425/// through [`badge`], where an unfiled word is a defect the probe reports.
426pub fn word_badge(word: &str) -> El {
427    let filed = DesignSystem::compiled()
428        .ok()
429        .is_some_and(|d| d.role_of_state(&css_word(word)).is_some());
430    if filed {
431        badge(word, word)
432    } else {
433        tag(word)
434    }
435}

504/// How many rows a listing shows at once before it is paged. A control plane is read a
505/// screenful at a time; a page that renders every one of nine hundred rows is a file
506/// dump with a header on it.
507pub const PER_PAGE: usize = 50;
508
509/// The window a listing is showing of its result set. The page number is a view
510/// parameter and lives in the URL, so a window is a deep link like every other state
511/// the Cockpit holds.
512#[derive(Debug, Clone, Copy, PartialEq, Eq)]
513pub struct Window {
514    /// The page being shown, 1-based and never past the last one.
515    pub page: usize,
516    /// How many rows a page holds.
517    pub per_page: usize,
518    /// How many rows matched, before the window was taken.
519    pub total: usize,
520}
521
522impl Window {
523    /// The window a request asks for, clamped to what the result set has: page 0 and page
524    /// 900 of a 3-page listing are both a page that exists.
525    pub fn new(page: usize, per_page: usize, total: usize) -> Self {
526        let per_page = per_page.max(1);
527        let pages = total.div_ceil(per_page).max(1);
528        Self {
529            page: page.clamp(1, pages),
530            per_page,
531            total,
532        }
533    }
534
535    /// How many pages the result set has; never zero, because an empty listing still has
536    /// a page to show its empty state on.
537    pub fn pages(&self) -> usize {
538        self.total.div_ceil(self.per_page).max(1)
539    }
540
541    /// The half-open range of rows this window covers, for slicing the matches.
542    pub fn range(&self) -> std::ops::Range<usize> {
543        let start = (self.page - 1) * self.per_page;
544        start..(start + self.per_page).min(self.total)
545    }
546}

548/// The page numbers a pagination control offers: the first, the last, the neighbours of
549/// the current one, and a gap marker for what is skipped. Nine hundred pages must not
550/// render nine hundred links.
551fn page_numbers(current: usize, pages: usize) -> Vec<Option<usize>> {
552    let mut wanted: Vec<usize> = [1, pages]
553        .into_iter()
554        .chain(current.saturating_sub(1).max(1)..=(current + 1).min(pages))
555        .filter(|n| *n >= 1 && *n <= pages)
556        .collect();
557    wanted.sort_unstable();
558    wanted.dedup();
559
560    let mut out = Vec::with_capacity(wanted.len() + 2);
561    let mut previous = 0usize;
562    for n in wanted {
563        if previous != 0 && n > previous + 1 {
564            out.push(None);
565        }
566        out.push(Some(n));
567        previous = n;
568    }
569    out
570}

572/// Pagination over a listing: what is being shown, and the way to the rest of it.
573///
574/// `href` builds the URL of one page — the caller owns the listing's filters and puts
575/// them back into every link, so paging never drops a filter and a link can be sent to
576/// somebody as it is.
577pub fn pagination(w: Window, href: impl Fn(usize) -> String) -> El {
578    if w.total == 0 {
579        return el("div").class("mj-hidden");
580    }
581    let range = w.range();
582    let summary = el("p").class("mj-pagination-summary").text(format!(
583        "{}–{} of {}",
584        range.start + 1,
585        range.end,
586        w.total
587    ));
588    if w.pages() == 1 {
589        return el("nav").class("mj-pagination").child(summary);
590    }
591
592    let step = |label: &str, to: Option<usize>| {
593        let li = el("li");
594        match to {
595            Some(n) => li.child(
596                el("a")
597                    .class("mj-page-link")
598                    .attr("href", href(n))
599                    .attr("rel", if n < w.page { "prev" } else { "next" })
600                    .text(label),
601            ),
602            None => li.child(
603                el("span")
604                    .class("mj-page-link mj-page-link--disabled")
605                    .attr("aria-disabled", "true")
606                    .text(label),
607            ),
608        }
609    };
610
611    let mut list = el("ul")
612        .class("mj-pagination-list")
613        .child(step("Previous", (w.page > 1).then(|| w.page - 1)));
614    for slot in page_numbers(w.page, w.pages()) {
615        list = list.child(match slot {
616            Some(n) if n == w.page => el("li").child(
617                el("span")
618                    .class("mj-page-link mj-page-link--current")
619                    .attr("aria-current", "page")
620                    .text(n.to_string()),
621            ),
622            Some(n) => el("li").child(
623                el("a")
624                    .class("mj-page-link")
625                    .attr("href", href(n))
626                    .attr("aria-label", format!("Page {n}"))
627                    .text(n.to_string()),
628            ),
629            None => el("li").child(
630                el("span")
631                    .class("mj-page-gap")
632                    .attr("aria-hidden", "true")
633                    .text("…"),
634            ),
635        });
636    }
637    list = list.child(step("Next", (w.page < w.pages()).then(|| w.page + 1)));
638
639    el("nav")
640        .class("mj-pagination")
641        .attr("aria-label", "Pagination")
642        .child(summary)
643        .child(list)
644}

646/// A row of filters a reader browses by: a label, where it goes, how many are behind it,
647/// and whether it is the one in force. It is the same catalogue the sidebar shows, put
648/// where the listing is, so a set of nine hundred rows is entered by its parts rather
649/// than scrolled.
650pub fn chips(items: Vec<(String, String, usize, bool)>) -> El {
651    if items.is_empty() {
652        return el("div").class("mj-hidden");
653    }
654    let mut list = el("ul").class("mj-chips");
655    for (label, href, count, current) in items {
656        list = list.child(
657            el("li").child(
658                el("a")
659                    .class(if current {
660                        "mj-chip mj-chip--current"
661                    } else {
662                        "mj-chip"
663                    })
664                    .attr("href", href)
665                    .attr_if("aria-current", current.then_some("true"))
666                    .child(el("span").class("mj-chip-label").text(label))
667                    .child(el("span").class("mj-chip-count").text(count.to_string())),
668            ),
669        );
670    }
671    list
672}

675/// of it in the tooltip. An identifier broken across three lines mid-word is what makes

677pub fn id_cell(href: impl Into<String>, id: impl Into<String>) -> El {

679    el("td").class("mj-cell-id").child(

684            .text(id),
685    )
686}

688/// A class-safe form of a status word: lowercase ASCII letters, digits and dashes, so a
689/// word from data can never build a class name that is not one.
690fn css_word(word: &str) -> String {
691    let cleaned: String = word
692        .chars()
693        .map(|c| {
694            if c.is_ascii_alphanumeric() {
695                c.to_ascii_lowercase()
696            } else {
697                '-'
698            }
699        })
700        .collect();
701    if cleaned.is_empty() {
702        "unknown".into()
703    } else {
704        cleaned
705    }
706}
