1//! One milestone in full, for the Roadmap's inspector (`/ui/roadmap?milestone=M004`): what it
2//! holds and where each issue stands, what can start now, the longest chain of open work
3//! through it, the intents its issues serve, and the milestones it waits for and unblocks.
4//!
5//! Read from `plan.roadmap` and `plan.issues`, both of which answer in milliseconds, so the
6//! inspector is part of the first paint. Every issue is a link to its own inspector on the PR
7//! Drain, and every milestone a link to its own here: a reader walks the plan by clicking it.
8

18/// A plan milestone id: letters, digits, `-` and `_`, as the plan's own ids are (`M004`,
19/// `development-runtime`), and nothing a link could carry anything else in.
20///
21/// ```text
22/// assert!(is_milestone_id("M004") && is_milestone_id("development-runtime"));
23/// assert!(!is_milestone_id("M0\"4") && !is_milestone_id(""));
24/// ```
25pub fn is_milestone_id(id: &str) -> bool {
26    !id.is_empty()
27        && id.len() <= 64
28        && id
29            .bytes()
30            .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_')
31}

33/// The milestone a query names, when it names one.
34pub fn picked(query: &[(String, String)]) -> Option<String> {
35    query
36        .iter()
37        .find(|(k, _)| k == "milestone")
38        .map(|(_, v)| v.trim().to_string())
39        .filter(|v| is_milestone_id(v))
40}

51fn strings(v: &Value, k: &str) -> Vec<String> {
52    v.get(k)
53        .and_then(Value::as_array)
54        .map(|a| {
55            a.iter()
56                .filter_map(Value::as_str)
57                .map(str::to_string)
58                .collect()
59        })
60        .unwrap_or_default()
61}

63fn closed(status: &str) -> bool {
64    matches!(status.to_ascii_uppercase().as_str(), "DONE" | "CANCELLED")
65}

67/// The longest chain of open issues through a milestone, first to last: each issue depends on
68/// the one before it, as the plan records `depends_on`, and none of them is done. The order a
69/// reader would have to work through, at the least, before the milestone can close. Issues of
70/// other milestones are not part of it; a cycle, which the plan's validator refuses, ends a
71/// chain rather than looping.
72pub fn critical_path(issues: &[Value]) -> Vec<String> {
73    let open: BTreeMap<String, Vec<String>> = issues
74        .iter()
75        .filter(|i| !closed(&text(i, "status")))
76        .map(|i| (text(i, "id"), strings(i, "depends_on")))
77        .collect();
78    fn longest(
79        id: &str,
80        open: &BTreeMap<String, Vec<String>>,
81        memo: &mut BTreeMap<String, Vec<String>>,
82        on_stack: &mut BTreeSet<String>,
83    ) -> Vec<String> {
84        if let Some(found) = memo.get(id) {
85            return found.clone();
86        }
87        if !on_stack.insert(id.to_string()) {
88            return Vec::new();
89        }
90        let mut best: Vec<String> = Vec::new();
91        for dep in open.get(id).into_iter().flatten() {
92            if open.contains_key(dep) {
93                let chain = longest(dep, open, memo, on_stack);
94                if chain.len() > best.len() {
95                    best = chain;
96                }
97            }
98        }
99        on_stack.remove(id);
100        best.push(id.to_string());
101        memo.insert(id.to_string(), best.clone());
102        best
103    }
104    let mut memo = BTreeMap::new();
105    let mut best: Vec<String> = Vec::new();
106    for id in open.keys() {
107        let chain = longest(id, &open, &mut memo, &mut BTreeSet::new());
108        if chain.len() > best.len() {
109            best = chain;
110        }
111    }
112    best
113}

115fn chip(href: String, label: String) -> El {
116    el("a")
117        .class("mj-ui-chip mj-ui-chip--link")
118        .attr("href", href)
119        .child(el("code").class("mj-ui-chip-value").text(label))
120}

122fn chips(ids: &[String], to: fn(&str) -> String) -> El {
123    if ids.is_empty() {
124        return el("span").class("mj-ui-dim").text("none");
125    }
126    el("span")
127        .class("mj-ui-badges")
128        .children(ids.iter().map(|id| chip(to(id), id.clone())))
129}

131/// The inspector of milestone `id`.
132pub fn inspector(roadmap: &Answer, issues: &Answer, id: &str) -> El {
133    let body = unanswered(roadmap)
134        .or_else(|| unanswered(issues))
135        .unwrap_or_else(|| {
136            let Some(m) = list(roadmap, "/milestones")
137                .into_iter()
138                .find(|m| text(m, "id") == id)
139            else {
140                return nothing(&format!("{id} is not a milestone of the plan."));
141            };
142            let held: BTreeSet<String> = strings(&m, "issues").into_iter().collect();
143            let mine: Vec<Value> = list(issues, "/issues")
144                .into_iter()
145                .filter(|i| held.contains(&text(i, "id")) || text(i, "milestone") == id)
146                .collect();
147            let ready: Vec<String> = mine
148                .iter()
149                .filter(|i| text(i, "status").eq_ignore_ascii_case("READY"))
150                .map(|i| text(i, "id"))
151                .collect();
152            let path = critical_path(&mine);
153            let serves: Vec<String> = mine
154                .iter()
155                .flat_map(|i| {
156                    strings(i, "serves")
157                        .into_iter()
158                        .map(|s| super::intent::intent_of(&s).to_string())
159                })
160                .collect::<BTreeSet<_>>()
161                .into_iter()
162                .collect();
163            let rows = mine
164                .iter()
165                .map(|i| {
166                    let iid = text(i, "id");
167                    el("tr")
168                        .child(
169                            el("td").child(
170                                el("a")
171                                    .class("mj-ui-row-link")
172                                    .attr("href", issue_href(&iid))
173                                    .child(
174                                        el("span").class("mj-ui-mono mj-ui-dim").text(iid.clone()),
175                                    )
176                                    .child(
177                                        el("span").class("mj-ui-cell-title").text(text(i, "title")),
178                                    ),
179                            ),
180                        )
181                        .child(el("td").child(state(i, "status")))
182                        .child(el("td").class("mj-ui-mono").text(text(i, "priority")))
183                        .child(
184                            el("td")
185                                .class("mj-ui-mono")
186                                .text(i.get("wave").map(Value::to_string).unwrap_or_default()),
187                        )
188                        .child(el("td").child(chips(&strings(i, "blocked_by"), issue_href)))
189                })
190                .collect::<Vec<_>>();
191            let fact = |label: &str, value: El| {
192                el("div")
193                    .class("mj-ui-fact")
194                    .child(el("dt").text(label))
195                    .child(el("dd").child(value))
196            };
197            let count = |s: &str| {
198                m.pointer(&format!("/counts/by_status/{s}"))
199                    .and_then(Value::as_u64)
200            };
201            el("section")
202                .class("mj-ui-detail")
203                .attr("id", "milestone")
204                .attr("aria-label", format!("Milestone {id}"))
205                .child(
206                    el("div")
207                        .class("mj-ui-detail-head")
208                        .child(el("p").class("mj-ui-mono mj-ui-dim").text(format!(
209                            "{id} · order {} · {}",
210                            m.get("order").map(Value::to_string).unwrap_or_else(|| "?".into()),
211                            text(&m, "priority")
212                        )))
213                        .child(el("h3").class("mj-ui-detail-title").text(text(&m, "title")))
214                        .child(
215                            el("p")
216                                .class("mj-ui-badges")
217                                .child(state(&m, "status"))
218                                .child(
219                                    el("a")
220                                        .class("mj-ui-more")
221                                        .attr(
222                                            "href",
223                                            format!("{}#milestone-{id}", super::roadmap::ROUTE),
224                                        )
225                                        .text("Close"),
226                                ),
227                        ),
228                )
229                .child(
230                    el("dl")
231                        .class("mj-ui-facts")
232                        .child(fact(
233                            "Done",
234                            meter(
235                                count("DONE"),
236                                m.pointer("/counts/total").and_then(Value::as_u64),
237                            ),
238                        ))
239                        .child(fact("Ready now", chips(&ready, issue_href)))
240                        .child(fact(
241                            "Blocked",
242                            el("span").class("mj-ui-mono").text(
243                                count("BLOCKED").map_or_else(|| "none".into(), |n| n.to_string()),
244                            ),
245                        ))
246                        .child(fact(
247                            "Waits for",
248                            chips(&strings(&m, "depends_on"), href_owned),
249                        ))
250                        .child(fact(
251                            "Blocked by",
252                            chips(&strings(&m, "blocked_by"), href_owned),
253                        ))
254                        .child(fact(
255                            "Unblocks",
256                            chips(&strings(&m, "dependents"), href_owned),
257                        ))
258                        .child(fact("Serves", chips(&serves, super::intent::href))),
259                )
260                .child(el("h4").class("mj-ui-panel-title").text("Critical path"))
261                .child(if path.is_empty() {
262                    nothing("No issue of it is open.")
263                } else {
264                    el("ol")
265                        .class("mj-ui-badges")
266                        .attr("aria-label", "The longest chain of open issues")
267                        .children(
268                            path.iter()
269                                .map(|i| el("li").child(chip(issue_href(i), i.clone()))),
270                        )
271                })
272                .child(el("h4").class("mj-ui-panel-title").text("Issues"))
273                .child(table(
274                    &["Issue", "Status", "Priority", "Wave", "Blocked by"],
275                    rows,
276                    "The milestone holds no issue.",
277                ))
278        });
279    panel(
280        "Milestone",
281        "One milestone of the plan: what it holds, what can start, and what it waits for.",
282        body,
283        &[roadmap, issues],
284    )
285}

336    /// A milestone that gives no order, counts nothing blocked and holds no open issue says
337    /// each of those rather than inventing a number or drawing an empty path.
338    #[test]
339    fn a_milestone_with_nothing_open_says_so() {
340        let roadmap = answer(
341            "plan.roadmap",
342            serde_json::json!({ "milestones": [
343                { "id": "M009", "status": "DONE", "priority": "p2", "title": "done already",
344                  "depends_on": [], "blocked_by": [], "dependents": [],
345                  "counts": { "total": 1, "by_status": { "DONE": 1 } }, "issues": ["I0009"] }
346            ]}),
347        );
348        let issues = answer(
349            "plan.issues",
350            serde_json::json!({ "issues": [issue("I0009", "DONE", &[])] }),
351        );
352        let html = inspector(&roadmap, &issues, "M009").render();
353        for want in ["M009 · order ? · p2", "No issue of it is open."] {
354            assert!(html.contains(want), "no {want}: {html}");
355        }
356        assert!(html.contains(">none<"), "no blocked count is none: {html}");
357    }

359    #[test]
360    fn the_critical_path_is_the_longest_chain_of_open_issues() {
361        let (_, issues) = plan();
362        let all = list(&issues, "/issues");
363        assert_eq!(critical_path(&all), ["I0002", "I0003", "I0004"]);
364        let cyclic = vec![issue("I1", "READY", &["I2"]), issue("I2", "READY", &["I1"])];
365        assert!(critical_path(&cyclic).len() <= 2, "a cycle ends a chain");
366    }

387    #[test]
388    fn an_unknown_milestone_says_so_and_a_hostile_one_is_never_picked() {
389        let (roadmap, issues) = plan();
390        assert!(inspector(&roadmap, &issues, "M999")
391            .render()
392            .contains("M999 is not a milestone of the plan."));
393        let q = |v: &str| vec![("milestone".to_string(), v.to_string())];
394        assert_eq!(picked(&q("M004")), Some("M004".into()));
395        assert_eq!(
396            picked(&q("development-runtime")),
397            Some("development-runtime".into()),
398            "a milestone named by a slug is a milestone"
399        );
400        assert_eq!(picked(&q("M0\" onclick=\"x")), None);
401        assert_eq!(picked(&q("")), None);
402    }
