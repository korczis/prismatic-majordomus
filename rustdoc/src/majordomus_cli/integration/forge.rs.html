1//! The forge adapter: the one place pull-request integration talks to the network.
2//!
3//! SECURITY.md declares it: `majordomus prs refresh` and `majordomus prs drain` — and
4//! nothing else — run the GitHub CLI (`gh`) and `git fetch` against this repository's own
5//! remote. The HTTP server, the MCP tools and the Cockpit never do: they render the last
6//! [`ForgeObservation`] recorded under `.ai/local/state/integration/`, with its age, so a
7//! page load can never reach the network. The observation is plain data, so every test
8//! builds one by hand and no test needs the forge.
9

19/// The recorded observation's schema version.
20pub const OBSERVATION_SCHEMA: u32 = 1;
21
22/// Where the pull-request heads are fetched to: a namespace of this tool's own, so no
23/// fetch ever moves a ref a person or another tool owns.
24pub const PR_REF_PREFIX: &str = "refs/majordomus/prs/";

26/// Everything the forge said, at one moment.
27#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
28pub struct ForgeObservation {
29    /// The format; this reads [`OBSERVATION_SCHEMA`].
30    pub schema: u32,
31    /// `owner/name`.
32    pub repository: String,
33    /// The integration base (the default branch).
34    pub base: String,
35    /// The base's commit as the forge reported it when observed.
36    pub base_sha: String,
37    /// When, RFC 3339.
38    pub observed_at: String,
39    /// The status contexts the base's protection requires; `None` when it could not be
40    /// read (every required-check verdict is then unknown).
41    pub required_checks: Option<Vec<String>>,
42    /// Whether the protection requires an approving review; `None` when unread.
43    pub reviews_required: Option<bool>,
44    /// The merge methods the repository allows, in the forge's words (`merge`, `squash`,
45    /// `rebase`).
46    pub merge_methods: Vec<String>,
47    /// Every open pull request.
48    pub pull_requests: Vec<PullRequestObservation>,
49}

51/// Why the forge could not be observed.

55impl std::fmt::Display for ForgeError {

71}

73fn gh(root: &Path, args: &[&str]) -> Result<(bool, String, String), ForgeError> {
74    let out = Command::new("gh")
75        .current_dir(root)
76        .args(args)
77        .output()
78        .map_err(|e| ForgeError(format!("the GitHub CLI (gh) could not run: {e}")))?;
79    Ok((
80        out.status.success(),
81        String::from_utf8_lossy(&out.stdout).into_owned(),
82        String::from_utf8_lossy(&out.stderr).into_owned(),
83    ))
84}
85
86/// `gh`, asked again on a transient failure ([`super::retry`]): every call here is a read,
87/// so asking twice is harmless. A refusal is returned as it came, for the caller to read.
88fn gh_retrying(root: &Path, args: &[&str]) -> Result<(bool, String, String), ForgeError> {
89    super::retry::forge(|| {
90        let (ok, out, err) = gh(root, args).map_err(|e| e.0)?;
91        if !ok && super::retry::transient(&err) {
92            Err(format!("gh {} failed: {}", args.join(" "), err.trim()))
93        } else {
94            Ok((ok, out, err))
95        }
96    })
97    .map_err(ForgeError)
98}
99
100fn gh_json(root: &Path, args: &[&str]) -> Result<Value, ForgeError> {
101    let (ok, out, err) = gh_retrying(root, args)?;
102    if !ok {
103        return Err(ForgeError(format!(
104            "gh {} failed: {}",
105            args.join(" "),
106            err.trim()
107        )));
108    }
109    serde_json::from_str(&out).map_err(|e| ForgeError(format!("gh {}: {e}", args.join(" "))))
110}

112/// The state word of one rollup entry: a check run (`status`/`conclusion`) or a commit
113/// status context (`state`).
114pub fn check_state(entry: &Value) -> CheckRunState {
115    let s = |k: &str| {
116        entry
117            .get(k)
118            .and_then(Value::as_str)
119            .unwrap_or("")
120            .to_ascii_uppercase()
121    };
122    if entry.get("__typename").and_then(Value::as_str) == Some("StatusContext") {
123        return match s("state").as_str() {
124            "SUCCESS" => CheckRunState::Passed,
125            "PENDING" | "EXPECTED" => CheckRunState::Pending,
126            _ => CheckRunState::Failed,
127        };
128    }
129    if s("status") != "COMPLETED" {
130        return CheckRunState::Pending;
131    }
132    match s("conclusion").as_str() {
133        "SUCCESS" => CheckRunState::Passed,
134        "SKIPPED" | "NEUTRAL" => CheckRunState::Skipped,
135        _ => CheckRunState::Failed,
136    }
137}

139/// One pull request from `gh pr list --json` output.
140pub fn pull_request_of(v: &Value) -> Option<PullRequestObservation> {
141    let s = |k: &str| v.get(k).and_then(Value::as_str).unwrap_or("").to_string();
142    Some(PullRequestObservation {
143        number: v.get("number")?.as_u64()?,
144        title: s("title"),
145        author: v
146            .pointer("/author/login")
147            .and_then(Value::as_str)
148            .unwrap_or("")
149            .to_string(),
150        head_ref: s("headRefName"),
151        head_sha: s("headRefOid"),
152        base_ref: s("baseRefName"),
153        draft: v.get("isDraft").and_then(Value::as_bool).unwrap_or(false),
154        labels: v
155            .get("labels")
156            .and_then(Value::as_array)
157            .map(|a| {
158                a.iter()
159                    .filter_map(|l| l.get("name").and_then(Value::as_str).map(str::to_string))
160                    .collect()
161            })
162            .unwrap_or_default(),
163        created_at: s("createdAt"),
164        updated_at: s("updatedAt"),
165        body: s("body"),
166        checks: v
167            .get("statusCheckRollup")
168            .and_then(Value::as_array)
169            .map(|a| {
170                a.iter()
171                    .map(|c| CheckObservation {
172                        name: c
173                            .get("name")
174                            .or_else(|| c.get("context"))
175                            .and_then(Value::as_str)
176                            .unwrap_or("")
177                            .to_string(),
178                        state: check_state(c),
179                    })
180                    .collect()
181            })
182            .unwrap_or_default(),
183        review_decision: s("reviewDecision"),
184        auto_merge: v
185            .get("autoMergeRequest")
186            .map(|a| !a.is_null())
187            .unwrap_or(false),
188        cross_repository: v
189            .get("isCrossRepository")
190            .and_then(Value::as_bool)
191            .unwrap_or(false),
192    })
193}

195/// The required contexts and review requirement from a branch-protection document.
196pub fn protection_of(v: &Value) -> (Option<Vec<String>>, Option<bool>) {
197    // a set: the protection lists a context under `contexts` and again under `checks`, and
198    // the order is the set's, so no sort sits beside what renders it
199    let mut contexts: std::collections::BTreeSet<String> = v
200        .pointer("/required_status_checks/contexts")
201        .and_then(Value::as_array)
202        .map(|a| {
203            a.iter()
204                .filter_map(|c| c.as_str().map(str::to_string))
205                .collect()
206        })
207        .unwrap_or_default();
208    if let Some(checks) = v
209        .pointer("/required_status_checks/checks")
210        .and_then(Value::as_array)
211    {
212        contexts.extend(
213            checks
214                .iter()
215                .filter_map(|c| c.get("context").and_then(Value::as_str).map(str::to_string)),
216        );
217    }
218    let reviews = v
219        .pointer("/required_pull_request_reviews/required_approving_review_count")
220        .and_then(Value::as_u64)
221        .map(|n| n > 0)
222        .or(Some(v.get("required_pull_request_reviews").is_some()));
223    (Some(contexts.into_iter().collect()), reviews)
224}

226impl Forge for GhForge<'_> {
227    fn observe(&self) -> Result<ForgeObservation, ForgeError> {
228        let root = self.root;
229        let repo = gh_json(
230            root,
231            &["repo", "view", "--json", "nameWithOwner,defaultBranchRef"],
232        )?;
233        let repository = repo
234            .get("nameWithOwner")
235            .and_then(Value::as_str)
236            .ok_or_else(|| ForgeError("gh repo view named no repository".into()))?
237            .to_string();
238        let base = repo
239            .pointer("/defaultBranchRef/name")
240            .and_then(Value::as_str)
241            .ok_or_else(|| ForgeError("gh repo view named no default branch".into()))?
242            .to_string();
243        let settings = gh_json(root, &["api", &format!("repos/{repository}")])?;
244        let mut merge_methods = Vec::new();
245        for (key, word) in [
246            ("allow_merge_commit", "merge"),
247            ("allow_squash_merge", "squash"),
248            ("allow_rebase_merge", "rebase"),
249        ] {
250            if settings.get(key).and_then(Value::as_bool).unwrap_or(false) {
251                merge_methods.push(word.to_string());
252            }
253        }
254        let base_sha = gh_json(
255            root,
256            &[
257                "api",
258                &format!("repos/{repository}/commits/{base}"),
259                "--jq",
260                "{sha: .sha}",
261            ],
262        )?
263        .get("sha")
264        .and_then(Value::as_str)
265        .unwrap_or("")
266        .to_string();
267        // an unprotected branch is a 404 whose message says so: nothing is required, which
268        // is a reading, not a failure; any other refusal leaves the requirement unread
269        let (ok, out, err) = gh_retrying(
270            root,
271            &[
272                "api",
273                &format!("repos/{repository}/branches/{base}/protection"),
274            ],
275        )?;
276        let (required_checks, reviews_required) = if ok {
277            serde_json::from_str::<Value>(&out)
278                .map(|v| protection_of(&v))
279                .unwrap_or((None, None))
280        } else if err.contains("Branch not protected") || out.contains("Branch not protected") {
281            (Some(Vec::new()), Some(false))
282        } else {
283            (None, None)
284        };
285        let list = gh_json(
286            root,
287            &[
288                "pr",
289                "list",
290                "--state",
291                "open",
292                "--limit",
293                "500",
294                "--json",
295                "number,title,author,headRefName,headRefOid,baseRefName,isDraft,labels,createdAt,updatedAt,body,statusCheckRollup,reviewDecision,autoMergeRequest,isCrossRepository",
296            ],
297        )?;
298        // keyed by number, so the observation is in number order whatever the forge listed
299        let pull_requests: Vec<PullRequestObservation> = list
300            .as_array()
301            .map(|a| {
302                a.iter()
303                    .filter_map(pull_request_of)
304                    .map(|p| (p.number, p))
305                    .collect::<std::collections::BTreeMap<_, _>>()
306                    .into_values()
307                    .collect()
308            })
309            .unwrap_or_default();
310        Ok(ForgeObservation {
311            schema: OBSERVATION_SCHEMA,
312            repository,
313            base,
314            base_sha,
315            observed_at: crate::peers::rfc3339(std::time::SystemTime::now()),
316            required_checks,
317            reviews_required,
318            merge_methods,
319            pull_requests,
320        })
321    }
322}

324/// Fetch the base and every observed head into this clone, the heads under
325/// [`PR_REF_PREFIX`]. One `git fetch`; a head that cannot be fetched stays unknown to the
326/// relation, which then says so.
327pub fn fetch(root: &Path, obs: &ForgeObservation) -> Result<(), ForgeError> {
328    let mut args: Vec<String> = vec![
329        "-C".into(),
330        root.display().to_string(),
331        "fetch".into(),
332        "--quiet".into(),
333        "--no-tags".into(),
334        "origin".into(),
335        format!("+refs/heads/{0}:refs/remotes/origin/{0}", obs.base),
336    ];
337    for p in &obs.pull_requests {
338        args.push(format!("+refs/pull/{0}/head:{PR_REF_PREFIX}{0}", p.number));
339    }
340    // a fetch is a read: a dropped connection is asked again, a refusal is not
341    super::retry::forge(|| {
342        let out = Command::new("git")
343            .args(&args)
344            .output()
345            .map_err(|e| format!("git fetch could not run: {e}"))?;
346        if out.status.success() {
347            Ok(())
348        } else {
349            Err(format!(
350                "git fetch failed: {}",
351                String::from_utf8_lossy(&out.stderr).trim()
352            ))
353        }
354    })
355    .map_err(ForgeError)
356}

360    //! The forge's JSON, read into the observation: every state word a rollup entry can

367    #[test]
368    fn every_rollup_state_word_has_one_meaning() {
369        let ctx = |s: &str| json!({"__typename": "StatusContext", "state": s});
370        assert_eq!(check_state(&ctx("success")), CheckRunState::Passed);
371        assert_eq!(check_state(&ctx("PENDING")), CheckRunState::Pending);
372        assert_eq!(check_state(&ctx("EXPECTED")), CheckRunState::Pending);
373        assert_eq!(check_state(&ctx("ERROR")), CheckRunState::Failed);
374        assert_eq!(check_state(&ctx("FAILURE")), CheckRunState::Failed);
375        let run = |status: &str, conclusion: &str| json!({"__typename": "CheckRun", "status": status, "conclusion": conclusion});
376        assert_eq!(check_state(&run("IN_PROGRESS", "")), CheckRunState::Pending);
377        assert_eq!(check_state(&run("QUEUED", "")), CheckRunState::Pending);
378        assert_eq!(
379            check_state(&run("COMPLETED", "SUCCESS")),
380            CheckRunState::Passed
381        );
382        assert_eq!(
383            check_state(&run("completed", "skipped")),
384            CheckRunState::Skipped
385        );
386        assert_eq!(
387            check_state(&run("COMPLETED", "NEUTRAL")),
388            CheckRunState::Skipped
389        );
390        assert_eq!(
391            check_state(&run("COMPLETED", "FAILURE")),
392            CheckRunState::Failed
393        );
394        assert_eq!(
395            check_state(&run("COMPLETED", "CANCELLED")),
396            CheckRunState::Failed
397        );
398        assert_eq!(
399            check_state(&run("COMPLETED", "TIMED_OUT")),
400            CheckRunState::Failed
401        );
402        // an entry that says nothing has not completed, so it has not passed
403        assert_eq!(check_state(&json!({})), CheckRunState::Pending);
404    }

406    #[test]
407    fn a_pull_request_is_read_with_every_field_and_without_any() {
408        let full = json!({
409            "number": 42,
410            "title": "a change",
411            "author": {"login": "someone"},
412            "headRefName": "feature/x",
413            "headRefOid": "abc",
414            "baseRefName": "master",
415            "isDraft": true,
416            "labels": [{"name": "hold"}, {"name": "docs"}],
417            "createdAt": "2026-09-01T00:00:00Z",
418            "updatedAt": "2026-09-02T00:00:00Z",
419            "body": "Depends on #7.",
420            "statusCheckRollup": [
421                {"__typename": "CheckRun", "name": "ci", "status": "COMPLETED", "conclusion": "SUCCESS"},
422                {"__typename": "StatusContext", "context": "legacy", "state": "PENDING"}
423            ],
424            "reviewDecision": "APPROVED",
425            "autoMergeRequest": {"enabledAt": "2026-09-03T00:00:00Z"},
426            "isCrossRepository": true
427        });
428        let p = pull_request_of(&full).expect("read");
429        assert_eq!(p.number, 42);
430        assert_eq!(p.author, "someone");
431        assert!(p.draft && p.auto_merge && p.cross_repository);
432        assert_eq!(p.labels, ["hold", "docs"]);
433        assert_eq!(p.checks.len(), 2);
434        assert_eq!(p.checks[0].name, "ci");
435        assert_eq!(p.checks[0].state, CheckRunState::Passed);
436        assert_eq!(
437            p.checks[1].name, "legacy",
438            "a status context is named by its context"
439        );
440        assert_eq!(p.checks[1].state, CheckRunState::Pending);
441        assert_eq!(p.review_decision, "APPROVED");
442
443        let bare = pull_request_of(&json!({"number": 1})).expect("a number is enough");
444        assert!(!bare.draft && !bare.auto_merge && !bare.cross_repository);
445        assert!(bare.labels.is_empty() && bare.checks.is_empty());
446        assert_eq!(bare.title, "");
447        let disarmed = pull_request_of(&json!({"number": 2, "autoMergeRequest": null})).unwrap();
448        assert!(
449            !disarmed.auto_merge,
450            "a null auto-merge request is not armed"
451        );
452        assert!(pull_request_of(&json!({"title": "no number"})).is_none());
453    }

455    #[test]
456    fn a_protection_names_its_contexts_once_and_its_review_requirement() {
457        let both = json!({
458            "required_status_checks": {
459                "contexts": ["ci", "lint"],
460                "checks": [{"context": "ci"}, {"context": "docs"}]
461            },
462            "required_pull_request_reviews": {"required_approving_review_count": 1}
463        });
464        let (contexts, reviews) = protection_of(&both);
465        assert_eq!(contexts.unwrap(), ["ci", "docs", "lint"]);
466        assert_eq!(reviews, Some(true));
467
468        let none_required = json!({
469            "required_status_checks": {"contexts": []},
470            "required_pull_request_reviews": {"required_approving_review_count": 0}
471        });
472        assert_eq!(protection_of(&none_required).1, Some(false));
473
474        let reviews_without_count = json!({"required_pull_request_reviews": {}});
475        let (contexts, reviews) = protection_of(&reviews_without_count);
476        assert_eq!(contexts.unwrap(), Vec::<String>::new());
477        assert_eq!(
478            reviews,
479            Some(true),
480            "a review section without a count still asks for one"
481        );
482
483        assert_eq!(protection_of(&json!({})).1, Some(false));
484    }
