1//! `reasoning.check`: the executable half of the rules `project.advisors-are-optional`
2//! and `project.review-is-recorded-not-claimed`. Each check reads files and says what it
3//! found; none of them asks an advisor anything.
4//!
5//! * `catalogue` — the advisor catalogue resolves against the provider table and the
6//!   model catalogue (one inventory, referenced, never restated);
7//! * `adapters` — every adapter the catalogue names has its transport module, so the Node
8//!   layer is derived from the catalogue rather than keeping a list of its own;
9//! * `coupling` — the provider-independent code (this module, its capability, its CLI
10//!   rendering, its Cockpit page, the transport driver) names no advisor, adapter,
11//!   executable, provider, vendor or model of the catalogue as a string literal;
12//! * `ci` — no CI workflow or gate names a model vendor's credential variable, so no
13//!   required gate can depend on a live model;
14//! * `claims` — no document asserts a claim the catalogue forbids ("consensus
15//!   guarantees ...") or that an advisor is required;
16//! * `records` — every stored record is readable, every consultation names an advisor
17//!   its plan selected and one this repository knows, every conclusion's computed review
18//!   matches the consultations it cites.
19//!
20//! The lifecycle: load the catalogue, its references and the store, then [`run`] every
21//! check over a repository root; an empty list is a clean repository.
22//!
23//! ```
24//! use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
25//! use majordomus_cli::reasoning::check::run;
26//! use majordomus_cli::reasoning::store::Store;
27//!
28//! let root = tempfile::tempdir().unwrap();
29//! let catalogue = AdvisorCatalogue {
30//!     forbidden_claims: vec!["consensus guarantees correctness".into()],
31//!     ..Default::default()
32//! };
33//! let loaded = Store::at(root.path()).load();
34//! assert!(run(root.path(), &catalogue, &References::default(), &loaded).is_empty());
35//!
36//! std::fs::write(root.path().join("README.md"), "Consensus guarantees correctness.\n").unwrap();
37//! let findings = run(root.path(), &catalogue, &References::default(), &loaded);
38//! assert_eq!(findings.len(), 1);
39//! assert_eq!((findings[0].check.as_str(), findings[0].subject.as_str()), ("claims", "README.md"));
40//! ```

51/// One finding of a reasoning check: which check, where, and what is wrong.
52///
53/// `check` is one of [`CHECKS`]; `subject` is a repository-relative path, a record id or
54/// an advisor; `message` is one sentence saying what the rule refuses. Findings are
55/// reported, never repaired.
56///
57/// ```
58/// use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
59/// use majordomus_cli::reasoning::check::{run, ReasoningFinding, CHECKS};
60/// use majordomus_cli::reasoning::store::Loaded;
61///
62/// let root = tempfile::tempdir().unwrap();
63/// let loaded = Loaded { records: vec![], unreadable: vec!["t/broken.json".into()] };
64/// let found = run(root.path(), &AdvisorCatalogue::default(), &References::default(), &loaded);
65/// let expected = ReasoningFinding {
66///     check: "records".into(),
67///     subject: "t/broken.json".into(),
68///     message: "a reasoning record that does not parse".into(),
69/// };
70/// assert_eq!(found, [expected]);
71/// assert!(CHECKS.contains(&found[0].check.as_str()));
72/// ```
73#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
74pub struct ReasoningFinding {
75    /// The check that found it.
76    pub check: String,
77    /// Where: a file, a record, an advisor.
78    pub subject: String,
79    /// What is wrong.
80    pub message: String,
81}

96fn files_under(root: &Path, rel: &str, ext: &[&str]) -> Vec<std::path::PathBuf> {
97    let path = root.join(rel);
98    // ordered by construction: a directory listing has no order of its own
99    let mut out = std::collections::BTreeSet::new();
100    if path.is_file() {
101        out.insert(path);
102    } else if let Ok(entries) = std::fs::read_dir(&path) {
103        for e in entries.flatten() {
104            let p = e.path();
105            if p.is_dir() {
106                if let Ok(r) = p.strip_prefix(root) {
107                    out.extend(files_under(root, &r.display().to_string(), ext));
108                }
109            } else if p
110                .extension()
111                .and_then(|x| x.to_str())
112                .is_some_and(|x| ext.contains(&x))
113            {
114                out.insert(p);
115            }
116        }
117    }
118    out.into_iter().collect()
119}

121/// The names the independent code must not spell.
122///
123/// Every advisor's id, adapter, executable, provider, vendor and model, and the peer
124/// adapter when peers are declared: each once, in canonical order. These are the
125/// literals the `coupling` check looks for.
126///
127/// ```
128/// use majordomus_cli::reasoning::catalogue::AdvisorCatalogue;
129/// use majordomus_cli::reasoning::check::catalogue_names;
130///
131/// let catalogue: AdvisorCatalogue = majordomus_cli::metadata::yaml::parse_into(
132///     "version: 1\nadvisors:\n  - id: alpha\n    title: Alpha\n    transport: cli\n    adapter: alpha-cli\n    executable: alpha\n",
133/// )
134/// .unwrap();
135/// assert_eq!(catalogue_names(&catalogue), ["alpha", "alpha-cli"]);
136/// ```
137pub fn catalogue_names(catalogue: &AdvisorCatalogue) -> Vec<String> {
138    let mut names = Vec::new();
139    for a in &catalogue.advisors {
140        names.push(a.id.clone());
141        names.push(a.adapter.clone());
142        names.extend(a.executable.clone());
143        names.extend(a.provider.clone());
144        names.extend(a.vendor.clone());
145        names.extend(a.model.clone());
146    }
147    if let Some(p) = &catalogue.peers {
148        names.push(p.adapter.clone());
149    }
150    crate::order::canonical(&mut names);
151    names.dedup();
152    names
153}

155/// String literals of a source naming any of `names`: `"name"` or `'name'`, exactly.
156///
157/// A name inside prose, a path or a longer literal is not a hit: only a literal that is
158/// the name and nothing else couples code to an advisor. Each name is reported once, in
159/// the order of `names`.
160///
161/// ```
162/// use majordomus_cli::reasoning::check::literals_naming;
163///
164/// let names = vec!["alpha".to_string(), "beta".to_string()];
165/// let source = "let a = \"alpha\"; let b = 'alpha'; // the beta advisor\nimport('./beta.mjs')";
166/// assert_eq!(literals_naming(source, &names), ["alpha"]);
167/// ```
168pub fn literals_naming(text: &str, names: &[String]) -> Vec<String> {
169    let mut hits = Vec::new();
170    for n in names {
171        for quote in ['"', '\''] {
172            let needle = format!("{quote}{n}{quote}");
173            if text.contains(&needle) && !hits.contains(n) {
174                hits.push(n.clone());
175            }
176        }
177    }
178    hits
179}

181/// Run every check over the repository at `root`, in the order of [`CHECKS`].
182///
183/// It reads the catalogue, the adapter modules, the provider-independent sources, the CI
184/// workflows, the documentation and the loaded records, and asks no advisor anything.
185/// The findings come back in check order; an empty list means every rule holds.
186///
187/// ```
188/// use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
189/// use majordomus_cli::reasoning::check::run;
190/// use majordomus_cli::reasoning::record::ReasoningRecord;
191/// use majordomus_cli::reasoning::store::Loaded;
192///
193/// let record: ReasoningRecord = serde_json::from_value(serde_json::json!({
194///     "schema": "majordomus/reasoning-record", "version": 1, "id": "consultation-1",
195///     "task": "t", "recorded_at": "2026-10-01T00:00:01Z",
196///     "body": {
197///         "kind": "consultation", "plan": "plan-0", "advisor": "alpha", "status": "timeout",
198///     },
199/// }))
200/// .unwrap();
201/// let loaded = Loaded { records: vec![record], unreadable: vec![] };
202/// let root = tempfile::tempdir().unwrap();
203/// let found = run(root.path(), &AdvisorCatalogue::default(), &References::default(), &loaded);
204/// let messages: Vec<&str> = found.iter().map(|f| f.message.as_str()).collect();
205/// assert_eq!(
206///     messages,
207///     [
208///         "names advisor 'alpha', which the catalogue does not declare",
209///         "cites plan plan-0, which is not recorded",
210///     ]
211/// );
212/// ```
213pub fn run(
214    root: &Path,
215    catalogue: &AdvisorCatalogue,
216    refs: &References,
217    loaded: &Loaded,
218) -> Vec<ReasoningFinding> {
219    run_checks(root, catalogue, refs, loaded, CHECKS)
220}

222/// Run the named `checks` over the repository at `root`, in the order of [`CHECKS`]; a
223/// name that is not one of them is ignored. [`run`] is this with every check, and
224/// [`applicable`] says which a repository is held to.
225///
226/// ```
227/// use majordomus_cli::reasoning::catalogue::{AdvisorCatalogue, References};
228/// use majordomus_cli::reasoning::check::run_checks;
229/// use majordomus_cli::reasoning::store::Store;
230///
231/// let root = tempfile::tempdir().unwrap();
232/// std::fs::write(root.path().join("README.md"), "Consensus guarantees correctness.\n").unwrap();
233/// let catalogue = AdvisorCatalogue {
234///     forbidden_claims: vec!["consensus guarantees correctness".into()],
235///     ..Default::default()
236/// };
237/// let loaded = Store::at(root.path()).load();
238/// let refs = References::default();
239/// assert_eq!(run_checks(root.path(), &catalogue, &refs, &loaded, &["claims"]).len(), 1);
240/// assert!(run_checks(root.path(), &catalogue, &refs, &loaded, &["catalogue", "records"]).is_empty());
241/// ```
242pub fn run_checks(
243    root: &Path,
244    catalogue: &AdvisorCatalogue,
245    refs: &References,
246    loaded: &Loaded,
247    checks: &[&str],
248) -> Vec<ReasoningFinding> {
249    let on = |c: &str| checks.contains(&c);
250    let mut out = Vec::new();
251    let mut push = |check: &str, subject: String, message: String| {
252        out.push(ReasoningFinding {
253            check: check.into(),
254            subject,
255            message,
256        })
257    };
258
259    if on("catalogue") {
260        for d in catalogue.diagnostics(refs) {
261            push("catalogue", "share/advisors.yaml".into(), d);
262        }
263    }
264
265    if on("adapters") && root.join(ADAPTER_DIR).is_dir() {
266        let mut adapters: Vec<&str> = catalogue
267            .advisors
268            .iter()
269            .map(|a| a.adapter.as_str())
270            .collect();
271        if let Some(p) = &catalogue.peers {
272            adapters.push(&p.adapter);
273        }
274        for adapter in adapters {
275            let module = format!("{ADAPTER_DIR}/{adapter}.mjs");
276            if !root.join(&module).is_file() {
277                push(
278                    "adapters",
279                    module.clone(),
280                    format!("the catalogue names adapter '{adapter}' and no transport module {module} implements it"),
281                );
282            }
283        }
284    }
285
286    let names = catalogue_names(catalogue);
287    for rel in INDEPENDENT_SOURCES.iter().filter(|_| on("coupling")) {
288        for file in files_under(root, rel, &["rs", "mjs"]) {
289            let Ok(text) = std::fs::read_to_string(&file) else {
290                continue;
291            };
292            for n in literals_naming(&text, &names) {
293                push(
294                    "coupling",
295                    file.strip_prefix(root).unwrap_or(&file).display().to_string(),
296                    format!("provider-independent code names '{n}'; it must reach advisors through the catalogue"),
297                );
298            }
299        }
300    }
301
302    let credentials = refs.models.credential_variables();
303    let mut ci_files = Vec::new();
304    if on("ci") {
305        ci_files.extend(files_under(root, ".github/workflows", &["yml", "yaml"]));
306        ci_files.extend(files_under(root, ".ai/repo/ci/gates.yaml", &["yaml"]));
307    }
308    for file in ci_files {
309        let Ok(text) = std::fs::read_to_string(&file) else {
310            continue;
311        };
312        for var in &credentials {
313            if text.contains(var) {
314                push(
315                    "ci",
316                    file.strip_prefix(root)
317                        .unwrap_or(&file)
318                        .display()
319                        .to_string(),
320                    format!(
321                        "CI names the model credential {var}; no gate may depend on a live model"
322                    ),
323                );
324            }
325        }
326    }
327
328    let mut phrases: Vec<String> = catalogue
329        .forbidden_claims
330        .iter()
331        .map(|p| p.to_lowercase())
332        .collect();
333    for a in &catalogue.advisors {
334        let t = a.title.to_lowercase();
335        phrases.push(format!("requires {t}"));
336        phrases.push(format!("{t} is required"));
337    }
338    let mut docs = Vec::new();
339    if on("claims") {
340        docs.extend(files_under(root, "docs", &["md"]));
341        docs.extend(files_under(root, "site/content", &["md"]));
342        docs.extend(files_under(root, ".ai/repo", &["md"]));
343        docs.push(root.join("README.md"));
344    }
345    for file in docs {
346        let Ok(text) = std::fs::read_to_string(&file) else {
347            continue;
348        };
349        let lower = text.to_lowercase();
350        for p in &phrases {
351            if lower.contains(p.as_str()) {
352                push(
353                    "claims",
354                    file.strip_prefix(root)
355                        .unwrap_or(&file)
356                        .display()
357                        .to_string(),
358                    format!("asserts '{p}', a claim the reasoning design refuses"),
359                );
360            }
361        }
362    }
363
364    let (unreadable, records) = if on("records") {
365        (&loaded.unreadable[..], &loaded.records[..])
366    } else {
367        (&[][..], &[][..])
368    };
369    for u in unreadable {
370        push(
371            "records",
372            u.clone(),
373            "a reasoning record that does not parse".into(),
374        );
375    }
376    for r in records {
377        match &r.body {
378            ReasoningBody::Consultation(c) => {
379                let known = catalogue.get(&c.advisor).is_some() || c.advisor.starts_with("peer:");
380                if !known {
381                    push(
382                        "records",
383                        r.id.clone(),
384                        format!(
385                            "names advisor '{}', which the catalogue does not declare",
386                            c.advisor
387                        ),
388                    );
389                }
390                match loaded.get(&c.plan).map(|p| &p.body) {
391                    Some(ReasoningBody::Plan(p)) => {
392                        if !p.plan.selected.iter().any(|s| s.advisor == c.advisor) {
393                            push(
394                                "records",
395                                r.id.clone(),
396                                format!(
397                                    "claims advisor '{}', which plan {} did not select",
398                                    c.advisor, c.plan
399                                ),
400                            );
401                        }
402                    }
403                    _ => push(
404                        "records",
405                        r.id.clone(),
406                        format!("cites plan {}, which is not recorded", c.plan),
407                    ),
408                }
409            }
410            ReasoningBody::Conclusion(k) => {
411                let mut computed: Vec<String> = Vec::new();
412                for id in &k.input.consultations {
413                    match loaded.get(id).map(|x| &x.body) {
414                        Some(ReasoningBody::Consultation(c))
415                            if c.status == ConsultationStatus::Completed =>
416                        {
417                            if !computed.contains(&c.advisor) {
418                                computed.push(c.advisor.clone());
419                            }
420                        }
421                        _ => push(
422                            "records",
423                            r.id.clone(),
424                            format!("cites {id}, which is not a completed consultation"),
425                        ),
426                    }
427                }
428                crate::order::canonical(&mut computed);
429                if computed != k.reviewed_by || k.independent_review_count != k.reviewed_by.len() {
430                    push(
431                        "records",
432                        r.id.clone(),
433                        format!(
434                            "claims review by [{}] ({}), and the consultations it cites show [{}]",
435                            k.reviewed_by.join(", "),
436                            k.independent_review_count,
437                            computed.join(", ")
438                        ),
439                    );
440                }
441            }
442            _ => {}
443        }
444    }
445    out
446}

448/// The checks, by name, in the order [`run`] performs them.
449pub const CHECKS: &[&str] = &[
450    "catalogue",
451    "adapters",
452    "coupling",
453    "ci",
454    "claims",
455    "records",
456];
457
458/// The rule whose executable half `ci` and `claims` are: what Majordomus's own CI and
459/// documents may say about advisors. A repository that does not declare it is an adopter,
460/// and an adopter's workflows and documents are its product, not its use of the
461/// reasoning layer (I1996).
462pub const REPOSITORY_RULE: &str = "project.advisors-are-optional";
463
464/// The checks that judge a repository's own workflows and documents rather than the
465/// reasoning layer it adopted; they apply only where [`REPOSITORY_RULE`] is declared.
466pub const REPOSITORY_CHECKS: &[&str] = &["ci", "claims"];
467
468/// The checks a repository declaring the rules `rule_ids` is held to, in [`CHECKS`] order:
469/// every check where [`REPOSITORY_RULE`] is among them, and only what an adopter adopted
470/// (its catalogue, adapters, coupling and records) where it is not.
471///
472/// ```
473/// use majordomus_cli::reasoning::check::{applicable, CHECKS};
474///
475/// assert_eq!(applicable(["project.advisors-are-optional"]), CHECKS);
476/// assert_eq!(
477///     applicable(["majordomus.some-other-rule"]),
478///     ["catalogue", "adapters", "coupling", "records"]
479/// );
480/// ```
481pub fn applicable<'a>(rule_ids: impl IntoIterator<Item = &'a str>) -> Vec<&'static str> {
482    let own = rule_ids.into_iter().any(|id| id == REPOSITORY_RULE);
483    CHECKS
484        .iter()
485        .copied()
486        .filter(|c| own || !REPOSITORY_CHECKS.contains(c))
487        .collect()
488}
