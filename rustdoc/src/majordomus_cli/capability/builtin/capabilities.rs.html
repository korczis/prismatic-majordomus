21/// The input of `capabilities.list`: optional filters by kind and by projection.
22pub struct CapabilitiesInput {

46/// One capability as a listing shows it: everything the descriptor says except its two
47/// schemas, which `capabilities.describe` answers for one capability. A listing of a
48/// repository's registry runs to hundreds of entries; their schemas would be megabytes
49/// of the same object view repeated.
50#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
51pub struct CapabilitySummary {
52    /// The canonical identity.
53    pub id: crate::capability::CapabilityId,
54    /// The module that composes it.
55    pub module: crate::capability::ModuleId,
56    /// Query, command or resource.
57    pub kind: CapabilityKind,
58    /// The short name.
59    pub title: String,
60    /// The one-paragraph description.
61    pub description: String,
62    /// Where it came from.
63    pub provenance: crate::capability::Provenance,
64    /// Where it is projected.
65    pub exposure: Exposure,
66    /// Where it stands.
67    pub stability: Stability,
68    #[serde(default, skip_serializing_if = "Vec::is_empty")]
69    /// Free tags.
70    pub tags: Vec<String>,
71    /// Whether it is a benchmark target.
72    pub benchmark: crate::capability::BenchmarkPolicy,
73    /// Whether the executor keeps its results.
74    pub cache: CachePolicy,
75}
76
77impl From<&Capability> for CapabilitySummary {
78    fn from(c: &Capability) -> Self {
79        CapabilitySummary {
80            id: c.id.clone(),
81            module: c.module.clone(),
82            kind: c.kind,
83            title: c.title.clone(),
84            description: c.description.clone(),
85            provenance: c.provenance.clone(),
86            exposure: c.exposure.clone(),
87            stability: c.stability,
88            tags: c.tags.clone(),
89            benchmark: c.benchmark,
90            cache: c.cache,
91        }
92    }
93}

95#[derive(Debug, Serialize, Deserialize, JsonSchema)]
96/// The answer of `capabilities.list`: the matching capabilities, summarised, and the
97/// registry counted.
98pub struct CapabilityList {
99    /// How many capabilities matched the filters.
100    pub count: usize,
101    /// The whole registry, counted by kind, stability and projection.
102    pub summary: Summary,
103    /// The matching capabilities, by id, without their schemas.
104    pub capabilities: Vec<CapabilitySummary>,
105}
106
107fn capabilities_list(
108    ctx: &Context,
109    input: CapabilitiesInput,
110) -> Result<CapabilityList, CapabilityError> {
111    let kind = match input.kind.as_deref() {
112        None => None,
113        Some("query") => Some(CapabilityKind::Query),
114        Some("command") => Some(CapabilityKind::Command),
115        Some("resource") => Some(CapabilityKind::Resource),
116        Some(other) => {
117            return Err(CapabilityError::InvalidInput(format!(
118                "kind '{other}' is not query, command or resource"
119            )))
120        }
121    };
122    let exposure = match input.exposure.as_deref() {
123        None | Some("mcp") | Some("http") | Some("cli") => input.exposure.clone(),
124        Some(other) => {
125            return Err(CapabilityError::InvalidInput(format!(
126                "exposure '{other}' is not mcp, http or cli"
127            )))
128        }
129    };
130    let capabilities: Vec<CapabilitySummary> = ctx
131        .registry
132        .iter()
133        .filter(|c| kind.is_none_or(|k| c.kind == k))
134        .filter(|c| match exposure.as_deref() {
135            None => true,
136            Some("mcp") => c.exposure.mcp.is_some(),
137            Some("http") => c.exposure.http.is_some(),
138            Some("cli") => c.exposure.cli.is_some(),
139            Some(_) => false,
140        })
141        .map(CapabilitySummary::from)
142        .collect();
143    Ok(CapabilityList {
144        count: capabilities.len(),
145        summary: ctx.registry.summary(),
146        capabilities,
147    })
148}

154/// The input of `capabilities.describe`: which capability.
155pub struct DescribeInput {
156    /// The canonical id, e.g. `repository.info` or `rule.majordomus.scope-integrity@1`.
157    pub id: String,
158}
159
160impl BenchmarkCases for DescribeInput {
161    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
162        vec![NamedCase::new(
163            "repository-info",
164            DescribeInput {
165                id: "repository.info".into(),
166            },
167        )]
168    }
169}
170
171fn capabilities_describe(
172    ctx: &Context,
173    input: DescribeInput,
174) -> Result<Capability, CapabilityError> {
175    ctx.registry
176        .get(&input.id)
177        .cloned()
178        .ok_or_else(|| CapabilityError::NotFound(format!("unknown capability: {}", input.id)))
179}

185/// The input of `capabilities.projections`: which rows, and whether only the unmet claims.
186pub struct ProjectionsInput {
187    /// Only capabilities composed in this module.
188    #[serde(default)]
189    pub module: Option<String>,
190    /// Only the capabilities whose declared exposures are not all answered by their
191    /// surface. Empty is the closure the rule asks for.
192    #[serde(default)]
193    pub unmet: bool,
194}
195
196impl BenchmarkCases for ProjectionsInput {
197    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
198        vec![
199            NamedCase::new("all", ProjectionsInput::default()),
200            NamedCase::new(
201                "unmet",
202                ProjectionsInput {
203                    module: None,
204                    unmet: true,
205                },
206            ),
207        ]
208    }
209}
210
211fn capabilities_projections(
212    ctx: &Context,
213    input: ProjectionsInput,
214) -> Result<crate::capability::closure::Matrix, CapabilityError> {
215    // the clap tree is a pure function of the declaration compiled into this executable:
216    // no repository is read, and the walk is the same one `cli::validate` runs
217    let mut m = crate::capability::closure::matrix(&ctx.registry, &crate::cli::tree());
218    if let Some(module) = &input.module {
219        if !ctx.registry.modules().any(|k| k.id.as_str() == module) {
220            return Err(CapabilityError::InvalidInput(format!(
221                "no module named '{module}'"
222            )));
223        }
224        m.rows.retain(|r| &r.module == module);
225    }
226    if input.unmet {
227        m.rows.retain(|r| !r.closed);
228    }
229    Ok(m)
230}

272                id: "capabilities.projections",

286            },
287        ],
288    }

295    /// The declaration is the only place these names exist, and every projection — the MCP
296    /// tool, the HTTP route, the OpenAPI operation, the benchmark target — is derived from
297    /// it. A refactor that dropped an exposure or renamed a route would still compile, and
298    /// the suites that exercise the behaviour behind it would still pass. This is the
299    /// assertion that would not.
300    #[test]
301    fn the_declaration_yields_the_projections_it_claims() {
302        let m = module();
303        assert_eq!(m.id.as_str(), "capabilities");
304        let expected: &[(&str, &str, &str)] = &[
305            (
306                "capabilities.list",
307                "majordomus_capabilities",
308                "/api/v1/capabilities",
309            ),
310            (
311                "capabilities.describe",
312                "majordomus_capability",
313                "/api/v1/capability",
314            ),
315            (
316                "capabilities.projections",
317                "majordomus_projections",
318                "/api/v1/capabilities/projections",
319            ),
320        ];
321        let ids: Vec<&str> = m
322            .capabilities
323            .iter()
324            .map(|e| e.capability.id.as_str())
325            .collect();
326        let want: Vec<&str> = expected.iter().map(|(id, _, _)| *id).collect();
327        assert_eq!(
328            ids, want,
329            "the module declares a different set of capabilities"
330        );
331        for (executable, (id, tool, path)) in m.capabilities.iter().zip(expected) {
332            let exposure = &executable.capability.exposure;
333            assert_eq!(
334                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
335                Some(*tool),
336                "{id} lost or renamed its MCP tool"
337            );
338            assert_eq!(
339                exposure.http.as_ref().map(|h| h.path.as_str()),
340                Some(*path),
341                "{id} lost or renamed its HTTP route"
342            );
343        }
344    }

346    /// The handler's two filters and its one refusal, over the registry this crate ships.
347    /// `Context::new` needs no repository: the filters read the registry and the clap tree,
348    /// and neither touches the index.
349    #[test]
350    fn projections_filters_by_module_and_refuses_a_module_that_does_not_exist() {
351        use crate::capability::{CapabilityRegistry, Context};
352        use crate::git::GitState;
353        use crate::index::{Index, RepositoryInfo, State};
354        use std::sync::Arc;
355
356        // an index with no objects: the filters read the registry and the clap tree, and
357        // neither of them touches the layer
358        let index = Index {
359            repository: RepositoryInfo {
360                root: "/tmp/projections".into(),
361                layer_schema: "ai-repository/v1".into(),
362                sections: Default::default(),
363                git: GitState::Unavailable {
364                    reason: "unit test".into(),
365                },
366                discovery: "filesystem".into(),
367                source_classes: vec![],
368                kind_sources: vec![],
369                scope_origin: crate::scope::Origin::Distribution,
370                scope_path: String::new(),
371            },
372            objects: vec![],
373            diagnostics: vec![],
374            state: State::Ok,
375            fingerprint: String::new(),
376            scoped: Default::default(),
377            distribution: None,
378            providers: Default::default(),
379            share: None,
380        };
381        let registry = Arc::new(
382            CapabilityRegistry::builder()
383                .with_modules(super::super::modules())
384                .build()
385                .expect("the builtin registry builds"),
386        );
387        let ctx = Context::new(Arc::new(index), Arc::clone(&registry));
388
389        // every row, unfiltered
390        let all = capabilities_projections(&ctx, ProjectionsInput::default()).expect("all rows");
391        assert_eq!(all.rows.len(), registry.len());
392        assert!(
393            all.unbacked.iter().any(|c| c == "majordomus serve"),
394            "the process commands are the debt this reports"
395        );
396
397        // one module
398        let one = capabilities_projections(
399            &ctx,
400            ProjectionsInput {
401                module: Some("capabilities".into()),
402                unmet: false,
403            },
404        )
405        .expect("one module");
406        assert!(!one.rows.is_empty());
407        assert!(one.rows.iter().all(|r| r.module == "capabilities"));
408        assert!(one.rows.len() < all.rows.len());
409
410        // the closure this crate ships: nothing unmet
411        let unmet = capabilities_projections(
412            &ctx,
413            ProjectionsInput {
414                module: None,
415                unmet: true,
416            },
417        )
418        .expect("unmet");
419        assert!(unmet.rows.is_empty(), "the shipped declaration is closed");
420
421        // a module nobody composes is a refusal naming it, not an empty answer that reads
422        // as "this module has no capabilities"
423        let err = capabilities_projections(
424            &ctx,
425            ProjectionsInput {
426                module: Some("nonesuch".into()),
427                unmet: false,
428            },
429        )
430        .expect_err("an unknown module is refused");
431        assert!(
432            matches!(err, CapabilityError::InvalidInput(ref m) if m.contains("nonesuch")),
433            "{err:?}"
434        );
435    }
