1//! The `distribution` module: how this project is packaged, published and installed, and
2//! what this particular build is.
3//!
4//! Every value comes from the distribution model the process was started with
5//! (`share/distribution.yaml`, carried on the index) and from the release records the
6//! repository holds. Nothing here decides a platform, an artifact name or a URL; the
7//! module exists so that the same answers reach a person at a command line, a script over
8//! HTTP, an assistant over MCP and the cockpit, without any of them restating a fact.

169/// The state of one check in the installability report. Three states and no more: a check
170/// either holds, does not, or could not be made from what this process can see.
171#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
172#[serde(rename_all = "lowercase")]
173pub enum CheckState {
174    /// The check holds.
175    Ok,
176    /// The check does not hold, and the public installation is affected.
177    Failed,
178    /// The check could not be made here; it says nothing either way.
179    Unknown,
180}
181
182/// One check: what was asked, what was seen, and — when it does not hold — why, and the
183/// command that changes it. The `cause` and `next` fields exist so that a report is
184/// actionable without a second document; a check that fails without naming its remedy is
185/// a diagnostic nobody can act on.
186#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
187pub struct InstallCheck {
188    /// A short stable name: `model`, `version`, `targets`, `release`, `artifacts`, `metadata`.
189    pub id: String,
190    /// Whether it holds.
191    pub state: CheckState,
192    /// What was observed, in one line.
193    pub observed: String,
194    /// Why it does not hold. Absent when it does.
195    pub cause: Option<String>,
196    /// The command or action that would change it. Absent when there is nothing to do.
197    pub next: Option<String>,
198}

200/// Whether the advertised one-line installation works right now, and if not, what is missing.
201///
202/// This answers the operator's actual question — *can a machine that has never seen this
203/// project install it with the published command?* — from local state alone. It performs no
204/// network access: a repository can be offline and this still answers, because everything it
205/// needs is the distribution model and the release records the repository itself carries.
206/// What it cannot see it reports as `unknown` rather than guessing; the public half of the
207/// question is answered by the release workflow's smoke phase, which installs from the
208/// published URL on every native runner and is the only thing that proves the public path.
209#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
210pub struct InstallabilityReport {
211    /// True when every check holds: a stable release is recorded and complete.
212    pub installable: bool,
213    /// One line for a person: what the state is, in the project's own terms.
214    pub summary: String,
215    /// The version this tree would release.
216    pub local_version: String,
217    /// The tag an unpinned installation resolves to, when a stable release is recorded.
218    pub stable_tag: Option<String>,
219    /// How many targets a release must publish for that release to be complete.
220    pub required_targets: usize,
221    /// How many of them the stable release actually publishes.
222    pub published_artifacts: usize,
223    /// The command a person would run, whether or not it currently works.
224    pub install_command: String,
225    /// Where the installer is served.
226    pub installer_url: String,
227    /// Where the stable pointer is served.
228    pub latest_url: String,
229    /// Every check, in the order they are worth reading.
230    pub checks: Vec<InstallCheck>,
231}

233/// The model this process was started with, or the reason there is none.
234fn model(ctx: &Context) -> Result<&Model, CapabilityError> {
235    ctx.index.distribution.as_ref().ok_or_else(|| {
236        CapabilityError::NotFound(
237            "this installation carries no share/distribution.yaml, so it declares no distribution"
238                .into(),

243fn distribution(ctx: &Context, _: Empty) -> Result<DistributionReport, CapabilityError> {
244    let m = model(ctx)?;
245    Ok(DistributionReport {
246        binary: m.project.binary.clone(),
247        repository: m.project.repository.clone(),
248        installer_url: m.installer_url(),
249        install_command: m.install_command(),
250        install_and_init_command: m.install_and_init_command(),
251        next_command: format!("{} init", m.project.binary),
252        checksum: m.archive.checksum.clone(),
253        install_dir: m.installer.install_dir.clone(),
254        prefix: m.installer.prefix.clone(),
255        latest_url: m.release_url(crate::distribution::release::LATEST),
256        supported: m.published().count(),
257        targets: m
258            .targets
259            .iter()
260            .map(|t| TargetView {
261                id: t.id.clone(),
262                title: t.title(&m.titles),
263                os: t.os,
264                arch: t.arch,
265                libc: t.libc,
266                rust_target: t.rust_target.clone(),
267                status: t.status,
268                reason: t.reason.clone(),
269                artifact: t.artifact_name(
270                    &m.project,
271                    &m.archive,
272                    crate::distribution::render::TAG_PLACEHOLDER,
273                ),
274            })
275            .collect(),
276    })

302fn build(ctx: &Context, _: Empty) -> Result<BuildReport, CapabilityError> {
303    let distribution_target = ctx
304        .index
305        .distribution
306        .as_ref()
307        .and_then(|m| m.targets.iter().find(|t| t.rust_target == crate::TARGET))
308        .map(|t| t.id.clone());
309    Ok(BuildReport {
310        version: crate::VERSION.to_string(),
311        target: crate::TARGET.to_string(),
312        profile: crate::PROFILE.to_string(),
313        commit: crate::COMMIT.to_string(),
314        distribution_target,
315    })
316}

318fn artifact(
319    ctx: &Context,
320    input: ReleaseArtifactInput,
321) -> Result<ReleaseArtifactView, CapabilityError> {
322    let m = model(ctx)?;
323    let t = m
324        .targets
325        .iter()
326        .find(|t| t.id == input.target || t.rust_target == input.target)
327        .ok_or_else(|| {
328            CapabilityError::NotFound(format!(
329                "the distribution model declares no target `{}`",
330                input.target
331            ))
332        })?;
333    if !t.status.is_published() {
334        return Err(CapabilityError::Refused(format!(
335            "`{}` is declared and not built: {}",
336            t.id,
337            t.reason.as_deref().unwrap_or("no reason recorded")
338        )));
339    }
340    let tag = input.tag;
341    if tag != crate::distribution::render::TAG_PLACEHOLDER
342        && !(tag.starts_with('v') && tag[1..].starts_with(|c: char| c.is_ascii_digit()))
343    {
344        return Err(CapabilityError::InvalidInput(format!(
345            "`{tag}` is not a tag: a tag is `v` followed by a version"
346        )));
347    }
348    let name = t.artifact_name(&m.project, &m.archive, &tag);
349    Ok(ReleaseArtifactView {
350        target: t.id.clone(),
351        rust_target: t.rust_target.clone(),
352        url: format!("{}{tag}/{name}", m.project.download_prefix()),
353        root: t.archive_root(&m.project, &tag),
354        name,
355        tag,
356    })
357}

359/// Whether the published one-line installation works, from what this repository can see.
360///
361/// The checks are ordered the way the pipeline is: the model must be valid before a target
362/// list means anything, targets must exist before a release can be complete, a release must
363/// be recorded before metadata can be published, and the metadata must agree with the model
364/// before an installer can use it. The first failure is the one that matters, and the
365/// summary names it; the rest are still reported, because an operator reading this wants the
366/// whole shape and not one line at a time.
367fn status(ctx: &Context, _: Empty) -> Result<InstallabilityReport, CapabilityError> {
368    let m = model(ctx)?;
369    let records = Releases::from_index(&ctx.index).map_err(CapabilityError::Internal)?;
370    Ok(installability(m, &records))
371}
372
373/// The report itself, over a model and a set of records and nothing else.
374///
375/// Separated from the capability so that the answer can be examined for states this
376/// repository is not in — a complete release, a partial one, a model that publishes
377/// nothing — without constructing an index for each. The capability above is the only
378/// caller in the executable; the tests are the rest.
379pub(crate) fn installability(m: &Model, records: &Releases) -> InstallabilityReport {
380    let required = m.published().count();
381    let mut checks = Vec::new();
382
383    // The model itself. `findings` is the same function `distribution validate` reports and
384    // the release workflow refuses a tag over; this asks it rather than restating any of it.
385    let model_findings = records.findings(m);
386    let model_ok = required > 0;
387    checks.push(InstallCheck {
388        id: "model".into(),
389        state: if model_ok {
390            CheckState::Ok
391        } else {
392            CheckState::Failed
393        },
394        observed: format!(
395            "{} declared target(s), {required} published",
396            m.targets.len()
397        ),
398        cause: (!model_ok)
399            .then(|| "the model publishes no target, so a release could build nothing".into()),
400        next: (!model_ok)
401            .then(|| "declare a target with `status: supported` in share/distribution.yaml".into()),
402    });
403
404    // The version this tree would release. It is the crate's, compiled in, so it is the same
405    // string the released executable would print.
406    let local_version = crate::VERSION.to_string();
407    checks.push(InstallCheck {
408        id: "version".into(),
409        state: CheckState::Ok,
410        observed: format!("this tree releases v{local_version}"),
411        cause: None,
412        next: None,
413    });
414
415    // A stable release, or none. This is the check that was failing while the public
416    // installer answered "no stable release is currently published".
417    let latest = records.latest_stable();
418    let stable_tag = latest.map(|r| r.tag.clone());
419    checks.push(InstallCheck {
420        id: "release".into(),
421        state: if latest.is_some() {
422            CheckState::Ok
423        } else {
424            CheckState::Failed
425        },
426        observed: match &stable_tag {
427            Some(t) => format!("the stable channel resolves to {t}"),
428            None => format!(
429                "no stable release is recorded ({} record(s) in total)",
430                records.releases.len()
431            ),
432        },
433        cause: latest.is_none().then(|| {
434            "there is no release record for the stable channel, so nothing derives the public \
435             metadata an unpinned installation reads"
436                .into()
437        }),
438        next: latest
439            .is_none()
440            .then(|| format!("git tag v{local_version} && git push origin v{local_version}")),
441    });
442
443    // Completeness: a release is all of its supported targets or it is not a release.
444    let published_artifacts = latest.map_or(0, |r| r.artifacts.len());
445    checks.push(InstallCheck {
446        id: "artifacts".into(),
447        state: match latest {
448            None => CheckState::Unknown,
449            Some(_) if published_artifacts >= required => CheckState::Ok,
450            Some(_) => CheckState::Failed,
451        },
452        observed: match latest {
453            None => "no release to count artifacts for".into(),
454            Some(_) => format!("{published_artifacts} of {required} required artifact(s)"),
455        },
456        cause: latest
457            .filter(|_| published_artifacts < required)
458            .map(|_| "the stable release does not publish every target the model requires; a partial release is not a release".into()),
459        next: latest
460            .filter(|_| published_artifacts < required)
461            .map(|_| "withdraw the record (yanked: true) and release again; the pipeline refuses to publish a partial release, so a record like this was not written by it".into()),
462    });
463
464    // Whether every record agrees with the model. A record that disagrees renders metadata
465    // the installer would refuse, so this is part of installability and not merely of hygiene.
466    checks.push(InstallCheck {
467        id: "metadata".into(),
468        state: if model_findings.is_empty() {
469            CheckState::Ok
470        } else {
471            CheckState::Failed
472        },
473        observed: if model_findings.is_empty() {
474            format!("{} record(s) agree with the model", records.releases.len())
475        } else {
476            format!(
477                "{} finding(s): {}",
478                model_findings.len(),
479                model_findings.join("; ")
480            )
481        },
482        cause: (!model_findings.is_empty()).then(|| {
483            "a release record states something the distribution model does not derive".into()
484        }),
485        next: (!model_findings.is_empty()).then(|| "majordomus distribution validate".into()),
486    });
487
488    // The public half is deliberately not guessed at. Nothing here reaches the network, so
489    // the only honest thing to say about the served bytes is who proves them.
490    checks.push(InstallCheck {
491        id: "public".into(),
492        state: CheckState::Unknown,
493        observed: "not checked here; this report reaches no network".into(),
494        cause: None,
495        next: Some(format!(
496            "curl -fsSL {} | head -1",
497            m.release_url(crate::distribution::release::LATEST)
498        )),
499    });
500
501    let installable = checks.iter().all(|c| c.state != CheckState::Failed);
502    let summary = if installable {
503        match &stable_tag {
504            Some(t) => format!(
505                "{t} is recorded, complete and consistent; the published command installs it"
506            ),
507            None => "installable".into(),
508        }
509    } else {
510        checks
511            .iter()
512            .find(|c| c.state == CheckState::Failed)
513            .and_then(|c| c.cause.clone())
514            .unwrap_or_else(|| "the published installation command does not currently work".into())
515    };
516
517    InstallabilityReport {
518        installable,
519        summary,
520        local_version,
521        stable_tag,
522        required_targets: required,
523        published_artifacts,
524        install_command: m.install_command(),
525        installer_url: m.installer_url(),
526        latest_url: m.release_url(crate::distribution::release::LATEST),
527        checks,
528    }
529}
