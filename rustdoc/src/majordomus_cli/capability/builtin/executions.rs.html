1//! The `executions` module: starting a capability as an execution, reading what it is
2//! doing, and asking it to stop.
3//!
4//! Every capability here is about executions; none of them is an execution engine of its
5//! own. What runs the work is [`crate::execution::ExecutionEngine`], reached through the
6//! context every handler already has, so a browser posting to `/api/v1/executions/start`,
7//! an agent calling `majordomus_execution_start` and a terminal running `majordomus run`
8//! reach one implementation and one store.
9//!
10//! There is no capability here that lists what may be run: that is `capabilities.list`,
11//! which already answers it, and a second list would be the same fact written twice. Every
12//! executable capability of the registry can be started as an execution, and the
13//! descriptor's own `execution` policy is what says what running it means.
14

50/// Where to read more about one execution, in this server's own terms.
51///
52/// Every value is derived from the registry's declared routes and from the live channel's
53/// own constant, so a route that moves moves here too and no client holds a path this
54/// server does not serve.
55#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
56pub struct ExecutionLinks {
57    /// The snapshot: `GET /api/v1/executions/get?id=…`.
58    #[serde(rename = "self")]
59    pub itself: String,
60    /// The retained history: `GET /api/v1/executions/events?id=…`.
61    pub events: String,
62    /// Where to ask it to stop: `POST /api/v1/executions/cancel`.
63    pub cancel: String,
64    /// The live channel, from the beginning: `GET /events?execution=…`.
65    pub websocket: String,
66    /// The page a person opens.
67    pub cockpit: String,
68}
69
70impl ExecutionLinks {
71    /// The links for one execution, built from what the registry declares.
72    pub fn of(ctx: &Context, id: &ExecutionId) -> Self {
73        let route = |capability: &str| {
74            ctx.registry
75                .get(capability)
76                .and_then(|c| c.exposure.http.as_ref())
77                .map(|h| h.path.clone())
78                .unwrap_or_default()
79        };
80        let query = crate::http::router::percent_encode(id.as_str());
81        ExecutionLinks {
82            itself: format!("{}?id={query}", route("executions.get")),
83            events: format!("{}?id={query}", route("executions.events")),
84            cancel: route("executions.cancel"),
85            websocket: format!("{}?execution={query}", crate::http::events::PATH),
86            cockpit: format!("{}/executions/{query}", crate::cockpit::PREFIX),
87        }
88    }
89}

108fn parse_id(text: &str) -> Result<ExecutionId, CapabilityError> {
109    ExecutionId::parse(text).ok_or_else(|| {
110        CapabilityError::InvalidInput(format!(
111            "'{text}' is not an execution id; they look like x-20260908T010203Z-0a1b2c3d"
112        ))
113    })

116fn actor(ctx: &Context) -> Actor {
117    match &ctx.caller {
118        Some(peer) => Actor {
119            kind: ActorKind::Mcp,
120            peer: Some(peer.to_string()),
121            pid: None,

123        None => Actor::of(ActorKind::Http),

131/// The input of `executions.start`: which capability to run, and with what.
132pub struct StartInput {
133    /// The canonical id of the capability to run (`health.report`, `objects.verify`).
134    pub capability: String,
135    /// Its input, as its own input schema describes it; an empty object when it takes none.
136    #[serde(default)]
137    pub input: Option<Value>,
138    /// A key that makes a repeated start one execution (I4000): sent again with the same
139    /// capability and input, it is answered the execution it started, `deduplicated`, and
140    /// nothing new runs; sent again with another input, it is refused 409 `conflict`,
141    /// naming that execution. 1 to 128 characters of `A-Z a-z 0-9 . _ : -`, scoped to this
142    /// repository and the capability, and held for as long as the server holds the
143    /// execution it started. Without one, every start starts an execution.
144    #[serde(default, skip_serializing_if = "Option::is_none")]
145    #[schemars(length(min = 1, max = 128), regex(pattern = r"^[A-Za-z0-9._:-]+$"))]
146    pub idempotency_key: Option<String>,
147    /// The issue the execution works for (I4004): `I4004`, `#184`. Before the handler is
148    /// entered the execution takes the issue's exclusive mesh claim — the issue record's
149    /// `scope`, or `issue/<id>` when it declares none — as the mesh session
150    /// `execution-<id>`, and moves `queued -> claimed -> running`; a claim another session
151    /// holds refuses it (`refused`, the holder named) and nothing runs. The claim is given
152    /// back at the final state and kept for an outcome that is not known. With no mesh in
153    /// the serving process it runs unclaimed and its `execution.issue_claim` event says so.
154    #[serde(default, skip_serializing_if = "Option::is_none")]
155    #[schemars(length(min = 1, max = 64), regex(pattern = r"^[A-Za-z0-9#._-]+$"))]
156    pub issue: Option<String>,
157}
158
159impl BenchmarkCases for StartInput {
160    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
161        vec![NamedCase::new(
162            "demonstrate",
163            StartInput {
164                capability: "executions.demonstrate".into(),
165                input: Some(serde_json::json!({ "steps": 1, "delay_ms": 0 })),
166                idempotency_key: None,
167                issue: None,
168            },
169        )]
170    }
171}

173/// The answer of `executions.start`: the execution, flattened as every other execution
174/// answer is, and whether this start began it or was answered one its idempotency key had
175/// already started.
176///
177/// ```
178/// use majordomus_cli::capability::builtin::StartedExecution;
179/// use majordomus_cli::synthetic::SyntheticRepository;
180/// let repo = SyntheticRepository::small().unwrap();
181/// let ctx = repo.context().unwrap();
182/// let input = serde_json::json!({
183///     "capability": "executions.demonstrate",
184///     "input": { "steps": 1, "delay_ms": 0 },
185///     "idempotency_key": "click-1",
186/// });
187/// let start = || -> StartedExecution {
188///     serde_json::from_value(ctx.execute("executions.start", input.clone()).unwrap()).unwrap()
189/// };
190/// let (first, again) = (start(), start());
191/// assert!(!first.deduplicated && again.deduplicated);
192/// assert_eq!(first.view.execution.id, again.view.execution.id);
193/// ```
194#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
195pub struct StartedExecution {
196    /// The execution, with the links to follow it.
197    #[serde(flatten)]
198    pub view: ExecutionView,
199    /// `true` when this start began nothing: its idempotency key had already started this
200    /// execution with the same input, and the answer is that execution as it stands now.
201    /// `false` for an execution this start began.
202    pub deduplicated: bool,
203}
204
205fn executions_start(ctx: &Context, input: StartInput) -> Result<StartedExecution, CapabilityError> {
206    let key = input
207        .idempotency_key
208        .as_deref()
209        .map(IdempotencyKey::parse)
210        .transpose()
211        .map_err(CapabilityError::InvalidInput)?;
212    let payload = input.input.unwrap_or_else(|| serde_json::json!({}));
213    let options = StartOptions {
214        idempotency_key: key,
215        issue: input.issue,
216    };
217    match ctx
218        .executions
219        .submit_with(ctx, &input.capability, payload, actor(ctx), options)
220    {
221        Ok(started) => Ok(StartedExecution {
222            view: ExecutionView::of(ctx, started.execution),
223            deduplicated: started.deduplicated,
224        }),
225        Err(SubmitError::Conflict(conflict)) => {
226            Err(CapabilityError::Conflict(conflict.to_string()))
227        }
228        Err(SubmitError::UnknownCapability(id)) => Err(CapabilityError::NotFound(format!(
229            "no capability '{id}'; the executable ones are listed by capabilities.list"
230        ))),
231        Err(e @ SubmitError::NotExecutable(_)) => Err(CapabilityError::Refused(e.to_string())),
232        Err(SubmitError::InvalidInput(m)) => Err(CapabilityError::InvalidInput(m)),
233        Err(e @ SubmitError::Unavailable(_)) => Err(CapabilityError::Refused(e.to_string())),
234    }
235}

270#[derive(Debug, Serialize, Deserialize, JsonSchema)]
271/// The answer of `executions.list`.
272pub struct ExecutionList {
273    /// How many are answered here.
274    pub count: usize,
275    /// How many this process remembers in total, before the filter.
276    pub remembered: usize,
277    /// How many are not finished.
278    pub active: usize,
279    /// How many are waiting for a worker.
280    pub queued: usize,
281    /// How many live channels are following them.
282    pub live_channels: usize,
283    /// The executions, newest first.
284    pub executions: Vec<ExecutionView>,
285}
286
287fn executions_list(ctx: &Context, input: ListInput) -> Result<ExecutionList, CapabilityError> {
288    let store = ctx.executions.store();
289    store.abandon_silent(REPORT_SILENCE);
290    let limit = input.limit.unwrap_or(MAX_LIST).min(MAX_LIST);
291    let executions = store.list(input.state, input.capability.as_deref(), limit);
292    Ok(ExecutionList {
293        count: executions.len(),
294        remembered: store.len(),
295        active: store.active(),
296        queued: ctx.executions.queued(),
297        live_channels: crate::http::events::open_connections(),
298        executions: executions
299            .into_iter()
300            .map(|e| ExecutionView::of(ctx, e))
301            .collect(),
302    })
303}

357#[derive(Debug, Serialize, Deserialize, JsonSchema)]
358/// The answer of `executions.events`: one page of an execution's retained history.
359pub struct EventHistory {
360    /// The execution the events belong to.
361    pub execution_id: String,
362    /// Where the execution stands now, so a client needs one request rather than two.
363    pub state: ExecutionState,
364    /// The events, oldest first.
365    pub events: Vec<ExecutionEvent>,
366    /// The sequence of the last event here: the cursor for the next page, and the one to
367    /// open the live channel with.
368    pub last_sequence: u64,
369    /// Whether more events follow this page right now.
370    pub more: bool,
371    /// Whether events before this page had already been dropped by the store's bound.
372    pub truncated: bool,
373}
374
375fn executions_events(ctx: &Context, input: EventsInput) -> Result<EventHistory, CapabilityError> {
376    let id = parse_id(&input.id)?;
377    let store = ctx.executions.store();
378    store.abandon_silent(REPORT_SILENCE);
379    let snapshot = store
380        .get(&id)
381        .ok_or_else(|| CapabilityError::NotFound(format!("no execution {id} in this process")))?;
382    let limit = input.limit.unwrap_or(MAX_EVENTS).min(MAX_EVENTS);
383    let page = store
384        .events(&id, input.after.unwrap_or(0), limit)
385        .ok_or_else(|| CapabilityError::NotFound(format!("no execution {id} in this process")))?;
386    Ok(EventHistory {
387        execution_id: id.to_string(),
388        state: snapshot.state,
389        events: page.events.iter().map(|e| (**e).clone()).collect(),
390        last_sequence: page.last_sequence,
391        more: page.more,
392        truncated: page.truncated,
393    })
394}

412#[derive(Debug, Serialize, Deserialize, JsonSchema)]
413/// The answer of `executions.cancel`.
414pub struct CancelReport {
415    /// What asking achieved: `requested`, `already_requested`, `already_finished`, or
416    /// `outcome_unknown` for an execution whose outcome waits for a reconciliation, which a
417    /// stop cannot change.
418    pub outcome: String,
419    /// Whether the capability declares that it looks at its cancellation flag. When it
420    /// does not, the request is recorded and the execution runs to completion.
421    pub cancellable: bool,
422    /// The execution as it stands after the request.
423    pub execution: ExecutionView,
424}
425
426fn executions_cancel(ctx: &Context, input: CancelInput) -> Result<CancelReport, CapabilityError> {
427    let id = parse_id(&input.id)?;
428    let by = match &ctx.caller {
429        Some(peer) => peer.to_string(),
430        None => "client".to_string(),
431    };
432    let outcome = match ctx.executions.cancel(&id, &by) {
433        CancelOutcome::Requested => "requested",
434        CancelOutcome::AlreadyRequested => "already_requested",
435        CancelOutcome::AlreadyFinished(_) => "already_finished",
436        CancelOutcome::OutcomeUnknown => "outcome_unknown",
437        CancelOutcome::Unknown => {
438            return Err(CapabilityError::NotFound(format!(
439                "no execution {id} in this process"
440            )))
441        }
442    };
443    let execution =
444        ctx.executions.store().get(&id).ok_or_else(|| {
445            CapabilityError::NotFound(format!("no execution {id} in this process"))
446        })?;
447    let cancellable = execution.cancellable;
448    Ok(CancelReport {
449        outcome: outcome.into(),
450        cancellable,
451        execution: ExecutionView::of(ctx, execution),
452    })
453}

490#[derive(Debug, Serialize, Deserialize, JsonSchema)]
491/// The answer of `executions.report`.
492pub struct ReportAnswer {
493    /// How many of the reported events were applied.
494    pub applied: usize,
495    /// The reporter's sequence this server holds everything up to: the next report sends
496    /// the events after it, so a lost or repeated report costs nothing.
497    pub through: u64,
498    /// Whether somebody asked this server to stop the execution. The reporter runs it, so
499    /// the reporter cancels it.
500    pub cancel_requested: bool,
501    /// The execution as this server now holds it.
502    pub execution: ExecutionView,
503}
504
505fn executions_report(ctx: &Context, input: ReportInput) -> Result<ReportAnswer, CapabilityError> {
506    let id = parse_id(&input.id)?;
507    if input.events.len() > MAX_EVENTS {
508        return Err(CapabilityError::InvalidInput(format!(
509            "{} events in one report; at most {MAX_EVENTS}, and the answer's `through` says where the next one starts",
510            input.events.len()
511        )));
512    }
513    if let Some(other) = input.events.iter().find(|e| e.execution_id != id) {
514        return Err(CapabilityError::InvalidInput(format!(
515            "an event of {} in a report about {id}",
516            other.execution_id
517        )));
518    }
519    let store = ctx.executions.store();
520    store.abandon_silent(REPORT_SILENCE);
521    let stored = match ctx.registry.get(&input.capability) {
522        Some(c) => redact(&c.input.schema, &input.input),
523        None => input.input,
524    };
525    store
526        .adopt(
527            id.clone(),
528            Seed {
529                capability: &input.capability,
530                title: &input.title,
531                input: stored,
532                cancellable: input.cancellable,
533                actor: Actor::reported(input.pid),
534                repository: repository_of(ctx),
535            },
536            input.pid,
537        )
538        .map_err(|e| CapabilityError::Refused(e.to_string()))?;
539    // adopted just now as a reported execution, so the store answers both readings; the
540    // refusal is named once, eagerly, rather than on two paths no report can take
541    let refused = CapabilityError::NotFound(format!("no reported execution {id} in this process"));
542    store
543        .report(
544            &id,
545            input.events.into_iter().map(|e| (e.sequence, e.payload)),
546        )
547        .zip(store.get(&id))
548        .map(|(outcome, execution)| ReportAnswer {
549            applied: outcome.applied,
550            through: outcome.through,
551            cancel_requested: outcome.cancel_requested,
552            execution: ExecutionView::of(ctx, execution),
553        })
554        .ok_or(refused)
555}

723#[derive(Debug, Serialize, Deserialize, JsonSchema)]
724/// The answer of `executions.demonstrate`.
725pub struct DemonstrateReport {
726    /// How many steps ran.
727    pub steps: u64,
728    /// How long it took, in milliseconds.
729    pub elapsed_ms: u64,
730    /// Whether anything was watching: false when it was called directly rather than as an
731    /// execution, which is the difference this capability exists to show.
732    pub observed: bool,
733}
734
735/// The longest one step of the demonstration may take.
736const MAX_STEP_MS: u64 = 10_000;
737
738/// The most steps the demonstration will walk.
739const MAX_STEPS: u64 = 100;
740
741fn executions_demonstrate(
742    ctx: &Context,
743    input: DemonstrateInput,
744) -> Result<DemonstrateReport, CapabilityError> {
745    let steps = input.steps.min(MAX_STEPS);
746    let delay = std::time::Duration::from_millis(input.delay_ms.min(MAX_STEP_MS));
747    let started = std::time::Instant::now();
748    let p = &ctx.progress;
749    for step in 1..=steps {
750        if p.cancelled() {
751            return Err(p.cancellation());
752        }
753        let name = format!("step-{step}");
754        p.step(&name, &format!("Step {step} of {steps}"));
755        p.log(format!("step {step} of {steps} is working"));
756        if !delay.is_zero() {
757            // slept in slices so that cancellation is noticed within a tenth of a second
758            // however long a step is
759            let slice = std::time::Duration::from_millis(100);
760            let mut left = delay;
761            while !left.is_zero() {
762                if p.cancelled() {
763                    p.step_done(&name, false, Some("cancelled".into()));
764                    return Err(p.cancellation());
765                }
766                let take = left.min(slice);
767                std::thread::sleep(take);
768                left -= take;
769            }
770        }
771        if input.fail_at == Some(step) {
772            p.step_done(&name, false, Some("asked to fail here".into()));
773            return Err(CapabilityError::Internal(format!(
774                "the demonstration was asked to fail at step {step} of {steps}"
775            )));
776        }
777        p.step_done(&name, true, None);
778        p.progress(step, Some(steps), format!("{step} of {steps} done"));
779    }
780    Ok(DemonstrateReport {
781        steps,
782        elapsed_ms: started.elapsed().as_millis() as u64,
783        observed: p.is_reporting(),
784    })
785}
