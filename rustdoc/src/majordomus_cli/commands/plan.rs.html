1//! `majordomus plan delivery` and `majordomus plan drift`: every issue's furthest delivery
2//! stage, and every contradiction between it and the declared status, through the registry's
3//! own `plan.delivery` and `plan.drift` capabilities.
4//!
5//! This file renders; it infers nothing. The stage, its proof and what was left unread are
6//! the capability's output, so the command line cannot call an issue merged that the HTTP
7//! route and the MCP tool would not. An unsettled stage is printed with the sources that
8//! could have moved it, never as a bare `none`.
9

19/// Run `majordomus plan <subcommand>`.
20pub fn run(args: PlanArgs) -> Result<u8> {
21    let app = App::load(&args.repo)?;
22    let v = match &args.command {
23        PlanCommand::Delivery { issue, at } => {
24            let mut input = json!({});
25            if let Some(i) = issue {
26                input["issue"] = json!(i);
27            }
28            if let Some(a) = at {
29                input["at"] = json!(a);
30            }
31            execute(&app.context, &["plan", "delivery"], input)?
32        }
33        PlanCommand::Drift { issue, at, gate } => {
34            let mut input = json!({});
35            if let Some(i) = issue {
36                input["issue"] = json!(i);
37            }
38            if let Some(a) = at {
39                input["at"] = json!(a);
40            }
41            if let Some(g) = gate {
42                input["gate"] = json!(match g {
43                    DriftGateArg::Head => "head",
44                    DriftGateArg::Trunk => "trunk",
45                });
46            }
47            execute(&app.context, &["plan", "drift"], input)?
48        }
49    };
50    let drift = matches!(args.command, PlanCommand::Drift { .. });
51    let body = match args.format {
52        OutputFormat::Json => format!("{v:#}"),
53        OutputFormat::Text if drift => drift_text(&v),
54        OutputFormat::Text => delivery_text(&v),
55    };
56    writeln!(std::io::stdout().lock(), "{body}").map_err(Error::Transport)?;
57    // a gate's verdict is its exit code; a report exits 0
58    Ok(if drift {
59        v["exit_code"]
60            .as_u64()
61            .and_then(|c| u8::try_from(c).ok())
62            .unwrap_or(0)
63    } else {
64        0
65    })
66}

68fn execute(ctx: &Context, path: &[&str], input: Value) -> Result<Value> {
69    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
70    let id = ctx
71        .registry
72        .by_cli(&words)
73        .map(|c| c.id.to_string())
74        .ok_or_else(|| Error::Protocol {
75            reason: format!(
76                "no capability is exposed as `majordomus {}`",
77                path.join(" ")
78            ),
79        })?;
80    ctx.execute(&id, input).map_err(|e| match e {
81        CapabilityError::NotFound(reason) => Error::NotFound { reason },
82        other => Error::Protocol {
83            reason: other.to_string(),
84        },
85    })
86}

88fn s<'a>(v: &'a Value, pointer: &str) -> &'a str {
89    v.pointer(pointer).and_then(Value::as_str).unwrap_or("")

92fn short(sha: &str) -> &str {
93    sha.get(..11).unwrap_or(sha)
94}

96/// What proved one issue's stage, in one phrase.
97fn proof(i: &Value) -> String {
98    match s(i, "/stage") {
99        "deployed" => format!(
100            "{} published as {} ({})",
101            s(i, "/released/tag"),
102            s(i, "/deployed/version"),
103            s(i, "/deployed/latest_json")
104        ),
105        "released" => format!(
106            "{} contains {}",
107            s(i, "/released/tag"),
108            short(s(i, "/merged/commit"))
109        ),
110        "merged" => {
111            let pr = i
112                .pointer("/merged/pull_request")
113                .and_then(Value::as_u64)
114                .map(|n| format!("#{n} "))
115                .unwrap_or_default();
116            format!("{pr}{}", short(s(i, "/merged/commit")))
117        }
118        "pr_open" => format!(
119            "#{} {}",
120            i.pointer("/pr_open/number")
121                .and_then(Value::as_u64)
122                .unwrap_or(0),
123            s(i, "/pr_open/head_ref")
124        ),
125        "branch" => s(i, "/branch/name").to_string(),
126        _ => String::new(),
127    }
128}

130fn delivery_text(v: &Value) -> String {
131    let mut out = format!(
132        "plan delivery — {} at {} {} ({})\n",
133        s(v, "/observed/mode"),
134        s(v, "/observed/trunk"),
135        short(s(v, "/observed/commit")),
136        s(v, "/observed/observed_at")
137    );
138    for src in v["sources"].as_array().into_iter().flatten() {
139        out.push_str(&format!(
140            "  {:<12} {:<7} {}\n",
141            s(src, "/name"),
142            s(src, "/state"),
143            s(src, "/detail")
144        ));
145    }
146    let tallies: Vec<String> = v["stages"]
147        .as_array()
148        .into_iter()
149        .flatten()
150        .filter_map(Value::as_str)
151        .map(|st| format!("{st} {}", v["tallies"][st].as_u64().unwrap_or(0)))
152        .collect();
153    out.push_str(&format!(
154        "stages: {} · unsettled {}\n\n",
155        tallies.join(" · "),
156        v["unsettled"].as_u64().unwrap_or(0)
157    ));
158    for i in v["issues"].as_array().into_iter().flatten() {
159        let unread: Vec<&str> = i["unread"]
160            .as_array()
161            .into_iter()
162            .flatten()
163            .filter_map(Value::as_str)
164            .collect();
165        let tail = if unread.is_empty() {
166            proof(i)
167        } else {
168            format!("{} (unread: {})", proof(i), unread.join(", "))
169        };
170        out.push_str(
171            format!(
172                "{:<6} {:<8} {:<9} {}\n",
173                s(i, "/issue"),
174                s(i, "/status"),
175                s(i, "/stage"),
176                tail.trim()
177            )
178            .as_str(),
179        );
180    }
181    out.trim_end().to_string()
182}

184/// One finding's proof, in one phrase.
185fn drift_proof(f: &Value) -> String {
186    let mut parts: Vec<String> = Vec::new();
187    if let Some(n) = f.pointer("/proof/pull_request").and_then(Value::as_u64) {
188        parts.push(format!("#{n}"));
189    }
190    if !s(f, "/proof/merge_commit").is_empty() {
191        parts.push(short(s(f, "/proof/merge_commit")).to_string());
192    }
193    if !s(f, "/proof/merged_at").is_empty() {
194        parts.push(format!("merged {}", s(f, "/proof/merged_at")));
195    }
196    if !s(f, "/proof/release").is_empty() {
197        parts.push(format!("in {}", s(f, "/proof/release")));
198    }
199    if !s(f, "/proof/started_at").is_empty() {
200        parts.push(format!("started {}", s(f, "/proof/started_at")));
201    }
202    if let Some(age) = f.pointer("/age_seconds").and_then(Value::as_u64) {
203        parts.push(format!("age {}h", age / 3600));
204    }
205    if parts.is_empty() {
206        parts.push("no merge names it".into());
207    }
208    parts.join(" ")
209}

211fn drift_text(v: &Value) -> String {
212    let mut out = format!(
213        "plan drift — {} at {} {} ({})\n",
214        s(v, "/observed/mode"),
215        s(v, "/observed/trunk"),
216        short(s(v, "/observed/commit")),
217        s(v, "/observed/observed_at")
218    );
219    out.push_str(&format!(
220        "  window       {:<7} {}: {}\n",
221        s(v, "/window/state"),
222        s(v, "/window/model"),
223        s(v, "/window/detail")
224    ));
225    for src in v["sources"].as_array().into_iter().flatten() {
226        out.push_str(&format!(
227            "  {:<12} {:<7} {}\n",
228            s(src, "/name"),
229            s(src, "/state"),
230            s(src, "/detail")
231        ));
232    }
233    let tallies: Vec<String> = v["classes"]
234        .as_array()
235        .into_iter()
236        .flatten()
237        .filter_map(Value::as_str)
238        .map(|c| format!("{c} {}", v["tallies"][c].as_u64().unwrap_or(0)))
239        .collect();
240    out.push_str(&format!("classes: {}\n\n", tallies.join(" · ")));
241    for f in v["findings"].as_array().into_iter().flatten() {
242        let tag = match s(f, "/verdict") {
243            "refused" => "REFUSED",
244            "pending" => "PENDING",
245            "reported" => "NOTE",
246            "advisory" => "ADVISORY",
247            _ => "FINDING",
248        };
249        let unread: Vec<&str> = f["unread"]
250            .as_array()
251            .into_iter()
252            .flatten()
253            .filter_map(Value::as_str)
254            .collect();
255        let unread = if unread.is_empty() {
256            String::new()
257        } else {
258            format!(" (unread: {})", unread.join(", "))
259        };
260        out.push_str(&format!(
261            "{:<8} {:<6} {:<8} {:<19} {}{} — {}\n",
262            tag,
263            s(f, "/issue"),
264            s(f, "/status"),
265            s(f, "/class"),
266            drift_proof(f),
267            unread,
268            s(f, "/remedy")
269        ));
270    }
271    let gate = s(v, "/gate");
272    if gate.is_empty() {
273        out.push_str(&format!(
274            "plan drift: {} finding(s), reported; no gate was asked",
275            v["findings"].as_array().map_or(0, Vec::len)
276        ));
277    } else {
278        out.push_str(&format!(
279            "plan drift ({gate}): {} — {} refused, {} pending{}",
280            s(v, "/verdict"),
281            v["refused"].as_u64().unwrap_or(0),
282            v["pending"].as_u64().unwrap_or(0),
283            match s(v, "/verdict") {
284                "advisory" => " (drift.enforcement is advisory: not enforced)",
285                "unjudged" => " (no window: nothing can be called young)",
286                _ => "",
287            }
288        ));
289    }
290    out
291}

298    fn every_stage_names_its_proof_and_unread_sources_are_said() {
299        let v = json!({
300            "observed": {"mode": "at", "trunk": "HEAD", "commit": "0123456789abcdef", "observed_at": "t"},
301            "sources": [{"name": "forge", "state": "unread", "detail": "never observed"}],
302            "stages": ["none", "merged"],
303            "tallies": {"none": 1, "merged": 1},
304            "unsettled": 1,
305            "issues": [
306                {"issue": "I0001", "status": "READY", "stage": "none", "unread": ["forge"]},
307                {"issue": "I0002", "status": "DONE", "stage": "merged", "unread": [],
308                 "merged": {"commit": "fedcba9876543210", "pull_request": 7}},
309                {"issue": "I0003", "status": "DONE", "stage": "released", "unread": [],
310                 "merged": {"commit": "fedcba9876543210"}, "released": {"tag": "v0.1.0"}},
311                {"issue": "I0004", "status": "DONE", "stage": "deployed", "unread": [],
312                 "released": {"tag": "v0.1.0"}, "deployed": {"version": "0.2.0", "latest_json": "x"}},
313                {"issue": "I0005", "status": "ACTIVE", "stage": "pr_open", "unread": [],
314                 "pr_open": {"number": 9, "head_ref": "fix/y"}},
315                {"issue": "I0006", "status": "ACTIVE", "stage": "branch", "unread": ["forge"],
316                 "branch": {"name": "feature/z"}}
317            ]
318        });
319        let t = delivery_text(&v);
320        assert!(
321            t.contains("plan delivery — at at HEAD 0123456789a (t)"),
322            "{t}"
323        );
324        assert!(t.contains("forge        unread  never observed"), "{t}");
325        assert!(t.contains("stages: none 1 · merged 1 · unsettled 1"), "{t}");
326        assert!(
327            t.contains("I0001  READY    none      (unread: forge)"),
328            "{t}"
329        );
330        assert!(t.contains("#7 fedcba98765"), "{t}");
331        assert!(t.contains("v0.1.0 contains fedcba98765"), "{t}");
332        assert!(t.contains("v0.1.0 published as 0.2.0 (x)"), "{t}");
333        assert!(t.contains("#9 fix/y"), "{t}");
334        assert!(t.contains("feature/z (unread: forge)"), "{t}");
335    }

369    fn every_verdict_has_its_tag_and_a_report_names_no_gate() {
370        let v = json!({
371            "observed": {"mode": "at", "trunk": "abc", "commit": "abc", "observed_at": "t"},
372            "window": {"model": "m", "state": "unread", "detail": "no window"},
373            "sources": [],
374            "classes": ["merged_not_recorded"],
375            "tallies": {"merged_not_recorded": 3},
376            "findings": [
377                {"issue": "I0001", "status": "READY", "class": "merged_not_recorded", "verdict": "pending",
378                 "proof": {"release": "v0.2.0"}, "unread": [], "remedy": "r1"},
379                {"issue": "I0002", "status": "ACTIVE", "class": "active_without_work", "verdict": "reported",
380                 "proof": {"started_at": "2026-10-01"}, "unread": [], "remedy": "r2"},
381                {"issue": "I0003", "status": "READY", "class": "merged_not_recorded",
382                 "proof": {}, "unread": [], "remedy": "r3"}
383            ],
384            "refused": 0, "pending": 1, "verdict": "report", "exit_code": 0
385        });
386        let t = drift_text(&v);
387        assert!(
388            t.contains("PENDING  I0001  READY    merged_not_recorded in v0.2.0 — r1"),
389            "{t}"
390        );
391        assert!(
392            t.contains("NOTE     I0002  ACTIVE   active_without_work started 2026-10-01 — r2"),
393            "{t}"
394        );
395        assert!(t.contains("FINDING  I0003"), "{t}");
396        assert!(
397            t.ends_with("plan drift: 3 finding(s), reported; no gate was asked"),
398            "{t}"
399        );
400
401        let mut unjudged = v.clone();
402        unjudged["gate"] = json!("head");
403        unjudged["verdict"] = json!("unjudged");
404        assert!(
405            drift_text(&unjudged).ends_with("(no window: nothing can be called young)"),
406            "{}",
407            drift_text(&unjudged)
408        );
409        let mut clean = v;
410        clean["gate"] = json!("head");
411        clean["verdict"] = json!("clean");
412        assert!(drift_text(&clean).ends_with("plan drift (head): clean — 0 refused, 1 pending"));
413    }

415    #[test]
416    fn a_capability_the_registry_lacks_or_refuses_is_a_typed_error() {
417        let repo = crate::synthetic::SyntheticRepository::small().expect("a repository");
418        let bare = Context::new(
419            std::sync::Arc::new(repo.index().expect("an index")),
420            std::sync::Arc::new(
421                crate::capability::CapabilityRegistry::builder()
422                    .build()
423                    .expect("an empty registry"),
424            ),
425        );
426        match execute(&bare, &["plan", "drift"], json!({})) {
427            Err(Error::Protocol { reason }) => {
428                assert_eq!(
429                    reason,
430                    "no capability is exposed as `majordomus plan drift`"
431                )
432            }
433            other => panic!("expected a protocol error, got {other:?}"),
434        }
435        let ctx = repo.context().expect("a context");
436        match execute(&ctx, &["plan", "delivery"], json!({"issue": "I9999"})) {
437            Err(Error::NotFound { reason }) => assert!(reason.contains("I9999"), "{reason}"),
438            other => panic!("expected not found, got {other:?}"),
439        }
440        match execute(&ctx, &["plan", "drift"], json!({"window": 60})) {
441            Err(Error::Protocol { reason }) => assert!(reason.contains("window"), "{reason}"),
442            other => panic!("expected a protocol error, got {other:?}"),
443        }
444        // a directory that is no git work tree has no history, and both say so
445        for path in [["plan", "delivery"], ["plan", "drift"]] {
446            match execute(&ctx, &path, json!({})) {
447                Err(Error::Protocol { reason }) => assert!(!reason.is_empty()),
448                other => panic!("{path:?} outside a repository answered {other:?}"),
449            }
450        }
451    }
