1//! Bounded retries for the forge's transient failures, and only for those.
2//!
3//! A timeout, a 5xx, a rate limit or a dropped connection says nothing about the pull
4//! request; asking again a little later is the right answer, a bounded number of times, with
5//! the wait doubling between attempts. Anything else — a refusal by the branch protection, a
6//! head that moved, an authentication failure, a pull request that does not exist — is the
7//! answer, and is returned at once: retrying a semantic failure only hammers the API with a
8//! question it already answered.
9//!
10//! What is retried is a read or an idempotent write. The merge itself is not
11//! ([`super::drain::ForgeIntegrator`]): a merge that timed out may have landed, and the
12//! verification after it — which *is* retried — is what finds out.
13
14use std::time::Duration;
15
16/// How many times, and how long to wait between them.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub struct Backoff {
19    /// Attempts in all, the first included. One means no retry.
20    pub attempts: u32,
21    /// The wait before the second attempt; doubled before each later one.
22    pub initial: Duration,
23    /// The longest single wait.
24    pub max: Duration,
25}
26
27/// The forge's policy: four attempts, waiting 2 s, 4 s and 8 s — under fifteen seconds in
28/// all, short enough that a drain step is not held for minutes by a forge that is down.
29pub const FORGE: Backoff = Backoff {
30    attempts: 4,
31    initial: Duration::from_secs(2),
32    max: Duration::from_secs(30),
33};
34
35impl Backoff {
36    /// The wait before attempt `n` (1-based, so `n >= 2`).
37    ///
38    /// ```text
39    /// FORGE.delay(2) == 2 s, FORGE.delay(3) == 4 s, FORGE.delay(4) == 8 s
40    /// ```text
41    pub fn delay(&self, n: u32) -> Duration {
42        let doublings = n.saturating_sub(2).min(16);
43        self.initial.saturating_mul(1u32 << doublings).min(self.max)
44    }
45}
46
47/// Whether a failure message describes a transient condition of the network or the forge,
48/// as `gh` and `git` word them. Conservative: a message that matches nothing is semantic.
49pub fn transient(message: &str) -> bool {
50    let m = message.to_ascii_lowercase();
51    const WORDS: &[&str] = &[
52        "timed out",
53        "timeout",
54        "rate limit",
55        "secondary rate",
56        "abuse detection",
57        "http 500",
58        "http 502",
59        "http 503",
60        "http 504",
61        "500 internal server error",
62        "502 bad gateway",
63        "503 service unavailable",
64        "504 gateway time",
65        "connection reset",
66        "connection refused",
67        "could not resolve host",
68        "temporary failure in name resolution",
69        "tls handshake",
70        "unexpected eof",
71        "the remote end hung up unexpectedly",
72        "early eof",
73    ];
74    WORDS.iter().any(|w| m.contains(w))
75}

77/// Run `op` until it succeeds, fails semantically, or `policy.attempts` are spent. `sleep`
78/// is how a wait is taken: the forge passes `std::thread::sleep`, a test records the waits.
79/// A failure after retries says how many attempts were made.
80pub fn with_backoff<T>(
81    policy: &Backoff,
82    sleep: &mut dyn FnMut(Duration),
83    mut op: impl FnMut() -> Result<T, String>,
84) -> Result<T, String> {
85    let mut attempt = 1;
86    loop {
87        match op() {
88            Ok(v) => return Ok(v),
89            Err(e) if transient(&e) && attempt < policy.attempts => {
90                attempt += 1;
91                sleep(policy.delay(attempt));
92            }
93            Err(e) if attempt > 1 => {
94                return Err(format!("{e} (after {attempt} attempts)"));
95            }
96            Err(e) => return Err(e),
97        }
98    }
99}
100
101/// [`with_backoff`] with the forge's policy and a real sleep.
102pub fn forge<T>(op: impl FnMut() -> Result<T, String>) -> Result<T, String> {
103    with_backoff(&FORGE, &mut std::thread::sleep, op)
104}

106#[cfg(test)]
107mod tests {
108    use super::*;
109
110    #[test]
111    fn a_transient_failure_is_retried_with_doubling_waits_and_then_succeeds() {
112        let mut waits = Vec::new();
113        let mut calls = 0;
114        let r = with_backoff(&FORGE, &mut |d| waits.push(d), || {
115            calls += 1;
116            if calls < 3 {
117                Err("HTTP 502: Bad Gateway (https://api.github.com/graphql)".to_string())
118            } else {
119                Ok(calls)
120            }
121        });
122        assert_eq!(r, Ok(3));
123        assert_eq!(waits, [Duration::from_secs(2), Duration::from_secs(4)]);
124    }
125
126    #[test]
127    fn a_semantic_failure_is_returned_at_once() {
128        let mut calls = 0;
129        let r: Result<(), String> = with_backoff(&FORGE, &mut |_| {}, || {
130            calls += 1;
131            Err(
132                "Pull request #7 is not mergeable: the base branch policy prohibits the merge"
133                    .into(),
134            )
135        });
136        assert_eq!(calls, 1, "a refusal is the answer, not an outage");
137        assert!(r.unwrap_err().starts_with("Pull request #7"));
138    }
139
140    #[test]
141    fn retries_are_bounded_and_the_failure_says_how_many() {
142        let mut waits = Vec::new();
143        let mut calls = 0;
144        let r: Result<(), String> = with_backoff(&FORGE, &mut |d| waits.push(d), || {
145            calls += 1;
146            Err("API rate limit exceeded for user".into())
147        });
148        assert_eq!(calls, FORGE.attempts as usize);
149        assert_eq!(waits.len(), FORGE.attempts as usize - 1);
150        let e = r.unwrap_err();
151        assert!(e.ends_with("(after 4 attempts)"), "{e}");
152    }
153
154    #[test]
155    fn the_wait_doubles_up_to_its_ceiling() {
156        let b = Backoff {
157            attempts: 10,
158            initial: Duration::from_secs(2),
159            max: Duration::from_secs(30),
160        };
161        let waits: Vec<u64> = (2..=7).map(|n| b.delay(n).as_secs()).collect();
162        assert_eq!(waits, [2, 4, 8, 16, 30, 30]);
163    }
164
165    #[test]
166    fn only_outage_words_are_transient() {
167        for t in [
168            "dial tcp: i/o timeout",
169            "error connecting to api.github.com: connection reset by peer",
170            "You have exceeded a secondary rate limit",
171            "fatal: the remote end hung up unexpectedly",
172            "HTTP 503",
173        ] {
174            assert!(transient(t), "{t}");
175        }
176        for s in [
177            "HTTP 401: Bad credentials",
178            "HTTP 404: Not Found",
179            "Head branch was modified. Review and try the merge again.",
180            "could not find pull request",
181            "",
182        ] {
183            assert!(!transient(s), "{s}");
184        }
185    }
