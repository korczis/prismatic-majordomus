1//! The mesh journal, folded into signals.
2//!
3//! The journal is read whole and in causal order, because what an event means depends on
4//! what came before it: a session's opening is news only when the session is new or what it
5//! says about itself changed (the server re-projects every attached board peer on each
6//! heartbeat, and those re-projections are most of the journal); a claim is news only when
7//! its session did not already hold the same one; an answer is for the reader when the
8//! request was theirs. Events up to the cursor only build that memory; events after it
9//! become signals or are counted as suppressed.
10

17/// A place in the journal: an event's causal key, unique and totally ordered.
18#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
19pub(crate) struct Position {
20    pub(crate) lamport: u64,
21    pub(crate) stream: String,
22    pub(crate) seq: u64,
23}
24
25impl Position {
26    pub(crate) fn of(event: &MeshEvent) -> Position {
27        Position {
28            lamport: event.lamport,
29            stream: event.stream.as_str().to_string(),
30            seq: event.seq,
31        }
32    }
33}

35/// What the fold needs to know beyond the events: who the reader is, which machine a
36/// node id is, and the oldest time a fresh page reaches back to.
37pub(crate) struct Context<'a> {
38    pub(crate) reader: &'a Reader,
39    /// Display names of nodes, by node id.
40    pub(crate) machines: &'a HashMap<String, String>,
41    /// Without a cursor, events older than this (Unix seconds) only build memory.
42    pub(crate) since: u64,
43}
44
45/// What makes two openings of one session the same news.
46type Fingerprint = [Option<String>; 7];
47
48/// What a session last said about itself that makes a change in it news. The head, the
49/// dirtiness and the context revision move on every commit and are not news.
50fn fingerprint(info: &SessionInfo) -> Fingerprint {
51    [
52        Some(info.client.clone()),
53        info.worker.clone(),
54        info.intent.clone(),
55        info.task.clone(),
56        info.issue.clone(),
57        info.milestone.clone(),
58        info.branch.clone(),
59    ]
60}

62#[derive(Default)]
63struct Memory {
64    /// Each session's fingerprint and how to call it, by (stream, session).
65    sessions: HashMap<(String, String), (Fingerprint, String)>,
66    /// The sessions whose arrival was shown, so that their departure is too.
67    shown: HashSet<(String, String)>,
68    /// Live claims by id: (stream, session, what makes it the same claim).
69    claims: HashMap<String, (String, String, String)>,
70    /// Review requests by `<stream>/<review>`: the requester and the subject, and whether
71    /// it was the reader's.
72    reviews: HashMap<String, (String, String, bool)>,
73    /// Handovers by content digest: the branch.
74    handovers: HashMap<String, String>,
75    /// Review requests the reader was asked for or answered, by `<stream>/<review>`: a
76    /// withdrawal of one of them is the reader's news.
77    involved: HashSet<String>,
78}

80/// Fold `events` (in causal order) after `after` into signals.
81pub(crate) fn fold(
82    events: &[MeshEvent],
83    after: Option<&Position>,
84    ctx: &Context<'_>,
85) -> SourceRead<Position> {
86    let mut memory = Memory::default();
87    let mut read = SourceRead::default();
88    for event in events {
89        let position = Position::of(event);
90        let news = match after {
91            Some(after) => position > *after,
92            None => event.at >= ctx.since,
93        };
94        let signal = memory.absorb(event, ctx);
95        if news {
96            match signal {
97                Some(signal) => read.signals.push((position.clone(), signal)),
98                None => read.suppressed += 1,
99            }
100            read.high = Some(position);
101        } else if after.is_none() {
102            // a fresh page stands after everything, even what it did not show
103            read.high = Some(position);
104        }
105    }
106    read
107}

109impl Memory {
110    /// Remember what `event` says and return the signal it is, if it is news.
111    fn absorb(&mut self, event: &MeshEvent, ctx: &Context<'_>) -> Option<MonitorSignal> {
112        let stream = event.stream.as_str().to_string();
113        let node = stream.split('-').next().unwrap_or_default().to_string();
114        let machine = ctx
115            .machines
116            .get(&node)
117            .cloned()
118            .unwrap_or_else(|| node.chars().take(8).collect());
119        // a kind newer than this executable's EventBody (ADR 0133 meets I2415): read raw
120        if event.kind() == "review_withdrawn" {
121            let mut s = self.withdrawn(event, &stream)?;
122            s.machine = Some(machine);
123            return Some(s);
124        }
125        let body = event.body()?;
126        let mut signal = match body {
127            EventBody::SessionOpened { info } => {
128                let key = (stream.clone(), info.session.clone());
129                let print = fingerprint(&info);
130                let name = who(&info);
131                let previous = self
132                    .sessions
133                    .insert(key.clone(), (print.clone(), name.clone()));
134                if previous.is_some_and(|(old, _)| old == print) {
135                    return None;
136                }
137                // a session that has said nothing about itself is a connection, not news:
138                // its arrival is shown when it first says what it does, and its departure
139                // only if its arrival was
140                let said = info.intent.as_deref().is_some_and(|i| !i.trim().is_empty());
141                if !said {
142                    return None;
143                }
144                let (kind, verb) = if self.shown.insert(key) {
145                    ("session_joined", "joined")
146                } else {
147                    ("session_changed", "now")
148                };
149                let intent = info.intent.clone().unwrap_or_default();
150                let mut s = MonitorSignal::new(
151                    event.at,
152                    MonitorSource::Mesh,
153                    kind,
154                    MonitorSeverity::Info,
155                    &format!("{name} {verb}: {intent}"),
156                );
157                s.session = Some(info.session.clone());
158                s.actor = Some(name);
159                s.branch = info.branch.clone();
160                s.issue = info.issue.clone();
161                s.for_you = ctx.reader.is(&info.session);
162                s
163            }
164            EventBody::SessionClosed { session } => {
165                let key = (stream.clone(), session.clone());
166                let name = self
167                    .sessions
168                    .remove(&key)
169                    .map(|(_, name)| name)
170                    .unwrap_or_else(|| session.clone());
171                if !self.shown.remove(&key) {
172                    return None;
173                }
174                let mut s = MonitorSignal::new(
175                    event.at,
176                    MonitorSource::Mesh,
177                    "session_left",
178                    MonitorSeverity::Info,
179                    &format!("{name} left"),
180                );
181                s.session = Some(session);
182                s.actor = Some(name);
183                s
184            }
185            EventBody::ClaimAcquired {
186                claim,
187                session,
188                scope,
189                intent,
190                mode,
191                issue,
192            } => {
193                let same = format!(
194                    "{mode:?}|{}|{}",
195                    scope.join(","),
196                    intent.clone().unwrap_or_default()
197                );
198                let held = self
199                    .claims
200                    .values()
201                    .any(|(st, se, sa)| *st == stream && *se == session && *sa == same);
202                self.claims
203                    .insert(claim, (stream.clone(), session.clone(), same));
204                if held {
205                    return None;
206                }
207                let name = self.name_of(&stream, &session);
208                let mode = match mode {
209                    ClaimMode::Exclusive => "exclusively",
210                    ClaimMode::Advisory => "advisory",
211                };
212                let what = intent.clone().unwrap_or_default();
213                let mut s = MonitorSignal::new(
214                    event.at,
215                    MonitorSource::Mesh,
216                    "claimed",
217                    MonitorSeverity::Info,
218                    &format!("{name} claims {mode} {}: {what}", short_scope(&scope)),
219                );
220                s.for_you = ctx.reader.named_in(&what);
221                s.session = Some(session);
222                s.actor = Some(name);
223                s.issue = issue;
224                s.paths = scope;
225                s
226            }
227            EventBody::ClaimReleased { claim } => {
228                let (st, session, _) = self.claims.remove(&claim)?;
229                let name = self.name_of(&st, &session);
230                let mut s = MonitorSignal::new(
231                    event.at,
232                    MonitorSource::Mesh,
233                    "released",
234                    MonitorSeverity::Info,
235                    &format!("{name} released claim {claim}"),
236                );
237                s.session = Some(session);
238                s.actor = Some(name);
239                s
240            }
241            EventBody::HandoverPublished { handover } => {
242                let HandoverBody {
243                    branch,
244                    head,
245                    body,
246                    issue,
247                    ..
248                } = handover;
249                let branch_name = branch.clone().unwrap_or_else(|| "(no branch)".into());
250                self.handovers
251                    .insert(HandoverBody::digest_of(&body), branch_name.clone());
252                let next = next_action(&body);
253                let head = head
254                    .map(|h| h.chars().take(10).collect::<String>())
255                    .unwrap_or_default();
256                let mut s = MonitorSignal::new(
257                    event.at,
258                    MonitorSource::Mesh,
259                    "handover",
260                    MonitorSeverity::Attention,
261                    &format!("{machine} handed over {branch_name}@{head}: {next}"),
262                );
263                s.for_you = ctx.reader.named_in(&body) || ctx.reader.named_in(&branch_name);
264                s.branch = branch;
265                s.issue = issue;
266                s
267            }
268            EventBody::HandoverConsumed { handover, session } => {
269                let branch = self.handovers.get(&handover).cloned();
270                let name = self.name_of(&stream, &session);
271                let mut s = MonitorSignal::new(
272                    event.at,
273                    MonitorSource::Mesh,
274                    "handover_taken",
275                    MonitorSeverity::Info,
276                    &format!(
277                        "{name} took the handover of {}",
278                        branch
279                            .clone()
280                            .unwrap_or_else(|| handover.chars().take(12).collect())
281                    ),
282                );
283                s.session = Some(session);
284                s.actor = Some(name);
285                s.branch = branch;
286                s
287            }
288            EventBody::ReviewRequested {
289                review,
290                session,
291                subject,
292                scope,
293                issue,
294                reviewer,
295                body,
296            } => {
297                let name = self.name_of(&stream, &session);
298                let mine = ctx.reader.is(&session) || ctx.reader.is(&name);
299                // the body (I2415, up to 8 KiB) can name the reader where the subject does not
300                let text = body.as_deref().unwrap_or_default();
301                let asked = reviewer.as_deref().is_some_and(|r| ctx.reader.is(r))
302                    || ctx.reader.named_in(&subject)
303                    || ctx.reader.named_in(text);
304                if asked {
305                    self.involved.insert(format!("{stream}/{review}"));
306                }
307                self.reviews.insert(
308                    format!("{stream}/{review}"),
309                    (name.clone(), subject.clone(), mine),
310                );
311                let mut s = MonitorSignal::new(
312                    event.at,
313                    MonitorSource::Mesh,
314                    "review_requested",
315                    if asked {
316                        MonitorSeverity::Action
317                    } else {
318                        MonitorSeverity::Attention
319                    },
320                    &format!(
321                        "{name} asks {} to review ({review}): {subject}",
322                        reviewer.as_deref().unwrap_or("anyone")
323                    ),
324                );
325                s.for_you = asked;
326                s.session = Some(session);
327                s.actor = Some(name);
328                s.issue = issue;
329                s.paths = scope;
330                s
331            }
332            EventBody::ReviewAnswered {
333                request,
334                session,
335                verdict,
336                note,
337            } => {
338                let name = self.name_of(&stream, &session);
339                if ctx.reader.is(&session) || ctx.reader.is(&name) {
340                    self.involved.insert(request.clone());
341                }
342                let (requester, subject, mine) = self
343                    .reviews
344                    .get(&request)
345                    .cloned()
346                    .unwrap_or_else(|| (String::new(), String::new(), false));
347                let review = request.rsplit('/').next().unwrap_or(&request).to_string();
348                let note = note.unwrap_or_default();
349                let severity = if mine {
350                    MonitorSeverity::Action
351                } else if verdict == "approved" || verdict == "approve" {
352                    MonitorSeverity::Info
353                } else {
354                    MonitorSeverity::Attention
355                };
356                let to = if requester.is_empty() {
357                    String::new()
358                } else {
359                    format!(" to {requester}")
360                };
361                let mut s = MonitorSignal::new(
362                    event.at,
363                    MonitorSource::Mesh,
364                    "review_answered",
365                    severity,
366                    &format!(
367                        "{name} answered {review}{to}: {verdict} — {}",
368                        if note.is_empty() { subject } else { note }
369                    ),
370                );
371                s.for_you = mine;
372                s.session = Some(session);
373                s.actor = Some(name);
374                s
375            }
376            // read raw above, before the body is typed: a withdrawal never reaches this match
377            EventBody::ReviewWithdrawn { .. } => return None,
378        };
379        signal.machine = Some(machine);
380        Some(signal)
381    }

383    /// A review withdrawn by its asker (I2415's `review_withdrawn` {request, session, note?}):
384    /// news to whoever was asked for it or answered it, a note to everyone else.
385    fn withdrawn(&mut self, event: &MeshEvent, stream: &str) -> Option<MonitorSignal> {
386        let body = &event.body;
387        let request = body.get("request")?.as_str()?.to_string();
388        let session = body
389            .get("session")
390            .and_then(|v| v.as_str())
391            .unwrap_or_default()
392            .to_string();
393        let note = body
394            .get("note")
395            .and_then(|v| v.as_str())
396            .unwrap_or_default();
397        let name = self.name_of(stream, &session);
398        let subject = self
399            .reviews
400            .remove(&request)
401            .map(|(_, subject, _)| subject)
402            .unwrap_or_default();
403        let concerns_reader = self.involved.remove(&request);
404        let review = request.rsplit('/').next().unwrap_or(&request).to_string();
405        let mut s = MonitorSignal::new(
406            event.at,
407            MonitorSource::Mesh,
408            "review_withdrawn",
409            if concerns_reader {
410                MonitorSeverity::Attention
411            } else {
412                MonitorSeverity::Info
413            },
414            &format!(
415                "{name} withdrew {review}: {}",
416                if note.is_empty() {
417                    subject.as_str()
418                } else {
419                    note
420                }
421            ),
422        );
423        s.for_you = concerns_reader;
424        s.session = Some(session);
425        s.actor = Some(name);
426        Some(s)
427    }

429    fn name_of(&self, stream: &str, session: &str) -> String {
430        self.sessions
431            .get(&(stream.to_string(), session.to_string()))
432            .map(|(_, name)| name.clone())
433            .unwrap_or_else(|| session.to_string())
434    }

437/// How to call a session: its worker when it named one, else its session and client.
438fn who(info: &SessionInfo) -> String {
439    match &info.worker {
440        Some(worker) if !worker.trim().is_empty() && worker != "Claude Code" => {
441            format!("{worker} ({})", info.session)
442        }
443        _ => format!("{} ({})", info.session, info.client),
444    }
445}

447/// The first line of a handover's `# Next Action`, or its first line when it has none.
448fn next_action(body: &str) -> String {
449    let after = body
450        .split("# Next Action")
451        .nth(1)
452        .unwrap_or(body)
453        .lines()
454        .map(str::trim)
455        .find(|l| !l.is_empty() && !l.starts_with('#'));
456    after.unwrap_or_default().to_string()
457}

459/// A scope as a reader can take in at a glance: the first two paths and how many more.
460fn short_scope(scope: &[String]) -> String {
461    let unique: Vec<&String> = {
462        let mut seen = HashSet::new();
463        scope.iter().filter(|p| seen.insert(*p)).collect()
464    };
465    match unique.len() {
466        0 => "nothing".into(),
467        1 | 2 => unique
468            .iter()
469            .map(|s| s.as_str())
470            .collect::<Vec<_>>()
471            .join(", "),
472        n => format!("{}, {} +{}", unique[0], unique[1], n - 2),
473    }
474}

803    /// The rarer shapes: an exclusive claim, a handover with no branch and the taking of one
804    /// this fold never saw, an approval without a note to a request it never saw (by the
805    /// reader, so they are involved), and a withdrawal that names no request.
806    #[test]
807    fn the_rare_shapes_still_read_as_sentences() {
808        let events = vec![
809            event(
810                1,
811                1,
812                json!({"kind": "claim_acquired", "claim": "c-9", "session": "board-p1",
813                "scope": ["apps/x"], "intent": "the lane", "mode": "exclusive"}),
814            ),
815            event(
816                2,
817                2,
818                json!({"kind": "handover_published", "handover": {
819                "id": "h", "body": "# Objective\nx\n# Next Action\nship it\n" }}),
820            ),
821            event(
822                3,
823                3,
824                json!({"kind": "handover_consumed", "handover": "0123456789abcdef0123",
825                "session": "board-p4"}),
826            ),
827            event(
828                4,
829                4,
830                json!({"kind": "review_answered", "request": format!("{STREAM}/r-7"),
831                "session": "1b", "verdict": "approved"}),
832            ),
833            event(
834                5,
835                5,
836                json!({"kind": "review_withdrawn", "session": "board-p3"}),
837            ),
838        ];
839        let reader = Reader::named(Some("1b"));
840        let machines = HashMap::new();
841        let read = fold(&events, None, &ctx(&reader, &machines));
842        let seen: Vec<(&str, &str)> = read
843            .signals
844            .iter()
845            .map(|(_, s)| (s.kind.as_str(), s.summary.as_str()))
846            .collect();
847        assert_eq!(
848            seen.len(),
849            4,
850            "the withdrawal without a request is no signal: {seen:?}"
851        );
852        assert!(seen[0].1.contains("exclusively"), "{seen:?}");
853        assert!(seen[1].1.contains("handed over (no branch)@"), "{seen:?}");
854        assert!(!read.signals[1].1.for_you, "nothing in it names 1b");
855        assert!(
856            seen[2].1.ends_with("took the handover of 0123456789ab"),
857            "{seen:?}"
858        );
859        assert_eq!(seen[3].0, "review_answered");
860        assert!(seen[3].1.contains("answered r-7: approved"), "{seen:?}");
861        assert!(
862            !seen[3].1.contains(" to "),
863            "no requester is known: {seen:?}"
864        );
865        assert_eq!(read.signals[3].1.severity, MonitorSeverity::Info);
866    }
