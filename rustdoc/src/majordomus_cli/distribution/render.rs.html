11/// What every projection of the distribution model says it was derived from. One string,
12/// so a reader of any of them is pointed at the same two files.
13pub const SOURCE: &str = "share/distribution.yaml and the release records";

15/// The placeholder the naming function leaves for the tag a release will have. The build
16/// matrix carries names with this in them, so that the workflow substitutes a tag rather
17/// than composing a name of its own.
18pub const TAG_PLACEHOLDER: &str = "{tag}";
19
20/// The line that opens the installer's generated region.
21pub const REGION_BEGIN: &str =
22    "# >>> generated from share/distribution.yaml — do not edit here >>>";
23
24/// The line that closes it.
25pub const REGION_END: &str = "# <<< generated <<<";

27/// The release build matrix: one entry per published target, with the runner it is built
28/// on and the artifact name the naming function derives for a tag yet to be chosen.
29pub fn matrix(model: &Model) -> Value {
30    let include: Vec<Value> = model
31        .published()
32        .map(|t| {
33            let build = t
34                .build
35                .as_ref()
36                .expect("a published target declares a build");
37            json!({
38                "id": t.id,
39                "target": t.rust_target,
40                "os": t.os,
41                "arch": t.arch,
42                "libc": t.libc,
43                "status": t.status,
44                "runner": build.runner,
45                "packages": build.packages.join(" "),
46                "native": build.native,
47                "artifact": t.artifact_name(&model.project, &model.archive, TAG_PLACEHOLDER),
48                "root": t.archive_root(&model.project, TAG_PLACEHOLDER),
49            })
50        })
51        .collect();
52    json!({
53        "schema": "majordomus/distribution-matrix/v1",
54        "generated": crate::generate::json_banner(SOURCE),
55        "binary": model.project.binary,
56        "repository": model.project.repository,
57        "checksums_file": model.archive.checksums_file,
58        "include": include,
59    })
60}

62/// The matrix as the workflow reads it.
63pub fn matrix_json(model: &Model) -> String {
64    let mut s = serde_json::to_string_pretty(&matrix(model)).unwrap_or_default();
65    s.push('\n');
66    s
67}

69/// The dataset the website renders: the commands to show, the platforms to list, and the
70/// release the pages name. The site states none of these itself.
71pub fn site_dataset(model: &Model, releases: &Releases) -> String {
72    let targets: Vec<Value> = model
73        .targets
74        .iter()
75        .map(|t| {
76            json!({
77                "id": t.id,
78                "title": t.title(&model.titles),
79                "os": t.os,
80                "arch": t.arch,
81                "libc": t.libc,
82                "rust_target": t.rust_target,
83                "status": t.status,
84                "reason": t.reason,
85            })
86        })
87        .collect();
88    let latest = releases.latest_stable().map(|r| {
89        json!({
90            "tag": r.tag,
91            "version": r.version,
92            "published_at": r.published_at,
93            "notes_url": r.notes_url,
94            "url": model.release_url(&r.tag),
95            "artifacts": r.artifacts.len(),
96        })
97    });
98    let value = json!({
99        "schema": "majordomus/distribution/v1",
100        "generated": crate::generate::json_banner(SOURCE),
101        "binary": model.project.binary,
102        "repository": model.project.repository,
103        "installer_url": model.installer_url(),
104        "install_command": model.install_command(),
105        "install_and_init_command": model.install_and_init_command(),
106        "next_command": format!("{} init", model.project.binary),
107        "verify_command": format!("{} --version", model.project.binary),
108        "install_dir": model.installer.install_dir,
109        "prefix": model.installer.prefix,
110        "latest_url": model.release_url(super::release::LATEST),
111        "checksum": model.archive.checksum,
112        "supported": model.published().count(),
113        "targets": targets,
114        "latest": latest,
115    });
116    let mut s = serde_json::to_string_pretty(&value).unwrap_or_default();
117    s.push('\n');
118    s
119}

121/// The installer's generated region: the facts the script must not restate. Shell, in the
122/// subset the installer itself is written in, single-quoted so that nothing in the model
123/// can be expanded by the shell that reads it.
124pub fn installer_region(model: &Model) -> String {
125    let mut s = String::new();
126    s.push_str(&format!("{REGION_BEGIN}\n"));
127    s.push_str(
128        "# Every value below comes from share/distribution.yaml. Change the model and run\n",
129    );
130    s.push_str("# `just derive`; editing this region is undone by the next generation and refused by CI.\n");
131    s.push_str(&format!("MJ_BINARY={}\n", sq(&model.project.binary)));
132    s.push_str(&format!(
133        "MJ_REPOSITORY={}\n",
134        sq(&model.project.repository)
135    ));
136    s.push_str(&format!("MJ_BASE_URL={}\n", sq(&model.installer.base_url)));
137    s.push_str(&format!(
138        "MJ_DOWNLOAD_PREFIX={}\n",
139        sq(&model.project.download_prefix())
140    ));
141    s.push_str(&format!(
142        "MJ_ARCHIVE_EXTENSION={}\n",
143        sq(&model.archive.extension)
144    ));
145    s.push_str(&format!(
146        "MJ_DEFAULT_INSTALL_DIR=\"{}\"\n",
147        model.installer.install_dir
148    ));
149    s.push_str(&format!(
150        "MJ_DEFAULT_PREFIX=\"{}\"\n",
151        model.installer.prefix
152    ));
153    s.push_str("\n# One line per target: <os> <arch> <libc or -> <rust target> <status>\n");
154    s.push_str("MJ_TARGETS='");
155    let rows: Vec<String> = model
156        .targets
157        .iter()
158        .map(|t| {
159            format!(
160                "{} {} {} {} {}",
161                json_str(&t.os),
162                json_str(&t.arch),
163                t.libc.map(|l| json_str(&l)).unwrap_or_else(|| "-".into()),
164                t.rust_target,
165                json_str(&t.status)
166            )
167        })
168        .collect();
169    s.push_str(&rows.join("\n"));
170    s.push_str("'\n");
171    s.push_str("\n# The supported platforms, as the unsupported-platform message lists them\n");
172    s.push_str("MJ_SUPPORTED_TITLES='");
173    let titles: Vec<String> = model.published().map(|t| t.title(&model.titles)).collect();
174    s.push_str(&titles.join("\n"));
175    s.push_str("'\n");
176    s.push_str("\n# Why a declared target is not built: <rust target> <reason>\n");
177    s.push_str("MJ_UNAVAILABLE='");
178    let unavailable: Vec<String> = model
179        .targets
180        .iter()
181        .filter(|t| t.status == Status::Unavailable)
182        .map(|t| {
183            format!(
184                "{} {}",
185                t.rust_target,
186                t.reason
187                    .as_deref()
188                    .unwrap_or("no reason recorded")
189                    .replace('\n', " ")
190            )
191        })
192        .collect();
193    s.push_str(&unavailable.join("\n"));
194    s.push_str("'\n");
195    s.push_str(&format!("{REGION_END}\n"));
196    s
197}

199/// Splice the generated region into the installer template. The template owns the
200/// behaviour; this owns the facts.
201pub fn installer(model: &Model, template: &str) -> std::result::Result<String, String> {
202    let begin = template
203        .find(REGION_BEGIN)
204        .ok_or_else(|| format!("the template carries no `{REGION_BEGIN}` line"))?;
205    let end_at = template
206        .find(REGION_END)
207        .ok_or_else(|| format!("the template carries no `{REGION_END}` line"))?;
208    if end_at < begin {
209        return Err("the installer template's generated region closes before it opens".into());
210    }
211    let end = end_at + REGION_END.len() + 1;
212    let mut out = String::new();
213    out.push_str(&template[..begin]);
214    out.push_str(&installer_region(model));
215    out.push_str(&template[end.min(template.len())..]);
216    Ok(out)
217}

219/// The installation guide: the template's prose with every derived value substituted.
220pub fn install_doc(
221    model: &Model,
222    releases: &Releases,
223    crate_version: &str,
224    template: &str,
225) -> std::result::Result<String, String> {
226    let example_tag = releases
227        .latest_stable()
228        .map(|r| r.tag.clone())
229        .unwrap_or_else(|| format!("v{crate_version}"));
230    let example = model
231        .published()
232        .next()
233        .map(|t| t.artifact_name(&model.project, &model.archive, &example_tag))
234        .unwrap_or_default();
235    let values: Vec<(&str, String)> = vec![
236        ("binary", model.project.binary.clone()),
237        ("repository", model.project.repository.clone()),
238        ("license", model.project.license.clone()),
239        ("base_url", model.installer.base_url.clone()),
240        ("installer_url", model.installer_url()),
241        ("install_command", model.install_command()),
242        ("install_and_init_command", model.install_and_init_command()),
243        ("install_dir", model.installer.install_dir.clone()),
244        ("prefix", model.installer.prefix.clone()),
245        ("checksum", model.archive.checksum.clone()),
246        ("checksums_file", model.archive.checksums_file.clone()),
247        ("latest_url", model.release_url(super::release::LATEST)),
248        ("example_tag", example_tag.clone()),
249        ("example_url", model.release_url(&example_tag)),
250        ("example_artifact", example),
251        ("supported_count", model.published().count().to_string()),
252        ("platforms", platform_table(model)),
253        ("unavailable", unavailable_section(model)),
254        ("releases", release_table(model, releases)),
255    ];
256    let mut out = template.to_string();
257    for (key, value) in &values {
258        out = out.replace(&format!("{{{{{key}}}}}"), value);
259    }
260    if let Some(at) = out.find("{{") {
261        let rest = &out[at..];
262        let name: String = rest.chars().take(40).collect();
263        return Err(format!(
264            "the guide's template has an unsubstituted placeholder: {name}"
265        ));
266    }
267    Ok(out)
268}

270/// The supported-platform table, derived from the model and written nowhere by hand.
271pub fn platform_table(model: &Model) -> String {
272    let mut s = String::from(
273        "| Platform | Architecture | libc | Rust target | Status |\n|---|---|---|---|---|\n",
274    );
275    for t in &model.targets {
276        let os = match t.os {
277            super::Os::Macos => &model.titles.os.macos,
278            super::Os::Linux => &model.titles.os.linux,
279            super::Os::Windows => &model.titles.os.windows,
280        };
281        let arch = match t.arch {
282            super::Arch::X86_64 => &model.titles.arch.x86_64,
283            super::Arch::Aarch64 => &model.titles.arch.aarch64,
284        };
285        let libc = match t.libc {
286            Some(super::Libc::Gnu) => model.titles.libc.gnu.as_str(),
287            Some(super::Libc::Musl) => model.titles.libc.musl.as_str(),
288            None => "—",
289        };
290        s.push_str(&format!(
291            "| {os} | {arch} | {libc} | `{}` | {} |\n",
292            t.rust_target,
293            json_str(&t.status)
294        ));
295    }
296    s
297}

299/// The reasons the unbuilt targets are unbuilt, or a sentence saying there are none.
300pub fn unavailable_section(model: &Model) -> String {
301    let rows: Vec<&super::Target> = model
302        .targets
303        .iter()
304        .filter(|t| t.status == Status::Unavailable)
305        .collect();
306    if rows.is_empty() {
307        return "Every target the model declares is built.\n".to_string();
308    }
309    let mut s = String::new();
310    for t in rows {
311        s.push_str(&format!(
312            "**{}** (`{}`) — {}\n\n",
313            t.title(&model.titles),
314            t.rust_target,
315            t.reason.as_deref().unwrap_or("no reason recorded")
316        ));
317    }
318    s
319}

321/// The releases this repository has published, or a sentence saying there are none yet.
322pub fn release_table(model: &Model, releases: &Releases) -> String {
323    if releases.is_empty() {
324        return format!(
325            "No release is published yet. When one is, its metadata appears at `{}` and this table lists it.\n",
326            model.release_url(super::release::LATEST)
327        );
328    }
329    let latest = releases.latest_stable().map(|r| r.tag.clone());
330    let mut s = String::from(
331        "| Release | Published | Channel | Artifacts | Metadata |\n|---|---|---|---|---|\n",
332    );
333    for r in &releases.releases {
334        let mark = if Some(&r.tag) == latest.as_ref() {
335            " (latest)"
336        } else if r.yanked {
337            " (withdrawn)"
338        } else {
339            ""
340        };
341        s.push_str(&format!(
342            "| `{}`{mark} | {} | {} | {} | [`{}.json`]({}) |\n",
343            r.tag,
344            &r.published_at[..10.min(r.published_at.len())],
345            match r.channel {
346                Channel::Stable => "stable",
347                Channel::Prerelease => "prerelease",
348            },
349            r.artifacts.len(),
350            r.tag,
351            model.release_url(&r.tag)
352        ));
353    }
354    s
355}

357/// A serde enum's wire name: the vocabulary is declared once, in the type.
358fn json_str<T: serde::Serialize>(value: &T) -> String {
359    serde_json::to_value(value)
360        .ok()
361        .and_then(|v| v.as_str().map(str::to_string))
362        .unwrap_or_default()
363}

365/// A single-quoted shell word. The model's values are constrained by its schema; a quote
366/// inside one would still not escape, and this makes that true rather than assumed.
367fn sq(value: &str) -> String {
368    format!("'{}'", value.replace('\'', "'\\''"))
