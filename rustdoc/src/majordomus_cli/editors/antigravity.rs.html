1//! Google Antigravity: what it reads from a project, and the projection of the environment
2//! onto it.
3//!
4//! Verified against Antigravity 2.22.0 (`com.google.antigravity`, the 2.0 desktop app) —
5//! the guide text shipped inside the application's language server, the changelog at
6//! <https://antigravity.google/docs/changelog/>, and the pages <https://antigravity.google/docs/rules>,
7//! `/docs/skills`, `/docs/mcp`, `/docs/ide/workflows` and `/docs/migration/workflows-to-skills`.
8//! Where the shipped guide and the web pages disagree the projection writes nothing and the
9//! capability says [`EditorSupport::Unknown`].
10

18/// The adapter's id.
19pub const ID: &str = "antigravity";
20
21/// The release the claims below were verified against.
22pub const VERIFIED_AGAINST: &str = "Antigravity 2.22.0";
23
24/// The values a modular rule's `trigger` may take; anything else, or none, and Antigravity
25/// discards the rule without a word.
26pub const TRIGGERS: &[&str] = &["always_on", "model_decision", "glob", "manual"];
27
28/// The most a rule file may hold, after its includes are expanded.
29pub const RULE_BYTES: usize = 24_000;
30
31/// What Antigravity does with each capability here.
32pub fn adapter(env: &EditorEnvironment) -> EditorAdapter {
33    let skills = env
34        .skills
35        .iter()
36        .filter(|s| super::is_skill_name(&s.id))
37        .map(|s| format!("{SKILLS_DIR}/{}/SKILL.md", s.id))
38        .collect();
39    EditorAdapter {
40        id: ID.into(),
41        title: "Google Antigravity".into(),
42        verified_against: VERIFIED_AGAINST.into(),
43        capabilities: vec![
44            EditorCapability {
45                id: "mcp".into(),
46                support: EditorSupport::Unknown,
47                artifacts: Vec::new(),
48                evidence: "antigravity.google/docs/mcp documents .agents/mcp_config.json; the MCP guide shipped in 2.22.0 names only the global and plugin locations".into(),
49                note: "No workspace MCP file is written while the installed product does not confirm it reads one. Add the server to ~/.gemini/config/mcp_config.json yourself: {\"mcpServers\": {\"majordomus\": {\"command\": \"<absolute path to this checkout>/bin/majordomus-mcp\", \"args\": []}}}.".into(),
50            },
51            EditorCapability {
52                id: "permissions".into(),
53                support: EditorSupport::RequiresUserAction,
54                artifacts: Vec::new(),
55                evidence: "antigravity.google/docs/agent-settings".into(),
56                note: "Approval and terminal-execution modes are personal and are never projected; a repository does not loosen them.".into(),
57            },
58            EditorCapability {
59                id: "rules".into(),
60                support: EditorSupport::Supported,
61                artifacts: vec!["AGENTS.md".into()],
62                evidence: "antigravity.google/docs/rules: AGENTS.md is always on for its directory".into(),
63                note: "The provider bootstrap AGENTS.md routes into .ai/; no rule corpus is copied into .agents/rules/.".into(),
64            },
65            EditorCapability {
66                id: "skills".into(),
67                support: EditorSupport::Supported,
68                artifacts: skills,
69                evidence: "antigravity.google/docs/skills; the built-in guide's skill section (name and description required)".into(),
70                note: "Read from .agents/skills/<name>/SKILL.md; each file points at the canonical definition.".into(),
71            },
72            EditorCapability {
73                id: "workflows".into(),
74                support: EditorSupport::Deprecated,
75                artifacts: Vec::new(),
76                evidence: "antigravity.google/docs/migration/workflows-to-skills: workflows stop working on 2026-10-19".into(),
77                note: "Never written; the skills carry what a workflow would.".into(),
78            },
79        ],
80    }
81}

83/// What this checkout holds that Antigravity would read differently from what the
84/// projection intends. Read only; nothing here is ever rewritten.
85pub fn legacy_findings(root: &Path) -> Vec<EditorDiagnostic> {
86    let mut out = Vec::new();
87    let finding =
88        |severity, code: &str, path: &str, message: String, remedy: &str| EditorDiagnostic {
89            severity,
90            code: code.into(),
91            adapter: Some(ID.into()),
92            path: Some(path.into()),
93            message,
94            remedy: remedy.into(),
95        };
96    if root.join(".agents/workflows").is_dir() {
97        out.push(finding(
98            Severity::Warning,
99            "antigravity-legacy-workflows",
100            ".agents/workflows",
101            "Antigravity stops running .agents/workflows on 2026-10-19".into(),
102            "turn each workflow into a skill under .ai/repo/skills/, which is projected to .agents/skills/",
103        ));
104    }
105    if root.join(".agents/skills.json").is_file() {
106        out.push(finding(
107            Severity::Info,
108            "antigravity-skills-manifest",
109            ".agents/skills.json",
110            "a skills manifest makes Antigravity load the directories it names as well as .agents/skills/; a manifest naming .ai/repo/skills loads canonical files that carry no `name`, which the shipped guide requires".into(),
111            "remove .agents/skills.json; the projected .agents/skills/ already carries every active skill",
112        ));
113    }
114    let Ok(entries) = std::fs::read_dir(root.join(".agents/rules")) else {
115        return out;
116    };
117    let mut rules: Vec<String> = entries
118        .filter_map(|e| e.ok())
119        .map(|e| e.path())
120        .filter(|p| p.extension().is_some_and(|x| x == "md") && p.is_file())
121        .map(|p| p.to_string_lossy().into_owned())
122        .collect();
123    crate::order::canonical_strings(&mut rules);
124    for path in rules.iter().map(std::path::Path::new) {
125        let rel = path
126            .strip_prefix(root)
127            .map(|p| p.to_string_lossy().replace('\\', "/"))
128            .unwrap_or_default();
129        let Ok(text) = std::fs::read_to_string(path) else {
130            continue;
131        };
132        if text.len() > RULE_BYTES {
133            out.push(finding(
134                Severity::Warning,
135                "antigravity-rule-too-large",
136                &rel,
137                format!(
138                    "{} bytes; Antigravity reads at most {RULE_BYTES}",
139                    text.len()
140                ),
141                "split the rule, or move its substance under .ai/ and link to it",
142            ));
143        }
144        match trigger_of(&text) {
145            Some(t) if TRIGGERS.contains(&t.as_str()) => {}
146            found => out.push(finding(
147                Severity::Warning,
148                "antigravity-rule-discarded",
149                &rel,
150                match found {
151                    Some(t) => format!("trigger {t:?} is not one of {}; Antigravity discards the rule", TRIGGERS.join(", ")),
152                    None => "no `trigger` in front matter; Antigravity discards the rule".into(),
153                },
154                "open the file with front matter `trigger: always_on` (or model_decision with a description, glob with globs, manual)",
155            )),
156        }
157    }
158    out
159}

161/// The `trigger` of a rule's front matter, when it has one.
162fn trigger_of(text: &str) -> Option<String> {
163    let front = text.strip_prefix("---\n")?.split_once("\n---")?.0;
164    front.lines().find_map(|l| {
165        l.strip_prefix("trigger:")
166            .map(|v| v.trim().trim_matches(['"', '\'']).to_string())
167    })
168}

170#[cfg(test)]
171mod tests {
172    use super::*;
173
174    #[test]
175    fn a_rule_without_a_valid_trigger_is_reported_and_one_with_it_is_not() {
176        let dir = tempfile::tempdir().unwrap();
177        let rules = dir.path().join(".agents/rules");
178        std::fs::create_dir_all(&rules).unwrap();
179        std::fs::write(rules.join("plain.md"), "# no front matter\n").unwrap();
180        std::fs::write(rules.join("camel.md"), "---\ntrigger: alwaysOn\n---\nx\n").unwrap();
181        std::fs::write(rules.join("good.md"), "---\ntrigger: always_on\n---\nx\n").unwrap();
182        let found = legacy_findings(dir.path());
183        let paths: Vec<&str> = found.iter().filter_map(|d| d.path.as_deref()).collect();
184        assert_eq!(paths, [".agents/rules/camel.md", ".agents/rules/plain.md"]);
185        assert!(found.iter().all(|d| d.code == "antigravity-rule-discarded"));
186    }
187
188    #[test]
189    fn legacy_workflows_and_a_skills_manifest_are_reported() {
190        let dir = tempfile::tempdir().unwrap();
191        std::fs::create_dir_all(dir.path().join(".agents/workflows")).unwrap();
192        std::fs::write(dir.path().join(".agents/skills.json"), "{}").unwrap();
193        let codes: Vec<String> = legacy_findings(dir.path())
194            .into_iter()
195            .map(|d| d.code)
196            .collect();
197        assert_eq!(
198            codes,
199            [
200                "antigravity-legacy-workflows",
201                "antigravity-skills-manifest"
202            ]
203        );
204    }
