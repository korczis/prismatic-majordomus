1//! The Swagger UI shell: one HTML page that loads the pinned Swagger UI distribution and
2//! points it at `/openapi.json`. It embeds no specification of its own; what it shows is
3//! whatever the server generated from the registry at the moment of the request. The
4//! page is rendered once per process and served from memory.
5//!
6//! The UI's own assets are fetched by the browser from the unpkg CDN. That is the one
7//! part of the HTTP projection that is not available offline; the OpenAPI document is.
8//!
9//! The frame around it is this repository's, and so is what the frame owes a reader: a
10//! `<main>` landmark and a level-one heading, which the page had neither of until the UI
11//! audit was first allowed to look at this surface. The widget itself carries
12//! `data-mj-foreign`, the declaration a page makes when a subtree is a third party's to
13//! answer for: the accessibility engine skips it and the audit's report says it was
14//! skipped and why, rather than reporting a clean page or an unfixable one.
15//!
16//! The frame around it is this repository's. The type stack and the accent come from
17//! `share/design/tokens.yaml` like every other surface, compiled in through `tokens.css`,
18//! so the API viewer reads as part of the same tool rather than as a stock installation of
19//! somebody else's. What is inside the widget — the operation blocks, the schema tables,
20//! the try-it form — is Swagger UI's own stylesheet and is left alone: restyling a third
21//! party's component tree against a pinned version is a maintenance bill this page does not
22//! need to take on.
23//!
24//! The page pins itself light. Swagger UI 5 ships no dark theme, so a dark frame around a
25//! permanently light widget is worse than a light one; `class="light"` is the escape hatch
26//! the generated token block provides for exactly this, and it also stops the browser from
27//! rendering dark form controls inside a light panel.
28
29use std::sync::LazyLock;
30
31/// The Swagger UI distribution version the page pins.
32pub const SWAGGER_UI_VERSION: &str = "5.17.14";
33
34/// The path the page loads the specification from.
35pub const SPEC_PATH: &str = "/openapi.json";
36
37/// The path the page is served at.
38///
39/// It is `/swagger` and not `/docs`: `/docs` is where this repository's own documentation
40/// is served, and a viewer for the API is not the documentation. The rule that holds the
41/// two apart is `project.web-surface-declared-once`.
42pub const SWAGGER_PATH: &str = "/swagger";
43
44static PAGE: LazyLock<String> = LazyLock::new(|| {
45    format!(
46        r##"<!doctype html>
47<html lang="en" class="light">
48<head>
49<meta charset="utf-8">
50<meta name="viewport" content="width=device-width, initial-scale=1">
51<title>Majordomus API</title>
52<link rel="stylesheet" href="https://unpkg.com/swagger-ui-dist@{v}/swagger-ui.css">
53<style>
54{tokens}
55/* The frame: the page around the widget, in this repository's type and colour. */
56body {{ margin: 0; background: var(--mj-bg); color: var(--mj-fg); font-family: var(--font-sans); }}
57.swagger-ui, .swagger-ui .info .title, .swagger-ui .opblock-tag {{ font-family: var(--font-sans); }}
58.swagger-ui .microlight, .swagger-ui code, .swagger-ui pre {{ font-family: var(--font-mono); }}
59/* Swagger UI's topbar is its own branding and a form for choosing a specification. This
60   page serves one specification and the process advertises its surfaces on its home page,
61   so the bar is a logo and a field that must not be used. */
62.swagger-ui .topbar {{ display: none; }}
63.swagger-ui .info .title small.version-stamp {{ background: var(--mj-accent-fill); }}
64.swagger-ui a {{ color: var(--mj-accent); }}
65/* The frame owes the page a landmark and a heading; the widget renders its own title as an
66   h2 beneath this one, so the order holds. Without them the page has neither, which is what
67   the audit found the first time it was allowed to look at this surface. */
68.mj-api-title {{ margin: 0; padding: 1rem 1.25rem 0; color: var(--mj-fg); }}
69</style>
70</head>
71<body>
72<h1 class="mj-api-title">Majordomus API</h1>
73<main id="swagger-ui" data-mj-foreign="swagger-ui-dist@{v}"></main>
74<script src="https://unpkg.com/swagger-ui-dist@{v}/swagger-ui-bundle.js" crossorigin></script>
75<script>
76window.ui = SwaggerUIBundle({{ url: "{spec}", dom_id: "#swagger-ui", deepLinking: true }});
77</script>
78</body>
79</html>
80"##,
81        v = SWAGGER_UI_VERSION,
82        spec = SPEC_PATH,
83        tokens = crate::web::html::TOKENS
84    )
85});

87/// The page, rendered once.
88///
89/// ```
90/// use majordomus_cli::http::swagger;
91/// let page = swagger::page();
92/// assert!(page.contains(swagger::SWAGGER_UI_VERSION));
93/// assert!(page.contains("url: \"/openapi.json\""));
94/// assert!(!page.contains("\"paths\""), "the shell embeds no specification");
95/// assert!(page.contains("--font-sans"), "the shell carries the repository's design tokens");
96/// ```
97pub fn page() -> &'static str {
98    PAGE.as_str()
99}
100
101/// Whether this shell may be offered as a link a reader can follow, in an environment
102/// where the surfaces are what [`crate::web::projection_routes`] resolved.
103///
104/// The console is served by a running process and nothing publishes it, so a published
105/// page must name it rather than link it. The answer comes from the surface's declared
106/// availability and from nowhere else: not from the page's own address, not from a build
107/// flag, not from a template that happens to know which site it is rendering.
108///
109/// ```
110/// use majordomus_cli::http::swagger;
111/// assert!(!swagger::offered_by_a_publication(), "a console needs the server behind it");
112/// ```
113pub fn offered_by_a_publication() -> bool {
114    crate::web::projection_routes()
115        .iter()
116        .find(|r| r.path == SWAGGER_PATH)
117        .is_some_and(|r| r.linkable())
118}

120#[cfg(test)]
121mod tests {
122    use super::*;
123
124    #[test]
125    fn the_shell_points_at_the_generated_document_and_carries_none_of_its_own() {
126        // the whole claim this module makes: what a reader sees is whatever the server
127        // generated from the registry at the moment of the request, never a copy
128        let shell = page();
129        assert!(shell.contains(&format!("url: \"{SPEC_PATH}\"")));
130        for embedded in ["\"paths\"", "\"openapi\"", "\"components\""] {
131            assert!(
132                !shell.contains(embedded),
133                "the shell embeds {embedded}, so it can disagree with the registry"
134            );
135        }
136        // and it is rendered once: two calls hand back the same allocation, not two
137        assert!(std::ptr::eq(shell.as_ptr(), page().as_ptr()));
138    }
139
140    #[test]
141    fn the_pinned_distribution_is_the_one_both_asset_urls_name() {
142        // a half-upgraded pin loads a stylesheet from one version and a bundle from another,
143        // which fails in the browser and nowhere else.
144        //
145        // The subject is the asset URLs, not every mention of the distribution: the widget
146        // also names it in `data-mj-foreign`, which is a declaration about whose component
147        // tree this is and not a thing the browser fetches. Counting bare occurrences
148        // conflated the two and made this test refuse a correct page, so it counts the
149        // fetched URLs, and separately holds every mention to the pinned version.
150        //
151        // master fixed the same failure in 4e04aa6ff by asking for 3 instead of 2. Both
152        // pass today; this one keeps passing when a fourth mention is added, and still
153        // refuses a pin that is half-upgraded wherever the stale version appears.
154        let shell = page();
155        let fetched = shell
156            .matches(&format!(
157                "https://unpkg.com/swagger-ui-dist@{SWAGGER_UI_VERSION}/"
158            ))
159            .count();
160        assert_eq!(
161            fetched, 2,
162            "the stylesheet and the bundle both come from the pinned version"
163        );
164        let pinned = shell.matches("swagger-ui-dist@").count();
165        assert_eq!(
166            shell
167                .matches(&format!("swagger-ui-dist@{SWAGGER_UI_VERSION}"))
168                .count(),
169            pinned,
170            "every mention of the distribution names the pinned version, not only the URLs"
171        );
172    }
