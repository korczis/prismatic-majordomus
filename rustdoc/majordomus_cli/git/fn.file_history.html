Expand description

The commits that changed path, newest first and across renames, as FileCommits: at most bound of them, with whether more exist (I3314).

None when git cannot answer, as in a directory that is no repository or one whose branch has no commit yet. An empty list is a file no commit has touched. Read through read_only, so asking moves nothing.

In a shallow clone the boundary commit appears to add every file it holds, which is not a change anybody made: the list stops before it and says it was cut, so an empty list that is cut means the commit that last changed the file is not in this clone.

use majordomus_cli::git::file_history;
use std::process::Command;

let dir = tempfile::tempdir().unwrap();
let git = |args: &[&str]| {
    let ok = Command::new("git").arg("-C").arg(dir.path()).args(args).status().unwrap();
    assert!(ok.success());
};
git(&["init", "-q"]);
git(&["config", "user.email", "t@example.com"]);
git(&["config", "user.name", "t"]);
assert_eq!(file_history(dir.path(), "a.md", 5), None, "no commit, no history");
for n in 1..=3 {
    std::fs::write(dir.path().join("a.md"), n.to_string()).unwrap();
    git(&["add", "a.md"]);
    git(&["commit", "-qm", &format!("change {n}")]);
}
let (commits, cut) = file_history(dir.path(), "a.md", 2).unwrap();
assert_eq!(commits.len(), 2);
assert!(cut, "a third commit exists past the bound");
assert_eq!(commits[0].2, "change 3", "newest first");
assert!(file_history(dir.path(), "never.md", 2).unwrap().0.is_empty());
