1//! The Cockpit's MCP page: the MCP projection of the registry as `mcp.projection` describes
2//! it, and where each executable capability is reachable as `capabilities.projections`
3//! derives it. The page holds no tool, no count, no client and no status of its own: a tool
4//! added to the registry is on it at the next request, and one removed is gone.
5

20const ASKED: &str = "mcp.projection";
21const HERE: &str = "/cockpit/mcp";

23/// The word an effect is written as, on the wire and in the address of this page.
24fn effect_word(effect: Effect) -> &'static str {
25    match effect {
26        Effect::Read => "read",
27        Effect::ProcessState => "process_state",
28        Effect::RepositoryMutation => "repository_mutation",
29    }
30}
31
32/// What a reader is told an effect means, in the words `docs/MCP.md` uses.
33fn effect_label(effect: Effect) -> &'static str {
34    match effect {
35        Effect::Read => "read",
36        Effect::ProcessState => "process state",
37        Effect::RepositoryMutation => "writes the repository",
38    }
39}
40
41/// An effect as a badge: the status says how careful a caller should be, the label says
42/// why, so the cue never rests on colour alone.
43fn effect_badge(effect: Effect) -> El {
44    badge(
45        match effect {
46            Effect::Read => "ok",
47            Effect::ProcessState => "info",
48            Effect::RepositoryMutation => "warn",
49        },
50        effect_label(effect),
51    )
52}
53
54fn capability_href(id: &str) -> String {
55    format!("/cockpit/capabilities/{}", percent_encode(id))
56}
57
58fn present(value: Option<&String>) -> El {
59    match value {
60        Some(v) => cell(mono(v.clone())),
61        None => text_cell("—"),
62    }
63}

65/// `/cockpit/mcp`, with `?effect=` narrowing the tools to one effect.
66pub fn page(ctx: &Context, query: &[(String, String)]) -> Page {
67    let asked_effect = query
68        .iter()
69        .find(|(k, _)| k == "effect")
70        .map(|(_, v)| v.as_str())
71        .filter(|v| !v.is_empty());
72    let input = match asked_effect {
73        Some(word) => json!({ "effect": word }),
74        None => json!({}),
75    };
76    let asked: Result<(McpProjection, Matrix), String> = ask(ctx, ASKED, input)
77        .and_then(|p| ask(ctx, "capabilities.projections", json!({})).map(|matrix| (p, matrix)));
78    match asked {
79        Ok((p, matrix)) => rendered(&p, &matrix, asked_effect),
80        Err(e) => failed(Area::Mcp, "MCP", e),
81    }
82}
83
84/// The page for one description of the projection and one matrix. Everything on it is
85/// read from those two values, so what a registry with no writer, a distribution with no
86/// client or a capability with an unanswered exposure looks like can be asked without
87/// building such a registry.
88fn rendered(p: &McpProjection, matrix: &Matrix, asked_effect: Option<&str>) -> Page {
89    // ------------------------------------------------------------ the projection
90    let mut stats = el("div").class("mj-stats").child(asked_statistic(
91        p.tool_count.to_string(),
92        "tools",
93        ASKED,
94        "tool_count",
95    ));
96    for e in &p.effects {
97        stats = stats.child(asked_statistic(
98            e.tools.to_string(),
99            effect_label(e.effect),
100            ASKED,
101            "effects",
102        ));
103    }
104    stats = stats
105        .child(asked_statistic(
106            p.resources.builtin.to_string(),
107            "capability resources",
108            ASKED,
109            "resources.builtin",
110        ))
111        .child(asked_statistic(
112            p.resources.objects.to_string(),
113            "object resources",
114            ASKED,
115            "resources.objects",
116        ));
117
118    let marks = |words: &[String]| {
119        el("span")
120            .class("mj-marks")
121            .children(words.iter().map(|w| tag(w.clone())).collect::<Vec<_>>())
122    };
123    let transports = table(
124        &["Transport", "Reached by", "Sessions attached now"],
125        p.transports
126            .iter()
127            .map(|t| {
128                row(vec![
129                    cell(mono(t.id.clone())),
130                    text_cell(t.reached_by.clone()),
131                    text_cell(t.attached.to_string()),
132                ])
133            })
134            .collect(),
135    );
136    let not_served: Vec<String> = [
137        (!p.serving.prompts).then_some("prompts"),
138        (!p.serving.notifications).then_some("subscriptions and list-change notifications"),
139    ]
140    .into_iter()
141    .flatten()
142    .map(String::from)
143    .collect();
144    // a row per method rather than a run of tags: a method name does not break, and a
145    // table scrolls inside itself where a run of tags pushes the page wide
146    let served = table(
147        &["Of the protocol", "Served"],
148        p.serving
149            .methods
150            .iter()
151            .map(|m| row(vec![cell(mono(m.clone())), cell(badge("ok", "served"))]))
152            .chain(not_served.iter().map(|w| {
153                row(vec![
154                    text_cell(w.clone()),
155                    cell(badge("info", "not served")),
156                ])
157            }))
158            .collect(),
159    );
160    let overview = card_with(
161        "The projection",
162        link("/cockpit/api", "HTTP routes"),
163        el("div")
164            .child(stats)
165            .child(facts(vec![
166                (
167                    "Server",
168                    Node::Element(mono(format!("{} {}", p.server.name, p.server.version))),
169                ),
170                ("Protocol versions", Node::Element(marks(&p.protocol_versions))),
171                ("Object resources", Node::Element(mono(p.resources.template.clone()))),
172            ]))
173            .child(transports)
174            .child(served)
175            .child(el("p").class("mj-note").text(
176                "A tool is a capability and a resource is a capability or an object of the layer: MCP declares nothing of its own. The attached sessions are this process's peer board at the moment the page was asked for, not a claim about any other server.",
177            )),
178    );
179
180    // ------------------------------------------------------------ the writers
181    let writers = if p.writers.is_empty() {
182        card(
183            "Tools that write the repository",
184            nothing("No tool of this registry writes the repository."),
185        )
186    } else {
187        let rows: Vec<El> = p
188            .writers
189            .iter()
190            .filter_map(|name| {
191                matrix
192                    .rows
193                    .iter()
194                    .find(|r| r.mcp_tool.as_ref() == Some(name))
195            })
196            .map(|r| {
197                row(vec![
198                    cell(mono(r.mcp_tool.clone().unwrap_or_default())),
199                    id_cell(capability_href(&r.id), r.id.clone()),
200                    present(r.http.as_ref()),
201                    present(r.cli.as_ref()),
202                ])
203            })
204            .collect();
205        card(
206            "Tools that write the repository",
207            el("div")
208                .child(table(&["Tool", "Capability", "HTTP", "Command line"], rows))
209                .child(el("p").class("mj-note").text(
210                    "Each is announced to a client as not read-only, destructive and not idempotent, and the `initialize` instructions name them. The handler a tool reaches is the one its HTTP route reaches: a refusal there is the same refusal here.",
211                )),
212        )
213    };
214
215    // ------------------------------------------------------------ the tools
216    let here = |effect: Option<Effect>| match effect {
217        Some(e) => format!("{HERE}?effect={}", effect_word(e)),
218        None => HERE.to_string(),
219    };
220    let mut filter = vec![(
221        "every tool".to_string(),
222        here(None),
223        p.tool_count,
224        asked_effect.is_none(),
225    )];
226    for e in &p.effects {
227        filter.push((
228            effect_label(e.effect).to_string(),
229            here(Some(e.effect)),
230            e.tools,
231            asked_effect == Some(effect_word(e.effect)),
232        ));
233    }
234    let tool_rows: Vec<El> = p
235        .tools
236        .iter()
237        .map(|t| {
238            row(vec![
239                cell(mono(t.name.clone())),
240                id_cell(capability_href(&t.capability), t.capability.clone()),
241                text_cell(t.title.clone()),
242                cell(effect_badge(t.effect)),
243            ])
244        })
245        .collect();
246    let tools = card(
247        "Tools",
248        el("div")
249            .child(chips(filter))
250            .child(if tool_rows.is_empty() {
251                nothing("No tool has this effect.")
252            } else {
253                table(&["Tool", "Capability", "Title", "Effect"], tool_rows)
254            })
255            .child(el("p").class("mj-note").text(
256                "The capability's page shows the input and output schemas a tool carries; they are the capability's own, never restated for MCP.",
257            )),
258    );
259
260    // ------------------------------------------------------------ the clients
261    let command = if p.launcher_in_repository {
262        format!("bin/{LAUNCHER}")
263    } else {
264        LAUNCHER.to_string()
265    };
266    let client_rows: Vec<El> = p
267        .clients
268        .iter()
269        .map(|c| {
270            let (status, label) = match c.standing {
271                McpClientStanding::Wired => ("ok", "starts this repository's server"),
272                McpClientStanding::Foreign => ("warn", "exists, does not name the launcher"),
273                McpClientStanding::Absent => ("info", "not written here"),
274            };
275            row(vec![
276                text_cell(c.title.clone()),
277                cell(mono(c.config.clone())),
278                cell(badge(status, label)),
279            ])
280        })
281        .collect();
282    let mut clients_body = el("div");
283    for f in &p.findings {
284        clients_body = clients_body.child(alert("warn", format!("{} — {}", f.message, f.remedy)));
285    }
286    clients_body = clients_body
287        .child(if client_rows.is_empty() {
288            nothing("The distribution declares no client configuration.")
289        } else {
290            table(&["Client", "Reads", "Here"], client_rows)
291        })
292        .child(el("p").class("mj-prose").text(
293            "A client is connected by one entry in the file it reads: a stdio server whose command is the launcher. It needs no argument, no environment variable and no secret.",
294        ))
295        .child(pre(
296            serde_json::to_string_pretty(&json!({
297                "mcpServers": { "majordomus": { "type": "stdio", "command": command, "args": [] } }
298            }))
299            .unwrap_or_default(),
300        ))
301        .child(el("p").class("mj-note").text(
302            "The launcher starts the server in the repository the client was opened in, wherever inside it the client's working directory is, and attaches to the one already running when there is one.",
303        ));
304    let clients = card("Clients", clients_body);
305
306    // ------------------------------------------------------------ the matrix
307    let executable: Vec<_> = matrix
308        .rows
309        .iter()
310        .filter(|r| r.cli.is_some() || r.http.is_some() || r.mcp_tool.is_some())
311        .collect();
312    let unmet = executable.iter().filter(|r| !r.closed).count();
313    let matrix_rows: Vec<El> = executable
314        .iter()
315        .map(|r| {
316            row(vec![
317                id_cell(capability_href(&r.id), r.id.clone()),
318                cell(effect_badge(r.effect)),
319                present(r.cli.as_ref()),
320                present(r.http.as_ref()),
321                present(r.mcp_tool.as_ref()),
322                present(r.mcp_resource.as_ref()),
323            ])
324        })
325        .collect();
326    let reach = card_with(
327        "Where each capability is reachable",
328        badge(
329            if unmet == 0 { "ok" } else { "fail" },
330            if unmet == 0 {
331                "every declared exposure is answered".to_string()
332            } else {
333                format!("{unmet} declared exposure(s) unanswered")
334            },
335        ),
336        el("div")
337            .child(table(
338                &[
339                    "Capability",
340                    "Effect",
341                    "Command line",
342                    "HTTP",
343                    "MCP tool",
344                    "MCP resource",
345                ],
346                matrix_rows,
347            ))
348            .child(el("p").class("mj-note").text(
349                "A row per capability that is executable from somewhere, from `capabilities.projections`. A dash is a projection the capability does not declare, which is a decision recorded on its declaration and not an omission of this table.",
350            )),
351    );
352
353    Page::new(
354        Area::Mcp,
355        "MCP",
356        el("div")
357            .class("mj-grid")
358            .child(overview)
359            .child(writers)
360            .child(clients)
361            .child(tools)
362            .child(reach),
363    )
364    .subtitle("The MCP projection of the capability registry: what a client is served, what each tool may change, and how a client is connected.")
365    .trail(vec![("Cockpit", Some("/cockpit")), ("MCP", None)])
366}

372    /// A description of the projection as the capability answers it, with the parts a
373    /// test varies passed in.
374    fn projection(
375        writers: &[&str],
376        tools: serde_json::Value,
377        launcher_in_repository: bool,
378        clients: serde_json::Value,
379        findings: serde_json::Value,
380    ) -> McpProjection {
381        serde_json::from_value(json!({
382            "server": { "name": "majordomus", "title": "Majordomus", "version": "0.0.0" },
383            "protocol_versions": ["2025-06-18"],
384            "transports": [{ "id": "stdio", "reached_by": "majordomus mcp", "attached": 1 }],
385            "serving": {
386                "methods": ["initialize", "tools/list"],
387                "tools": true, "resources": true, "prompts": false, "notifications": false
388            },
389            "tool_count": 2,
390            "tools": tools,
391            "effects": [{ "effect": "read", "tools": 2 }],
392            "writers": writers,
393            "resources": { "builtin": 1, "objects": 3, "template": "majordomus://{kind}/{id}" },
394            "launcher_in_repository": launcher_in_repository,
395            "clients": clients,
396            "findings": findings
397        }))
398        .expect("a projection as the capability answers it")
399    }

401    fn matrix(rows: serde_json::Value) -> Matrix {
402        serde_json::from_value(json!({ "rows": rows, "unbacked": [], "unclassified": [] }))
403            .expect("a matrix as the capability answers it")
404    }

406    fn row_of(id: &str, tool: Option<&str>, effect: &str, closed: bool) -> serde_json::Value {
407        let mut row = json!({
408            "id": id, "module": "m", "kind": "query", "stability": "stable",
409            "effect": effect, "http": format!("GET /api/v1/{id}"), "closed": closed,
410            "source": "builtin"
411        });
412        if let Some(tool) = tool {
413            row["mcp_tool"] = json!(tool);
414        }
415        row
416    }

426    #[test]
427    fn a_registry_with_no_writer_says_so_and_one_with_a_writer_names_its_routes() {
428        let rows = matrix(json!([
429            row_of("a.read", Some("majordomus_read"), "read", true),
430            row_of(
431                "a.write",
432                Some("majordomus_write"),
433                "repository_mutation",
434                true
435            ),
436        ]));
437        let none = rendered(
438            &projection(&[], a_read_tool(), false, json!([]), json!([])),
439            &rows,
440            None,
441        )
442        .main
443        .render();
444        assert!(none.contains("No tool of this registry writes the repository."));
445
446        let one = rendered(
447            &projection(
448                &["majordomus_write"],
449                a_read_tool(),
450                false,
451                json!([]),
452                json!([]),
453            ),
454            &rows,
455            None,
456        )
457        .main
458        .render();
459        assert!(!one.contains("No tool of this registry writes the repository."));
460        assert!(one.contains("majordomus_write"));
461        assert!(one.contains("GET /api/v1/a.write"));
462    }

491    #[test]

493        let rows = matrix(json!([row_of(
494            "a.read",
495            Some("majordomus_read"),
496            "read",
497            true
498        )]));
499        let clients = json!([
500            { "id": "claude", "title": "Claude Code", "config": ".mcp.json", "standing": "wired" },
501            { "id": "codex", "title": "Codex", "config": ".codex/config.toml", "standing": "foreign" },
502            { "id": "gemini", "title": "Gemini CLI", "config": ".gemini/settings.json", "standing": "absent" },
503        ]);
504        let findings = json!([{
505            "code": "mcp_client_config_foreign",
506            "message": "the codex file names something else",
507            "remedy": "name the launcher in it"
508        }]);
509        let here = rendered(
510            &projection(&[], a_read_tool(), true, clients.clone(), findings),
511            &rows,
512            None,
513        )
514        .main
515        .render();
516        assert!(
517            here.contains("starts this repository&#39;s server")
518                || here.contains("starts this repository's server")
519        );
520        assert!(here.contains("exists, does not name the launcher"));
521        assert!(here.contains("not written here"));
522        assert!(here.contains("the codex file names something else — name the launcher in it"));
523        assert!(here.contains(&format!("bin/{LAUNCHER}")));
524
525        let installed = rendered(
526            &projection(&[], a_read_tool(), false, clients, json!([])),
527            &rows,
528            None,
529        )
530        .main
531        .render();
532        assert!(!installed.contains(&format!("bin/{LAUNCHER}")));
533        assert!(installed.contains(LAUNCHER));
534
535        let undeclared = rendered(
536            &projection(&[], a_read_tool(), false, json!([]), json!([])),
537            &rows,
538            None,

541        .render();
542        assert!(undeclared.contains("The distribution declares no client configuration."));
543    }
