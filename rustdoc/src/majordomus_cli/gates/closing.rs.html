1//! The closing readings: what a task leaves behind that was not there when it started.
2//!
3//! A person ending a session asks the same things every time — is any debt added, are
4//! there more branches and worktrees than at the start, are there more open pull requests —
5//! and a worker answers from what it remembers doing, which is the one source that cannot
6//! see what another worker did in the same hour and has every reason to round up. On
7//! 2026-10-10 that question was put to one session seven times and answered seven times from
8//! recollection; the eighth answer, measured, differed from all of them.
9//!
10//! So the three things asked are read here, each from the record that already holds it:
11//!
12//! - **recorded debt** from the ratchet baselines under `.ai/repo/` — every
13//!   `*-baseline.txt` is a list of known violations a gate tolerates, one a line, so the
14//!   number of entries at the commit the task started at and in the working tree now are
15//!   two facts git has;
16//! - **branches and worktrees** from git's own reflogs, which carry the moment each was
17//!   created: the first line of `logs/refs/heads/<branch>` and of a linked worktree's
18//!   `logs/HEAD`;
19//! - **the backlog** from the last recorded forge observation, whose pull requests carry
20//!   the moment each was opened.
21//!
22//! Nothing here judges. A reading is taken or it is an `Err` with the reason, and
23//! [`super::done`] is where a reading becomes an answer — where an unread one is `unknown`
24//! and never a pass.
25//!
26//! # What these readings are not
27//!
28//! They are the repository's, not one worker's. A branch created by another session since
29//! this task started is counted, because nothing in git says who created a branch, and the
30//! question a person asks at the end — did this get bigger while we worked — is about the
31//! repository. And "created since and still here" is not "more than at the start": what was
32//! removed in the meantime leaves no record, so the readings name what was added and
33//! remains and claim nothing about the difference of two totals.
34

40/// The directory a repository's debt baselines live in, repository-relative.
41pub(crate) const DEBT_DIRECTORY: &str = ".ai/repo";
42
43/// What makes a file there a debt baseline. Discovered by this suffix and never listed: a
44/// baseline added tomorrow is read tomorrow.
45pub(crate) const DEBT_SUFFIX: &str = "-baseline.txt";
46
47/// One baseline's entries, at the commit the task started at and in the working tree.
48#[derive(Debug, Clone, PartialEq, Eq)]
49pub(crate) struct DebtReading {
50    /// The baseline, repository-relative.
51    pub path: String,
52    /// Its entries at the start; `None` when the file did not exist there.
53    pub before: Option<usize>,
54    /// Its entries now.
55    pub after: usize,
56}
57
58impl DebtReading {
59    /// How many entries it gained. A baseline that did not exist at the start gained
60    /// nothing: the debt it lists was there unrecorded, and recording it is not adding it.
61    pub(crate) fn grown_by(&self) -> usize {
62        match self.before {
63            Some(before) => self.after.saturating_sub(before),
64            None => 0,
65        }
66    }
67}

69/// What was created since the task started and is still there.
70#[derive(Debug, Clone, Default, PartialEq, Eq)]
71pub(crate) struct Accumulated {
72    /// The linked worktrees registered since, by path. A set: each is there once, and its
73    /// order is the paths' own.
74    pub worktrees: BTreeSet<String>,
75    /// The local branches created since, by name.
76    pub branches: BTreeSet<String>,
77    /// Every linked worktree registered now.
78    pub worktrees_now: usize,
79    /// Every local branch now.
80    pub branches_now: usize,
81    /// Branches and worktrees whose creation git has no record of, and which are therefore
82    /// counted on neither side.
83    pub undated: usize,
84}

86/// The pull requests opened since the task started and still open when the forge was last

98/// Everything the closing questions are answered from. Each is its own `Result`: one
99/// reading that could not be taken does not take the others with it.
100#[derive(Debug, Clone)]
101pub(crate) struct Closing {
102    /// The task's starting moment the other two are measured from, RFC 3339.
103    pub since: String,
104    /// Every debt baseline, sorted by path.
105    pub debt: Result<Vec<DebtReading>, String>,
106    /// Branches and worktrees.
107    pub accumulated: Result<Accumulated, String>,
108    /// Pull requests.
109    pub backlog: Result<Backlog, String>,
110}

112/// Take every closing reading for a task that started at commit `base`, at `since`.
113pub(crate) fn read(root: &Path, base: &str, since: &str) -> Closing {
114    let started = crate::peers::parse_rfc3339(since)
115        .ok_or_else(|| format!("the task's starting moment '{since}' is not an RFC 3339 UTC time"));
116    Closing {
117        since: since.to_string(),
118        debt: debt(root, base),
119        accumulated: started.clone().and_then(|at| accumulated(root, at)),
120        backlog: started.and_then(|at| {
121            let observation = crate::integration::load_observation(root)?;
122            backlog(observation.as_ref(), at)
123        }),
124    }
125}

127/// How many entries a baseline's text holds: every line that is neither blank nor a comment.
128pub(crate) fn entries(text: &str) -> usize {
129    text.lines()
130        .map(str::trim)
131        .filter(|line| !line.is_empty() && !line.starts_with('#'))
132        .count()
133}

135/// Every debt baseline of the working tree, with its entries at `base` and now.
136pub(crate) fn debt(root: &Path, base: &str) -> Result<Vec<DebtReading>, String> {
137    if base.is_empty() || base == "NONE" {
138        return Err("the task records no commit it started at".into());
139    }
140    let directory = root.join(DEBT_DIRECTORY);
141    let names: BTreeSet<String> = match std::fs::read_dir(&directory) {
142        Ok(listing) => listing
143            .filter_map(|entry| entry.ok())
144            .filter_map(|entry| entry.file_name().into_string().ok())
145            .filter(|name| name.ends_with(DEBT_SUFFIX))
146            .collect(),
147        Err(e) if e.kind() == std::io::ErrorKind::NotFound => BTreeSet::new(),
148        Err(e) => return Err(format!("{}: {e}", directory.display())),
149    };
150    let paths: Vec<String> = names
151        .iter()
152        .map(|name| format!("{DEBT_DIRECTORY}/{name}"))
153        .collect();
154    // The commit itself is the first object asked for, so that one batch both proves the
155    // start exists and reads every baseline as it stood there.
156    let mut asked = vec![format!("{base}^{{commit}}")];
157    asked.extend(paths.iter().map(|path| format!("{base}:{path}")));
158    let mut then = batch(root, &asked)?.into_iter();
159    if then.next().flatten().is_none() {
160        return Err(format!(
161            "the commit the task started at ({base}) is not in this repository"
162        ));
163    }
164    let mut readings = Vec::with_capacity(paths.len());
165    for (path, before) in paths.into_iter().zip(then) {
166        let now = std::fs::read_to_string(root.join(&path)).map_err(|e| format!("{path}: {e}"))?;
167        readings.push(DebtReading {
168            before: before.as_deref().map(entries),
169            after: entries(&now),
170            path,
171        });
172    }
173    Ok(readings)
174}

176/// The body of each object `asked` names, in order; `None` for one the repository lacks.
177///
178/// One `git cat-file --batch` for all of them: this is read on every `majordomus check`, and
179/// a process per baseline would be paid there sixteen times over in this repository. Whatever
180/// goes wrong on the way — git absent, a write cut short, a directory that is no repository —
181/// ends in an answer that is not a whole batch, and that is the one refusal.
182fn batch(root: &Path, asked: &[String]) -> Result<Vec<Option<String>>, String> {
183    use std::io::Write;
184    use std::process::Stdio;
185
186    let request: String = asked.iter().map(|object| format!("{object}\n")).collect();
187    let answer = crate::git::read_only(root)
188        .args(["cat-file", "--batch"])
189        .stdin(Stdio::piped())
190        .stdout(Stdio::piped())
191        .stderr(Stdio::null())
192        .spawn()
193        .and_then(|mut child| {
194            // the request is a few hundred bytes, written whole before anything is read; the
195            // pipe closes when `input` is dropped, which is what ends the batch, and a write
196            // cut short leaves an answer that is refused below
197            let _ = child
198                .stdin
199                .take()
200                .map(|mut input| input.write_all(request.as_bytes()));
201            child.wait_with_output()
202        })
203        .map(|out| out.stdout)
204        .unwrap_or_default();
205    parse_batch(&answer, asked.len()).ok_or_else(|| {
206        format!(
207            "git cat-file --batch did not answer for {} object(s) in {}",
208            asked.len(),
209            root.display()
210        )
211    })
212}

214/// The bodies of a `git cat-file --batch` answer: `<oid> <type> <size>\n<body>\n` for an
215/// object that exists, one line ending in ` missing` for one that does not.
216pub(crate) fn parse_batch(mut answer: &[u8], expected: usize) -> Option<Vec<Option<String>>> {
217    let mut bodies = Vec::with_capacity(expected);
218    for _ in 0..expected {
219        let end = answer.iter().position(|b| *b == b'\n')?;
220        let header = std::str::from_utf8(&answer[..end]).ok()?;
221        answer = &answer[end + 1..];
222        if header.ends_with(" missing") {
223            bodies.push(None);
224            continue;
225        }
226        let (_, size) = header.rsplit_once(' ')?;
227        let size: usize = size.parse().ok()?;
228        let body = answer.get(..size)?;
229        bodies.push(Some(String::from_utf8_lossy(body).into_owned()));
230        // the body is followed by one newline of the protocol's own
231        answer = answer.get(size + 1..)?;
232    }
233    Some(bodies)
234}

236/// The moment a reflog's first line records, in seconds since the epoch.
237///
238/// A reflog line is `<old> <new> <name> <email> <seconds> <zone>\t<message>`; the name may
239/// hold spaces, so the two fields are taken from the end of what precedes the tab.
240pub(crate) fn first_recorded(reflog: &str) -> Option<u64> {
241    let line = reflog.lines().next()?;
242    let head = line.split_once('\t').map_or(line, |(head, _)| head);
243    let mut fields = head.split_whitespace().rev();
244    let _zone = fields.next()?;
245    fields.next()?.parse().ok()
246}

248/// What git answers to `args` in `root`, or the reason it did not. A git that cannot be
249/// started and a git that refuses are one answer here: neither said anything to read.
250fn git_text(root: &Path, args: &[&str]) -> Result<String, String> {
251    crate::git::read_only(root)
252        .args(args)
253        .output()
254        .ok()
255        .filter(|out| out.status.success())
256        .map(|out| String::from_utf8_lossy(&out.stdout).into_owned())
257        .ok_or_else(|| {
258            format!(
259                "git {} did not answer in {}",
260                args.join(" "),
261                root.display()
262            )
263        })
264}

266/// The branches and linked worktrees created at or after `started` that still exist.
267pub(crate) fn accumulated(root: &Path, started: u64) -> Result<Accumulated, String> {
268    // one question: the directory every worktree shares, then every local branch by name
269    let answer = git_text(
270        root,
271        &[
272            "rev-parse",
273            "--path-format=absolute",
274            "--git-common-dir",
275            "--symbolic-full-name",
276            "--branches",
277        ],
278    )?;
279    let mut lines = answer.lines();
280    let common = PathBuf::from(lines.next().unwrap_or_default().trim());
281    let branches: Vec<&str> = lines
282        .filter_map(|line| line.strip_prefix("refs/heads/"))
283        .collect();
284    let mut found = Accumulated::default();
285    for branch in branches {
286        found.branches_now += 1;
287        let log = common.join("logs/refs/heads").join(branch);
288        match std::fs::read_to_string(&log)
289            .ok()
290            .as_deref()
291            .and_then(first_recorded)
292        {
293            Some(created) if created >= started => {
294                found.branches.insert(branch.to_string());
295            }
296            Some(_) => {}
297            None => found.undated += 1,
298        }
299    }
300    // A linked worktree is a directory under <common>/worktrees/; its `gitdir` names the
301    // checkout and its own HEAD reflog begins when it was added.
302    let registered = match std::fs::read_dir(common.join("worktrees")) {
303        Ok(listing) => listing.filter_map(|entry| entry.ok()).collect::<Vec<_>>(),
304        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Vec::new(),
305        Err(e) => return Err(format!("{}: {e}", common.join("worktrees").display())),
306    };
307    for entry in registered {
308        let admin = entry.path();
309        let Ok(gitdir) = std::fs::read_to_string(admin.join("gitdir")) else {
310            continue;
311        };
312        found.worktrees_now += 1;
313        let checkout = gitdir.trim().trim_end_matches("/.git").to_string();
314        let log = admin.join("logs/HEAD");
315        match std::fs::read_to_string(&log)
316            .ok()
317            .as_deref()
318            .and_then(first_recorded)
319        {
320            Some(created) if created >= started => {
321                found.worktrees.insert(checkout);
322            }
323            Some(_) => {}
324            None => found.undated += 1,
325        }
326    }
327    Ok(found)
328}

330/// The pull requests of `observation` opened at or after `started`.
331///
332/// An observation taken before the task started cannot have seen what the task's hours
333/// brought, so it is an `Err` and not an empty answer.
334pub(crate) fn backlog(
335    observation: Option<&ForgeObservation>,
336    started: u64,
337) -> Result<Backlog, String> {
338    let Some(observation) = observation else {
339        return Err("the forge has never been observed from this checkout".into());
340    };
341    let observed = crate::peers::parse_rfc3339(&observation.observed_at).ok_or_else(|| {
342        format!(
343            "the observation's moment '{}' could not be read",
344            observation.observed_at
345        )
346    })?;
347    if observed < started {
348        return Err(format!(
349            "the forge was last observed at {}, before this task started",
350            observation.observed_at
351        ));
352    }
353    let opened_since: BTreeSet<u64> = observation
354        .pull_requests
355        .iter()
356        .filter(|pull| {
357            crate::peers::parse_rfc3339(&pull.created_at).is_some_and(|at| at >= started)
358        })
359        .map(|pull| pull.number)
360        .collect();
361    Ok(Backlog {
362        observed_at: observation.observed_at.clone(),
363        open: observation.pull_requests.len(),
364        opened_since,
365    })
366}

619        let (_dir, root, _) = repository();
620        let trunk_created = first_recorded(
621            &std::fs::read_to_string(root.join(".git/logs/refs/heads/master")).unwrap(),
622        )
623        .unwrap();
624        git(&root, &["branch", "feature/kept"]);
625        let wt = root.parent().unwrap().join("wt");
626        git(
627            &root,
628            &[
629                "worktree",
630                "add",
631                "-q",
632                "-b",
633                "feature/held",
634                wt.to_str().unwrap(),
635            ],
636        );
637
638        // measured from before anything existed, everything is new
639        let all = accumulated(&root, 0).unwrap();
640        assert_eq!(
641            names(&all.branches),
642            ["feature/held", "feature/kept", "master"]
643        );
644        assert_eq!(names(&all.worktrees), [wt.to_str().unwrap()]);
645        assert_eq!(
646            (all.branches_now, all.worktrees_now, all.undated),
647            (3, 1, 0)
648        );
649
650        // measured from after everything existed, nothing is — and the totals are the same
651        let none = accumulated(&root, trunk_created + 86_400).unwrap();
652        assert!(none.branches.is_empty() && none.worktrees.is_empty());
653        assert_eq!((none.branches_now, none.worktrees_now), (3, 1));
654
655        // what is removed again is no longer there to name
656        git(&root, &["worktree", "remove", wt.to_str().unwrap()]);
657        git(&root, &["branch", "-D", "feature/held", "feature/kept"]);
658        let after = accumulated(&root, 0).unwrap();
659        assert_eq!(names(&after.branches), ["master"]);
660        assert!(after.worktrees.is_empty());

673    fn observation(observed_at: &str, opened: &[(u64, &str)]) -> ForgeObservation {
674        let pulls: Vec<serde_json::Value> = opened
675            .iter()
676            .map(|(number, created_at)| {
677                serde_json::json!({
678                    "number": number, "title": "t", "body": "", "author": "a",
679                    "author_association": "OWNER", "draft": false, "auto_merge": false,
680                    "head_ref": format!("feature/{number}"), "head_sha": "0".repeat(40),
681                    "base_ref": "master", "cross_repository": false,
682                    "created_at": created_at, "updated_at": created_at,
683                    "labels": [], "checks": [], "latest_reviews": [], "review_requests": [],
684                    "review_decision": "", "cross_references": "whole",
685                })
686            })
687            .collect();
688        serde_json::from_value(serde_json::json!({
689            "schema": crate::integration::forge::OBSERVATION_SCHEMA,
690            "repository": "o/r", "base": "master", "base_sha": "0".repeat(40),
691            "observed_at": observed_at, "required_checks": null, "review_policy": null,
692            "up_to_date_required": null, "merge_methods": ["merge"],
693            "pull_requests": pulls, "resolved": {}, "delete_branch_on_merge": null,
694        }))
695        .unwrap()
696    }
