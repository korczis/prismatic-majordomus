1//! `majordomus completion`: the query a shell adapter makes, and the adapter itself.
2//!
3//! The query is the hot path — it runs on every TAB — so it takes the fast load: the clap
4//! tree, the shell tool's registry, and the workflows from the cache the last
5//! materialisation wrote. No index, no subprocess, no network, no server. When the cache is
6//! absent the answer is smaller and still correct, which is the right failure for a
7//! completion.
8

19/// Run `majordomus completion`.
20pub fn run(args: CompletionArgs) -> Result<u8> {
21    match args.command {
22        None => init(&CompletionInitArgs {
23            shell: CompletionShell::Zsh,
24        }),
25        Some(CompletionCommand::Init(ref init_args)) => init(init_args),
26        Some(CompletionCommand::Query(ref query_args)) => query(&args, query_args),
27        Some(CompletionCommand::Install(ref install_args)) => install(install_args),
28    }
29}

31/// The markers the managed block is written between.
32///
33/// Everything outside them belongs to the person whose file this is and is never read,
34/// rewritten or reordered. Everything between them belongs to this command and is replaced
35/// wholesale, which is what makes installing twice the same as installing once.
36const BEGIN: &str = "# >>> MAJORDOMUS >>>";
37const END: &str = "# <<< MAJORDOMUS <<<";
38
39/// `completion install`.
40///
41/// The only thing in this executable that writes outside the repository, and it is its own
42/// command for that reason: putting a line into somebody's shell startup file is a decision
43/// a person makes, never a side effect of initialising a repository.
44fn install(args: &CompletionInstallArgs) -> Result<u8> {
45    let path = match &args.rc {
46        Some(p) => p.clone(),
47        None => default_rc(args.shell)?,
48    };
49
50    let existing = std::fs::read_to_string(&path).unwrap_or_default();
51    let wanted = if args.remove {
52        None
53    } else {
54        Some(format!(
55            "{BEGIN}\n\
56             # Written by `majordomus completion install`. Everything between these markers is\n\
57             # replaced when it runs again, and removed by `--remove`; nothing outside them is read.\n\
58             # The integration carries no command of its own — it asks the executable for every\n\
59             # candidate — so it serves every repository and never goes stale.\n\
60             eval \"$({} completion init --shell {})\"\n\
61             {END}",
62            program(),
63            shell_name(args.shell),
64        ))
65    };
66
67    let updated = splice(&existing, wanted.as_deref());
68    let stdout = std::io::stdout();
69    let mut out = stdout.lock();
70
71    if updated == existing {
72        writeln!(
73            out,
74            "completion: {} is already as asked; nothing written",
75            path.display()
76        )
77        .map_err(Error::Transport)?;
78        return Ok(0);
79    }
80    if args.dry_run {
81        writeln!(
82            out,
83            "completion: would {} the managed block in {}",
84            if args.remove { "remove" } else { "write" },
85            path.display()
86        )
87        .map_err(Error::Transport)?;
88        return Ok(0);
89    }
90
91    // A backup only when there was something to lose, and only the first time: a person who
92    // runs this twice does not want two copies of their shell configuration lying about.
93    if !existing.is_empty() {
94        let backup = path.with_extension("majordomus.bak");
95        if !backup.exists() {
96            std::fs::write(&backup, &existing).map_err(|e| Error::io(backup.clone(), e))?;
97            writeln!(out, "completion: kept {}", backup.display()).map_err(Error::Transport)?;
98        }
99    }
100    if let Some(parent) = path.parent() {
101        let _ = std::fs::create_dir_all(parent);
102    }
103    std::fs::write(&path, &updated).map_err(|e| Error::io(path.clone(), e))?;
104    writeln!(
105        out,
106        "completion: {} the managed block in {}\ncompletion: open a new shell, or `source {}`",
107        if args.remove { "removed" } else { "wrote" },
108        path.display(),
109        path.display()
110    )
111    .map_err(Error::Transport)?;
112    Ok(0)
113}

115/// The program name the installed line calls.
116///
117/// `majordomus` rather than this executable's path: the integration is generic and outlives
118/// any one build directory, and inside a repository the environment names the executable
119/// that answers through `MAJORDOMUS_COMPLETION_BIN`.
120fn program() -> &'static str {
121    "majordomus"
122}

124/// The shell's own name, as its `--shell` value spells it.
125fn shell_name(shell: CompletionShell) -> &'static str {
126    match shell {
127        CompletionShell::Zsh => "zsh",
128        CompletionShell::Bash => "bash",
129        CompletionShell::Fish => "fish",
130    }

133/// Where a shell reads its startup from, when the caller does not say.
134fn default_rc(shell: CompletionShell) -> Result<std::path::PathBuf> {
135    let home = std::env::var_os("HOME")
136        .map(std::path::PathBuf::from)
137        .ok_or_else(|| Error::Refused {
138            code: 12,
139            reason:
140                "HOME is not set, so there is no startup file to install into; name one with --rc"
141                    .into(),
142        })?;
143    Ok(match shell {
144        CompletionShell::Zsh => home.join(".zshrc"),
145        CompletionShell::Bash => home.join(".bashrc"),
146        CompletionShell::Fish => home.join(".config/fish/config.fish"),
147    })
148}

150/// Replace the managed block with `block`, or remove it when `block` is `None`.
151///
152/// Pure, so the whole of the decision is testable without a filesystem: the caller compares
153/// the answer with what the file already held and writes only on a difference, which is why
154/// running this twice is not two writes.
155fn splice(existing: &str, block: Option<&str>) -> String {
156    let (before, after) = match (existing.find(BEGIN), existing.find(END)) {
157        (Some(b), Some(e)) if e > b => {
158            let end = e + END.len();
159            let after = existing[end..]
160                .strip_prefix('\n')
161                .unwrap_or(&existing[end..]);
162            (&existing[..b], after)
163        }
164        // No block, or one whose markers are damaged: leave every byte alone and append.
165        _ => (existing, ""),
166    };
167    let mut out = String::new();
168    out.push_str(before);
169    if let Some(block) = block {
170        if !out.is_empty() && !out.ends_with('\n') {
171            out.push('\n');
172        }
173        out.push_str(block);
174        out.push('\n');
175    }
176    out.push_str(after);
177    out
178}

180/// `completion init`.
181fn init(args: &CompletionInitArgs) -> Result<u8> {
182    let stdout = std::io::stdout();
183    let mut out = stdout.lock();
184    let script = match args.shell {
185        CompletionShell::Zsh => shell::ZSH,
186        CompletionShell::Bash => shell::BASH,
187        CompletionShell::Fish => shell::FISH,
188    };
189    write!(out, "{script}").map_err(Error::Transport)?;
190    Ok(0)
191}

193/// `completion query`.
194fn query(args: &CompletionArgs, query_args: &CompletionQueryArgs) -> Result<u8> {
195    // A completion never fails a shell. A repository that cannot be found, a cache that
196    // cannot be read and a command line that makes no sense all produce no candidates,
197    // and the shell falls back to its own file completion.
198    let Ok(loaded) = load::fast(&args.repo) else {
199        return Ok(0);
200    };
201
202    let surface = match query_args.surface {
203        CompletionSurface::Cli => Surface::Cli,
204        CompletionSurface::Workflow => Surface::Workflow,
205    };
206    let cursor = query_args
207        .cursor
208        .unwrap_or_else(|| query_args.words.len().saturating_sub(1));
209    let request = Request {
210        surface,
211        words: query_args.words.clone(),
212        cursor,
213        cwd: std::env::current_dir().ok(),
214    };
215
216    let resolver = Repository {
217        root: loaded.root.clone(),
218        graph: &loaded.graph,
219    };
220    let response = complete::complete(&loaded.graph, &request, &resolver);
221
222    let stdout = std::io::stdout();
223    let mut out = stdout.lock();
224    match query_args.format {
225        OutputFormat::Json => writeln!(
226            out,
227            "{}",
228            serde_json::to_string_pretty(&response).unwrap_or_else(|_| "{}".into())
229        )
230        .map_err(Error::Transport)?,
231        OutputFormat::Text => {
232            for candidate in &response.candidates {
233                match &candidate.description {
234                    Some(d) if !d.is_empty() => {
235                        writeln!(out, "{}\t{}", candidate.value, one_line(d))
236                    }
237                    _ => writeln!(out, "{}", candidate.value),
238                }
239                .map_err(Error::Transport)?;
240            }
241        }
242    }
243    Ok(0)
244}

246/// One line, with nothing that would break a shell's parsing of the protocol.
247fn one_line(text: &str) -> String {
248    text.lines()
249        .next()
250        .unwrap_or_default()
251        .chars()
252        .filter(|c| !c.is_control())
253        .collect()
254}

256/// The value sources this repository can answer without a subprocess or a server.
257///
258/// Deliberately small. A source that would cost a process, a network call or an index build
259/// answers nothing here rather than making TAB wait: the graph still says what the argument
260/// *is*, so the Cockpit's form and a machine surface can resolve it where they can afford
261/// to.
262///
263/// What *is* answered here is answered from something already in hand. A command id comes
264/// from the graph this very query resolved against; a capability id comes from composing
265/// the builtin modules, which reads no file and builds no index. Neither costs a syscall,
266/// so neither is a reason to make TAB wait.
267struct Repository<'a> {
268    root: std::path::PathBuf,
269    graph: &'a crate::command_graph::CommandGraph,
270}
271
272impl ValueResolver for Repository<'_> {
273    fn values(&self, source: ValueSource) -> Vec<String> {
274        match source {
275            // git keeps the local branches as files under refs/heads and as lines in
276            // packed-refs. Reading them is two file reads; asking git is a process.
277            ValueSource::Branch => branches(&self.root),
278            ValueSource::Shell => shell::ALL.iter().map(|(name, _)| (*name).into()).collect(),
279            // The graph is the answer to `what commands are there`, and it is already
280            // loaded — this query resolved a command line against it a moment ago.
281            ValueSource::Command => self
282                .graph
283                .commands
284                .iter()
285                .map(|c| c.id.to_string())
286                .collect(),
287            // Composing the builtin modules is compiled-in arithmetic: no index, no walk.
288            // The declarative capabilities of the layer are absent for that reason, and
289            // they are the ones no command names as an argument.
290            ValueSource::Capability => load::registry()
291                .map(|r| r.iter().map(|c| c.id.to_string()).collect())
292                .unwrap_or_default(),
293            _ => Vec::new(),
294        }
295    }
296}

298/// The local branches, read from git's own files.
299fn branches(root: &std::path::Path) -> Vec<String> {
300    let git = root.join(".git");
301    // A linked worktree's `.git` is a file naming the real directory; the branches live
302    // with the common directory, not with the worktree.
303    let common = if git.is_file() {
304        std::fs::read_to_string(&git)
305            .ok()
306            .and_then(|s| {
307                s.strip_prefix("gitdir:")
308                    .map(|p| std::path::PathBuf::from(p.trim()))
309            })
310            .and_then(|p| p.parent().and_then(|p| p.parent()).map(Path_to_owned))
311            .unwrap_or(git)
312    } else {
313        git
314    };
315
316    let mut out = Vec::new();
317    let heads = common.join("refs/heads");
318    walk(&heads, &heads, &mut out);
319    if let Ok(packed) = std::fs::read_to_string(common.join("packed-refs")) {
320        for line in packed.lines() {
321            if let Some(name) = line.split_once(" refs/heads/").map(|(_, n)| n) {
322                out.push(name.to_string());
323            }
324        }
325    }
326    out.sort();
327    out.dedup();
328    out
329}

331/// `PathBuf::to_path_buf` as a function, for the option chain above.
332#[allow(non_snake_case)]
333fn Path_to_owned(p: &std::path::Path) -> std::path::PathBuf {
334    p.to_path_buf()
335}
336
337/// Every ref file under `dir`, named relative to `base`.
338fn walk(base: &std::path::Path, dir: &std::path::Path, out: &mut Vec<String>) {
339    let Ok(entries) = std::fs::read_dir(dir) else {
340        return;
341    };
342    for entry in entries.flatten() {
343        let path = entry.path();
344        if path.is_dir() {
345            walk(base, &path, out);
346        } else if let Ok(name) = path.strip_prefix(base) {
347            out.push(name.to_string_lossy().replace('\\', "/"));
348        }
349    }
350}

356    #[test]
357    fn branches_are_read_from_git_and_never_from_a_process() {
358        let dir = tempfile::tempdir().expect("a temporary directory");
359        let heads = dir.path().join(".git/refs/heads/feature");
360        std::fs::create_dir_all(&heads).unwrap();
361        std::fs::write(heads.join("x"), "0000\n").unwrap();
362        std::fs::write(dir.path().join(".git/refs/heads/master"), "0000\n").unwrap();
363        std::fs::write(
364            dir.path().join(".git/packed-refs"),
365            "# pack-refs with: peeled\n0000 refs/heads/packed\n",
366        )
367        .unwrap();
368        let found = branches(dir.path());
369        assert!(found.contains(&"master".to_string()), "{found:?}");
370        assert!(found.contains(&"feature/x".to_string()), "{found:?}");
371        assert!(found.contains(&"packed".to_string()), "{found:?}");
372    }

374    #[test]
375    fn a_description_is_one_line_without_control_characters() {
376        assert_eq!(one_line("a\nb"), "a");
377        assert_eq!(one_line("a\tb"), "ab");
378    }

381#[cfg(test)]
382mod install_tests {
383    use super::*;
384
385    const BLOCK: &str = "# >>> MAJORDOMUS >>>\nline\n# <<< MAJORDOMUS <<<";
386
387    #[test]
388    fn an_empty_file_gains_the_block() {
389        assert_eq!(splice("", Some(BLOCK)), format!("{BLOCK}\n"));
390    }
391
392    #[test]
393    fn installing_twice_writes_the_same_bytes() {
394        let once = splice("export FOO=1\n", Some(BLOCK));
395        let twice = splice(&once, Some(BLOCK));
396        assert_eq!(once, twice, "a second install changed the file");
397    }
398
399    #[test]
400    fn nothing_outside_the_markers_is_touched() {
401        let before = "export FOO=1\nalias x=y\n";
402        let after = "unset BAR\n";
403        let installed = splice(&format!("{before}{BLOCK}\n{after}"), Some(BLOCK));
404        assert!(
405            installed.starts_with(before),
406            "the head was rewritten: {installed:?}"
407        );
408        assert!(
409            installed.ends_with(after),
410            "the tail was rewritten: {installed:?}"
411        );
412    }
413
414    #[test]
415    fn removing_leaves_the_file_as_it_was() {
416        let original = "export FOO=1\nunset BAR\n";
417        let installed = splice(original, Some(BLOCK));
418        assert_eq!(splice(&installed, None), original);
419    }
420
421    #[test]
422    fn a_file_with_damaged_markers_keeps_every_byte() {
423        // Only the opening marker: the block cannot be located, so nothing is replaced and
424        // the person's own text survives. Appending is the safe answer, never a rewrite.
425        let odd = "# >>> MAJORDOMUS >>>\nsomething a person edited\n";
426        let out = splice(odd, Some(BLOCK));
427        assert!(
428            out.starts_with(odd),
429            "a damaged block ate the file: {out:?}"
430        );
431    }
