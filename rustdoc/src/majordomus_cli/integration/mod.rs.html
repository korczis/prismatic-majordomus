1//! Pull-request integration: the next provably safe change, onto the current master, one
2//! at a time (ADR 0101).
3//!
4//! # The pipeline
5//!
6//! ```text
7//! forge (gh) ──► ForgeObservation ──► relation to master (git merge-tree, cached by SHA pair)
8//!                (recorded locally)            │
9//!                                              ▼
10//!                               classify ──► PullRequestAssessment (one per PR, with evidence)
11//!                                              │
12//!                                              ▼
13//!                                 plan ──► IntegrationQueue (ranked, deterministic)
14//!                                              │
15//!           CLI · HTTP · MCP · Cockpit ◄───────┤
16//!                                              ▼
17//!                           drain: lease ► re-observe ► re-decide ► act ► verify ► re-plan
18//! ```text
19//!
20//! The observation is the only network product and it is recorded with its moment; every
21//! read surface renders it offline. A decision names the master and head it was taken
22//! against ([`EvaluatedAgainst`]), and the executor acts only when a fresh observation
23//! still says the same — a merge invalidates every earlier plan by construction, because
24//! the next step starts from a new observation of a new master.
25//!
26//! # Why this module is not public
27//!
28//! It is `pub(crate)`. Integration is an operation of this executable, reached through its
29//! command line (`majordomus prs`) and its capability registry (`integration.*` on HTTP, MCP
30//! and OpenAPI), and not a library other crates link against, so it is not part of the
31//! crate's exported API and the exported-surface gate (`quality report`) does not cover it.
32//! What covers it instead: the unit tests beside it (`tests.rs`: every disposition, the
33//! stale-decision refusal, the re-plan after every merge, the cleanup threshold, relations
34//! computed by real git, and two properties), and
35//! `test/cases/720_integration_follows_the_current_master.sh`, which drives the real
36//! command line against a scripted forge.
37//!
38//! # Why the forge's `mergeable` is not read
39//!
40//! This repository resolves derived files with a per-clone merge driver the forge cannot
41//! run, so the forge calls nearly every pull request conflicting. The relation to master is
42//! decided here, by git, with the drivers ([`relation`]).

106/// The repository's audit trail. A checkout that still carries a trail of its own from
107/// before the trail was the repository's (`.ai/local/state/integration/events.jsonl`) has it
108/// appended to the repository's the first time this is asked while the repository has none;
109/// a marker keeps that from ever happening twice. A second checkout's old trail is left
110/// where it is: appended after another's, its lines would fold out of order.
111pub fn events_path(root: &Path) -> Result<PathBuf, String> {
112    let path = common_state_path(root, EVENTS_FILE)?;
113    let old = state_path(root, EVENTS_FILE);
114    if path.exists() || !old.is_file() {
115        return Ok(path);
116    }
117    move_trail(&old, &path).map(|()| path)
118}

120/// Append `old` to the repository's trail at `path`, once. The marker is taken with
121/// `create_new` before anything is appended, so of two processes that both find no trail,
122/// one appends; and a trail removed afterwards is not refilled.
123fn move_trail(old: &Path, path: &Path) -> Result<(), String> {
124    let dir = path.parent().unwrap_or(path);
125    let marker = dir.join(TRAIL_MOVED_MARKER);
126    // a directory that cannot be made is said by the marker's open below, which then fails
127    let _ = std::fs::create_dir_all(dir);
128    match std::fs::OpenOptions::new()
129        .write(true)
130        .create_new(true)
131        .open(&marker)
132    {
133        Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => Ok(()),
134        taken => taken
135            .and_then(|mut m| writeln!(m, "{}", old.display()))
136            .and_then(|()| std::fs::read_to_string(old))
137            .and_then(|text| {
138                std::fs::OpenOptions::new()
139                    .create(true)
140                    .append(true)
141                    .open(path)
142                    .and_then(|mut trail| writeln!(trail, "{}", text.trim_end_matches('\n')))
143            })
144            .map_err(at(path)),
145    }
146}

148/// The last recorded observation, if any.
149///
150/// A record of another schema is refused, older or newer, and never read as empty or as
151/// current: its schema is read before anything else of it, so a record a newer executable
152/// wrote is refused by its number and not by whatever field this one cannot parse. The
153/// message names both schemas and the one command that records an observation this reads.
154pub fn load_observation(root: &Path) -> Result<Option<ForgeObservation>, String> {
155    let path = state_path(root, OBSERVATION_FILE);
156    let text = match std::fs::read_to_string(&path) {
157        Ok(t) => t,
158        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(None),
159        Err(e) => return Err(format!("{}: {e}", path.display())),
160    };
161    let at = |e: serde_json::Error| format!("{}: {e}", path.display());
162    let record: serde_json::Value = serde_json::from_str(&text).map_err(at)?;
163    let other = record
164        .get("schema")
165        .and_then(serde_json::Value::as_u64)
166        .filter(|schema| *schema != u64::from(OBSERVATION_SCHEMA));
167    match other {
168        Some(schema) => Err(format!(
169            "{}: schema {schema}, this reads {OBSERVATION_SCHEMA}; majordomus prs refresh \
170             records an observation this reads",
171            path.display()
172        )),
173        // one that names no schema is no observation: the fields say so
174        None => serde_json::from_value(record).map(Some).map_err(at),
175    }
176}

178fn write_atomic(path: &Path, text: &str) -> Result<(), String> {
179    if let Some(dir) = path.parent() {
180        std::fs::create_dir_all(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
181    }
182    let tmp = path.with_extension(format!("tmp.{}", std::process::id()));
183    std::fs::write(&tmp, text).map_err(|e| format!("{}: {e}", tmp.display()))?;
184    std::fs::rename(&tmp, path).map_err(|e| format!("{}: {e}", path.display()))
185}

193/// The relation cache: `master..head` → relation, and beside it under the same key what the
194/// merge changes by kind ([`ChangeShape`]). Both SHAs are immutable, and both answers read
195/// derived attributes from master's own `.gitattributes`, so an entry is true forever and a
196/// change of either commit stales both together; the file is bounded by dropping entries
197/// whose master is not the current.
198#[derive(Debug, Default, Serialize, Deserialize)]
199struct RelationCache {
200    entries: BTreeMap<String, RelationToMaster>,
201    #[serde(default)]
202    shapes: BTreeMap<String, ChangeShape>,
203}
204
205fn relation_cached(
206    root: &Path,
207    cache: &mut RelationCache,
208    master: &str,
209    head: &str,
210) -> RelationToMaster {
211    // a ref or an abbreviation may name another commit tomorrow: only a pair of commit ids
212    // is decided here, so that every key of the cache is a fact forever
213    if !is_object_id(master) || !is_object_id(head) {
214        return RelationToMaster::Unknown {
215            reason: format!("{master}..{head} is not a pair of full commit ids"),
216        };
217    }
218    let key = format!("{master}..{head}");
219    if let Some(r) = cache.entries.get(&key) {
220        return r.clone();
221    }
222    let r = relation::relation_to_master(root, master, head);
223    // an unknown is not a fact about the SHAs; it is not cached
224    if !matches!(r, RelationToMaster::Unknown { .. }) {
225        cache.entries.insert(key, r.clone());
226    }
227    r
228}

230/// [`relation::change_shape`] through the cache, under exactly the relation's key: only a pair
231/// of full commit ids is decided, and an unanswered shape is not cached.
232fn shape_cached(
233    root: &Path,
234    cache: &mut RelationCache,
235    master: &str,
236    head: &str,
237) -> Option<ChangeShape> {
238    let key = format!("{master}..{head}");
239    (is_object_id(master) && is_object_id(head))
240        .then(|| {
241            cache.shapes.get(&key).cloned().or_else(|| {
242                relation::change_shape(root, master, head)
243                    .inspect(|s| drop(cache.shapes.insert(key.clone(), s.clone())))
244            })
245        })
246        .flatten()
247}

249/// Whether `s` is a full commit id, SHA-1 or SHA-256, as git prints one.
250fn is_object_id(s: &str) -> bool {
251    matches!(s.len(), 40 | 64) && s.bytes().all(|b| matches!(b, b'0'..=b'9' | b'a'..=b'f'))
252}

323/// The policy of an observation: required checks and reviews as the forge reported them,
324/// the label policy declared once in [`LABEL_POLICY`], and the merge method — a merge commit
325/// when the repository's settings allow one (`allow_merge_commit`, which the observation
326/// carries as `merge`), and none otherwise. Never a squash or a rebase: the derived-file
327/// driver resolves merges, and either would replay commits it never saw. With none, every
328/// pull request not already on master is held (`merge_commit_not_allowed`); one that is stays
329/// `redundant`, since closing it needs no merge.
330pub fn policy_of(obs: &ForgeObservation) -> IntegrationPolicy {
331    let merge_method = obs
332        .merge_methods
333        .iter()
334        .any(|m| m == "merge")
335        .then(|| "merge".to_string());
336    IntegrationPolicy {
337        base: obs.base.clone(),
338        required_checks: obs.required_checks.clone(),
339        review_policy: obs.review_policy,
340        skipped_permitted: Vec::new(),
341        labels: LABEL_POLICY.to_vec(),
342        merge_method,
343        up_to_date_required: obs.up_to_date_required,
344    }
345}

347/// The rank key: lane, then disposition, then risk, then how many other *ready or
348/// refreshable* pull requests it overlaps (fewer first: landing it invalidates less), then how many declared dependents wait on it (more first: landing it unblocks
349/// them), then how many authored paths it changes (fewer first), then age (older first, so
350/// easy new work cannot starve old work), then number. Every component is a value of the
351/// assessment or of the queue around it; the order is total and input-order free.
352fn rank_factors(
353    a: &PullRequestAssessment,
354    contenders: &BTreeSet<u64>,
355    dependents: &BTreeMap<u64, usize>,
356) -> RankFactors {
357    RankFactors {
358        lane: a.lane,
359        disposition: a.disposition,
360        risk: a.risk,
361        contention: a
362            .overlaps
363            .iter()
364            .filter(|o| contenders.contains(&o.number))
365            .count(),
366        dependents: dependents.get(&a.number).copied().unwrap_or(0),
367        authored_paths: a.authored_paths.len(),
368        created_at: a.created_at.clone(),
369        number: a.number,
370    }
371}
372
373fn rank_key(
374    f: &RankFactors,
375) -> (
376    u8,
377    u8,
378    u8,
379    usize,
380    std::cmp::Reverse<usize>,
381    usize,
382    String,
383    u64,
384) {
385    (
386        f.lane as u8,
387        f.disposition as u8,
388        f.risk as u8,
389        f.contention,
390        std::cmp::Reverse(f.dependents),
391        f.authored_paths,
392        f.created_at.clone(),
393        f.number,
394    )
395}

397/// Rank assessments. Deterministic whatever order they arrive in; each carries the factors
398/// it was ranked by.
399pub fn rank(assessments: Vec<PullRequestAssessment>) -> Vec<PullRequestAssessment> {
400    let contenders: BTreeSet<u64> = assessments
401        .iter()
402        .filter(|a| {
403            matches!(
404                a.disposition,
405                PullRequestDisposition::Ready | PullRequestDisposition::NeedsRefresh
406            )
407        })
408        .map(|a| a.number)
409        .collect();
410    // declared, unsatisfied edges into each pull request: what landing it would unblock
411    let mut dependents: BTreeMap<u64, usize> = BTreeMap::new();
412    for a in &assessments {
413        for d in &a.dependencies {
414            if d.certainty == DependencyCertainty::Confirmed && !d.satisfied {
415                *dependents.entry(d.number).or_default() += 1;
416            }
417        }
418    }
419    let mut ranked: Vec<PullRequestAssessment> = assessments
420        .into_iter()
421        .map(|mut a| {
422            a.rank_factors = Some(rank_factors(&a, &contenders, &dependents));
423            a
424        })
425        .collect();
426    ranked.sort_by_cached_key(|a| rank_key(a.rank_factors.as_ref().expect("set above")));
427    ranked
428}

440/// Build the queue from an observation, against `master_sha`. `relation` answers what a
441/// head is to master; the command line passes git, a test passes a table. It is asked about
442/// every open pull request, and about every authorised declared successor that is no longer
443/// open (a pull request built from what the forge reported of it, at its head and then, when
444/// the head is not on master, at its merge commit): such a successor landed exactly when one
445/// of the two is `contained`, whatever the forge calls it. A declaration nobody entitled made
446/// asks nothing of it. `shape` answers what each open pull request's merge changes
447/// by kind ([`ChangeShape`]): its authored paths decide overlaps and risk, and two that raise
448/// the version or change a release overlap as such. A pull request it does not answer is
449/// assessed on its relation's paths alone.
450pub fn build_queue_shaped(
451    obs: &ForgeObservation,
452    master_sha: &str,
453    mut relation: impl FnMut(&PullRequestObservation) -> RelationToMaster,
454    shape: impl FnMut(&PullRequestObservation) -> Option<ChangeShape>,
455) -> IntegrationQueue {
456    let policy = policy_of(obs);
457    let mut diagnostics = Vec::new();
458    if !obs.base_sha.is_empty() && obs.base_sha != master_sha {
459        diagnostics.push(format!(
460            "stale observation: the forge reported {} at {} but this clone's master is {}; majordomus prs refresh",
461            short(&obs.base_sha),
462            obs.observed_at,
463            short(master_sha)
464        ));
465    }
466    match &policy.required_checks {
467        None => diagnostics.push(
468            "the branch protection or rulesets could not be read: no pull request can be ready"
469                .into(),
470        ),
471        Some(r) if r.is_empty() => diagnostics.push(format!(
472            "{} requires no check, in its protection or any ruleset: a head can prove nothing \
473             to it, so no pull request can be ready (owner decision D5)",
474            policy.base
475        )),
476        // a context bound to an app whose check runs were not read with their writer
477        Some(required) => {
478            diagnostics.extend(classify::unread_writers(&obs.pull_requests, required))
479        }
480    }
481    if policy.merge_method.is_none() {
482        diagnostics.push(format!(
483            "{} allows no merge commit (it allows: {}): the executor never squashes or rebases, \
484             so no pull request can be ready",
485            obs.repository,
486            if obs.merge_methods.is_empty() {
487                "nothing".to_string()
488            } else {
489                obs.merge_methods.join(", ")
490            }
491        ));
492    }
493    // in number order, whatever order the forge listed them in
494    let armed: BTreeSet<u64> = obs
495        .pull_requests
496        .iter()
497        .filter(|p| p.auto_merge)
498        .map(|p| p.number)
499        .collect();
500    if !armed.is_empty() {
501        diagnostics.push(format!(
502            "auto-merge is armed on {}: the forge would merge them on its own, outside the \
503             executor, so they are unsafe until it is disarmed (gh pr merge <n> --disable-auto)",
504            armed
505                .iter()
506                .map(|n| format!("#{n}"))
507                .collect::<Vec<_>>()
508                .join(" ")
509        ));
510    }
511    let relations: Vec<RelationToMaster> = obs.pull_requests.iter().map(&mut relation).collect();
512    let mut queue = QueueContext {
513        open: obs.pull_requests.iter().map(|p| p.number).collect(),
514        // a fork's branch is not a branch of this repository, whatever it is called: a fork
515        // named `master` would otherwise stack every pull request onto itself
516        heads: obs
517            .pull_requests
518            .iter()
519            .filter(|p| !p.cross_repository)
520            .map(|p| (p.head_ref.clone(), p.number))
521            .collect(),
522        authored: BTreeMap::new(),
523        shapes: obs
524            .pull_requests
525            .iter()
526            .map(shape)
527            .zip(&obs.pull_requests)
528            .filter_map(|(s, p)| s.map(|s| (p.number, s)))
529            .collect(),
530        superseded_by: BTreeMap::new(),
531        possible_supersessions: BTreeMap::new(),
532        references_unread: BTreeMap::new(),
533        dependency_states: obs
534            .resolved
535            .iter()
536            .map(|(n, r)| {
537                let state = if r.merged {
538                    DependencyState::Merged
539                } else {
540                    DependencyState::ClosedUnmerged
541                };
542                (*n, state)
543            })
544            .collect(),
545        cycles: BTreeMap::new(),
546    };
547    queue.cycles = dependency_cycles(obs, &queue.heads);
548    if obs.pull_requests.len() >= forge::OPEN_LIMIT {
549        diagnostics.push(format!(
550            "the forge listed {} open pull requests, its limit: more may be open; a dependency on one of those reads as unread, and a listed one that one of those says it supersedes is held as unread",
551            obs.pull_requests.len()
552        ));
553    }
554    let declared = successors(obs, &queue.open, &mut relation);
555    queue.superseded_by = declared.authorised;
556    queue.possible_supersessions = declared.possible;
557    // a pull request whose cross-references were not read whole is held, and the queue says
558    // which and why: a declaration that was not read is never taken for none
559    queue.references_unread = obs
560        .pull_requests
561        .iter()
562        .filter(|p| p.cross_references != CrossReferenceRead::Whole)
563        .map(|p| (p.number, p.cross_references))
564        .collect();
565    diagnostics.extend(references_unread_diagnostic(&queue.references_unread));
566    for (p, r) in obs.pull_requests.iter().zip(&relations) {
567        let authored = match (queue.shapes.get(&p.number), r) {
568            (Some(s), _) => s.authored.clone(),
569            (
570                None,
571                RelationToMaster::UpToDate { authored } | RelationToMaster::Behind { authored, .. },
572            ) => authored.clone(),
573            (None, RelationToMaster::Conflicting { paths }) => paths.clone(),
574            _ => Vec::new(),
575        };
576        queue.authored.insert(p.number, authored);
577    }
578    let assessments: Vec<PullRequestAssessment> = obs
579        .pull_requests
580        .iter()
581        .zip(&relations)
582        .map(|(p, r)| classify(p, r, master_sha, &obs.observed_at, &policy, &queue))
583        .collect();
584    let mut queue = IntegrationQueue {
585        repository: obs.repository.clone(),
586        base: obs.base.clone(),
587        master_sha: master_sha.to_string(),
588        observed_at: obs.observed_at.clone(),
589        observed_base_sha: obs.base_sha.clone(),
590        next_merge: None,
591        next_refresh: Vec::new(),
592        policy,
593        assessments: rank(assessments),
594        tallies: QueueTallies::default(),

602/// The open pull requests caught in a cycle of confirmed dependencies
603/// ([`classify::confirmed_dependencies`], between open pull requests only), each with the
604/// others it can reach and that reach it. A pull request on such a cycle waits for itself,
605/// so nothing about the others can let it land first. The module is private, so the
606/// example is text; `dependency_queue` runs the same assertions.
607///
608/// ```text
609/// use majordomus_cli::integration::{forge, ForgeObservation, OBSERVATION_SCHEMA};
610/// let pr = |n: u64, body: &str| forge::pull_request_of(&serde_json::json!({
611///     "number": n, "title": "t", "author": {"login": "a"}, "headRefName": format!("f/{n}"),
612///     "headRefOid": format!("{n:040}"), "baseRefName": "master", "isDraft": false,
613///     "labels": [], "createdAt": "t", "updatedAt": "t", "body": body,
614///     "statusCheckRollup": [], "reviewDecision": "", "autoMergeRequest": null,
615///     "isCrossRepository": false})).unwrap();
616/// let mut obs: ForgeObservation = serde_json::from_value(serde_json::json!({
617///     "schema": OBSERVATION_SCHEMA, "repository": "o/r", "base": "master", "base_sha": "m",
618///     "observed_at": "t", "required_checks": null, "review_policy": null,
619///     "merge_methods": ["merge"], "pull_requests": []})).unwrap();
620/// obs.pull_requests = vec![pr(1, "Depends on #2"), pr(2, "Depends on #1"), pr(3, "Depends on #1")];
621/// let cycles = majordomus_cli::integration::dependency_cycles(&obs, &Default::default());
622/// assert_eq!(cycles.get(&1), Some(&vec![2]));
623/// assert!(cycles.get(&3).is_none(), "waiting on a cycle is not being in one");
624/// ```
625pub fn dependency_cycles(
626    obs: &ForgeObservation,
627    heads: &BTreeMap<String, u64>,
628) -> BTreeMap<u64, Vec<u64>> {
629    let open: BTreeSet<u64> = obs.pull_requests.iter().map(|p| p.number).collect();
630    let edges: BTreeMap<u64, Vec<u64>> = obs
631        .pull_requests
632        .iter()
633        .map(|p| {
634            let (declared, stacked) = classify::confirmed_dependencies(p, &obs.base, heads);
635            let to: Vec<u64> = declared
636                .into_iter()
637                .chain(stacked)
638                .filter(|n| open.contains(n))
639                .collect();
640            (p.number, to)
641        })
642        .collect();
643    // everything each pull request reaches through its edges
644    let reach = |from: u64| -> BTreeSet<u64> {
645        let mut seen = BTreeSet::new();
646        let mut stack = vec![from];
647        while let Some(n) = stack.pop() {
648            for m in edges.get(&n).into_iter().flatten() {
649                if seen.insert(*m) {
650                    stack.push(*m);
651                }
652            }
653        }
654        seen
655    };
656    let reaches: BTreeMap<u64, BTreeSet<u64>> = open.iter().map(|n| (*n, reach(*n))).collect();
657    open.iter()
658        .filter(|n| reaches[n].contains(n))
659        .map(|n| {
660            let others: Vec<u64> = reaches[n]
661                .iter()
662                .filter(|m| *m != n && reaches[*m].contains(n))
663                .copied()
664                .collect();
665            (*n, others)
666        })
667        .collect()
668}

670/// Add the dependencies git implies: a pull request whose observed head contains another's
671/// observed head carries that one's commits, so landing it lands both. Inferred, and evidence
672/// only — a block is only ever declared ([`DependencyCertainty::Confirmed`]), never guessed,
673/// and nothing here changes a disposition or a rank. `containing` answers, for a head, the
674/// pull requests whose mirrored head contains it and the commit each mirror names; a mirror
675/// that is not the head the queue decided on implies nothing.
676pub fn infer_dependencies(
677    queue: &mut IntegrationQueue,
678    mut containing: impl FnMut(&str) -> Vec<(u64, String)>,
679) {
680    let heads: BTreeMap<u64, String> = queue
681        .assessments
682        .iter()
683        .map(|a| (a.number, a.evaluated_against.head_sha.clone()))
684        .collect();
685    let mut implied: BTreeMap<u64, BTreeSet<u64>> = BTreeMap::new();
686    for (n, head) in &heads {
687        for (m, tip) in containing(head) {
688            if m != *n && heads.get(&m) == Some(&tip) {
689                implied.entry(m).or_default().insert(*n);
690            }
691        }
692    }
693    for a in &mut queue.assessments {
694        for n in implied.get(&a.number).into_iter().flatten() {
695            if a.dependencies.iter().any(|d| d.number == *n) {
696                continue;
697            }
698            a.dependencies.push(PullRequestDependency {
699                number: *n,
700                certainty: DependencyCertainty::Inferred,
701                satisfied: false,
702                state: DependencyState::Open,
703            });
704            a.evidence.push(IntegrationEvidence {
705                kind: EvidenceKind::Dependency,
706                status: "inferred".into(),
707                detail: format!("#{n}'s head is in this head: landing it lands #{n} too"),
708                source: Some(EvidenceSource::Git {
709                    master_sha: a.evaluated_against.master_sha.clone(),
710                    head_sha: a.evaluated_against.head_sha.clone(),
711                }),
712            });
713        }
714    }
715}

717/// What the queue's ranked assessments imply: the next merge, the refresh candidates and the
718/// counts. Derived again whenever a disposition changes after the queue was built.
719fn derive_heads(queue: &mut IntegrationQueue) {
720    let mut tallies = QueueTallies {
721        open: queue.assessments.len(),
722        ..Default::default()
723    };
724    for a in &queue.assessments {
725        *tallies
726            .by_disposition
727            .entry(a.disposition.as_str().to_string())
728            .or_default() += 1;
729        *tallies.by_lane.entry(classify::word(&a.lane)).or_default() += 1;
730    }
731    queue.tallies = tallies;
732    queue.next_merge = queue
733        .assessments
734        .iter()
735        .find(|a| a.disposition == PullRequestDisposition::Ready)
736        .map(|a| a.number);
737    queue.next_refresh = queue
738        .assessments
739        .iter()
740        .filter(|a| a.disposition == PullRequestDisposition::NeedsRefresh)
741        .map(|a| a.number)
742        .collect();
743}

745/// The executor's own refusals, folded from the trail: a candidate the forge refused to
746/// merge, or whose refresh failed, at the head and master it still has, is held back as
747/// `needs_repair` instead of being chosen again, so one refused change does not block the
748/// queue. A new head or a new master clears it, because the key no longer matches; a
749/// transient failure never holds anything back, because it says nothing about the change.
750/// No plan survives: the trail is the only memory, and every queue is folded from it again.
751pub fn executor_feedback(queue: &mut IntegrationQueue, trail: &[drain::IntegrationEvent]) {
752    use drain::{FailureClass, IntegrationAction};
753    let refused = |action: IntegrationAction| -> BTreeSet<(u64, String, String)> {
754        trail
755            .iter()
756            .filter(|e| e.action == action && e.class != Some(FailureClass::Transient))
757            .filter_map(|e| Some((e.pr?, e.head_sha.clone()?, e.master_before.clone()?)))
758            .collect()
759    };
760    let merges = refused(IntegrationAction::MergeFailed);
761    let refreshes = refused(IntegrationAction::RefreshFailed);
762    let master = queue.master_sha.clone();
763    let mut changed = false;
764    for a in &mut queue.assessments {
765        let key = (
766            a.number,
767            a.evaluated_against.head_sha.clone(),
768            master.clone(),
769        );
770        let held = match a.disposition {
771            PullRequestDisposition::Ready if merges.contains(&key) => {
772                Some(ReasonCode::ExecutorMergeRefused {
773                    head: key.1.clone(),
774                })
775            }
776            PullRequestDisposition::NeedsRefresh if refreshes.contains(&key) => {
777                Some(ReasonCode::ExecutorRefreshFailed {
778                    master: master.clone(),
779                })
780            }
781            _ => None,
782        };
783        if let Some(reason) = held {
784            a.disposition = PullRequestDisposition::NeedsRepair;
785            a.lane = a.disposition.lane();
786            a.reasons.insert(0, reason);
787            changed = true;
788        }
789    }
790    if changed {
791        queue.assessments = rank(std::mem::take(&mut queue.assessments));
792        derive_heads(queue);
793    }
794}

834/// What the bodies of an observation declare about replacement, sorted by who said it.
835struct DeclaredSuccessors {
836    /// Replaced → its authorised successors, each with what became of it.
837    authorised: BTreeMap<u64, Vec<classify::Successor>>,
838    /// Replaced → the declarations nobody entitled made about it.
839    possible: BTreeMap<u64, Vec<classify::PossibleSupersession>>,
840}
841
842/// One declaration, by the pull request whose body carries it.
843struct Declarer<'a> {
844    number: u64,
845    login: &'a str,
846    association: &'a str,
847    cross_repository: bool,
848    /// An open one of this repository of which nothing was read: neither who its author is
849    /// to the repository nor the pull requests that mention it.
850    unread: bool,
851}
852
853impl Declarer<'_> {
854    fn authorised(&self) -> bool {
855        classify::declaration_is_authorised(self.association, self.cross_repository)
856    }
857
858    /// How much its word weighs when two bodies declare one pair: an authorised one over one
859    /// that could not be read, and that over one that is not authorised.
860    fn weight(&self) -> u8 {
861        if self.authorised() {
862            2
863        } else {
864            u8::from(self.unread)
865        }
866    }
867}

869/// The declared successors of every open pull request. A declaration counts only from someone
870/// the repository lets declare one ([`classify::declaration_is_authorised()`]), tested on the
871/// author of the body that speaks: the replaced one's own (`Superseded by`), an open
872/// successor's, or a resolved one's (`Supersedes`). An authorised successor comes with what
873/// became of it: open; or no longer open and landed when `relation` finds its head, or else
874/// its merge commit, contained in master — whatever the forge calls it; or not landed; or
875/// unread, when the forge did not report it or git could not answer. Anyone else's declaration
876/// is listed as possible and nothing is asked of git about it. When two bodies declare one
877/// pair, an authorised one stands over an unauthorised one, and of two alike the replaced
878/// one's own body stands.
879///
880/// Git's word is asked only of a successor that can have landed. One the forge says changes
881/// no file brought nothing to master wherever its head points, and one closed unmerged with
882/// its head in a fork has a head its owner could point anywhere before closing: neither is a
883/// landing, whatever master contains. And an open pull request of which nothing could be read
884/// may be one whose word counts: what it says it supersedes is held until a refresh reads it,
885/// never released on a read that failed.
886fn successors<'a>(
887    obs: &'a ForgeObservation,
888    open: &BTreeSet<u64>,
889    relation: &mut impl FnMut(&PullRequestObservation) -> RelationToMaster,
890) -> DeclaredSuccessors {
891    use classify::{declared_supersessions, PossibleSupersession, Successor, SuccessorState};
892    // (replaced, successor) → whose body said so
893    let mut declared: BTreeMap<(u64, u64), Declarer<'a>> = BTreeMap::new();
894    let mut declare = |replaced: u64, successor: u64, by: Declarer<'a>| {
895        let pair = (replaced, successor);
896        let stands = declared
897            .get(&pair)
898            .is_some_and(|kept| kept.weight() >= by.weight());
899        if replaced != successor && open.contains(&replaced) && !stands {
900            declared.insert(pair, by);
901        }
902    };
903    let of_open = |p: &'a PullRequestObservation| Declarer {
904        number: p.number,
905        login: &p.author,
906        association: &p.author_association,
907        cross_repository: p.cross_repository,
908        unread: !p.cross_repository
909            && p.author_association.is_empty()
910            && p.cross_references == CrossReferenceRead::Unread,
911    };
912    // the replaced one's own body first, so that of two authorised bodies it is the one named
913    for p in &obs.pull_requests {
914        for m in declared_supersessions(&p.body).superseded_by {
915            declare(p.number, m, of_open(p));
916        }
917    }
918    for p in &obs.pull_requests {
919        for n in declared_supersessions(&p.body).supersedes {
920            declare(n, p.number, of_open(p));
921        }
922    }
923    for (m, r) in &obs.resolved {
924        for n in declared_supersessions(&r.body).supersedes {
925            let by = Declarer {
926                number: *m,
927                login: &r.author,
928                association: &r.author_association,
929                cross_repository: r.cross_repository,
930                unread: false,
931            };
932            declare(n, *m, by);
933        }
934    }
935    let mut state_of = |m: u64| -> SuccessorState {
936        if open.contains(&m) {
937            return SuccessorState::Open;
938        }
939        let Some(r) = obs.resolved.get(&m) else {
940            return SuccessorState::Unread;
941        };
942        // the two vetoes below read "a fork's" and "no file" as facts, and each releases what
943        // a successor closed unmerged held. A reading that left either out said neither.
944        if !r.whole {
945            return SuccessorState::Unread;
946        }
947        // what cannot be a landing is decided before git is asked where its head is
948        if r.changed_files == 0 {
949            return SuccessorState::Empty { merged: r.merged };
950        }
951        if r.cross_repository && !r.merged {
952            return SuccessorState::ForkClosed;
953        }
954        // the pull request the forge described, at one commit of it
955        let at = |sha: &str| PullRequestObservation {
956            number: m,
957            title: String::new(),
958            author: r.author.clone(),
959            head_ref: String::new(),
960            head_sha: sha.to_string(),
961            base_ref: obs.base.clone(),
962            draft: false,
963            labels: Vec::new(),
964            created_at: String::new(),
965            updated_at: String::new(),
966            body: r.body.clone(),
967            checks: Vec::new(),
968            review_decision: String::new(),
969            auto_merge: false,
970            cross_repository: r.cross_repository,
971            latest_reviews: Vec::new(),
972            review_requests: Vec::new(),
973            author_association: r.author_association.clone(),
974            cross_references: CrossReferenceRead::Whole,
975        };
976        // landed is git's fact: the head, or the merge commit, is in master. The forge's
977        // word that it merged is in no condition here.
978        let head = relation(&at(&r.head_sha));
979        let contained = |r: &RelationToMaster| matches!(r, RelationToMaster::Contained);
980        if contained(&head) {
981            SuccessorState::Landed {
982                head_sha: r.head_sha.clone(),
983                merged: r.merged,
984                by_merge_commit: false,
985            }
986        } else if !r.merge_commit.is_empty() && contained(&relation(&at(&r.merge_commit))) {
987            SuccessorState::Landed {
988                head_sha: r.merge_commit.clone(),
989                merged: r.merged,
990                by_merge_commit: true,
991            }
992        } else if matches!(head, RelationToMaster::Unknown { .. }) {
993            SuccessorState::Unread
994        } else {
995            SuccessorState::NotLanded { merged: r.merged }
996        }
997    };
998    let mut states: BTreeMap<u64, SuccessorState> = BTreeMap::new();
999    let mut found = DeclaredSuccessors {
1000        authorised: BTreeMap::new(),
1001        possible: BTreeMap::new(),
1002    };
1003    for ((replaced, successor), by) in declared {
1004        // its own body's word that it replaces another, when nothing was read of it
1005        let unread = by.unread && by.number == successor;
1006        if by.authorised() || unread {
1007            let state = if unread {
1008                SuccessorState::DeclarerUnread
1009            } else {
1010                states
1011                    .entry(successor)
1012                    .or_insert_with(|| state_of(successor))
1013                    .clone()
1014            };
1015            found
1016                .authorised
1017                .entry(replaced)
1018                .or_default()
1019                .push(Successor {
1020                    number: successor,
1021                    declared_in: by.number,
1022                    state,
1023                });
1024        } else {
1025            found
1026                .possible
1027                .entry(replaced)
1028                .or_default()
1029                .push(PossibleSupersession {
1030                    successor,
1031                    declared_in: by.number,
1032                    declared_by: by.login.to_string(),
1033                    association: by.association.to_string(),
1034                    cross_repository: by.cross_repository,
1035                });
1036        }
1037    }
1038    found
1039}

1112fn computed(
1113    root: &Path,
1114    now: std::time::SystemTime,
1115) -> Result<(IntegrationQueue, RelationCache), String> {
1116    let obs = load_observation(root)?.ok_or_else(|| {
1117        "no forge observation is recorded in this checkout; majordomus prs refresh".to_string()
1118    })?;
1119    let master = local_master(root, &obs.base).ok_or_else(|| {
1120        format!(
1121            "this clone has no refs/remotes/origin/{}; majordomus prs refresh fetches it",
1122            obs.base
1123        )
1124    })?;
1125    let cache_path = state_path(root, RELATIONS_FILE);
1126    let mut cache: RelationCache = std::fs::read_to_string(&cache_path)
1127        .ok()
1128        .and_then(|t| serde_json::from_str(&t).ok())
1129        .unwrap_or_default();
1130    // bounded: only the current master's entries are worth keeping, and only a pair of
1131    // commit ids is a fact — a key written before refs were refused names a ref, which may
1132    // hold another commit tomorrow, and is dropped even while master stands still
1133    let current = |k: &String| {
1134        k.split_once("..")
1135            .is_some_and(|(m, h)| m == master && is_object_id(m) && is_object_id(h))
1136    };
1137    cache.entries.retain(|k, _| current(k));
1138    cache.shapes.retain(|k, _| current(k));
1139    // the relation and the shape share the cache, and each is asked from its own closure
1140    let cell = std::cell::RefCell::new(cache);
1141    let mut queue = build_queue_shaped(
1142        &obs,
1143        &master,
1144        |p| {
1145            // decided on exactly the head the forge reported, which the assessment names as
1146            // evaluated: a head that moved during the refresh is not in this clone, and what
1147            // the fetched ref holds now is another head nobody observed
1148            if !relation::has_commit(root, &p.head_sha) {
1149                return RelationToMaster::Unknown {
1150                reason: format!(
1151                    "the observed head {} is not fetched; the pull request moved during the refresh — majordomus prs refresh",
1152                    p.head_sha
1153                ),
1154            };
1155            }
1156            relation_cached(root, &mut cell.borrow_mut(), &master, &p.head_sha)
1157        },
1158        |p| shape_cached(root, &mut cell.borrow_mut(), &master, &p.head_sha),
1159    );
1160    let cache = cell.into_inner();
1161    infer_dependencies(&mut queue, |head| relation::containing(root, head));
1162    link_issues(&mut queue, &issue_milestones(root));
1163    let now_secs = now
1164        .duration_since(std::time::UNIX_EPOCH)
1165        .map_or(0, |d| i64::try_from(d.as_secs()).unwrap_or(i64::MAX));
1166    if let Some(age) = crate::peers::epoch_seconds(&obs.observed_at)
1167        .map(|then| now_secs.saturating_sub(then))
1168        .filter(|age| *age > i64::try_from(OBSERVATION_STALE_AFTER.as_secs()).unwrap_or(i64::MAX))
1169    {
1170        queue.diagnostics.push(format!(
1171            "the observation is {} min old (observed {}): the forge has moved on since; majordomus prs refresh",
1172            age / 60,
1173            obs.observed_at
1174        ));
1175    }
1176    let trail = drain::events(root);
1177    executor_feedback(&mut queue, &trail);
1178    wait::annotate(&mut queue, &trail);
1179    Ok((queue, cache))
1180}

1232/// Observe the forge, fetch what it names, and record the observation — in the checkout's
1233/// state, and as an `observed` line of the repository's trail. The network step; the only
1234/// one outside the executor.
1235pub fn refresh(root: &Path) -> Result<ForgeObservation, String> {
1236    use forge::Forge;
1237    let obs = forge::GhForge { root }.observe().map_err(|e| e.0)?;
1238    forge::fetch(root, &obs).map_err(|e| e.0)?;
1239    store_observation(root, &obs)?;
1240    drain::record(
1241        root,
1242        drain::IntegrationEvent {
1243            master_before: Some(obs.base_sha.clone()),
1244            detail: format!(
1245                "{} open pull request(s) of {} at {}",
1246                obs.pull_requests.len(),
1247                obs.repository,
1248                obs.observed_at
1249            ),
1250            ..drain::IntegrationEvent::of(drain::IntegrationAction::Observed)
1251        },
1252    )?;
1253    Ok(obs)
1254}
