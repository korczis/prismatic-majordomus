1//! The boundary between the layer's two halves, checked (I2871, I1102).
2//!
3//! I2870 lets a directory's contract declare its layout — the lifecycle of what lives there
4//! and what must never appear — and the manifest's `local:` block declares the local half's.
5//! This module reads those declarations and refuses what contradicts them, naming the path
6//! and the contract. It judges only what is declared: a contract without a layout says
7//! nothing, and nothing is refused under it.
8//!
9//! What a file is, is read rather than guessed from its name:
10//!
11//! - a tracked file under the tracked half that git's own ignore rules match is state that
12//!   does not travel — `cache` or `runtime` — committed by mistake;
13//! - a file under the local half whose bytes are a tracked file of the tracked half, or
14//!   whose front matter declares the schema of a kind `share/kinds.yaml` calls canonical,
15//!   is a copy of authored knowledge.
16//!
17//! ```
18//! use majordomus_cli::layout::{LayoutFinding, LayoutReport};
19//! let report = LayoutReport::of(true, None, 3, Vec::new());
20//! assert!(report.passes, "measured and nothing found");
21//! let found = LayoutReport::of(true, None, 3, vec![LayoutFinding {
22//!     path: ".ai/local/rules/copied.md".into(),
23//!     contract: "manifest local".into(),
24//!     excludes: "canonical".into(),
25//!     reason: "a copy of .ai/repo/rules/copied.md".into(),
26//! }]);
27//! assert!(!found.passes);
28//! assert!(!LayoutReport::of(false, Some("no git".into()), 0, Vec::new()).passes);
29//! ```

42/// The largest local file whose head is read for a declared schema; larger files are
43/// compared by their bytes only.
44const HEAD_BYTES: u64 = 256 * 1024;

46/// One misplaced artifact: a path found in a half whose declared layout excludes it, with the
47/// contract that excludes it and why, so a refusal names what to move and where it belongs.
48///
49/// It travels as the JSON of the `layout.check` capability, so its member names are the
50/// wire contract every projection reads:
51///
52/// ```
53/// use majordomus_cli::layout::LayoutFinding;
54/// let finding = LayoutFinding {
55///     path: ".ai/local/notes/rule.md".into(),
56///     contract: "manifest local".into(),
57///     excludes: "canonical".into(),
58///     reason: "a copy of .ai/repo/rules/rule.md".into(),
59/// };
60/// let json = serde_json::to_value(&finding).unwrap();
61/// assert_eq!(json["contract"], "manifest local");
62/// assert_eq!(json["excludes"], "canonical");
63/// let back: LayoutFinding = serde_json::from_value(json).unwrap();
64/// assert_eq!(back, finding);
65/// ```
66#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
67pub struct LayoutFinding {
68    /// The path, repository-relative.
69    pub path: String,
70    /// The contract it violates: a context document's id, or `manifest local`.
71    pub contract: String,
72    /// What the contract excludes that this path is.
73    pub excludes: String,
74    /// What the path was found to be, and how.
75    pub reason: String,
76}
77
78/// Findings are listed by path, the contract breaking the tie, as everywhere else.
79impl crate::order::Ordered for LayoutFinding {
80    fn order_key(&self) -> crate::order::OrderKey<'_> {
81        crate::order::OrderKey::plain(&self.path, &self.contract)
82    }
83}

85/// The answer of the layout check: whether the halves were read, how many paths were
86/// judged, and every path a contract excludes. A report that could not read the halves
87/// never passes, so silence about a tree nobody measured cannot read as a clean tree.
88///
89/// ```
90/// use majordomus_cli::layout::LayoutReport;
91/// let clean = LayoutReport::of(true, None, 12, Vec::new());
92/// assert!(clean.passes);
93/// assert_eq!(clean.checked, 12);
94/// let unread = LayoutReport::of(false, Some("git could not start".into()), 0, Vec::new());
95/// assert!(!unread.passes, "nothing found because nothing was read");
96/// assert_eq!(unread.reason.as_deref(), Some("git could not start"));
97/// ```
98#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
99pub struct LayoutReport {
100    /// Whether the halves could be read at all. `false` is not a pass.
101    pub measured: bool,
102    /// Why they could not, when they could not.
103    pub reason: Option<String>,
104    /// How many paths were judged.
105    pub checked: usize,
106    /// Every path a contract excludes, in canonical order. Empty is the passing answer.
107    pub findings: Vec<LayoutFinding>,
108    /// `true` when the halves were read and nothing was found.
109    pub passes: bool,
110}
111
112impl LayoutReport {
113    /// A report, its verdict derived from what was measured and found. The findings are
114    /// put in canonical order — by path, the contract breaking the tie — whatever order the
115    /// walk found them in, so two runs over one tree answer the same bytes.
116    ///
117    /// ```
118    /// use majordomus_cli::layout::{LayoutFinding, LayoutReport};
119    /// let at = |path: &str| LayoutFinding {
120    ///     path: path.into(),
121    ///     contract: "manifest local".into(),
122    ///     excludes: "canonical".into(),
123    ///     reason: "a copy".into(),
124    /// };
125    /// let found = vec![at(".ai/local/z.md"), at(".ai/local/a.md")];
126    /// let report = LayoutReport::of(true, None, 5, found);
127    /// assert!(!report.passes, "measured, but something was found");
128    /// let paths: Vec<&str> = report.findings.iter().map(|f| f.path.as_str()).collect();
129    /// assert_eq!(paths, [".ai/local/a.md", ".ai/local/z.md"]);
130    /// ```
131    pub fn of(
132        measured: bool,
133        reason: Option<String>,
134        checked: usize,
135        mut findings: Vec<LayoutFinding>,
136    ) -> Self {
137        crate::order::canonical(&mut findings);
138        let passes = measured && findings.is_empty();
139        LayoutReport {
140            measured,
141            reason,
142            checked,
143            findings,
144            passes,
145        }
146    }
147
148    fn unmeasured(reason: String) -> Self {
149        Self::of(false, Some(reason), 0, Vec::new())
150    }
151}

159/// Check both halves of `repository` against their declared layouts.
160///
161/// Reads git at the repository's root — the tracked half's index entries and the ones its
162/// ignore rules match — and the local half's Markdown and YAML files. A root git cannot
163/// read is answered unmeasured rather than passing; a local half the manifest does not
164/// declare `excludes: [canonical]` for is counted but not judged.
165///
166/// ```
167/// use majordomus_cli::layout;
168/// use majordomus_cli::metadata::KindSchema;
169/// use majordomus_cli::repository::Repository;
170/// use majordomus_cli::share::Share;
171/// use majordomus_cli::synthetic::{crate_share, SyntheticRepository};
172///
173/// let repo = SyntheticRepository::small().unwrap();
174/// let root = repo.root();
175/// // the local half declares that authored knowledge never lives there
176/// let manifest = root.join(".ai/manifest.yaml");
177/// let text = std::fs::read_to_string(&manifest).unwrap().replace(
178///     "  implicit_context: false\n",
179///     "  implicit_context: false\n  lifecycle: runtime\n  excludes: [canonical]\n",
180/// );
181/// std::fs::write(&manifest, text).unwrap();
182/// let git = |args: &[&str]| {
183///     let mut git = std::process::Command::new("git");
184///     assert!(git.arg("-C").arg(root).args(args).status().unwrap().success());
185/// };
186/// git(&["init", "-q"]);
187/// git(&["add", ".ai"]);
188/// // a byte copy of a tracked rule, left in the local half
189/// let rule = std::fs::read(root.join(".ai/repo/rules/project/rule-0.v1.md")).unwrap();
190/// std::fs::create_dir_all(root.join(".ai/local/notes")).unwrap();
191/// std::fs::write(root.join(".ai/local/notes/rule-0.md"), rule).unwrap();
192///
193/// let repository = Repository::discover(root).unwrap();
194/// let share = Share::locate(Some(&crate_share()), root).unwrap();
195/// let kinds = KindSchema::load(&share, &repository).unwrap();
196/// let report = layout::check(&repository, &repo.index().unwrap(), &kinds);
197/// assert!(report.measured && !report.passes);
198/// assert_eq!(report.findings.len(), 1);
199/// assert_eq!(report.findings[0].path, ".ai/local/notes/rule-0.md");
200/// assert!(report.findings[0].reason.contains(".ai/repo/rules/project/rule-0.v1.md"));
201/// ```
202pub fn check(repository: &Repository, index: &Index, kinds: &KindSchema) -> LayoutReport {
203    let root = repository.root();
204    let manifest = repository.manifest();
205    let tracked_half = format!(".ai/{}", manifest.repo.path.trim_end_matches('/'));
206    let local_half = format!(".ai/{}", manifest.local.path.trim_end_matches('/'));
207
208    // the tracked half with its blob ids, then which of it git's ignore rules match: both
209    // listings or neither, so a half-read tree is never judged
210    let listed = git_lines(root, &["ls-files", "-s", "--", &tracked_half]).and_then(|tracked| {
211        git_lines(
212            root,
213            &[
214                "ls-files",
215                "-c",
216                "-i",
217                "--exclude-standard",
218                "--",
219                &tracked_half,
220            ],
221        )
222        .map(|ignored| (tracked, ignored))
223    });
224    let (tracked, ignored) = match listed {
225        Ok(lists) => lists,
226        Err(e) => return LayoutReport::unmeasured(e),
227    };
228
229    let mut findings = Vec::new();
230    let contracts = declared_layouts(index);
231    for path in &ignored {
232        let Some(d) = nearest(&contracts, path) else {
233            continue;
234        };
235        let Some(excluded) = d
236            .excludes
237            .iter()
238            .find(|e| matches!(e.as_str(), "cache" | "runtime"))
239        else {
240            continue;
241        };
242        findings.push(LayoutFinding {
243            path: path.clone(),
244            contract: d.contract.clone(),
245            excludes: excluded.clone(),
246            reason: "tracked although git's ignore rules match it: state that does not travel"
247                .into(),
248        });
249    }
250
251    let mut checked = tracked.len();
252    if manifest.local.excludes.iter().any(|e| e == "canonical") {
253        // the tracked half's blobs, by id: a local file with one of these ids is that file
254        let blobs: BTreeMap<String, String> =
255            tracked.iter().filter_map(|l| staged_blob(l)).collect();
256        let canonical: BTreeSet<String> = kinds
257            .kinds()
258            .filter(|(_, k)| k.lifecycle.unwrap_or(Lifecycle::Canonical) == Lifecycle::Canonical)
259            .filter_map(|(_, k)| k.schema.as_deref())
260            .flat_map(|s| {
261                let short = s.split_once('.').map_or(s, |(_, rest)| rest);
262                [s.to_string(), short.to_string()]
263            })
264            .collect();
265        let files = local_files(&root.join(&local_half));
266        checked += files.len();
267        let ids = match hash_objects(root, &files) {
268            Ok(ids) => ids,
269            Err(e) => return LayoutReport::unmeasured(e),
270        };
271        for (file, id) in files.iter().zip(ids) {
272            let rel = file
273                .strip_prefix(root)
274                .unwrap_or(file)
275                .to_string_lossy()
276                .replace('\\', "/");
277            if let Some(original) = blobs.get(&id) {
278                findings.push(LayoutFinding {
279                    path: rel,
280                    contract: "manifest local".into(),
281                    excludes: "canonical".into(),
282                    reason: format!(
283                        "a copy of {original}: authored knowledge belongs in the tracked half"
284                    ),
285                });
286                continue;
287            }
288            if let Some(schema) = declared_schema(file).filter(|s| canonical.contains(s)) {
289                findings.push(LayoutFinding {
290                    path: rel,
291                    contract: "manifest local".into(),
292                    excludes: "canonical".into(),
293                    reason: format!("declares `schema: {schema}`, a canonical kind: authored knowledge belongs in the tracked half"),
294                });
295            }
296        }
297    }
298    LayoutReport::of(true, None, checked, findings)
299}

301/// Every contract that declares a layout, by the directory it governs.
302fn declared_layouts(index: &Index) -> BTreeMap<String, Declared> {
303    index
304        .objects
305        .iter()
306        .filter(|o| o.kind == "context")
307        .filter_map(|o| {
308            let layout = o.metadata.get("layout")?;
309            let dir = Path::new(&o.provenance.path)
310                .parent()
311                .map(|p| p.to_string_lossy().replace('\\', "/"))
312                .unwrap_or_default();
313            let excludes = layout
314                .get("excludes")
315                .and_then(|e| e.as_array())
316                .map(|a| {
317                    a.iter()
318                        .filter_map(|v| v.as_str().map(str::to_string))
319                        .collect()
320                })
321                .unwrap_or_default();
322            Some((
323                dir,
324                Declared {
325                    contract: o.identity.clone(),
326                    excludes,
327                },
328            ))
329        })
330        .collect()
331}

333/// The contract nearest above `path` that declares a layout.
334fn nearest<'a>(contracts: &'a BTreeMap<String, Declared>, path: &str) -> Option<&'a Declared> {
335    Path::new(path)
336        .ancestors()
337        .skip(1)
338        .find_map(|dir| contracts.get(&dir.to_string_lossy().replace('\\', "/")))
339}

341/// The `schema:` a file declares in its first lines (front matter or a YAML document).
342fn declared_schema(file: &Path) -> Option<String> {
343    let meta = std::fs::metadata(file).ok()?;
344    if meta.len() > HEAD_BYTES {
345        return None;
346    }
347    let text = std::fs::read_to_string(file).ok()?;
348    text.lines().take(40).find_map(|l| {
349        let v = l.strip_prefix("schema:")?.trim();
350        Some(v.trim_matches(|c| c == '"' || c == '\'').to_string())
351    })
352}

354/// Every regular file under `dir` that could carry knowledge (Markdown or YAML). Links are
355/// not followed; the order is the walk's, and the report orders what it finds.
356fn local_files(dir: &Path) -> Vec<PathBuf> {
357    let mut out = Vec::new();
358    let mut stack = vec![dir.to_path_buf()];
359    while let Some(d) = stack.pop() {
360        let Ok(entries) = std::fs::read_dir(&d) else {
361            continue;
362        };
363        for e in entries.flatten() {
364            let p = e.path();
365            match e.file_type() {
366                Ok(t) if t.is_dir() => stack.push(p),
367                Ok(t)
368                    if t.is_file()
369                        && p.extension()
370                            .is_some_and(|x| x == "md" || x == "yaml" || x == "yml")
371                        && e.metadata().is_ok_and(|m| m.len() > 0) =>
372                {
373                    out.push(p)
374                }
375                // another kind of file, or one whose type cannot be read: not a record
376                _ => {}
377            }
378        }
379    }
380    out
381}

383/// How many paths one `git hash-object` is handed, so that the argument list stays far
384/// below any platform's limit however many local files there are.
385const HASH_CHUNK: usize = 256;
386
387/// git's blob id of each file, in order. The paths are arguments, never lines of input: git
388/// prints one id per path it is given or fails, so the ids pair with the files one to one,
389/// whatever characters a file name holds.
390fn hash_objects(root: &Path, files: &[PathBuf]) -> Result<Vec<String>, String> {
391    let mut ids = Vec::with_capacity(files.len());
392    for chunk in files.chunks(HASH_CHUNK) {
393        let paths: Vec<String> = chunk.iter().map(|f| f.display().to_string()).collect();
394        let mut args = vec!["hash-object", "--no-filters", "--"];
395        args.extend(paths.iter().map(String::as_str));
396        ids.extend(git_lines(root, &args)?);
397    }
398    Ok(ids)
399}

401/// The tracked path and its blob id, from one line of `git ls-files -s`
402/// (`<mode> <blob> <stage>\t<path>`); none from a line of another shape.
403fn staged_blob(line: &str) -> Option<(String, String)> {
404    let (meta, path) = line.split_once('\t')?;
405    let blob = meta.split_whitespace().nth(1)?;
406    Some((blob.to_string(), path.to_string()))
407}

409/// The lines git prints for `args` at `root`.
410fn git_lines(root: &Path, args: &[&str]) -> Result<Vec<String>, String> {
411    program_lines("git", root, args)
412}
413
414/// The lines `program`, a git, prints for `args` at `root`: the seam a test names a program
415/// that cannot start through.
416fn program_lines(program: &str, root: &Path, args: &[&str]) -> Result<Vec<String>, String> {
417    let out = Command::new(program)
418        .arg("-C")
419        .arg(root)
420        .args(args)
421        .output()
422        .map_err(|e| format!("git could not start: {e}"))?;
423    if !out.status.success() {
424        return Err(format!(
425            "git {} failed: {}",
426            args.join(" "),
427            String::from_utf8_lossy(&out.stderr).trim()
428        ));
429    }
430    Ok(String::from_utf8_lossy(&out.stdout)
431        .lines()
432        .filter(|l| !l.is_empty())
433        .map(str::to_string)
434        .collect())
435}

599    /// A context document under the tracked half that declares a layout.
600    fn contract(repo: &SyntheticRepository, dir: &str, id: &str, excludes: &str) {
