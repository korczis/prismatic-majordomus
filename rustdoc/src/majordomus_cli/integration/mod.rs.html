1//! Pull-request integration: the next provably safe change, onto the current master, one
2//! at a time (ADR 0101).
3//!
4//! # The pipeline
5//!
6//! ```text
7//! forge (gh) ──► ForgeObservation ──► relation to master (git merge-tree, cached by SHA pair)
8//!                (recorded locally)            │
9//!                                              ▼
10//!                               classify ──► PullRequestAssessment (one per PR, with evidence)
11//!                                              │
12//!                                              ▼
13//!                                 plan ──► IntegrationQueue (ranked, deterministic)
14//!                                              │
15//!           CLI · HTTP · MCP · Cockpit ◄───────┤
16//!                                              ▼
17//!                           drain: lease ► re-observe ► re-decide ► act ► verify ► re-plan
18//! ```text
19//!
20//! The observation is the only network product and it is recorded with its moment; every
21//! read surface renders it offline. A decision names the master and head it was taken
22//! against ([`EvaluatedAgainst`]), and the executor acts only when a fresh observation
23//! still says the same — a merge invalidates every earlier plan by construction, because
24//! the next step starts from a new observation of a new master.
25//!
26//! # Why this module is not public
27//!
28//! It is `pub(crate)`. Integration is an operation of this executable, reached through its
29//! command line (`majordomus prs`) and its capability registry (`integration.*` on HTTP, MCP
30//! and OpenAPI), and not a library other crates link against, so it is not part of the
31//! crate's exported API and the exported-surface gate (`quality report`) does not cover it.
32//! What covers it instead: the unit tests beside it (`tests.rs`: every disposition, the
33//! stale-decision refusal, the re-plan after every merge, the cleanup threshold, relations
34//! computed by real git, and two properties), and
35//! `test/cases/720_integration_follows_the_current_master.sh`, which drives the real
36//! command line against a scripted forge.
37//!
38//! # Why the forge's `mergeable` is not read
39//!
40//! This repository resolves derived files with a per-clone merge driver the forge cannot
41//! run, so the forge calls nearly every pull request conflicting. The relation to master is
42//! decided here, by git, with the drivers ([`relation`]).

79    root.join(STATE_DIR).join(file)

82/// The last recorded observation, if any.
83pub fn load_observation(root: &Path) -> Result<Option<ForgeObservation>, String> {
84    let path = state_path(root, OBSERVATION_FILE);
85    let text = match std::fs::read_to_string(&path) {
86        Ok(t) => t,
87        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(None),
88        Err(e) => return Err(format!("{}: {e}", path.display())),
89    };
90    let obs: ForgeObservation =
91        serde_json::from_str(&text).map_err(|e| format!("{}: {e}", path.display()))?;
92    if obs.schema != OBSERVATION_SCHEMA {
93        return Err(format!(
94            "{}: schema {}, this reads {OBSERVATION_SCHEMA}",
95            path.display(),
96            obs.schema
97        ));
98    }
99    Ok(Some(obs))
100}

102fn write_atomic(path: &Path, text: &str) -> Result<(), String> {
103    if let Some(dir) = path.parent() {
104        std::fs::create_dir_all(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
105    }
106    let tmp = path.with_extension(format!("tmp.{}", std::process::id()));
107    std::fs::write(&tmp, text).map_err(|e| format!("{}: {e}", tmp.display()))?;
108    std::fs::rename(&tmp, path).map_err(|e| format!("{}: {e}", path.display()))
109}

111/// Record an observation.
112pub fn store_observation(root: &Path, obs: &ForgeObservation) -> Result<(), String> {
113    let text = serde_json::to_string_pretty(obs).map_err(|e| e.to_string())?;
114    write_atomic(&state_path(root, OBSERVATION_FILE), &(text + "\n"))
115}

117/// The relation cache: `master..head` → relation. Both SHAs are immutable, so an entry is
118/// true forever; the file is bounded by dropping entries whose master is not the current.
119#[derive(Debug, Default, Serialize, Deserialize)]
120struct RelationCache {
121    entries: BTreeMap<String, RelationToMaster>,
122}
123
124fn relation_cached(
125    root: &Path,
126    cache: &mut RelationCache,
127    master: &str,
128    head: &str,
129) -> RelationToMaster {
130    let key = format!("{master}..{head}");
131    if let Some(r) = cache.entries.get(&key) {
132        return r.clone();
133    }
134    let r = relation::relation_to_master(root, master, head);
135    // an unknown is not a fact about the SHAs; it is not cached
136    if !matches!(r, RelationToMaster::Unknown { .. }) {
137        cache.entries.insert(key, r.clone());
138    }
139    r
140}

142/// The master commit this clone has for `base`, as fetched.
143pub fn local_master(root: &Path, base: &str) -> Option<String> {
144    let out = std::process::Command::new("git")
145        .arg("-C")
146        .arg(root)
147        .args([
148            "rev-parse",
149            "--verify",
150            "--quiet",
151            &format!("refs/remotes/origin/{base}^{{commit}}"),
152        ])
153        .output()
154        .ok()?;
155    out.status
156        .success()
157        .then(|| String::from_utf8_lossy(&out.stdout).trim().to_string())
158        .filter(|s| !s.is_empty())
159}

172/// The integration queue: every open pull request assessed against one master, ranked.
173#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
174pub struct IntegrationQueue {
175    /// `owner/name`.
176    pub repository: String,
177    /// The integration base.
178    pub base: String,
179    /// The master commit every assessment was decided against (this clone's fetched ref).
180    pub master_sha: String,
181    /// When the forge was observed.
182    pub observed_at: String,
183    /// The base commit the forge reported at that moment.
184    pub observed_base_sha: String,
185    /// The policy in force.
186    pub policy: IntegrationPolicy,
187    /// Every open pull request, in rank order: the first `ready` one is the next merge.
188    pub assessments: Vec<PullRequestAssessment>,
189    /// The next merge, when anything is ready.
190    pub next_merge: Option<u64>,
191    /// The pull requests the executor could refresh (bring master in), in rank order.
192    pub next_refresh: Vec<u64>,
193    /// Counts.
194    pub tallies: QueueTallies,
195    /// What makes this queue less than a full answer.
196    pub diagnostics: Vec<String>,
197    /// The actionable pull requests the executor has passed over at least
198    /// [`wait::STARVING_AFTER`] times in their current wait, in rank order. Visible, never
199    /// promoted: see [`wait`].
200    #[serde(default)]
201    pub starving: Vec<u64>,
202}

211/// The policy of an observation: required checks and reviews as the forge reported them,
212/// the blocking labels declared once in [`BLOCKING_LABELS`], and the merge method the
213/// repository allows — a merge commit when it allows one, because the derived-file driver
214/// resolves merges and a squash or rebase would replay commits it never saw.
215pub fn policy_of(obs: &ForgeObservation) -> IntegrationPolicy {
216    let merge_method = ["merge", "squash", "rebase"]
217        .iter()
218        .find(|m| obs.merge_methods.iter().any(|x| x == *m))
219        .unwrap_or(&"merge")
220        .to_string();
221    IntegrationPolicy {
222        base: obs.base.clone(),
223        required_checks: obs.required_checks.clone(),
224        reviews_required: obs.reviews_required,
225        blocking_labels: BLOCKING_LABELS.iter().map(|s| s.to_string()).collect(),
226        merge_method,
227    }
228}

230/// The rank key: lane, then disposition, then risk, then how many other *ready or
231/// refreshable* pull requests share an authored path (fewer first: landing it invalidates
232/// less), then age (older first, so easy new work cannot starve old work), then number.
233/// Every component is a value of the assessment; the order is total and input-order free.
234fn rank_key(
235    a: &PullRequestAssessment,
236    contenders: &BTreeSet<u64>,
237) -> (u8, u8, u8, usize, String, u64) {
238    let lane = a.lane as u8;
239    let disposition = a.disposition as u8;
240    let risk = a.risk as u8;
241    let contention = a
242        .overlaps
243        .iter()
244        .filter(|o| contenders.contains(&o.number))
245        .count();
246    (
247        lane,
248        disposition,
249        risk,
250        contention,
251        a.created_at.clone(),
252        a.number,
253    )
254}
255
256/// Rank assessments. Deterministic whatever order they arrive in.
257pub fn rank(mut assessments: Vec<PullRequestAssessment>) -> Vec<PullRequestAssessment> {
258    let contenders: BTreeSet<u64> = assessments
259        .iter()
260        .filter(|a| {
261            matches!(
262                a.disposition,
263                PullRequestDisposition::Ready | PullRequestDisposition::NeedsRefresh
264            )
265        })
266        .map(|a| a.number)
267        .collect();
268    assessments.sort_by_cached_key(|a| rank_key(a, &contenders));
269    assessments
270}

272/// Build the queue from an observation, against `master_sha`. `relation` answers what a
273/// head is to master; the command line passes git, a test passes a table.
274pub fn build_queue(
275    obs: &ForgeObservation,
276    master_sha: &str,
277    relation: impl FnMut(&PullRequestObservation) -> RelationToMaster,
278) -> IntegrationQueue {
279    let policy = policy_of(obs);
280    let mut diagnostics = Vec::new();
281    if !obs.base_sha.is_empty() && obs.base_sha != master_sha {
282        diagnostics.push(format!(
283            "stale observation: the forge reported {} at {} but this clone's master is {}; majordomus prs refresh",
284            short(&obs.base_sha),
285            obs.observed_at,
286            short(master_sha)
287        ));
288    }
289    if policy.required_checks.is_none() {
290        diagnostics
291            .push("the branch protection could not be read: no pull request can be ready".into());
292    }
293    let relations: Vec<RelationToMaster> = obs.pull_requests.iter().map(relation).collect();
294    let mut queue = QueueContext {
295        open: obs.pull_requests.iter().map(|p| p.number).collect(),
296        heads: obs
297            .pull_requests
298            .iter()
299            .map(|p| (p.head_ref.clone(), p.number))
300            .collect(),
301        authored: BTreeMap::new(),
302    };
303    for (p, r) in obs.pull_requests.iter().zip(&relations) {
304        let authored = match r {
305            RelationToMaster::UpToDate { authored } | RelationToMaster::Behind { authored, .. } => {
306                authored.clone()
307            }
308            RelationToMaster::Conflicting { paths } => paths.clone(),
309            _ => Vec::new(),
310        };
311        queue.authored.insert(p.number, authored);
312    }
313    let assessments: Vec<PullRequestAssessment> = obs
314        .pull_requests
315        .iter()
316        .zip(&relations)
317        .map(|(p, r)| classify(p, r, master_sha, &policy, &queue))
318        .collect();
319    let assessments = rank(assessments);
320    let mut tallies = QueueTallies {
321        open: assessments.len(),
322        ..Default::default()
323    };
324    for a in &assessments {
325        *tallies
326            .by_disposition
327            .entry(a.disposition.as_str().to_string())
328            .or_default() += 1;
329        *tallies.by_lane.entry(classify::word(&a.lane)).or_default() += 1;
330    }
331    IntegrationQueue {
332        repository: obs.repository.clone(),
333        base: obs.base.clone(),
334        master_sha: master_sha.to_string(),
335        observed_at: obs.observed_at.clone(),
336        observed_base_sha: obs.base_sha.clone(),
337        next_merge: assessments
338            .iter()
339            .find(|a| a.disposition == PullRequestDisposition::Ready)
340            .map(|a| a.number),
341        next_refresh: assessments
342            .iter()
343            .filter(|a| a.disposition == PullRequestDisposition::NeedsRefresh)
344            .map(|a| a.number)
345            .collect(),
346        policy,
347        assessments,
348        tallies,
349        diagnostics,
350        starving: Vec::new(),
351    }
352}

354/// The queue of this checkout, from the last recorded observation, with relations decided
355/// by git against this clone's fetched master. Offline: it reads the network's last answer
356/// and never asks it again.
357pub fn queue_of(root: &Path) -> Result<IntegrationQueue, String> {
358    let obs = load_observation(root)?.ok_or_else(|| {
359        "no forge observation is recorded in this checkout; majordomus prs refresh".to_string()
360    })?;
361    let master = local_master(root, &obs.base).ok_or_else(|| {
362        format!(
363            "this clone has no refs/remotes/origin/{}; majordomus prs refresh fetches it",
364            obs.base
365        )
366    })?;
367    let cache_path = state_path(root, RELATIONS_FILE);
368    let mut cache: RelationCache = std::fs::read_to_string(&cache_path)
369        .ok()
370        .and_then(|t| serde_json::from_str(&t).ok())
371        .unwrap_or_default();
372    // bounded: only the current master's entries are worth keeping
373    cache
374        .entries
375        .retain(|k, _| k.starts_with(&format!("{master}..")));
376    let mut queue = build_queue(&obs, &master, |p| {
377        let head = format!("{PR_REF_PREFIX}{}", p.number);
378        // the fetched ref must still be the observed head: a head that moved since is
379        // decided against the SHA the forge reported, which the fetch brought in
380        let sha = if relation::has_commit(root, &p.head_sha) {
381            p.head_sha.clone()
382        } else {
383            head
384        };
385        relation_cached(root, &mut cache, &master, &sha)
386    });
387    if let Ok(text) = serde_json::to_string(&cache) {
388        let _ = write_atomic(&cache_path, &text);
389    }
390    wait::annotate(&mut queue, &drain::events(root));
391    // the summary a briefing reads without deciding a single relation (QueueSummary)
392    if let Ok(text) = serde_json::to_string_pretty(&QueueSummary::of(&queue)) {
393        let _ = write_atomic(&state_path(root, SUMMARY_FILE), &(text + "\n"));
394    }
395    Ok(queue)
396}

401/// What the last queue built said, in a few numbers: what a session briefing prints without
402/// building a queue, which would decide every relation again. Written by [`queue_of`] every
403/// time it builds one, so it is exactly as current as the last read of the queue, and says
404/// when that was.
405#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
406pub struct QueueSummary {
407    /// The integration base.
408    pub base: String,
409    /// The master commit the queue was decided against.
410    pub master_sha: String,
411    /// When the forge was observed.
412    pub observed_at: String,
413    /// When the queue was built, RFC 3339.
414    pub built_at: String,
415    /// The counts.
416    pub tallies: QueueTallies,
417    /// The next merge, when anything was ready.
418    pub next_merge: Option<u64>,
419    /// The starving pull requests.
420    pub starving: Vec<u64>,
421    /// Whether the queue carried diagnostics (a stale observation first).
422    pub diagnostics: usize,
423}
424
425impl QueueSummary {
426    /// The summary of a queue, built now.
427    pub fn of(q: &IntegrationQueue) -> Self {
428        QueueSummary {
429            base: q.base.clone(),
430            master_sha: q.master_sha.clone(),
431            observed_at: q.observed_at.clone(),
432            built_at: crate::peers::rfc3339(std::time::SystemTime::now()),
433            tallies: q.tallies.clone(),
434            next_merge: q.next_merge,
435            starving: q.starving.clone(),
436            diagnostics: q.diagnostics.len(),
437        }
438    }
439
440    /// The summary recorded in this checkout, if any.
441    pub fn load(root: &Path) -> Option<Self> {
442        std::fs::read_to_string(state_path(root, SUMMARY_FILE))
443            .ok()
444            .and_then(|t| serde_json::from_str(&t).ok())
445    }
446}

448/// Observe the forge, fetch what it names, and record the observation. The network step;
449/// the only one outside the executor.
450pub fn refresh(root: &Path) -> Result<ForgeObservation, String> {
451    use forge::Forge;
452    let obs = forge::GhForge { root }.observe().map_err(|e| e.0)?;
453    forge::fetch(root, &obs).map_err(|e| e.0)?;
454    store_observation(root, &obs)?;
455    Ok(obs)
456}
