1//! `majordomus guard`: the edit guard on the command line (I2735).
2//!
3//! `guard hook` is what a provider's pre-tool hook runs. It reads the provider's payload from
4//! stdin, answers `guard.edit` for the path it names without building the index — it runs
5//! before every edit — and says the answer in the one code the provider reads as a refusal:
6//! 2, with the reason and the remedy on stderr, which the provider hands to the model. Every
7//! other outcome is 0, an undecidable one included: a guard that refused because it could
8//! not read its own inputs would stop every edit of a checkout nobody has served.
9
10use std::io::{Read, Write};
11use std::path::{Path, PathBuf};
12
13use crate::cli::{GuardArgs, GuardCommand, GuardHookArgs};
14use crate::error::Result;
15use crate::guard::{from_payload, judge, server_claims, ClaimSource, EditVerdict};
16
17/// The code a provider's pre-tool hook reads as "do not run this tool".
18const EXIT_REFUSED: u8 = 2;
19
20/// Run `majordomus guard`.
21pub fn run(args: GuardArgs) -> Result<u8> {
22    match args.command {
23        GuardCommand::Hook(a) => hook(a),
24    }
25}
26
27fn hook(args: GuardHookArgs) -> Result<u8> {
28    // a payload that cannot be read whole names no path, and an edit nobody can name is let
29    // through like every other undecidable one
30    let mut payload = String::new();
31    let _ = std::io::stdin().read_to_string(&mut payload);
32    let base = args.repo.clone().or_else(|| std::env::current_dir().ok());
33    let verdict = answer(&args.provider, base, &payload, server_claims);
34    Ok(report(verdict.as_ref(), &mut std::io::stderr()))
35}
36
37/// The verdict on the edit `payload` names, or `None` when it names no file of a repository.
38///
39/// The checkout is the one the written file belongs to, not the one the hook was installed
40/// in: a session started in the primary checkout edits files of linked worktrees, and each of
41/// those has its own task, its own server and its own claims. A relative path is `base`'s,
42/// the hook's working directory, which is the provider's project.
43fn answer(
44    provider: &str,
45    base: Option<PathBuf>,
46    payload: &str,
47    claims: impl FnOnce(&Path) -> ClaimSource,
48) -> Option<EditVerdict> {
49    let (path, session) = from_payload(payload)?;
50    let target = base.unwrap_or_default().join(path);
51    let start = target.ancestors().find(|a| a.is_dir())?.to_path_buf();
52    let repository = crate::repository::Repository::discover(&start).ok()?;
53    let root = repository.root().to_path_buf();
54    // the root is canonical, so the path is spelled the same way: its existing ancestor
55    // canonicalised (discover has just done so without failing), and whatever of it does not
56    // exist yet appended (`start` is one of its ancestors, so it is a prefix)
57    let rest = target.strip_prefix(&start).unwrap_or(target.as_path());
58    let path = start.canonicalize().unwrap_or_default().join(rest);
59    let session = session.map(|s| format!("{provider}-{s}"));
60    Some(judge(
61        &root,
62        &path.to_string_lossy(),
63        session.as_deref(),
64        || claims(&root),
65    ))
66}
67
68/// The exit code for `verdict`, with the reason and the remedy written to `err` for a
69/// refusal: the provider hands that text to the model.
70fn report(verdict: Option<&EditVerdict>, err: &mut dyn Write) -> u8 {
71    let Some(refused) = verdict.filter(|v| !v.allowed) else {
72        return 0;
73    };
74    let remedy = refused.remedy.as_deref().unwrap_or_default();
75    let _ = writeln!(
76        err,
77        "majordomus guard: {}\n  {remedy}\n  \
78         (session.edits_require_claim in .ai/repo/policy.yaml turns this guard off; I2735)",
79        refused.detail
80    );
81    EXIT_REFUSED
82}

84#[cfg(test)]
85mod tests {
86    use super::*;

88    /// A repository on the mesh whose policy asks every edit for a claim.
89    fn repository() -> tempfile::TempDir {
90        let skeleton = Path::new(env!("CARGO_MANIFEST_DIR")).join("../../share/skeleton");
91        let dir = tempfile::tempdir().unwrap();
92        let ai = dir.path().join(".ai");
93        std::fs::create_dir_all(ai.join("repo/mesh")).unwrap();
94        std::fs::copy(skeleton.join("ai/manifest.yaml"), ai.join("manifest.yaml")).unwrap();
95        let policy = std::fs::read_to_string(skeleton.join("policy.yaml"))
96            .unwrap()
97            .replace("edits_require_claim: false", "edits_require_claim: true");
98        std::fs::write(ai.join("repo/policy.yaml"), policy).unwrap();
99        std::fs::write(
100            ai.join("repo/mesh/mesh.yaml"),
101            "enabled: true\ntrust:\n  policy: tofu\n",
102        )
103        .unwrap();
104        dir
105    }
106
107    fn edit(path: &str) -> String {
108        serde_json::json!({ "session_id": "s1", "tool_input": { "file_path": path } }).to_string()
109    }
110
111    #[test]
112    fn an_unclaimed_edit_is_refused_with_the_reason_and_the_remedy() {
113        let dir = repository();
114        let root = dir.path().canonicalize().unwrap();
115        let verdict = answer(
116            "claude-code",
117            Some(dir.path().to_path_buf()),
118            &edit("docs/new.md"),
119            |asked| {
120                assert_eq!(asked, root.as_path(), "the checkout of the file is asked");
121                ClaimSource::Claims(vec![])
122            },
123        )
124        .expect("a file of the repository");
125        assert!(!verdict.allowed);
126        assert_eq!(
127            verdict.path, "docs/new.md",
128            "a file that does not exist yet is placed"
129        );
130        let mut err = Vec::new();
131        assert_eq!(report(Some(&verdict), &mut err), EXIT_REFUSED);
132        let text = String::from_utf8(err).unwrap();
133        assert!(text.contains("no claim covers docs/new.md"), "{text}");
134        assert!(text.contains("--session claude-code-s1"), "{text}");
135    }

137    #[test]
138    fn an_edit_naming_no_file_of_a_repository_is_let_through() {
139        let dir = repository();
140        let none = |_: &Path| -> ClaimSource { unreachable!("nothing to judge") };
141        let base = Some(dir.path().to_path_buf());
142        assert!(answer("claude-code", base.clone(), "not json", none).is_none());
143        let ls = r#"{"tool_input":{"command":"ls"}}"#;
144        assert!(answer("claude-code", base, ls, none).is_none());
145        let bare = tempfile::tempdir().unwrap();
146        let outside = edit(&bare.path().join("a.md").to_string_lossy());
147        assert!(
148            answer("claude-code", None, &outside, none).is_none(),
149            "no manifest is above it"
150        );
151        let nowhere = edit("no-such-directory/y.md");
152        assert!(answer("claude-code", None, &nowhere, none).is_none());
153        let mut err = Vec::new();
154        assert_eq!(report(None, &mut err), 0);
155        assert!(err.is_empty());
156    }

158    #[test]
159    fn an_allowed_edit_says_nothing() {
160        let dir = repository();
161        let claims = |_: &Path| ClaimSource::Unavailable("no server".into());
162        let verdict = answer(
163            "claude-code",
164            Some(dir.path().into()),
165            &edit("a.md"),
166            claims,
167        )
168        .expect("a file of the repository");
169        assert_eq!(verdict.ground, "undecided");
170        let mut err = Vec::new();

172        assert!(err.is_empty());
173    }
