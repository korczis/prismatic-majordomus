1//! The rules module: a rule, and everything the repository can show that proves it.
2//!
3//! A rule of this repository is a Markdown object under the rules section with YAML front
4//! matter: an id, a version, a class (`blocking` or `advisory`), a statement, and — since
5//! the proof relation became load-bearing — an `x-majordomus` block naming what enforces
6//! it. This module is the one typed reading of that object, and the one place the
7//! repository decides what a rule's proof amounts to.
8//!
9//! # Why this exists rather than another list
10//!
11//! Before it, "is this rule enforced?" was answered by `o.metadata.get("x-majordomus")
12//! .is_some()` — a green badge for the presence of a YAML key. A rule could name
13//! `test/cases/28_no_hardcoded_values.sh` after that case was deleted and still read as
14//! enforced on every surface that asked. The proof graph here answers the question the
15//! badge pretended to: the named validator is in the tree, the named cases are in the tree,
16//! a runner owns them, a run of them was recorded, and that run is not older than what it
17//! is about.
18//!
19//! # What it does not do
20//!
21//! It does not re-implement evidence. [`crate::evidence`] already binds a test path to the
22//! executions recorded against it and already decides, with [`ProofState`], what a passing
23//! run from an older commit is worth. A rule's test-level proof *is* that vocabulary; this
24//! module joins rules to it rather than inventing a second one. Nor does it dispatch: the
25//! portable shell layer's `lib/doctrine.sh` runs the validators, and `scripts/ci/
26//! rule-proof-check` is the gate. This is the reading all of those project from.
27//!
28//! # The two enforcement modes
29//!
30//! The canonical front matter allows exactly two shapes, and the distinction is not
31//! cosmetic:
32//!
33//! * **dispatched** — the block names a `validator`, which `lib/doctrine.sh` calls at run
34//!   time. It carries `category`, `exit_code` and `enforced_by` with it; a half-declaration
35//!   (a category without a validator) is a defect the loader refuses rather than a rule
36//!   with a missing field.
37//! * **gated** — the block names `tests` and no validator. Nothing dispatches it; the
38//!   behavioural cases are what prove it, and a CI gate is what runs them.
39//!
40//! Either mode names at least one test. A blocking rule that names neither is `Unproven`,
41//! which is the state this module exists to make visible rather than absent.
42
43//! # Example
44//!
45//! The vocabulary, without a repository: a class is read rather than guessed, the mode is
46//! whatever the block names, and every state ranks and says what it means.
47//!
48//! ```
49//! use majordomus_cli::rules::{Class, Mode, RuleState};
50//!
51//! assert_eq!(Class::parse("blocking"), Class::Blocking);
52//! assert_eq!(Class::parse("strict"), Class::Unknown);
53//! assert_eq!(Mode::Reviewed.label(), "reviewed");
54//! assert!(RuleState::Proven < RuleState::Dangling);
55//! assert!(!RuleState::Gated.passing());
56//! ```

88/// What a rule declares about how strongly it binds.
89///
90/// The repository has exactly two, and they are not severities on a scale: `blocking` means
91/// a gate refuses the work, `advisory` means a reader is expected to have read it. A rule
92/// whose front matter says neither is reported as [`Class::Unknown`] rather than defaulted,
93/// because guessing here is how a blocking rule quietly becomes advice.
94#[derive(
95    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
96)]
97#[serde(rename_all = "snake_case")]
98#[schemars(rename = "RuleClass")]
99/// # Example
100///
101/// ```
102/// use majordomus_cli::rules::Class;
103/// assert_eq!(Class::Blocking.label(), "blocking");
104/// assert_eq!(Class::parse("advisory"), Class::Advisory);
105/// ```
106pub enum Class {
107    /// A gate refuses work that violates it.
108    Blocking,
109    /// Normative for whoever reads it; nothing refuses a violation.
110    Advisory,
111    /// The front matter declared no class, or one this repository does not know.
112    Unknown,
113}
114
115impl Class {
116    /// The word every surface prints for this class, written here once so the command
117    /// line, the API and the Cockpit cannot disagree about what to call it.
118    ///
119    /// ```
120    /// use majordomus_cli::rules::Class;
121    /// assert_eq!(Class::Unknown.label(), "unknown");
122    /// ```
123    pub fn label(self) -> &'static str {
124        match self {
125            Class::Blocking => "blocking",
126            Class::Advisory => "advisory",
127            Class::Unknown => "unknown",
128        }
129    }
130
131    /// Read a class from the front matter, without defaulting.
132    /// ```
133    /// use majordomus_cli::rules::Class;
134    /// assert_eq!(Class::parse(" blocking "), Class::Blocking);
135    /// assert_eq!(Class::parse(""), Class::Unknown);
136    /// ```
137    pub fn parse(word: &str) -> Class {
138        match word.trim() {
139            "blocking" => Class::Blocking,
140            "advisory" => Class::Advisory,
141            _ => Class::Unknown,
142        }
143    }
144}

146/// How a rule is enforced, as its own front matter declares it.
147#[derive(
148    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
149)]
150#[serde(rename_all = "snake_case")]
151#[schemars(rename = "RuleMode")]
152/// # Example
153///
154/// ```
155/// use majordomus_cli::rules::Mode;
156/// assert_eq!(Mode::Dispatched.label(), "dispatched");
157/// assert_eq!(Mode::Declarative.label(), "declarative");
158/// ```
159pub enum Mode {
160    /// Names a validator the doctrine dispatcher calls at run time.
161    Dispatched,
162    /// Names behavioural cases and no validator; a gate runs them.
163    Gated,
164    /// Carries a reason why nothing executable can express it, and so declares that a
165    /// reader enforces it. An honest declaration, not an absence: the reason *is* the
166    /// declaration — there is no flag beside it, because a flag can be set and a reason
167    /// cannot be set without writing one — it is counted apart from every executable mode,
168    /// and it separates a rule nobody could automate from a rule nobody got round to.
169    Reviewed,
170    /// Names neither. Correct for a rule nothing executable can express; a defect for a
171    /// blocking one.
172    Declarative,
173}
174
175impl Mode {
176    /// The word every surface prints for this enforcement mode, written here once so no
177    /// projection invents its own spelling of a mode.
178    ///
179    /// ```
180    /// use majordomus_cli::rules::Mode;
181    /// assert_eq!(Mode::Gated.label(), "gated");
182    /// ```
183    pub fn label(self) -> &'static str {
184        match self {
185            Mode::Dispatched => "dispatched",
186            Mode::Gated => "gated",
187            Mode::Reviewed => "reviewed",
188            Mode::Declarative => "declarative",
189        }
190    }
191}

196/// # Example
197///
198/// Read from front matter rather than built by hand; this is the shape a gated rule

200///
201/// ```
202/// use majordomus_cli::rules::{Enforcement, Mode};

216    pub mode: Mode,
217    /// The validator the dispatcher calls, when the mode is dispatched.
218    #[serde(default, skip_serializing_if = "Option::is_none")]
219    pub validator: Option<String>,
220    /// The diagnostic category the validator reports under.
221    #[serde(default, skip_serializing_if = "Option::is_none")]
222    pub category: Option<String>,
223    /// The exit code the validator uses to refuse.
224    #[serde(default, skip_serializing_if = "Option::is_none")]
225    pub exit_code: Option<i64>,
226    /// Which boundary enforces it: the word the rule's own block uses.
227    #[serde(default, skip_serializing_if = "Option::is_none")]
228    pub enforced_by: Option<String>,
229    /// Every test the block names, in the order it names them.
230    #[serde(default, skip_serializing_if = "Vec::is_empty")]
231    pub tests: Vec<String>,
232    /// Why nothing executable can express this rule, when the mode is reviewed. Required
233    /// there: a declared exemption with no reason is the debt it exists to replace.
234    #[serde(default, skip_serializing_if = "Option::is_none")]
235    pub reviewed_because: Option<String>,
236}

241///

245/// ```
246/// use majordomus_cli::rules::{Class, RuleDefinition};
247/// # use majordomus_cli::rules::{Enforcement, Mode};
248/// let d = RuleDefinition {
249///     id: "project.scope-is-declared".into(),
250///     identity: "project.scope-is-declared@1".into(),

256///     status: "active".into(),

258///     namespace: "project".into(),
259///     depends_on: vec![],
260///     tags: vec![],
261///     path: ".ai/repo/rules/project/scope-is-declared.v1.md".into(),
262///     enforcement: Enforcement {

264///         enforced_by: None, tests: vec![], reviewed_because: None,

267/// assert_eq!(d.class, Class::Blocking);

269pub struct RuleDefinition {
270    /// The rule id, `<namespace>.<stem>`, without the version.
271    pub id: String,
272    /// The identity the index holds, with the version.
273    pub identity: String,

275    pub uri: String,
276    /// The declared version.
277    pub version: u64,
278    /// The title.
279    pub title: String,
280    /// The one-line description.
281    #[serde(default, skip_serializing_if = "Option::is_none")]
282    pub description: Option<String>,
283    /// The normative sentence.
284    #[serde(default, skip_serializing_if = "Option::is_none")]
285    pub statement: Option<String>,
286    /// `active`, or whatever else the front matter declares.
287    pub status: String,
288    /// How strongly it binds.
289    pub class: Class,
290    /// Whether it came with the vendored package or belongs to this project.
291    pub namespace: String,
292    /// The rules it depends on, by id.
293    #[serde(default, skip_serializing_if = "Vec::is_empty")]
294    pub depends_on: Vec<String>,
295    /// The declared tags.
296    #[serde(default, skip_serializing_if = "Vec::is_empty")]
297    pub tags: Vec<String>,
298    /// Repository-relative path of the rule object.
299    pub path: String,
300    /// How it is enforced, as declared.
301    pub enforcement: Enforcement,
302}

323/// The validator a dispatched rule names, and whether anything defines it.
324///
325/// A validator is **not a path**. `validator: adr` names the shell function
326/// `mj_validate_adr`, which the doctrine dispatcher calls; the tool finds it by scanning
327/// `lib/`, exactly as `majordomus doctor` does. Reading it as a path is a mistake worth
328/// recording here, because it is the one this type was introduced to fix: it makes every
329/// dispatched rule in the vendored package report as naming something that is not in the
330/// tree, which is the opposite of true — those are the best-enforced rules there are.
331#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
332#[schemars(rename = "RuleValidator")]
333/// # Example
334///
335/// ```
336/// use majordomus_cli::rules::ValidatorRef;
337/// let v = ValidatorRef {
338///     name: "adr".into(),
339///     function: "mj_validate_adr".into(),
340///     defined_in: Some("lib/adr.sh".into()),
341///     present: true,
342/// };
343/// assert_eq!(v.function, "mj_validate_adr");
344/// ```
345pub struct ValidatorRef {
346    /// The name the rule declares, without the `mj_validate_` prefix.
347    pub name: String,
348    /// The function the dispatcher calls.
349    pub function: String,
350    /// The file that defines it, when one does.
351    #[serde(default, skip_serializing_if = "Option::is_none")]
352    pub defined_in: Option<String>,
353    /// Whether anything in `lib/` defines it.
354    pub present: bool,
355}

357/// What a path a rule names actually is, decided by what drives it rather than by where it
358/// sits.
359///
360/// The corpus names two kinds and this distinction is the difference between a verdict and
361/// a mechanism. A **case** is driven by a runner, so an execution of it can be recorded and
362/// the repository can show that it passed. A **gate** is an executable check a CI gate runs
363/// as its own command: it refuses violations on every run, and this repository records no
364/// verdict for it, so what can be shown is the mechanism and not the result. Anything else
365/// a rule names is proof only in prose.
366#[derive(
367    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
368)]
369#[serde(rename_all = "snake_case")]
370#[schemars(rename = "RuleArtifactKind")]
371/// # Example
372///
373/// ```
374/// use majordomus_cli::rules::ArtifactKind;
375/// assert_eq!(ArtifactKind::Case.label(), "case");
376/// assert_eq!(ArtifactKind::Gate.label(), "gate");
377/// ```
378pub enum ArtifactKind {
379    /// A behavioural case a runner drives; executions of it are recorded.
380    Case,
381    /// An executable check a CI gate runs as its own command.
382    Gate,
383    /// Neither. Nothing can run it, so nothing can ever record it.
384    Unknown,
385}
386
387impl ArtifactKind {
388    /// The word every surface prints for what drives this artifact, so that a gate and a
389    /// case are never rendered as the same kind of thing.
390    ///
391    /// ```
392    /// use majordomus_cli::rules::ArtifactKind;
393    /// assert_eq!(ArtifactKind::Unknown.label(), "unknown");
394    /// ```
395    pub fn label(self) -> &'static str {
396        match self {
397            ArtifactKind::Case => "case",
398            ArtifactKind::Gate => "gate",
399            ArtifactKind::Unknown => "unknown",
400        }
401    }
402}

451/// What the repository can say about one rule's proof, as a whole.
452///
453/// Ordered strongest to weakest so that a summary sorted by this reads as a ranking. A rule's
454/// state is made from its parts' by [`crate::evidence::freshness::aggregate`]: a failing part
455/// makes it failing, and otherwise the weakest part that can carry proof decides.
456#[derive(
457    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
458)]
459#[serde(rename_all = "snake_case")]
460/// # Example
461///
462/// ```
463/// use majordomus_cli::rules::RuleState;
464/// assert!(RuleState::Proven < RuleState::Stale);
465/// assert!(RuleState::Stale.passing());
466/// assert!(!RuleState::Dangling.passing());
467/// ```
468pub enum RuleState {
469    /// Every artifact it names is present, and every test it names has a passing run that
470    /// nothing has invalidated.
471    Proven,
472    /// Every artifact is present and every test passed, but at least one run is only
473    /// un-invalidated rather than measured against this tree.
474    InputsUnchanged,
475    /// A passing run exists, but something it is about has changed since.
476    Stale,
477    /// An executable gate runs what this rule names, on every CI run that selects it. The
478    /// mechanism is in the tree and wired; no verdict for it is recorded here.
479    Gated,
480    /// The latest run of a test this rule names did not pass.
481    Failing,
482    /// Every artifact is present, and no run of at least one named test was ever recorded.
483    NotRun,
484    /// The rule declares, with a stated reason, that nothing executable can express it. An
485    /// honest exemption: counted apart from every proven rule, never counted as one.
486    Reviewed,
487    /// The rule names a path no runner in this repository drives.
488    Unrunnable,
489    /// The rule names a path that is not in the tree. A proof that does not exist is worse
490    /// than no proof, because the rule reads as proven.
491    Dangling,
492    /// The rule names neither a validator nor a test. Correct for a rule nothing executable
493    /// can express; a defect for a blocking one.
494    Unproven,
495}
496
497impl RuleState {
498    /// The word every surface prints for this state, written here once so that `proven`
499    /// and `inputs unchanged` cannot be collapsed into one badge by a renderer.
500    ///
501    /// ```
502    /// use majordomus_cli::rules::RuleState;
503    /// assert_eq!(RuleState::InputsUnchanged.label(), "inputs unchanged");
504    /// ```
505    pub fn label(self) -> &'static str {
506        match self {
507            RuleState::Proven => "proven",
508            RuleState::InputsUnchanged => "inputs unchanged",
509            RuleState::Stale => "stale",
510            RuleState::Gated => "gated",
511            RuleState::Failing => "failing",
512            RuleState::NotRun => "not run",
513            RuleState::Reviewed => "reviewed",
514            RuleState::Unrunnable => "unrunnable",
515            RuleState::Dangling => "dangling",
516            RuleState::Unproven => "unproven",
517        }
518    }
519
520    /// One sentence: what the state means, worded once so that every surface says the same
521    /// thing rather than each inventing its own gloss.
522    /// ```
523    /// use majordomus_cli::rules::RuleState;
524    /// assert!(RuleState::Dangling.meaning().contains("not in the tree"));
525    /// ```
526    pub fn meaning(self) -> &'static str {
527        match self {
528            RuleState::Proven => {
529                "Everything this rule names is in the tree, and every test it names has a passing \
530                 run that nothing in the repository has changed since."
531            }
532            RuleState::InputsUnchanged => {
533                "Everything this rule names is in the tree and passed, but at least one run is \
534                 only un-invalidated: nothing it names has changed since, which is the absence of \
535                 a known invalidation rather than proof against this commit."
536            }
537            RuleState::Stale => {
538                "A passing run exists, but a file it is about has changed since. The proof is \
539                 older than its subject."
540            }
541            RuleState::Gated => {
542                "An executable check refuses violations of this rule, and a CI gate runs it. \
543                 That is the mechanism; this repository records no verdict for a gate, so what \
544                 can be shown is that violations are refused and not that the last run passed."
545            }
546            RuleState::Failing => {
547                "The most recent recorded run of a test this rule names did not pass."
548            }
549            RuleState::NotRun => {
550                "This rule names a test a runner owns, and no run of it has ever been recorded."
551            }
552            RuleState::Reviewed => {
553                "This rule declares that nothing executable can express it and gives its reason. \
554                 A reader enforces it. That is a weaker guarantee than any gate, and it is \
555                 counted apart from every rule that has one."
556            }
557            RuleState::Unrunnable => {
558                "This rule names a path no runner in this repository drives, so no execution of \
559                 it can ever be recorded."
560            }
561            RuleState::Dangling => {
562                "This rule names a path that is not in the tree. It reads as proven on every \
563                 surface that checks only the name."
564            }
565            RuleState::Unproven => "This rule names neither a validator nor a test.",
566        }
567    }
568
569    /// Does this state carry a passing execution behind it, of any freshness?
570    /// ```
571    /// use majordomus_cli::rules::RuleState;
572    /// assert!(RuleState::Proven.passing());
573    /// assert!(!RuleState::Unproven.passing());
574    /// ```
575    pub fn passing(self) -> bool {
576        matches!(
577            self,
578            RuleState::Proven | RuleState::InputsUnchanged | RuleState::Stale
579        )
580    }
581}

583/// One rule, joined to everything the repository can show about it.
584#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
585/// # Example
586///
587/// A `RuleProof` is what [`report`] produces per rule; `satisfied` is whether the state
588/// supports the class the rule declares, which is the whole question.
589///
590/// ```
591/// use majordomus_cli::rules::{report, RuleProof, RuleState};
592/// use majordomus_cli::evidence::Ledger;
593/// use majordomus_cli::synthetic::SyntheticRepository;
594/// let repo = SyntheticRepository::small().unwrap();
595/// let ledger = Ledger::load(repo.root()).unwrap();
596/// let r = report(&repo.index().unwrap(), &ledger);
597/// let first: &RuleProof = &r.rules[0];

654/// How much of the rule corpus carries each part of a proof.

672    /// Rules discovered.
673    pub rules: usize,

675    pub blocking: usize,

702/// What the whole corpus can be said to prove, in one word, before any number is read.
703///
704/// `satisfied` answers a narrower question — are there findings — and a corpus in which no
705/// blocking rule was ever run has none, because `not run`, `gated` and `reviewed` are weaker
706/// states rather than defects (docs/DOCTRINE.md). Read alone, `satisfied: true` over such a
707/// corpus says "every rule is enforced" about a repository that has shown nothing. The
708/// verdict is the answer that cannot be read that way: it is `proven` only when every
709/// blocking rule is `proven` — a passing run recorded on a clean tree, at a commit that
710/// nothing but the evidence ledger has changed since. A rule whose inputs are merely
711/// unchanged is not that (ADR 0041), and leaves the corpus `unproven`.
712#[derive(
713    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
714)]
715#[serde(rename_all = "snake_case")]
716#[schemars(rename = "RulesVerdict")]
717/// # Example
718///
719/// ```
720/// use majordomus_cli::rules::{RuleState, RulesVerdict};
721/// // a corpus with a finding is failing, whatever else is true of it
722/// assert_eq!(RulesVerdict::of([RuleState::Proven], true), RulesVerdict::Failing);
723/// // one blocking rule never run is enough to leave the corpus unproven
724/// assert_eq!(
725///     RulesVerdict::of([RuleState::Proven, RuleState::NotRun], false),
726///     RulesVerdict::Unproven
727/// );
728/// assert_eq!(RulesVerdict::of([RuleState::Proven], false), RulesVerdict::Proven);
729/// // a pass whose inputs are merely unchanged is not proven
730/// assert_eq!(
731///     RulesVerdict::of([RuleState::Proven, RuleState::InputsUnchanged], false),
732///     RulesVerdict::Unproven
733/// );
734/// ```
735pub enum RulesVerdict {
736    /// Every blocking rule is proven: a passing run recorded on a clean tree, at a commit
737    /// that nothing but the evidence ledger has changed since.
738    Proven,
739    /// No rule is a finding, and at least one blocking rule is not proven — never run,
740    /// gated, reviewed, stale, or passing with only its inputs unchanged — or there is no
741    /// blocking rule at all.
742    Unproven,
743    /// At least one rule declares a class its proof does not support.
744    Failing,
745}
746
747impl RulesVerdict {
748    /// Decide the verdict from the state of every blocking rule and whether any finding
749    /// exists.
750    ///
751    /// Only `proven` counts. `stale` does not: a pass older than its subject is a pass of
752    /// something else. Nor does `inputs unchanged`: a run recorded on a dirty tree, or one
753    /// whose repository has moved since, is the absence of a known invalidation and not
754    /// proof (ADR 0041), and a corpus verdict of `proven` over it would be the one badge the
755    /// evidence states exist to refuse. An empty set of blocking rules is `unproven`, not
756    /// `proven` — a corpus that claims nothing has shown nothing, and success by vacancy is
757    /// the number that stops meaning anything.
758    /// ```
759    /// use majordomus_cli::rules::{RuleState, RulesVerdict};
760    /// assert_eq!(RulesVerdict::of([RuleState::Stale], false), RulesVerdict::Unproven);
761    /// assert_eq!(RulesVerdict::of([], false), RulesVerdict::Unproven);
762    /// assert_eq!(
763    ///     RulesVerdict::of([RuleState::InputsUnchanged], false),
764    ///     RulesVerdict::Unproven
765    /// );
766    /// ```
767    pub fn of(blocking: impl IntoIterator<Item = RuleState>, any_finding: bool) -> Self {
768        if any_finding {
769            return RulesVerdict::Failing;
770        }
771        let mut any = false;
772        for state in blocking {
773            any = true;
774            if state != RuleState::Proven {
775                return RulesVerdict::Unproven;
776            }
777        }
778        if any {
779            RulesVerdict::Proven
780        } else {
781            RulesVerdict::Unproven
782        }
783    }
784
785    /// The one word every surface prints for this verdict, and the one word a script greps
786    /// for: the serialised form and the rendered form are the same string, so a reader and
787    /// a parser can never be told different things about the same report. The three are
788    /// `proven`, `unproven` and `failing`, and the middle one exists because "no finding"
789    /// and "proven" are not the same answer — [`RulesVerdict::meaning`] says why in prose.
790    /// ```
791    /// use majordomus_cli::rules::RulesVerdict;
792    /// assert_eq!(RulesVerdict::Unproven.label(), "unproven");
793    /// assert_eq!(RulesVerdict::Proven.label(), "proven");
794    /// assert_eq!(RulesVerdict::Failing.label(), "failing");
795    /// ```
796    pub fn label(self) -> &'static str {
797        match self {
798            RulesVerdict::Proven => "proven",
799            RulesVerdict::Unproven => "unproven",
800            RulesVerdict::Failing => "failing",
801        }
802    }
803
804    /// What the verdict asserts, in one sentence, carried beside the label so that a reader
805    /// is never left to infer it from a colour or a word.
806    ///
807    /// The sentence is the whole point of the type. A report that found no fault used to be
808    /// rendered as a pass, and "no rule is a finding" is a far weaker statement than "every
809    /// blocking rule is proven": the first is satisfied by a corpus nothing has ever run. `unproven` says that distinction out loud, in the one place a
810    /// person reads, so that an absence of findings cannot be quoted as proof.
811    /// ```
812    /// use majordomus_cli::rules::RulesVerdict;
813    /// assert!(RulesVerdict::Unproven.meaning().contains("not a finding"));
814    /// assert!(RulesVerdict::Proven.meaning().contains("Every blocking rule"));
815    /// ```
816    pub fn meaning(self) -> &'static str {
817        match self {
818            RulesVerdict::Proven => {
819                "Every blocking rule is proven: a passing run recorded on a clean tree, at a \
820                 commit that nothing but the evidence ledger has changed since."
821            }
822            RulesVerdict::Unproven => {
823                "No rule is a finding, and at least one blocking rule is not proven — never run, \
824                 gated, reviewed, stale, or passing with only its inputs unchanged: not a finding \
825                 is not proven."
826            }
827            RulesVerdict::Failing => {
828                "At least one rule declares a class its proof does not support."
829            }
830        }
831    }
832}

836#[schemars(rename = "RulesReport")]
837/// # Example
838///
839/// The whole corpus joined to the tree and the ledger. `verdict` is what the corpus can be
840/// said to prove; `satisfied` is only that there are no findings, and the two differ

874    pub findings: Vec<Finding>,

890/// Read a string field of the front matter, treating an empty value as absence.
891fn field(meta: &Value, key: &str) -> Option<String> {
892    let v = meta.get(key)?;
893    let s = match v {
894        Value::String(s) => s.trim().trim_matches('\'').to_string(),
895        Value::Number(n) => n.to_string(),
896        Value::Bool(b) => b.to_string(),
897        _ => return None,
898    };
899    if s.is_empty() {
900        None
901    } else {
902        Some(s)
903    }
904}
905
906/// Read a list-of-strings field, accepting both the YAML flow list the rules use and a
907/// single scalar, which a hand-written rule occasionally is.
908fn list(meta: &Value, key: &str) -> Vec<String> {
909    match meta.get(key) {
910        Some(Value::Array(a)) => a
911            .iter()
912            .filter_map(|v| v.as_str())
913            .map(|s| s.trim().to_string())
914            .filter(|s| !s.is_empty())
915            .collect(),
916        Some(Value::String(s)) if !s.trim().is_empty() => vec![s.trim().to_string()],
917        _ => Vec::new(),
918    }
919}

921/// Read the enforcement block, deciding the mode from what it actually names rather than
922/// from a word it could also declare. A block that names a validator is dispatched; one
923/// that names only tests is gated; one that names neither is declarative, whatever else it
924/// carries.
925fn enforcement_of(meta: &Value) -> Enforcement {
926    let block = meta.get("x-majordomus");
927    let Some(block) = block else {
928        return Enforcement {
929            mode: Mode::Declarative,
930            validator: None,
931            category: None,
932            exit_code: None,
933            enforced_by: None,
934            tests: Vec::new(),
935            reviewed_because: None,
936        };
937    };
938    let validator = field(block, "validator");
939    let tests = list(block, "tests");
940    let enforced_by = field(block, "enforced_by");
941    let reviewed_because = field(block, "reviewed_because");
942    // The order matters. An executable proof wins over a declaration of review, so a rule
943    // that acquires a case stops being review-enforced without anyone remembering to
944    // delete the declaration. And a review declaration with no reason is not a mode: it
945    // falls through to declarative, which is the finding it was meant to replace.
946    let mode = if validator.is_some() {
947        Mode::Dispatched
948    } else if !tests.is_empty() {
949        Mode::Gated
950    } else if reviewed_because.is_some() {
951        Mode::Reviewed
952    } else {
953        Mode::Declarative
954    };
955    Enforcement {
956        mode,
957        validator,
958        category: field(block, "category"),
959        exit_code: block.get("exit_code").and_then(Value::as_i64),
960        enforced_by,
961        tests,
962        reviewed_because,
963    }
964}

966/// Every rule the index holds, as a canonical definition.
967///
968/// The denominator is read from the index, so a rule added tomorrow is measured tomorrow
969/// and this function never carries a list of what exists.
970/// ```
971/// use majordomus_cli::rules::definitions;
972/// use majordomus_cli::synthetic::SyntheticRepository;
973/// let repo = SyntheticRepository::small().unwrap();
974/// let defs = definitions(&repo.index().unwrap());
975/// assert_eq!(defs.len(), repo.shape.rules);
976/// ```
977pub fn definitions(index: &Index) -> Vec<RuleDefinition> {
978    let mut out: Vec<RuleDefinition> = index
979        .objects
980        .iter()
981        .filter(|o| o.kind == "rule")
982        .map(|o| {
983            let id = field(&o.metadata, "id").unwrap_or_else(|| o.identity.clone());
984            let namespace = id.split('.').next().unwrap_or_default().to_string();
985            RuleDefinition {
986                version: o
987                    .metadata
988                    .get("version")
989                    .and_then(Value::as_u64)
990                    .unwrap_or(1),
991                title: o.title.clone().unwrap_or_else(|| id.clone()),
992                description: o.description.clone(),
993                statement: field(&o.metadata, "statement"),
994                status: field(&o.metadata, "status").unwrap_or_else(|| "unknown".into()),
995                class: Class::parse(&field(&o.metadata, "class").unwrap_or_default()),
996                namespace,
997                depends_on: list(&o.metadata, "depends_on"),
998                tags: list(&o.metadata, "tags"),
999                path: o.provenance.path.clone(),
1000                enforcement: enforcement_of(&o.metadata),
1001                identity: o.identity.clone(),
1002                uri: o.uri.clone(),
1003                id,
1004            }
1005        })
1006        .collect();
1007    // canonical order, from the one comparator the crate has: two runs of this over the
1008    // same tree, whatever order discovery handed it, produce the same report
1009    crate::order::canonical(&mut out);
1010    out
1011}

1013/// Which gates this repository declares, and the command each runs.
1014///
1015/// `gates.yaml` is the CI model: `scripts/ci-plan` reads it and nothing else to decide what
1016/// a change must run. What a rule needs from it is narrower — which gate would run the
1017/// thing this rule names — and that is answerable from the one line each gate states
1018/// directly, its `runs:`. No gate id is written down here: a gate renamed in that file is
1019/// renamed in this answer, which is the whole reason to read it rather than to restate it.
1020fn gate_commands(root: &Path) -> Vec<(String, String)> {
1021    let Ok(text) = std::fs::read_to_string(root.join(GATES_PATH)) else {
1022        return Vec::new();
1023    };
1024    let mut out = Vec::new();
1025    let mut gate: Option<String> = None;
1026    for line in text.lines() {
1027        let t = line.trim();
1028        if let Some(rest) = t.strip_prefix("- id: ") {
1029            gate = Some(rest.trim().to_string());
1030        } else if let Some(rest) = t.strip_prefix("runs: ") {
1031            if let Some(g) = gate.take() {
1032                out.push((g, rest.trim().to_string()));
1033            }
1034        }
1035    }
1036    out
1037}

1039/// The gates that run one path the rule names.
1040///
1041/// Three bindings, each read from the gate's own command rather than assumed:
1042/// the gate whose command *is* that path (a validator wired as its own gate); the gate that
1043/// drives the behavioural suite, for a path under the cases directory; and the gate that
1044/// drives the crate's tests, for a path under the crate's test directory. A path none of
1045/// these resolves gets no gate, which is reported rather than guessed — an over-claimed
1046/// gate is the same defect as an over-claimed test.
1047fn gates_for(path: &str, commands: &[(String, String)]) -> Vec<String> {
1048    let mut out: BTreeSet<String> = BTreeSet::new();
1049    for (gate, runs) in commands {
1050        let names_path = runs.split_whitespace().any(|w| w == path);
1051        let drives_suite = path.starts_with(CASES_DIR) && runs.contains(SUITE_RUNNER);
1052        let drives_crate = path.starts_with(CRATE_TESTS_DIR) && runs.contains(CRATE_RUNNER);
1053        if names_path || drives_suite || drives_crate {
1054            out.insert(gate.clone());
1055        }
1056    }
1057    out.into_iter().collect()
1058}

1060/// Where `lib/` defines `mj_validate_<name>`, if anywhere.
1061///
1062/// The same scan `majordomus doctor` does, and the same one `scripts/generate-site-data`
1063/// does before it will build a doctrine page: a validator is a shell function, and the
1064/// question "is it there" is answered by reading the library rather than the filesystem.
1065fn validator_defined_in(root: &Path, name: &str) -> Option<String> {
1066    let needle = format!("mj_validate_{name}()");
1067    let dir = root.join("lib");
1068    // `min` rather than collect-sort-first: one pass, no sort site, and the same answer —
1069    // the least file name, deterministically, whatever order the directory was read in.
1070    std::fs::read_dir(&dir)
1071        .ok()?
1072        .filter_map(Result::ok)
1073        .filter(|e| e.path().extension().is_some_and(|x| x == "sh"))
1074        .filter(|e| {
1075            std::fs::read_to_string(e.path())
1076                .map(|t| t.lines().any(|l| l.trim_start().starts_with(&needle)))
1077                .unwrap_or(false)
1078        })
1079        .map(|e| format!("lib/{}", e.file_name().to_string_lossy()))
1080        .min()
1081}

1089        .any(|(g, runs)| g == gate && runs.split_whitespace().any(|w| w == path))

1092/// Map one test's [`ProofState`] onto the rule vocabulary. The two agree everywhere they
1093/// overlap; `Dangling` and `Unproven` are the rule-level states evidence has no word for,
1094/// because a claim naming a missing file is already `Unrunnable` to it.
1095fn rule_state_of(test: &TestProof) -> RuleState {
1096    if !test.present {
1097        return RuleState::Dangling;
1098    }
1099    // A gate is a mechanism, not a verdict: it is not joined to the ledger, so the
1100    // evidence vocabulary has nothing to say about it and reading its `Unrunnable` as a
1101    // defect would be wrong in the opposite direction from a fake green.
1102    if test.kind == ArtifactKind::Gate {
1103        return RuleState::Gated;
1104    }
1105    match test.state {
1106        ProofState::Proven => RuleState::Proven,
1107        ProofState::InputsUnchanged => RuleState::InputsUnchanged,
1108        ProofState::Stale => RuleState::Stale,
1109        ProofState::Failing => RuleState::Failing,
1110        ProofState::NotRun => RuleState::NotRun,
1111        ProofState::Unrunnable => RuleState::Unrunnable,
1112        ProofState::NoTest => RuleState::Unproven,
1113    }
1114}

1116/// Why a rule's state does not support the class it declares, or `None` when it does.
1117///
1118/// A blocking rule is the subject: it claims a gate refuses work that violates it. The
1119/// states that make that claim false are findings — `unproven`, `failing` and `unrunnable`.
1120/// `not run`, `gated`, `reviewed` and `stale` are not findings: they are weaker states, said
1121/// out loud and counted, which is a different thing from a defect (docs/DOCTRINE.md). They
1122/// are not proof either, and [`RulesVerdict`] is where the report says so. An advisory rule
1123/// claims nothing executable, so only a name that does not resolve is a finding — a dangling
1124/// path is a defect at any class, because it reads as proof and is not.
1125fn unsupported(class: Class, state: RuleState) -> Option<String> {
1126    match (class, state) {
1127        (_, RuleState::Dangling) => {
1128            Some("it names a path that is not in the tree, so it reads as proven and is not".into())
1129        }
1130        // Reviewed is deliberately not a finding: the rule says in its own front matter
1131        // that nothing executable can express it and why, which is a declaration a reader
1132        // can audit. What it is not is proof, and the report counts it apart.
1133        (Class::Blocking, RuleState::Unproven) => {
1134            Some("it is blocking and names neither a validator nor a test that proves it".into())
1135        }
1136        (Class::Blocking, RuleState::Failing) => {
1137            Some("it is blocking and the latest run of what proves it did not pass".into())
1138        }
1139        // Gated is deliberately not a finding: an executable check refuses violations, which
1140        // is exactly what a blocking rule claims. What it does not have is a recorded
1141        // verdict, which the state itself says and the report counts separately.
1142        (Class::Blocking, RuleState::Unrunnable) => Some(
1143            "it is blocking and names a path no runner drives, so nothing can ever record it"
1144                .into(),
1145        ),
1146        _ => None,
1147    }
1148}

1150/// Join every rule of the index with the tree and the ledger, and decide, per rule, what
1151/// the repository can honestly say.
1152///
1153/// The git comparison is done once per distinct recorded commit and shared by every rule
1154/// that recorded against it, as [`crate::evidence::report`] does — the two derivations read
1155/// the same ledger and must not disagree about what a commit's diff is.
1156/// ```
1157/// use majordomus_cli::rules::report;
1158/// use majordomus_cli::evidence::Ledger;
1159/// use majordomus_cli::synthetic::SyntheticRepository;
1160/// let repo = SyntheticRepository::small().unwrap();
1161/// let ledger = Ledger::load(repo.root()).unwrap();
1162/// let r = report(&repo.index().unwrap(), &ledger);
1163/// assert_eq!(r.coverage.rules, repo.shape.rules);
1164/// assert!(r.satisfied(), "the synthetic rules are advisory and owe no proof");
1165/// ```
1166pub fn report(index: &Index, ledger: &Ledger) -> RulesReport {
1167    let root = PathBuf::from(&index.repository.root);
1168    let git = crate::git::inspect(&root);
1169    let (head, working_tree) = match &git {
1170        crate::git::GitState::Available(i) => (i.head.clone(), i.working_tree.clone()),
1171        crate::git::GitState::Unavailable { .. } => (None, "unknown".to_string()),
1172    };
1173
1174    // Rules are judged at the working tree, through the one freshness function the evidence
1175    // report uses: one comparison per commit the ledger names, shared by every rule.
1176    let presented = Presented::WorkingTree;
1177    let mut comparisons: BTreeMap<String, Comparison> = BTreeMap::new();
1178    for e in &ledger.executions {
1179        comparisons
1180            .entry(e.commit.clone())
1181            .or_insert_with(|| freshness::compare(&root, &e.commit, &presented));
1182    }
1183    let gate_commands = gate_commands(&root);
1184
1185    let defs = definitions(index);
1186    // the converse of depends_on, which no rule declares
1187    let mut required_by: BTreeMap<String, Vec<String>> = BTreeMap::new();
1188    for d in &defs {
1189        for dep in &d.depends_on {
1190            required_by
1191                .entry(dep.clone())
1192                .or_default()
1193                .push(d.id.clone());
1194        }
1195    }
1196
1197    let mut rules = Vec::new();
1198    let mut states: BTreeMap<String, usize> = BTreeMap::new();
1199    let mut findings = Vec::new();
1200    let mut coverage = Coverage::default();
1201    let mut blocking_states = Vec::new();
1202
1203    for def in defs {
1204        let validator = def.enforcement.validator.as_ref().map(|v| {
1205            let defined_in = validator_defined_in(&root, v);
1206            ValidatorRef {
1207                name: v.clone(),
1208                function: format!("mj_validate_{v}"),
1209                present: defined_in.is_some(),
1210                defined_in,
1211            }
1212        });
1213
1214        let mut tests = Vec::new();
1215        for path in &def.enforcement.tests {
1216            let present = root.join(path).exists();
1217            let id = TestId::of(path);
1218            let gates = gates_for(path, &gate_commands);
1219            let kind = if id.is_some() {
1220                ArtifactKind::Case
1221            } else if gates
1222                .iter()
1223                .any(|g| gate_runs_exactly(g, path, &gate_commands))
1224            {
1225                ArtifactKind::Gate
1226            } else {
1227                ArtifactKind::Unknown
1228            };
1229            let execution = id
1230                .as_ref()
1231                .and_then(|t| ledger.latest(&t.as_string()))
1232                .cloned();
1233            let recorded = match (&id, &execution) {
1234                (None, _) => Recorded::Unrunnable,
1235                (Some(_), None) => Recorded::NotRun,
1236                (Some(_), Some(e)) => Recorded::Ran(e),
1237            };
1238            // what this rule names, and nothing else: the test and the rule's own definition
1239            let test_source = id.as_ref().map(TestId::source).unwrap_or_default();
1240            let inputs = [test_source.clone(), def.path.clone()];
1241            let test_moved = execution
1242                .as_ref()
1243                .is_some_and(|e| e.outcome.proves() && e.digest_matches(&root) == Some(false));
1244            let state = freshness::freshness(
1245                recorded,
1246                Some(&inputs),
1247                &test_source,
1248                test_moved,
1249                execution.as_ref().and_then(|e| comparisons.get(&e.commit)),
1250                TreeState::Clean,
1251            )
1252            .state;
1253            tests.push(TestProof {
1254                gates,
1255                kind,
1256                path: path.clone(),
1257                present,
1258                test: id.as_ref().map(TestId::as_string),
1259                state,
1260                meaning: state.meaning().to_string(),
1261                execution,
1262                reproduce: id.as_ref().map(TestId::reproduce),
1263            });
1264        }
1265
1266        // A failing test makes the rule failing, whatever its other tests say; otherwise the
1267        // rule's state is the weakest of the parts that can carry proof — and which parts
1268        // those are is the whole subtlety, learned by running this against the real corpus.
1269        //
1270        // `dangling` always dominates: a named path that is not in the tree is a lie
1271        // whatever else the rule names, because the rule reads as proven on its strength.
1272        //
1273        // The failure comes next because the declared order ranks `failing` above `not_run`:
1274        // a plain weakest-of over a rule whose tests are one fail and one skip (which reads
1275        // `not_run`) would read `not_run`, which is not a finding, and the counter-evidence
1276        // would be hidden behind an absence.
1277        //
1278        // An artifact no runner and no gate drives is different. It contributes nothing,
1279        // and that is not a defect in the rule: `project.web-surface-declared-once` names a
1280        // case, two gates and `scripts/generate-site-data`, which nothing runs as its own
1281        // command. Letting that one path drag the rule to `unrunnable` reported six rules
1282        // as unprovable while each had a perfectly good case — a false finding, which is
1283        // the failure this whole subsystem exists to refuse. So such a path is reported,
1284        // and judged only when it is all the rule has. `freshness::aggregate` is that rule,
1285        // written once for every aggregation of states.
1286        let mut state = if def.enforcement.mode == Mode::Reviewed {
1287            RuleState::Reviewed
1288        } else if def.enforcement.mode == Mode::Declarative {
1289            RuleState::Unproven
1290        } else if tests.iter().any(|t| !t.present) {
1291            RuleState::Dangling
1292        } else {
1293            let parts: Vec<(RuleState, bool)> = tests
1294                .iter()
1295                .map(|t| (rule_state_of(t), t.kind != ArtifactKind::Unknown))
1296                .collect();
1297            freshness::aggregate(&parts, RuleState::Failing).unwrap_or(RuleState::Unproven)
1298        };
1299        if let Some(v) = &validator {
1300            if !v.present {
1301                state = RuleState::Dangling;
1302            } else if tests.is_empty() {
1303                // a dispatched rule with a present validator and no case: the dispatcher
1304                // runs it, and nothing behavioural proves it does the right thing
1305                state = state.max(RuleState::NotRun);
1306            }
1307        }
1308
1309        let gates: Vec<String> = tests
1310            .iter()
1311            .flat_map(|t| t.gates.iter().cloned())
1312            .collect::<BTreeSet<_>>()
1313            .into_iter()
1314            .collect();
1315
1316        let reason = unsupported(def.class, state);
1317        let satisfied = reason.is_none();
1318        if let Some(reason) = reason {
1319            findings.push(Finding {
1320                rule: def.id.clone(),
1321                class: def.class,
1322                state,
1323                reason,
1324                reproduce: "scripts/ci/rule-proof-check".to_string(),
1325            });
1326        }
1327
1328        // the tallies, all counted from what was just derived
1329        coverage.rules += 1;
1330        match def.class {
1331            Class::Blocking => {
1332                coverage.blocking += 1;
1333                blocking_states.push(state);
1334            }
1335            Class::Advisory => coverage.advisory += 1,
1336            Class::Unknown => {}
1337        }
1338        if matches!(def.enforcement.mode, Mode::Dispatched | Mode::Gated) {
1339            coverage.named_proof += 1;
1340        }
1341        if state != RuleState::Dangling
1342            && matches!(def.enforcement.mode, Mode::Dispatched | Mode::Gated)
1343        {
1344            coverage.artifacts_present += 1;
1345        }
1346        if tests.iter().any(|t| t.test.is_some()) {
1347            coverage.runnable += 1;
1348        }
1349        if !tests.is_empty() && tests.iter().all(|t| t.execution.is_some()) {
1350            coverage.recorded += 1;
1351        }
1352        if state.passing() {
1353            coverage.passing += 1;
1354        }
1355        if state == RuleState::Gated {
1356            coverage.mechanism_only += 1;
1357        }
1358        if state == RuleState::Reviewed {
1359            coverage.review_only += 1;
1360        }
1361        if !gates.is_empty() {
1362            coverage.gated += 1;
1363        }
1364        if satisfied {
1365            coverage.satisfied += 1;
1366        }
1367        *states.entry(state.label().to_string()).or_insert(0) += 1;
1368
1369        rules.push(RuleProof {
1370            required_by: required_by.get(&def.id).cloned().unwrap_or_default(),
1371            validator,
1372            tests,
1373            gates,
1374            state,
1375            meaning: state.meaning().to_string(),
1376            satisfied,
1377            rule: def,
1378        });
1379    }
1380
1381    RulesReport {
1382        head,
1383        working_tree,
1384        verdict: RulesVerdict::of(blocking_states, !findings.is_empty()),
1385        rules,
1386        states,
1387        coverage,
1388        satisfied: findings.is_empty(),
1389        findings,
1390    }
1391}
