1//! The bar at the top of every screen: the site's, not a second one.
2//!
3//! majordomus.dev is the reference for how the product looks
4//! (`project.the-ui-wears-the-sites-navbar`), and its navbar is declared once, in
5//! `site/data/nav.toml`, and rendered once, by `site/templates/partials/navbar.html`. The UI
6//! does not draw a bar after that one. It wears that rendering: `scripts/ui navbar --write` cuts
7//! the bar out of the built site into `navbar.html` beside this file, which is compiled in, and
8//! `scripts/ui navbar` refuses a UI whose bar differs from the site's in an element, a box or a
9//! computed style, in both themes and at every declared width. The UI's stylesheet compiles the
10//! utilities the bar is written in from that same file (`share/ui/src/ui.css`), and the bar's
11//! two behaviours are the site's own scripts, served from `share/ui/` as byte copies
12//! (`scripts/cockpit-assets`).
13//!
14//! What the executable decides is what only it knows:
15//!
16//! - **the version** the badge names: the executable's, where the site names the release it
17//!   offers;
18//! - **the live surfaces**: an entry the site leads to a published snapshot of a surface this
19//!   process serves leads to the surface itself, and stops saying "snapshot".
20//!
21//! Every other link leads to the site, at its own address, because it is the site's page.
22
23/// The site's bar, as the site rendered it, with the version left to fill in.
24pub const FRAGMENT: &str = include_str!("navbar.html");
25
26/// What the fragment holds where the site's build named the release it offers.
27pub const VERSION_SLOT: &str = "@MJ_VERSION@";
28
29/// The attributes the site's bar marks an entry with when it leads to a surface's snapshot:
30/// the bar's own entry for the product, and an entry of a group's menu.
31const SURFACE_MARKS: [&str; 2] = ["data-nav-primary", "data-nav-snapshot"];
32
33/// The attribute of the note that says an entry leads to a snapshot.
34const SNAPSHOT_NOTE: &str = "data-nav-snapshot-note";
35
36/// The bar for a page of this process: `version` in the badge, and every entry the site leads
37/// to the snapshot of a surface in `live` (its id and the mount this process serves it at)
38/// leading to the mount instead.
39pub fn render(version: &str, live: &[(&str, &str)]) -> String {
40    relink(&FRAGMENT.replace(VERSION_SLOT, version), live)
41}
42
43/// Every `<a>` that names a surface of `live` by one of [`SURFACE_MARKS`] leads to its mount,
44/// without the snapshot note.
45fn relink(bar: &str, live: &[(&str, &str)]) -> String {
46    let mut out = String::with_capacity(bar.len());
47    let mut rest = bar;
48    while let Some(at) = rest.find("<a ") {
49        out.push_str(&rest[..at]);
50        rest = &rest[at..];
51        let Some(close) = rest.find('>') else { break };
52        let Some(end) = rest.find("</a>") else { break };
53        let (tag, inner) = (&rest[..=close], &rest[close + 1..end]);
54        match live.iter().find(|(id, _)| {
55            SURFACE_MARKS
56                .iter()
57                .any(|m| tag.contains(&format!("{m}=\"{id}\"")))
58        }) {
59            Some((_, mount)) => {
60                out.push_str(&with_href(tag, mount));
61                out.push_str(&without_note(inner));
62            }
63            None => out.push_str(&rest[..end]),
64        }
65        out.push_str("</a>");
66        rest = &rest[end + "</a>".len()..];
67    }
68    out.push_str(rest);
69    out
70}
71
72/// The tag with its `href` replaced.
73fn with_href(tag: &str, href: &str) -> String {
74    match tag.find("href=\"") {
75        Some(at) => {
76            let from = at + "href=\"".len();
77            let to = tag[from..].find('"').map_or(tag.len(), |n| from + n);
78            format!("{}{}{}", &tag[..from], href, &tag[to..])
79        }
80        None => tag.to_string(),
81    }
82}
83
84/// An entry's content without the note that says it is a snapshot, and the space before it.
85fn without_note(inner: &str) -> String {
86    match inner.find(&format!("<span {SNAPSHOT_NOTE}")) {
87        Some(at) => {
88            let end = inner[at..]
89                .find("</span>")
90                .map_or(inner.len(), |n| at + n + "</span>".len());
91            format!("{}{}", inner[..at].trim_end(), &inner[end..])
92        }
93        None => inner.to_string(),
94    }
95}

97#[cfg(test)]
98mod tests {
99    use super::*;
100
101    /// The compiled bar is the site's: its landmark, its brand, its four questions and its call
102    /// to action, and a slot for the version and nothing of the build that cut it.
103    #[test]
104    fn the_compiled_bar_is_the_sites() {
105        assert!(FRAGMENT.contains("aria-label=\"Main\""), "{FRAGMENT}");
106        assert!(FRAGMENT.contains("id=\"navbar-main\""));
107        assert!(FRAGMENT.contains("id=\"theme-toggle\""));
108        assert!(FRAGMENT.contains("data-site-search-open"));
109        assert_eq!(FRAGMENT.matches("<nav ").count(), 1);
110        assert!(
111            FRAGMENT.contains(VERSION_SLOT),
112            "the badge names a version to fill in"
113        );
114        assert!(
115            !FRAGMENT.contains("href=\"#"),
116            "a link to the site map of a site page leads to the site"
117        );
118        assert!(!FRAGMENT.contains("<script"), "the bar carries no script");
119    }
120
121    /// The version is the executable's, and an entry that leads to a surface this process
122    /// serves leads to it and no longer says "snapshot"; every other entry is left alone.
123    #[test]
124    fn a_live_surface_is_led_to_and_the_rest_is_the_sites() {
125        let bar = "<span data-release=\"@MJ_VERSION@\">v@MJ_VERSION@</span>\
126            <a href=\"https://site/ui/\" data-nav-primary=\"ui\" class=\"x\">Product UI</a>\
127            <a href=\"https://site/ui/\" data-nav-snapshot=\"ui\" class=\"y\">Product UI \
128            <span data-nav-snapshot-note class=\"z\">&middot; snapshot</span></a>\
129            <a href=\"https://site/docs/\">Docs</a>";
130        let out = relink(&bar.replace(VERSION_SLOT, "1.2.3"), &[("ui", "/ui")]);
131        assert!(out.contains(">v1.2.3<"), "{out}");
132        assert!(out.contains("<a href=\"/ui\" data-nav-primary=\"ui\" class=\"x\">Product UI</a>"));
133        assert!(
134            out.contains("<a href=\"/ui\" data-nav-snapshot=\"ui\" class=\"y\">Product UI</a>"),
135            "{out}"
136        );
137        assert!(!out.contains("snapshot</span>"), "{out}");
138        assert!(out.contains("<a href=\"https://site/docs/\">Docs</a>"));
139        assert_eq!(relink(bar, &[]), bar, "nothing live, nothing moved");
140    }
141
142    /// The bar the UI draws keeps every link of the compiled one but those it leads home.
143    #[test]
144    fn the_rendered_bar_keeps_every_other_link() {
145        let out = render("9.9.9", &[(super::super::SURFACE, super::super::PREFIX)]);
146        assert!(!out.contains(VERSION_SLOT));
147        let links = |s: &str| s.matches("<a ").count();
148        assert_eq!(links(&out), links(FRAGMENT));
149    }
150
151    /// A bar whose entry is cut short is passed on as it came, and a live entry with no
152    /// `href` keeps its tag while losing its snapshot note.
153    #[test]
154    fn a_cut_entry_is_left_as_it_came_and_one_without_a_link_keeps_its_tag() {
155        let live = [("ui", "/ui")];
156        let unclosed = "<b>bar</b><a data-nav-primary=\"ui\"";
157        assert_eq!(relink(unclosed, &live), unclosed, "no end to the tag");
158        let unended = "<b>bar</b><a data-nav-primary=\"ui\">Product UI";
159        assert_eq!(relink(unended, &live), unended, "no end to the entry");
160        let bare =
161            "<a data-nav-snapshot=\"ui\">UI <span data-nav-snapshot-note>snapshot</span></a>";
162        assert_eq!(
163            relink(bare, &live),
164            "<a data-nav-snapshot=\"ui\">UI</a>",
165            "no href to move"
166        );
167    }
