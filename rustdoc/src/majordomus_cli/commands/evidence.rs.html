1//! `majordomus evidence`: the claims matrix against the runs actually recorded for it.
2//!
3//! Every subcommand runs a capability of the `evidence` module through the one executor, so
4//! the answer a person reads here and the answer a client reads over MCP or HTTP are the
5//! same answer rendered twice, not two derivations that happen to agree today.
6//!
7//! The text rendering is deliberately blunt about the distinction the whole subsystem
8//! exists for: `proven` and `inputs unchanged` are printed differently, and the second one
9//! says what it means, because a column that showed both as a tick would be the badge whose
10//! derivation cannot be inspected.
11

22/// The exit code when `--check` finds a claim the evidence does not support.
23pub const EXIT_UNSUPPORTED: u8 = 10;
24
25/// Run `majordomus evidence`.
26pub fn run(args: EvidenceArgs) -> Result<u8> {
27    let app = App::load(&args.repo)?;
28    let ctx = &app.context;
29    let stdout = std::io::stdout();
30    let mut out = stdout.lock();
31
32    match args.command {
33        EvidenceCommand::Show {
34            state,
35            status,
36            findings,
37            check,
38            presented,
39            presented_tree,
40        } => {
41            // a presented commit is judged from the ledger it holds, and the runs the working
42            // ledger holds beside it can only withhold `proven`: the capability reads both,
43            // and an absent revision or tree is the null an optional input reads as absent
44            let mut input = json!({
45                "findings_only": findings,
46                "presented": presented,
47                "presented_tree": presented_tree,
48            });
49            if let Some(s) = &state {
50                input["state"] = json!(s);
51            }
52            if let Some(s) = &status {
53                input["status"] = json!(s);
54            }
55            let v = execute(ctx, &["evidence", "show"], input)?;
56            match args.format {
57                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
58                OutputFormat::Text => show_text(&mut out, &v)?,
59            }
60            if check {
61                let unsupported = !v["findings"].as_array().map(Vec::is_empty).unwrap_or(true);
62                // An empty finding list over an index that lost entries is not a pass; it
63                // is the absence of an answer. `--check` must not spell the two the same.
64                let partial = !v["subject"]["complete"].as_bool().unwrap_or(false);
65                if unsupported || partial {
66                    return Ok(EXIT_UNSUPPORTED);
67                }
68            }
69            Ok(0)
70        }
71
72        EvidenceCommand::Claim { id } => {
73            let v = execute(ctx, &["evidence", "claim"], json!({ "claim": id }))?;
74            match args.format {
75                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
76                OutputFormat::Text => claim_text(&mut out, &v)?,
77            }
78            Ok(0)
79        }
80
81        EvidenceCommand::Proves { id } => {
82            let v = execute(ctx, &["evidence", "proves"], json!({ "test": id }))?;
83            match args.format {
84                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
85                OutputFormat::Text => test_text(&mut out, &v)?,
86            }
87            Ok(0)
88        }
89
90        EvidenceCommand::Record {
91            suite,
92            crate_output,
93            origin,
94            provenance,
95            coverage,
96            ledger,
97            run_record,
98        } => {
99            let mut input = json!({});
100            if !provenance.is_empty() {
101                input["provenance"] = json!(provenance);
102            }
103            if let Some(p) = &coverage {
104                input["coverage"] = json!(p.to_string_lossy());
105            }
106            if let Some(l) = &ledger {
107                input["ledger"] = json!(l);
108            }
109            if let Some(p) = &run_record {
110                input["run_record"] = json!(p.to_string_lossy());
111            }
112            if let Some(p) = &suite {
113                input["suite"] = json!(p.to_string_lossy());
114            }
115            if let Some(p) = &crate_output {
116                input["crate_output"] = json!(p.to_string_lossy());
117            }
118            if let Some(o) = &origin {
119                input["origin"] = json!(o);
120            }
121            let v = execute(ctx, &["evidence", "record"], input)?;
122            match args.format {
123                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
124                OutputFormat::Text => {
125                    return record_text(&mut out, &v, run_record.as_deref()).map(|()| 0)
126                }
127            }
128            Ok(0)
129        }
130        EvidenceCommand::Stamp {
131            producer,
132            report,
133            exclude,
134            out: file,
135        } => {
136            let mut input = json!({ "exclude": exclude });
137            if let Some(p) = &producer {
138                input["producer"] = json!(p);
139            }
140            if let Some(p) = &report {
141                input["report"] = json!(p.to_string_lossy());
142            }
143            let v = execute(ctx, &["evidence", "stamp"], input)?;
144            write_stamp(&mut out, &v, file.as_deref(), args.format).map(|()| 0)
145        }
146    }
147}

149/// A measurement written to `file` when one is named, then answered on `out` in `format`;
150/// a file or a line that cannot be written is a transport error.
151fn write_stamp(
152    out: &mut impl Write,
153    v: &Value,
154    file: Option<&Path>,
155    format: OutputFormat,
156) -> Result<()> {
157    if let Some(f) = file {
158        // a bare file name's parent is the empty path, which create_dir_all accepts as is
159        f.parent()
160            .map_or(Ok(()), std::fs::create_dir_all)
161            .map_err(Error::Transport)?;
162        std::fs::write(f, format!("{}\n", pretty(v))).map_err(Error::Transport)?;
163    }
164    match format {
165        OutputFormat::Json => writeln!(out, "{}", pretty(v)).map_err(Error::Transport),
166        OutputFormat::Text => stamp_text(out, v, file),
167    }
168}

172/// A recording, one line per fact: what reached the ledger, against what, where, what the
173/// reports held that no claim can name yet, and what they named that this repository lacks.
174fn record_text(out: &mut impl Write, v: &Value, run_record: Option<&Path>) -> Result<()> {
175    writeln!(
176        out,
177        "recorded     {} execution(s), {} passing",
178        v["recorded"], v["passed"]
179    )
180    .map_err(Error::Transport)?;
181    writeln!(
182        out,
183        "against      {} ({})",
184        short(v["commit"].as_str().unwrap_or("?")),
185        v["working_tree"].as_str().unwrap_or("?")
186    )
187    .map_err(Error::Transport)?;
188    writeln!(out, "ledger       {}", v["ledger"].as_str().unwrap_or("?"))
189        .map_err(Error::Transport)?;
190    let r = &v["run_record"];
191    writeln!(out, "run          {}", r["id"].as_str().unwrap_or("?")).map_err(Error::Transport)?;
192    let measured: Vec<String> = r["provenance"]
193        .as_array()
194        .into_iter()
195        .flatten()
196        .map(|p| {
197            let producer = p["producer"].as_str().unwrap_or("?");
198            format!("{producer} ({})", p["working_tree"].as_str().unwrap_or("?"))
199        })
200        .collect();
201    if measured.is_empty() {
202        writeln!(out, "measured     nothing: the tree is recorded as unknown")
203    } else {
204        writeln!(out, "measured     {}", measured.join(", "))
205    }
206    .map_err(Error::Transport)?;
207    for a in r["absent"].as_array().into_iter().flatten() {
208        writeln!(
209            out,
210            "absent       {} ({})",
211            a["producer"].as_str().unwrap_or("?"),
212            a["reason"].as_str().unwrap_or("?")
213        )
214        .map_err(Error::Transport)?;
215    }
216    // what the reports held that no claim can name yet, listed, never hidden
217    for d in v["dropped"].as_array().into_iter().flatten() {
218        writeln!(
219            out,
220            "dropped      {} ({})",
221            d["what"].as_str().unwrap_or("?"),
222            d["reason"].as_str().unwrap_or("?")
223        )
224        .map_err(Error::Transport)?;
225    }
226    for u in v["unknown"].as_array().into_iter().flatten() {
227        writeln!(out, "unknown      {} (no such test here)", u).map_err(Error::Transport)?;
228    }
229    if let Some(p) = run_record {
230        writeln!(out, "run record   {}", p.display()).map_err(Error::Transport)?;
231    }
232    Ok(())
233}

235/// A measurement, one line per fact: what was stamped at which commit and tree, what was
236/// excluded, the report it names, and where it was written.
237fn stamp_text(out: &mut impl Write, v: &Value, written: Option<&Path>) -> Result<()> {
238    let commit = v["commit"].as_str().unwrap_or("?");
239    writeln!(
240        out,
241        "stamped      {} at {} ({})",
242        v["producer"].as_str().unwrap_or("a run"),
243        commit.get(..8).unwrap_or(commit),
244        v["working_tree"].as_str().unwrap_or("?")
245    )
246    .map_err(Error::Transport)?;
247    let excluded: Vec<&str> = v["excluded"]
248        .as_array()
249        .into_iter()
250        .flatten()
251        .filter_map(Value::as_str)
252        .collect();
253    if !excluded.is_empty() {
254        writeln!(out, "excluded     {}", excluded.join(", ")).map_err(Error::Transport)?;
255    }
256    if let Some(r) = v.get("report") {
257        let digest = r["digest"].as_str().unwrap_or("?");
258        writeln!(
259            out,
260            "report       {} {}",
261            r["path"].as_str().unwrap_or("?"),
262            digest.get(..19).unwrap_or(digest)
263        )
264        .map_err(Error::Transport)?;
265    }
266    if let Some(p) = written {
267        writeln!(out, "written      {}", p.display()).map_err(Error::Transport)?;
268    }
269    Ok(())
270}

272fn show_text(out: &mut std::io::StdoutLock<'_>, v: &Value) -> Result<()> {
273    let w =
274        |o: &mut std::io::StdoutLock<'_>, s: String| writeln!(o, "{s}").map_err(Error::Transport);
275
276    w(
277        out,
278        format!(
279            "head         {} ({})\n{}",
280            short(v["head"].as_str().unwrap_or("unknown")),
281            v["working_tree"].as_str().unwrap_or("?"),
282            judged_at(v).join("\n")
283        ),
284    )?;
285    let l = &v["ledger"];
286    w(
287        out,
288        format!(
289            "ledger       {} — {} execution(s){}",
290            l["path"].as_str().unwrap_or("?"),
291            l["executions"],
292            l["newest"]
293                .as_str()
294                .map(|n| format!(", newest {n}"))
295                .unwrap_or_default()
296        ),
297    )?;
298    if !l["present"].as_bool().unwrap_or(false) {
299        w(
300            out,
301            "             nothing has been recorded; every claim reads `not run`, which is true"
302                .to_string(),
303        )?;
304    }
305    w(out, String::new())?;
306
307    let sub = &v["subject"];
308    if !sub["complete"].as_bool().unwrap_or(true) {
309        w(
310            out,
311            format!(
312                "SUBJECT      INCOMPLETE — the index excluded {} file(s), so every count below \
313                 is over a smaller matrix than this repository has:",
314                sub["excluded"].as_array().map_or(0, Vec::len)
315            ),
316        )?;
317        for e in sub["excluded"].as_array().into_iter().flatten() {
318            w(out, format!("             {}", e.as_str().unwrap_or("?")))?;
319        }
320        w(out, String::new())?;
321    } else {
322        w(
323            out,
324            format!("subject      {} claim(s), whole", sub["examined"]),
325        )?;
326        w(out, String::new())?;
327    }
328
329    if let Some(t) = v["totals"].as_object() {
330        for (state, count) in t {
331            w(out, format!("{state:<18} {count}"))?;
332        }
333        w(out, String::new())?;
334    }
335
336    for c in v["claims"].as_array().into_iter().flatten() {
337        let state = c["state"].as_str().unwrap_or("?");
338        w(
339            out,
340            format!(
341                "{:<18} {:<12} {}{}",
342                state,
343                c["status"].as_str().unwrap_or("?"),
344                c["id"].as_str().unwrap_or("?"),
345                detail_line(c)
346            ),
347        )?;
348        if let Some(e) = c["execution"].as_object() {
349            w(
350                out,
351                format!(
352                    "                   {} at {} · {}s · {} · {}",
353                    e["outcome"].as_str().unwrap_or("?"),
354                    short(e["commit"].as_str().unwrap_or("?")),
355                    e["seconds"],
356                    e["origin"].as_str().unwrap_or("?"),
357                    e["at"].as_str().unwrap_or("?")
358                ),
359            )?;
360        }
361        for p in c["changed"].as_array().into_iter().flatten() {
362            w(
363                out,
364                format!(
365                    "                   changed since: {}",
366                    p.as_str().unwrap_or("?")
367                ),
368            )?;
369        }
370    }
371
372    let findings = v["findings"].as_array().cloned().unwrap_or_default();
373    if !findings.is_empty() {
374        w(out, String::new())?;
375        w(
376            out,
377            format!(
378                "{} claim(s) declare a guarantee the evidence does not support:",
379                findings.len()
380            ),
381        )?;
382        for f in &findings {
383            w(
384                out,
385                format!(
386                    "  {:<40} {}",
387                    f["claim"].as_str().unwrap_or("?"),
388                    f["reason"].as_str().unwrap_or("?")
389                ),
390            )?;
391            if let Some(r) = f["reproduce"].as_str() {
392                w(out, format!("  {:<40} reproduce: {r}", ""))?;
393            }
394        }
395    }
396    Ok(())
397}

399fn claim_text(out: &mut std::io::StdoutLock<'_>, v: &Value) -> Result<()> {
400    let w =
401        |o: &mut std::io::StdoutLock<'_>, s: String| writeln!(o, "{s}").map_err(Error::Transport);
402    w(
403        out,
404        format!("claim        {}", v["id"].as_str().unwrap_or("?")),
405    )?;
406    w(
407        out,
408        format!("             {}", v["claim"].as_str().unwrap_or("?")),
409    )?;
410    w(
411        out,
412        format!("status       {}", v["status"].as_str().unwrap_or("?")),
413    )?;
414    w(
415        out,
416        format!("state        {}", v["state"].as_str().unwrap_or("?")),
417    )?;
418    w(
419        out,
420        format!("             {}", v["meaning"].as_str().unwrap_or("")),
421    )?;
422    for (label, key) in [
423        ("source", "source"),
424        ("implementation", "implementation"),
425        ("test", "test_path"),
426    ] {
427        if let Some(s) = v[key].as_str() {
428            w(out, format!("{label:<12} {s}"))?;
429        }
430    }
431    if let Some(e) = v["execution"].as_object() {
432        w(out, String::new())?;
433        w(
434            out,
435            format!("execution    {}", e["outcome"].as_str().unwrap_or("?")),
436        )?;
437        w(
438            out,
439            format!("commit       {}", e["commit"].as_str().unwrap_or("?")),
440        )?;
441        w(
442            out,
443            format!("tree         {}", e["working_tree"].as_str().unwrap_or("?")),
444        )?;
445        w(out, format!("duration     {}s", e["seconds"]))?;
446        w(
447            out,
448            format!(
449                "recorded     {} ({})",
450                e["at"].as_str().unwrap_or("?"),
451                e["origin"].as_str().unwrap_or("?")
452            ),
453        )?;
454        w(
455            out,
456            format!("digest       {}", e["digest"].as_str().unwrap_or("?")),
457        )?;
458    }
459    for p in v["changed"].as_array().into_iter().flatten() {
460        w(out, format!("changed      {}", p.as_str().unwrap_or("?")))?;
461    }
462    let also: Vec<&str> = v["also_proves"]
463        .as_array()
464        .into_iter()
465        .flatten()
466        .filter_map(Value::as_str)
467        .collect();
468    if !also.is_empty() {
469        w(out, String::new())?;
470        w(
471            out,
472            format!("the same test also proves {} claim(s):", also.len()),
473        )?;
474        for a in also {
475            w(out, format!("  {a}"))?;
476        }
477    }
478    if let Some(r) = v["reproduce"].as_str() {
479        w(out, String::new())?;
480        w(out, format!("reproduce    {r}"))?;
481    }
482    Ok(())
483}

485fn test_text(out: &mut std::io::StdoutLock<'_>, v: &Value) -> Result<()> {
486    let w =
487        |o: &mut std::io::StdoutLock<'_>, s: String| writeln!(o, "{s}").map_err(Error::Transport);
488    w(
489        out,
490        format!("test         {}", v["test"].as_str().unwrap_or("?")),
491    )?;
492    w(
493        out,
494        format!("runner       {}", v["runner"].as_str().unwrap_or("?")),
495    )?;
496    w(
497        out,
498        format!(
499            "source       {}{}",
500            v["source"].as_str().unwrap_or("?"),
501            if v["present"].as_bool().unwrap_or(false) {
502                ""
503            } else {
504                "  (not in this checkout)"
505            }
506        ),
507    )?;
508    if let Some(e) = v["execution"].as_object() {
509        w(
510            out,
511            format!(
512                "latest       {} at {} · {}s · {}",
513                e["outcome"].as_str().unwrap_or("?"),
514                short(e["commit"].as_str().unwrap_or("?")),
515                e["seconds"],
516                e["at"].as_str().unwrap_or("?")
517            ),
518        )?;
519        match v["digest_matches"].as_bool() {
520            Some(true) => w(out, "digest       matches the source as recorded".into())?,
521            Some(false) => w(
522                out,
523                "digest       DOES NOT match: the test has changed since this run".into(),
524            )?,
525            None => {}
526        }
527    } else {
528        w(
529            out,
530            "latest       no run of this test has ever been recorded".into(),
531        )?;
532    }
533    w(
534        out,
535        format!("reproduce    {}", v["reproduce"].as_str().unwrap_or("?")),
536    )?;
537
538    let proves = v["proves"].as_array().cloned().unwrap_or_default();
539    w(out, String::new())?;
540    w(out, format!("proves {} claim(s):", proves.len()))?;
541    for c in &proves {
542        w(
543            out,
544            format!(
545                "  {:<18} {:<12} {}",
546                c["state"].as_str().unwrap_or("?"),
547                c["status"].as_str().unwrap_or("?"),
548                c["id"].as_str().unwrap_or("?")
549            ),
550        )?;
551    }
552    Ok(())
553}

555/// Why a claim's state is what it is, as the line under the claim that says so: empty when
556/// the state alone says it, and otherwise the detail on a line of its own.
557fn detail_line(c: &Value) -> String {
558    c["detail"]
559        .as_str()
560        .map(|d| format!("\n                   {d}"))
561        .unwrap_or_default()
562}

564/// The line naming what the verdicts below were judged at, and — for a presented commit whose
565/// checkout holds runs its own ledger does not — the line saying what those runs are for.
566fn judged_at(v: &Value) -> Vec<String> {
567    let p = &v["presented"];
568    let tree = p["tree"].as_str().unwrap_or("unknown");
569    let mut lines = vec![match p["revision"].as_str() {
570        None | Some("working_tree") => format!(
571            "judged at the working tree (HEAD {}, {tree})",
572            twelve(v["head"].as_str().unwrap_or("unknown"))
573        ),
574        Some(commit) => format!("judged at {} as committed ({tree})", twelve(commit)),
575    }];
576    let held: Vec<&str> = p["uncommitted"]
577        .as_array()
578        .into_iter()
579        .flatten()
580        .filter_map(Value::as_str)
581        .collect();
582    if !held.is_empty() {
583        lines.push(format!(
584            "             the working ledger holds executions this commit does not ({}), read \
585             only to withhold `proven`",
586            held.join(", ")
587        ));
588    }
589    lines
590}

592/// The first twelve characters of a commit: the spelling the judged line and every detail
593/// sentence use, so a reader can match one against the other.
594fn twelve(commit: &str) -> String {
595    commit.chars().take(12).collect()
596}

610/// Run one of this module's capabilities by the command line it declares, so that the
611/// command line cannot reach a capability it does not claim.
612fn execute(ctx: &crate::capability::Context, path: &[&str], input: Value) -> Result<Value> {
613    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
614    let id = ctx
615        .registry
616        .by_cli(&words)
617        .map(|c| c.id.as_str())
618        .ok_or_else(|| Error::Protocol {
619            reason: format!(
620                "no capability is exposed as `majordomus {}`",
621                path.join(" ")
622            ),
623        })?;
624    ctx.execute(id, input).map_err(map)
625}
626
627fn map(e: CapabilityError) -> Error {
628    match e {
629        CapabilityError::InvalidInput(reason) | CapabilityError::Refused(reason) => {
630            Error::Protocol { reason }
631        }
632        other => Error::Protocol {
633            reason: other.to_string(),
634        },
635    }
636}

642    /// The first line says what was judged, and a presented commit whose checkout holds runs
643    /// its own ledger does not says what those runs are read for.
644    #[test]
645    fn the_report_says_what_it_was_judged_at() {
646        let here = json!({
647            "head": "abcdef0123456789abcdef0123456789abcdef01",
648            "presented": { "revision": "working_tree", "tree": "dirty" }
649        });
650        assert_eq!(
651            judged_at(&here),
652            ["judged at the working tree (HEAD abcdef012345, dirty)"]
653        );
654
655        let commit = json!({
656            "presented": {
657                "revision": "0123456789abcdef0123456789abcdef01234567",
658                "tree": "clean",
659                "uncommitted": ["suite:01_alpha", "suite:02_beta"]
660            }
661        });
662        let lines = judged_at(&commit);
663        assert_eq!(lines[0], "judged at 0123456789ab as committed (clean)");

723    /// A writer that takes whole lines until it holds `lines` of them, then refuses every
724    /// write, and counts the writes it was asked for after its first refusal.
725    struct ClosesAfter {
726        lines: usize,
727        held: Vec<u8>,
728        refused: usize,
729    }
730
731    impl Write for ClosesAfter {
732        fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
733            if self.held.iter().filter(|&&b| b == b'\n').count() >= self.lines {
734                self.refused += 1;
735                return Err(std::io::Error::new(
736                    std::io::ErrorKind::BrokenPipe,
737                    "the reader went away",
738                ));
739            }
740            self.held.extend_from_slice(bytes);
741            Ok(bytes.len())
742        }
743
744        fn flush(&mut self) -> std::io::Result<()> {
745            Ok(())
746        }
747    }

752    fn a_recording_whose_output_cannot_be_written_says_so_at_the_first_line() {

763            lines: usize::MAX,
764            held: Vec::new(),
765            refused: 0,
