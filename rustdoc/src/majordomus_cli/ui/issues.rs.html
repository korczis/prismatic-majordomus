1//! Issues: every issue of the plan, narrowed by what the reader picks — a status, a
2//! milestone, a priority, text — each a link to its inspector, its milestone a link to the
3//! milestone's, and what blocks it a link to that issue.
4//!
5//! Read from `plan.issues`, `plan.status` and `plan.roadmap`, all of which answer in
6//! milliseconds, so the screen is one paint. Every control is a link or a GET form: the
7//! status tabs, the milestone and priority choices, the search. A value a link carries that is
8//! not one of the plan's own words is dropped, never echoed.
9

21/// The route of the screen.

24const ISSUES: Source = Source::of("plan.issues");

28/// The query parameters this screen reads.
29const KEYS: &[&str] = &["status", "milestone", "priority", "q"];
30
31/// What the reader picked.
32#[derive(Debug, Clone, Default, PartialEq, Eq)]
33pub struct Pick {
34    /// A status, as the plan writes it (`READY`).
35    pub status: Option<String>,
36    /// A milestone id.
37    pub milestone: Option<String>,
38    /// A priority (`p0`).
39    pub priority: Option<String>,
40    /// Text an id, title or objective must contain.
41    pub q: Option<String>,
42}
43
44/// A word of the plan's vocabulary: letters, digits, `-` and `_`, as its statuses,
45/// priorities and milestone ids are.
46fn is_word(w: &str) -> bool {
47    !w.is_empty()
48        && w.len() <= 64
49        && w.bytes()
50            .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_')
51}
52
53impl Pick {
54    /// The pick a query names; a value that is not a word of the plan is dropped.
55    pub fn from_query(query: &[(String, String)]) -> Self {
56        let get = |k: &str| {
57            query
58                .iter()
59                .find(|(key, _)| key == k)
60                .map(|(_, v)| v.trim().to_string())
61                .filter(|v| !v.is_empty())
62        };
63        Pick {
64            status: get("status").filter(|w| is_word(w)),
65            milestone: get("milestone").filter(|w| is_word(w)),
66            priority: get("priority").filter(|w| is_word(w)),
67            q: get("q"),
68        }
69    }
70
71    fn get(&self, key: &str) -> Option<String> {
72        match key {
73            "status" => self.status.clone(),
74            "milestone" => self.milestone.clone(),
75            "priority" => self.priority.clone(),
76            "q" => self.q.clone(),
77            _ => None,
78        }
79    }
80
81    /// This pick with `key` set to `value`, or cleared, as a link to this screen.
82    fn href(&self, key: &str, value: Option<&str>) -> String {
83        let pairs: Vec<String> = KEYS
84            .iter()
85            .filter_map(|k| {
86                let v = if *k == key {
87                    value.map(str::to_string)
88                } else {
89                    self.get(k)
90                };
91                v.map(|v| format!("{k}={}", encode(&v)))
92            })
93            .collect();
94        if pairs.is_empty() {
95            ROUTE.to_string()
96        } else {
97            format!("{ROUTE}?{}", pairs.join("&"))
98        }
99    }
100
101    /// Does this issue pass the pick?
102    fn admits(&self, i: &Value) -> bool {
103        let is = |k: &str, want: &Option<String>| {
104            want.as_deref()
105                .is_none_or(|w| text(i, k).eq_ignore_ascii_case(w))
106        };
107        let found = self.q.as_deref().is_none_or(|q| {
108            let q = q.to_lowercase();
109            ["id", "title", "objective", "slug"]
110                .iter()
111                .any(|k| text(i, k).to_lowercase().contains(&q))
112        });
113        is("status", &self.status)
114            && is("milestone", &self.milestone)
115            && is("priority", &self.priority)
116            && found
117    }
118}
119
120fn encode(v: &str) -> String {
121    v.bytes()
122        .map(|b| match b {
123            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
124                (b as char).to_string()
125            }
126            other => format!("%{other:02X}"),
127        })
128        .collect()
129}
130
131fn strings(v: &Value, k: &str) -> Vec<String> {
132    v.get(k)
133        .and_then(Value::as_array)
134        .map(|a| {
135            a.iter()
136                .filter_map(Value::as_str)
137                .map(str::to_string)
138                .collect()
139        })
140        .unwrap_or_default()
141}
142
143fn chip(href: String, label: String) -> El {
144    el("a")
145        .class("mj-ui-chip mj-ui-chip--link")
146        .attr("href", href)
147        .child(el("code").class("mj-ui-chip-value").text(label))
148}
149
150fn issue_href(id: &str) -> String {
151    format!("{}?issue={id}#issue", super::prs::ROUTE)
152}
153
154/// The values of `key` the issues hold, each with how many hold it, in the plan's own words.
155fn tally(issues: &[Value], key: &str) -> Vec<(String, u64)> {
156    let mut t: BTreeMap<String, u64> = BTreeMap::new();
157    for i in issues {
158        let w = text(i, key);
159        if !w.is_empty() {
160            *t.entry(w).or_default() += 1;
161        }
162    }
163    t.into_iter().collect()
164}

166/// The Issues screen.
167pub fn screen(ctx: &Context, query: &[(String, String)]) -> View {
168    let pick = Pick::from_query(query);
169    let answers = ask_all(ctx, &[ISSUES, PLAN, ROADMAP]);
170    let (issues, plan, roadmap) = (&answers[&ISSUES], &answers[&PLAN], &answers[&ROADMAP]);
171    let top = hero(
172        "list",
173        "cyan",
174        "Issues",
175        "Every piece of work the plan holds, and where each stands.",
176        "Narrow the plan by status, milestone, priority or text; every issue opens its inspector, every milestone its own.",
177        vec![
178            kpi("Issues", &plan.read("/counts/total")),
179            kpi("Ready", &plan.read("/counts/by_status/READY")),
180            kpi("Blocked", &plan.read("/counts/by_status/BLOCKED")),
181            kpi("Done", &plan.read("/counts/by_status/DONE")),
182        ],
183    );
184    View::new(
185        "Issues",
186        el("div").class("mj-ui-stack").child(top).child(
187            el("div")
188                .class("mj-ui-grid")
189                .child(wide(browser(issues, roadmap, &pick))),
190        ),
191    )
192}

194/// The browser: the status tabs, the filters, the issues the pick admits.
195pub fn browser(issues: &Answer, roadmap: &Answer, pick: &Pick) -> El {
196    let body = unanswered(issues).unwrap_or_else(|| {
197        let all = list(issues, "/issues");
198        let shown: Vec<&Value> = all.iter().filter(|i| pick.admits(i)).collect();
199        let tab = |label: String, href: String, current: bool| {
200            el("a")
201                .class("mj-ui-tab")
202                .attr("href", href)
203                .attr_if("aria-current", current.then_some("page"))
204                .text(label)
205        };
206        let tabs = el("nav")
207            .class("mj-ui-tabs")
208            .attr("aria-label", "Which issues")
209            .child(tab(
210                format!("All {}", all.len()),
211                pick.href("status", None),
212                pick.status.is_none(),
213            ))
214            .children(tally(&all, "status").into_iter().map(|(s, n)| {
215                let current = pick
216                    .status
217                    .as_deref()
218                    .is_some_and(|p| p.eq_ignore_ascii_case(&s));
219                tab(
220                    format!("{} {n}", s.to_lowercase()),
221                    pick.href("status", Some(&s)),
222                    current,
223                )
224            }));
225        let milestones: Vec<(String, String)> = list(roadmap, "/milestones")
226            .iter()
227            .map(|m| (text(m, "id"), text(m, "title")))
228            .collect();
229        let select =
230            |name: &str, label: &str, options: Vec<(String, String)>, picked: &Option<String>| {
231                el("label")
232                    .class("mj-ui-line")
233                    .child(el("span").class("mj-ui-dim").text(format!("{label} ")))
234                    .child(
235                        el("select")
236                            .attr("name", name)
237                            .child(el("option").attr("value", "").text("any"))
238                            .children(options.into_iter().map(|(value, shown)| {
239                                el("option")
240                                    .attr_if(
241                                        "selected",
242                                        (picked.as_deref() == Some(value.as_str()))
243                                            .then_some("selected"),
244                                    )
245                                    .attr("value", value)
246                                    .text(shown)
247                            })),
248                    )
249            };
250        let form = el("form")
251            .class("mj-ui-search")
252            .attr("role", "search")
253            .attr("method", "get")
254            .attr("action", ROUTE)
255            .child(
256                el("label")
257                    .class("mj-ui-sr-only")
258                    .attr("for", "mj-ui-issues-q")
259                    .text("Search issues"),
260            )
261            .child(
262                el("input")
263                    .attr("id", "mj-ui-issues-q")
264                    .attr("type", "search")
265                    .attr("name", "q")
266                    .attr("placeholder", "Id, title or objective")
267                    .attr("value", pick.q.clone().unwrap_or_default()),
268            )
269            .children(pick.status.clone().map(|s| {
270                el("input")
271                    .attr("type", "hidden")
272                    .attr("name", "status")
273                    .attr("value", s)
274            }))
275            .child(select(
276                "milestone",
277                "Milestone",
278                milestones
279                    .iter()
280                    .map(|(id, t)| (id.clone(), format!("{id} {t}")))
281                    .collect(),
282                &pick.milestone,
283            ))
284            .child(select(
285                "priority",
286                "Priority",
287                tally(&all, "priority")
288                    .into_iter()
289                    .map(|(p, n)| (p.clone(), format!("{p} ({n})")))
290                    .collect(),
291                &pick.priority,
292            ))
293            .child(
294                el("button")
295                    .attr("type", "submit")
296                    .class("mj-ui-more")
297                    .text("Apply"),
298            )
299            .when(pick != &Pick::default(), |f| {
300                f.child(
301                    el("a")
302                        .class("mj-ui-more")
303                        .attr("href", ROUTE)
304                        .text("Clear"),
305                )
306            });
307        let rows = shown
308            .iter()
309            .map(|i| {
310                let id = text(i, "id");
311                let milestone = text(i, "milestone");
312                let blocked = strings(i, "blocked_by");
313                el("tr")
314                    .child(
315                        el("td").child(
316                            el("a")
317                                .class("mj-ui-row-link")
318                                .attr("href", issue_href(&id))
319                                .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))
320                                .child(el("span").class("mj-ui-cell-title").text(text(i, "title"))),
321                        ),
322                    )
323                    .child(el("td").child(state(i, "status")))
324                    .child(el("td").child(if milestone.is_empty() {
325                        el("span").class("mj-ui-dim").text("none")
326                    } else {
327                        chip(super::milestone::href(&milestone), milestone)
328                    }))
329                    .child(el("td").class("mj-ui-mono").text(text(i, "priority")))
330                    .child(el("td").child(if blocked.is_empty() {
331                        el("span").class("mj-ui-dim").text("nothing")
332                    } else {
333                        el("span")
334                            .class("mj-ui-badges")
335                            .children(blocked.iter().map(|b| chip(issue_href(b), b.clone())))
336                    }))
337                    .child(el("td").class("mj-ui-cell-meter").child(meter(
338                        i.get("evidence_have").and_then(Value::as_u64),
339                        i.get("evidence_need").and_then(Value::as_u64),
340                    )))
341            })
342            .collect::<Vec<_>>();
343        el("div")
344            .class("mj-ui-stack")
345            .child(tabs)
346            .child(form)
347            .child(table(
348                &[
349                    "Issue",
350                    "Status",
351                    "Milestone",
352                    "Priority",
353                    "Blocked by",
354                    "Evidence",
355                ],
356                rows,
357                "No issue of the plan matches.",
358            ))
359            .child(el("p").class("mj-ui-line").text(format!(
360                "{} of {} issues shown",
361                shown.len(),
362                all.len()
363            )))
364    });
365    panel(
366        "Issues",
367        "Every issue of the plan, as the plan records it.",
368        body,
369        &[issues, roadmap],
370    )
371}

379    fn answer(capability: &'static str, value: Value) -> Answer {
380        Answer {
381            source: Source::of(capability),
382            route: None,
383            exposure: Exposure::default(),
384            value: Ok(value),
385            took: Duration::from_millis(1),
386            standing: crate::ui::source::Standing::NOW,
387        }
388    }
389
390    fn q(pairs: &[(&str, &str)]) -> Vec<(String, String)> {
391        pairs
392            .iter()
393            .map(|(k, v)| (k.to_string(), v.to_string()))
394            .collect()
395    }
396
397    fn plan() -> (Answer, Answer) {
398        let i = |id: &str, status: &str, m: &str, p: &str, blocked: &[&str]| {
399            serde_json::json!({ "id": id, "status": status, "milestone": m, "priority": p, "title": format!("issue {id}"),
400                "objective": format!("do {id}"), "blocked_by": blocked, "evidence_have": 0, "evidence_need": 1 })
401        };
402        (
403            answer(
404                "plan.issues",
405                serde_json::json!({ "issues": [
406                    i("I0001", "DONE", "M1", "p0", &[]),
407                    i("I0002", "READY", "M1", "p1", &[]),
408                    i("I0003", "BLOCKED", "M2", "p1", &["I0002"]),
409                ]}),
410            ),
411            answer(
412                "plan.roadmap",
413                serde_json::json!({ "milestones": [
414                { "id": "M1", "title": "first" }, { "id": "M2", "title": "second" } ]}),
415            ),
416        )
417    }

419    #[test]
420    fn the_tabs_are_the_plans_statuses_and_each_narrows_the_issues() {
421        let (issues, roadmap) = plan();
422        let all = browser(&issues, &roadmap, &Pick::default()).render();
423        for want in [
424            "All 3",
425            "blocked 1",
426            "done 1",
427            "ready 1",
428            "3 of 3 issues shown",
429        ] {
430            assert!(all.contains(want), "no {want}: {all}");
431        }
432        let blocked = browser(
433            &issues,
434            &roadmap,
435            &Pick::from_query(&q(&[("status", "BLOCKED")])),
436        )
437        .render();
438        assert!(blocked.contains("1 of 3 issues shown") && blocked.contains("I0003"));
439        assert!(
440            blocked.contains("/ui/prs?issue=I0002#issue"),
441            "what blocks it is a link"
442        );
443        assert!(
444            blocked.contains("/ui/roadmap?milestone=M2#milestone"),
445            "its milestone is a link"
446        );
447    }

482    /// An issue filed under no milestone says none, and a key the pick does not read gives
483    /// nothing back, so a link never carries a parameter it did not mean.
484    #[test]
485    fn an_issue_without_a_milestone_says_none() {
486        let issues = answer(
487            "plan.issues",
488            serde_json::json!({ "issues": [{ "id": "I0042", "title": "loose", "status": "READY", "priority": "p3" }] }),
489        );
490        let roadmap = answer("plan.roadmap", serde_json::json!({ "milestones": [] }));
491        let html = browser(&issues, &roadmap, &Pick::default()).render();
492        assert!(html.contains("I0042") && html.contains(">none<"), "{html}");
493        assert_eq!(Pick::default().get("nothing-of-ours"), None);
494    }
