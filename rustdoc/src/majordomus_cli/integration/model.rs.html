1//! The canonical pull-request state and the one disposition model.
2//!
3//! Everything a surface says about a pull request is a field of [`PullRequestAssessment`], and every
4//! surface — the command line, the HTTP route, the MCP tool, the Cockpit — renders that one
5//! value. Nothing downstream re-derives a verdict from raw forge data.
6

10/// What the forge said about one open pull request, as observed. Plain data: the forge
11/// adapter fills it, a test builds it by hand.
12#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
13pub struct PullRequestObservation {
14    /// The number.
15    pub number: u64,
16    /// The title.
17    pub title: String,
18    /// The author's login.
19    pub author: String,
20    /// The branch it asks to merge.
21    pub head_ref: String,
22    /// The commit that branch pointed at when observed.
23    pub head_sha: String,
24    /// The branch it asks to merge into.
25    pub base_ref: String,
26    /// A draft is not asking to be merged.
27    pub draft: bool,
28    /// The label names.
29    #[serde(default)]
30    pub labels: Vec<String>,
31    /// When it was opened, RFC 3339.
32    pub created_at: String,
33    /// When it last changed, RFC 3339.
34    pub updated_at: String,
35    /// The body, for dependency declarations; never rendered.
36    #[serde(default)]
37    pub body: String,
38    /// Every check reported on the head commit.
39    #[serde(default)]
40    pub checks: Vec<CheckObservation>,
41    /// The forge's review decision, verbatim (`APPROVED`, `CHANGES_REQUESTED`,
42    /// `REVIEW_REQUIRED`) or empty when the repository asks for none.
43    #[serde(default)]
44    pub review_decision: String,
45    /// Whether the forge has auto-merge armed on it.
46    #[serde(default)]
47    pub auto_merge: bool,
48    /// Whether the head lives in another repository (a fork): its branch cannot be
49    /// refreshed from here.
50    #[serde(default)]
51    pub cross_repository: bool,
52    /// Each reviewer's latest review, with the commit it was given on. A review is about one
53    /// commit: an approval of another commit is not an approval of the head.
54    #[serde(default)]
55    pub latest_reviews: Vec<ReviewObservation>,
56    /// Who has been asked to review and has not yet: logins, or team slugs.
57    #[serde(default)]
58    pub review_requests: Vec<String>,
59}

160/// The state of the repository's *required* checks on one head, which is the only check

172    Pending,
173    /// A required check failed.
174    Failed,
175    /// A required check has not reported on this head at all.
176    Missing,
177    /// The repository's required checks could not be read.
178    Unknown,

202/// What the pull request's head is to the current master, decided locally by git with
203/// this repository's own merge drivers — never by the forge's `mergeable`, which cannot
204/// run the derived-file driver and calls nearly every pull request here conflicting.
205#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
206#[serde(rename_all = "snake_case", tag = "kind")]
207pub enum RelationToMaster {
208    /// The head is an ancestor of master: every commit already landed.
209    Contained,
210    /// Merging the head into master changes nothing: its patch is already fully there.
211    Superseded,
212    /// The merge would conflict or change only derived output, but every commit the head has
213    /// and master lacks is on master already as an equal patch (`git cherry` marks each `-`):
214    /// the change landed, and master moved on past it. Never a partial match, and never a
215    /// head that carries a merge commit of its own, whose resolution has no patch to compare.
216    PatchIdsUpstream {
217        /// How many commits, every one of them already on master.
218        commits: u64,
219    },
220    /// Merging changes only derived artifacts: the authored change is already on master,
221    /// and what differs is output the generators rewrite.
222    DerivedOnly {
223        /// The derived paths that would change.
224        paths: Vec<String>,
225    },
226    /// The head already contains master and merges cleanly.
227    UpToDate {
228        /// Authored (non-derived) paths the merge changes.
229        authored: Vec<String>,
230    },
231    /// The head does not contain master, but the merge is clean with the derived driver.
232    Behind {
233        /// Commits on master the head does not have.
234        behind: u64,
235        /// Authored paths the merge changes.
236        authored: Vec<String>,
237    },
238    /// Merging conflicts on authored paths: a person must resolve it.
239    Conflicting {
240        /// The conflicting authored paths.
241        paths: Vec<String>,
242    },
243    /// Git could not answer (the head is not fetched, or git failed); the reason.
244    Unknown {
245        /// Why.
246        reason: String,
247    },
248}

250/// Explicit ordering between pull requests, as declared.

252pub struct PullRequestDependency {

254    pub number: u64,
255    /// How it is known.

268pub enum DependencyState {
269    /// Still open.
270    #[default]

273    Merged,

307/// The one classification of an open pull request. Every value carries a machine-readable
308/// reason in [`PullRequestAssessment::reasons`]; there is no `mergeable: bool`.
309#[derive(
310    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
311)]
312#[serde(rename_all = "snake_case")]
313pub enum PullRequestDisposition {
314    /// Contains the current master, every required check passed on its head, nothing
315    /// blocks it: the executor may merge it.
316    Ready,
317    /// Merges cleanly but does not contain master: the executor may bring master in (a
318    /// merge commit on the branch, never a rewrite), after which CI must run again.
319    NeedsRefresh,
320    /// Contains master; a required check has not finished on this head.
321    WaitingForChecks,
322    /// A required review is missing or changes were requested.
323    WaitingForReview,
324    /// A declared dependency has not landed.
325    WaitingForDependency,
326    /// A draft.
327    Draft,
328    /// A required check failed on this head.
329    NeedsRepair,
330    /// Conflicts on authored paths: a person resolves them.
331    Conflicting,
332    /// A label whose policy is to hold holds it, or the repository allows no merge commit,
333    /// which is the only way the executor merges.
334    Blocked,
335    /// The forge has auto-merge armed on it: the forge would merge it on its own, outside the
336    /// executor and against whatever master is then. Held until a person disarms it.
337    Unsafe,
338    /// Its head already landed, its merge changes nothing, or every one of its commits is on
339    /// master as an equal patch: strong evidence from git, eligible for closure under policy.
340    /// (Before 0.13 this was the word `superseded`.)
341    Redundant,
342    /// Its body, or the body of another pull request, declares that a successor replaces it,
343    /// and that successor landed: no longer open, its head contained in master. The successor
344    /// is [`PullRequestAssessment::superseded_by`]. Eligible for closure under policy.
345    Superseded,
346    /// Weak evidence: only derived artifacts would change, or its declared successor was closed
347    /// without its head landing. Surfaced for a person; never closed automatically.
348    PossiblyRedundant,
349    /// Targets a branch other than the integration base.
350    OtherBase,
351    /// Something needed to decide could not be observed.
352    Unknown,
353    /// A person marked it obsolete with a label the label policy declares (owner decision
354    /// D3): the only evidence that reaches this word. Listed for a person by cleanup and
355    /// never closed automatically; age, shared paths or a similar title never make one.
356    Obsolete,
357}

391            PullRequestDisposition::WaitingForReview => "waiting_for_review",

393            PullRequestDisposition::Draft => "draft",

400            PullRequestDisposition::PossiblyRedundant => "possibly_redundant",
401            PullRequestDisposition::OtherBase => "other_base",
402            PullRequestDisposition::Unknown => "unknown",
403            PullRequestDisposition::Obsolete => "obsolete",
404        }
405    }
406
407    /// The queue a disposition belongs to on every surface.
408    pub fn lane(self) -> IntegrationLane {
409        match self {
410            PullRequestDisposition::Ready => IntegrationLane::Ready,
411            PullRequestDisposition::NeedsRefresh
412            | PullRequestDisposition::WaitingForChecks
413            | PullRequestDisposition::WaitingForReview
414            | PullRequestDisposition::WaitingForDependency => IntegrationLane::Waiting,
415            PullRequestDisposition::NeedsRepair | PullRequestDisposition::Conflicting => {
416                IntegrationLane::Repair
417            }
418            PullRequestDisposition::Redundant
419            | PullRequestDisposition::Superseded
420            | PullRequestDisposition::PossiblyRedundant
421            | PullRequestDisposition::Obsolete => IntegrationLane::Cleanup,
422            PullRequestDisposition::Draft
423            | PullRequestDisposition::Blocked
424            | PullRequestDisposition::Unsafe
425            | PullRequestDisposition::OtherBase
426            | PullRequestDisposition::Unknown => IntegrationLane::Held,
427        }
428    }

550/// The revisions a decision was taken against, and when the forge was observed. A decision is
551/// valid only while both revisions still hold: the executor compares them with what it
552/// re-reads before it acts.
553#[derive(Debug, Clone, Default, Eq, Serialize, Deserialize, JsonSchema)]

555    /// The master commit.
556    pub master_sha: String,

564/// Two are equal when they name the same revisions. The moment is when, not what: the same
565/// master and head observed a moment later is the same decision, which is what the executor's
566/// stale-decision comparison asks.
567impl PartialEq for EvaluatedAgainst {
568    fn eq(&self, other: &Self) -> bool {
569        self.master_sha == other.master_sha && self.head_sha == other.head_sha
570    }

573/// One question of the policy, in the order [`IntegrationGate::ALL`] asks them. The
574/// disposition is the first that fails; every one is answered.
575#[derive(
576    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
577)]
578#[serde(rename_all = "snake_case")]
579pub enum IntegrationGate {
580    /// It targets the integration base.
581    Base,
582    /// It is not a draft.
583    Draft,
584    /// It carries no label whose policy is to hold.
585    Label,
586    /// The forge has no auto-merge armed on it.
587    AutoMerge,
588    /// No successor is declared to replace it. Asked before the relation to master, because
589    /// a pull request whose successor landed usually conflicts with what the successor
590    /// brought, and is superseded rather than conflicting.
591    Supersession,
592    /// Its head is not on master already, and git could say that it merges cleanly.
593    RelationToMaster,
594    /// The repository allows a merge commit, the only way the executor merges. Asked after
595    /// the relation, so work already on master is still `redundant` and may be closed.
596    MergeMethod,
597    /// Every declared dependency has landed.
598    Dependency,
599    /// The review policy is satisfied on the head.
600    Review,
601    /// No required check failed on the head.
602    NoFailingCheck,
603    /// The head contains the current master.
604    Freshness,
605    /// Every required check passed on the head.
606    RequiredChecks,
607}

670/// One machine-readable reason, typed. Its wire form is the `code` or `code:payload` string
671/// the reasons always had ([`std::fmt::Display`] and [`std::str::FromStr`]), so the trail,

696    MergeCommitNotAllowed,

720    },
721    /// `head_reachable_from_master`: every commit already landed.

723    /// `merge_changes_nothing`: its patch is already fully on master.

727    PatchIdsUpstream,
728    /// `only_derived_artifacts_differ`.

751    /// `review_policy_unread`.
752    ReviewPolicyUnread,
753    /// `required_check_failed`.
754    RequiredCheckFailed,
755    /// `behind_master:COMMITS`: the head does not contain master.
756    BehindMaster {
757        /// Commits on master the head does not have.
758        commits: u64,
759    },
760    /// `fork_head`: the head is a fork's branch, which cannot be refreshed from here.

900impl std::str::FromStr for ReasonCode {
901    type Err = String;
902
903    /// The reason a wire string names; an error for a code outside the vocabulary or a
904    /// payload that is not the code's.
905    fn from_str(s: &str) -> Result<Self, Self::Err> {
906        let bad = || format!("not a reason code: {s:?}");
907        let number = |p: &str| p.strip_prefix('#').and_then(|n| n.parse::<u64>().ok());
908        let word = |p: &str| serde_json::Value::String(p.to_string());
909        let (code, payload) = match s.split_once(':') {
910            Some((c, p)) => (c, Some(p)),
911            None => (s, None),
912        };
913        let reason = match (code, payload) {
914            ("stacked_on", Some(p)) => ReasonCode::StackedOn {
915                number: number(p).ok_or_else(bad)?,
916            },
917            ("depends_on", Some(p)) => ReasonCode::DependsOn {
918                number: number(p).ok_or_else(bad)?,
919            },
920            ("superseded_by", Some(p)) => ReasonCode::SupersededBy {
921                number: number(p).ok_or_else(bad)?,
922            },
923            ("successor_open", Some(p)) => ReasonCode::SuccessorOpen {
924                number: number(p).ok_or_else(bad)?,
925            },
926            ("successor_not_landed", Some(p)) => ReasonCode::SuccessorNotLanded {
927                number: number(p).ok_or_else(bad)?,
928            },
929            ("successor_unread", Some(p)) => ReasonCode::SuccessorUnread {
930                number: number(p).ok_or_else(bad)?,
931            },
932            ("dependency_cycle", Some(p)) => ReasonCode::DependencyCycle {
933                number: number(p).ok_or_else(bad)?,
934            },
935            ("dependency_closed_unmerged", Some(p)) => ReasonCode::DependencyClosedUnmerged {
936                number: number(p).ok_or_else(bad)?,
937            },
938            ("dependency_unread", Some(p)) => ReasonCode::DependencyUnread {
939                number: number(p).ok_or_else(bad)?,
940            },
941            ("base_is", Some(p)) => ReasonCode::BaseIs { base: p.into() },
942            ("label", Some(p)) => ReasonCode::Label { name: p.into() },
943            ("label_obsolete", Some(p)) => ReasonCode::LabelObsolete { name: p.into() },
944            ("relation_unknown", Some(p)) => ReasonCode::RelationUnknown { reason: p.into() },
945            ("conflicts_on", Some(p)) => ReasonCode::ConflictsOn {
946                count: p.parse().map_err(|_| bad())?,
947            },
948            ("behind_master", Some(p)) => ReasonCode::BehindMaster {
949                commits: p.parse().map_err(|_| bad())?,
950            },
951            ("review", Some(p)) => ReasonCode::Review {
952                state: serde_json::from_value(word(p)).map_err(|_| bad())?,
953            },
954            ("required_checks", Some(p)) => ReasonCode::RequiredChecks {
955                state: serde_json::from_value(word(p)).map_err(|_| bad())?,
956            },
957            ("draft", None) => ReasonCode::Draft,
958            ("auto_merge_armed", None) => ReasonCode::AutoMergeArmed,
959            ("merge_commit_not_allowed", None) => ReasonCode::MergeCommitNotAllowed,
960            ("head_reachable_from_master", None) => ReasonCode::HeadReachableFromMaster,
961            ("merge_changes_nothing", None) => ReasonCode::MergeChangesNothing,
962            ("patch_ids_upstream", None) => ReasonCode::PatchIdsUpstream,
963            ("only_derived_artifacts_differ", None) => ReasonCode::OnlyDerivedArtifactsDiffer,
964            ("review_policy_unread", None) => ReasonCode::ReviewPolicyUnread,
965            ("required_check_failed", None) => ReasonCode::RequiredCheckFailed,
966            ("fork_head", None) => ReasonCode::ForkHead,
967            ("no_required_checks", None) => ReasonCode::NoRequiredChecks,
968            ("required_checks_unread", None) => ReasonCode::RequiredChecksUnread,
969            ("contains_master", None) => ReasonCode::ContainsMaster,
970            ("required_checks_passed", None) => ReasonCode::RequiredChecksPassed,
971            ("required_checks_skipped", None) => ReasonCode::RequiredChecksSkipped,
972            ("executor_merge_refused", Some(p)) => {
973                ReasonCode::ExecutorMergeRefused { head: p.into() }
974            }
975            ("executor_refresh_failed", Some(p)) => {
976                ReasonCode::ExecutorRefreshFailed { master: p.into() }
977            }
978            _ => return Err(bad()),
979        };
980        Ok(reason)
981    }
982}

1016/// Whether `reason`'s wire form is `wire`, compared as it is written, without building it.
1017fn wire_is(reason: &ReasonCode, wire: &str) -> bool {
1018    struct Against<'a> {
1019        rest: &'a str,
1020        same: bool,
1021    }
1022    impl std::fmt::Write for Against<'_> {
1023        fn write_str(&mut self, s: &str) -> std::fmt::Result {
1024            match self.rest.strip_prefix(s) {
1025                Some(rest) if self.same => self.rest = rest,
1026                _ => self.same = false,
1027            }
1028            Ok(())
1029        }
1030    }
1031    let mut against = Against {
1032        rest: wire,
1033        same: true,
1034    };
1035    let _ = std::fmt::Write::write_fmt(&mut against, format_args!("{reason}"));
1036    against.same && against.rest.is_empty()
1037}

1169    /// The disposition, within the lane.
1170    pub disposition: PullRequestDisposition,
1171    /// The planning risk: lower first.
1172    pub risk: IntegrationRisk,
1173    /// How many other ready or refreshable pull requests it overlaps ([`PathOverlap`]): an
1174    /// authored path, a version bump or a release in common. Fewer first, because landing it
1175    /// invalidates less.
1176    pub contention: usize,
1177    /// How many open pull requests declare that they wait for this one and are not yet
1178    /// satisfied: more first, because landing it unblocks them.
1179    pub dependents: usize,
1180    /// How many authored paths it changes: fewer first, a smaller change is cheaper to land
1181    /// and to undo.
1182    pub authored_paths: usize,
1183    /// When it was opened: older first, so easy new work cannot starve old work.
1184    pub created_at: String,
1185    /// The number: the last tie-break, total.
1186    pub number: u64,
1187}

1189/// How long a pull request has waited for the executor, from the audit trail. Starvation is
1190/// made visible here, never resolved by changing the rank: the rank's age tie-break already
1191/// prefers the older of two otherwise equal candidates, and a wait this long is a fact a
1192/// person reads, not a reason to merge anything less safe.
1193#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
1194pub struct ExecutorWait {
1195    /// When it was first observed ready or refreshable in its current run of being so,
1196    /// RFC 3339.
1197    pub actionable_since: String,
1198    /// How many times the executor selected another pull request while this one was ready
1199    /// or refreshable.
1200    pub passed_over: u32,
1201    /// The last of those times.
1202    pub last_passed_over: Option<PassedOver>,
1203}
1204
1205/// One time the executor chose another pull request.
1206#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
1207pub struct PassedOver {
1208    /// When, RFC 3339.
1209    pub at: String,
1210    /// The pull request chosen instead.
1211    pub for_pr: u64,
1212    /// What it was chosen for (`selected` to merge, `refresh_selected` to bring master in).
1213    pub action: super::drain::IntegrationAction,
1214}

1216/// Shared authored paths with one other open pull request.
1217#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
1218pub struct PathOverlap {
1219    /// The other pull request.
1220    pub number: u64,
1221    /// The paths both change.
1222    pub paths: Vec<String>,
1223    /// What the two share. `authored` in a record written before it was said.
1224    #[serde(default)]
1225    pub kind: OverlapKind,
1226}
1227
1228/// What two open pull requests have in common that makes landing one change the other.
1229#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
1230#[serde(rename_all = "snake_case")]
1231pub enum OverlapKind {
1232    /// Authored paths both change.
1233    #[default]
1234    Authored,
1235    /// Both raise the crate's version: whichever lands second must be re-derived on the first.
1236    VersionBump,
1237    /// Both change the release records under `.ai/repo/releases/`.
1238    Release,
1239}
