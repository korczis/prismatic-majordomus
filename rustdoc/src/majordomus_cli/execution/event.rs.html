1//! The typed execution event: one envelope, a closed set of payloads, and a version.
2//!
3//! An event is a domain fact, not a message of a transport. The WebSocket endpoint, the
4//! history route, the audit line and anything added later render the same value; nothing
5//! downstream may add a field, and nothing upstream may send a string a consumer has to
6//! parse. The schema every client validates against is derived from these types by
7//! `schemars`, exactly as a capability's input and output schemas are.
8
9use schemars::JsonSchema;
10use serde::{Deserialize, Serialize};
11use serde_json::Value;
12
13use super::model::{
14    ExecutionDiagnostic, ExecutionError, ExecutionId, ExecutionState, IssueClaim, LogStream,
15    ProgressView,
16};

18/// The version of the event protocol this executable speaks.
19///
20/// It is carried on every event and negotiated when a client connects. A change that a
21/// client written against version 1 could not ignore safely — a renamed field, a removed
22/// variant, a changed meaning — increments it. A new variant does not, and neither does a
23/// new state or a new envelope field: a client is required to ignore a payload type it
24/// does not know, which is what [`EventPayload::is_terminal`] and the Cockpit's own reader
25/// are written to do. A reader of this executable refuses an event of a newer version by
26/// name rather than guessing at it ([`check_schema_version`]).
27pub const PROTOCOL_VERSION: &str = "1";
28
29/// May a reader of this executable read an event of `version`? An older version or this
30/// one, yes; a newer one, or a word that is not a version, is refused with a sentence that
31/// names it, so a reader that meets an event from a newer writer says so instead of
32/// misreading it.
33///
34/// ```
35/// use majordomus_cli::execution::event::check_schema_version;
36/// assert!(check_schema_version("1").is_ok());
37/// let refused = check_schema_version("2").unwrap_err();
38/// assert!(refused.contains("version 2") && refused.contains("newer"), "{refused}");
39/// assert!(check_schema_version("one").is_err());
40/// ```
41pub fn check_schema_version(version: &str) -> Result<(), String> {
42    let ours: u32 = PROTOCOL_VERSION.parse().unwrap_or(1);
43    match version.parse::<u32>() {
44        Ok(theirs) if (1..=ours).contains(&theirs) => Ok(()),
45        Ok(theirs) if theirs > ours => Err(format!(
46            "an execution event of schema version {theirs} is newer than version {ours}, the newest this executable reads; upgrade majordomus to read it"
47        )),
48        _ => Err(format!(
49            "{version:?} is not an execution event schema version this executable reads (1 to {ours})"
50        )),
51    }
52}
53
54/// Deserialise `schema_version`, refusing one [`check_schema_version`] refuses: every
55/// reader of an event, whatever transport it came by, refuses a newer one the same way.
56fn read_schema_version<'de, D>(deserializer: D) -> Result<String, D::Error>
57where
58    D: serde::Deserializer<'de>,
59{
60    let version = String::deserialize(deserializer)?;
61    check_schema_version(&version).map_err(serde::de::Error::custom)?;
62    Ok(version)
63}

65/// One event about one execution.
66///
67/// `sequence` is dense and starts at 1 within an execution: a client that has seen
68/// sequence `n` knows it has missed something when the next event it reads is not `n + 1`,
69/// and asks for the gap by cursor rather than reloading the world.
70#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
71pub struct ExecutionEvent {
72    /// The protocol version, [`PROTOCOL_VERSION`]. A reader refuses a newer one by name.
73    #[serde(deserialize_with = "read_schema_version")]
74    #[schemars(with = "String")]
75    pub schema_version: String,
76    /// The identity of this event: the execution's id and the sequence, so a client that
77    /// deduplicates needs no composite key of its own.
78    pub event_id: String,
79    /// The execution it is about.
80    pub execution_id: ExecutionId,
81    /// Its position in that execution's stream, from 1, without gaps.
82    pub sequence: u64,
83    /// When the event was produced, RFC 3339 in UTC.
84    pub timestamp: String,
85    /// The id this execution's log lines and tracing spans carry: the execution id, under
86    /// the name a reader of a log expects, as the snapshot's own `correlation_id` is. Absent
87    /// from an event an older writer produced, and read as empty there.
88    #[serde(default)]
89    pub correlation_id: String,
90    /// What happened.
91    #[serde(flatten)]
92    pub payload: EventPayload,
93}
94
95impl ExecutionEvent {
96    /// The event for an execution at a sequence, stamped now.
97    pub fn new(execution_id: ExecutionId, sequence: u64, payload: EventPayload) -> Self {
98        ExecutionEvent {
99            schema_version: PROTOCOL_VERSION.to_string(),
100            event_id: format!("{execution_id}#{sequence}"),
101            correlation_id: execution_id.to_string(),
102            execution_id,
103            sequence,
104            timestamp: crate::peers::rfc3339(std::time::SystemTime::now()),
105            payload,
106        }
107    }
108
109    /// The discriminator on the wire, for a log line or a filter.
110    pub fn type_name(&self) -> &'static str {
111        self.payload.type_name()
112    }
113}

115/// What happened, as a closed set.
116///
117/// The tag is `type` and the body is `data`, so every event on the wire has the same two
118/// members whatever it says, and a client switches on one string.
119#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
120#[serde(tag = "type", content = "data")]
121pub enum EventPayload {
122    /// The execution was accepted and exists. Carries what it will run and with what,
123    /// redacted, so a client that joins at the first event needs nothing else to render a
124    /// heading.
125    #[serde(rename = "execution.created")]
126    Created {
127        /// The capability's canonical id.
128        capability: String,
129        /// Its title.
130        title: String,
131        /// The input, with sensitive values already replaced.
132        input: Value,
133    },
134    /// A policy decision on whether it may run is being taken.
135    #[serde(rename = "execution.authorizing")]
136    Authorizing,
137    /// It is waiting for a worker.
138    #[serde(rename = "execution.queued")]
139    Queued {
140        /// How many executions are ahead of it.
141        ahead: usize,
142    },
143    /// A worker holds it and has not entered the handler yet.
144    #[serde(rename = "execution.claimed")]
145    Claimed {
146        /// Who holds it, in the holder's own words: a worker, a node, a session.
147        holder: String,
148    },
149    /// Where the mesh claim of an execution started for an issue stands (I4004): wanted at
150    /// the start, then held, unclaimed or refused when a worker takes it up, and released
151    /// or kept as it ends. Moves no state: the `claimed`, `refused` and final events that
152    /// follow it do.
153    #[serde(rename = "execution.issue_claim")]
154    IssueClaim(IssueClaim),
155    /// A worker picked it up and is about to enter the handler.
156    #[serde(rename = "execution.started")]
157    Started,
158    /// The handler returned, and what it produced is being checked.
159    #[serde(rename = "execution.verifying")]
160    Verifying,
161    /// How far along it is.
162    #[serde(rename = "execution.progress")]
163    Progress(ProgressView),
164    /// A named phase was entered.
165    #[serde(rename = "execution.step.started")]
166    StepStarted {
167        /// The step's stable name.
168        name: String,
169        /// One line for a reader.
170        title: String,
171    },
172    /// A named phase finished.
173    #[serde(rename = "execution.step.completed")]
174    StepCompleted {
175        /// The step's stable name.
176        name: String,
177        /// Whether what it was asked to do happened.
178        ok: bool,
179        /// What it said, when it said anything.
180        #[serde(skip_serializing_if = "Option::is_none")]
181        detail: Option<String>,
182    },
183    /// A line of output.
184    #[serde(rename = "execution.log")]
185    Log {
186        /// Which stream it came from.
187        stream: LogStream,
188        /// The text, with control characters and terminal escapes already removed.
189        message: String,
190    },
191    /// A structured finding, distinct from the outcome.
192    #[serde(rename = "execution.diagnostic")]
193    Diagnostic(ExecutionDiagnostic),
194    /// Cancellation was asked for; the handler has not stopped yet.
195    #[serde(rename = "execution.cancelling")]
196    Cancelling {
197        /// Who asked, in one word.
198        by: String,
199    },
200    /// It stopped because it was asked to.
201    #[serde(rename = "execution.cancelled")]
202    Cancelled,
203    /// It finished, and the handler returned an output.
204    ///
205    /// The output is on this event and not on one of its own, because a client must never
206    /// be able to read "succeeded" and find no result: the state and the value it produced
207    /// are one fact and travel together.
208    #[serde(rename = "execution.completed")]
209    Completed {
210        /// What the handler returned.
211        output: Value,
212    },
213    /// It finished, and it did not.
214    #[serde(rename = "execution.failed")]
215    Failed {
216        /// Why.
217        error: ExecutionError,
218    },
219    /// A policy refused it before it ran: at the request, or when a worker claimed it.
220    #[serde(rename = "execution.refused")]
221    Refused {
222        /// Why, with the code `refused` or `forbidden`.
223        error: ExecutionError,
224        /// The policy that decided, when one can be named.
225        #[serde(default, skip_serializing_if = "Option::is_none")]
226        policy: Option<String>,
227    },
228    /// Whether the work happened is not known: the worker went silent, or its answer could
229    /// not be read. Not the last event: a reconciliation that reads evidence follows it.
230    #[serde(rename = "execution.unknown_outcome")]
231    UnknownOutcome {
232        /// What is known about why the outcome is not.
233        error: ExecutionError,
234    },
235}

237impl EventPayload {
238    /// The discriminator on the wire.
239    ///
240    /// ```
241    /// use majordomus_cli::execution::EventPayload;
242    /// assert_eq!(EventPayload::Started.type_name(), "execution.started");
243    /// assert_eq!(EventPayload::Cancelled.type_name(), "execution.cancelled");
244    /// ```
245    pub fn type_name(&self) -> &'static str {
246        match self {
247            EventPayload::Created { .. } => "execution.created",
248            EventPayload::Authorizing => "execution.authorizing",
249            EventPayload::Queued { .. } => "execution.queued",
250            EventPayload::Claimed { .. } => "execution.claimed",
251            EventPayload::IssueClaim(_) => "execution.issue_claim",
252            EventPayload::Started => "execution.started",
253            EventPayload::Verifying => "execution.verifying",
254            EventPayload::Progress(_) => "execution.progress",
255            EventPayload::StepStarted { .. } => "execution.step.started",
256            EventPayload::StepCompleted { .. } => "execution.step.completed",
257            EventPayload::Log { .. } => "execution.log",
258            EventPayload::Diagnostic(_) => "execution.diagnostic",
259            EventPayload::Cancelling { .. } => "execution.cancelling",
260            EventPayload::Cancelled => "execution.cancelled",
261            EventPayload::Completed { .. } => "execution.completed",
262            EventPayload::Failed { .. } => "execution.failed",
263            EventPayload::Refused { .. } => "execution.refused",
264            EventPayload::UnknownOutcome { .. } => "execution.unknown_outcome",
265        }
266    }
267
268    /// Is this the last event of its execution? A subscriber that has read one may stop.
269    ///
270    /// ```
271    /// use majordomus_cli::execution::EventPayload;
272    /// assert!(EventPayload::Cancelled.is_terminal());
273    /// assert!(!EventPayload::Started.is_terminal());
274    /// assert!(!EventPayload::Verifying.is_terminal());
275    /// ```
276    pub fn is_terminal(&self) -> bool {
277        matches!(
278            self,
279            EventPayload::Completed { .. }
280                | EventPayload::Failed { .. }
281                | EventPayload::Cancelled
282                | EventPayload::Refused { .. }
283        )
284    }
285
286    /// The state this event puts the execution into, when it decides one.
287    pub fn implies_state(&self) -> Option<ExecutionState> {
288        match self {
289            EventPayload::Authorizing => Some(ExecutionState::Authorizing),
290            EventPayload::Queued { .. } => Some(ExecutionState::Queued),
291            EventPayload::Claimed { .. } => Some(ExecutionState::Claimed),
292            EventPayload::Started => Some(ExecutionState::Running),
293            EventPayload::Verifying => Some(ExecutionState::Verifying),
294            EventPayload::Cancelling { .. } => Some(ExecutionState::Cancelling),
295            EventPayload::Cancelled => Some(ExecutionState::Cancelled),
296            EventPayload::Completed { .. } => Some(ExecutionState::Succeeded),
297            EventPayload::Failed { .. } => Some(ExecutionState::Failed),
298            EventPayload::Refused { .. } => Some(ExecutionState::Refused),
299            EventPayload::UnknownOutcome { .. } => Some(ExecutionState::UnknownOutcome),
300            _ => None,
301        }
302    }
303}

305/// Text as an event may carry it: no control character, no terminal escape sequence, and
306/// bounded.
307///
308/// Every line a handler or a child process produces goes through here. A browser is not
309/// the only reason — a terminal reading `majordomus run --follow` would be repainted by an
310/// escape it did not choose — and doing it once, here, is why no consumer has to.
311///
312/// ```
313/// use majordomus_cli::execution::event::sanitise;
314/// assert_eq!(sanitise("plain", 100), "plain");
315/// assert_eq!(sanitise("\u{1b}[31mred\u{1b}[0m", 100), "red");
316/// assert_eq!(sanitise("a\u{0}b\tc", 100), "a b\tc");
317/// assert_eq!(sanitise("abcdef", 4), "abc…");
318/// ```
319pub fn sanitise(text: &str, max_chars: usize) -> String {
320    let mut out = String::with_capacity(text.len().min(max_chars));
321    let mut chars = text.chars().peekable();
322    let mut count = 0usize;
323    while let Some(c) = chars.next() {
324        if c == '\u{1b}' {
325            // the character after the escape says which sequence it is, and the two have
326            // different terminators: a control sequence ends at its final byte, an
327            // operating-system command at a BEL or a string terminator. Treating them
328            // alike leaves half of one on the wire.
329            match chars.next() {
330                Some('[') => {
331                    for next in chars.by_ref() {
332                        if ('\u{40}'..='\u{7e}').contains(&next) {
333                            break;
334                        }
335                    }
336                }
337                Some(']') => {
338                    while let Some(next) = chars.next() {
339                        if next == '\u{7}' {
340                            break;
341                        }
342                        if next == '\u{1b}' {
343                            if chars.peek() == Some(&'\\') {
344                                chars.next();
345                            }
346                            break;
347                        }
348                    }
349                }
350                // any other two-character escape: both are dropped
351                _ => {}
352            }
353            continue;
354        }
355        let c = if c == '\t' || !c.is_control() { c } else { ' ' };
356        if count + 1 >= max_chars {
357            out.push('…');
358            return out;
359        }
360        out.push(c);
361        count += 1;
362    }
363    out
364}
