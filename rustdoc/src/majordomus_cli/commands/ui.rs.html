1//! `majordomus ui`: the primary surface in one step (ADR 0135, I2484).
2//!
3//! The command is a projection of what already exists and nothing more: the screen is
4//! resolved against [`SCREENS`], the list the navigation is drawn from; the server is the
5//! one [`serve::converge_with`] converges on, the whole of `serve ensure`, so a server
6//! already answering is used and a second one is never started; and the address is the
7//! one the lease publishes. There is no `ui status` and no `ui stop`, because `serve
8//! status` and `serve stop` already answer for that server.
9//!
10//! A screen is refused before anything is started: a typo must not cost a server. The
11//! browser is the platform's own opener; when it cannot be launched the command says so
12//! and still prints the address, because the address is the answer and the browser only a
13//! convenience.
14

29/// What `majordomus ui --format json` prints.
30#[derive(Debug, Serialize)]
31struct UiReport {
32    /// Where this checkout's server stands.
33    standing: ServerStanding,
34    /// The address of the screen, when a server is there.
35    #[serde(skip_serializing_if = "Option::is_none")]
36    url: Option<String>,
37    /// Whether this call started the server that answers.
38    started: bool,
39    /// Whether a browser was opened on the address.
40    opened: bool,
41    /// Why the standing is not `ready`, or why no browser was opened.
42    #[serde(skip_serializing_if = "Option::is_none")]
43    reason: Option<String>,
44}

46/// Run `majordomus ui`.
47pub fn run(args: UiArgs) -> Result<u8> {
48    run_to(
49        args,
50        open_browser,
51        &mut std::io::stdout().lock(),
52        &mut std::io::stderr().lock(),
53    )
54}
55
56/// [`run`], with the browser opened by `open` and the report written into `out` and `err`.
57fn run_to(
58    args: UiArgs,
59    open: impl FnOnce(&str) -> std::result::Result<(), String>,
60    out: &mut impl Write,
61    err: &mut impl Write,
62) -> Result<u8> {
63    // refused before the repository is read or a server started: a typo costs nothing
64    let route = screen_route(args.screen.as_deref())?;
65    let repo = serve::discover(&args.repo)?;
66    let layer = serve::layer_args(&args.repo)?;
67    // the declared server timings, the idle life among them, as `serve ensure` reads them
68    let _ = crate::policy::LoadedPolicy::load(&repo);
69    let wait = Duration::from_secs(args.wait);
70    let c = serve::converge_with(
71        &repo,
72        args.port
73            .unwrap_or_else(|| crate::cli::checkout_port(repo.root())),
74        args.idle.unwrap_or_else(|| lease::timings().idle.as_secs()),
75        wait,
76        &layer,
77    )?;
78    let (report, code) = conclude(&c, route, wait, args.no_open, open);
79    write_report(&report, args.format, out, err)?;
80    Ok(code)
81}

83/// What a convergence comes to: the report and the exit code, after `open` was asked to
84/// open the screen's address when a server is ready there and opening was not declined.
85fn conclude(
86    c: &serve::Convergence,
87    route: &str,
88    wait: Duration,
89    no_open: bool,
90    open: impl FnOnce(&str) -> std::result::Result<(), String>,
91) -> (UiReport, u8) {
92    let url = c
93        .document
94        .as_ref()
95        .and_then(|d| d.url.as_deref())
96        .map(|base| format!("{}{route}", base.trim_end_matches('/')));
97    let mut report = UiReport {
98        standing: c.standing,
99        url: url.clone(),
100        started: c.started,
101        opened: false,
102        reason: serve::not_ready(c, wait),
103    };
104    let ready = c.ready() && url.is_some();
105    if let (true, Some(url), false) = (ready, &url, no_open) {
106        match open(url) {
107            Ok(()) => report.opened = true,
108            Err(why) => report.reason = Some(format!("no browser was opened: {why}")),
109        }
110    }
111    (report, if ready { 0 } else { 10 })
112}

114/// The route of the named screen, the dashboard's when none is named; an unknown name is
115/// refused with every name there is, in the navigation's order.
116///
117/// A screen is named by the last word of its route, which is what a person reads in the
118/// address bar: `roadmap` is `/ui/roadmap`.
119fn screen_route(name: Option<&str>) -> Result<&'static str> {
120    let Some(name) = name else {
121        return Ok(PREFIX);
122    };
123    let named = |route: &'static str| {
124        route
125            .strip_prefix(PREFIX)
126            .and_then(|rest| rest.strip_prefix('/'))
127    };
128    if let Some(screen) = SCREENS.iter().find(|s| named(s.route) == Some(name)) {
129        return Ok(screen.route);
130    }
131    let known: Vec<&str> = SCREENS.iter().filter_map(|s| named(s.route)).collect();
132    Err(Error::Refused {
133        code: 2,
134        reason: format!(
135            "ui: no screen is called `{name}`; the screens are {} (none names the dashboard)",
136            known.join(", ")
137        ),
138    })
139}

141/// Open `url` with the platform's own opener and wait for the opener, which returns as
142/// soon as it has handed the address on.
143fn open_browser(url: &str) -> std::result::Result<(), String> {
144    launch(opener(url))
145}
146
147/// The platform's opener, given `url` and nothing to read or write: built here and run by
148/// [`launch`], so what is run can be read without running it.
149fn opener(url: &str) -> Command {
150    #[cfg(target_os = "macos")]
151    let mut cmd = Command::new("open");
152    #[cfg(windows)]
153    let mut cmd = {
154        let mut c = Command::new("cmd");
155        c.args(["/C", "start", ""]);
156        c
157    };
158    #[cfg(not(any(target_os = "macos", windows)))]
159    let mut cmd = Command::new("xdg-open");
160    cmd.arg(url)
161        .stdin(Stdio::null())
162        .stdout(Stdio::null())
163        .stderr(Stdio::null());
164    cmd
165}
166
167/// Run `cmd` to its end: an opener that cannot be run, or that exits unsuccessfully, is
168/// the reason no browser was opened.
169fn launch(mut cmd: Command) -> std::result::Result<(), String> {
170    let program = format!("{:?}", cmd.get_program());
171    let status = cmd
172        .status()
173        .map_err(|e| format!("{program} could not be run: {e}"))?;
174    if status.success() {
175        Ok(())
176    } else {
177        Err(format!("{program} exited {status}"))
178    }
179}

181/// The report, into `out` and `err`: the address on `out`, alone, so that
182/// `$(majordomus ui --no-open)` is it; what else there is to say goes to `err`.
183fn write_report(
184    r: &UiReport,
185    format: OutputFormat,
186    out: &mut impl Write,
187    err: &mut impl Write,
188) -> Result<()> {
189    match format {
190        OutputFormat::Json => writeln!(
191            out,
192            "{}",
193            serde_json::to_string_pretty(r).unwrap_or_default()
194        )
195        .map_err(Error::Transport)?,
196        OutputFormat::Text => {
197            if r.standing == ServerStanding::Ready {
198                if let Some(url) = &r.url {
199                    writeln!(out, "{url}").map_err(Error::Transport)?;
200                }
201            }
202            if r.standing != ServerStanding::Ready {
203                let why = r.reason.as_deref().unwrap_or("no server answers");
204                let _ = writeln!(err, "majordomus ui: {}: {why}", r.standing.as_str());
205            } else if let Some(why) = &r.reason {
206                let _ = writeln!(err, "majordomus ui: {why}; open the address yourself");
207            }
208        }
209    }
210    Ok(())
211}

353    fn written(r: &UiReport, format: OutputFormat) -> (String, String) {
354        let (mut out, mut err) = (Vec::new(), Vec::new());

416    /// A stream that takes nothing: every write is refused, as a closed pipe would.
417    struct Closed;
418
419    impl Write for Closed {
420        fn write(&mut self, _: &[u8]) -> std::io::Result<usize> {
421            Err(std::io::Error::from(std::io::ErrorKind::BrokenPipe))
422        }
423        fn flush(&mut self) -> std::io::Result<()> {
424            Ok(())
425        }
426    }

447    fn ui_args(argv: &[&str]) -> UiArgs {
448        use clap::Parser;
449        let mut all = vec!["majordomus", "ui"];
450        all.extend_from_slice(argv);

453            panic!("ui");
454        };

530        let parse = |extra: &[&str]| {

542        assert!(!lease::lease_path(&repo).exists(), "nothing was started");
543        let refused = run(parse(&["nowhere"])).unwrap_err();
544        assert_eq!(refused.exit_code(), 2);
