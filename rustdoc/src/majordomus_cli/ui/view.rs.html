1//! The UI's components: the hero band of a screen, its KPI group, a panel, a donut, a
2//! timeline and a meter. Each is a function of readings (`super::source`), never of a value
3//! of its own, and each says where it read from.
4//!
5//! The source of a value is shown in the UI itself — a tile's title, a panel's "Source"
6//! line. The UI is its own product; the Cockpit is its low-level reference, so each source
7//! carries one quiet "technical view" link to the Cockpit's page for that capability, for the
8//! reader who wants the registry behind a value. An object a screen names and the UI has no
9//! screen of its own for — a document, a test, a declaring file, an ADR — leads to the
10//! Cockpit's page for that object, its technical view, and only when the index holds it.
11//! Nothing else leads there (ADR 0135: the diagnostic view of the capability, object or graph
12//! a panel read).
13//!
14//! Every entity a screen names is a link: to the UI's own screen or inspector for it when
15//! there is one, to its technical view when there is not, and to the address its source gives
16//! when it lives elsewhere. A value a source does not have is said as `none`, never left blank,
17//! and a list cut short says how much is left out and where the rest is.
18//!
19//! Charts are drawn here as SVG, on the server: they are complete without JavaScript, they
20//! carry their numbers as text beside them, and they draw only what a capability answered —
21//! a series nobody measured is not drawn at all.
22

121/// one count a screen may take of an answer, because it is the answer's own size.

176/// What a reading shows: its value, or which way it is missing. A number is shown as the
177/// source gave it, rounded to two places when it is a fraction.
178pub fn shown(r: &Reading) -> String {
179    match &r.state {
180        Availability::Available(Value::String(s)) => s.clone(),
181        Availability::Available(Value::Number(n)) => number(n),
182        Availability::Available(Value::Bool(b)) => if *b { "yes" } else { "no" }.into(),
183        Availability::Available(v) => v.to_string(),
184        Availability::Absent => "none".into(),
185        Availability::Failed(_) => "not answered".into(),
186    }
187}
188
189fn number(n: &serde_json::Number) -> String {
190    // an integer is written as the source wrote it; only a fraction is rounded
191    match n.as_f64() {
192        Some(f) if n.is_f64() => {
193            let s = format!("{f:.2}");
194            s.trim_end_matches('0').trim_end_matches('.').to_string()
195        }
196        _ => n.to_string(),
197    }
198}

260/// The sources of a panel: for each, the capability and how long it took, then every other
261/// way to ask it the same question — the API route a browser opens, the command a person
262/// types, the MCP tool an agent calls — and its technical reference in the Cockpit. Each is
263/// the capability's own declaration, as `capabilities.describe` answered it.
264pub fn sources(answers: &[&Answer]) -> El {
265    let mut box_ = el("div").class("mj-ui-sources");
266    for a in answers {
267        let mut line = el("p")
268            .class("mj-ui-source")
269            .child(
270                el("span")
271                    .class("mj-ui-source-name")
272                    .text(a.source.capability),
273            )
274            .child(
275                el("span")
276                    .class("mj-ui-dim")
277                    .text(format!("{} ms", a.took.as_millis())),
278            );
279        // a reading that is not simply current says so, and one a newer reading is about to
280        // replace marks its line, which is what the page's script reads the screen again for
281        if let Some(note) = a.standing.note() {
282            line = line.child(el("span").class("mj-ui-dim").text(note));
283        }
284        if a.standing.catching_up() {
285            line = line.attr(CATCHING_UP, "");
286        }
287        if let Some(path) = &a.exposure.get {
288            line = line.child(chip_link("API", path, "open the same answer as JSON"));
289        }
290        if let Some(cli) = &a.exposure.cli {
291            line = line.child(chip("CLI", cli));
292        }
293        if let Some(tool) = &a.exposure.mcp {
294            line = line.child(chip("MCP", tool));
295        }
296        line = line.child(
297            el("a")
298                .class("mj-ui-technical")
299                .attr("href", technical(a.source.capability))
300                .attr("title", "the capability behind this value, in the Cockpit")
301                .text("technical view"),
302        );
303        box_ = box_.child(line);
304    }
305    box_
306}

353/// [`object`] when the index holds `uri`, the label as text when it does not, so no link
354/// leads to a page that answers 404, and `none` when there is no label at all. Whether the
355/// index holds it is asked of `objects.get`, the capability every surface resolves a URI with
356/// (ADR 0012), never read from the index beside it. Asked as a reading, like every source of
357/// a screen (`project.no-screen-waits-for-a-source`): an object whose first reading has not
358/// arrived is its label as text until it has.
359pub fn object_if(ctx: &Context, uri: &str, label: &str) -> El {
360    if label.is_empty() {
361        return none();
362    }
363    match ctx
364        .read("objects.get", serde_json::json!({ "uri": uri }))
365        .value
366    {
367        Ok(_) => object(uri, label),
368        Err(_) => el("span").class("mj-ui-mono").text(label),
369    }
370}

411/// What a list cut short leaves out: "the first `shown` of `total`", and where the whole list
412/// is when there is a place that shows it. Nothing when nothing was left out.
413pub fn more(shown: usize, total: usize, whole: Option<&str>) -> Option<El> {
414    (total > shown).then(|| {
415        let line = el("p").class("mj-ui-line mj-ui-dim").text(format!(
416            "The first {shown} of {total}; {} more not shown here. ",
417            total - shown
418        ));
419        match whole {
420            Some(href) => line.child(link(href, "Open the whole list")),
421            None => line,
422        }
423    })
424}

426/// An identifier as a fragment a page can carry: letters, digits, `-`, `_` and `.` kept,
427/// anything else a `-`, so an id a source writes never breaks out of the attribute.
428pub fn anchor(id: &str) -> String {
429    id.chars()
430        .map(|c| {
431            if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.') {
432                c
433            } else {
434                '-'
435            }
436        })
437        .collect()
438}

566/// A node of a relationship graph: an entity the source named, its label, the tone its
567/// state is drawn in, and where clicking it leads.
568pub struct GraphNode {
569    /// The entity's id, as the source gave it.
570    pub id: String,
571    /// The label drawn on the node.
572    pub label: String,
573    /// The tone of the design system it is drawn in.
574    pub tone: &'static str,
575    /// Where it leads.
576    pub href: String,
577    /// `leaf` for a node drawn small and labelled only when selected; anything else is a
578    /// node of the graph's main kind.
579    pub kind: &'static str,
580}

582/// How a graph is laid out: `directed` for a dependency order, `force` for a topology.
583#[derive(Debug, Clone, Copy, PartialEq, Eq)]
584pub enum Layout {
585    /// Breadth-first along the edges' direction.
586    Directed,
587    /// Force-directed, for a topology with no order.
588    Force,
589}
590
591/// A relationship graph: the nodes and the typed edges a source answered, drawn by the
592/// vendored Cytoscape once the screen has painted. Without JavaScript the `fallback` — a
593/// table of the same edges — is what the reader gets, and with it the graph is a second view
594/// of the same data, never a different one. An edge whose end is not a node is dropped
595/// rather than drawn to nothing.
596///
597/// Only nodes that take part in an edge are drawn: a node with no relation says nothing in a
598/// graph that its row in the fallback does not say better.
599pub fn graph(
600    label: &str,
601    layout: Layout,
602    nodes: &[GraphNode],
603    edges: &[(String, String)],
604    fallback: El,
605) -> El {
606    let ids: std::collections::BTreeSet<&str> = nodes.iter().map(|n| n.id.as_str()).collect();
607    let related: std::collections::BTreeSet<&str> = edges
608        .iter()
609        .filter(|(a, b)| ids.contains(a.as_str()) && ids.contains(b.as_str()))
610        .flat_map(|(a, b)| [a.as_str(), b.as_str()])
611        .collect();
612    let data = serde_json::json!({
613        "layout": match layout { Layout::Directed => "directed", Layout::Force => "force" },
614        "nodes": nodes.iter().filter(|n| related.contains(n.id.as_str())).map(|n| serde_json::json!({
615            "id": n.id, "label": n.label, "tone": n.tone, "href": n.href, "kind": n.kind,
616        })).collect::<Vec<_>>(),
617        "edges": edges.iter()
618            .filter(|(a, b)| ids.contains(a.as_str()) && ids.contains(b.as_str()))
619            .map(|(a, b)| serde_json::json!({ "source": a, "target": b }))
620            .collect::<Vec<_>>(),
621    });
622    el("div")
623        .class("mj-ui-graph")
624        .attr("data-graph", data.to_string())
625        .child(
626            el("div")
627                .class("mj-ui-graph-canvas")
628                .attr("role", "img")
629                .attr("aria-label", label)
630                .flag("hidden"),
631        )
632        .child(el("div").class("mj-ui-graph-fallback").child(fallback))
633}

636/// an unknown word gets the neutral tone rather than a meaning.
637pub fn status_tone(status: &str) -> &'static str {

641        "READY" => "cyan",
642        "BLOCKED" => "rose",
643        "PLANNED" => "violet",
644        _ => "orange",
645    }
646}

650/// A donut of named counts, each in a tone of the design system, with the counts beside it
651/// as a legend: a reader without the picture reads the same numbers. Segments that are
652/// zero are listed and not drawn; a donut of nothing says so.
653pub fn donut(center: &str, segments: &[(String, u64)]) -> El {
654    let total: u64 = segments.iter().map(|(_, n)| n).sum();
655    if total == 0 {
656        return nothing("Nothing has been counted yet.");
657    }
658    // a circle of circumference 100, so a segment's dash is its share in percent
659    let radius = 15.915_494_309;
660    let mut offset = 25.0_f64;
661    let mut svg = el("svg")
662        .class("mj-ui-donut-svg")
663        .attr("viewBox", "0 0 42 42")
664        .attr("aria-hidden", "true")
665        .child(
666            el("circle")
667                .class("mj-ui-donut-track")
668                .attr("cx", "21")
669                .attr("cy", "21")
670                .attr("r", format!("{radius}")),
671        );
672    for (i, (_, n)) in segments.iter().enumerate() {
673        if *n == 0 {
674            continue;
675        }
676        let share = *n as f64 * 100.0 / total as f64;
677        svg = svg.child(
678            el("circle")
679                .class(format!(
680                    "mj-ui-donut-segment mj-tone--{}",
681                    TONES[i % TONES.len()]
682                ))
683                .attr("cx", "21")
684                .attr("cy", "21")
685                .attr("r", format!("{radius}"))
686                .attr(
687                    "stroke-dasharray",
688                    format!("{share:.3} {:.3}", 100.0 - share),
689                )
690                .attr("stroke-dashoffset", format!("{offset:.3}")),
691        );
692        offset -= share;
693    }
694    let legend = el("ul")
695        .class("mj-ui-legend")
696        .children(segments.iter().enumerate().map(|(i, (label, n))| {
697            el("li")
698                .class(format!(
699                    "mj-ui-legend-item mj-tone--{}",
700                    TONES[i % TONES.len()]
701                ))
702                .child(
703                    el("span")
704                        .class("mj-ui-legend-dot")
705                        .attr("aria-hidden", "true"),
706                )
707                .child(el("span").class("mj-ui-legend-name").text(label))
708                .child(el("span").class("mj-ui-legend-value").text(n.to_string()))
709        }));
710    el("div")
711        .class("mj-ui-donut")
712        .child(
713            el("div").class("mj-ui-donut-figure").child(svg).child(
714                el("span")
715                    .class("mj-ui-donut-center")
716                    .child(el("strong").text(total.to_string()))
717                    .child(el("span").text(center)),
718            ),
719        )
720        .child(legend)
721}

723/// The counts of an object at `pointer`, as donut segments in the order the source gave.
724pub fn counts(answer: &Answer, pointer: &str) -> Vec<(String, u64)> {
725    answer
726        .ok()
727        .and_then(|v| v.pointer(pointer))
728        .and_then(Value::as_object)
729        .map(|m| {
730            m.iter()
731                .filter_map(|(k, v)| v.as_u64().map(|n| (k.to_lowercase(), n)))
732                .collect()
733        })
734        // ui-integrity: an absent object is no segments, never a zero; the donut then says nothing was counted
735        .unwrap_or_default()
736}
