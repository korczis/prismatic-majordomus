1//! The worktree service: every decision about where a worktree belongs, what stands where,
2//! whether a call is where it should be, and what may be created or removed.
3//!
4//! This is the only implementation. The command line renders what it answers, the
5//! capability registry projects its read-only questions to MCP, HTTP, OpenAPI and the
6//! Cockpit, the git hooks ask its guard, and the migration in [`super::migrate`] moves what
7//! it judged misplaced. None of them decides anything; a second decision somewhere else is
8//! the defect this file exists to prevent.
9//!
10//! Nothing here touches the network. `git fetch` is never run, a ref that does not resolve
11//! locally does not resolve, and every operation works with the machine offline.
12

201    /// Everything the topology says about one registered work tree. The one place a
202    /// standing is decided.
203    pub fn judge(
204        &self,
205        record: &WorktreeRecord,
206        detail: Detail,
207        branches: &BTreeMap<String, BranchRef>,
208    ) -> WorktreeState {
209        let resolved = ResolvedPath::of(&record.path);
210        let is_primary = self.identity.is_primary(record);
211        let exists = record.path.is_dir();
212        let trunk = self.identity.trunk();
213        let mut diagnostics = Vec::new();
214        let path_text = display(&record.path);
215
216        let label = match (&record.branch, &record.head) {
217            (Some(b), _) => b.clone(),
218            (None, Some(h)) => path::detached_label(h),
219            (None, None) => "(unborn)".to_string(),
220        };
221
222        // where the branch belongs, if it has one and the name derives
223        let expected: Option<PathBuf> = match record.branch.as_deref() {
224            Some(b) => match BranchName::parse(b)
225                .and_then(|n| path::expected_path(&self.container.path, &n))
226            {
227                Ok(p) => Some(p),
228                Err(e) => {
229                    diagnostics.push(TopologyDiagnostic {
230                        code: match e {
231                            WorktreeError::PathEscape { .. } => DiagnosticCode::PathEscape,
232                            _ => DiagnosticCode::InvalidBranchName,
233                        },
234                        severity: Severity::Error,
235                        path: Some(path_text.clone()),
236                        branch: Some(b.to_string()),
237                        expected: None,
238                        message: e.to_string(),
239                        remedy: "rename the branch (git branch -m) to a name git and the filesystem both accept".into(),
240                    });
241                    None
242                }
243            },
244            None => None,
245        };
246        let expected_text = expected.as_deref().map(display);
247
248        let standing = if is_primary {
249            if let Some(b) = record.branch.as_deref() {
250                if let Some(t) = trunk.branch.as_deref() {
251                    if b != t {
252                        diagnostics.push(TopologyDiagnostic {
253                            code: DiagnosticCode::PrimaryOnNonTrunk,
254                            severity: Severity::Error,
255                            path: Some(path_text.clone()),
256                            branch: Some(b.to_string()),
257                            expected: expected_text.clone(),
258                            message: format!(
259                                "the primary checkout holds '{b}', and the primary checkout hosts the trunk '{t}'"
260                            ),
261                            remedy: format!(
262                                "when the primary checkout is clean: git switch {t}; then majordomus worktree create {b}"
263                            ),
264                        });
265                    }
266                }
267            }
268            Standing::Primary
269        } else if record.prunable.is_some() || !exists {
270            diagnostics.push(TopologyDiagnostic {
271                code: if record.prunable.is_some() {
272                    DiagnosticCode::StaleRegistration
273                } else {
274                    DiagnosticCode::Missing
275                },
276                severity: Severity::Warning,
277                path: Some(path_text.clone()),
278                branch: record.branch.clone(),
279                expected: expected_text.clone(),
280                message: match &record.prunable {
281                    Some(reason) if !reason.is_empty() => {
282                        format!("git reports it prunable: {reason}")
283                    }
284                    _ => "the directory git registered does not exist".into(),
285                },
286                remedy: "majordomus worktree repair".into(),
287            });
288            Standing::Missing
289        } else if record.branch.is_none() {
290            diagnostics.push(TopologyDiagnostic {
291                code: DiagnosticCode::Detached,
292                severity: Severity::Info,
293                path: Some(path_text.clone()),
294                branch: None,
295                expected: None,
296                message: "HEAD is detached: no branch, so no canonical path; nothing moves it"
297                    .into(),
298                remedy: "git switch -c <branch> there, then majordomus worktree migrate".into(),
299            });
300            Standing::Detached
301        } else if let Some(root) = self.scratch_root_of(&resolved) {
302            diagnostics.push(TopologyDiagnostic {
303                code: DiagnosticCode::Ephemeral,
304                severity: Severity::Warning,
305                path: Some(path_text.clone()),
306                branch: record.branch.clone(),
307                expected: expected_text.clone(),
308                message: format!(
309                    "a session's scratch checkout under {}{}, holding branch '{}' which belongs at {}; nothing moves it while the session that made it may be running",
310                    display(&root.path),
311                    root.provider
312                        .as_deref()
313                        .map(|p| format!(" (created by {p})"))
314                        .unwrap_or_default(),
315                    label,
316                    expected_text.as_deref().unwrap_or("-")
317                ),
318                remedy: format!(
319                    "detach it (git switch --detach) and continue in the canonical worktree (majordomus worktree create {}); or majordomus worktree migrate --include-ephemeral --only {}",
320                    label, label
321                ),
322            });
323            Standing::Ephemeral
324        } else if resolved.same_as(&self.container) {
325            diagnostics.push(TopologyDiagnostic {
326                code: DiagnosticCode::ContainerOccupied,
327                severity: Severity::Error,
328                path: Some(path_text.clone()),
329                branch: record.branch.clone(),
330                expected: expected_text.clone(),
331                message: format!(
332                    "it occupies the container {} itself; the container holds worktrees and cannot be one",
333                    display(&self.container.path)
334                ),
335                remedy: "majordomus worktree migrate".into(),
336            });
337            Standing::Misplaced
338        } else if expected.as_deref().is_some_and(|e| {
339            ResolvedPath::of(e).same_as(&resolved) && !symlink_below(&self.container.path, e)
340        }) {
341            Standing::Canonical
342        } else if expected.is_some() {
343            diagnostics.push(TopologyDiagnostic {
344                code: DiagnosticCode::PathMismatch,
345                severity: Severity::Error,
346                path: Some(path_text.clone()),
347                branch: record.branch.clone(),
348                expected: expected_text.clone(),
349                message: format!(
350                    "branch '{}' belongs at {}",
351                    label,
352                    expected_text.as_deref().unwrap_or("-")
353                ),
354                remedy: "majordomus worktree migrate".into(),
355            });
356            let inside_primary = resolved.is_inside(self.identity.primary_worktree());
357            let inside_other = self
358                .identity
359                .registered_worktrees()
360                .iter()
361                .filter(|r| r.path != record.path)
362                .any(|r| resolved.is_inside(&ResolvedPath::of(&r.path)));
363            if inside_primary || inside_other {
364                diagnostics.push(TopologyDiagnostic {
365                    code: DiagnosticCode::Nested,
366                    severity: Severity::Warning,
367                    path: Some(path_text.clone()),
368                    branch: record.branch.clone(),
369                    expected: expected_text.clone(),
370                    message: if inside_primary {
371                        "it sits inside the primary checkout".into()
372                    } else {
373                        "it sits inside another worktree of this repository".into()
374                    },
375                    remedy: "majordomus worktree migrate".into(),
376                });
377            }
378            Standing::Misplaced
379        } else {
380            // a branch whose name did not derive: already diagnosed above
381            Standing::Misplaced
382        };
383
384        if standing == Standing::Misplaced {
385            if let Some(reason) = &record.locked {
386                diagnostics.push(TopologyDiagnostic {
387                    code: DiagnosticCode::Locked,
388                    severity: Severity::Warning,
389                    path: Some(path_text.clone()),
390                    branch: record.branch.clone(),
391                    expected: expected_text.clone(),
392                    message: if reason.is_empty() {
393                        "it is locked, so git will not move it".into()
394                    } else {
395                        format!("it is locked ({reason}), so git will not move it")
396                    },
397                    remedy: format!("git worktree unlock {path_text}"),
398                });
399            }
400            if let Some(e) = expected.as_deref() {
401                if let Some(d) = self.destination_conflict(e, Some(record)) {
402                    diagnostics.push(d);
403                }
404            }
405        }
406
407        if !is_primary && record.branch.as_deref().is_some_and(|b| trunk.is(b)) {
408            diagnostics.push(TopologyDiagnostic {
409                code: DiagnosticCode::TrunkInLinkedWorktree,
410                severity: Severity::Warning,
411                path: Some(path_text.clone()),
412                branch: record.branch.clone(),
413                expected: Some(display(&self.identity.primary_worktree().path)),
414                message: "the trunk is checked out in a linked worktree; the primary checkout hosts it".into(),
415                remedy: "switch the primary checkout to the trunk and remove this worktree when it is clean".into(),
416            });
417        }
418
419        let dirty = if detail == Detail::Full && exists && !record.bare {
420            state::dirty_state(&record.path).ok()
421        } else {
422            None
423        };
424
425        WorktreeState {
426            path: path_text,
427            kind: if is_primary {
428                WorktreeKind::Primary
429            } else {
430                WorktreeKind::Linked
431            },
432            standing,
433            branch: record.branch.clone(),
434            label,
435            head: record.head.clone(),
436            detached: record.detached,
437            expected_path: if is_primary && record.branch.as_deref().is_some_and(|b| trunk.is(b)) {
438                None
439            } else {
440                expected_text
441            },
442            exists,
443            current: self.identity.is_current(record),
444            locked: record.locked.clone(),
445            prunable: record.prunable.clone(),
446            dirty,
447            upstream: record
448                .branch
449                .as_deref()
450                .and_then(|b| branches.get(b))
451                .and_then(|b| b.upstream.clone()),
452            issue: record
453                .branch
454                .as_deref()
455                .and_then(|b| state::issue_of(b, &self.issue_ids)),
456            diagnostics,
457        }
458    }

460    /// The scratch root a path lies under, when it does. The roots are the declarations'
461    /// (`share/providers.yaml`): the tool's own temporary directories and every provider's,
462    /// such as `.claude/worktrees/` under the primary checkout, which Claude Code creates and
463    /// removes with the session that asked for it, or the thread directory an orchestrator
464    /// keeps under its data directory. A root the primary checkout itself lives under is
465    /// skipped, as a test fixture's temporary directory is: nothing beside the repository
466    /// is a scratch checkout of it.
467    pub fn scratch_root_of(&self, path: &ResolvedPath) -> Option<ScratchRoot> {
468        let primary = self.identity.primary_worktree();
469        for root in &self.scratch_roots {
470            let resolved = ResolvedPath::of(root.path.clone());
471            if primary.is_inside(&resolved) || primary.same_as(&resolved) {
472                continue;
473            }
474            if path.is_inside(&resolved) {
475                return Some(ScratchRoot {
476                    provider: root.provider.clone(),
477                    path: resolved.path,
478                });
479            }
480        }
481        None
482    }

486        self.scratch_root_of(path).map(|r| r.path)

489    /// What occupies `expected`, when something other than `own` does.
490    pub(crate) fn destination_conflict(
491        &self,
492        expected: &Path,
493        own: Option<&WorktreeRecord>,
494    ) -> Option<TopologyDiagnostic> {
495        let meta = std::fs::symlink_metadata(expected).ok()?;
496        let resolved = ResolvedPath::of(expected);
497        // A link is a conflict whatever it points at — including this very worktree: the
498        // canonical path has to be the directory, not a name for it somewhere else.
499        if !meta.file_type().is_symlink() {
500            if let Some(r) = self.identity.record_at(&resolved) {
501                if own.is_some_and(|o| o.path == r.path) {
502                    return None;
503                }
504                return Some(TopologyDiagnostic {
505                code: DiagnosticCode::DestinationConflict,
506                severity: Severity::Error,
507                path: own.map(|o| display(&o.path)),
508                branch: own.and_then(|o| o.branch.clone()),
509                expected: Some(display(expected)),
510                message: format!(
511                    "the canonical path is occupied by a registered worktree of this repository holding {}",
512                    r.branch.clone().unwrap_or_else(|| "a detached HEAD".into())
513                ),
514                remedy: "move that worktree to its own canonical path first (majordomus worktree migrate)".into(),
515            });
516            }
517        }
518        let what = if meta.file_type().is_symlink() {
519            "a symbolic link".to_string()
520        } else if meta.is_dir() {
521            if expected.join(".git").exists() {
522                "a git checkout this repository has not registered".to_string()
523            } else if std::fs::read_dir(expected)
524                .map(|mut d| d.next().is_none())
525                .unwrap_or(false)
526            {
527                "an empty directory".to_string()
528            } else {
529                "an unrelated directory with content".to_string()
530            }
531        } else {
532            "a file".to_string()
533        };
534        Some(TopologyDiagnostic {
535            code: DiagnosticCode::DestinationConflict,
536            severity: Severity::Error,
537            path: own.map(|o| display(&o.path)),
538            branch: own.and_then(|o| o.branch.clone()),
539            expected: Some(display(expected)),
540            message: format!("the canonical path is occupied by {what}; nothing overwrites it"),
541            remedy: format!(
542                "inspect and move it aside by hand: ls -la {}",
543                display(expected)
544            ),
545        })
546    }

550    /// The whole topology.
551    pub fn topology(&self, detail: Detail) -> Result<RepositoryTopology> {
552        let primary = &self.identity.primary_worktree().path;
553        let trunk = self.identity.trunk();
554        let branch_refs = state::branches(primary)?;
555        let branches: BTreeMap<String, BranchRef> = branch_refs
556            .iter()
557            .map(|b| (b.name.clone(), b.clone()))
558            .collect();
559        let merged: Option<BTreeSet<String>> = match trunk.branch.as_deref() {
560            Some(t) => Some(state::merged_into_trunk(primary, t)?),
561            None => None,
562        };
563
564        let mut worktrees: Vec<WorktreeState> = self
565            .identity
566            .registered_worktrees()
567            .iter()
568            .map(|r| self.judge(r, detail, &branches))
569            .collect();
570
571        // repository-wide facts
572        let mut repository_diagnostics = Vec::new();
573        if trunk.branch.is_none() {
574            repository_diagnostics.push(TopologyDiagnostic {
575                code: DiagnosticCode::TrunkUnknown,
576                severity: Severity::Warning,
577                path: None,
578                branch: None,
579                expected: None,
580                message: "the trunk could not be determined: no remote HEAD, no init.defaultBranch, neither main nor master alone, and the primary checkout is detached".into(),
581                remedy: "git remote set-head origin --auto, or git config init.defaultBranch <branch>".into(),
582            });
583        }
584        let mut by_folded: BTreeMap<String, Vec<String>> = BTreeMap::new();
585        for w in &worktrees {
586            if let (Some(e), Some(b)) = (&w.expected_path, &w.branch) {
587                by_folded
588                    .entry(e.to_lowercase())
589                    .or_default()
590                    .push(b.clone());
591            }
592        }
593        for (_, names) in by_folded.iter().filter(|(_, n)| n.len() > 1) {
594            repository_diagnostics.push(TopologyDiagnostic {
595                code: DiagnosticCode::CaseCollision,
596                severity: Severity::Warning,
597                path: None,
598                branch: Some(names.join(", ")),
599                expected: None,
600                message: "these branch names derive paths that differ only by case, which one directory on a case-insensitive filesystem".into(),
601                remedy: "rename one of the branches (git branch -m)".into(),
602            });
603        }
604
605        let by_path: BTreeMap<String, &WorktreeState> =
606            worktrees.iter().map(|w| (w.path.clone(), w)).collect();
607        let mut branch_states: Vec<BranchState> = branch_refs
608            .iter()
609            .map(|b| {
610                let is_trunk = trunk.is(&b.name);
611                let worktree_path = b.worktree.as_deref().map(display);
612                let holder = worktree_path.as_deref().and_then(|p| by_path.get(p));
613                let merged_into_trunk = merged.as_ref().map(|m| m.contains(&b.name));
614                // A session's scratch checkout is never cleanup-eligible however merged and
615                // clean: the harness that made it removes it, and a listing that offered it
616                // would offer another session's floor.
617                let cleanup_eligible = !is_trunk
618                    && merged_into_trunk == Some(true)
619                    && match holder {
620                        None => true,
621                        Some(h) => {
622                            h.standing != Standing::Ephemeral
623                                && h.dirty.as_ref().is_some_and(|d| d.clean)
624                        }
625                    };
626                BranchState {
627                    name: b.name.clone(),
628                    head: b.head.clone(),
629                    trunk: is_trunk,
630                    expected_path: if is_trunk {
631                        None
632                    } else {
633                        self.expected_path_of(&b.name).ok().map(|p| display(&p))
634                    },
635                    worktree: worktree_path,
636                    upstream: b.upstream.clone(),
637                    merged_into_trunk,
638                    cleanup_eligible,
639                    issue: state::issue_of(&b.name, &self.issue_ids),
640                }
641            })
642            .collect();
643        crate::order::canonical(&mut branch_states);
644
645        let mut tallies = TopologyTallies {
646            worktrees: worktrees.len(),
647            branches: branch_states.len(),
648            branches_without_worktree: branch_states
649                .iter()
650                .filter(|b| b.worktree.is_none())
651                .count(),
652            cleanup_eligible: branch_states.iter().filter(|b| b.cleanup_eligible).count(),
653            ..TopologyTallies::default()
654        };
655        for w in &worktrees {
656            match w.standing {
657                Standing::Canonical => tallies.canonical += 1,
658                Standing::Misplaced => tallies.misplaced += 1,
659                Standing::Detached => tallies.detached += 1,
660                Standing::Ephemeral => tallies.ephemeral += 1,
661                Standing::Missing => tallies.missing += 1,
662                Standing::Primary => {}
663            }
664            if w.locked.is_some() {
665                tallies.locked += 1;
666            }
667            if w.dirty.as_ref().is_some_and(|d| !d.clean) {
668                tallies.dirty += 1;
669            }
670        }
671        let mut diagnostics: Vec<TopologyDiagnostic> = worktrees
672            .iter_mut()
673            .flat_map(|w| w.diagnostics.clone())
674            .collect();
675        diagnostics.extend(repository_diagnostics);
676        tallies.errors = diagnostics
677            .iter()
678            .filter(|d| d.severity == Severity::Error)
679            .count();
680        tallies.warnings = diagnostics
681            .iter()
682            .filter(|d| d.severity == Severity::Warning)
683            .count();
684
685        Ok(RepositoryTopology {
686            schema: SCHEMA.to_string(),
687            repository: self.repository_view(),
688            container: self.container_view(),
689            trunk: self.trunk_view(),
690            observed_from: display(&self.identity.current_worktree().path),
691            valid: tallies.errors == 0,
692            worktrees,
693            branches: branch_states,
694            diagnostics,
695            tallies,
696        })
697    }

699    /// Where this call is, and whether that is where it belongs.
700    pub fn status(&self) -> Result<StatusReport> {
701        let topology = self.topology(Detail::Fast)?;
702        let record = self.identity.current_record().cloned().ok_or_else(|| {
703            WorktreeError::NotThisRepository {
704                path: self.identity.current_worktree().path.clone(),
705                git_common_dir: self.identity.git_common_dir().path.clone(),
706            }
707        })?;
708        let branches: BTreeMap<String, BranchRef> =
709            state::branches(&self.identity.primary_worktree().path)?
710                .into_iter()
711                .map(|b| (b.name.clone(), b))
712                .collect();
713        let worktree = self.judge(&record, Detail::Full, &branches);
714        let canonical = is_in_place(&worktree, self.identity.trunk().source);
715        Ok(StatusReport {
716            schema: SCHEMA.to_string(),
717            repository: topology.repository,
718            container: topology.container,
719            trunk: topology.trunk,
720            worktree,
721            canonical,
722            repository_errors: topology.tallies.errors,
723        })
724    }

726    /// The guard: may a mutation proceed from the work tree this call came from?
727    pub fn guard(&self) -> Result<GuardVerdict> {
728        let status = self.status()?;
729        let w = status.worktree;
730        let exempt = matches!(w.standing, Standing::Detached)
731            || (w.standing == Standing::Primary && w.expected_path.is_none());
732        let reason = w
733            .diagnostics
734            .iter()
735            .find(|d| d.severity == Severity::Error || d.code == DiagnosticCode::Ephemeral)
736            .cloned();
737        Ok(GuardVerdict {
738            ok: status.canonical,
739            path: w.path,
740            branch: w.branch,
741            expected_path: w.expected_path,
742            standing: w.standing,
743            exempt,
744            reason: if status.canonical { None } else { reason },
745        })
746    }

748    /// One branch: where its worktree belongs and what is there.
749    pub fn inspect(&self, branch: &str) -> Result<InspectReport> {
750        let name = BranchName::parse(branch)?;
751        let expected = path::expected_path(&self.container.path, &name)?;
752        let primary = &self.identity.primary_worktree().path;
753        let branches: BTreeMap<String, BranchRef> = state::branches(primary)?
754            .into_iter()
755            .map(|b| (b.name.clone(), b))
756            .collect();
757        let worktree = self
758            .identity
759            .record_of_branch(branch)
760            .map(|r| self.judge(r, Detail::Full, &branches));
761        let canonical = worktree
762            .as_ref()
763            .is_some_and(|w| w.standing == Standing::Canonical);
764        let mut diagnostics: Vec<TopologyDiagnostic> = worktree
765            .as_ref()
766            .map(|w| w.diagnostics.clone())
767            .unwrap_or_default();
768        if worktree.is_none() {
769            if let Some(d) = self.destination_conflict(&expected, None) {
770                diagnostics.push(d);
771            }
772        }
773        Ok(InspectReport {
774            branch: branch.to_string(),
775            branch_exists: branches.contains_key(branch),
776            expected_path: display(&expected),
777            destination_exists: std::fs::symlink_metadata(&expected).is_ok(),
778            worktree,
779            canonical,
780            diagnostics,
781        })
782    }

786    /// Resolve a selector to exactly one registered work tree: an exact branch name, or an
787    /// exact path. Nothing is matched by prefix, substring or similarity — these selectors
788    /// are given to commands that delete things.
789    pub fn resolve(&self, selector: &str) -> Result<WorktreeRecord> {
790        if let Some(r) = self.identity.record_of_branch(selector) {
791            return Ok(r.clone());
792        }
793        let wanted = ResolvedPath::of(PathBuf::from(selector));
794        if let Some(r) = self.identity.record_at(&wanted) {
795            return Ok(r.clone());
796        }
797        Err(WorktreeError::NoSuchWorktree {
798            selector: selector.to_string(),
799        })
800    }

812    /// already there.

817    fn create_inner(&self, request: &CreateRequest, ensure: bool) -> Result<CreateReport> {
818        let name = BranchName::parse(&request.branch)?;
819        let target = path::expected_path(&self.container.path, &name)?;
820        let resolved_target = ResolvedPath::of(&target);
821        let primary = self.identity.primary_worktree().path.clone();
822
823        // One lock around the whole check-then-act, and the topology re-read under it.
824        let _lock = WorktreeLock::acquire(&self.identity.git_common_dir().path)?;
825        let mut identity = self.identity.clone();
826        identity.refresh()?;
827
828        if let Some(existing) = identity.record_of_branch(name.as_str()) {
829            if ResolvedPath::of(&existing.path).same_as(&resolved_target) {
830                if ensure {
831                    // Found rather than made, and approved all the same: a worktree that
832                    // exists and is blocked is what `ensure` is most often asked about.
833                    let envrc = direnv::approve(&primary, &existing.path);
834                    let envrc_local = direnv::link_local(&primary, &existing.path);
835                    return Ok(CreateReport {
836                        path: display(&existing.path),
837                        branch: name.as_str().to_string(),
838                        branch_created: false,
839                        base: None,
840                        container: display(&self.container.path),
841                        container_created: false,
842                        existed: true,
843                        envrc,
844                        envrc_local,
845                    });
846                }
847                return Err(WorktreeError::WorktreeAlreadyExists {
848                    path: existing.path.clone(),
849                    branch: name.as_str().to_string(),
850                });
851            }
852            return Err(WorktreeError::BranchAlreadyCheckedOut {
853                branch: name.as_str().to_string(),
854                path: existing.path.clone(),
855                expected: target,
856            });
857        }
858        if let Some(d) = self.destination_conflict(&target, None) {
859            return Err(WorktreeError::DestinationConflict {
860                path: target,
861                what: d.message,
862            });
863        }
864        if !git::branch_name_is_valid(&primary, name.as_str())? {
865            return Err(WorktreeError::InvalidBranchName {
866                given: name.as_str().to_string(),
867                reason: "git does not accept it as a reference name".into(),
868            });
869        }
870        let branch_exists = git::branch_exists(&primary, name.as_str())?;
871        let base = if branch_exists {
872            None
873        } else {
874            let base = request
875                .base
876                .clone()
877                .or_else(|| self.identity.trunk().branch.clone())
878                .unwrap_or_else(|| "HEAD".to_string());
879            if !git::rev_exists(&primary, &base)? {
880                return Err(WorktreeError::BaseDoesNotExist { base });
881            }
882            Some(base)
883        };
884
885        let container_created = self.ensure_container()?;
886        if let Some(parent) = target.parent() {
887            std::fs::create_dir_all(parent).map_err(|e| WorktreeError::io(parent, &e))?;
888        }
889        let target_text = display(&target);
890        let mut args: Vec<String> = vec!["worktree".into(), "add".into()];
891        match &base {
892            None => {
893                args.push("--".into());
894                args.push(target_text.clone());
895                args.push(name.as_str().to_string());
896            }
897            Some(base) => {
898                args.push("-b".into());
899                args.push(name.as_str().to_string());
900                args.push("--".into());
901                args.push(target_text.clone());
902                args.push(base.clone());
903            }
904        }
905        git::run(&primary, &args)?;
906
907        // Verify rather than assume: the answer is what the topology says.
908        identity.refresh()?;
909        let registered = identity
910            .record_at(&resolved_target)
911            .cloned()
912            .ok_or_else(|| WorktreeError::MigrationUnsafe {
913                path: target.clone(),
914                reason: "git reported success and `git worktree list` does not show the worktree; \
915                         nothing was cleaned up because that would risk deleting real work"
916                    .into(),
917            })?;
918        // The path exists now; this is the moment direnv's approval is carried to it.
919        let envrc = direnv::approve(&primary, &registered.path);
920        let envrc_local = direnv::link_local(&primary, &registered.path);
921        Ok(CreateReport {
922            path: display(&registered.path),
923            branch: name.as_str().to_string(),
924            branch_created: base.is_some(),
925            base,
926            container: display(&self.container.path),
927            container_created,
928            existed: false,
929            envrc,
930            envrc_local,
931        })
932    }

934    /// Make sure the container exists, and answer whether this call created it. It lives
935    /// beside the repository and is never committed, so it is created lazily, by the first
936    /// worktree that needs it.
937    pub(crate) fn ensure_container(&self) -> Result<bool> {
938        let p = &self.container.path;
939        if p.is_dir() {
940            return Ok(false);
941        }
942        if std::fs::symlink_metadata(p).is_ok() {
943            return Err(WorktreeError::ContainerNotADirectory { path: p.clone() });
944        }
945        std::fs::create_dir_all(p).map_err(|e| WorktreeError::io(p, &e))?;
946        Ok(true)
947    }

951    /// Remove one linked worktree. Never a branch, never the primary checkout, never
952    /// uncommitted work without `force`.
953    pub fn remove(&self, selector: &str, force: bool) -> Result<RemoveReport> {
954        let _lock = WorktreeLock::acquire(&self.identity.git_common_dir().path)?;
955        let mut identity = self.identity.clone();
956        identity.refresh()?;
957        let record = self.resolve(selector)?;
958        if identity.is_primary(&record) {
959            return Err(WorktreeError::PrimaryWorktreeProtected {
960                path: record.path.clone(),
961                operation: "remove".into(),
962            });
963        }
964        let locked = record.locked.is_some();
965        if locked && !force {
966            return Err(WorktreeError::LockedWorktree {
967                path: record.path.clone(),
968                reason: record.locked.clone(),
969            });
970        }
971        let mut forced = false;
972        if record.path.is_dir() {
973            let dirty = state::dirty_state(&record.path)?;
974            if !dirty.clean {
975                if !force {
976                    return Err(WorktreeError::DirtyWorktree {
977                        path: record.path.clone(),
978                        summary: dirty.summary(),
979                        operation: "remove".into(),
980                    });
981                }
982                forced = true;
983            }
984        }
985        let mut args: Vec<String> = vec!["worktree".into(), "remove".into()];
986        if forced || locked {
987            args.push("--force".into());
988        }
989        if locked {
990            args.push("--force".into());
991        }
992        args.push("--".into());
993        args.push(display(&record.path));
994        git::run(&identity.primary_worktree().path, &args)?;
995        Ok(RemoveReport {
996            path: display(&record.path),
997            branch: record.branch.clone(),
998            forced: forced || locked,
999        })
1000    }

1004    /// Drop the registrations of work trees whose directories are gone, and let git repair
1005    /// the administrative links of the ones that exist. Deletes no directory.
1006    pub fn repair(&self, dry_run: bool) -> Result<RepairReport> {
1007        let _lock = WorktreeLock::acquire(&self.identity.git_common_dir().path)?;
1008        let primary = &self.identity.primary_worktree().path;
1009        let mut args: Vec<String> = vec!["worktree".into(), "prune".into(), "--verbose".into()];
1010        if dry_run {
1011            args.push("--dry-run".into());
1012        }
1013        let pruned = lines_of(&git::run(primary, &args)?.stdout);
1014        let repaired = if dry_run {
1015            Vec::new()
1016        } else {
1017            let mut args: Vec<String> = vec!["worktree".into(), "repair".into()];
1018            for r in self.identity.registered_worktrees().iter().skip(1) {
1019                if r.path.is_dir() {
1020                    args.push(display(&r.path));
1021                }
1022            }
1023            let out = git::run(primary, &args)?;
1024            let mut lines = lines_of(&out.stdout);
1025            lines.extend(lines_of(out.stderr.as_bytes()));
1026            lines
1027        };
1028        Ok(RepairReport {
1029            pruned,
1030            repaired,
1031            applied: !dry_run,
1032        })
1033    }

1036/// Is any component of `path` below `container` a symbolic link? A link inside the container
1037/// pointing at a worktree elsewhere would make that worktree canonicalise to its expected
1038/// path; the expected path has to be a real directory, not a name for one somewhere else.
1039fn symlink_below(container: &Path, path: &Path) -> bool {
1040    let Ok(rest) = path.strip_prefix(container) else {
1041        return false;
1042    };
1043    let mut cursor = container.to_path_buf();
1044    for c in rest.components() {
1045        cursor.push(c);
1046        if std::fs::symlink_metadata(&cursor)
1047            .map(|m| m.file_type().is_symlink())
1048            .unwrap_or(false)
1049        {
1050            return true;
1051        }
1052    }
1053    false
1054}

1056/// Whether a judged work tree is where it belongs: canonical, or the primary checkout on
1057/// the trunk (or with no trunk to be held to), or detached.
1058fn is_in_place(w: &WorktreeState, trunk_source: TrunkSource) -> bool {
1059    match w.standing {
1060        Standing::Canonical | Standing::Detached => true,
1061        Standing::Primary => {
1062            trunk_source == TrunkSource::Unknown
1063                || !w
1064                    .diagnostics
1065                    .iter()
1066                    .any(|d| d.code == DiagnosticCode::PrimaryOnNonTrunk)
1067        }
1068        Standing::Misplaced | Standing::Missing | Standing::Ephemeral => false,
1069    }
1070}

1072/// The scratch roots the distribution declares, expanded against `primary`, in declaration
1073/// order: the tool's own first, then each provider's. Located the way every command locates
1074/// the share (`--share` excepted, which this path does not carry), from `current` — the
1075/// checkout the command runs in, so that a repository supervising itself reads the
1076/// declarations of the branch it is on rather than the primary checkout's; a checkout with
1077/// no share in reach declares none. The standard library's answer for the temporary
1078/// directory is added when the declarations name `$TMPDIR` and the environment does not, so
1079/// that what the tool itself creates under `std::env::temp_dir()` is recognised on every
1080/// platform.
1081pub fn declared_scratch_roots(current: &Path, primary: &Path) -> Vec<ScratchRoot> {
1082    let Ok(share) = crate::share::Share::locate(None, current) else {
1083        return Vec::new();
1084    };
1085    let Ok(decls) = share.providers() else {
1086        return Vec::new();
1087    };
1088    scratch_roots_from(&decls, primary)
1089}
1090
1091/// [`declared_scratch_roots`] over declarations already read.
1092pub fn scratch_roots_from(
1093    decls: &crate::share::ProviderDeclarations,
1094    primary: &Path,
1095) -> Vec<ScratchRoot> {
1096    let mut out: Vec<ScratchRoot> = decls
1097        .scratch_roots(primary)
1098        .into_iter()
1099        .map(|(provider, path)| ScratchRoot { provider, path })
1100        .collect();
1101    if decls.scratch_roots.iter().any(|r| r == "$TMPDIR") {
1102        let temp = std::env::temp_dir();
1103        if !out.iter().any(|r| r.provider.is_none() && r.path == temp) {
1104            out.push(ScratchRoot {
1105                provider: None,
1106                path: temp,
1107            });
1108        }
1109    }
1110    out
1111}
