1//! The derive offload: a branch's `scripts/derive` run on another machine the repository
2//! declares, in that machine's turn, and the derived tree brought back staged.
3//!
4//! A derive costs from ten minutes on an idle machine to an hour on the shared one, and
5//! every session needs one before it can commit. Until this module the offload was a script
6//! in a home directory with its hosts typed into it and a lock that was a race. Now:
7//!
8//! - [`declaration`]: the hosts and their limits, read from the layer (kind
9//!   `derive-offload`, `.ai/repo/offload/*.yaml`) and nowhere else;
10//! - [`host`]: which host a derive goes to — free space, then the fewest derives running or
11//!   waiting, then declaration order;
12//! - [`queue`]: the FIFO queue on the host, decided here over what the host reports;
13//! - [`remote`]: how a script reaches a host, over ssh or directly;
14//! - [`run`]: the round trip — bundle in, derive, bundle back, stage.
15//!
16//! `docs/OFFLOAD.md` is the reference; `majordomus offload` is the command.
17
18pub mod declaration;
19pub mod host;
20pub mod queue;
21pub mod remote;
22pub mod run;
23
24use declaration::{OffloadDeclaration, KIND};

26/// The repository's declaration, read from its index: `Ok(None)` when it declares none,
27/// an error naming the object when one is declared and invalid, or when two are.
28pub fn from_objects(
29    objects: &[crate::model::Object],
30) -> Result<Option<OffloadDeclaration>, String> {
31    let mut found: Vec<&crate::model::Object> = objects.iter().filter(|o| o.kind == KIND).collect();
32    match found.len() {
33        0 => Ok(None),
34        1 => OffloadDeclaration::parse(found.remove(0)).map(Some),
35        _ => Err(format!(
36            "{} derive-offload declarations ({}); a repository declares one",
37            found.len(),
38            found
39                .iter()
40                .map(|o| o.uri.as_str())
41                .collect::<Vec<_>>()
42                .join(", ")
43        )),
44    }
45}

47#[cfg(test)]
48mod tests {
49    use super::*;
50    use crate::model::{Object, Provenance};
51
52    fn object(kind: &str, identity: &str, metadata: serde_json::Value) -> Object {
53        Object {
54            kind: kind.into(),
55            identity: identity.into(),
56            uri: format!("majordomus://{kind}/{identity}"),
57            title: None,
58            description: None,
59            metadata,
60            body: String::new(),
61            content: String::new(),
62            media_type: "application/yaml",
63            provenance: Provenance {
64                path: format!(".ai/repo/offload/{identity}.yaml"),
65                directory: ".ai/repo/offload".into(),
66                source_class: "offload".into(),
67                section: None,
68                bytes: 0,
69                member: None,
70            },
71        }
72    }
73
74    fn declared(id: &str) -> Object {
75        object(
76            KIND,
77            id,
78            serde_json::json!({
79                "schema": declaration::SCHEMA_VERSION, "kind": KIND, "id": id,
80                "hosts": [{ "id": "lundra", "address": "u@lundra" }]
81            }),
82        )
83    }
84
85    #[test]
86    fn the_one_declaration_of_the_index_is_found_among_other_objects() {
87        let rule = object("rule", "project.alpha", serde_json::json!({}));
88        assert_eq!(from_objects(std::slice::from_ref(&rule)).unwrap(), None);
89
90        let d = from_objects(&[rule.clone(), declared("derive")])
91            .unwrap()
92            .expect("one declaration is declared");
93        assert_eq!(
94            (d.id.as_str(), d.hosts[0].id.as_str()),
95            ("derive", "lundra")
96        );
97
98        let invalid = object(KIND, "broken", serde_json::json!({ "kind": KIND }));
99        assert!(from_objects(&[invalid])
100            .unwrap_err()
101            .starts_with("majordomus://derive-offload/broken: "));
102    }
103
104    #[test]
105    fn two_declarations_are_refused_naming_both() {
106        assert_eq!(
107            from_objects(&[declared("one"), declared("two")]).unwrap_err(),
108            "2 derive-offload declarations (majordomus://derive-offload/one, \
109             majordomus://derive-offload/two); a repository declares one"
110        );
111    }
