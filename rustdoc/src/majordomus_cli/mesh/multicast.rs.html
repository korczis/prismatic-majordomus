1//! The UDP multicast provider: the preferred LAN mechanism. One socket, bound to the
2//! configured port and joined to the group; a listener thread hands every datagram to
3//! the manager, an announcer thread transmits the beacon on the interval with jitter.
4//! IPv4 only, TTL from the declaration (1 by default: the local segment and no further).
5//!
6//! Multicast does not work on every network, and this provider assumes nothing: a socket
7//! that cannot bind (another local Majordomus already listens on the port) or a group
8//! that cannot be joined marks the provider `Failed` with the reason, and every other
9//! provider runs on. The datagrams this socket hears may also be unicast or broadcast
10//! aimed at the port — provenance names the socket that heard, not the routing that
11//! delivered.
12//!
13//! ```
14//! use majordomus_cli::mesh::multicast::MulticastProvider;
15//! use majordomus_cli::mesh::config::MulticastConfig;
16//! use majordomus_cli::mesh::provider::{MeshProvider, MeshProviderState};
17//!
18//! // Before the manager starts it, a provider is stopped and has heard nothing.
19//! let provider = MulticastProvider::new(MulticastConfig::default());
20//! assert_eq!(provider.id(), "udp_multicast");
21//! assert_eq!(provider.status().state, MeshProviderState::Stopped);
22//! ```
23

38/// How long a blocked read waits before checking the stop flag: every wait is bounded.
39const READ_TIMEOUT: Duration = Duration::from_millis(500);
40
41/// The multicast provider: the declared group and port, the socket its threads share,
42/// and the counters its status reports. Constructed by the manager from the declaration.
43pub struct MulticastProvider {
44    config: MulticastConfig,
45    counters: Arc<Counters>,
46    state: Arc<Mutex<(MeshProviderState, Option<String>)>>,
47}
48
49impl MulticastProvider {
50    /// A provider over its declaration; nothing opens until the manager calls `start`,
51    /// so constructing one is free and never touches the network.
52    pub fn new(config: MulticastConfig) -> Self {
53        MulticastProvider {
54            config,
55            counters: Arc::new(Counters::default()),
56            state: Arc::new(Mutex::new((MeshProviderState::Stopped, None))),
57        }
58    }
59
60    fn open(&self) -> Result<(UdpSocket, Ipv4Addr), MeshError> {
61        let group: Ipv4Addr = self.config.group.parse().map_err(|_| {
62            MeshError::Provider(format!("'{}' is not an IPv4 group", self.config.group))
63        })?;
64        if !group.is_multicast() {
65            return Err(MeshError::Provider(format!(
66                "{group} is not a multicast address"
67            )));
68        }
69        let socket = bind_shared(self.config.port).map_err(|e| {
70            MeshError::Provider(format!("cannot bind udp port {}: {e}", self.config.port))
71        })?;
72        socket
73            .join_multicast_v4(&group, &Ipv4Addr::UNSPECIFIED)
74            .map_err(|e| MeshError::Provider(format!("cannot join {group}: {e}")))?;
75        socket
76            .set_multicast_ttl_v4(self.config.ttl)
77            .map_err(|e| MeshError::Provider(format!("cannot set ttl {}: {e}", self.config.ttl)))?;
78        socket
79            .set_read_timeout(Some(READ_TIMEOUT))
80            .map_err(|e| MeshError::Provider(format!("cannot bound the socket read: {e}")))?;
81        Ok((socket, group))
82    }
83}

85/// Bind the multicast port so that every Majordomus runtime of this machine can bind it
86/// too: `SO_REUSEADDR` and `SO_REUSEPORT` before the bind. The kernel then delivers each
87/// group datagram to every socket joined to the group, so two worktrees' servers hear the
88/// group — and each other — instead of the second one failing to bind.
89#[cfg(unix)]
90fn bind_shared(port: u16) -> std::io::Result<UdpSocket> {
91    use std::os::fd::FromRawFd;
92    // SAFETY: one socket descriptor, owned here until it is handed to UdpSocket (which
93    // closes it); every early return closes it first. The sockaddr_in is zeroed and then
94    // filled field by field, with the BSD length byte set where the platform has one.
95    unsafe {
96        let fd = libc::socket(libc::AF_INET, libc::SOCK_DGRAM, 0);
97        if fd < 0 {
98            return Err(std::io::Error::last_os_error());
99        }
100        let fail = |fd| {
101            let error = std::io::Error::last_os_error();
102            libc::close(fd);
103            Err(error)
104        };
105        let one: libc::c_int = 1;
106        for option in [libc::SO_REUSEADDR, libc::SO_REUSEPORT] {
107            if libc::setsockopt(
108                fd,
109                libc::SOL_SOCKET,
110                option,
111                &one as *const libc::c_int as *const libc::c_void,
112                std::mem::size_of::<libc::c_int>() as libc::socklen_t,
113            ) != 0
114            {
115                return fail(fd);
116            }
117        }
118        let mut address: libc::sockaddr_in = std::mem::zeroed();
119        address.sin_family = libc::AF_INET as libc::sa_family_t;
120        address.sin_port = port.to_be();
121        address.sin_addr = libc::in_addr {
122            s_addr: u32::from(Ipv4Addr::UNSPECIFIED).to_be(),
123        };
124        #[cfg(any(target_os = "macos", target_os = "ios", target_os = "freebsd"))]
125        {
126            address.sin_len = std::mem::size_of::<libc::sockaddr_in>() as u8;
127        }
128        if libc::bind(
129            fd,
130            &address as *const libc::sockaddr_in as *const libc::sockaddr,
131            std::mem::size_of::<libc::sockaddr_in>() as libc::socklen_t,
132        ) != 0
133        {
134            return fail(fd);
135        }
136        Ok(UdpSocket::from_raw_fd(fd))
137    }
138}

140/// Off Unix, a plain bind: one runtime per machine hears the group.
141#[cfg(not(unix))]
142fn bind_shared(port: u16) -> std::io::Result<UdpSocket> {
143    UdpSocket::bind((Ipv4Addr::UNSPECIFIED, port))
144}

146impl MeshProvider for MulticastProvider {
147    fn id(&self) -> &'static str {
148        "udp_multicast"
149    }
150
151    fn start(&mut self, ctx: &ProviderContext) -> Result<(), MeshError> {
152        let (socket, group) = match self.open() {
153            Ok(opened) => opened,
154            Err(e) => {
155                *self.state.lock().expect("multicast state") =
156                    (MeshProviderState::Failed, Some(e.to_string()));
157                return Err(e);
158            }
159        };
160        let destination = SocketAddrV4::new(group, self.config.port);
161        *self.state.lock().expect("multicast state") = (
162            MeshProviderState::Running,
163            Some(format!("group {destination}, ttl {}", self.config.ttl)),
164        );
165
166        // The listener: every datagram to the manager, raw. Parsing happens once, above.
167        {
168            let socket = socket.try_clone().map_err(|e| {
169                MeshError::Provider(format!("cannot clone the multicast socket: {e}"))
170            })?;
171            let tx = ctx.tx.clone();
172            let stop = Arc::clone(&ctx.stop);
173            let counters = Arc::clone(&self.counters);
174            let state = Arc::clone(&self.state);
175            let _ = std::thread::Builder::new()
176                .name("majordomus-mesh-multicast-listen".into())
177                .spawn(move || {
178                    let mut buffer = [0u8; MAX_DATAGRAM + 1];
179                    while !stop.load(Ordering::SeqCst) {
180                        match socket.recv_from(&mut buffer) {
181                            Ok((len, from)) => {
182                                counters.received.fetch_add(1, Ordering::Relaxed);
183                                let _ = tx.send(Observation {
184                                    source: MeshSource::UdpMulticast,
185                                    path: from.to_string(),
186                                    bytes: buffer[..len].to_vec(),
187                                });
188                            }
189                            Err(e)
190                                if e.kind() == std::io::ErrorKind::WouldBlock
191                                    || e.kind() == std::io::ErrorKind::TimedOut => {}
192                            Err(e) => {
193                                *state.lock().expect("multicast state") =
194                                    (MeshProviderState::Failed, Some(format!("recv: {e}")));
195                                return;
196                            }
197                        }
198                    }
199                    *state.lock().expect("multicast state") = (MeshProviderState::Stopped, None);
200                });
201        }
202
203        // The announcer: the beacon on the interval, jittered, and once immediately —
204        // a node that starts should be seen now, not an interval from now.
205        {
206            let tx_socket = socket;
207            let stop = Arc::clone(&ctx.stop);
208            let beacon = Arc::clone(&ctx.beacon);
209            let counters = Arc::clone(&self.counters);
210            let interval = Duration::from_secs(self.config.interval_seconds);
211            let _ = std::thread::Builder::new()
212                .name("majordomus-mesh-multicast-announce".into())
213                .spawn(move || {
214                    while !stop.load(Ordering::SeqCst) {
215                        if let Ok(bytes) = encode(&beacon.next_envelope()) {
216                            if tx_socket.send_to(&bytes, destination).is_ok() {
217                                counters.sent.fetch_add(1, Ordering::Relaxed);
218                            }
219                        }
220                        // Sleep the interval plus jitter, in bounded slices so a stop
221                        // is honoured within half a second.
222                        let total = interval + Duration::from_millis(jitter_ms(2000));
223                        let mut slept = Duration::ZERO;
224                        while slept < total && !stop.load(Ordering::SeqCst) {
225                            let slice = READ_TIMEOUT.min(total - slept);
226                            std::thread::sleep(slice);
227                            slept += slice;
228                        }
229                    }
230                });
231        }
232        Ok(())
233    }
234
235    fn status(&self) -> ProviderStatus {
236        let (state, detail) = self.state.lock().expect("multicast state").clone();
237        ProviderStatus {
238            id: self.id().into(),
239            state,
240            detail,
241            sent: self.counters.sent.load(Ordering::Relaxed),
242            received: self.counters.received.load(Ordering::Relaxed),
243        }
244    }
245}

252    /// The first observation carrying exactly `bytes`, passing over every other datagram the
253    /// listener hands up before `within` runs out; `None` when it never arrives.
254    fn heard_among(
255        rx: &std::sync::mpsc::Receiver<Observation>,
256        bytes: &[u8],
257        within: Duration,
258    ) -> Option<Observation> {
259        let deadline = std::time::Instant::now() + within;
260        loop {
261            let left = deadline.checked_duration_since(std::time::Instant::now())?;
262            let heard = rx.recv_timeout(left).ok()?;
263            if heard.bytes == bytes {
264                return Some(heard);
265            }
266        }
267    }

269    #[test]
270    fn the_datagram_a_test_sent_is_found_behind_the_providers_own_beacon() {
271        let (tx, rx) = std::sync::mpsc::channel();
272        let seen = |bytes: &[u8]| Observation {
273            source: MeshSource::UdpMulticast,
274            path: "127.0.0.1:1".into(),
275            bytes: bytes.to_vec(),
276        };
277        tx.send(seen(b"{\"v\":1,\"pk\":\"1fb5\"}")).unwrap();
278        tx.send(seen(b"a datagram for the listener")).unwrap();
279        let heard = heard_among(
280            &rx,
281            b"a datagram for the listener",
282            Duration::from_millis(200),
283        );
284        assert_eq!(
285            heard.map(|o| o.bytes),
286            Some(b"a datagram for the listener".to_vec())
287        );
288        // and a datagram that never comes is an absence, not a hang
289        assert!(heard_among(&rx, b"never sent", Duration::from_millis(50)).is_none());
290    }

315    #[test]
316    fn the_listener_hears_and_the_announcer_transmits() {
317        // What this holds without depending on the host's network: the provider binds
318        // its socket, joins its group and reaches Running, and its listener and announcer
319        // threads are alive. Delivery and the send count are then observed *if the host
320        // routes multicast* — a CI runner and a locked-down desktop often route none, and
321        // an assertion on delivery there is testing the network, not the provider. The
322        // datagram round-trip and the send accounting are proven host-independently by
323        // the manager's own tests (a_synthetic_provider_reaches_the_registry...) and by
324        // the two-runtime rendezvous test in tests/mesh.rs; here they are a best-effort
325        // observation, reported, never a failure.
326        //
327        // The probe binds the wildcard the provider binds (bind_shared), so the port it
328        // offers is free on every local address. A loopback probe may offer a port another
329        // socket holds on a host address: macOS lets the provider's SO_REUSEADDR wildcard
330        // bind share it, Linux refuses unless that other socket set SO_REUSEADDR too.
331        let port = {
332            let probe = UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)).unwrap();
333            probe.local_addr().unwrap().port()
334        };
335        let mut provider = MulticastProvider::new(MulticastConfig {
336            port,
337            interval_seconds: 1,
338            ..MulticastConfig::default()
339        });
340        let (tx, rx) = std::sync::mpsc::channel();
341        let stop = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
342        let ctx = crate::mesh::provider::ProviderContext {
343            tx,
344            stop: std::sync::Arc::clone(&stop),
345            beacon: std::sync::Arc::new(crate::mesh::provider::Beacon::new(
346                std::sync::Arc::new(crate::mesh::identity::NodeIdentity::ephemeral().unwrap()),
347                vec![],
348                vec![],
349                vec![],
350                "test",
351            )),
352        };
353
354        // The one hard assertion: the provider comes up. A host that cannot bind the port
355        // or join the group is a real failure of the provider's own contract.
356        provider.start(&ctx).unwrap();
357        assert_eq!(provider.status().state, MeshProviderState::Running);
358        assert_eq!(provider.id(), "udp_multicast");
359
360        // Best-effort: aim a datagram at the port and see whether the listener hands it
361        // up. On a host that delivers it, assert it is well-formed; on one that does not,
362        // observe the silence and move on — the socket bound, which is what was under test.
363        //
364        // The listener hears more than this datagram: the announcer sends the provider's own
365        // beacon the moment it starts, and a host that routes multicast loops it back to this
366        // socket, so the first datagram up was often that beacon, and the test failed on it
367        // (master's rust job in run 36557029377, `left` the beacon's JSON). Datagrams that are
368        // not this one are passed over.
369        let sender = UdpSocket::bind((Ipv4Addr::LOCALHOST, 0)).unwrap();
370        let _ = sender.send_to(b"a datagram for the listener", (Ipv4Addr::LOCALHOST, port));
371        if let Some(heard) =
372            heard_among(&rx, b"a datagram for the listener", Duration::from_secs(2))
373        {
374            assert_eq!(heard.source, MeshSource::UdpMulticast);
375        }
376
377        // Best-effort: the announcer transmits on a host with a multicast route. Where
378        // there is none, send_to fails and the count stays zero; that is the network's
379        // answer, not the provider's.
380        let deadline = std::time::Instant::now() + Duration::from_secs(2);
381        while provider.status().sent == 0 && std::time::Instant::now() < deadline {
382            std::thread::sleep(Duration::from_millis(50));
383        }
384        // Either it transmitted, or it did not for want of a route — both are consistent
385        // with a Running provider. The status is readable either way, which is the invariant.
386        let _ = provider.status().sent;
387        stop.store(true, std::sync::atomic::Ordering::SeqCst);
388    }
