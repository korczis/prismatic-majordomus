1//! The resolver: one function that builds a [`RepositoryEnvironment`], and the only place
2//! any surface gets one.
3//!
4//! # The two resolutions
5//!
6//! A full resolution has the index; a fast one does not, and may not build it. That is not
7//! a preference. Building the index reads and validates every declared file of the layer,
8//! which is measured in seconds even in a release build, and a fast resolution is what
9//! runs when a person types `cd`.
10//!
11//! Everything else is cheap enough for either: the project identity is compile-time
12//! constants, the repository comes from one manifest, version control from one `git`
13//! call, the toolchains from a handful of `exists()`, the provider projections from the
14//! policy and its templates, the services from one small file and at most one bounded
15//! connection attempt. The three answers that are not cheap — what the layer holds,
16//! what the workflow runner says, and what versions are installed — come from
17//! [`super::cache`], each with its own fingerprint, and are reported as unavailable when
18//! no cache can supply them.
19//!
20//! Unavailable is a real answer here, and the discipline the whole module is built on: a
21//! count that was not counted is absent, never zero, because a person reads a zero as a
22//! broken repository and acts on it.
23

39/// How many refused files the `layer_degraded` warning names before it counts the rest. A
40/// warning a terminal truncates says nothing; the whole list is in `layer.refused`.
41const LAYER_REFUSALS_NAMED: usize = 3;
42
43/// What a caller wants resolved, and what it will pay for.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub struct EnvironmentQuery {
46    /// How much may be read.
47    pub resolution: Resolution,
48    /// Whether a running server's address may be contacted. A resolution that says no
49    /// reports every service `unknown`, which is what it knows.
50    pub probe_services: bool,
51    /// Whether the cache may supply the tiers this resolution cannot compute.
52    pub use_cache: bool,
53    /// Whether what was computed is written back for the next fast resolution.
54    pub write_cache: bool,
55}
56
57impl EnvironmentQuery {
58    /// What a shell prompt asks for: everything cheap, the cache for the rest, a bounded
59    /// probe of the local address, and no write of tiers it did not compute.
60    pub fn fast() -> Self {
61        EnvironmentQuery {
62            resolution: Resolution::Fast,
63            probe_services: true,
64            use_cache: true,
65            write_cache: true,
66        }
67    }
68
69    /// What a person waiting on a command asks for: everything, written back.
70    pub fn full() -> Self {
71        EnvironmentQuery {
72            resolution: Resolution::Full,
73            probe_services: true,
74            use_cache: false,
75            write_cache: true,
76        }
77    }
78
79    /// The same, without touching anything outside this process: no probe, no cache read,
80    /// no cache write. What a test and a hosted request use.
81    pub fn sealed(self) -> Self {
82        EnvironmentQuery {
83            probe_services: false,
84            use_cache: false,
85            write_cache: false,
86            ..self
87        }
88    }
89}

91/// What the resolver may read. A fast resolution carries no index and no registry; a full
92/// one carries both, and the presence of the index is what makes it full.
93pub struct Inputs<'a> {
94    /// The repository, already discovered.
95    pub repository: &'a Repository,
96    /// The distribution directory, when it was located.
97    pub share: Option<&'a Share>,
98    /// The index, in a full resolution.
99    pub index: Option<&'a Index>,
100    /// The registry, in a full resolution.
101    pub registry: Option<&'a CapabilityRegistry>,
102    /// The policy, when the caller has already read it. The resolver reads it itself when
103    /// this is `None`, which is what every caller that needs nothing else from it does.
104    ///
105    /// It is here for the one caller that does: `majordomus env enter` asks the policy
106    /// whether entry ensures a runtime, and then resolves a snapshot whose provider
107    /// projections come out of the same file. Two reads of one canonical file inside one
108    /// invocation is what `project.hot-path-reads-once` is about, and this is the hot path
109    /// by definition — it runs on every `cd`.
110    pub policy: Option<&'a crate::policy::LoadedPolicy>,
111}

113/// Build a snapshot.
114///
115/// Never fails. Anything that cannot be read becomes a diagnostic and an absent value; a
116/// snapshot is the report of what could be learnt, and a resolver that refused to answer
117/// because one thing was missing would take the shell down with it.
118pub fn resolve(inputs: &Inputs<'_>, query: &EnvironmentQuery) -> RepositoryEnvironment {
119    let root = inputs.repository.root();
120    let local_half = inputs.repository.local_path();
121    let mut diagnostics = Vec::new();
122    let mut provenance = Vec::new();
123
124    let project = ProjectIdentity::of_this_build();
125    provenance.push(FieldSource::exact(
126        "project.version",
127        Some(project.version.clone()),
128        "the crate manifest, at compile time (CARGO_PKG_VERSION)",
129        "environment::resolve",
130    ));
131    provenance.push(FieldSource::exact(
132        "project.summary",
133        Some(project.summary.clone()),
134        "majordomus_cli::about::SUMMARY",
135        "environment::resolve",
136    ));
137    provenance.push(FieldSource::exact(
138        "project.commit",
139        Some(project.commit.clone()),
140        "build.rs, at compile time (MAJORDOMUS_COMMIT)",
141        "environment::resolve",
142    ));
143
144    let repository = repository_identity(inputs.repository);
145    provenance.push(FieldSource::exact(
146        "repository.name",
147        Some(repository.name.clone()),
148        "the base name of the repository root",
149        "environment::resolve",
150    ));
151    provenance.push(FieldSource::exact(
152        "repository.sections",
153        Some(format!("{} section(s)", repository.sections.len())),
154        crate::repository::MANIFEST,
155        "environment::resolve",
156    ));
157
158    // ---------------------------------------------------------------- version control
159    let vcs = vcs::inspect(root);
160    match &vcs {
161        VcsState::Git(tree) => {
162            provenance.push(FieldSource::exact(
163                "vcs.branch",
164                tree.branch.clone(),
165                vcs::SOURCE,
166                "environment::vcs",
167            ));
168            provenance.push(FieldSource::exact(
169                "vcs.clean",
170                Some(tree.clean.to_string()),
171                vcs::SOURCE,
172                "environment::vcs",
173            ));
174            provenance.push(match (tree.ahead, tree.behind) {
175                (Some(a), Some(b)) => FieldSource::exact(
176                    "vcs.ahead_behind",
177                    Some(format!("{a}/{b}")),
178                    vcs::SOURCE,
179                    "environment::vcs",
180                ),
181                _ => FieldSource::unknown(
182                    "vcs.ahead_behind",
183                    "the branch tracks no upstream",
184                    "environment::vcs",
185                ),
186            });
187        }
188        VcsState::Unavailable { reason } => {
189            diagnostics.push(Diagnostic::warning(
190                "vcs_unavailable",
191                None,
192                format!("version control could not be read: {reason}"),
193            ));
194            provenance.push(FieldSource::unknown(
195                "vcs.branch",
196                format!("{}: {reason}", vcs::SOURCE),
197                "environment::vcs",
198            ));
199        }
200    }
201
202    let mut cache = if query.use_cache || query.write_cache {
203        Cache::load(root, &local_half)
204    } else {
205        Cache::default()
206    };
207
208    // ---------------------------------------------------------------- what the layer holds
209    let layer_fingerprint = layer_fingerprint(inputs.repository, &vcs);
210    let layer = match (inputs.index, inputs.registry) {
211        (Some(index), Some(registry)) => {
212            let summary = summarise(index, registry);
213            provenance.push(FieldSource::exact(
214                "layer.objects",
215                summary.objects.map(|n| n.to_string()),
216                "the index this process built",
217                "environment::resolve",
218            ));
219            summary
220        }
221        _ => {
222            let cached = query
223                .use_cache
224                .then_some(cache.tiers.layer.as_ref())
225                .flatten()
226                .and_then(|e| e.fresh(&layer_fingerprint, None));
227            match cached {
228                Some(summary) => {
229                    let mut summary = summary.clone();
230                    summary.state = TierState::Cached;
231                    provenance.push(FieldSource::cached(
232                        "layer.objects",
233                        summary.objects.map(|n| n.to_string()),
234                        format!(
235                            "{}, written by the last full resolution",
236                            super::cache::CACHE_PATH
237                        ),
238                        "environment::cache",
239                    ));
240                    summary
241                }
242                None => {
243                    diagnostics.push(Diagnostic::info(
244                        "environment_layer_not_counted",
245                        None,
246                        "what the layer holds is not in the cache for this state of the repository; `majordomus env status` counts it and writes the cache",
247                    ));
248                    provenance.push(FieldSource::unknown(
249                        "layer.objects",
250                        "a fast resolution does not build the index, and no cache entry matched",
251                        "environment::cache",
252                    ));
253                    LayerSummary::unavailable()
254                }
255            }
256        }
257    };
258    if layer.degraded == Some(true) {
259        // Name them here. The count alone sent a reader to `majordomus capabilities
260        // validate`, which validates capabilities against their interfaces and reports
261        // nothing about a file the index refused — the remedy named an answer it does not
262        // have, so the only place the list existed was a log line.
263        let named: Vec<String> = layer
264            .refused
265            .iter()
266            .take(LAYER_REFUSALS_NAMED)
267            .map(|d| match &d.path {
268                Some(p) => format!("{p} — {}", d.message),
269                None => d.message.clone(),
270            })
271            .collect();
272        let total = layer.invalid.unwrap_or(layer.refused.len());
273        let rest = total.saturating_sub(named.len());
274        let list = match (named.is_empty(), rest) {
275            (true, _) => String::new(),
276            (false, 0) => format!(": {}", named.join("; ")),
277            (false, n) => format!(": {}; and {n} more", named.join("; ")),
278        };
279        diagnostics.push(Diagnostic::warning(
280            "layer_degraded",
281            None,
282            format!("{total} declared file(s) did not become objects{list}"),
283        ));
284    }
285
286    // ---------------------------------------------------------------- workflows
287    let workflow_fingerprint = workflow_fingerprint(root);
288    let cached_workflows = query
289        .use_cache
290        .then_some(cache.tiers.workflows.as_ref())
291        .flatten()
292        .and_then(|e| e.fresh(&workflow_fingerprint, None));
293    let workflows = match cached_workflows {
294        Some(catalogue) => {
295            let mut catalogue = catalogue.clone();
296            catalogue.state = TierState::Cached;
297            provenance.push(FieldSource::cached(
298                "workflows",
299                Some(format!("{} workflow(s)", catalogue.workflows.len())),
300                format!("{}, keyed by the workflow files", super::cache::CACHE_PATH),
301                "environment::cache",
302            ));
303            catalogue
304        }
305        None => {
306            // Asking the runner costs two short subprocesses, which a fast resolution can
307            // afford when the answer is not already known; it is the index it cannot.
308            let catalogue = workflows::resolve(root);
309            provenance.push(match catalogue.state {
310                TierState::Unavailable => FieldSource::unknown(
311                    "workflows",
312                    "no workflow runner answered for this repository",
313                    "environment::workflows",
314                ),
315                _ => FieldSource::exact(
316                    "workflows",
317                    Some(format!("{} workflow(s)", catalogue.workflows.len())),
318                    workflows::SOURCE,
319                    "environment::workflows",
320                ),
321            });
322            catalogue
323        }
324    };
325
326    // ---------------------------------------------------------------- toolchains
327    let declared = toolchain::declared(root);
328    let toolchain_fingerprint = fingerprint_of(
329        root,
330        &declared
331            .iter()
332            .map(|t| t.declared_by.clone())
333            .collect::<Vec<_>>(),
334        &["toolchains"],
335    );
336    let toolchains = if query.resolution == Resolution::Full {
337        let resolved = toolchain::with_installed(declared);
338        provenance.push(FieldSource::exact(
339            "toolchains",
340            Some(format!("{} declared", resolved.len())),
341            "each toolchain's own --version, bounded",
342            "environment::toolchain",
343        ));
344        resolved
345    } else {
346        let cached = query
347            .use_cache
348            .then_some(cache.tiers.toolchains.as_ref())
349            .flatten()
350            .and_then(|e| e.fresh(&toolchain_fingerprint, Some(TOOLCHAIN_LIFETIME)))
351            .cloned();
352        match cached {
353            Some(resolved) => {
354                provenance.push(FieldSource::cached(
355                    "toolchains",
356                    Some(format!("{} declared", resolved.len())),
357                    format!("{}, within its lifetime", super::cache::CACHE_PATH),
358                    "environment::cache",
359                ));
360                resolved
361            }
362            None => {
363                provenance.push(FieldSource::exact(
364                    "toolchains",
365                    Some(format!("{} declared", declared.len())),
366                    "the file that declares each one; no version was asked",
367                    "environment::toolchain",
368                ));
369                declared
370            }
371        }
372    };
373    for missing in toolchains
374        .iter()
375        .filter(|t| t.availability == ToolchainAvailability::Missing)
376    {
377        diagnostics.push(Diagnostic::warning(
378            "toolchain_missing",
379            Some(missing.declared_by.clone()),
380            format!("{} is declared here and is not installed", missing.title),
381        ));
382    }
383
384    // ---------------------------------------------------------------- providers
385    let (providers, provider_source) =
386        provider_states(inputs.repository, inputs.share, inputs.policy);
387    provenance.push(FieldSource::exact(
388        "providers",
389        Some(format!("{} projection(s)", providers.len())),
390        provider_source,
391        "environment::resolve",
392    ));
393    for stale in providers
394        .iter()
395        .filter(|p| matches!(p.state, ProjectionState::Stale | ProjectionState::Absent))
396    {
397        diagnostics.push(Diagnostic::warning(
398            "projection_stale",
399            Some(stale.target.clone()),
400            format!(
401                "{} is {} against the policy that renders it; `majordomus generate` rewrites it",
402                stale.target,
403                match stale.state {
404                    ProjectionState::Absent => "missing",
405                    _ => "stale",
406                }
407            ),
408        ));
409    }
410
411    // ---------------------------------------------------------------- services
412    let services = services::resolve(root, &local_half, query.probe_services);
413    provenance.push(match services.first().and_then(|s| s.url.clone()) {
414        Some(_) => FieldSource::exact(
415            "services.url",
416            services.first().and_then(|s| s.url.clone()),
417            format!("{}/{}", local_half, crate::lease::LEASE_PATH),
418            "environment::services",
419        ),
420        None => FieldSource::unknown(
421            "services.url",
422            format!(
423                "no server holds the lease at {}/{}",
424                local_half,
425                crate::lease::LEASE_PATH
426            ),
427            "environment::services",
428        ),
429    });
430
431    let environment = RepositoryEnvironment {
432        schema: RepositoryEnvironment::schema_id(),
433        generated_at: crate::peers::rfc3339(std::time::SystemTime::now()),
434        resolution: query.resolution,
435        project,
436        repository,
437        vcs,
438        toolchains,
439        layer,
440        workflows,
441        providers,
442        services,
443        diagnostics,
444        provenance,
445    };
446
447    if query.write_cache {
448        // Only what this resolution actually computed is written back; a tier taken from
449        // the cache is rewritten with the same fingerprint, which is a no-op, and a tier
450        // nothing resolved is left as it was rather than being replaced with an absence.
451        if environment.layer.state == TierState::Resolved {
452            cache.tiers.layer = Some(Cache::entry(layer_fingerprint, environment.layer.clone()));
453        }
454        if environment.workflows.state == TierState::Resolved {
455            cache.tiers.workflows = Some(Cache::entry(
456                workflow_fingerprint,
457                environment.workflows.clone(),
458            ));
459        }
460        if environment
461            .toolchains
462            .iter()
463            .any(|t| t.availability != ToolchainAvailability::Unknown)
464        {
465            cache.tiers.toolchains = Some(Cache::entry(
466                toolchain_fingerprint,
467                environment.toolchains.clone(),
468            ));
469        }
470        if let Err(e) = cache.store(root, &local_half) {
471            tracing::debug!(error = %e, "the environment cache could not be written");
472        }
473    }
474
475    environment
476}

478/// The identity of the checkout.
479fn repository_identity(repository: &Repository) -> RepositoryIdentity {
480    let root = repository.root();
481    RepositoryIdentity {
482        name: root
483            .file_name()
484            .map(|n| n.to_string_lossy().into_owned())
485            .unwrap_or_else(|| root.display().to_string()),
486        root: root.display().to_string(),
487        layer_schema: repository.manifest().schema.clone(),
488        sections: repository
489            .manifest()
490            .sections
491            .iter()
492            .map(|(name, path)| (name.clone(), format!(".ai/{path}")))
493            .collect(),
494        local_path: repository.local_path(),
495        // A linked work tree has a `.git` file rather than a `.git` directory. One stat.
496        linked_worktree: root.join(".git").is_file(),
497    }
498}

500/// What the layer holds, from the index and the registry that already hold it.
501fn summarise(index: &Index, registry: &CapabilityRegistry) -> LayerSummary {
502    // The index already decided each of these and said why; carrying the diagnostics rather
503    // than their number is what lets every reader of the snapshot name the file.
504    let refused: Vec<crate::model::Diagnostic> = index
505        .diagnostics
506        .iter()
507        .filter(|d| d.severity == Severity::Error)
508        .cloned()
509        .collect();
510    LayerSummary {
511        state: TierState::Resolved,
512        kinds: index
513            .kinds()
514            .into_iter()
515            .map(|(kind, count)| KindCount {
516                kind: kind.to_string(),
517                count,
518            })
519            .collect(),
520        objects: Some(index.objects.len()),
521        capabilities: Some(registry.summary().total),
522        invalid: Some(refused.len()),
523        refused,
524        degraded: Some(index.state == crate::index::State::Degraded),
525    }
526}

528/// The provider projections the policy declares, each against the file it renders to.
529fn provider_states(
530    repository: &Repository,
531    share: Option<&Share>,
532    already: Option<&crate::policy::LoadedPolicy>,
533) -> (Vec<ProviderState>, String) {
534    let source = format!(
535        "{}, projections[]",
536        repository
537            .section_path("policy")
538            .unwrap_or_else(|| ".ai/repo/policy.yaml".into())
539    );
540    let read;
541    let policy = match already {
542        Some(policy) => policy,
543        None => {
544            let Ok(loaded) = crate::policy::LoadedPolicy::load(repository) else {
545                return (Vec::new(), format!("{source} (unreadable)"));
546            };
547            read = loaded;
548            &read
549        }
550    };
551    // Rendering needs the templates, which live in the distribution. Without it the
552    // projections are still known — the policy declares them — but whether each file is
553    // current is not, and `Unknown` says exactly that.
554    let rendered = share.and_then(|share| {
555        crate::providers::artifacts(repository, share, policy)
556            .ok()
557            .map(|artifacts| {
558                artifacts
559                    .into_iter()
560                    .map(|a| {
561                        let state = crate::providers::state_of(repository.root(), &a);
562                        (a.path, state)
563                    })
564                    .collect::<Vec<_>>()
565            })
566    });
567    let states = policy
568        .policy
569        .projections
570        .iter()
571        .map(|p| ProviderState {
572            id: p.provider.clone(),
573            target: p.target.clone(),
574            always_loaded: p.always_loaded,
575            state: match rendered
576                .as_ref()
577                .and_then(|r| r.iter().find(|(path, _)| *path == p.target))
578            {
579                Some((_, crate::providers::TargetState::Current)) => ProjectionState::Current,
580                Some((_, crate::providers::TargetState::Stale)) => ProjectionState::Stale,
581                Some((_, crate::providers::TargetState::Absent)) => ProjectionState::Absent,
582                None => ProjectionState::Unknown,
583            },
584        })
585        .collect();
586    (states, source)
587}

589/// The fingerprint of everything that can change what the layer holds: the commit, the
590/// files that configure discovery, and whatever is dirty under a section of the layer.
591///
592/// The dirty set is what makes this exact rather than approximate. A rule edited and not
593/// committed moves no commit and touches no configuration file, so a fingerprint over
594/// those alone would keep serving the old count; the file appears in the status report
595/// this snapshot has already paid for, and its size and modification time go into the
596/// fingerprint at the cost of one `stat`.
597fn layer_fingerprint(repository: &Repository, vcs: &VcsState) -> String {
598    let mut paths: Vec<String> = vec![
599        crate::repository::MANIFEST.to_string(),
600        ".ai/repo/knowledge/sources.yaml".to_string(),
601        ".ai/repo/knowledge/kinds.yaml".to_string(),
602        "share/kinds.yaml".to_string(),
603    ];
604    if let Some(policy) = repository.section_path("policy") {
605        paths.push(policy);
606    }
607    if let Some(scope) = repository.section_path("scope") {
608        paths.push(scope);
609    }
610    let head = vcs
611        .tree()
612        .and_then(|t| t.head.clone())
613        .unwrap_or_else(|| "unborn".into());
614    if let Some(tree) = vcs.tree() {
615        // only the dirty paths that could be objects of the layer, so that a change to an
616        // unrelated file costs nothing
617        paths.extend(
618            tree.changed_paths
619                .iter()
620                .filter(|p| is_layer_path(repository, p))
621                .cloned(),
622        );
623    }
624    fingerprint_of(repository.root(), &paths, &["layer", &head])
625}

627/// Could a repository-relative path be an object of the layer? Derived from the manifest's
628/// own sections plus the two trees outside `.ai/` that the source classes reach into.
629fn is_layer_path(repository: &Repository, path: &str) -> bool {
630    if path.starts_with(&repository.local_path()) {
631        return false;
632    }
633    path.starts_with(".ai/")
634        || path.starts_with("docs/")
635        || path.starts_with("share/")
636        || path.starts_with("lib/")
637        || path.starts_with("test/cases/")
638        || path.starts_with("bin/")
639}

641/// The fingerprint of the workflow files: the justfile and whatever it imports.
642fn workflow_fingerprint(root: &Path) -> String {
643    let mut paths = vec!["justfile".to_string(), ".justfile".to_string()];
644    // One directory read of the module directory, when there is one. A justfile split into
645    // modules must expire when any of them changes, and reading one directory is the whole
646    // cost of knowing that.
647    if let Ok(entries) = std::fs::read_dir(root.join(".just")) {
648        let mut names: Vec<String> = entries
649            .flatten()
650            .filter_map(|e| e.file_name().into_string().ok())
651            .map(|name| format!(".just/{name}"))
652            .collect();
653        crate::order::canonical(&mut names);
654        paths.extend(names);
655    } else {
656        // The absence of the directory is itself part of the fingerprint: creating it must
657        // expire the catalogue.
658        paths.push(".just".to_string());
659    }
660    fingerprint_of(root, &paths, &["workflows"])
661}

825    /// The digest answers "is there anything new to show a person", so it moves when a
826    /// count moves. It deliberately does not move when a file's bytes change without
827    /// changing what the snapshot says: a banner that turned into news on every keystroke
828    /// would be one a person learns to ignore.
829    #[test]
830    fn the_digest_follows_what_the_snapshot_says_and_not_every_byte() {
831        let repo = repository();
832        let before = full(&repo).digest();
833
834        repo.touch_rule(0, "-changed").expect("a changed rule");
835        assert_eq!(
836            before,
837            full(&repo).digest(),
838            "editing a rule's body changes no count and so is not news"
839        );
840
841        std::fs::write(
842            repo.root().join(".ai/repo/rules/project/rule-99.v1.md"),
843            "---\nid: project.rule-99\nversion: 1\nkind: rule\ntitle: T\ndescription: D\nstatement: S\nstatus: active\nclass: advisory\n---\n\n# Rationale\n\nx\n",
844        )
845        .expect("a new rule");
846        assert_ne!(
847            before,
848            full(&repo).digest(),
849            "a rule that did not exist before is news"
850        );
851    }
