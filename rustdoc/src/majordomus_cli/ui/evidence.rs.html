1//! Evidence: which claims of the product hold, judged at the presented revision. Read from
2//! `evidence.report`: its totals, its claims with their state, and its findings.
3
4use crate::capability::Context;
5use crate::cockpit::html::el;

7use super::dashboard::ROWS;
8use super::shell::View;
9use super::source::{ask_all, Source};
10use super::view::{
11    counts, donut, hero, kpi, kpi_count, list, panel, state, table, text, unanswered, wide,
12};

14/// The route of the screen.
15pub const ROUTE: &str = "/ui/evidence";
16
17const EVIDENCE: Source = Source::of("evidence.report");
18
19/// The Evidence screen; `query` may pick one claim to inspect (`?claim=<id>`).
20pub fn screen(ctx: &Context, query: &[(String, String)]) -> View {
21    let answers = ask_all(ctx, &[EVIDENCE]);
22    let evidence = &answers[&EVIDENCE];
23    let picked = super::claim::picked(query);
24    let inspected = picked
25        .as_deref()
26        .map(|id| super::claim::inspector(ctx, &super::claim::ask(ctx, id), id));
27
28    let top = hero(
29        "file-check",
30        "green",
31        "Evidence",
32        "Truth over assumptions: each claim with the test that settles it.",
33        "Every claim the product makes, judged against the revision presented, with the findings that keep a claim from holding.",
34        vec![
35            kpi_count("Claims", &evidence.read("/claims")),
36            kpi_count("Findings", &evidence.read("/findings")),
37            kpi("Revision", &evidence.read("/presented/revision")),
38            kpi("Ledger runs", &evidence.read("/ledger/executions")),
39        ],
40    );
41
42    let findings = unanswered(evidence).unwrap_or_else(|| {
43        let rows = list(evidence, "/findings")
44            .iter()
45            .take(ROWS * 2)
46            .map(|f| {
47                el("tr")
48                    .child(
49                        el("td").child(
50                            el("a")
51                                .class("mj-ui-mono")
52                                .attr("href", super::claim::href(&text(f, "claim")))
53                                .text(text(f, "claim")),
54                        ),
55                    )
56                    .child(el("td").child(state(f, "state")))
57                    .child(el("td").class("mj-ui-dim").text(text(f, "reason")))
58            })
59            .collect::<Vec<_>>();
60        table(
61            &["Claim", "State", "Why it does not hold"],
62            rows,
63            "Every claim holds.",
64        )
65    });
66
67    let claims = unanswered(evidence).unwrap_or_else(|| {
68        let rows = list(evidence, "/claims")
69            .iter()
70            .take(ROWS * 2)
71            .map(|c| {
72                el("tr")
73                    .child(
74                        el("td").child(
75                            el("a")
76                                .class("mj-ui-row-link")
77                                .attr("href", super::claim::href(&text(c, "id")))
78                                .child(el("span").class("mj-ui-mono mj-ui-dim").text(text(c, "id")))
79                                .child(el("span").class("mj-ui-cell-title").text(text(c, "claim"))),
80                        ),
81                    )
82                    .child(el("td").child(state(c, "state")))
83                    .child(
84                        el("td")
85                            .class("mj-ui-mono mj-ui-dim")
86                            .text(text(c, "test_path")),
87                    )
88            })
89            .collect::<Vec<_>>();
90        table(&["Claim", "State", "Settled by"], rows, "No claim is made.")
91    });
92
93    let mut grid = el("div").class("mj-ui-grid");
94    if let Some(inspector) = inspected {
95        grid = grid.child(wide(inspector));
96    }
97    let grid = grid
98        .child(panel(
99            "Claims without a verdict",
100            "The claims the report could not settle, by why.",
101            unanswered(evidence).unwrap_or_else(|| donut("claims", &counts(evidence, "/totals"))),
102            &[evidence],
103        ))
104        .child(panel(
105            "Findings",
106            "What keeps a claim from holding, as the report found it.",
107            findings,
108            &[evidence],
109        ))
110        .child(wide(panel(
111            "Claims",
112            "The product's claims with the test that settles each.",
113            claims,
114            &[evidence],
115        )));
116
117    View::new(
118        "Evidence",
119        el("div").class("mj-ui-stack").child(top).child(grid),
120    )
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126    use crate::ui::deployments::tests::furnished;
127
128    /// A guarantee no test supports is a finding with its reason, and every claim is a row
129    /// with its state and the test that would settle it.
130    #[test]
131    fn every_claim_is_a_row_and_an_unsupported_guarantee_is_a_finding() {
132        let repo = furnished();
133        let ctx = repo.context().expect("a context");
134        let html = screen(&ctx, &[]).main.render();
135        assert!(
136            html.contains("the claim guarantees a behaviour and names no test at all"),
137            "{html}"
138        );
139        assert!(html.contains("synthetic-guarantee"), "{html}");
140        assert!(html.contains("A guarantee no test supports"), "{html}");
141        assert!(html.contains("An advisory claim"), "{html}");
142        assert!(html.contains("test/cases/1_synthetic.sh"), "{html}");
143        assert!(
144            html.contains("no_test") || html.contains("no test"),
145            "{html}"
146        );
147        assert!(!html.contains("Every claim holds."), "{html}");
148        assert!(!html.contains("No claim is made."), "{html}");
149    }
