1//! The `executions` module: starting a capability as an execution, reading what it is
2//! doing, and asking it to stop.
3//!
4//! Every capability here is about executions; none of them is an execution engine of its
5//! own. What runs the work is [`crate::execution::ExecutionEngine`], reached through the
6//! context every handler already has, so a browser posting to `/api/v1/executions/start`,
7//! an agent calling `majordomus_execution_start` and a terminal running `majordomus run`
8//! reach one implementation and one store.
9//!
10//! There is no capability here that lists what may be run: that is `capabilities.list`,
11//! which already answers it, and a second list would be the same fact written twice. Every
12//! executable capability of the registry can be started as an execution, and the
13//! descriptor's own `execution` policy is what says what running it means.
14

48/// Where to read more about one execution, in this server's own terms.
49///
50/// Every value is derived from the registry's declared routes and from the live channel's
51/// own constant, so a route that moves moves here too and no client holds a path this
52/// server does not serve.
53#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
54pub struct ExecutionLinks {
55    /// The snapshot: `GET /api/v1/executions/get?id=…`.
56    #[serde(rename = "self")]
57    pub itself: String,
58    /// The retained history: `GET /api/v1/executions/events?id=…`.
59    pub events: String,
60    /// Where to ask it to stop: `POST /api/v1/executions/cancel`.
61    pub cancel: String,
62    /// The live channel, from the beginning: `GET /events?execution=…`.
63    pub websocket: String,
64    /// The page a person opens.
65    pub cockpit: String,
66}
67
68impl ExecutionLinks {
69    /// The links for one execution, built from what the registry declares.
70    pub fn of(ctx: &Context, id: &ExecutionId) -> Self {
71        let route = |capability: &str| {
72            ctx.registry
73                .get(capability)
74                .and_then(|c| c.exposure.http.as_ref())
75                .map(|h| h.path.clone())
76                .unwrap_or_default()
77        };
78        let query = crate::http::router::percent_encode(id.as_str());
79        ExecutionLinks {
80            itself: format!("{}?id={query}", route("executions.get")),
81            events: format!("{}?id={query}", route("executions.events")),
82            cancel: route("executions.cancel"),
83            websocket: format!("{}?execution={query}", crate::http::events::PATH),
84            cockpit: format!("{}/executions/{query}", crate::cockpit::PREFIX),
85        }
86    }
87}

107    ExecutionId::parse(text).ok_or_else(|| {
108        CapabilityError::InvalidInput(format!(
109            "'{text}' is not an execution id; they look like x-20260908T010203Z-0a1b2c3d"

114fn actor(ctx: &Context) -> Actor {
115    match &ctx.caller {
116        Some(peer) => Actor {
117            kind: ActorKind::Mcp,
118            peer: Some(peer.to_string()),
119        },
120        None => Actor::of(ActorKind::Http),
121    }

149fn executions_start(ctx: &Context, input: StartInput) -> Result<ExecutionView, CapabilityError> {
150    let payload = input.input.unwrap_or_else(|| serde_json::json!({}));
151    match ctx
152        .executions
153        .submit(ctx, &input.capability, payload, actor(ctx))
154    {
155        Ok(execution) => Ok(ExecutionView::of(ctx, execution)),
156        Err(SubmitError::UnknownCapability(id)) => Err(CapabilityError::NotFound(format!(
157            "no capability '{id}'; the executable ones are listed by capabilities.list"
158        ))),
159        Err(e @ SubmitError::NotExecutable(_)) => Err(CapabilityError::Refused(e.to_string())),
160        Err(SubmitError::InvalidInput(m)) => Err(CapabilityError::InvalidInput(m)),
161        Err(e @ SubmitError::Unavailable(_)) => Err(CapabilityError::Refused(e.to_string())),
162    }
163}

198#[derive(Debug, Serialize, Deserialize, JsonSchema)]
199/// The answer of `executions.list`.
200pub struct ExecutionList {
201    /// How many are answered here.
202    pub count: usize,
203    /// How many this process remembers in total, before the filter.
204    pub remembered: usize,
205    /// How many are not finished.
206    pub active: usize,
207    /// How many are waiting for a worker.
208    pub queued: usize,
209    /// How many live channels are following them.
210    pub live_channels: usize,
211    /// The executions, newest first.
212    pub executions: Vec<ExecutionView>,
213}
214
215fn executions_list(ctx: &Context, input: ListInput) -> Result<ExecutionList, CapabilityError> {
216    let store = ctx.executions.store();
217    let limit = input.limit.unwrap_or(MAX_LIST).min(MAX_LIST);
218    let executions = store.list(input.state, input.capability.as_deref(), limit);
219    Ok(ExecutionList {
220        count: executions.len(),
221        remembered: store.len(),
222        active: store.active(),
223        queued: ctx.executions.queued(),
224        live_channels: crate::http::events::open_connections(),
225        executions: executions
226            .into_iter()
227            .map(|e| ExecutionView::of(ctx, e))
228            .collect(),
229    })
230}

283#[derive(Debug, Serialize, Deserialize, JsonSchema)]
284/// The answer of `executions.events`: one page of an execution's retained history.
285pub struct EventHistory {
286    /// The execution the events belong to.
287    pub execution_id: String,
288    /// Where the execution stands now, so a client needs one request rather than two.
289    pub state: ExecutionState,
290    /// The events, oldest first.
291    pub events: Vec<ExecutionEvent>,
292    /// The sequence of the last event here: the cursor for the next page, and the one to
293    /// open the live channel with.
294    pub last_sequence: u64,
295    /// Whether more events follow this page right now.
296    pub more: bool,
297    /// Whether events before this page had already been dropped by the store's bound.
298    pub truncated: bool,
299}
300
301fn executions_events(ctx: &Context, input: EventsInput) -> Result<EventHistory, CapabilityError> {
302    let id = parse_id(&input.id)?;
303    let store = ctx.executions.store();
304    let snapshot = store
305        .get(&id)
306        .ok_or_else(|| CapabilityError::NotFound(format!("no execution {id} in this process")))?;
307    let limit = input.limit.unwrap_or(MAX_EVENTS).min(MAX_EVENTS);
308    let page = store
309        .events(&id, input.after.unwrap_or(0), limit)
310        .ok_or_else(|| CapabilityError::NotFound(format!("no execution {id} in this process")))?;
311    Ok(EventHistory {
312        execution_id: id.to_string(),
313        state: snapshot.state,
314        events: page.events.iter().map(|e| (**e).clone()).collect(),
315        last_sequence: page.last_sequence,
316        more: page.more,
317        truncated: page.truncated,
318    })
319}

337#[derive(Debug, Serialize, Deserialize, JsonSchema)]
338/// The answer of `executions.cancel`.
339pub struct CancelReport {
340    /// What asking achieved: `requested`, `already_requested` or `already_finished`.
341    pub outcome: String,
342    /// Whether the capability declares that it looks at its cancellation flag. When it
343    /// does not, the request is recorded and the execution runs to completion.
344    pub cancellable: bool,
345    /// The execution as it stands after the request.
346    pub execution: ExecutionView,
347}
348
349fn executions_cancel(ctx: &Context, input: CancelInput) -> Result<CancelReport, CapabilityError> {
350    let id = parse_id(&input.id)?;
351    let by = match &ctx.caller {
352        Some(peer) => peer.to_string(),
353        None => "client".to_string(),
354    };
355    let outcome = match ctx.executions.cancel(&id, &by) {
356        CancelOutcome::Requested => "requested",
357        CancelOutcome::AlreadyRequested => "already_requested",
358        CancelOutcome::AlreadyFinished(_) => "already_finished",
359        CancelOutcome::Unknown => {
360            return Err(CapabilityError::NotFound(format!(
361                "no execution {id} in this process"
362            )))
363        }
364    };
365    let execution =
366        ctx.executions.store().get(&id).ok_or_else(|| {
367            CapabilityError::NotFound(format!("no execution {id} in this process"))
368        })?;
369    let cancellable = execution.cancellable;
370    Ok(CancelReport {
371        outcome: outcome.into(),
372        cancellable,
373        execution: ExecutionView::of(ctx, execution),
374    })
375}

433fn variants_of(schema: &Value) -> Vec<String> {
434    schema
435        .get("oneOf")
436        .and_then(Value::as_array)
437        .map(|all| {
438            all.iter()
439                .filter_map(|v| {
440                    v.pointer("/properties/type/const")
441                        .and_then(Value::as_str)
442                        .map(str::to_string)
443                })
444                .collect()
445        })
446        .unwrap_or_default()
447}

450    let event_schema = CanonicalSchema::of::<ExecutionEvent>().schema;

452    let limits = ctx.executions.store().limits();

454        protocol_version: PROTOCOL_VERSION.into(),
455        websocket: crate::http::events::PATH.into(),
456        subscription: vec![

458                name: "execution".into(),
459                description: "An execution id to follow. Absent, the channel follows every execution of this process.".into(),
460                required: false,
461            },
462            ParameterView {
463                name: "after".into(),
464                description: "The sequence number the client last saw. The server replays the retained events after it, then goes live with no gap and nothing written twice.".into(),

466            },

468        heartbeat_seconds: crate::http::events::HEARTBEAT.as_secs(),

489/// The input of `executions.demonstrate`.
490pub struct DemonstrateInput {
491    /// How many steps to walk through.
492    #[serde(default = "three")]
493    pub steps: u64,
494    /// How long each step takes, in milliseconds. Bounded at ten seconds a step, so this
495    /// cannot be used to hold a worker.
496    #[serde(default)]
497    pub delay_ms: u64,
498    /// Fail on this step instead of completing, to show what a failure looks like.
499    #[serde(default, skip_serializing_if = "Option::is_none")]
500    pub fail_at: Option<u64>,
501}
502
503fn three() -> u64 {
504    3
505}

507impl BenchmarkCases for DemonstrateInput {
508    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
509        vec![NamedCase::new(
510            "immediate",
511            DemonstrateInput {
512                steps: 1,
513                delay_ms: 0,
514                fail_at: None,
515            },
516        )]
517    }

520#[derive(Debug, Serialize, Deserialize, JsonSchema)]
521/// The answer of `executions.demonstrate`.
522pub struct DemonstrateReport {
523    /// How many steps ran.
524    pub steps: u64,
525    /// How long it took, in milliseconds.
526    pub elapsed_ms: u64,
527    /// Whether anything was watching: false when it was called directly rather than as an
528    /// execution, which is the difference this capability exists to show.
529    pub observed: bool,
530}
531
532/// The longest one step of the demonstration may take.
533const MAX_STEP_MS: u64 = 10_000;
534
535/// The most steps the demonstration will walk.
536const MAX_STEPS: u64 = 100;
537
538fn executions_demonstrate(
539    ctx: &Context,
540    input: DemonstrateInput,
541) -> Result<DemonstrateReport, CapabilityError> {
542    let steps = input.steps.min(MAX_STEPS);
543    let delay = std::time::Duration::from_millis(input.delay_ms.min(MAX_STEP_MS));
544    let started = std::time::Instant::now();
545    let p = &ctx.progress;
546    for step in 1..=steps {
547        if p.cancelled() {
548            return Err(p.cancellation());
549        }
550        let name = format!("step-{step}");
551        p.step(&name, &format!("Step {step} of {steps}"));
552        p.log(format!("step {step} of {steps} is working"));
553        if !delay.is_zero() {
554            // slept in slices so that cancellation is noticed within a tenth of a second
555            // however long a step is
556            let slice = std::time::Duration::from_millis(100);
557            let mut left = delay;
558            while !left.is_zero() {
559                if p.cancelled() {
560                    p.step_done(&name, false, Some("cancelled".into()));
561                    return Err(p.cancellation());
562                }
563                let take = left.min(slice);
564                std::thread::sleep(take);
565                left -= take;
566            }
567        }
568        if input.fail_at == Some(step) {
569            p.step_done(&name, false, Some("asked to fail here".into()));
570            return Err(CapabilityError::Internal(format!(
571                "the demonstration was asked to fail at step {step} of {steps}"
572            )));
573        }
574        p.step_done(&name, true, None);
575        p.progress(step, Some(steps), format!("{step} of {steps} done"));
576    }
577    Ok(DemonstrateReport {
578        steps,
579        elapsed_ms: started.elapsed().as_millis() as u64,
580        observed: p.is_reporting(),
581    })
582}

693    /// from it. A dropped exposure would still compile and every behavioural test would

699        let expected: &[(&str, &str, &str, CapabilityKind)] = &[
700            (
701                "executions.start",
702                "majordomus_execution_start",

704                CapabilityKind::Command,

706            (
707                "executions.list",
708                "majordomus_executions",
709                "/api/v1/executions",
710                CapabilityKind::Query,
711            ),
712            (
713                "executions.get",
714                "majordomus_execution",
715                "/api/v1/executions/get",
716                CapabilityKind::Query,
717            ),
718            (
719                "executions.events",
720                "majordomus_execution_events",
721                "/api/v1/executions/events",
722                CapabilityKind::Query,
723            ),
724            (
725                "executions.cancel",
726                "majordomus_execution_cancel",
727                "/api/v1/executions/cancel",
728                CapabilityKind::Command,
729            ),
730            (
731                "executions.protocol",
732                "majordomus_execution_protocol",
733                "/api/v1/executions/protocol",
734                CapabilityKind::Query,
735            ),
736            (
737                "executions.demonstrate",
738                "majordomus_demonstrate_execution",
739                "/api/v1/executions/demonstrate",
740                CapabilityKind::Query,
741            ),
742        ];
743        let ids: Vec<&str> = m
744            .capabilities
745            .iter()
746            .map(|e| e.capability.id.as_str())
747            .collect();
748        assert_eq!(ids, expected.iter().map(|(id, ..)| *id).collect::<Vec<_>>());
749        for (executable, (id, tool, path, kind)) in m.capabilities.iter().zip(expected) {
750            let c = &executable.capability;
751            assert_eq!(
752                c.exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
753                Some(*tool),
754                "{id} lost or renamed its MCP tool"
755            );
756            assert_eq!(
757                c.exposure.http.as_ref().map(|h| h.path.as_str()),
758                Some(*path),
759                "{id} lost or renamed its HTTP route"
760            );
761            assert_eq!(c.kind, *kind, "{id} changed kind");
762            let base = crate::capability::ExecutionPolicy::classify(*kind);
763            assert!(
764                c.execution == base || c.execution == base.stoppable(),
765                "{id} carries a policy neither its kind nor a cancellation declaration gives it"
766            );
767        }
