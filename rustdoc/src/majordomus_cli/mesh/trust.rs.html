1//! Trust: what a verified advertisement is allowed to mean. Verification (the signature
2//! proves the key) happens in the protocol; trust (this key is one of ours) is decided
3//! here, by policy — and by default the policy is that an unknown key is observed,
4//! recorded, visible in every listing, and trusted for nothing.
5//!
6//! Discovery creates awareness, not authority: even a `Trusted` node gains no execution
7//! rights anywhere in this codebase. Trust gates which nodes surfaces mark as ours and
8//! which endpoints later negotiation may approach — nothing else.
9//!
10//! ```
11//! use majordomus_cli::mesh::trust::{evaluate, TrustPolicy, TrustState};
12//!
13//! // the default: a valid unknown key is observed and trusted for nothing
14//! assert_eq!(evaluate(&TrustPolicy::DenyUnknown, &[], "aa", None, None), TrustState::Observed);
15//! // an allowlisted key is trusted under every policy
16//! let listed = evaluate(&TrustPolicy::DenyUnknown, &["aa".into()], "aa", None, None);
17//! assert!(listed.is_trusted());
18//! ```
19//!
20//! A key can also be trusted through a [`Succession`] its predecessor signed, and only while
21//! the declaration in force trusts that predecessor ([`evaluate_with_successions`], ADR 0067
22//! addendum of 2026-10-15).
23

31/// The policy, from the mesh declaration.
32#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
33#[serde(rename_all = "snake_case")]
34pub enum TrustPolicy {
35    /// The default: a valid unknown node is recorded as observed and trusted for
36    /// nothing. The safe answer when nobody decided otherwise.
37    #[default]
38    DenyUnknown,
39    /// Trust-on-first-use: the first valid appearance of a key is trusted, a later
40    /// appearance of the same node id under a different key is rejected. A development
41    /// convenience, and every listing says the trust came from TOFU rather than a person.
42    Tofu,
43    /// Only keys the declaration lists are trusted; everything else is observed.
44    Allowlist,
45}
46
47impl TrustPolicy {
48    /// The policy's wire word, as declarations write it and listings show it.
49    ///
50    /// ```
51    /// use majordomus_cli::mesh::trust::TrustPolicy;
52    /// assert_eq!(TrustPolicy::DenyUnknown.as_str(), "deny_unknown");
53    /// ```
54    pub fn as_str(&self) -> &'static str {
55        match self {
56            TrustPolicy::DenyUnknown => "deny_unknown",
57            TrustPolicy::Tofu => "tofu",
58            TrustPolicy::Allowlist => "allowlist",
59        }
60    }
61}

63/// What the policy decided about one node.
64#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
65#[serde(rename_all = "snake_case", tag = "state", content = "detail")]
66pub enum TrustState {
67    /// Valid, recorded, trusted for nothing. The resting state of an unknown node.
68    Observed,
69    /// The policy trusts this key; `detail` says which rule did (`allowlist`, `tofu`).
70    Trusted(String),
71    /// The policy rejected this node; `detail` says why (for instance a TOFU key
72    /// change). A rejected node stays listed — an operator debugging a mesh needs to
73    /// see what was refused, not wonder about silence.
74    Rejected(String),
75    /// The key was succeeded by a trusted successor, whose key is the `detail`: its history
76    /// stays legitimate, and it is trusted for nothing new. Retired, not withdrawn.
77    Retired(String),
78}
79
80impl TrustState {
81    /// Whether this state trusts the node — the one question consumers ask; the detail
82    /// stays on the variant for whoever renders it.
83    pub fn is_trusted(&self) -> bool {
84        matches!(self, TrustState::Trusted(_))
85    }
86}

88/// Evaluate the policy for a node's key against what the registry already knows.
89///
90/// ```
91/// use majordomus_cli::mesh::trust::{evaluate, TrustPolicy, TrustState};
92/// // TOFU trusts first use; a key change under a known node id is rejected regardless
93/// assert!(evaluate(&TrustPolicy::Tofu, &[], "aa", None, None).is_trusted());
94/// assert!(matches!(
95///     evaluate(&TrustPolicy::Tofu, &[], "bb", Some("aa"), None),
96///     TrustState::Rejected(_)
97/// ));
98/// ```
99///
100/// `known_key` is the key the registry last associated with this node id — with node
101/// ids derived from keys the two cannot disagree, so a mismatch can only mean a digest
102/// collision or registry corruption, and it is rejected loudly.
103pub fn evaluate(
104    policy: &TrustPolicy,
105    allow: &[String],
106    public_key: &str,
107    known_key: Option<&str>,
108    previous: Option<&TrustState>,
109) -> TrustState {
110    if let Some(known) = known_key {
111        if known != public_key {
112            return TrustState::Rejected("the node id is already bound to a different key".into());
113        }
114    }
115    if allow.iter().any(|k| k == public_key) {
116        return TrustState::Trusted("allowlist".into());
117    }
118    match policy {
119        TrustPolicy::DenyUnknown => TrustState::Observed,
120        TrustPolicy::Allowlist => TrustState::Observed,
121        TrustPolicy::Tofu => match previous {
122            // The first use is what TOFU trusts; a node once rejected stays rejected.
123            Some(state @ TrustState::Rejected(_)) => state.clone(),
124            _ => TrustState::Trusted("tofu".into()),
125        },
126    }
127}

129/// [`evaluate`], following the successions a runtime holds (ADR 0067, addendum of
130/// 2026-10-15). Only statements that verify count; of those:
131///
132/// - a key that two different successors claim to succeed is a fork, which only a stolen
133///   key can produce: the key and both successors are `Rejected("succession fork")`, unless
134///   the allowlist names a successor itself;
135/// - a key with one successor that is trusted (the allowlist names the key or the successor)
136///   is `Retired(successor)`, whatever the allowlist says of the key;
137/// - a key the allowlist does not name is `Trusted("succession")` when a predecessor the
138///   allowlist names hands over to it. The depth is one: a key trusted only through a
139///   succession confers nothing on a successor of its own until the allowlist names it, and
140///   withdrawing a predecessor withdraws its successor with it.
141///
142/// ```
143/// use majordomus_cli::mesh::identity::{NodeIdentity, Succession};
144/// use majordomus_cli::mesh::trust::{evaluate_with_successions, TrustPolicy, TrustState};
145/// let old = NodeIdentity::ephemeral().unwrap();
146/// let new = NodeIdentity::ephemeral().unwrap();
147/// let (old_key, new_key) = (old.public.public_key.clone(), new.public.public_key.clone());
148/// let statements = [Succession::sign(&old, &new, "2026-10-15T08:00:00Z").unwrap()];
149/// let trust = |allow: &[String], key: &str| {
150///     evaluate_with_successions(&TrustPolicy::Allowlist, allow, &statements, key, None, None)
151/// };
152/// let allow = [old_key.clone()];
153/// assert_eq!(trust(&allow, &new_key), TrustState::Trusted("succession".into()));
154/// assert_eq!(trust(&allow, &old_key), TrustState::Retired(new_key.clone()));
155/// // a withdrawn predecessor confers nothing
156/// assert_eq!(trust(&[], &new_key), TrustState::Observed);
157/// ```
158pub fn evaluate_with_successions(
159    policy: &TrustPolicy,
160    allow: &[String],
161    successions: &[Succession],
162    public_key: &str,
163    known_key: Option<&str>,
164    previous: Option<&TrustState>,
165) -> TrustState {
166    if let Some(known) = known_key {
167        if known != public_key {
168            return TrustState::Rejected("the node id is already bound to a different key".into());
169        }
170    }
171    let valid: Vec<&Succession> = successions.iter().filter(|s| s.verify().is_ok()).collect();
172    let listed = |key: &str| allow.iter().any(|k| k == key);
173    let successors = |key: &str| -> BTreeSet<&str> {
174        valid
175            .iter()
176            .filter(|s| s.old_key == key)
177            .map(|s| s.new_key.as_str())
178            .collect()
179    };
180
181    let own = successors(public_key);
182    if own.len() > 1 {
183        return TrustState::Rejected("succession fork".into());
184    }
185    if let Some(successor) = own.first() {
186        if listed(public_key) || listed(successor) {
187            return TrustState::Retired((*successor).to_string());
188        }
189    }
190    if listed(public_key) {
191        return TrustState::Trusted("allowlist".into());
192    }
193    let predecessors: BTreeSet<&str> = valid
194        .iter()
195        .filter(|s| s.new_key == public_key)
196        .map(|s| s.old_key.as_str())
197        .collect();
198    if predecessors.iter().any(|p| successors(p).len() > 1) {
199        return TrustState::Rejected("succession fork".into());
200    }
201    if predecessors.iter().any(|p| listed(p)) {
202        return TrustState::Trusted("succession".into());
203    }
204    evaluate(policy, allow, public_key, None, previous)
205}

276    /// I2705: the successor is trusted through the trunk's trust in its predecessor and only
277    /// while it holds; the predecessor is retired once succeeded; a withdrawn predecessor
278    /// takes its successor with it, so a rotation cannot escape a revocation.
279    #[test]
280    fn a_successor_is_trusted_only_while_its_predecessor_is() {
281        let (nodes, k) = keys();
282        let rotation = [hand_over(&nodes[0], &nodes[1])];
283
284        let trunk = [k[0].clone()];
285        assert_eq!(
286            judge(&trunk, &rotation, &k[1]),
287            TrustState::Trusted("succession".into())
288        );
289        assert_eq!(
290            judge(&trunk, &rotation, &k[0]),
291            TrustState::Retired(k[1].clone()),
292            "the succeeded key is retired although the trunk still lists it"
293        );
294        assert!(!judge(&trunk, &rotation, &k[0]).is_trusted());
295
296        // the predecessor withdrawn: the successor its thief may have named is not trusted,
297        // and the old key is not retired either (nothing trusted took over)
298        assert_eq!(judge(&[], &rotation, &k[1]), TrustState::Observed);
299        assert_eq!(judge(&[], &rotation, &k[0]), TrustState::Observed);
300        // nor does any policy but the declaration trust the successor through the statement
301        assert_eq!(
302            evaluate_with_successions(&TrustPolicy::DenyUnknown, &[], &rotation, &k[1], None, None),
303            TrustState::Observed
304        );
305
306        // the replacement committed: the new key trusted directly, the old one retired
307        let committed = [k[1].clone()];
308        assert_eq!(
309            judge(&committed, &rotation, &k[1]),
310            TrustState::Trusted("allowlist".into())
311        );
312        assert_eq!(
313            judge(&committed, &rotation, &k[0]),
314            TrustState::Retired(k[1].clone())
315        );
316
317        // a forged countersignature confers nothing and retires nothing
318        let mut forged = rotation[0].clone();
319        forged.new_signature = nodes[2].sign(&forged.countersigning_bytes());
320        let forged = [forged];
321        assert_eq!(judge(&trunk, &forged, &k[1]), TrustState::Observed);
322        assert_eq!(
323            judge(&trunk, &forged, &k[0]),
324            TrustState::Trusted("allowlist".into())
325        );
326
327        // the depth is one: a key trusted only through a succession names no successor
328        let chain = [rotation[0].clone(), hand_over(&nodes[1], &nodes[2])];
329        assert_eq!(judge(&trunk, &chain, &k[2]), TrustState::Observed);
330        assert_eq!(
331            judge(&trunk, &chain, &k[1]),
332            TrustState::Trusted("succession".into()),
333            "an inert statement does not retire the key that made it"
334        );
335        // until the trunk lists it
336        let listed = [k[1].clone()];
337        assert_eq!(
338            judge(&listed, &chain, &k[2]),
339            TrustState::Trusted("succession".into())
340        );
341        assert_eq!(
342            judge(&listed, &chain, &k[1]),
343            TrustState::Retired(k[2].clone())
344        );
345    }

347    /// I2705: two different successors of one key are a fork, which only a stolen key can
348    /// produce: neither successor is trusted and the old key is rejected, listed or not.
349    #[test]
350    fn a_fork_trusts_neither() {
351        let (nodes, k) = keys();
352        let fork = [
353            hand_over(&nodes[0], &nodes[1]),
354            hand_over(&nodes[0], &nodes[2]),
355        ];
356        let trunk = [k[0].clone()];
357        let fork_state = TrustState::Rejected("succession fork".into());
358        for key in &k {
359            assert_eq!(judge(&trunk, &fork, key), fork_state, "{key}");
360        }
361        // the same statement twice is one succession, not a fork
362        let twice = [fork[0].clone(), fork[0].clone()];
363        assert_eq!(
364            judge(&trunk, &twice, &k[1]),
365            TrustState::Trusted("succession".into())
366        );
367        // a forged second statement makes no fork
368        let mut forged = fork[1].clone();
369        forged.old_signature = nodes[2].sign(&forged.signing_bytes());
370        let one = [fork[0].clone(), forged];
371        assert_eq!(
372            judge(&trunk, &one, &k[1]),
373            TrustState::Trusted("succession".into())
374        );
375        // a person who lists a successor directly is the trust root
376        let chosen = [k[1].clone()];
377        assert_eq!(
378            judge(&chosen, &fork, &k[1]),
379            TrustState::Trusted("allowlist".into())
380        );
381        assert_eq!(judge(&chosen, &fork, &k[0]), fork_state);
382    }
