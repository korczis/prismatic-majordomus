1//! Intent workspace: one intent and everything that serves it — its criteria, its
2//! milestones, their issues with what blocks each, the waves those issues occupy and the
3//! dependency graph between them — with every ready issue one link from the move that
4//! starts it. Read from `intents.list`, `plan.roadmap`, `plan.issues`, `plan.waves` and
5//! `plan.next`.
6//!
7//! The screen decides nothing. Which issues belong to an intent is the plan's own relation:
8//! the issues of the milestones the intent names, and any issue whose `serves` names it.
9//! Status, wave, blockers and the next issue are the capabilities' words, drawn as given; a
10//! new intent, milestone or issue appears here with no edit but its own record. The one
11//! choice the screen makes is which intent to open when the link names none, and that rule
12//! ([`PREFERRED_STAGES`]) is printed on the screen.
13//!
14//! The screen at `/ui/idd?intent=<id>` reads five capabilities — `intents.list`,
15//! `plan.roadmap`, `plan.issues`, `plan.waves` and `plan.next` — and links to the milestone,
16//! issue and criterion inspectors; the move itself (`plan.transition`) is offered on the
17//! issue page.

62fn strings(v: &Value, k: &str) -> Vec<String> {
63    v.get(k)
64        .and_then(Value::as_array)
65        .map(|a| {

74/// The intent to open: the one the query names, else the first of `intents` at the earliest
75/// of [`PREFERRED_STAGES`], else the first. `None` when there is no intent, and the named
76/// id as given when the query names one, so that an unknown id is refused by name rather
77/// than silently replaced.
78pub fn chosen(query: &[(String, String)], intents: &[Value]) -> Option<String> {
79    if let Some(id) = super::intent::picked(query) {
80        return Some(id);
81    }
82    PREFERRED_STAGES
83        .iter()
84        .find_map(|stage| intents.iter().find(|i| text(i, "stage") == *stage))
85        .or_else(|| intents.first())
86        .map(|i| text(i, "id"))
87}

89/// The issues that serve intent `id`: those of the milestones it names and those whose
90/// `serves` names it, in the plan's own order of `plan.issues`.
91pub fn serving(intent: &Value, issues: &[Value]) -> Vec<Value> {
92    let id = text(intent, "id");
93    let milestones: BTreeSet<String> = intent
94        .get("milestones")
95        .and_then(Value::as_array)
96        .map(|ms| ms.iter().map(|m| text(m, "id")).collect())
97        .unwrap_or_default();
98    issues
99        .iter()
100        .filter(|x| {
101            milestones.contains(&text(x, "milestone"))
102                || strings(x, "serves")
103                    .iter()
104                    .any(|s| super::intent::intent_of(s) == id)
105        })
106        .cloned()
107        .collect()
108}

130/// The workspace drawn from the five answers, whichever of them could not answer.
131fn render(
132    intents: &Answer,
133    roadmap: &Answer,
134    issues: &Answer,
135    waves: &Answer,
136    next: &Answer,
137    query: &[(String, String)],
138) -> View {
139    let all = list(intents, "/intents");
140    let id = chosen(query, &all);
141    let intent = id
142        .as_deref()
143        .and_then(|id| all.iter().find(|i| text(i, "id") == id));
144
145    let top = hero(
146        "target",
147        "violet",
148        "Intent workspace",
149        "From what the work is for to the move that starts it.",
150        "One intent and everything that serves it: its milestones, their issues and what blocks them, the waves the dependencies allow and the graph of what waits for what.",
151        vec![
152            kpi_count("Intents", &intents.read("/intents")),
153            kpi_count("Milestones", &roadmap.read("/milestones")),
154            kpi("Roadmap now", &roadmap.read("/now")),
155            kpi("Plan's next issue", &next.read("/issue/id")),
156        ],
157    );
158
159    let mut grid = el("div")
160        .class("mj-ui-grid")
161        .child(wide(choice(intents, id.as_deref())));
162    match (id.as_deref(), intent) {
163        (None, _) => {}
164        (Some(id), None) => {
165            if unanswered(intents).is_none() {
166                grid = grid.child(wide(panel(
167                    "Intent",
168                    "The intent the link names.",
169                    nothing(&format!(
170                        "{id} is not an intent of this repository: pick one above."
171                    )),
172                    &[intents],
173                )));
174            }
175        }
176        (Some(id), Some(intent)) => {
177            let mine = if unanswered(issues).is_none() {
178                serving(intent, &list(issues, "/issues"))
179            } else {
180                Vec::new()
181            };
182            grid = grid
183                .child(wide(super::intent::inspector_closing_to(
184                    intents, issues, id, ROUTE,
185                )))
186                .child(wide(milestones(roadmap, intent)))
187                .child(wide(issue_table(issues, waves, next, &mine)))
188                .child(wide(wave_panel(waves, &mine)))
189                .child(wide(dependency_graph(issues, &mine)));
190        }
191    }
192
193    View::new(
194        "Intent workspace",
195        el("div").class("mj-ui-stack").child(top).child(grid),
196    )
197}

199/// Every intent as a link to its workspace, the open one marked, with the rule that chose
200/// it when the link named none.
201fn choice(intents: &Answer, open: Option<&str>) -> El {
202    let body = unanswered(intents).unwrap_or_else(|| {
203        let rows = list(intents, "/intents")
204            .iter()
205            .map(|i| {
206                let id = text(i, "id");
207                let current = open == Some(id.as_str());
208                el("tr")
209                    .child(
210                        el("td").child(
211                            el("a")
212                                .class("mj-ui-row-link")
213                                .attr("href", href(&id))
214                                .attr_if("aria-current", current.then_some("page"))
215                                .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))
216                                .child(el("span").class("mj-ui-cell-title").text(text(i, "title"))),
217                        ),
218                    )
219                    .child(el("td").child(state(i, "stage")))
220                    .child(el("td").child(super::intent::verdict(i)))
221            })
222            .collect::<Vec<_>>();
223        if rows.is_empty() {
224            return nothing(
225                "No intent is recorded. Propose one with `majordomus intent propose`, or the intents.propose capability.",
226            );
227        }
228        el("div")
229            .class("mj-ui-stack")
230            .child(table(&["Intent", "Stage", "Verdict"], rows, ""))
231            .child(el("p").class("mj-ui-dim").text(format!(
232                "A link that names no intent opens the first one intents.list reports {}, in that order.",
233                PREFERRED_STAGES.join(", then ")
234            )))
235    });
236    panel(
237        "Intents",
238        "What the work is for: pick one to see what serves it.",
239        body,
240        &[intents],
241    )
242}

244/// The milestones the intent names, in the roadmap's order, and any it names that the plan
245/// does not hold.
246fn milestones(roadmap: &Answer, intent: &Value) -> El {
247    let named: Vec<String> = intent
248        .get("milestones")
249        .and_then(Value::as_array)
250        .map(|ms| ms.iter().map(|m| text(m, "id")).collect())
251        .unwrap_or_default();
252    let body = unanswered(roadmap).unwrap_or_else(|| {
253        let plan = list(roadmap, "/milestones");
254        let mut rows: Vec<El> = plan
255            .iter()
256            .filter(|m| named.contains(&text(m, "id")))
257            .map(|m| {
258                let waits: Vec<String> = strings(m, "blocked_by");
259                let row = milestone_row(m);
260                row.child(el("td").child(if waits.is_empty() {
261                    el("span").class("mj-ui-dim").text("nothing")
262                } else {
263                    el("p").class("mj-ui-badges").children(
264                        waits
265                            .into_iter()
266                            .map(|w| chip(super::milestone::href(&w), w)),
267                    )
268                }))
269            })
270            .collect();
271        let held: BTreeSet<String> = plan.iter().map(|m| text(m, "id")).collect();
272        rows.extend(named.iter().filter(|m| !held.contains(*m)).map(|m| {
273            el("tr")
274                .child(el("td").class("mj-ui-mono").text(m.clone()))
275                .child(el("td").child(state(&Value::Null, "status")))
276                .child(el("td").class("mj-ui-dim").text("not in the plan"))
277                .child(el("td"))
278        }));
279        table(
280            &["Milestone", "State", "Issues done", "Waits for"],
281            rows,
282            "The intent names no milestone.",
283        )
284    });
285    panel_to(
286        "Milestones",
287        "The milestones this intent names, in the roadmap's order; issues done is a count, not the milestone's acceptance.",
288        Some(super::roadmap::ROUTE),
289        body,
290        &[roadmap],
291    )
292}

294/// The issues in the order `plan.waves` lays them out — wave by wave, each wave in the
295/// capability's own order — and any it does not lay out after them in `plan.issues`'
296/// order. The screen sorts nothing: the order is the plan's.
297pub fn in_wave_order<'a>(waves: &Answer, mine: &'a [Value]) -> Vec<&'a Value> {
298    let by_id: BTreeMap<String, &Value> = mine.iter().map(|x| (text(x, "id"), x)).collect();
299    let mut seen = BTreeSet::new();
300    let mut ordered: Vec<&Value> = super::view::items(waves, "/waves")
301        .iter()
302        .flat_map(|w| {
303            w.get("issues")
304                .and_then(Value::as_array)
305                .map(Vec::as_slice)
306                .unwrap_or_default()
307        })
308        .filter_map(|x| by_id.get(&text(x, "id")).copied())
309        .filter(|x| seen.insert(text(x, "id")))
310        .collect();
311    ordered.extend(mine.iter().filter(|x| !seen.contains(&text(x, "id"))));
312    ordered
313}

315/// The issues that serve the intent, in the plan's wave order, each with what blocks it and
316/// the way to its move.
317fn issue_table(issues: &Answer, waves: &Answer, next: &Answer, mine: &[Value]) -> El {
318    let next_id = next
319        .ok()
320        .and_then(|v| v.pointer("/issue/id"))
321        .and_then(Value::as_str)
322        .map(str::to_string);
323    let body = unanswered(issues).unwrap_or_else(|| {
324        let rows = in_wave_order(waves, mine)
325            .into_iter()
326            .map(|x| {
327                let id = text(x, "id");
328                let status = text(x, "status").to_ascii_uppercase();
329                let blocked_by = strings(x, "blocked_by");
330                let mut title = el("a")
331                    .class("mj-ui-row-link")
332                    .attr("href", issue_href(&id))
333                    .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))
334                    .child(el("span").class("mj-ui-cell-title").text(text(x, "title")));
335                if next_id.as_deref() == Some(id.as_str()) {
336                    title = title.child(el("span").class("mj-ui-dim").text(" — the plan's next"));
337                }
338                let mv = match status.as_str() {
339                    "READY" => Some("start"),
340                    "ACTIVE" => Some("verify"),
341                    "VERIFY" => Some("done"),
342                    _ => None,
343                };
344                el("tr")
345                    .attr("id", format!("issue-{id}"))
346                    .child(el("td").child(title))
347                    .child(el("td").child(chip(
348                        super::milestone::href(&text(x, "milestone")),
349                        text(x, "milestone"),
350                    )))
351                    .child(el("td").child(state(x, "status")))
352                    .child(
353                        el("td")
354                            .class("mj-ui-mono")
355                            .text(x.get("wave").map(Value::to_string).unwrap_or_default()),
356                    )
357                    .child(el("td").class("mj-ui-mono").text(text(x, "priority")))
358                    .child(el("td").child(if blocked_by.is_empty() {
359                        el("span").class("mj-ui-dim").text("nothing")
360                    } else {
361                        el("p")
362                            .class("mj-ui-badges")
363                            .children(blocked_by.into_iter().map(|b| chip(issue_href(&b), b)))
364                    }))
365                    .child(
366                        el("td").child(match mv {
367                            Some(word) => el("a")
368                                .class("mj-ui-more")
369                                .attr("href", move_href(&id))
370                                .attr(
371                                    "title",
372                                    format!("the move plan.transition makes: {word} {id}"),
373                                )
374                                .text(word),
375                            None => el("span").class("mj-ui-dim").text("none"),
376                        }),
377                    )
378            })
379            .collect::<Vec<_>>();
380        table(
381            &[
382                "Issue",
383                "Milestone",
384                "Status",
385                "Wave",
386                "Priority",
387                "Blocked by",
388                "Move",
389            ],
390            rows,
391            "No issue serves this intent yet.",
392        )
393    });
394    panel_to(
395        "Issues",
396        "Every issue of its milestones or naming it, by wave; a move opens the issue's page, where plan.transition makes it and refuses one the plan does not allow.",
397        Some(super::issues::ROUTE),
398        body,
399        &[issues, waves, next],
400    )
401}

403/// The waves the intent's issues occupy, as `plan.waves` layers the whole plan, and the
404/// scope conflicts among them that serialise issues of one wave.
405fn wave_panel(waves: &Answer, mine: &[Value]) -> El {
406    let ids: BTreeSet<String> = mine.iter().map(|x| text(x, "id")).collect();
407    let body = unanswered(waves).unwrap_or_else(|| {
408        let rows = list(waves, "/waves")
409            .iter()
410            .filter_map(|w| {
411                let members: Vec<Value> = w
412                    .get("issues")
413                    .and_then(Value::as_array)
414                    .map(|a| {
415                        a.iter()
416                            .filter(|x| ids.contains(&text(x, "id")))
417                            .cloned()
418                            .collect()
419                    })
420                    .unwrap_or_default();
421                if members.is_empty() {
422                    return None;
423                }
424                Some(
425                    el("tr")
426                        .child(
427                            el("td")
428                                .class("mj-ui-mono")
429                                .text(w.get("wave").map(Value::to_string).unwrap_or_default()),
430                        )
431                        .child(el("td").child(el("p").class("mj-ui-badges").children(
432                            members.iter().map(|x| {
433                                el("span")
434                                    .child(chip(issue_href(&text(x, "id")), text(x, "id")))
435                                    .child(state(x, "status"))
436                            }),
437                        ))),
438                )
439            })
440            .collect::<Vec<_>>();
441        let conflicts: Vec<El> = list(waves, "/serialised_by_scope")
442            .iter()
443            .filter(|f| ids.contains(&text(f, "subject")))
444            .map(|f| {
445                el("li")
446                    .child(el("code").class("mj-ui-mono").text(text(f, "subject")))
447                    .child(el("span").text(format!(" {}", text(f, "message"))))
448            })
449            .collect();
450        let mut body = el("div").class("mj-ui-stack").child(table(
451            &["Wave", "Issues of this intent in it"],
452            rows,
453            "No issue of this intent is in a wave.",
454        ));
455        if !conflicts.is_empty() {
456            body = body
457                .child(
458                    el("h4")
459                        .class("mj-ui-panel-title")
460                        .text("Shared scope: these may not run at the same time"),
461                )
462                .child(el("ul").class("mj-ui-list").children(conflicts));
463        }
464        body
465    });
466    panel(
467        "Waves",
468        "An issue enters a wave once every dependency has left it; sharing a wave is necessary to run together, not sufficient.",
469        body,
470        &[waves],
471    )
472}

474/// The dependency graph of the intent's issues: each a node in the tone of its status, each
475/// dependency the plan records an edge, and a dependency outside the intent drawn small. The
476/// table beneath it is the same edges, for a reader without the picture.
477fn dependency_graph(issues: &Answer, mine: &[Value]) -> El {
478    let body = unanswered(issues).unwrap_or_else(|| {
479        let by_id: BTreeMap<String, &Value> = super::view::items(issues, "/issues")
480            .iter()
481            .map(|x| (text(x, "id"), x))
482            .collect();
483        let own: BTreeSet<String> = mine.iter().map(|x| text(x, "id")).collect();
484        let edges: Vec<(String, String)> = mine
485            .iter()
486            .flat_map(|x| {
487                let to = text(x, "id");
488                strings(x, "depends_on")
489                    .into_iter()
490                    .map(move |from| (from, to.clone()))
491            })
492            .collect();
493        let mut ids: BTreeSet<String> = own.clone();
494        ids.extend(edges.iter().map(|(from, _)| from.clone()));
495        let nodes: Vec<GraphNode> = ids
496            .iter()
497            .map(|id| {
498                let status = by_id.get(id).map(|x| text(x, "status")).unwrap_or_default();
499                GraphNode {
500                    id: id.clone(),
501                    label: id.clone(),
502                    tone: status_tone(&status),
503                    href: issue_href(id),
504                    kind: if own.contains(id) { "node" } else { "leaf" },
505                }
506            })
507            .collect();
508        let rows = edges
509            .iter()
510            .map(|(from, to)| {
511                el("tr")
512                    .child(el("td").child(chip(issue_href(to), to.clone())))
513                    .child(el("td").child(chip(issue_href(from), from.clone())))
514                    .child(el("td").child(match by_id.get(from) {
515                        Some(x) => state(x, "status"),
516                        None => state(&Value::Null, "status"),
517                    }))
518            })
519            .collect::<Vec<_>>();
520        let fallback = table(
521            &["Issue", "Depends on", "Its status"],
522            rows,
523            "No issue of this intent depends on another.",
524        );
525        graph(
526            "The issues of this intent and the dependencies between them",
527            Layout::Directed,
528            &nodes,
529            &edges,
530            fallback,
531        )
532    });
533    panel(
534        "Dependencies",
535        "What waits for what, as the plan records it; a dependency outside this intent is drawn small.",
536        body,
537        &[issues],
538    )
539}

671    /// An issue added to the plan's record appears in the workspace with no other edit.
672    #[test]
673    fn an_issue_added_to_the_plan_appears_with_no_registration() {
674        let repo = planned_with_an_intent();
675        let before = screen(&repo.context().expect("a context"), &q("intent", "i-first"))
676            .main
677            .render();
678        assert!(!before.contains("I9999"), "{before}");
679        let issues = repo.root().join(".ai/repo/project/issues");
680        let template = std::fs::read_dir(&issues)
681            .expect("the issues")
682            .filter_map(Result::ok)
683            .map(|e| e.path())
684            .find(|p| std::fs::read_to_string(p).is_ok_and(|t| t.contains("milestone: m-first")))
685            .expect("an issue of m-first");
686        let text = std::fs::read_to_string(&template).expect("the issue");
687        let id = text
688            .lines()
689            .find_map(|l| l.strip_prefix("id: "))
690            .expect("an id")
691            .trim()
692            .trim_matches('"')
693            .to_string();
694        let added: String = text
695            .lines()
696            .map(|l| match l {
697                l if l.starts_with("id: ") => "id: I9999".to_string(),
698                l if l.starts_with("slug: ") => "slug: the-added-one".to_string(),
699                l => l.to_string(),
700            })
701            .collect::<Vec<_>>()
702            .join("\n");
703        std::fs::write(issues.join("I9999.yaml"), format!("{added}\n")).expect("an issue");
704        assert_ne!(id, "I9999");
705        let after = screen(&repo.context().expect("a context"), &q("intent", "i-first"))
706            .main
707            .render();
708        assert!(after.contains("I9999"), "the added issue is drawn: {after}");
709    }
