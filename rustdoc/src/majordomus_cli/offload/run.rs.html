1//! One offloaded derive, end to end: the worktree goes to a host as a git bundle, the host
2//! derives it in its turn, and the derived files come back staged.
3//!
4//! What it does, and what it never does:
5//!
6//! - The tree that travels is the worktree as it stands: HEAD when it is clean, otherwise a
7//!   `git stash create` snapshot of its tracked and staged changes. That writes a commit
8//!   object and touches neither the index, the worktree nor a hook — which matters, because
9//!   the commit hook refuses a tree that is not derived yet, so the authored change cannot
10//!   be committed before its derive. Untracked files would not travel and are refused.
11//! - A merge in progress travels as a real merge commit with both parents, written by
12//!   `git commit-tree` from the staged tree once every conflict is resolved: a stash would
13//!   drop the second parent, and the changelog — composed from the log — would be derived
14//!   without the merged commits (PR #854, 2026-10-10). No ref of the branch moves.
15//! - Nothing is pushed to origin, so no CI run starts, and no hook of this repository runs.
16//! - On the host, the derive runs in a detached worktree `offload-<tag>`, under the queue,
17//!   with a build directory of its own: a copy-on-write clone of the shared one where the
18//!   filesystem has clones, removed when the derive ends. A build into the shared directory
19//!   by a derive that never took the lock cannot replace this derive's executable (it did,
20//!   on the second MacBook, 2026-10-11 00:25Z).
21//! - Back here the files the derive changed are checked out of the derived snapshot and
22//!   staged. Nothing is committed. A derive that deletes a tracked file is refused rather
23//!   than half-applied.
24//! - The machine's own derive lock is not taken: it guards this machine, and this machine
25//!   is not deriving.
26

104fn git(dir: &Path, args: &[&str]) -> Result<String, Refusal> {
105    let out = Command::new("git")
106        .arg("-C")
107        .arg(dir)
108        .args(args)
109        .output()
110        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("git could not be run: {e}")))?;
111    if !out.status.success() {
112        return Err(refuse(
113            EXIT_UNAVAILABLE,
114            format!(
115                "git {} failed: {}",
116                args.join(" "),
117                String::from_utf8_lossy(&out.stderr).trim()
118            ),
119        ));
120    }
121    Ok(String::from_utf8_lossy(&out.stdout).into_owned())
122}

127        .arg(dir)

130        .map(|o| o.status.success())
131        .unwrap_or(false)
132}

140/// The commit that travels, after the worktree's preconditions are checked.
141pub fn input_commit(worktree: &Path) -> Result<String, Refusal> {
142    let merge_head = git(
143        worktree,
144        &[
145            "rev-parse",
146            "--path-format=absolute",
147            "--git-path",
148            "MERGE_HEAD",
149        ],
150    )?;
151    let in_merge = Path::new(merge_head.trim()).exists();
152    if in_merge {
153        if !git(worktree, &["diff", "--name-only", "--diff-filter=U"])?
154            .trim()
155            .is_empty()
156        {
157            return Err(refuse(
158                EXIT_PRECONDITION,
159                "the merge still has unresolved paths — resolve and git add them first",
160            ));
161        }
162        if !git_ok(worktree, &["diff", "--quiet"]) {
163            return Err(refuse(EXIT_PRECONDITION, "the merge has unstaged changes — git add them first, the staged tree is what travels"));
164        }
165    }
166    let status = git(
167        worktree,
168        &["status", "--porcelain", "--untracked-files=all"],
169    )?;
170    let untracked: Vec<&str> = status
171        .lines()
172        .filter_map(|l| l.strip_prefix("?? "))
173        .collect();
174    if !untracked.is_empty() {
175        return Err(refuse(
176            EXIT_PRECONDITION,
177            format!(
178                "untracked files would not travel — git add them first (a derive sees what the commit will): {}",
179                untracked.join(", ")
180            ),
181        ));
182    }
183    if in_merge {
184        let tree = git(worktree, &["write-tree"])?;
185        let other = git(worktree, &["rev-parse", "MERGE_HEAD"])?;
186        let head = git(
187            worktree,
188            &[
189                "-c",
190                SCRATCH_NAME,
191                "-c",
192                SCRATCH_EMAIL,
193                "commit-tree",
194                tree.trim(),
195                "-p",
196                "HEAD",
197                "-p",
198                other.trim(),
199                "-m",
200                "derive-offload input: a merge in progress",
201            ],
202        )?;
203        return Ok(head.trim().to_string());
204    }
205    let stash = git(
206        worktree,
207        &[
208            "-c",
209            SCRATCH_NAME,
210            "-c",
211            SCRATCH_EMAIL,
212            "stash",
213            "create",
214            "derive-offload input",
215        ],
216    )?;
217    if !stash.trim().is_empty() {
218        return Ok(stash.trim().to_string());
219    }
220    Ok(git(worktree, &["rev-parse", "HEAD"])?.trim().to_string())
221}

223/// Prepare the host's worktree from the uploaded bundle, or — when the input is already in
224/// the base, so there was nothing to bundle — from the commit itself, which the host has
225/// once it has fetched origin. Arguments: checkout, worktrees directory, spool, tag, input.
226///
227/// This runs before the queue's lock is taken, and every offload to the host fetches into
228/// the same checkout: two arriving together race on git's ref locks, and git refuses the
229/// loser ("cannot lock ref", "Another git process seems to be running"), which failed a
230/// whole derive on 2026-10-29 (I4375). A fetch is therefore asked again, bounded:
231/// `MJ_OFFLOAD_FETCH_TRIES` attempts (default 5), `MJ_OFFLOAD_FETCH_BACKOFF` seconds times
232/// the attempt between them (default 3), and the last refusal is what the caller reports.
233const PREPARE: &str = r#"set -e
234c=$1; w=$2; spool=$3; tag=$4; input=$5
235case $c in /*) ;; *) c=$HOME/$c ;; esac
236case $w in /*) ;; *) w=$HOME/$w ;; esac
237fetch() {
238  n=1
239  until git fetch -q "$@"; do
240    [ "$n" -lt "${MJ_OFFLOAD_FETCH_TRIES:-5}" ] || return 1
241    sleep $(( n * ${MJ_OFFLOAD_FETCH_BACKOFF:-3} ))
242    n=$((n + 1))
243  done
244}
245cd "$c"
246fetch origin
247if [ -f "$spool/offload-$tag.bundle" ]; then
248  fetch "$spool/offload-$tag.bundle" "refs/offload/in-$tag:refs/offload/$tag"
249else
250  git update-ref "refs/offload/$tag" "$input"
251fi
252d=$w/offload-$tag
253[ -d "$d" ] || git worktree add -q --detach "$d" "refs/offload/$tag"
254cd "$d"
255git checkout -q -f --detach "refs/offload/$tag"
256git clean -fdq"#;

258/// Derive on the host, under a lock the caller holds. Its own pid joins the lock's holders,
259/// so the lock stays held exactly as long as this derive runs.
260///
261/// Everything the derive compiles stays in its own run directory, `<shared>-<tag>`, which
262/// the traps remove however the derive ends. Pinning the target directory alone is not
263/// that: the checkout declares a shared `build-dir`, so cargo kept every intermediate in the
264/// host's `build-shared/majordomus`, outside what the traps remove, and on 2026-10-18 that
265/// directory held 87 GB on lundra and the volume reached 0 bytes. The build directory is
266/// pinned to the run directory too. Arguments: lock, worktrees
267/// directory, tag, shared build directory, derive command, spool, PATH prefix
268/// (colon-separated), the commit the derived files may observe (exported to the derive as
269/// `MJ_DERIVE_OBSERVE`; empty leaves it unset), whether the caller holds standard input open
270/// (`1`: its end means the caller is gone, and the run is stopped), the run-record directory,
271/// the owner's session, peer, issue, host and pid ([`super::owner::RunOwner::args`]), then the
272/// files to touch; [`derive_command`] builds them. Prints `pid <shell>`, `target <dir>`,
273/// `group <pgid|none>`, then `nothing` or `derived <sha>`; exit 10 when the derive fails, 13
274/// when the lock is not there, 14 when `offload stop` stopped it, 129 when the caller went
275/// away.
276///
277/// Who owns the run (I3956): the owner is exported into every process the derive starts
278/// (`MAJORDOMUS_OWNER_*`, and the run's own tag as `MAJORDOMUS_OWNER_RUN`) and written, with
279/// the run's process group, into `<runs>/<tag>.run` — atomically, through a temporary file
280/// and a rename — for as long as the run lives; every ending this shell sees removes it, so
281/// a record that outlives its run is the evidence of a crash. The derive runs in a process
282/// group of its own: `setsid`, or Python's `os.setsid` where a host has no `setsid`. A child
283/// started with `&` by a non-interactive shell is never a group leader, so `setsid` makes
284/// the new group in the derive's own process and its pid is the group's id. A run that is
285/// stopped — its caller gone, a signal — is stopped as a group, `TERM` and then `KILL`, so no
286/// `rustc` under it is left running with init as its parent.
287const DERIVE: &str = r#"l=$1; w=$2; tag=$3; shared=$4; derive=$5; spool=$6; prefix=$7; observe=$8; held=$9
288runs=${10}; os=${11}; op=${12}; oi=${13}; oh=${14}; opid=${15}; shift 15
289[ -d "$l" ] || { echo "the lock $l is not held" >&2; exit 13; }
290echo "pid $$"
291: > "$l/$$"
292case $w in /*) ;; *) w=$HOME/$w ;; esac
293case $shared in /*) ;; *) shared=$HOME/$shared ;; esac
294case $runs in /*) ;; *) runs=$HOME/$runs ;; esac
295p=""; IFS=:; for e in $prefix; do case $e in /*) ;; *) e=$HOME/$e ;; esac; p="$p$e:"; done; unset IFS
296export PATH="$p$PATH"
297cd "$w/offload-$tag" || exit 13
298private="$shared-$tag"; log="$spool/offload-$tag.log"; record="$runs/$tag.run"
299mkdir -p "$shared" "$spool" "$runs"; rm -rf "$private"
300if [ "$(uname)" = Darwin ]; then cp -cR "$shared" "$private" 2>/dev/null || { rm -rf "$private"; mkdir -p "$private"; }
301else cp -a --reflink=always "$shared" "$private" 2>/dev/null || { rm -rf "$private"; mkdir -p "$private"; }; fi
302child=""; pg=""; watch=""
303ended() {
304  [ -n "$watch" ] && kill "$watch" 2>/dev/null
305  rm -f "$record" "$record.tmp" "$runs/$tag.stopped"; rm -rf "$private"
306}
307stopped() {
308  if [ -n "$pg" ]; then
309    kill -TERM -"$pg" 2>/dev/null; i=0
310    while kill -0 -"$pg" 2>/dev/null && [ "$i" -lt 10 ]; do sleep 0.2; i=$((i+1)); done
311    kill -KILL -"$pg" 2>/dev/null
312  fi
313  [ -n "$child" ] && kill -TERM "$child" 2>/dev/null
314  ended
315}
316trap 'ended' EXIT
317trap 'stopped; exit 129' HUP
318trap 'stopped; exit 130' INT
319trap 'stopped; exit 143' TERM
320if [ "$held" = 1 ]; then
321  main=$$
322  exec 3<&0
323  ( while IFS= read -r _; do :; done <&3; kill -HUP "$main" 2>/dev/null ) &
324  watch=$!
325  exec 3<&- 0</dev/null
326fi
327MAJORDOMUS_OWNER_RUN=$tag; export MAJORDOMUS_OWNER_RUN
328for v in "MAJORDOMUS_OWNER_SESSION=$os" "MAJORDOMUS_OWNER_PEER=$op" "MAJORDOMUS_OWNER_ISSUE=$oi" \
329  "MAJORDOMUS_OWNER_HOST=$oh" "MAJORDOMUS_OWNER_PID=$opid"; do
330  case $v in *=) ;; *) export "$v" ;; esac
331done
332export CARGO_TARGET_DIR="$private"
333export CARGO_BUILD_BUILD_DIR="$private"
334[ -n "$observe" ] && { MJ_DERIVE_OBSERVE=$observe; export MJ_DERIVE_OBSERVE; }
335echo "target $private"
336for f in "$@"; do [ -e "$f" ] && touch "$f"; done
337group='import os,sys; os.setsid(); os.execvp(sys.argv[1], sys.argv[1:])'
338if command -v setsid >/dev/null 2>&1; then how=setsid
339elif python3 -c 'import os' >/dev/null 2>&1; then how=python3
340else how=; fi
341case $how in
342  setsid) env -u MAJORDOMUS_SHARE -u MAJORDOMUS_BIN setsid "$derive" > "$log" 2>&1 < /dev/null & ;;
343  python3) env -u MAJORDOMUS_SHARE -u MAJORDOMUS_BIN python3 -c "$group" "$derive" > "$log" 2>&1 < /dev/null & ;;
344  *) env -u MAJORDOMUS_SHARE -u MAJORDOMUS_BIN "$derive" > "$log" 2>&1 < /dev/null & ;;
345esac
346child=$!
347[ -n "$how" ] && pg=$child
348echo "group ${pg:-none}"
349{ echo "majordomus-offload-run/v1"; echo "tag $tag"; echo "pid $$"; echo "pgid $pg"
350  echo "started_at $(date +%s)"; echo "host $(hostname)"; echo "cwd $PWD"
351  echo "owner_session $os"; echo "owner_peer $op"; echo "owner_issue $oi"
352  echo "owner_host $oh"; echo "owner_pid $opid"; } > "$record.tmp" && mv -f "$record.tmp" "$record"
353rc=0; wait "$child" || rc=$?
354child=""
355if [ "$rc" != 0 ] && [ -f "$runs/$tag.stopped" ]; then
356  echo "the derive was stopped on $(hostname) by session $(cat "$runs/$tag.stopped")" >&2
357  exit 14
358fi
359if [ "$rc" != 0 ]; then
360  echo "derive failed on $(hostname); the last lines of $log:" >&2
361  grep -vE ' INFO |^ +(Compiling|Checking)' "$log" | tail -8 >&2
362  exit 10
363fi
364grep -E '^==> derive: done' "$log" >&2 || true
365git add -A
366sha=$(git -c user.name=offload -c user.email=offload@invalid stash create derived || true)
367if [ -z "$sha" ]; then echo nothing; exit 0; fi
368git update-ref "refs/offload/$tag-derived" "$sha"
369git bundle create "$spool/offload-$tag-back.bundle" "refs/offload/$tag..refs/offload/$tag-derived" >/dev/null 2>&1
370echo "derived $sha""#;

390    /// The commit the derived files may observe; empty leaves it unset.
391    pub observe: &'a str,

395    pub runs: &'a str,

402/// The host's derive as a script and its positional arguments, in the order [`DERIVE`]
403/// reads them: the one place that order is written.
404pub(crate) fn derive_command(a: &DeriveArgs) -> (&'static str, Vec<String>) {
405    let mut args: Vec<String> = [
406        a.lock,
407        a.worktrees,
408        a.tag,
409        a.shared,
410        a.derive,
411        a.spool,
412        a.prefix,
413        a.observe,
414        if a.held { "1" } else { "0" },
415        a.runs,
416    ]
417    .iter()
418    .map(|s| s.to_string())
419    .collect();
420    args.extend(a.owner.iter().cloned());
421    args.extend(a.touch.iter().cloned());
422    (DERIVE, args)
423}

452/// The lock held on a host, released however this process ends short of a crash; after a
453/// crash the holders are dead and the next waiter reclaims it.
454struct HeldLock<'a, 'q> {
455    queue: &'q Queue<'a>,
456    pids: Vec<u32>,
457}
458
459impl Drop for HeldLock<'_, '_> {
460    fn drop(&mut self) {
461        self.queue.release(&self.pids);
462    }
463}

465fn pick_host(
466    d: &OffloadDeclaration,
467    named: Option<&str>,
468    log: &mut dyn Write,
469) -> Result<(OffloadHost, Vec<HostReport>), Refusal> {
470    if let Some(name) = named {
471        let host = d.host(name).cloned().ok_or_else(|| {
472            refuse(
473                EXIT_UNAVAILABLE,
474                format!(
475                    "'{name}' is not a declared host; declared: {}",
476                    d.hosts
477                        .iter()
478                        .map(|h| h.id.as_str())
479                        .collect::<Vec<_>>()
480                        .join(", ")
481                ),
482            )
483        })?;
484        return Ok((host, Vec::new()));
485    }
486    let reports = probe(d);
487    for r in &reports {
488        if let Some(why) = &r.reason {
489            let _ = writeln!(log, "derive-offload: {} skipped: {why}", r.id);
490        }
491    }
492    let Some(chosen) = reports.iter().find(|r| r.chosen) else {
493        return Err(refuse(
494            EXIT_UNAVAILABLE,
495            "no declared host answers with its free-space minimum free",
496        ));
497    };
498    let _ = writeln!(
499        log,
500        "derive-offload: chose {} ({} running or waiting there, {} GiB free)",
501        chosen.id,
502        chosen.load.unwrap_or(0),
503        chosen.free_gib.unwrap_or(0)
504    );
505    let host = d
506        .host(&chosen.id)
507        .cloned()
508        .expect("a chosen host is declared");
509    Ok((host, reports))
510}

512/// Offload one derive. Progress goes to `log`; the answer is the report or a refusal.
513pub fn offload(
514    d: &OffloadDeclaration,
515    request: &OffloadRequest,
516    log: &mut dyn Write,
517) -> Result<OffloadReport, Refusal> {
518    let started = Instant::now();
519    let worktree = std::fs::canonicalize(&request.worktree).map_err(|e| {
520        refuse(
521            EXIT_PRECONDITION,
522            format!("{}: {e}", request.worktree.display()),
523        )
524    })?;
525    let top = git(&worktree, &["rev-parse", "--show-toplevel"])?;
526    let worktree = PathBuf::from(top.trim());
527
528    // The commit the derived files may observe: the worktree's HEAD, never the input. The
529    // input of a dirty tree is a `git stash create` snapshot and that of a merge in progress
530    // a `commit-tree` commit; neither reaches origin, so a derived file that named one
531    // (delivery.json's observed.commit) could not be checked anywhere else, and CI's
532    // `generate-site-data --check` called it stale (#1759, I3836). Read before the input, so
533    // a branch with no commit yet is refused here, by what it lacks, before anything is
534    // snapshot.
535    let observe = git(&worktree, &["rev-parse", "--verify", "HEAD"])
536        .map_err(|_| {
537            refuse(
538                EXIT_PRECONDITION,
539                "the worktree's branch has no commit yet: a derive observes HEAD, so commit first",
540            )
541        })?
542        .trim()
543        .to_string();
544    let input = input_commit(&worktree)?;
545    let (host, probe_reports) = pick_host(d, request.host.as_deref(), log)?;
546    let remote = remote_of(&host);
547    // The base must be current here so the bundle carries only what the host lacks.
548    let _ = git(&worktree, &["fetch", "-q", "origin"]);
549
550    let tag = input[..10].to_string();
551    let in_ref = format!("refs/offload/in-{tag}");
552    git(&worktree, &["update-ref", &in_ref, &input])?;
553    let _in_guard = ScratchRef {
554        dir: &worktree,
555        name: in_ref.clone(),
556    };
557    let scratch = tempfile_dir()?;
558    let spool = d.remote.spool.as_str();
559    // An input the base already reaches has nothing to bundle (git refuses an empty bundle);
560    // the host has it once it fetches origin.
561    if !git_ok(&worktree, &["merge-base", "--is-ancestor", &input, &d.base]) {
562        let bundle = scratch.path.join("in.bundle");
563        let range = format!("{}..{in_ref}", d.base);
564        git(
565            &worktree,
566            &["bundle", "create", &bundle.to_string_lossy(), &range],
567        )?;
568        let upload = remote
569            .run(
570                UPLOAD,
571                &[spool, &format!("offload-{tag}.bundle")],
572                Some(&bundle),
573            )
574            .map_err(|e| {
575                refuse(
576                    EXIT_UNAVAILABLE,
577                    format!("{}: the bundle could not be sent: {e}", host.id),
578                )
579            })?;
580        if !upload.ok() {
581            return Err(refuse(
582                EXIT_UNAVAILABLE,
583                format!(
584                    "{}: the bundle could not be sent: {}",
585                    host.id,
586                    upload.stderr.trim()
587                ),
588            ));
589        }
590    }
591    let _ = writeln!(
592        log,
593        "derive-offload: {tag} on {} — deriving (log: {}:{spool}/offload-{tag}.log)",
594        host.id, host.address
595    );
596    let prep = remote
597        .run(
598            PREPARE,
599            &[&d.remote.checkout, &d.remote.worktrees, spool, &tag, &input],
600            None,
601        )
602        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{}: {e}", host.id)))?;
603    if !prep.ok() {
604        return Err(refuse(
605            EXIT_UNAVAILABLE,
606            format!(
607                "{}: the host could not prepare the worktree: {}",
608                host.id,
609                prep.stderr.trim()
610            ),
611        ));
612    }
613
614    let queue = Queue {
615        remote: &remote,
616        directory: &d.queue.directory,
617        lock: &d.queue.lock,
618        poll: Duration::from_secs(d.queue.poll_seconds),
619        wait_max: Duration::from_secs(d.queue.wait_max_seconds),
620    };
621    let holder = Holder::start(&remote).map_err(|e| {
622        refuse(
623            EXIT_UNAVAILABLE,
624            format!("{}: no holder process could be started: {e}", host.id),
625        )
626    })?;
627    let ticket = queue
628        .enqueue(holder.pid, &tag)
629        .map_err(|e| refuse(EXIT_UNAVAILABLE, e.to_string()))?;
630    let wait_started = Instant::now();
631    queue.wait(&ticket, holder.pid, log).map_err(|e| match e {
632        QueueError::TimedOut(_) => refuse(EXIT_TIMED_OUT, e.to_string()),
633        _ => refuse(EXIT_UNAVAILABLE, e.to_string()),
634    })?;
635    let waited = wait_started.elapsed().as_secs();
636    let mut lock = HeldLock {
637        queue: &queue,
638        pids: vec![holder.pid],
639    };
640
641    let prefix = d.remote.path.join(":");
642    let owner = super::owner::RunOwner::of_this_process(&worktree);
643    let runs = super::owner::runs_directory(&d.queue.directory);
644    let _ = writeln!(
645        log,
646        "derive-offload: {tag} runs for session {}, peer {}, issue {} (record {}:{runs}/{tag}.run; \
647         stop it with `majordomus offload stop {tag} --host {}`)",
648        owner.session.as_deref().unwrap_or("(unnamed)"),
649        owner.peer.as_deref().unwrap_or("(none)"),
650        owner.issue.as_deref().unwrap_or("(none)"),
651        host.id,
652        host.id
653    );
654    let (script, args) = derive_command(&DeriveArgs {
655        lock: &d.queue.lock,
656        worktrees: &d.remote.worktrees,
657        tag: &tag,
658        shared: &d.remote.target,
659        derive: &d.remote.derive,
660        spool,
661        prefix: &prefix,
662        observe: &observe,
663        held: true,
664        runs: &runs,
665        owner: &owner.args(),
666        touch: &d.remote.touch,
667    });
668    let args: Vec<&str> = args.iter().map(String::as_str).collect();
669    let out = remote
670        .run_held(script, &args)
671        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{}: {e}", host.id)))?;
672    let _ = log.write_all(out.stderr.as_bytes());
673    // the derive's own pid file is dead now; release it with the holder's
674    lock.pids.extend(out.stdout.lines().filter_map(|l| {
675        l.strip_prefix("pid ")
676            .and_then(|p| p.trim().parse::<u32>().ok())
677    }));
678    // released now rather than at the end of the scope, so the next waiter starts while
679    // this one applies its result
680    queue.release(&lock.pids);
681    lock.pids.clear();
682    drop(holder);
683    let cleanup = |keep: bool| {
684        let _ = remote.run(
685            CLEANUP,
686            &[
687                &d.remote.checkout,
688                &d.remote.worktrees,
689                spool,
690                &tag,
691                if keep { "1" } else { "0" },
692            ],
693            None,
694        );
695    };
696    match out.code {
697        Some(0) => {}
698        Some(14) => {
699            cleanup(true);
700            return Err(refuse(
701                EXIT_STOPPED,
702                format!(
703                    "the derive was stopped on {} with `offload stop`; its worktree is kept there",
704                    host.id
705                ),
706            ));
707        }
708        Some(10) => {
709            cleanup(true);
710            return Err(refuse(
711                EXIT_DERIVE_FAILED,
712                format!(
713                    "the derive failed on {}; its worktree is kept there",
714                    host.id
715                ),
716            ));
717        }
718        other => {
719            cleanup(true);
720            return Err(refuse(
721                EXIT_UNAVAILABLE,
722                format!(
723                    "{}: the derive step ended with {other:?}: {}",
724                    host.id,
725                    out.stderr.trim()
726                ),
727            ));
728        }
729    }
730    let derived = out
731        .stdout
732        .lines()
733        .find_map(|l| l.strip_prefix("derived "))
734        .map(|s| s.trim().to_string());
735    let mut report = OffloadReport {
736        host: host.id.clone(),
737        tag: tag.clone(),
738        input: input.clone(),
739        derived: derived.clone(),
740        staged: Vec::new(),
741        waited_seconds: waited,
742        seconds: 0,
743        probe: probe_reports,
744        owner,
745    };
746    let Some(sha) = derived else {
747        cleanup(request.keep);
748        let _ = writeln!(
749            log,
750            "derive-offload: the derive changed nothing; {} is already current",
751            worktree.display()
752        );
753        report.seconds = started.elapsed().as_secs();
754        return Ok(report);
755    };
756
757    let back = scratch.path.join("back.bundle");
758    let dl = remote
759        .run_to_file(
760            DOWNLOAD,
761            &[spool, &format!("offload-{tag}-back.bundle")],
762            &back,
763        )
764        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{}: {e}", host.id)))?;
765    if !dl.ok() {
766        return Err(refuse(
767            EXIT_UNAVAILABLE,
768            format!(
769                "{}: the derived bundle could not be fetched: {}",
770                host.id,
771                dl.stderr.trim()
772            ),
773        ));
774    }
775    cleanup(request.keep);
776    let derived_ref = format!("refs/offload/{tag}-derived");
777    // The ref is this process's scratch and goes however the apply ends: a refused apply
778    // used to leave it behind, and the next offload of the same tree (same tag) then fetched
779    // a different derived commit (`stash create` stamps the time) onto it, which git refuses
780    // as a non-fast-forward with nothing on stderr under -q. `+` overwrites whatever an
781    // earlier process left, and the guard removes it on every path.
782    let _derived_guard = ScratchRef {
783        dir: &worktree,
784        name: derived_ref.clone(),
785    };
786    git(
787        &worktree,
788        &[
789            "fetch",
790            "-q",
791            &back.to_string_lossy(),
792            &format!("+{derived_ref}:{derived_ref}"),
793        ],
794    )?;
795    apply(&worktree, &input, &sha, &derived_ref, &mut report)?;
796    let _ = writeln!(
797        log,
798        "derive-offload: applied {} derived file(s), staged, not committed",
799        report.staged.len()
800    );
801    report.seconds = started.elapsed().as_secs();
802    Ok(report)
803}

805/// Check the derived snapshot against the input and stage what it changed.
806fn apply(
807    worktree: &Path,
808    input: &str,
809    sha: &str,
810    derived_ref: &str,
811    report: &mut OffloadReport,
812) -> Result<(), Refusal> {
813    apply_with(worktree, input, sha, derived_ref, report, git, git_ok)
814}
815
816/// [`apply`], running `git` and `git_ok` as given: the real ones in earnest, a test's own
817/// failures in a test.
818fn apply_with(
819    worktree: &Path,
820    input: &str,
821    sha: &str,
822    derived_ref: &str,
823    report: &mut OffloadReport,
824    git: impl Fn(&Path, &[&str]) -> Result<String, Refusal>,
825    git_ok: impl Fn(&Path, &[&str]) -> bool,
826) -> Result<(), Refusal> {
827    let status = git(worktree, &["diff", "--name-status", input, sha])?;
828    let deleted: Vec<&str> = status.lines().filter(|l| l.starts_with('D')).collect();
829    if !deleted.is_empty() {
830        return Err(refuse(
831            EXIT_DELETES,
832            format!(
833                "the derive deleted tracked files; nothing was applied — apply by hand from {derived_ref}:\n{}",
834                deleted.join("\n")
835            ),
836        ));
837    }
838    let names = git(worktree, &["diff", "--name-only", input, sha])?;
839    let files: Vec<String> = names
840        .lines()
841        .filter(|l| !l.is_empty())
842        .map(str::to_string)
843        .collect();
844    for chunk in files.chunks(200) {
845        let mut args = vec!["checkout", sha, "--"];
846        args.extend(chunk.iter().map(String::as_str));
847        git(worktree, &args)?;
848    }
849    if !git_ok(worktree, &["diff", "--quiet", sha, "--", "."]) {
850        return Err(refuse(
851            EXIT_UNAVAILABLE,
852            "the worktree does not match the derived tree after applying it",
853        ));
854    }
855    report.staged = files;
856    report.staged.sort();
857    Ok(())
858}

860/// A scratch directory removed when dropped.
861struct Scratch {
862    path: PathBuf,
863}
864
865impl Drop for Scratch {
866    fn drop(&mut self) {
867        let _ = std::fs::remove_dir_all(&self.path);
868    }
869}
870
871fn tempfile_dir() -> Result<Scratch, Refusal> {
872    let base = std::env::temp_dir();
873    for n in 0..100u32 {
874        let path = base.join(format!("derive-offload.{}.{n}", std::process::id()));
875        if std::fs::create_dir(&path).is_ok() {
876            return Ok(Scratch { path });
877        }
878    }
879    Err(refuse(
880        EXIT_UNAVAILABLE,
881        "no scratch directory could be made",
882    ))
883}

1196    /// The host's prepare, run here as a host runs it, against a `git` whose fetch is refused
1197    /// the way a concurrent offload's ref lock refuses it: a fetch that loses the race is
1198    /// asked again and the worktree is prepared; one that never wins fails the prepare after
1199    /// its declared tries, with git's refusal on stderr for the caller to report (I4375).
1200    #[test]
1201    fn a_fetch_that_loses_a_ref_lock_race_is_asked_again_and_bounded() {
1202        use std::os::unix::fs::PermissionsExt;
1203        for (refusals, tries, prepared) in [(2, 5, true), (9, 3, false)] {
1204            let root = tempfile::tempdir().expect("a temporary directory");
1205            let at = |p: &str| root.path().join(p);
1206            let (checkout, worktrees, spool, bin) = (at("c"), at("wt"), at("spool"), at("bin"));
1207            for dir in [&checkout, &worktrees.join("offload-t"), &spool, &bin] {
1208                std::fs::create_dir_all(dir).unwrap();
1209            }
1210            let count = at("fetches");
1211            // every git subcommand succeeds but fetch, which is refused until it has been
1212            // asked more than `refusals` times
1213            std::fs::write(
1214                bin.join("git"),
1215                format!(
1216                    "#!/bin/sh\n[ \"$1\" = fetch ] || exit 0\n\
1217                     n=$(( $(cat {count} 2>/dev/null || echo 0) + 1 )); echo $n > {count}\n\
1218                     [ $n -gt {refusals} ] && exit 0\n\
1219                     echo \"error: cannot lock ref 'refs/remotes/origin/x': is at a but expected b\" >&2\n\
1220                     exit 1\n",
1221                    count = count.display()
1222                ),
1223            )
1224            .unwrap();
1225            std::fs::set_permissions(bin.join("git"), std::fs::Permissions::from_mode(0o755))
1226                .unwrap();
1227            let path = format!(
1228                "{}:{}",
1229                bin.display(),
1230                std::env::var("PATH").unwrap_or_default()
1231            );
1232            let out = std::process::Command::new("sh")
1233                .arg("-c")
1234                .arg(PREPARE)
1235                .arg("sh")
1236                .args([
1237                    checkout.to_str().unwrap(),
1238                    worktrees.to_str().unwrap(),
1239                    spool.to_str().unwrap(),
1240                    "t",
1241                    "HEAD",
1242                ])
1243                .env("PATH", path)
1244                .env("MJ_OFFLOAD_FETCH_TRIES", tries.to_string())
1245                .env("MJ_OFFLOAD_FETCH_BACKOFF", "0")
1246                .output()
1247                .expect("sh runs");
1248            let asked: u32 = std::fs::read_to_string(&count)
1249                .unwrap()
1250                .trim()
1251                .parse()
1252                .unwrap();
1253            if prepared {
1254                assert!(out.status.success(), "a lost race is asked again: {out:?}");
1255                assert_eq!(asked, refusals + 1, "asked until it won, and no more");
1256            } else {
1257                assert!(
1258                    !out.status.success(),
1259                    "a fetch that never wins fails: {out:?}"
1260                );
1261                assert_eq!(asked, tries, "asked exactly its declared tries");
1262                assert!(
1263                    String::from_utf8_lossy(&out.stderr).contains("cannot lock ref"),
1264                    "git's refusal reaches the caller: {out:?}"
1265                );
1266            }
1267        }
1268    }
