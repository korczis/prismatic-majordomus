1//! One shared server per checkout: the lease that decides who it is. The first
2//! `majordomus mcp` (or `serve`) to create `state/mcp/server.json` under the checkout's
3//! local half owns the server and publishes its URL there; every later process reads the
4//! file, checks that the server answers for this root, and attaches to it. A lease whose
5//! server does not answer is stale, and the next process takes it over; so is a file that
6//! is not a lease document, an empty one, or one whose owner never published a URL. A
7//! server that is slow to answer while its process is alive is busy, not stale: the
8//! election waits for it for [`BUSY_GRACE`] before it takes anything over. The file lives
9//! under `.ai/local/` (never tracked, by the layer's contract), and it is removed when the
10//! server stops — including when `SIGTERM`, `SIGINT` or `SIGHUP` asks it to — but only by
11//! the process that still holds it: every removal and every rewrite re-reads the file under
12//! the lock beside it (`server.lock`) and acts only on a lease that carries the actor's own
13//! token (`take_over_with` says why it is a lock).
14//!
15//! Per *checkout*, and this file said "per repository" until ADR 0044: a linked worktree
16//! is a checkout, so it has a manifest, a root, a lease and a server of its own, and on
17//! 2026-09-11 this repository had seven of them at once. What is repository-wide is the
18//! board, which `peers.list` gathers over these very leases; the election is unchanged.
19

46/// How long the probe of a published URL waits.

49/// How long a live server that does not answer its probe is waited for before its lease is
50/// taken over.
51///
52/// A probe that times out is not a dead server. On 2026-09-15 ten MCP clients attached at
53/// once to a server a shell entry had started; nine were bridged to it, and the tenth
54/// probed while the server was answering the other nine, heard nothing within
55/// [`PROBE_TIMEOUT`], took the lease over and started a second server for one checkout
56/// (test/cases/192). So a silent probe of a lease whose process is still alive is waited
57/// on, and asked again; only a server that stays silent for this long, or whose process is
58/// gone, loses its lease. The bound keeps the recovery from a hung server, and it sits
59/// well inside [`JOIN_TIMEOUT`] so that the take-over still happens within one election.
60pub const BUSY_GRACE: Duration = Duration::from_secs(10);
61
62/// How long a process keeps trying to acquire or join the lease before it gives up and
63/// says so: the bind grace with a margin for the probes. The caller then serves its
64/// client alone.
65pub const JOIN_TIMEOUT: Duration = Duration::from_secs(20);
66
67/// How long the shared server lets a connection go without a request read in full on it
68/// before it closes the connection: waiting for a request head, kept alive between requests,
69/// or reading a body. A client that sends one byte a second would otherwise hold a request
70/// thread forever, and enough of them would leave the probe above nobody to answer it (I2156).
71/// Generous, because a person's client on a slow link is a client too; far below anything
72/// that would let a handful of slow sockets outlast a [`BUSY_GRACE`] contest many times over.
73pub const READ_DEADLINE: Duration = Duration::from_secs(30);
74
75/// How long a server started for a checkout no client owns — by `serve ensure` or by a shell
76/// entering the checkout — outlives its last peer before it stops by itself:
77/// [`crate::cli::DEFAULT_IDLE_SECONDS`]. `serve --idle` on a command line overrides whatever
78/// is declared (I2701).
79pub const IDLE: Duration = Duration::from_secs(crate::cli::DEFAULT_IDLE_SECONDS);

81/// The timings above are the defaults. What a process actually judges a lease contest by is
82/// declared in `.ai/repo/policy.yaml`'s `server:` block and read once, here.
83///
84/// They were compiled constants until 2026-09-15: unchangeable without a rebuild, stated
85/// nowhere a reader would look, and invisible to every projection — while `probe_timeout` is
86/// the number that decides whether a live but slow owner keeps its lease or is taken over
87/// while it is still serving. That is a decision about how this repository is supervised, not
88/// an implementation detail, so it belongs in the model.
89///
90/// A `OnceLock`, because a process reads one policy: the repository is opened once and the
91/// answer must not change underneath a contest that is already being judged. Absent keys keep
92/// the constants, so a policy that cannot be read does not silently change behaviour.
93///
94/// ```
95/// use majordomus_cli::lease::{
96///     Timings, BIND_GRACE, BUSY_GRACE, IDLE, JOIN_TIMEOUT, PROBE_TIMEOUT, READ_DEADLINE,
97/// };
98/// // the defaults are the compiled constants, one for one
99/// let t = Timings::default();
100/// assert_eq!(t.bind_grace, BIND_GRACE);
101/// assert_eq!(t.probe_timeout, PROBE_TIMEOUT);
102/// assert_eq!(t.join_timeout, JOIN_TIMEOUT);
103/// assert_eq!(t.busy_grace, BUSY_GRACE);
104/// assert_eq!(t.read_deadline, READ_DEADLINE);
105/// assert_eq!(t.idle, IDLE);
106/// // a busy owner is given up on before the election that waits on it gives up itself
107/// assert!(t.busy_grace < t.join_timeout);
108/// ```
109#[derive(Debug, Clone, Copy, PartialEq, Eq)]
110pub struct Timings {
111    /// `server.bind_grace_seconds:`
112    pub bind_grace: Duration,
113    /// `server.probe_timeout_seconds:`
114    pub probe_timeout: Duration,
115    /// `server.join_timeout_seconds:`
116    pub join_timeout: Duration,
117    /// `server.busy_grace_seconds:`
118    pub busy_grace: Duration,
119    /// `server.read_deadline_seconds:`
120    pub read_deadline: Duration,
121    /// `server.idle_seconds:`
122    pub idle: Duration,
123}

154impl Timings {
155    /// What a declaration means, as a value rather than as a side effect.
156    ///
157    /// The mapping is the part that can be wrong; the `OnceLock` below is plumbing. Kept
158    /// separate so it can be tested directly — a test that went through the lock would
159    /// depend on which test ran first, since a process reads one policy by construction.
160    ///
161    /// ```
162    /// use majordomus_cli::lease::Timings;
163    /// use majordomus_cli::policy::ServerPolicy;
164    /// use std::time::Duration;
165    ///
166    /// // an empty declaration keeps every default
167    /// assert_eq!(Timings::from_policy(&ServerPolicy::default()), Timings::default());
168    ///
169    /// // and a declared value is the one used, rather than the constant
170    /// let declared = ServerPolicy {
171    ///     probe_timeout_seconds: Some(9),
172    ///     ..ServerPolicy::default()
173    /// };
174    /// let t = Timings::from_policy(&declared);
175    /// assert_eq!(t.probe_timeout, Duration::from_secs(9));
176    /// assert_ne!(t.probe_timeout, Timings::default().probe_timeout);
177    /// // the keys it says nothing about are untouched
178    /// assert_eq!(t.bind_grace, Timings::default().bind_grace);
179    ///
180    /// // the idle timeout is declared beside the others (I2701)
181    /// let declared = ServerPolicy { idle_seconds: Some(3), ..ServerPolicy::default() };
182    /// assert_eq!(Timings::from_policy(&declared).idle, Duration::from_secs(3));
183    /// ```
184    pub fn from_policy(policy: &crate::policy::ServerPolicy) -> Self {
185        let d = Self::default();
186        Self {
187            bind_grace: policy
188                .bind_grace_seconds
189                .map(Duration::from_secs)
190                .unwrap_or(d.bind_grace),
191            probe_timeout: policy
192                .probe_timeout_seconds
193                .map(Duration::from_secs)
194                .unwrap_or(d.probe_timeout),
195            join_timeout: policy
196                .join_timeout_seconds
197                .map(Duration::from_secs)
198                .unwrap_or(d.join_timeout),
199            busy_grace: policy
200                .busy_grace_seconds
201                .map(Duration::from_secs)
202                .unwrap_or(d.busy_grace),
203            read_deadline: policy
204                .read_deadline_seconds
205                .map(Duration::from_secs)
206                .unwrap_or(d.read_deadline),
207            idle: policy
208                .idle_seconds
209                .map(Duration::from_secs)
210                .unwrap_or(d.idle),
211        }
212    }
213}

241/// The identity of the file an executable was started from: where it is, and the mtime
242/// and size of the file at that path. A server outlives its own binary — a rebuild replaces
243/// the file under a process that keeps serving the code it loaded hours ago — and nothing
244/// about the process itself says so. This is what makes that visible.
245///
246/// ```
247/// use majordomus_cli::lease::ExecutableIdentity;
248/// let recorded = ExecutableIdentity { path: "/opt/majordomus".into(), mtime: 1_700_000_000, size: 42 };
249/// let text = serde_json::to_string(&recorded).unwrap();
250/// assert_eq!(serde_json::from_str::<ExecutableIdentity>(&text).unwrap(), recorded);
251/// ```
252#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
253pub struct ExecutableIdentity {
254    /// The path the process was started from.
255    pub path: PathBuf,
256    /// The file's modification time, seconds since the epoch.
257    pub mtime: u64,
258    /// The file's size in bytes.
259    pub size: u64,
260}
261
262impl ExecutableIdentity {
263    /// Has the file at the recorded path been replaced or removed since it was recorded?
264    /// The reason, for a reader; `None` when the same file is still there.
265    ///
266    /// ```
267    /// use majordomus_cli::lease::ExecutableIdentity;
268    /// let gone = ExecutableIdentity { path: "/nowhere/majordomus".into(), mtime: 1, size: 1 };
269    /// assert!(gone.replaced().unwrap().contains("no longer on disk"));
270    /// ```
271    pub fn replaced(&self) -> Option<String> {
272        let Ok(meta) = fs::metadata(&self.path) else {
273            return Some(format!(
274                "the executable it was started from, {}, is no longer on disk",
275                self.path.display()
276            ));
277        };
278        let mtime = meta
279            .modified()
280            .ok()
281            .and_then(|m| m.duration_since(SystemTime::UNIX_EPOCH).ok())
282            .map(|d| d.as_secs())
283            .unwrap_or(0);
284        if mtime == self.mtime && meta.len() == self.size {
285            return None;
286        }
287        Some(format!(
288            "the executable it was started from, {}, has been replaced since (it is serving \
289             code that is no longer on disk)",
290            self.path.display()
291        ))
292    }
293}

295/// The lease file, typed: what the process serving a checkout wrote down about itself.
296/// Every reader of the file — the election, a command asking which server is running, the
297/// environment snapshot, the server's own status — reads it through [`LeaseFile::read`],
298/// so that a field added here is added once and a file that is not a lease is refused in
299/// one place. A key this version does not know is ignored, so a lease written by a newer
300/// executable still reads; a key it needs that an older lease lacks takes its default.
301///
302/// ```
303/// use majordomus_cli::lease::{LeaseDocument, SCHEMA};
304/// // a lease an older server wrote: no pid, no version, a key this version has never heard of
305/// let older = format!(r#"{{"schema":"{SCHEMA}","token":"t","root":"/r","url":"http://127.0.0.1:1","later":1}}"#);
306/// let doc: LeaseDocument = serde_json::from_str(&older).unwrap();
307/// assert_eq!(doc.pid, 0);
308/// assert!(doc.version.is_none(), "older than any executable that reads this");
309/// assert_eq!(doc.url.as_deref(), Some("http://127.0.0.1:1"));
310/// // and what this version writes reads back as itself
311/// let text = serde_json::to_string(&doc).unwrap();
312/// assert_eq!(serde_json::from_str::<LeaseDocument>(&text).unwrap(), doc);
313/// ```
314#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
315pub struct LeaseDocument {
316    /// [`SCHEMA`].
317    pub schema: String,
318    /// The process id of the server. A live pid is not a live server: only an answering
319    /// probe says that. What a live pid does decide is patience — a probe that goes
320    /// unanswered by a live owner is asked again within [`BUSY_GRACE`] before the lease
321    /// counts as stale (the election's busy wait, [`probe_patiently`]); a dead or zero pid
322    /// gets no patience at all.
323    #[serde(default)]
324    pub pid: u32,
325    /// What makes the file this process's: only the process holding this token removes it.
326    #[serde(default)]
327    pub token: String,
328    /// The checkout root the server serves.
329    #[serde(default)]
330    pub root: PathBuf,
331    /// The address, once bound; `null` while the owner is still binding.
332    pub url: Option<String>,
333    /// When the server took the lease, RFC 3339.
334    #[serde(default)]
335    pub started_at: String,
336    /// The executable the server was started from, when it could be located.
337    #[serde(default)]
338    pub executable: Option<ExecutableIdentity>,
339    /// The executable's version. Absent in a lease written before this field existed,
340    /// which is itself a fact: that server is older than any executable that reads this.
341    #[serde(default, skip_serializing_if = "Option::is_none")]
342    pub version: Option<String>,
343}

345/// What a lease file holds, read once. Every state of the file is a variant, so that a
346/// reader decides what to do about each rather than learning it from an error.
347///
348/// ```
349/// use majordomus_cli::lease::LeaseFile;
350/// let dir = tempfile::tempdir().unwrap();
351/// let path = dir.path().join("server.json");
352/// assert_eq!(LeaseFile::read(&path), LeaseFile::Absent);
353/// assert!(LeaseFile::read(&path).document().is_none(), "no document to hand back");
354/// ```
355#[derive(Debug, Clone, PartialEq, Eq)]
356pub enum LeaseFile {
357    /// There is no file.
358    Absent,
359    /// A file with nothing in it: its owner created it and has not written it yet.
360    Empty,
361    /// A file that is not a lease document; the reason, for the log.
362    Corrupt(String),
363    /// A lease document.
364    Document(LeaseDocument),
365}
366
367impl LeaseFile {
368    /// Read and type the file at `path`. Never fails: every state of the file is one of the
369    /// variants, and what to do about each is the reader's decision.
370    ///
371    /// ```
372    /// use majordomus_cli::lease::{LeaseFile, SCHEMA};
373    /// let dir = tempfile::tempdir().unwrap();
374    /// let path = dir.path().join("server.json");
375    /// assert_eq!(LeaseFile::read(&path), LeaseFile::Absent);
376    /// std::fs::write(&path, "").unwrap();
377    /// assert_eq!(LeaseFile::read(&path), LeaseFile::Empty);
378    /// std::fs::write(&path, "{not json").unwrap();
379    /// assert!(matches!(LeaseFile::read(&path), LeaseFile::Corrupt(r) if r.starts_with("not JSON")));
380    /// std::fs::write(&path, r#"{"schema":"other/v1"}"#).unwrap();
381    /// assert!(matches!(LeaseFile::read(&path), LeaseFile::Corrupt(r) if r.contains(SCHEMA)));
382    /// std::fs::write(&path, format!(r#"{{"schema":"{SCHEMA}","token":"x","root":"/r","url":null}}"#)).unwrap();
383    /// match LeaseFile::read(&path) {
384    ///     LeaseFile::Document(d) => { assert_eq!(d.token, "x"); assert!(d.url.is_none()); assert!(d.version.is_none()); }
385    ///     other => panic!("{other:?}"),
386    /// }
387    /// ```
388    pub fn read(path: &Path) -> LeaseFile {
389        let text = match fs::read_to_string(path) {
390            Ok(text) => text,
391            Err(e) if e.kind() == std::io::ErrorKind::NotFound => return LeaseFile::Absent,
392            Err(e) => return LeaseFile::Corrupt(format!("cannot be read ({e})")),
393        };
394        if text.trim().is_empty() {
395            return LeaseFile::Empty;
396        }
397        let value: Value = match serde_json::from_str(&text) {
398            Ok(value) => value,
399            Err(e) => return LeaseFile::Corrupt(format!("not JSON ({e})")),
400        };
401        if !value.is_object() || value["schema"] != SCHEMA {
402            return LeaseFile::Corrupt(format!("not a {SCHEMA} document"));
403        }
404        match serde_json::from_value::<LeaseDocument>(value) {
405            Ok(document) => LeaseFile::Document(document),
406            Err(e) => LeaseFile::Corrupt(format!("not a {SCHEMA} document ({e})")),
407        }
408    }
409
410    /// The document, when the file holds one; `None` for an absent, empty or corrupt file,
411    /// for a reader that has no different answer for those three.
412    ///
413    /// ```
414    /// use majordomus_cli::lease::{LeaseDocument, LeaseFile, SCHEMA};
415    /// let doc = LeaseDocument {
416    ///     schema: SCHEMA.into(), pid: 7, token: "t".into(), root: "/r".into(), url: None,
417    ///     started_at: "2026-09-10T00:00:00Z".into(), executable: None, version: None,
418    /// };
419    /// assert_eq!(LeaseFile::Document(doc.clone()).document(), Some(&doc));
420    /// assert_eq!(LeaseFile::Corrupt("not JSON".into()).document(), None);
421    /// ```
422    pub fn document(&self) -> Option<&LeaseDocument> {
423        match self {
424            LeaseFile::Document(d) => Some(d),
425            _ => None,
426        }
427    }
428}

449/// Is `doc` the lease this very process holds, published and not lost?
450///
451/// A server asked where its own checkout stands must answer from memory. Probing the
452/// address its lease names is a request to itself, served by the same small pool of HTTP
453/// workers the question arrived on: with every worker answering such a question at once,
454/// none is left to answer the probes, each waits out [`PROBE_TIMEOUT`], the server calls
455/// itself stale — and a client electing in that window times out on the same queue and
456/// takes a live lease over. Measured on 2026-09-15: ten MCP clients calling
457/// `majordomus_peers` at once, and one of them started a second server of the checkout.
458///
459/// ```
460/// use majordomus_cli::lease::{is_own, LeaseDocument};
461/// let doc: LeaseDocument = serde_json::from_str(
462///     r#"{"schema":"lease/v1","pid":1,"token":"t","url":"http://127.0.0.1:1"}"#,
463/// )
464/// .unwrap();
465/// // a process that holds no lease owns no document
466/// assert!(!is_own(&doc));
467/// ```
468pub fn is_own(doc: &LeaseDocument) -> bool {
469    !was_lost()
470        && held().is_some_and(|h| h.token == doc.token && h.pid == doc.pid && h.url == doc.url)
471}

536/// Decide whether this process serves the repository or attaches to the process that does.
537///
538/// An existing file is read on every attempt and classified: the lease of a live server
539/// (it answers for this root) is attached to; a stale one (its server does not answer),
540/// a corrupt one (not a lease document), an empty one, or an abandoned one (no URL after
541/// [`BIND_GRACE`]) is taken over; a fresh lease without a URL means its owner is still
542/// binding, and this process waits for it. Nothing a client leaves behind can lock the
543/// others out; when the file can neither be created nor removed within [`JOIN_TIMEOUT`],
544/// the error names the path and the caller serves its client alone.
545pub fn elect(repo: &Repository) -> Result<Role> {
546    let path = lease_path(repo);
547    let root = repo.root().to_path_buf();
548    if let Some(dir) = path.parent() {
549        fs::create_dir_all(dir).map_err(|e| Error::io(dir, e))?;
550    }
551    let waited_since = Instant::now();
552    // when a live server first failed to answer this election; cleared whenever it answers,
553    // or the file stops naming a busy server
554    let mut busy_since: Option<Instant> = None;
555    loop {
556        match fs::OpenOptions::new()
557            .write(true)
558            .create_new(true)
559            .open(&path)
560        {
561            Ok(mut file) => {
562                let token = format!(
563                    "{}-{:x}",
564                    std::process::id(),
565                    SystemTime::now()
566                        .duration_since(SystemTime::UNIX_EPOCH)
567                        .map(|d| d.as_nanos())
568                        .unwrap_or(0)
569                );
570                let lease = Lease {
571                    path: path.clone(),
572                    token,
573                    root,
574                    released: false,
575                    started_at: crate::peers::rfc3339(SystemTime::now()),
576                    binding: Arc::new(Mutex::new(true)),
577                };
578                let text =
579                    serde_json::to_string(&lease.document(None)).map_err(|e| Error::Lease {
580                        reason: format!("cannot render the lease: {e}"),
581                    })?;
582                file.write_all(text.as_bytes())
583                    .map_err(|e| Error::io(&path, e))?;
584                hold(&path, &lease.token);
585                return Ok(Role::Server(lease));
586            }
587            Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
588                let (found, seen) = inspect(&path, &root);
589                if !matches!(found, Found::Busy(_)) {
590                    busy_since = None;
591                }
592                match found {
593                    Found::Live(url) => return Ok(Role::Peer { url }),
594                    Found::Busy(url) => {
595                        let since = *busy_since.get_or_insert_with(Instant::now);
596                        if since.elapsed() < timings().busy_grace {
597                            std::thread::sleep(Duration::from_millis(200));
598                        } else {
599                            tracing::warn!(
600                                lease = %path.display(),
601                                "the server at {url} is alive and has not answered for {} seconds; taking it over",
602                                timings().busy_grace.as_secs()
603                            );
604                            take_over(&path, &seen)?;
605                            busy_since = None;
606                        }
607                    }
608                    Found::Stale(reason) => {
609                        tracing::warn!(lease = %path.display(), "{reason}; taking it over");
610                        take_over(&path, &seen)?;
611                    }
612                    // an owner that is still binding keeps its file young (`keep_alive`), so
613                    // a file that is old enough to be abandoned is classified so by `inspect`
614                    // and this arm only ever waits; the whole attempt is bounded below
615                    Found::Binding => std::thread::sleep(Duration::from_millis(100)),
616                }
617            }
618            Err(e) => return Err(Error::io(&path, e)),
619        }
620        if waited_since.elapsed() > timings().join_timeout {
621            return Err(Error::Lease {
622                reason: format!(
623                    "could not acquire or join the lease at {} within {} seconds",
624                    path.display(),
625                    timings().join_timeout.as_secs()
626                ),
627            });
628        }
629    }
630}

645/// What the running executable is, for the lease to record: the path it was started from
646/// and the identity of the file at that path. A server outlives its own binary — a rebuild
647/// replaces the file under a process that keeps serving the code it loaded hours ago — and
648/// nothing about the process itself says so. This is what makes that visible.
649///
650/// `None` when the executable cannot be located or stat'ed; the lease then carries no
651/// claim, and the election's `superseded` check makes none either.
652///
653/// ```
654/// use majordomus_cli::lease::executable_identity;
655/// // the process running this example can locate itself, and the file it names is there
656/// let mine = executable_identity().expect("a test binary knows where it is");
657/// assert!(mine.path.is_file());
658/// assert!(mine.replaced().is_none(), "the file that is running has not been replaced");
659/// ```
660pub fn executable_identity() -> Option<ExecutableIdentity> {
661    let path = std::env::current_exe().ok()?;
662    let meta = fs::metadata(&path).ok()?;
663    let mtime = meta
664        .modified()
665        .ok()?
666        .duration_since(SystemTime::UNIX_EPOCH)
667        .ok()?
668        .as_secs();
669    Some(ExecutableIdentity {
670        path,
671        mtime,
672        size: meta.len(),
673    })
674}

695/// Why this process is serving code that is no longer on disk, or `None` while the file it
696/// was started from is still the file at that path. The same judgement the lease reader

708/// The judgement [`serving_replaced_code`] makes, over an identity it is handed rather than

719            .to_string()
720    } else {
721        "the executable this process was started from has been removed since it started: \

723            .to_string()
724    })

727/// Has the executable behind a lease been replaced since that server started?
728///
729/// Only a lease naming *this process's own* executable path can answer: same path, a file
730/// that is now a different file, and the server on the other end is provably running code
731/// that no longer exists on disk. A lease naming some other path — a release install next
732/// to a debug build — makes no claim either way and is left alone, so two legitimate
733/// binaries never fight over the lease.
734fn superseded(doc: &LeaseDocument) -> Option<String> {
735    executable_identity().and_then(|me| superseded_by(doc, &me, crate::VERSION))
736}

738/// The judgement [`superseded`] makes, over the identity and version it is handed.
739///
740/// A file replaced at the lease's path says the server runs code no longer on disk, not that
741/// this process runs newer code: a build of an older tree written to the same path is a
742/// replacement too. So a server whose lease records a newer version than `my_version` is
743/// never taken over on that ground (I3097): this process is the older one, and attaching to
744/// the newer server is the verdict [`version_verdict`] already gives everywhere else.
745fn superseded_by(
746    doc: &LeaseDocument,
747    mine: &ExecutableIdentity,
748    my_version: &str,
749) -> Option<String> {
750    let recorded = doc.executable.as_ref()?;
751    if recorded.path != mine.path {
752        // a server started from the build output this process runs a copy of (I2059)
753        let origin = crate::served_executable::copied_from(&mine.path)?;
754        let same = origin == recorded.path
755            || matches!(
756                (fs::canonicalize(&origin), fs::canonicalize(&recorded.path)),
757                (Ok(a), Ok(b)) if a == b
758            );
759        if !same || recorded.replaced().is_none() {
760            return None;
761        }
762    } else if recorded.mtime == mine.mtime && recorded.size == mine.size {
763        return None;
764    }
765    if version_verdict(doc.version.as_deref(), my_version) == VersionVerdict::ServerNewer {
766        tracing::warn!(
767            server = doc.version.as_deref().unwrap_or_default(),
768            mine = my_version,
769            "the server's executable was replaced, but the server runs a newer version than this \
770             executable; attaching to it rather than taking it over"
771        );
772        return None;
773    }
774    Some(format!(
775        "superseded lease: the server was started from {} and that file has been replaced since \
776         (it is serving code that is no longer on disk)",
777        recorded.path.display()
778    ))
779}

781/// How the version a server published in its lease stands to this executable's.
782///
783/// One verdict, read by the election, by `serve ensure` and by `server.status`, so that the
784/// three never disagree about whether a server is older (I2163). The installer puts each
785/// release in a directory of its own, so an upgrade changes the executable's path and
786/// `superseded` — same path, different file — never fires on it: before this verdict a
787/// new client attached to the old server and was served yesterday's code until it idled out.
788///
789/// ```
790/// use majordomus_cli::lease::{version_verdict, VersionVerdict};
791/// // only an older release is replaced; every other verdict attaches
792/// let replaces = |served| version_verdict(served, "0.20.0") == VersionVerdict::ServerOlder;
793/// assert!(replaces(Some("0.19.1")));
794/// assert!(!replaces(Some("0.20.0")) && !replaces(Some("0.21.0")) && !replaces(None));
795/// ```
796#[derive(Debug, Clone, Copy, PartialEq, Eq)]
797pub enum VersionVerdict {
798    /// The version this executable is.
799    Same,
800    /// An older release than this executable: a client of this executable replaces it.
801    ServerOlder,
802    /// A newer release: a client of this executable attaches to it and says it is older.
803    ServerNewer,
804    /// No version published: a server from before leases carried one. Reported as outdated,
805    /// never taken over on that ground alone.
806    Unpublished,
807    /// Two versions that do not order as releases — one is not a semantic version (a local
808    /// build tag), or the two differ only in build metadata: reported as different, never
809    /// acted on.
810    Unordered,
811}

813/// Decide how `served`, the version in a lease, stands to `mine`.
814///
815/// Ordered by [`Version`], the crate's one reading of a version, so a server is older or
816/// newer by exactly the semantic-version precedence the release writer and the stable
817/// pointer use (ADR 0004): a pre-release sits below its own final and above every earlier
818/// release. A leading `v` is a tag's spelling and is read past.
819///
820/// ```
821/// use majordomus_cli::lease::{version_verdict, VersionVerdict};
822/// assert_eq!(version_verdict(Some("0.19.1"), "0.19.1"), VersionVerdict::Same);
823/// assert_eq!(version_verdict(Some("0.19.1"), "0.20.0"), VersionVerdict::ServerOlder);
824/// assert_eq!(version_verdict(Some("0.9.10"), "0.10.0"), VersionVerdict::ServerOlder);
825/// assert_eq!(version_verdict(Some("1.0.0"), "0.20.0"), VersionVerdict::ServerNewer);
826/// assert_eq!(version_verdict(None, "0.20.0"), VersionVerdict::Unpublished);
827/// // a pre-release orders by precedence: below its own final, above an earlier release
828/// assert_eq!(version_verdict(Some("0.20.0-rc.1"), "0.20.0"), VersionVerdict::ServerOlder);
829/// assert_eq!(version_verdict(Some("1.0.0-alpha.1"), "0.29.0"), VersionVerdict::ServerNewer);
830/// // what is not a semantic version orders nothing
831/// assert_eq!(version_verdict(Some("dev"), "0.20.0"), VersionVerdict::Unordered);
832/// ```
833pub fn version_verdict(served: Option<&str>, mine: &str) -> VersionVerdict {
834    let release = |v: &str| Version::parse(v.strip_prefix('v').unwrap_or(v));
835    let Some(served) = served else {
836        return VersionVerdict::Unpublished;
837    };
838    if served == mine {
839        return VersionVerdict::Same;
840    }
841    match (release(served), release(mine)) {
842        (Some(theirs), Some(ours)) if theirs < ours => VersionVerdict::ServerOlder,
843        (Some(theirs), Some(ours)) if theirs > ours => VersionVerdict::ServerNewer,
844        _ => VersionVerdict::Unordered,
845    }
846}

848/// Read and classify an existing lease file. What was read comes back beside the verdict,
849/// so that a take-over can insist on removing the file it judged and not one that arrived
850/// in the meantime.
851fn inspect(path: &Path, root: &Path) -> (Found, LeaseFile) {
852    let age = file_age(path);
853    let seen = LeaseFile::read(path);
854    let doc = match &seen {
855        // gone between the failed create and this read: the next attempt creates it
856        LeaseFile::Absent => return (Found::Binding, seen),
857        LeaseFile::Empty if age > timings().bind_grace => {
858            return (
859                Found::Stale("empty lease: its owner never wrote it".into()),
860                seen,
861            )
862        }
863        LeaseFile::Empty => return (Found::Binding, seen),
864        LeaseFile::Corrupt(reason) => {
865            return (Found::Stale(format!("corrupt lease: {reason}")), seen)
866        }
867        LeaseFile::Document(doc) => doc.clone(),
868    };
869    // before asking whether it answers: a server that answers from a binary that has been
870    // replaced answers with yesterday's code, which is the harder failure to see
871    if let Some(reason) = superseded(&doc) {
872        return (Found::Stale(reason), seen);
873    }
874    let found = match doc.url.as_deref() {
875        Some(url) => match ask(url, root, timings().probe_timeout) {
876            // an older release answering is replaced, not joined: its sessions move on to the
877            // new server once it sees its lease taken (I2127), and nobody is served
878            // yesterday's code because the installer put today's at another path (I2163)
879            Answer::Ours(_)
880                if version_verdict(doc.version.as_deref(), crate::VERSION)
881                    == VersionVerdict::ServerOlder =>
882            {
883                Found::Stale(format!(
884                    "superseded lease: the server at {url} runs version {}, older than this executable's {}",
885                    doc.version.as_deref().unwrap_or_default(),
886                    crate::VERSION
887                ))
888            }
889            Answer::Ours(_) => Found::Live(url.to_string()),
890            // a live owner that is silent is busy and waited on — unless the process that
891            // started this one already waited out that patience on this very lease
892            Answer::Silent if alive(doc.pid) && !judged_stale(&doc) => Found::Busy(url.to_string()),
893            Answer::Silent | Answer::NotOurs => Found::Stale(format!(
894                "stale lease: the server it names at {url} does not answer for this repository"
895            )),
896        },
897        None if age > timings().bind_grace => {
898            Found::Stale("abandoned lease: its owner never published a URL".into())
899        }
900        None => Found::Binding,
901    };
902    (found, seen)
903}

905/// How long ago the file was last written; zero when that cannot be read, so that a file
906/// whose age is unknown is never taken for an old one.
907///
908/// ```
909/// use majordomus_cli::lease::file_age;
910/// use std::time::Duration;
911/// let dir = tempfile::tempdir().unwrap();
912/// let path = dir.path().join("server.json");
913/// assert_eq!(file_age(&path), Duration::ZERO, "no file: no age");
914/// std::fs::write(&path, "").unwrap();
915/// assert!(file_age(&path) < Duration::from_secs(60), "just written");
916/// ```
917pub fn file_age(path: &Path) -> Duration {
918    fs::metadata(path)
919        .and_then(|m| m.modified())
920        .ok()
921        .and_then(|m| m.elapsed().ok())
922        .unwrap_or(Duration::ZERO)
923}

925/// Remove a lease that cannot be used, so that the next attempt creates a fresh one — but
926/// only the file that was judged: when another process has replaced it since, the file on
927/// disk is somebody's fresh lease, and the next round of the election reads that one.
928fn take_over(path: &Path, seen: &LeaseFile) -> Result<()> {
929    take_over_with(path, seen, LOCK_PATIENCE, || {})

932/// [`take_over`], with the patience for the [change lock](lock_for_change) stated and a seam
933/// between the comparison and the removal, where the unit tests put a second elector.
934///
935/// # Why a lock, and not a rename aside
936///
937/// Until I2128 this was three steps — read, compare with what was judged, remove — and two
938/// electors that had both judged one lease stale could interleave: A removes the stale lease
939/// and creates its own, and B, which had already compared, removes A's. A then serves with no
940/// lease on disk, and the next client starts a second server for the checkout.
941///
942/// The textbook repair for a compare-and-delete on a file is to rename it aside under a name
943/// of one's own and look at what was moved: the rename is atomic, so whatever was moved is
944/// the mover's to judge. It was considered and not taken, because it cannot be used for the
945/// other half of the problem. A lease is also *rewritten* — published, kept young while the
946/// layer loads — and a rewrite that moved the file aside to check it would leave the path
947/// empty for a moment on every rewrite, during which any elector's `create_new` succeeds:
948/// the owner would lose its lease to its own bookkeeping. A moved-aside lease that turned out
949/// to be somebody else's has the same hole on the way back.
950///
951/// So every change to the file — this removal, [`Lease::publish`], the keep-alive rewrite,
952/// the release — is made under one advisory lock beside the lease (`server.lock`), and each
953/// re-reads the file under it before acting. Creating the lease needs no lock: `create_new`
954/// only succeeds on an empty path, and a path holding a lease can only be emptied under the
955/// lock, by a holder that has just read it. Two electors that judged one lease stale then end
956/// with exactly one lease: whichever takes the lock first removes the stale file, the other
957/// finds a file that is not the one it judged and leaves it, and `create_new` lets exactly
958/// one of them create the next.
959///
960/// What it does not cover is an executable older than this one, which takes no lock; two
961/// builds side by side keep the old window between them until the older one is gone.
962fn take_over_with(
963    path: &Path,
964    seen: &LeaseFile,
965    patience: Duration,
966    between: impl FnOnce(),
967) -> Result<()> {
968    let Some(_lock) = lock_for_change(path, patience) else {
969        tracing::debug!(lease = %path.display(), "another process is changing the lease; reading it again");
970        return Ok(());
971    };
972    if LeaseFile::read(path) != *seen {
973        tracing::debug!(lease = %path.display(), "the lease changed under the take-over; reading it again");
974        return Ok(());
975    }
976    between();
977    match fs::remove_file(path) {
978        Ok(()) => Ok(()),
979        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
980        Err(e) => Err(Error::Lease {
981            reason: format!(
982                "cannot remove the unusable lease at {}: {e}",
983                path.display()
984            ),
985        }),
986    }
987}

989/// How long a change to the lease waits for another process's change to finish. A change
990/// holds the lock for one small read and one rename or removal, so anything near this bound
991/// is a holder that has been stopped (`SIGSTOP`, a debugger), and the change is then not
992/// made: a lease left in place is recovered by the next election, a lease removed or
993/// overwritten in error is not.
994const LOCK_PATIENCE: Duration = Duration::from_secs(2);
995
996/// The advisory lock every change to a lease file is made under: an exclusive `flock(2)` on
997/// `server.lock` beside it, held for as long as the value lives. The kernel releases it when
998/// the descriptor closes, so a holder that is killed cannot keep it.
999///
1000/// Where the lock file cannot be opened at all (a directory that refuses writes) or the
1001/// filesystem refuses `flock`, the change goes ahead without it: the check that the file is
1002/// the actor's own is still made, as it was before the lock existed.
1003struct ChangeLock {
1004    #[cfg(unix)]
1005    _file: Option<fs::File>,
1006}
1007
1008/// The lock file beside a lease.
1009fn lock_path(lease: &Path) -> PathBuf {
1010    lease.with_extension("lock")
1011}
1012
1013/// Take the [`ChangeLock`] of the lease at `lease`, waiting up to `patience` for another
1014/// holder; `None` when it is still held after that. A `patience` of zero is one attempt.
1015fn lock_for_change(lease: &Path, patience: Duration) -> Option<ChangeLock> {
1016    #[cfg(unix)]
1017    {
1018        use std::os::unix::io::AsRawFd;
1019        lock_for_change_with(lease, patience, |file| {
1020            // SAFETY: flock(2) on a descriptor the caller owns; no memory is touched.
1021            if unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX | libc::LOCK_NB) } == 0 {
1022                Ok(())
1023            } else {
1024                Err(std::io::Error::last_os_error())
1025            }
1026        })
1027    }
1028    #[cfg(not(unix))]
1029    {
1030        let _ = (lease, patience);
1031        Some(ChangeLock {})
1032    }
1033}
1034
1035/// [`lock_for_change`], taking the lock with `try_lock`: `flock(2)` in earnest, a test's own
1036/// refusals in a test, since a file just opened is one the system never refuses for anything
1037/// but contention.
1038#[cfg(unix)]
1039fn lock_for_change_with(
1040    lease: &Path,
1041    patience: Duration,
1042    try_lock: impl Fn(&fs::File) -> std::io::Result<()>,
1043) -> Option<ChangeLock> {
1044    let Ok(file) = fs::OpenOptions::new()
1045        .create(true)
1046        .truncate(false)
1047        .write(true)
1048        .open(lock_path(lease))
1049    else {
1050        return Some(ChangeLock { _file: None });
1051    };
1052    let deadline = Instant::now() + patience;
1053    loop {
1054        let Err(error) = try_lock(&file) else {
1055            return Some(ChangeLock { _file: Some(file) });
1056        };
1057        let contended = error.raw_os_error() == Some(libc::EWOULDBLOCK)
1058            || error.kind() == std::io::ErrorKind::Interrupted;
1059        if !contended {
1060            return Some(ChangeLock { _file: None });
1061        }
1062        if Instant::now() >= deadline {
1063            return None;
1064        }
1065        std::thread::sleep(Duration::from_millis(2));
1066    }
1067}

1069/// Does the file at `path` hold the lease named by `token`? The token is minted once per
1070/// election, from the pid and the clock, so it names one process's one generation of the
1071/// lease: a process that took the lease, lost it and took it again holds a different token.
1072fn holds(path: &Path, token: &str) -> bool {
1073    LeaseFile::read(path)
1074        .document()
1075        .is_some_and(|d| d.token == token)
1076}

1078/// Remove the lease at `path` if, and only if, it is still the one `token` names — under the
1079/// [`ChangeLock`], so that nobody can replace the file between the check and the removal.
1080/// Whether it was removed.
1081fn remove_if_held(path: &Path, token: &str) -> bool {
1082    let Some(_lock) = lock_for_change(path, LOCK_PATIENCE) else {

1084        return false;

1087}

1117/// The lease this process holds, by path and token, from the election until it is released
1118/// or lost: what [`release_held`] removes when the process is asked to stop by a signal and
1119/// the [`Lease`] itself belongs to a thread that will not get to drop it.
1120static HOLDING: Mutex<Option<(PathBuf, String)>> = Mutex::new(None);
1121
1122/// This process now holds the lease at `path` under `token`; install the signal handlers the
1123/// first time.
1124fn hold(path: &Path, token: &str) {
1125    if let Ok(mut holding) = HOLDING.lock() {
1126        *holding = Some((path.to_path_buf(), token.to_string()));
1127    }
1128    signals::install();
1129}

1131/// This process no longer holds the lease `token` names (or, with `None`, any lease).
1132fn forget(token: Option<&str>) {
1133    if let Ok(mut holding) = HOLDING.lock() {
1134        if token.is_none_or(|t| holding.as_ref().is_some_and(|(_, h)| h == t)) {
1135            *holding = None;
1136        }
1137    }
1138}

1140/// Release the lease this process holds, if the file is still its own: the release a
1141/// process makes when it is stopped by a signal, from a thread that does not own the
1142/// [`Lease`]. A lease taken over since is somebody else's and stays where it is.
1143pub(crate) fn release_held() {
1144    let taken = HOLDING.lock().ok().and_then(|mut h| h.take());
1145    release_record(taken);
1146}

1148/// [`release_held`] over a record it is handed rather than the process-wide one, so that a
1149/// test can show it without touching what a concurrent test's election recorded.
1150fn release_record(record: Option<(PathBuf, String)>) {
1151    if let Some((path, token)) = record {
1152        remove_if_held(&path, &token);
1153        if let Ok(mut slot) = published().lock() {
1154            if slot.as_ref().is_some_and(|d| d.token == token) {
1155                *slot = None;
1156            }
1157        }
1158    }

1179/// End the process the way the signal would have: release the lease if it is still this
1180/// process's, restore the signal's default disposition and send it again, so that the exit
1181/// status still names the signal.
1182pub(crate) fn end_by_signal(signal: i32) -> ! {
1183    release_held();
1184    #[cfg(unix)]
1185    // SAFETY: restoring a default disposition and signalling this process; no memory is
1186    // touched.
1187    unsafe {
1188        libc::signal(signal, libc::SIG_DFL);
1189        libc::kill(libc::getpid(), signal);
1190    }
1191    // the default disposition of every signal the handlers take ends the process; this is
1192    // the moment between the sending and the delivery
1193    std::thread::sleep(Duration::from_secs(1));
1194    std::process::exit(128 + signal)
1195}

1197/// What the signal watcher does about the signals received so far.

1202    /// End the process now.

1206/// The watcher's decision, as a value: no signal is nothing to do; the first, in a process
1207/// that answers stop requests, is left to it for [`STOP_BOUND`]; a first one nobody answers,

1217/// Set by [`lost`], never cleared: a lease this process was given and no longer has.

1229/// This process's lease was taken over by another process: from now on nothing this process
1230/// does on its way out touches the file, which is somebody else's, and it stops claiming to be the
1231/// checkout's server — [`held`] answers `None` and [`was_lost`] answers `true` from here

1233/// ones, or answer a stranger's probe as though it were still the one.

1235/// ```
1236/// // a process that holds no lease has nothing to lose; saying so twice changes nothing

1238/// majordomus_cli::lease::lost();

1240/// assert!(majordomus_cli::lease::was_lost(), "it was told the lease is gone");
1241/// ```
1242pub fn lost() {
1243    let _ = LOST_AT.set(Instant::now());
1244    LOST.store(true, Ordering::SeqCst);
1245    if let Ok(mut published) = published().lock() {
1246        *published = None;
1247    }
1248    forget(None);
1249}

1284/// Does a Majordomus server answer at `url` for the repository at `root`, *as* that
1285/// repository's server?
1286///
1287/// Three questions, and the third is the one a remembered address needs: it is a
1288/// Majordomus server, it serves this root, and it still holds this checkout's lease.
1289/// Without the third, a process whose lease was taken over answers every probe exactly as
1290/// the current server does — same name, same `repository_id`, its own peer board, its own
1291/// generation of the layer — and a client holding an address from before the takeover is
1292/// served plausible answers by a server nobody else is talking to. That was measured on
1293/// 2026-09-10: two servers of one checkout, one lease, and the older one indistinguishable
1294/// from the current one on this endpoint.
1295///
1296/// A server too old to answer the question is accepted. It cannot be told from a current
1297/// one here, and refusing it would be the worse failure: a live server taken for dead is
1298/// taken over, which is how one checkout comes to have two.

1303/// [`probe`], keeping what the server answered: its index document when it answers as the
1304/// leaseholder of this checkout, `None` otherwise. One decision, so a caller that also wants
1305/// the surfaces the server lists makes neither a second request nor a second judgement.
1306///
1307/// ```
1308/// use majordomus_cli::lease::probe_reply;
1309/// use std::time::Duration;
1310/// // nothing listens on port 1 of the loopback address
1311/// let root = std::path::Path::new("/r");
1312/// assert!(probe_reply("http://127.0.0.1:1", root, Duration::from_millis(50)).is_none());
1313/// ```
1314pub fn probe_reply(url: &str, root: &Path, timeout: Duration) -> Option<Value> {
1315    match ask(url, root, timeout) {
1316        Answer::Ours(index) => Some(index),
1317        Answer::NotOurs | Answer::Silent => None,
1318    }
1319}

1321/// What a probe of a published URL heard.
1322enum Answer {
1323    /// A Majordomus server that serves this root and holds its lease, with the index
1324    /// document it answered.
1325    Ours(Value),
1326    /// Something answered that is not this checkout's server, or nothing is listening.
1327    NotOurs,
1328    /// The connection was made and no answer came within the timeout: a server that is
1329    /// busy looks exactly like this, and so does a hung one.
1330    Silent,
1331}

1333/// Ask a published URL the probe's three questions within `timeout`, telling a server that
1334/// said no apart from one that said nothing in time.
1335fn ask(url: &str, root: &Path, timeout: Duration) -> Answer {
1336    // marked as a probe, so that asking whether a server is there does not keep it there
1337    // (I4007): a server's idle timer counts a client reading its home page, not this
1338    let probe = [(crate::idle::PROBE_HEADER, "1")];
1339    match bridge::request(url, "GET", "/", &probe, None, timeout) {
1340        Ok(reply) if reply.status == 200 => {
1341            let v: Value = serde_json::from_str(&reply.body).unwrap_or(Value::Null);
1342            // the identity and not the path: the index names the repository it serves
1343            // without telling every caller where the checkout sits
1344            let ours = v["name"] == "majordomus"
1345                && v["repository_id"].as_str() == Some(crate::repository::identity(root).as_str())
1346                && v[LEASEHOLDER_KEY] != Value::Bool(false);
1347            if ours {
1348                Answer::Ours(v)
1349            } else {
1350                Answer::NotOurs
1351            }
1352        }
1353        Ok(_) => Answer::NotOurs,
1354        Err(e)
1355            if matches!(
1356                e.kind(),
1357                std::io::ErrorKind::TimedOut | std::io::ErrorKind::WouldBlock
1358            ) =>
1359        {
1360            Answer::Silent
1361        }
1362        Err(_) => Answer::NotOurs,
1363    }
1364}

1366/// Is the process with this id alive on this machine? A lease is a file of this checkout,
1367/// so its pid is a pid of this host. `kill(pid, 0)` delivers nothing and asks only whether
1368/// the process exists; a process owned by somebody else answers `EPERM`, which is alive.
1369/// Whether a process with this pid exists (signal 0: checked, nothing sent).
1370pub(crate) fn alive(pid: u32) -> bool {
1371    if pid == 0 {
1372        return false;
1373    }
1374    let Ok(pid) = libc::pid_t::try_from(pid) else {
1375        return false;
1376    };
1377    // SAFETY: signal 0 performs the existence and permission check and sends nothing.
1378    let rc = unsafe { libc::kill(pid, 0) };
1379    rc == 0 || std::io::Error::last_os_error().raw_os_error() == Some(libc::EPERM)
1380}

1382/// [`probe`], with patience for an owner that is alive but slow.
1383///
1384/// A probe that times out says nothing about whether the server is gone: a server loading
1385/// a large layer, or one on a machine under load (an instrumented test run, a loaded CI
1386/// runner), answers late. Judging it gone from one silent probe makes `serve ensure` start
1387/// a second server beside it and report the old one's address as though it had started it.
1388/// So when the probe fails and the lease's `pid` is a live process on this machine, the
1389/// probe is repeated every second for up to `patience` — the caller's share of
1390/// [`Timings::busy_grace`], the same wait the election gives a busy owner. The answer is
1391/// `false` only when every attempt fails.
1392///
1393/// The patience is bounded on purpose. A live pid can be a wedged server that never answers
1394/// again, and a probe that waited on it forever would make every client hang on a ghost. A
1395/// dead pid, or pid 0, gets no patience at all, so a killed server is recovered at once,
1396/// and a `patience` of zero is a single probe.
1397///
1398/// ```
1399/// use majordomus_cli::lease::probe_patiently;
1400/// use std::path::Path;
1401/// use std::time::{Duration, Instant};
1402/// // nothing listens on port 1 and pid 0 names no process: one refused probe, no patience
1403/// let t0 = Instant::now();
1404/// let patience = Duration::from_secs(10);
1405/// assert!(!probe_patiently("http://127.0.0.1:1", Path::new("/nowhere"), 0, patience));
1406/// assert!(t0.elapsed() < Duration::from_secs(5));
1407/// ```
1408pub fn probe_patiently(url: &str, root: &Path, pid: u32, patience: Duration) -> bool {
1409    if probe(url, root) {
1410        return true;
1411    }
1412    let deadline = Instant::now() + patience;
1413    while alive(pid) && Instant::now() < deadline {
1414        std::thread::sleep(Duration::from_secs(1));
1415        if probe(url, root) {
1416            return true;
1417        }
1418    }
1419    false
1420}

1423/// one? Its token is then in [`JUDGED_STALE_ENV`], and the election takes exactly that
1424/// lease over at once instead of spending the same patience on it a second time: waiting
1425/// twice would only double the delay before a wedged owner is replaced. Any other lease —
1426/// a new owner, or one that changed in between — still gets the full wait.

1428    !doc.token.is_empty() && std::env::var(JUDGED_STALE_ENV).is_ok_and(|token| token == doc.token)
1429}

1455    /// waits for it only as long as the file is younger than [`BIND_GRACE`]; a cold start
1456    /// that takes longer than that — a large layer, a machine under load — would otherwise
1457    /// be taken for an abandoned one and taken over while its owner is still binding. So
1458    /// until [`Lease::publish`], a thread rewrites the same document every third of the
1459    /// grace, under the lock `publish` also takes, so that a touch can never land after the
1460    /// URL. It writes only while the file is still this process's: a lease that was taken
1461    /// over anyway is somebody else's to write.

1463    /// ```
1464    /// use majordomus_cli::lease::{elect, held, LeaseFile, Role};

1469    ///     "schema: ai-repository/v1\nrepo:\n  path: repo\nlocal:\n  path: local\n  tracked: false\n  implicit_context: false\nsections:\n  policy: repo/policy.yaml\n").unwrap();
1470    /// let repo = Repository::discover(dir.path()).unwrap();
1471    /// let Role::Server(lease) = elect(&repo).unwrap() else { panic!("nobody else serves a fresh directory") };
1472    /// lease.keep_alive();                         // returns at once; the touching runs beside the load
1473    /// assert!(held().is_none(), "nothing is published yet");
1474    /// lease.publish("http://127.0.0.1:1").unwrap();  // stops the touching, under the same lock
1475    /// assert_eq!(held().unwrap().url.as_deref(), Some("http://127.0.0.1:1"));
1476    /// assert!(matches!(LeaseFile::read(lease.path()), LeaseFile::Document(d) if d.url.is_some()));
1477    /// lease.release();
1478    /// assert_eq!(LeaseFile::read(&dir.path().join(".ai/local/state/mcp/server.json")), LeaseFile::Absent);
1479    /// ```
1480    pub fn keep_alive(&self) {
1481        let path = self.path.clone();
1482        let token = self.token.clone();
1483        let binding = Arc::clone(&self.binding);
1484        let document = serde_json::to_string(&self.document(None)).unwrap_or_default();
1485        let tick = timings().bind_grace / 3;
1486        let _ = std::thread::Builder::new()
1487            .name("majordomus-lease-keep-alive".into())
1488            .spawn(move || loop {
1489                std::thread::sleep(tick);
1490                let Ok(guard) = binding.lock() else { return };
1491                if !*guard {
1492                    return;
1493                }
1494                if !holds(&path, &token) {
1495                    return;
1496                }
1497                // the same discipline as `publish`: a file of its own, synced, renamed only
1498                // while the lease is still this process's, under the lock a take-over takes;
1499                // a lock held by somebody else skips this touch rather than wait on it
1500                let tmp = tmp_path(&path, &token);
1501                if write_synced(&tmp, &document).is_err() {
1502                    let _ = fs::remove_file(&tmp);
1503                    continue;
1504                }
1505                let renamed = lock_for_change(&path, Duration::from_millis(200))
1506                    .filter(|_| holds(&path, &token))
1507                    .is_some_and(|_lock| fs::rename(&tmp, &path).is_ok());
1508                if !renamed {
1509                    let _ = fs::remove_file(&tmp);
1510                }
1511            });
1512    }

1541    /// [`Lease::publish`], with a seam between the check and the rename, where the unit
1542    /// tests put an elector taking the lease over.
1543    fn publish_with(&self, url: &str, between: impl FnOnce()) -> Result<()> {
1544        let mut binding = self.binding.lock().map_err(|_| Error::Lease {
1545            reason: "the lease's binding lock is poisoned".into(),
1546        })?;
1547        *binding = false;
1548        let document = self.document(Some(url));
1549        let text = serde_json::to_string(&document).map_err(|e| Error::Lease {
1550            reason: format!("cannot render the lease: {e}"),
1551        })?;
1552        let tmp = tmp_path(&self.path, &self.token);
1553        let renamed = write_synced(&tmp, &text)
1554            .map_err(|e| Error::io(&tmp, e))
1555            .and_then(|()| self.replace_with(&tmp, between));
1556        if renamed.is_err() {
1557            let _ = fs::remove_file(&tmp);
1558        }
1559        renamed?;
1560        sync_dir(&self.path);
1561        if let Ok(mut slot) = published().lock() {
1562            *slot = Some(document);
1563        }
1564        Ok(())
1565    }

1567    /// Rename `tmp` over the lease, under the change lock, only while the lease is still this
1568    /// process's.
1569    fn replace_with(&self, tmp: &Path, between: impl FnOnce()) -> Result<()> {
1570        let Some(_lock) = lock_for_change(&self.path, LOCK_PATIENCE) else {
1571            return Err(Error::Lease {
1572                reason: format!(
1573                    "the lock beside the lease at {} stayed held for {} seconds; not publishing over a lease another process is changing",
1574                    self.path.display(),
1575                    LOCK_PATIENCE.as_secs()
1576                ),
1577            });
1578        };
1579        if !self.is_mine() {
1580            return Err(Error::Lease {
1581                reason: format!(
1582                    "the lease at {} was taken over while this server was starting; another process serves this checkout",
1583                    self.path.display()
1584                ),
1585            });
1586        }
1587        between();
1588        fs::rename(tmp, &self.path).map_err(|e| Error::io(&self.path, e))
1589    }

1596    /// Remove the lease: the server has stopped. Only while the file is still this process's,
1597    /// checked under the lock a take-over takes, so that a lease taken over since is never
1598    /// removed by the process it was taken from.
1599    pub fn release(mut self) {
1600        self.release_now();
1601    }
1602
1603    fn release_now(&mut self) {
1604        if let Ok(mut binding) = self.binding.lock() {
1605            *binding = false;
1606        }
1607        if !self.released {
1608            remove_if_held(&self.path, &self.token);
1609        }
1610        self.released = true;
1611        if let Ok(mut slot) = published().lock() {
1612            if slot.as_ref().is_some_and(|d| d.token == self.token) {
1613                *slot = None;
1614            }
1615        }
1616        forget(Some(&self.token));
1617    }
1618}

1621    fn drop(&mut self) {
1622        self.release_now();
1623    }

1626/// `SIGTERM`, `SIGINT` and `SIGHUP` — `serve stop`, Ctrl-C, a closing terminal — ask a
1627/// process that holds a lease to stop. The handler records the request and nothing else.
1628///
1629/// It used to unlink the lease's path and die of the signal, whenever the process had held a
1630/// lease. That was async-signal-safe and wrong: a lease taken over a moment earlier is the
1631/// successor's, and the old owner's reader notices the takeover only at its next tick, so a
1632/// signal in that window removed the *new* owner's lease (I2128). Checking the token first in
1633/// the handler would only shrink the window, not close it — a check and an unlink are two
1634/// system calls and a takeover fits between them — and reading and parsing a file is not
1635/// work a handler may do. So the handler counts, and normal code acts:
1636///
1637/// * the server's loop (`serve`) polls [`stop_requested`] and stops the server in order —
1638///   episodes closed, listeners closed, the lease released through the check under the
1639///   lock, the mesh drained — and the process returns from `main`;
1640/// * a server whose loop does not poll (`majordomus mcp`, whose main thread is its client's
1641///   stdio session) is stopped in the same order by its own reader thread, which then ends
1642///   the process with the signal it was sent;
1643/// * a watcher thread ends the process when nobody answers the request, when a second signal
1644///   arrives, or when the ordered stop takes longer than [`STOP_BOUND`] — releasing the
1645///   lease through the same check first. A third signal is the kernel's: the handler restores
1646///   the default disposition on the second.
1647///
1648/// A `kill -9` cannot be caught: the next process finds the stale lease and takes it over.
1649#[cfg(unix)]
1650mod signals {
1651    use std::sync::atomic::{AtomicBool, AtomicI32, AtomicUsize, Ordering};
1652    use std::sync::Once;
1653    use std::time::{Duration, Instant};
1654
1655    static RECEIVED: AtomicUsize = AtomicUsize::new(0);
1656    static SIGNAL: AtomicI32 = AtomicI32::new(0);
1657    static ANSWERED: AtomicBool = AtomicBool::new(false);
1658    static INSTALL: Once = Once::new();
1659
1660    /// Install the handlers and start the watcher, once per process.
1661    pub fn install() {
1662        INSTALL.call_once(|| {
1663            let handler = on_signal as extern "C" fn(libc::c_int) as libc::sighandler_t;
1664            for signal in [libc::SIGTERM, libc::SIGINT, libc::SIGHUP] {
1665                // SAFETY: installing a handler that only touches lock-free atomics and calls
1666                // signal(2), both async-signal-safe
1667                unsafe { libc::signal(signal, handler) };
1668            }
1669            let _ = std::thread::Builder::new()
1670                .name("majordomus-signal-watcher".into())
1671                .spawn(watch);
1672        });
1673    }
1674
1675    pub fn requested() -> Option<i32> {
1676        (RECEIVED.load(Ordering::SeqCst) > 0).then(|| SIGNAL.load(Ordering::SeqCst))
1677    }
1678
1679    pub fn answer() {
1680        ANSWERED.store(true, Ordering::SeqCst);
1681    }
1682
1683    fn watch() {
1684        let mut first: Option<Instant> = None;
1685        loop {
1686            std::thread::sleep(Duration::from_millis(50));
1687            let received = RECEIVED.load(Ordering::SeqCst);
1688            if received == 0 {
1689                continue;
1690            }
1691            let since = first.get_or_insert_with(Instant::now).elapsed();
1692            let answered = ANSWERED.load(Ordering::SeqCst);
1693            if super::watch_verdict(received, answered, since) == super::Watch::End {
1694                if received == 1 && answered {
1695                    tracing::warn!(
1696                        "the ordered stop did not finish within {} seconds; ending the process",
1697                        super::STOP_BOUND.as_secs()
1698                    );
1699                }
1700                super::end_by_signal(SIGNAL.load(Ordering::SeqCst));
1701            }
1702        }
1703    }
1704
1705    extern "C" fn on_signal(signal: libc::c_int) {
1706        SIGNAL.store(signal, Ordering::SeqCst);
1707        if RECEIVED.fetch_add(1, Ordering::SeqCst) >= 1 {
1708            // SAFETY: restoring the default disposition is async-signal-safe; the next one
1709            // of this signal ends the process whatever the watcher is doing
1710            unsafe { libc::signal(signal, libc::SIG_DFL) };
1711        }
1712    }
1713}
