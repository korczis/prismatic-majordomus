29/// Where a module's entry in the registry came from.
30#[derive(

33#[serde(rename_all = "snake_case")]
34pub enum ModuleSource {
35    /// Declared with `module!` and composed with `compose_modules!`.
36    Builtin,
37    /// Derived from the namespace of executables composed without a module descriptor.
38    Derived,
39    /// One kind of declarative objects of the repository's layer.

43/// One module as the registry knows it.
44#[derive(Debug, Clone, PartialEq, Eq, Serialize, serde::Deserialize, schemars::JsonSchema)]
45pub struct ModuleInfo {
46    /// The identity.
47    pub id: ModuleId,
48    /// The short name.
49    pub title: String,
50    /// One paragraph.
51    pub description: String,
52    /// Where the module stands, when it declared it.
53    #[serde(skip_serializing_if = "Option::is_none")]
54    pub stability: Option<Stability>,
55    /// Where the entry came from.
56    pub source: ModuleSource,
57    /// How many capabilities it composes.
58    pub capabilities: usize,
59}

70/// Why the registry could not be built. Every variant names the provenance of every party.
71#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error, Serialize)]
72#[serde(tag = "code", rename_all = "snake_case")]
73pub enum RegistryError {
74    #[error("capability '{id}' ({provenance}): invalid id: {reason}")]
75    /// The id does not satisfy the grammar.
76    InvalidId {
77        /// The id as written.
78        id: String,
79        /// Where the descriptor came from.
80        provenance: String,
81        /// What the grammar objects to.
82        reason: String,
83    },
84    #[error("capability '{id}' is defined twice: {first} and {second}")]
85    /// Two descriptors claim one id.
86    DuplicateId {
87        /// The id.
88        id: String,
89        /// The provenance of the descriptor seen first, in id order.
90        first: String,
91        /// The provenance of the other.
92        second: String,
93    },
94    #[error("MCP tool '{name}' is claimed by '{first}' and '{second}'")]
95    /// Two capabilities claim one MCP tool name.
96    DuplicateMcpName {
97        /// The tool name.
98        name: String,
99        /// The id of the capability seen first.
100        first: String,
101        /// The id of the other.
102        second: String,
103    },
104    #[error("MCP resource '{uri}' is claimed by '{first}' and '{second}'")]
105    /// Two capabilities claim one MCP resource URI.
106    DuplicateMcpUri {
107        /// The URI.
108        uri: String,
109        /// The id of the capability seen first.
110        first: String,
111        /// The id of the other.
112        second: String,
113    },
114    #[error("HTTP route {method} {path} is claimed by '{first}' and '{second}'")]
115    /// Two capabilities claim one HTTP method and path.
116    DuplicateHttpRoute {
117        /// The method.
118        method: String,
119        /// The path.
120        path: String,
121        /// The id of the capability seen first.
122        first: String,
123        /// The id of the other.
124        second: String,
125    },
126    #[error("CLI path '{path}' is claimed by '{first}' and '{second}'")]
127    /// Two capabilities claim one CLI path.
128    DuplicateCliPath {
129        /// The words, joined with spaces.
130        path: String,
131        /// The id of the capability seen first.
132        first: String,
133        /// The id of the other.
134        second: String,
135    },
136    #[error("capability '{id}' ({provenance}): invalid {projection} exposure: {reason}")]
137    /// A declared exposure cannot be served by its projection.
138    InvalidExposure {
139        /// The id.
140        id: String,
141        /// Where the descriptor came from.
142        provenance: String,
143        /// `MCP`, `HTTP` or `CLI`.
144        projection: String,
145        /// What is wrong with the exposure.
146        reason: String,
147    },
148    #[error("capability '{id}' ({provenance}) is {stability} and cannot be exposed as executable through {projection}")]
149    /// A planned or unsupported capability declares an executable exposure.
150    NotExecutable {
151        /// The id.
152        id: String,
153        /// Where the descriptor came from.
154        provenance: String,
155        /// The stability that forbids execution.
156        stability: String,
157        /// The projection the exposure is for.
158        projection: String,
159    },
160    #[error("module '{id}': invalid id: {reason}")]
161    /// A module id does not satisfy the grammar.
162    InvalidModuleId {
163        /// The id as written.
164        id: String,
165        /// What the grammar objects to.
166        reason: String,
167    },
168    #[error("module '{id}' is composed twice ({first} and {second})")]
169    /// Two modules claim one id, or a builtin module's id is a declarative kind.
170    DuplicateModule {
171        /// The id.
172        id: String,
173        /// The source seen first.
174        first: String,
175        /// The other.
176        second: String,
177    },
178    #[error("capability '{id}' ({provenance}) is composed in module '{module}' but its namespace is '{namespace}'")]
179    /// A capability's id does not belong to the module that composes it.
180    ModuleMismatch {
181        /// The capability id.
182        id: String,
183        /// Where the descriptor came from.
184        provenance: String,
185        /// The module that composed it.
186        module: String,
187        /// The id's namespace.
188        namespace: String,
189    },
190    #[error("capability '{id}' ({provenance}): invalid cache policy: {reason}")]
191    /// The cache policy keeps nothing or contradicts itself.
192    InvalidCachePolicy {
193        /// The id.
194        id: String,
195        /// Where the descriptor came from.
196        provenance: String,
197        /// What is wrong.
198        reason: String,
199    },
200    #[error("capability '{id}' ({provenance}): {reason}")]
201    /// The descriptor's shape contradicts its kind: a query without a handler, a resource with one or with a callable exposure.
202    Shape {
203        /// The id.
204        id: String,
205        /// Where the descriptor came from.
206        provenance: String,
207        /// What is contradictory.
208        reason: String,
209    },
210}

213/// Every capability, once, validated, in id order, with the lookups projections need.
214pub struct CapabilityRegistry {

216    modules: BTreeMap<ModuleId, ModuleInfo>,

218    by_mcp_tool: BTreeMap<String, CapabilityId>,
219    by_mcp_uri: BTreeMap<String, CapabilityId>,

224/// The one composition point.
225#[derive(Default)]
226pub struct Builder {
227    pending: Vec<Entry>,
228    modules: Vec<ModuleInfo>,
229    index_fingerprint: String,
230}

232impl Builder {
233    /// Add the application's modules, as `compose_modules!` produced them: their metadata
234    /// and every executable they compose.
235    pub fn with_modules(mut self, modules: Vec<ModuleDescriptor>) -> Self {
236        for m in modules {
237            self.modules.push(ModuleInfo {
238                id: m.id.clone(),
239                title: m.title,
240                description: m.description,
241                stability: Some(m.stability),
242                source: ModuleSource::Builtin,
243                capabilities: m.capabilities.len(),
244            });
245            self = self.with_builtin(m.capabilities);
246        }
247        self
248    }
249
250    /// Add executables without a module descriptor; each one's module is the namespace of
251    /// its id.
252    pub fn with_builtin(mut self, executables: Vec<Executable>) -> Self {
253        for e in executables {
254            self.pending.push(Entry {
255                capability: e.capability,
256                handler: Some(e.handler),
257                cases: Some(e.cases),
258            });
259        }
260        self
261    }
262
263    /// Every object of the index becomes a resource capability; the index's fingerprint
264    /// joins the registry's.
265    pub fn with_index(mut self, index: &Index) -> Self {
266        self.index_fingerprint = index.fingerprint.clone();
267        for object in &index.objects {
268            self.pending.push(Entry {
269                capability: super::declarative::capability_of(object),
270                handler: None,
271                cases: None,
272            });
273        }
274        self
275    }

277    /// Validate every invariant and build. Errors are collected, not stopped at the first,
278    /// and reported in a deterministic order.
279    pub fn build(self) -> Result<CapabilityRegistry, Vec<RegistryError>> {
280        let _phase = crate::perf::phase(crate::perf::Phase::RegistryBuild);
281        crate::perf::Counters::bump(&crate::perf::COUNTERS.registry_builds);
282        let mut errors = Vec::new();
283        let mut registry = CapabilityRegistry::default();
284        for m in self.modules {
285            let id = m.id.as_str().to_string();
286            if let Err(reason) = ModuleId::parse(&id) {
287                errors.push(RegistryError::InvalidModuleId { id, reason });
288                continue;
289            }
290            if let Some(first) = registry.modules.get(&m.id) {
291                errors.push(RegistryError::DuplicateModule {
292                    id,
293                    first: format!("{:?}", first.source).to_lowercase(),
294                    second: "builtin".into(),
295                });
296                continue;
297            }
298            registry.modules.insert(m.id.clone(), m);
299        }
300        let mut pending = self.pending;
301        pending.sort_by(|a, b| {
302            a.capability.id.cmp(&b.capability.id).then_with(|| {
303                a.capability
304                    .provenance
305                    .to_string()
306                    .cmp(&b.capability.provenance.to_string())
307            })
308        });
309        for mut entry in pending {
310            let c = &entry.capability;
311            let id = c.id.as_str().to_string();
312            let prov = c.provenance.to_string();
313            if let Err(reason) = CapabilityId::parse(&id) {
314                errors.push(RegistryError::InvalidId {
315                    id: id.clone(),
316                    provenance: prov.clone(),
317                    reason,
318                });
319                continue;
320            }
321            if let Some(first) = registry.entries.get(&c.id) {
322                errors.push(RegistryError::DuplicateId {
323                    id: id.clone(),
324                    first: first.capability.provenance.to_string(),
325                    second: prov.clone(),
326                });
327                continue;
328            }
329            // the module: stamped by `module!`, derived from the namespace otherwise
330            let namespace = c.id.namespace().to_string();
331            if entry.capability.module.as_str().is_empty() {
332                entry.capability.module = ModuleId::unchecked(&namespace);
333            }
334            let c = &entry.capability;
335            let is_builtin = matches!(c.provenance, Provenance::Builtin { .. });
336            if is_builtin && c.module.as_str() != namespace {
337                errors.push(RegistryError::ModuleMismatch {
338                    id: id.clone(),
339                    provenance: prov.clone(),
340                    module: c.module.as_str().to_string(),
341                    namespace,
342                });
343                continue;
344            }
345            match registry.modules.get_mut(&c.module) {
346                Some(m) if m.source == ModuleSource::Builtin && is_builtin => {}
347                Some(m) if m.source == ModuleSource::Builtin => {
348                    errors.push(RegistryError::DuplicateModule {
349                        id: c.module.as_str().to_string(),
350                        first: "builtin".into(),
351                        second: format!("declarative kind of {prov}"),
352                    });
353                    continue;
354                }
355                Some(m) => m.capabilities += 1,
356                None => {
357                    if let Err(reason) = ModuleId::parse(c.module.as_str()) {
358                        errors.push(RegistryError::InvalidModuleId {
359                            id: c.module.as_str().to_string(),
360                            reason,
361                        });
362                        continue;
363                    }
364                    let (source, title, description) = if is_builtin {
365                        (
366                            ModuleSource::Derived,
367                            c.module.as_str().to_string(),
368                            format!("Executables in the '{}' namespace, composed without a module descriptor.", c.module),
369                        )
370                    } else {
371                        (
372                            ModuleSource::Declarative,
373                            c.module.as_str().to_string(),
374                            format!(
375                                "Declarative objects of kind '{}' from the repository's AI layer.",
376                                c.module
377                            ),
378                        )
379                    };
380                    registry.modules.insert(
381                        c.module.clone(),
382                        ModuleInfo {
383                            id: c.module.clone(),
384                            title,
385                            description,
386                            stability: None,
387                            source,
388                            capabilities: 1,
389                        },
390                    );
391                }
392            }
393            if let Err(reason) = c.cache.validate() {
394                errors.push(RegistryError::InvalidCachePolicy {
395                    id: id.clone(),
396                    provenance: prov.clone(),
397                    reason,
398                });
399                continue;
400            }
401            // classification is not a value a descriptor may hold an opinion about: it
402            // follows from the kind and the transports it declares, and a capability
403            // arriving from outside this crate with a different one is refused rather
404            // than believed. A projection reads these fields; if they could drift, every
405            // surface downstream would be reading a claim instead of a fact.
406            let availability = Availability::classify(c.kind, &c.exposure);
407            let visibility = Visibility::classify(&c.exposure);
408            if c.availability != availability || c.visibility != visibility {
409                errors.push(RegistryError::Shape {
410                    id: id.clone(),
411                    provenance: prov.clone(),
412                    reason: format!(
413                        "declares availability {:?} and visibility {:?}; what it exposes makes it {availability:?} and {visibility:?}. Classification follows the kind and the transports, so change those or let the declaration classify itself",
414                        c.availability, c.visibility
415                    ),
416                });
417                continue;
418            }
419            match (c.kind, c.benchmark) {
420                (
421                    CapabilityKind::Resource,
422                    BenchmarkPolicy::Waived {
423                        reason: WaiverReason::NotExecutable,
424                    },
425                ) => {}
426                (CapabilityKind::Resource, _) => {
427                    errors.push(RegistryError::Shape {
428                        id: id.clone(),
429                        provenance: prov.clone(),
430                        reason: "a resource is not a benchmark target: its policy is waived as not executable".into(),
431                    });
432                    continue;
433                }
434                (
435                    _,
436                    BenchmarkPolicy::Waived {
437                        reason: WaiverReason::NotExecutable,
438                    },
439                ) => {
440                    errors.push(RegistryError::Shape {
441                        id: id.clone(),
442                        provenance: prov.clone(),
443                        reason:
444                            "an executable cannot be waived as not executable; name a real reason"
445                                .into(),
446                    });
447                    continue;
448                }
449                _ => {}
450            }
451            if c.cache.is_enabled() && c.kind == CapabilityKind::Command {
452                errors.push(RegistryError::InvalidCachePolicy {
453                    id: id.clone(),
454                    provenance: prov.clone(),
455                    reason: "a command changes state and is never cached".into(),
456                });
457                continue;
458            }
459            // classification is the registry's, not the declaration's, for the same reason
460            // availability and visibility are: a descriptor arriving from outside this
461            // crate with a policy of its own would be a claim a projection then reads as a
462            // fact
463            let policy = ExecutionPolicy::classify(c.kind);
464            // The documented additions, and their combination: cancellation, which any kind
465            // may declare because any handler may look at its flag, and writing the
466            // repository, which only a command may — a query that wrote tracked files would
467            // be a query in name only, bound to GET and announced to MCP as read-only.
468            // Everything else about the policy still follows the kind.
469            let mut permitted = vec![policy, policy.stoppable()];
470            if c.kind == CapabilityKind::Command {
471                permitted.push(policy.writes_repository());
472                permitted.push(policy.stoppable().writes_repository());
473            }
474            if !permitted.contains(&c.execution) {
475                errors.push(RegistryError::Shape {
476                    id: id.clone(),
477                    provenance: prov.clone(),
478                    reason: format!(
479                        "declares the execution policy {:?}; its kind makes it {policy:?}. A declaration may add cancellation (`.cancellable()`), and a command may add writing the repository (`.writes_repository()`); the effect of a query and the concurrency of anything follow the kind",
480                        c.execution
481                    ),
482                });
483                continue;
484            }
485            match (c.kind, entry.handler.is_some()) {
486                (CapabilityKind::Query, false) => {
487                    errors.push(RegistryError::Shape {
488                        id: id.clone(),
489                        provenance: prov.clone(),
490                        reason: "a query needs a handler".into(),
491                    });
492                    continue;
493                }
494                (CapabilityKind::Command, false) => {
495                    errors.push(RegistryError::Shape {
496                        id: id.clone(),
497                        provenance: prov.clone(),
498                        reason: "a command needs a handler".into(),
499                    });
500                    continue;
501                }
502                (CapabilityKind::Resource, true) => {
503                    errors.push(RegistryError::Shape {
504                        id: id.clone(),
505                        provenance: prov.clone(),
506                        reason: "a resource has no handler".into(),
507                    });
508                    continue;
509                }
510                _ => {}
511            }
512            if let Some(mcp) = &c.exposure.mcp {
513                if let Some(tool) = &mcp.tool {
514                    if !c.stability.executable() {
515                        errors.push(not_executable(c, "MCP tool"));
516                    }
517                    if tool.is_empty()
518                        || !tool
519                            .bytes()
520                            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
521                    {
522                        errors.push(RegistryError::InvalidExposure {
523                            id: id.clone(),
524                            provenance: prov.clone(),
525                            projection: "MCP".into(),
526                            reason: format!("tool name '{tool}' is not [a-z0-9_]+"),
527                        });
528                    } else if let Some(first) = registry.by_mcp_tool.get(tool) {
529                        errors.push(RegistryError::DuplicateMcpName {
530                            name: tool.clone(),
531                            first: first.to_string(),
532                            second: id.clone(),
533                        });
534                    } else {
535                        registry.by_mcp_tool.insert(tool.clone(), c.id.clone());
536                    }
537                }
538                if let Some(res) = &mcp.resource {
539                    if !res.uri.starts_with("majordomus://")
540                        || res.uri.contains(char::is_whitespace)
541                    {
542                        errors.push(RegistryError::InvalidExposure {
543                            id: id.clone(),
544                            provenance: prov.clone(),
545                            projection: "MCP".into(),
546                            reason: format!(
547                                "resource uri '{}' is not a majordomus:// uri",
548                                res.uri
549                            ),
550                        });
551                    } else if let Some(first) = registry.by_mcp_uri.get(&res.uri) {
552                        errors.push(RegistryError::DuplicateMcpUri {
553                            uri: res.uri.clone(),
554                            first: first.to_string(),
555                            second: id.clone(),
556                        });
557                    } else {
558                        registry.by_mcp_uri.insert(res.uri.clone(), c.id.clone());
559                    }
560                    // a read supplies no input: a query read as a resource must accept `{}`
561                    let (_, required) = c.input.properties();
562                    if c.kind.is_executable() && !required.is_empty() {
563                        errors.push(RegistryError::Shape {
564                            id: id.clone(),
565                            provenance: prov.clone(),
566                            reason: format!(
567                                "a query read as the resource '{}' is called with no input, and its input requires {}",
568                                res.uri,
569                                required.join(", ")
570                            ),
571                        });
572                    }
573                }
574            }
575            if let Some(http) = &c.exposure.http {
576                if !c.stability.executable() {
577                    errors.push(not_executable(c, "HTTP"));
578                }
579                if let Err(reason) = http.validate() {
580                    errors.push(RegistryError::InvalidExposure {
581                        id: id.clone(),
582                        provenance: prov.clone(),
583                        projection: "HTTP".into(),
584                        reason,
585                    });
586                } else if let Some(first) = registry.by_http.get(&(http.method, http.path.clone()))
587                {
588                    errors.push(RegistryError::DuplicateHttpRoute {
589                        method: http.method.as_str().into(),
590                        path: http.path.clone(),
591                        first: first.to_string(),
592                        second: id.clone(),
593                    });
594                } else {
595                    registry
596                        .by_http
597                        .insert((http.method, http.path.clone()), c.id.clone());
598                }
599            }
600            if let Some(cli) = &c.exposure.cli {
601                if !c.stability.executable() {
602                    errors.push(not_executable(c, "CLI"));
603                }
604                if cli.path.is_empty()
605                    || cli
606                        .path
607                        .iter()
608                        .any(|w| w.is_empty() || w.contains(char::is_whitespace))
609                {
610                    errors.push(RegistryError::InvalidExposure {
611                        id: id.clone(),
612                        provenance: prov.clone(),
613                        projection: "CLI".into(),
614                        reason: format!("path {:?} has an empty or blank word", cli.path),
615                    });
616                } else if let Some(first) = registry.by_cli.get(&cli.path) {
617                    errors.push(RegistryError::DuplicateCliPath {
618                        path: cli.path.join(" "),
619                        first: first.to_string(),
620                        second: id.clone(),
621                    });
622                } else {
623                    registry.by_cli.insert(cli.path.clone(), c.id.clone());
624                }
625            }
626            if c.kind == CapabilityKind::Resource
627                && (c.exposure.http.is_some()
628                    || c.exposure.cli.is_some()
629                    || c.exposure.mcp.as_ref().is_some_and(|m| m.tool.is_some()))
630            {
631                errors.push(RegistryError::Shape {
632                    id: id.clone(),
633                    provenance: prov.clone(),
634                    reason: "a resource is read, not called: only an MCP resource exposure applies"
635                        .into(),
636                });
637            }
638            if c.kind == CapabilityKind::Command {
639                let word = c.kind.as_str();
640                if c.exposure
641                    .mcp
642                    .as_ref()
643                    .is_some_and(|m| m.resource.is_some())
644                {
645                    errors.push(RegistryError::Shape {
646                        id: id.clone(),
647                        provenance: prov.clone(),
648                        reason: format!(
649                            "a {word} is called, not read: it has no MCP resource exposure"
650                        ),
651                    });
652                }
653                if let Some(http) = &c.exposure.http {
654                    if http.method != HttpMethod::Post {
655                        errors.push(RegistryError::InvalidExposure {
656                            id: id.clone(),
657                            provenance: prov.clone(),
658                            projection: "HTTP".into(),
659                            reason: format!(
660                                "a {word} changes state and is bound to POST, not {}",
661                                http.method.as_str()
662                            ),
663                        });
664                    }
665                }
666            }
667            registry.entries.insert(c.id.clone(), entry);
668        }
669        if errors.is_empty() {
670            registry.fingerprint = fingerprint(&self.index_fingerprint, &registry);
671            Ok(registry)
672        } else {
673            Err(errors)
674        }
675    }
676}

678/// The registry's fingerprint: a hash of the index's fingerprint (every declarative
679/// object's path and content), of every id, and of every builtin descriptor in full, in
680/// id order. Stable across processes for the same code and the same repository state.
681fn fingerprint(index: &str, registry: &CapabilityRegistry) -> String {
682    use sha2::{Digest, Sha256};
683    let mut h = Sha256::new();
684    h.update(index.as_bytes());
685    for c in registry.iter() {
686        h.update(c.id.as_str().as_bytes());
687        h.update(b"\0");
688        // a declarative descriptor is a function of the object the index fingerprint
689        // already covers; hashing its (large, shared) schema per object would only cost
690        if matches!(c.provenance, Provenance::Builtin { .. }) {
691            h.update(serde_json::to_string(c).unwrap_or_default().as_bytes());
692        }
693        h.update(b"\n");
694    }
695    format!("{:x}", h.finalize())
696}
697
698fn not_executable(c: &Capability, projection: &str) -> RegistryError {
699    RegistryError::NotExecutable {
700        id: c.id.to_string(),
701        provenance: c.provenance.to_string(),
702        stability: serde_json::to_value(c.stability)
703            .ok()
704            .and_then(|v| v.as_str().map(str::to_string))
705            .unwrap_or_default(),
706        projection: projection.into(),
707    }
708}

710impl CapabilityRegistry {
711    /// Start composing a registry.
712    ///
713    /// ```
714    /// use majordomus_cli::capability::{builtin, CapabilityRegistry, RegistryError};
715    /// let registry = CapabilityRegistry::builder().with_builtin(builtin::all()).build().unwrap();
716    /// assert!(registry.get("repository.info").is_some());
717    /// assert_eq!(registry.by_mcp_tool("majordomus_get").unwrap().id.as_str(), "objects.get");
718    /// // the same executables twice: every id is claimed twice, and the registry says so
719    /// let twice = CapabilityRegistry::builder().with_builtin(builtin::all()).with_builtin(builtin::all()).build();
720    /// let errors = twice.unwrap_err();
721    /// assert!(errors.iter().all(|e| matches!(e, RegistryError::DuplicateId { .. })));
722    /// ```
723    pub fn builder() -> Builder {
724        Builder::default()
725    }
726
727    /// Every capability, by id.
728    pub fn iter(&self) -> impl Iterator<Item = &Capability> {
729        self.entries.values().map(|e| &e.capability)
730    }
731
732    /// How many capabilities the registry holds.
733    pub fn len(&self) -> usize {
734        self.entries.len()
735    }
736
737    /// Does the registry hold nothing?
738    pub fn is_empty(&self) -> bool {
739        self.entries.is_empty()
740    }
741
742    /// Every module, by id: the composed ones, the derived ones, and one per declarative kind.
743    pub fn modules(&self) -> impl Iterator<Item = &ModuleInfo> {
744        self.modules.values()
745    }
746
747    /// One module by id.
748    pub fn module(&self, id: &str) -> Option<&ModuleInfo> {
749        self.modules.get(&ModuleId::unchecked(id))
750    }
751
752    /// The benchmark case provider of an executable; `None` for a resource or an unknown id.
753    pub fn cases(&self, id: &str) -> Option<CaseProvider> {
754        self.entries
755            .get(&CapabilityId::unchecked(id))
756            .and_then(|e| e.cases)
757    }

759    /// A capability by canonical id.
760    pub fn get(&self, id: &str) -> Option<&Capability> {
761        self.entries
762            .get(&CapabilityId::unchecked(id))
763            .map(|e| &e.capability)
764    }

766    /// The capability an MCP tool name projects.
767    pub fn by_mcp_tool(&self, name: &str) -> Option<&Capability> {
768        self.by_mcp_tool
769            .get(name)
770            .and_then(|id| self.get(id.as_str()))
771    }

773    /// The capability an MCP resource URI projects, with the resource exposure it matched:
774    /// a declarative object, or a query read as a document (`majordomus://repository`).
775    ///
776    /// ```
777    /// use majordomus_cli::capability::{builtin, CapabilityKind, CapabilityRegistry};
778    /// let r = CapabilityRegistry::builder().with_builtin(builtin::all()).build().unwrap();
779    /// let (c, res) = r.by_mcp_uri(builtin::REPOSITORY_URI).unwrap();
780    /// assert_eq!((c.id.as_str(), c.kind, res.name.as_str()), ("repository.info", CapabilityKind::Query, "repository"));
781    /// assert!(r.by_mcp_uri("majordomus://rule/none@1").is_none());
782    /// ```
783    pub fn by_mcp_uri(&self, uri: &str) -> Option<(&Capability, &McpResource)> {
784        let c = self
785            .by_mcp_uri
786            .get(uri)
787            .and_then(|id| self.get(id.as_str()))?;
788        let res = c.exposure.mcp.as_ref()?.resource.as_ref()?;
789        Some((c, res))
790    }

792    /// The capability an HTTP method and path project.
793    pub fn by_http(&self, method: HttpMethod, path: &str) -> Option<&Capability> {
794        self.by_http
795            .get(&(method, path.to_string()))
796            .and_then(|id| self.get(id.as_str()))
797    }

799    /// The capability a CLI path projects.
800    pub fn by_cli(&self, path: &[String]) -> Option<&Capability> {
801        self.by_cli.get(path).and_then(|id| self.get(id.as_str()))
802    }

804    /// The fingerprint of this registry and the repository state it was built from.
805    pub fn fingerprint(&self) -> &str {
806        &self.fingerprint
807    }

809    /// Run the handler of an executable by id, with no counters and no cache: the raw
810    /// dispatch the executor wraps. Everything else calls [`Context::execute`]. A
811    /// resource, or an unknown id, is `NotFound`.
812    pub fn dispatch(
813        &self,
814        ctx: &Context,
815        id: &str,
816        input: Value,
817    ) -> Result<Value, CapabilityError> {
818        let entry = self
819            .entries
820            .get(&CapabilityId::unchecked(id))
821            .ok_or_else(|| CapabilityError::NotFound(format!("capability {id}")))?;
822        let handler = entry.handler.as_ref().ok_or_else(|| {
823            CapabilityError::NotFound(format!("capability {id} is a resource, not a query"))
824        })?;
825        tracing::debug!(capability_id = id, "call");
826        handler.call(ctx, input)
827    }
828
829    /// Counts by kind, by stability and by projection, for introspection.
830    pub fn summary(&self) -> Summary {
831        let mut s = Summary {
832            modules: self.modules.len(),
833            ..Default::default()
834        };
835        for c in self.iter() {
836            *s.by_kind
837                .entry(format!("{:?}", c.kind).to_lowercase())
838                .or_default() += 1;
839            let st = serde_json::to_value(c.stability)
840                .ok()
841                .and_then(|v| v.as_str().map(str::to_string))
842                .unwrap_or_default();
843            *s.by_stability.entry(st).or_default() += 1;
844            if c.exposure.mcp.as_ref().is_some_and(|m| m.tool.is_some()) {
845                s.mcp_tools += 1;
846            }
847            if c.exposure
848                .mcp
849                .as_ref()
850                .is_some_and(|m| m.resource.is_some())
851            {
852                s.mcp_resources += 1;
853            }
854            if c.exposure.http.is_some() {
855                s.http_routes += 1;
856            }
857            if c.exposure.cli.is_some() {
858                s.cli_commands += 1;
859            }
860            match c.benchmark {
861                BenchmarkPolicy::Required | BenchmarkPolicy::RequiredWhen { .. } => {
862                    s.benchmark_required += 1
863                }
864                BenchmarkPolicy::Waived {
865                    reason: WaiverReason::NotExecutable,
866                } => {}
867                BenchmarkPolicy::Waived { .. } => s.benchmark_waived += 1,
868            }
869            if c.cache.is_enabled() {
870                s.cached += 1;
871            }
872            match c.provenance {
873                Provenance::Builtin { .. } => s.builtin += 1,
874                Provenance::Declarative { .. } => s.declarative += 1,
875            }
876        }
877        s.total = self.len();
878        s
879    }
880}
