1//! The state of every declared gate, as this checkout's record holds it: which gates the CI
2//! model declares, what the last recorded run of each one said, over which tree and at which
3//! head — and, said as plainly as a verdict, which gates no run has ever reported.
4//!
5//! # Why this is not the completion report
6//!
7//! `gates.completion` asks a narrower question — may *this task* be called finished — and so
8//! reads only the active task's own runs over the gates its change set selects. With no task
9//! open it attributes nothing, and every gate is `unknown`. A person asking "how did the
10//! gates last go?" is asking about the gates, not about whichever task is open, so this
11//! reads the newest run of each gate whatever task recorded it.
12//!
13//! # What it refuses to do
14//!
15//! It never renders a missing verdict as a passing one. A gate with no recorded run is
16//! [`GateStatus::Queued`] and is named in [`GateStateReport::never_reported`]; a run whose
17//! inputs have changed since is [`GateStatus::Stale`]; a run whose inputs cannot be hashed
18//! here is [`GateStatus::Unknown`]. Only a run over the files as they stand now, exiting 0,
19//! is `pass`. And it never reaches the network: what it did not consult — the forge's own
20//! CI runs — is listed in [`GateStateReport::unreached`] rather than silently absent.
21//!
22//! ```
23//! use majordomus_cli::gates::GateStatus;
24//!
25//! // the words this report answers in are the completion report's own
26//! assert_eq!(GateStatus::Queued.as_str(), "queued");
27//! assert_ne!(GateStatus::Queued, GateStatus::Pass, "never reported is not passed");
28//! ```

38/// What this capability says it did not consult, every time it answers: the CI runs the
39/// forge holds. Reading them needs a network and a token, and `gates.state` is the offline
40/// half; the sentence is part of the answer so that an absent forge verdict is never read as
41/// a passing one.
42pub(crate) const FORGE_UNREACHED: &str =
43    "the forge's CI runs (GitHub Actions) were not consulted: \
44     this answer is this checkout's own record, read offline, and a verdict that exists only \
45     on the forge is not in it";

48/// One declared gate and the standing of its last recorded run.
49///
50/// ```
51/// use majordomus_cli::gates::{GateState, GateStatus};
52///
53/// let never: GateState = serde_json::from_value(serde_json::json!({
54///     "id": "rust-check",
55///     "job": "rust",
56///     "always": true,
57///     "on_demand": false,
58///     "status": "queued",
59///     "reason": "no run of this gate has ever been recorded in this checkout",
60/// }))
61/// .unwrap();
62/// // no run, no head, and a word that is not `pass`
63/// assert!(never.last_run.is_none() && never.on_head.is_none());
64/// assert_eq!(never.status, GateStatus::Queued);
65/// ```
66pub struct GateState {
67    /// The gate's identity in the model.
68    pub id: String,
69    /// The CI job that runs it.
70    pub job: String,
71    /// Whether every plan selects it.
72    pub always: bool,
73    /// Whether its runner can be had only on demand, so no routine plan runs it.
74    pub on_demand: bool,
75    /// What is known about its last run: `pass`, `fail`, `stale`, `queued` (never
76    /// reported) or `unknown` (a run exists and cannot be checked against this tree).
77    pub status: GateStatus,
78    /// Why it is in that status, in words a person can act on.
79    pub reason: String,
80    /// The run that last reported, whichever task recorded it.
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    pub last_run: Option<GateRun>,
83    /// Whether that run was taken at the commit this checkout is at now. Absent when there
84    /// is no run or when the current commit could not be read.
85    #[serde(default, skip_serializing_if = "Option::is_none")]
86    pub on_head: Option<bool>,
87    /// The hash of the gate's inputs as they stand now, when it could be taken.
88    #[serde(default, skip_serializing_if = "String::is_empty")]
89    pub inputs_hash: String,
90}

93/// The declared gate set and the standing of each gate's last recorded run.
94///
95/// `count`, `tallies` and `never_reported` are measured from `gates`, never written down,
96/// and `unreached` always names what was not consulted.
97///
98/// ```
99/// use majordomus_cli::gates::GateStateReport;
100///
101/// let r: GateStateReport = serde_json::from_value(serde_json::json!({
102///     "source": ".ai/repo/ci/gates.yaml",
103///     "ledger": ".ai/local/state/ledger.jsonl",
104///     "count": 0,
105///     "tallies": {},
106///     "never_reported": [],
107///     "gates": [],
108///     "unreached": ["the forge's CI runs were not consulted"],
109///     "findings": [],
110///     "evaluated_at": "2026-10-15T00:00:00Z",
111/// }))
112/// .unwrap();
113/// assert_eq!(r.count, r.gates.len());
114/// assert!(r.head.is_none(), "a checkout without git still answers, and says so");
115/// assert!(!r.unreached.is_empty());
116/// ```
117pub struct GateStateReport {
118    /// Where the gate set was read from, repository-relative.
119    pub source: String,
120    /// Where the recorded runs were read from, repository-relative.
121    pub ledger: String,
122    /// The commit this checkout is at, when git could say.
123    #[serde(default, skip_serializing_if = "Option::is_none")]
124    pub head: Option<String>,
125    /// How many gates the model declares; measured.
126    pub count: usize,
127    /// How many gates stand in each status, by the status word.
128    pub tallies: BTreeMap<String, usize>,
129    /// The gates no run has ever reported, in declaration order.
130    pub never_reported: Vec<String>,
131    /// Every declared gate, in declaration order.
132    pub gates: Vec<GateState>,
133    /// What this answer did not consult, so its silence about it is not read as a verdict.
134    pub unreached: Vec<String>,
135    /// What could not be read, and what that means for the answer.
136    pub findings: Vec<String>,
137    /// When this answer was taken, RFC 3339 UTC.
138    pub evaluated_at: String,
139}

141/// The standing of one gate given its last run and the hash of its inputs now.
142fn standing_of(run: Option<&GateRun>, hash_now: Option<&str>) -> (GateStatus, String) {
143    let Some(r) = run else {
144        return (
145            GateStatus::Queued,
146            "no run of this gate has ever been recorded in this checkout: never reported, \
147             which is not a pass"
148                .to_string(),
149        );
150    };
151    let Some(now) = hash_now else {
152        return (
153            GateStatus::Unknown,
154            "the files that select this gate could not be hashed here, so its last run cannot \
155             be checked against this tree"
156                .to_string(),
157        );
158    };
159    if r.inputs_hash != now {
160        return (
161            GateStatus::Stale,
162            format!(
163                "the last run (exit {} at {}) was taken over inputs {} and this tree hashes to \
164                 {}; it no longer describes what it proved",
165                r.exit,
166                short(&r.head),
167                short(&r.inputs_hash),
168                short(now)
169            ),
170        );
171    }
172    if r.exit == 0 {
173        (
174            GateStatus::Pass,
175            format!("exit 0 at {} over inputs {}", short(&r.head), short(now)),
176        )
177    } else {
178        (
179            GateStatus::Fail,
180            format!(
181                "exit {} at {} over inputs {}",
182                r.exit,
183                short(&r.head),
184                short(now)
185            ),
186        )
187    }
188}

190/// Twelve characters of a hash or a commit, as every gate report here prints them.
191fn short(v: &str) -> String {
192    if v.is_empty() {
193        "(none)".to_string()
194    } else {
195        v.chars().take(12).collect()
196    }
197}

199/// Every gate of `model` with the standing of its last run in `runs`.
200///
201/// `hashes` holds, per gate, the hash of its inputs now (`None` when it could not be taken;
202/// the empty string for a gate with no inputs, as the ledger records it), and `head` the
203/// commit this checkout is at. Everything is passed in so that a test drives the same
204/// function the capability does.
205pub(crate) fn state(
206    model: &GateModel,
207    runs: &BTreeMap<String, GateRun>,
208    hashes: &BTreeMap<String, Option<String>>,
209    head: Option<&str>,
210) -> Vec<GateState> {
211    model
212        .gates
213        .iter()
214        .map(|g| {
215            let run = runs.get(&g.id);
216            let hash_now = hashes.get(&g.id).cloned().flatten();
217            let (status, reason) = standing_of(run, hash_now.as_deref());
218            let on_head = match (run, head) {
219                (Some(r), Some(h)) if !r.head.is_empty() && !h.is_empty() => {
220                    Some(h.starts_with(&r.head) || r.head.starts_with(h))
221                }
222                (Some(_), Some(_)) => Some(false),
223                _ => None,
224            };
225            GateState {
226                id: g.id.clone(),
227                job: g.job.clone(),
228                always: g.always,
229                on_demand: g.on_demand,
230                status,
231                reason,
232                last_run: run.cloned(),
233                on_head,
234                inputs_hash: hash_now.unwrap_or_default(),
235            }
236        })
237        .collect()
238}

240/// Fold the per-gate states into the whole report.
241pub(crate) fn report(
242    source: &str,
243    ledger: &str,
244    head: Option<String>,
245    gates: Vec<GateState>,
246    findings: Vec<String>,
247    now: &str,
248) -> GateStateReport {
249    let mut tallies: BTreeMap<String, usize> = BTreeMap::new();
250    for g in &gates {
251        *tallies.entry(g.status.as_str().to_string()).or_default() += 1;
252    }
253    GateStateReport {
254        source: source.to_string(),
255        ledger: ledger.to_string(),
256        head,
257        count: gates.len(),
258        tallies,
259        never_reported: gates
260            .iter()
261            .filter(|g| g.last_run.is_none())
262            .map(|g| g.id.clone())
263            .collect(),
264        gates,
265        unreached: vec![FORGE_UNREACHED.to_string()],
266        findings,
267        evaluated_at: now.to_string(),
268    }
269}

305    fn model() -> GateModel {
306        yaml::parse_into(MODEL).expect("the fixture model parses")

309    fn run(exit: i64, hash: &str, head: &str) -> GateRun {
310        GateRun {
311            recorded_at: "2026-10-15T00:00:00Z".into(),
312            head: head.into(),
313            branch: "feature/x".into(),
314            exit,
315            command: "the gate".into(),
316            inputs_hash: hash.into(),
317            session: "s-1".into(),
318        }
319    }

321    fn by_id<'a>(states: &'a [GateState], id: &str) -> &'a GateState {
322        states.iter().find(|g| g.id == id).expect("declared")
323    }

325    #[test]
326    fn every_gate_stands_in_exactly_the_word_its_last_run_earns() {
327        let m = model();
328        let mut runs = BTreeMap::new();
329        runs.insert("build".to_string(), run(0, "aaaa", "c0ffee"));
330        runs.insert("site".to_string(), run(0, "old", "c0ffee"));
331        runs.insert("lint".to_string(), run(3, "", "beef"));
332        let hashes: BTreeMap<String, Option<String>> = [
333            ("build", Some("aaaa")),
334            ("site", Some("new")),
335            ("lint", Some("")),
336            ("probe", None),
337        ]
338        .into_iter()
339        .map(|(k, v)| (k.to_string(), v.map(str::to_string)))
340        .collect();
341
342        let s = state(&m, &runs, &hashes, Some("c0ffee0000"));
343        assert_eq!(s.len(), 4, "every declared gate, reported or not");
344        assert_eq!(by_id(&s, "build").status, GateStatus::Pass);
345        assert_eq!(by_id(&s, "build").on_head, Some(true));
346        assert_eq!(by_id(&s, "site").status, GateStatus::Stale);
347        assert!(by_id(&s, "site").reason.contains("no longer describes"));
348        assert_eq!(by_id(&s, "lint").status, GateStatus::Fail);
349        assert_eq!(by_id(&s, "lint").on_head, Some(false));
350        let probe = by_id(&s, "probe");
351        assert_eq!(
352            probe.status,
353            GateStatus::Queued,
354            "never reported is not a pass"
355        );
356        assert!(probe.last_run.is_none() && probe.on_head.is_none());
357
358        let r = report(
359            ".ai/repo/ci/gates.yaml",
360            ".ai/local/state/ledger.jsonl",
361            Some("c0ffee0000".into()),
362            s,
363            Vec::new(),
364            "now",
365        );
366        assert_eq!(r.count, 4);
367        assert_eq!(r.never_reported, ["probe"]);
368        assert_eq!(r.tallies.get("pass"), Some(&1));
369        assert_eq!(r.tallies.get("queued"), Some(&1));
370        assert_eq!(r.unreached, [FORGE_UNREACHED]);
371    }

373    #[test]
374    fn a_run_that_cannot_be_checked_against_this_tree_is_unknown_never_a_pass() {
375        let m = model();
376        let mut runs = BTreeMap::new();
377        runs.insert("build".to_string(), run(0, "aaaa", "c0ffee"));
378        // no hash for any gate, and no head: a checkout git cannot read
379        let s = state(&m, &runs, &BTreeMap::new(), None);
380        let build = by_id(&s, "build");
381        assert_eq!(build.status, GateStatus::Unknown);
382        assert!(build.on_head.is_none(), "no head, so no claim about it");
383        assert_eq!(by_id(&s, "site").status, GateStatus::Queued);
384    }
