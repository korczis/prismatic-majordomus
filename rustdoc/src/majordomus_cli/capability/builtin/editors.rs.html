1//! The `editors` module: what Zed and Google Antigravity read from this repository, and how
2//! each file they read was derived.
3//!
4//! The answer is `crate::editors::report` and nothing else: the same plan
5//! `majordomus generate editors` writes, with every adapter's capabilities and the evidence
6//! for them, every projected file's state in this checkout and content address, and every
7//! finding. Read-only — the files are written by `majordomus generate editors` and checked by
8//! `majordomus generate --check` (ADR 0140).
9
10use crate::capability::handler::{CapabilityError, Context};
11use crate::capability::model::{Exposure, McpExposure, McpResource, Stability};
12use crate::capability::module::ModuleDescriptor;
13use crate::editors::EditorProjection;
14use crate::{capability, module};
15
16use super::{get, Empty};
17
18/// The URI under which `editors.status` is read as an MCP resource.
19pub const EDITORS_URI: &str = "majordomus://editors";
20
21fn editors_status(ctx: &Context, _: Empty) -> Result<EditorProjection, CapabilityError> {
22    let root = std::path::PathBuf::from(&ctx.index.repository.root);
23    Ok(crate::editors::report(
24        &ctx.index,
25        &ctx.registry,
26        &root,
27        crate::VERSION,
28    ))
29}

31/// The `editors` module: the editor and agent projections, explained.
32pub fn module() -> ModuleDescriptor {
33    module! {
34        id: "editors",
35        title: "Editor and agent projections",
36        description: "The files Zed and Google Antigravity read from this repository, derived from the skills and the tracked justfile: what each product supports and on what evidence, the state and content address of every projected file, and what to do about each finding.",
37        stability: Stability::BehaviorallyVerified,
38        capabilities: [
39            capability! {
40                id: "editors.status",
41                title: "Editor and agent projections",
42                description: "Every adapter (Zed, Antigravity) with each capability it supports, does not, or cannot be verified for, and the evidence; every projected file with the inputs it is derived from, its content address and its state here (current, stale, absent, or a conflict with a file this generator did not write); orphaned projections; and every finding with its remedy. Writes nothing: `majordomus generate editors` does.",
43                input: Empty,
44                output: EditorProjection,
45                stability: Stability::BehaviorallyVerified,
46                exposure: Exposure {
47                    mcp: Some(McpExposure {
48                        tool: Some("majordomus_editors".into()),
49                        resource: Some(McpResource { uri: EDITORS_URI.into(), name: "editors".into() }),
50                    }),
51                    http: get("/api/v1/editors"),
52                    cli: None,
53                },
54                tags: ["editors", "projections", "introspection"],
55                handler: editors_status,
56            },
57        ],
58    }
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64
65    /// The declaration is the one place the tool, the route and the resource are named; a
66    /// refactor that dropped one would still compile and every behavioural suite would
67    /// still pass. This would not.
68    #[test]
69    fn the_declaration_yields_the_projections_it_claims() {
70        let m = module();
71        assert_eq!(m.id.as_str(), "editors");
72        let ids: Vec<&str> = m
73            .capabilities
74            .iter()
75            .map(|e| e.capability.id.as_str())
76            .collect();
77        assert_eq!(ids, ["editors.status"]);
78        let exposure = &m.capabilities[0].capability.exposure;
79        let mcp = exposure.mcp.as_ref().expect("an MCP exposure");
80        assert_eq!(mcp.tool.as_deref(), Some("majordomus_editors"));
81        assert_eq!(
82            mcp.resource.as_ref().map(|r| r.uri.as_str()),
83            Some(EDITORS_URI)
84        );
85        assert_eq!(
86            exposure.http.as_ref().map(|h| h.path.as_str()),
87            Some("/api/v1/editors")
88        );
89        assert!(
90            exposure.cli.is_none(),
91            "the files are written by `generate editors`"
92        );
93    }
94
95    /// The capability answers with the projection's own report, over the repository the
96    /// context was built from.
97    #[test]
98    fn the_status_is_the_projections_report_for_this_repository() {
99        let repo = crate::synthetic::SyntheticRepository::small().unwrap();
100        let ctx = repo.context().unwrap();
101        let answer = editors_status(&ctx, Empty {}).unwrap();
102        assert_eq!(answer.schema, crate::editors::SCHEMA);
103        let ids: Vec<&str> = answer.adapters.iter().map(|a| a.id.as_str()).collect();
104        assert_eq!(ids, ["antigravity", "claude-code", "zed"]);
105        // no skill of the layer and no justfile: what is projected is what the registry
106        // itself declares, the commands and the domains every repository has
107        assert!(answer
108            .artifacts
109            .iter()
110            .all(|a| a.adapters == ["claude-code"]));
111        let intent = answer
112            .artifacts
113            .iter()
114            .find(|a| a.path == ".claude/skills/intent/SKILL.md")
115            .expect("the command skill");
116        assert_eq!(intent.sources, ["capability:intents.propose"]);
117        let mesh = answer
118            .artifacts
119            .iter()
120            .find(|a| a.path == ".claude/skills/mj-mesh/SKILL.md")
121            .expect("the mesh domain skill");
122        assert_eq!(mesh.sources, ["module:mesh"]);
123        assert_eq!(mesh.document, "editors/domain");
124        assert!(
125            answer
126                .artifacts
127                .iter()
128                .all(|a| a.path.starts_with(".claude/skills/")),
129            "every derived file is a skill Claude Code reads"
130        );
131    }
