1//! The shared server: one per checkout (ADR 0035, ADR 0044), holding the lease, serving every web surface the
2//! process resolved — the home page, the Cockpit, Swagger UI, OpenAPI, the capability
3//! routes, the documentation and every generated report — and MCP over HTTP for every
4//! peer that attaches. It is started by the first `majordomus mcp` or `serve` in a repository and
5//! ends when its owner's session is over and the last peer has left.
6

18/// How often the server looks for expired sessions while it waits for peers to leave.
19pub const REAP_INTERVAL: Duration = Duration::from_millis(500);

21/// A running shared server.
22pub struct SharedServer {
23    running: Running,
24    endpoint: Arc<McpEndpoint>,
25    lease: Lease,
26    /// Set by [`SharedServer::stop`]; the reader thread ends at its next tick.
27    stopping: Arc<AtomicBool>,
28    /// The mesh runtime this server activated (or left inactive, with the reason).
29    mesh: Arc<crate::mesh::MeshRuntime>,
30}

32/// Activate the mesh from the repository's declaration, when there is one and it is
33/// enabled. Every failure is a reason on `mesh.status`, never a failed server.
34fn activate_mesh(ctx: &Arc<Context>, version: &str, url: &str) {
35    let Some(parsed) = crate::capability::builtin::mesh::declaration(ctx) else {
36        return;
37    };
38    let config = match parsed {
39        Ok(config) => config,
40        Err(e) => {
41            tracing::warn!(error = %e, "the mesh declaration does not parse; the mesh stays off");
42            ctx.mesh
43                .decline(&format!("the declaration does not parse: {e}"));
44            return;
45        }
46    };
47    if !config.enabled {
48        ctx.mesh.decline("the mesh declaration is disabled");
49        return;
50    }
51    let Some(identity_path) = crate::mesh::default_identity_path() else {
52        ctx.mesh
53            .decline("no HOME and no XDG_STATE_HOME: nowhere to keep a node identity");
54        return;
55    };
56    let identity = match crate::mesh::NodeIdentity::load_or_create(&identity_path) {
57        Ok(identity) => identity,
58        Err(e) => {
59            tracing::warn!(error = %e, "the node identity did not load; the mesh stays off");
60            ctx.mesh.decline(&e.to_string());
61            return;
62        }
63    };
64    // Where peers can reach this server: the bound address, or every interface's address
65    // when it is bound to all of them. A loopback bind is reachable from this machine
66    // only, and `mesh doctor` says so.
67    let bound = url.trim_start_matches("http://").trim_end_matches('/');
68    let endpoints = crate::mesh::address::advertised_endpoints(bound);
69    let root = std::path::Path::new(&ctx.index.repository.root);
70    // One runtime per checkout, and one repository identity per repository on every
71    // machine — derived from content or declared, never from a path.
72    let runtime = crate::mesh::repository::runtime_id(&crate::repository::identity(root));
73    let repository =
74        crate::mesh::repository::resolve(root, config.cooperation.repository.as_deref());
75    let repos = repository
76        .as_ref()
77        .map(|r| vec![r.id.clone()])
78        .unwrap_or_default();
79    let node = identity.public.node_id.clone();
80    let identity = Arc::new(identity);
81    match ctx.mesh.activate_as(
82        &config,
83        Arc::clone(&identity),
84        &runtime,
85        endpoints.clone(),
86        repos,
87        version,
88    ) {
89        Ok(()) => {
90            tracing::info!(node = %node, runtime_id = %runtime, "mesh active: this node announces and listens per the declaration")
91        }
92        Err(e) => {
93            tracing::warn!(error = %e, "the mesh did not activate");
94            ctx.mesh.decline(&e.to_string());
95            return;
96        }
97    }
98    if !config.cooperation.enabled {
99        ctx.mesh
100            .decline_cooperation("cooperation is disabled in the mesh declaration");
101        return;
102    }
103    let repository = match repository {
104        Ok(repository) => repository,
105        Err(e) => {
106            tracing::warn!(error = %e, "the repository has no mesh identity; cooperation stays off");
107            ctx.mesh.decline_cooperation(&e.to_string());
108            return;
109        }
110    };
111    let setup = crate::mesh::cooperation::CooperationSetup {
112        identity,
113        runtime,
114        repository,
115        endpoints,
116        version: version.into(),
117        config: config.cooperation.clone(),
118        trust: config.trust.clone(),
119        journal_path: Some(root.join(".ai/local/state/mesh/journal.jsonl")),
120        registry: Arc::clone(ctx.mesh.registry()),
121        transport: Arc::new(crate::mesh::link::HttpTransport),
122        board: Some(Arc::clone(&ctx.peers)),
123        checkout: crate::mesh::cooperation::CheckoutFacts {
124            id: Some(crate::repository::identity(root)),
125            root: Some(root.to_path_buf()),
126        },
127    };
128    match crate::mesh::cooperation::Cooperation::new(setup) {
129        Ok(cooperation) => {
130            cooperation.start();
131            ctx.mesh.attach_cooperation(cooperation);
132        }
133        Err(e) => {
134            tracing::warn!(error = %e, "cooperation did not start");
135            ctx.mesh.decline_cooperation(&e.to_string());
136        }
137    }
138}

140impl SharedServer {
141    /// Bind, start serving, then publish the URL into the lease. With `fallback`, a taken
142    /// port is replaced by a free one and said so; without it, a taken port is an error.
143    #[allow(clippy::too_many_arguments)]
144    pub fn start(
145        live: Arc<crate::live::Live>,
146        version: &'static str,
147        host: &str,
148        port: u16,
149        fallback: bool,
150        declared_by: Option<&str>,
151        lease: Lease,
152        share_dir: Option<&std::path::Path>,
153    ) -> Result<Self> {
154        // Three ways to arrive at a socket, and they are not interchangeable: a canonical
155        // object declared this address (bind it, say which object said so), the port is a
156        // convenience (take a free one when it is taken), or the port is what was asked
157        // for (bind it or fail).
158        let bound = match declared_by {
159            Some(source) => server::bind_declared(host, port, source)?,
160            None if fallback => server::bind_or_fallback(host, port)?,
161            None => server::bind(host, port)?,
162        };
163        let url = bound.url();
164        let ctx_for_mesh = live.current();
165        let endpoint = Arc::new(McpEndpoint::new(Arc::clone(&live), version, url.clone()));
166        let router = Router::new(live, version)
167            .with_mcp(Arc::clone(&endpoint))
168            .with_cockpit(share_dir);
169        // The mesh, when the repository declares one — before the workers pick up their
170        // first request, so a client that connects on the "listening" line already sees
171        // the activated runtime. Never blocking: providers open sockets on their own
172        // threads, and a declaration that is absent, disabled or malformed leaves the
173        // runtime inactive with the reason `mesh.status` reports.
174        activate_mesh(&ctx_for_mesh, version, &url);
175        // what it serves is read off the resolution, so this line cannot name a route the
176        // process does not have or miss one it does
177        let surfaces = router.served()?.summary(&url);
178        // A published URL must already answer. Publishing before mesh activation and
179        // surface resolution lets direnv probe a bound socket with no request workers,
180        // report a failed server, and reload before anything can answer it.
181        let running = start_published(bound, router, || lease.publish(&url)).inspect_err(|_| {
182            ctx_for_mesh.mesh.stop();
183        })?;
184        // The server's own reader: every REAP_INTERVAL it forgets the HTTP sessions that
185        // stopped pinging — on every path, not only while the owner waits for peers to
186        // leave, so that a dead peer never stays `attached` on the board — and it checks
187        // that the lease is still its own. A lease another process took over is that
188        // process's to remove; this one stops claiming it, serves the peers it has, and
189        // ends with them.
190        let stopping = Arc::new(AtomicBool::new(false));
191        {
192            let endpoint = Arc::clone(&endpoint);
193            let stopping = Arc::clone(&stopping);
194            let path = lease.path().to_path_buf();
195            let token = lease.token().to_string();
196            let _ = std::thread::Builder::new()
197                .name("majordomus-server-tick".into())
198                .spawn(move || {
199                    let mut lost = false;
200                    while !stopping.load(Ordering::SeqCst) {
201                        std::thread::sleep(REAP_INTERVAL);
202                        let gone = endpoint.reap();
203                        if !gone.is_empty() {
204                            tracing::info!(peers = ?gone, "peer(s) expired: no message within the idle timeout");
205                        }
206                        if !lost
207                            && LeaseFile::read(&path)
208                                .document()
209                                .is_none_or(|d| d.token != token)
210                        {
211                            lost = true;
212                            crate::lease::lost();
213                            tracing::warn!(
214                                lease = %path.display(),
215                                "the lease is no longer this server's: another process took it over; this server serves the peers it has and ends with them"
216                            );
217                        }
218                    }
219                });
220        }
221        tracing::info!(
222            url = %url,
223            lease = %lease.path().display(),
224            surfaces = %surfaces,
225            // "of this checkout", not "for this repository": a linked worktree is a
226            // checkout with a lease and a server of its own, and this line is the first
227            // thing a person reads when a client starts one (ADR 0044).
228            "shared server listening on {url} — {surfaces}; the one server of this checkout: every later `majordomus mcp` here attaches to it, and it ends when the last peer leaves"
229        );
230        Ok(SharedServer {
231            running,
232            endpoint,
233            lease,
234            stopping,
235            mesh: ctx_for_mesh.mesh.clone(),
236        })
237    }
238
239    /// `http://host:port`.
240    pub fn url(&self) -> String {
241        self.running.url()
242    }
243
244    /// The MCP-over-HTTP endpoint, for whoever needs to count or reap its sessions.
245    pub fn endpoint(&self) -> &Arc<McpEndpoint> {
246        &self.endpoint
247    }
248
249    /// How many HTTP sessions are open right now, expired ones already forgotten.
250    pub fn peers_attached(&self) -> usize {
251        self.endpoint.reap();
252        self.endpoint.active()
253    }
254
255    /// Block until no HTTP session remains.
256    pub fn wait_until_peers_leave(&self) {
257        let mut announced = false;
258        loop {
259            let n = self.peers_attached();
260            if n == 0 {
261                break;
262            }
263            if !announced {
264                tracing::info!(
265                    peers = n,
266                    "the owner's session ended; serving until the last peer leaves"
267                );
268                announced = true;
269            }
270            std::thread::sleep(REAP_INTERVAL);
271        }
272    }
273
274    /// Stop serving and release the lease.
275    ///
276    /// The order is what matters. Telling the mesh to stop is immediate, but draining its
277    /// link table needs a lock a worker can hold across a dial that waits out the link
278    /// timeout — several seconds. Everything a waiting `serve stop` measures happens before
279    /// that: the listeners close and the lease is released, and only then does the mesh
280    /// drain. A lease released late is read by the next process as a server still holding
281    /// the port, which is what `serve stop` answers 10 for.
282    pub fn stop(self) {
283        self.mesh.begin_stop();
284        self.stopping.store(true, Ordering::SeqCst);
285        self.endpoint.close_all();
286        self.running.stop();
287        self.lease.release();
288        self.mesh.stop();
289        tracing::info!("shared server stopped");
290    }
291}

293/// Publish only after requests can be handled, and close the socket if publication fails.
294fn start_published(
295    bound: server::Bound,
296    router: Router,
297    publish: impl FnOnce() -> Result<()>,
298) -> Result<Running> {
299    let running = bound.start(router);
300    if let Err(error) = publish() {
301        running.stop();
302        return Err(error);
303    }
304    Ok(running)
305}

315    /// Whether the listening socket at `address` stops accepting within `deadline`. Dropping
316    /// a `tiny_http::Server` closes its listener on the library's own accept thread, which
317    /// nothing joins, so the socket outlives `stop` by a scheduling delay; under a loaded
318    /// test run an immediate connect can still land in the backlog. The wait is bounded and
319    /// a listener that never closes still fails.
320    fn listener_closes(address: std::net::SocketAddr, deadline: Duration) -> bool {
321        let until = std::time::Instant::now() + deadline;
322        loop {
323            if std::net::TcpStream::connect_timeout(&address, Duration::from_millis(200)).is_err() {
324                return true;
325            }
326            if std::time::Instant::now() >= until {
327                return false;
328            }
329            std::thread::sleep(Duration::from_millis(20));
330        }
331    }
332
333    #[test]
334    fn publication_observes_a_serving_socket_and_failure_closes_it() {
335        let fixture = SyntheticRepository::small().unwrap();
336        let app = App::load(&RepoArgs {
337            repo: Some(fixture.root().to_path_buf()),
338            discovery: DiscoveryMode::Filesystem,
339            share: Some(std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../../share")),
340            ..Default::default()
341        })
342        .unwrap();
343        let router = Router::new(Arc::clone(&app.context), crate::VERSION);
344        router.served().unwrap();
345
346        // Check at the publication boundary, not after start returns or its log appears.
347        // No retry can hide a URL being announced before requests are handled.
348        for refused in [false, true] {
349            let bound = server::bind("127.0.0.1", 0).unwrap();
350            let url = bound.url();
351            let address: std::net::SocketAddr = bound.address().parse().unwrap();
352            let result = start_published(bound, router.clone(), || {
353                let reply =
354                    crate::lease::probe_reply(&url, app.repository.root(), Duration::from_secs(2));
355                assert!(reply.is_some(), "publication preceded request handling");
356                if refused {
357                    Err(Error::Lease {
358                        reason: "publication refused".into(),
359                    })
360                } else {
361                    Ok(())
362                }
363            });
364            if refused {
365                assert!(result.is_err());
366            } else {
367                result.unwrap().stop();
368            }
369            assert!(
370                listener_closes(address, Duration::from_secs(5)),
371                "request workers retained the socket after shutdown"
372            );
373        }
374
375        // Losing the lease while routes are prepared must also unwind the real startup.
376        let crate::lease::Role::Server(lease) = crate::lease::elect(&app.repository).unwrap()
377        else {
378            panic!("fixture unexpectedly has a server");
379        };
380        std::fs::remove_file(lease.path()).unwrap();
381        assert!(SharedServer::start(
382            Arc::new(crate::live::Live::pinned(app.context)),
383            crate::VERSION,
384            "127.0.0.1",
385            0,
386            false,
387            None,
388            lease,
389            None,
390        )
391        .is_err());
392    }
