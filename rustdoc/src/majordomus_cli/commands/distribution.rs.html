1//! `majordomus distribution`: the distribution model through the registry's own
2//! capabilities. Every subcommand here executes a capability and renders what it answered;
3//! the facts are the model's, the phrasing is this file's, and nothing between them
4//! restates a platform, a name or a URL.

16/// The exit code when the model or a release record breaks an invariant.
17pub const EXIT_INVALID: u8 = 10;

19/// Run `majordomus distribution`.
20pub fn run(args: DistributionArgs) -> Result<u8> {
21    let app = App::load(&args.repo)?;
22    let stdout = std::io::stdout();
23    let mut out = stdout.lock();
24    match args.command {
25        None | Some(DistributionCommand::Show) => show(&app, args.format, &mut out),
26        Some(DistributionCommand::Status) => status(&app, args.format, &mut out),
27        Some(DistributionCommand::Targets) => targets(&app, args.format, &mut out),
28        Some(DistributionCommand::Validate) => validate(&app, args.format, &mut out),
29        Some(DistributionCommand::Matrix) => {
30            let model = Model::load(&app.share)?;
31            w(&mut out, render::matrix_json(&model).trim_end().to_string())?;
32            Ok(0)
33        }
34        Some(DistributionCommand::Artifact { target, tag }) => {
35            artifact(&app, &target, &tag, args.format, &mut out)
36        }
37        Some(DistributionCommand::Releases) => releases(&app, args.format, &mut out),
38        Some(DistributionCommand::Metadata { record }) => metadata(&app, &record, &mut out),
39        Some(DistributionCommand::Build) => build(&app, args.format, &mut out),
40    }
41}

43type Out<'a> = std::io::StdoutLock<'a>;

45fn w(out: &mut Out<'_>, s: String) -> Result<()> {
46    writeln!(out, "{s}").map_err(Error::Transport)
47}

49fn pretty(v: &Value) -> String {
50    serde_json::to_string_pretty(v).unwrap_or_else(|_| v.to_string())

53fn map(e: CapabilityError) -> Error {
54    Error::Protocol {
55        reason: e.to_string(),
56    }

59/// Execute the capability exposed at a command path: the command line is a projection of
60/// the registry, not a second implementation beside it.
61fn call(app: &App, path: &[&str], input: Value) -> Result<Value> {
62    let ctx = &app.context;
63    let words: Vec<String> = path.iter().map(|w| (*w).to_string()).collect();
64    let id = ctx
65        .registry
66        .by_cli(&words)
67        .map(|c| c.id.as_str())
68        .ok_or_else(|| Error::Protocol {
69            reason: format!(
70                "no capability is exposed as `majordomus {}`",
71                path.join(" ")
72            ),
73        })?;
74    ctx.execute(id, input).map_err(map)
75}

77/// `majordomus distribution status`. Exits 10 when the published installation does not
78/// work, so that a script and a person get the same answer without reading prose.
79fn status(app: &App, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
80    let v = call(app, &["distribution", "status"], json!({}))?;
81    let installable = v["installable"].as_bool().unwrap_or(false);
82    match format {
83        OutputFormat::Json => w(out, pretty(&v))?,
84        OutputFormat::Text => {
85            let s = |k: &str| v[k].as_str().unwrap_or("?").to_string();
86            for c in v["checks"].as_array().into_iter().flatten() {
87                let state = match c["state"].as_str().unwrap_or("") {
88                    "ok" => "OK  ",
89                    "failed" => "FAIL",
90                    _ => "?   ",
91                };
92                w(
93                    out,
94                    format!(
95                        "{state} {:<10} {}",
96                        c["id"].as_str().unwrap_or(""),
97                        c["observed"].as_str().unwrap_or("")
98                    ),
99                )?;
100            }
101            w(out, String::new())?;
102            w(out, format!("install      {}", s("install_command")))?;
103            w(
104                out,
105                format!(
106                    "stable       {}",
107                    v["stable_tag"].as_str().unwrap_or("none")
108                ),
109            )?;
110            w(
111                out,
112                format!("installable  {}", if installable { "yes" } else { "NO" }),
113            )?;
114            // A failure states its cause and the command that changes it, in that order,
115            // because an operator reading this is deciding what to do next.
116            if !installable {
117                for c in v["checks"].as_array().into_iter().flatten() {
118                    if c["state"].as_str() != Some("failed") {
119                        continue;
120                    }
121                    w(out, String::new())?;
122                    w(out, format!("CAUSE  {}", c["cause"].as_str().unwrap_or("")))?;
123                    if let Some(next) = c["next"].as_str() {
124                        w(out, format!("NEXT   {next}"))?;
125                    }
126                }
127            }
128        }
129    }
130    Ok(if installable { 0 } else { EXIT_INVALID })
131}

133fn show(app: &App, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
134    let v = call(app, &["distribution", "show"], json!({}))?;
135    match format {
136        OutputFormat::Json => w(out, pretty(&v))?,
137        OutputFormat::Text => {
138            let s = |k: &str| v[k].as_str().unwrap_or("?").to_string();
139            w(out, format!("binary       {}", s("binary")))?;
140            w(out, format!("repository   {}", s("repository")))?;
141            w(out, format!("installer    {}", s("installer_url")))?;
142            w(out, format!("install      {}", s("install_command")))?;
143            w(out, format!("then         {}", s("next_command")))?;
144            w(out, format!("launchers    {}", s("install_dir")))?;
145            w(out, format!("prefix       {}", s("prefix")))?;
146            w(out, format!("checksum     {}", s("checksum")))?;
147            w(out, format!("latest       {}", s("latest_url")))?;
148            w(
149                out,
150                format!(
151                    "targets      {} published of {} declared",
152                    v["supported"],
153                    v["targets"].as_array().map_or(0, Vec::len)
154                ),
155            )?;
156        }
157    }
158    Ok(0)
159}

161fn targets(app: &App, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
162    let v = call(app, &["distribution", "show"], json!({}))?;
163    let rows = v["targets"].as_array().cloned().unwrap_or_default();
164    match format {
165        OutputFormat::Json => w(out, pretty(&json!({ "targets": rows })))?,
166        OutputFormat::Text => {
167            w(
168                out,
169                format!(
170                    "{:<20} {:<32} {:<13} {}",
171                    "ID", "RUST TARGET", "STATUS", "TITLE"
172                ),
173            )?;
174            for t in &rows {
175                w(
176                    out,
177                    format!(
178                        "{:<20} {:<32} {:<13} {}",
179                        t["id"].as_str().unwrap_or("?"),
180                        t["rust_target"].as_str().unwrap_or("?"),
181                        t["status"].as_str().unwrap_or("?"),
182                        t["title"].as_str().unwrap_or("?")
183                    ),
184                )?;
185            }
186        }
187    }
188    Ok(0)
189}

191fn validate(app: &App, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
192    let model = Model::load(&app.share)?;
193    let records = Releases::load(app.repository.root())?;
194    let mut findings = model.findings();
195    findings.extend(records.findings(&model));
196    match format {
197        OutputFormat::Json => w(
198            out,
199            pretty(&json!({
200                "valid": findings.is_empty(),
201                "targets": model.targets.len(),
202                "published": model.published().count(),
203                "releases": records.releases.len(),
204                "findings": findings,
205            })),
206        )?,
207        OutputFormat::Text => {
208            for f in &findings {
209                w(out, format!("FAIL distribution  {f}"))?;
210            }
211            if findings.is_empty() {
212                w(
213                    out,
214                    format!(
215                        "OK   distribution  {} target(s), {} published, {} release(s) recorded",
216                        model.targets.len(),
217                        model.published().count(),
218                        records.releases.len()
219                    ),
220                )?;
221            }
222        }
223    }
224    Ok(if findings.is_empty() { 0 } else { EXIT_INVALID })
225}

227fn artifact(
228    app: &App,
229    target: &str,
230    tag: &str,
231    format: OutputFormat,
232    out: &mut Out<'_>,
233) -> Result<u8> {
234    let v = call(
235        app,
236        &["distribution", "artifact"],
237        json!({ "target": target, "tag": tag }),
238    )?;
239    match format {
240        OutputFormat::Json => w(out, pretty(&v))?,
241        // two lines, name then root: what `scripts/release-package` reads
242        OutputFormat::Text => {
243            w(out, v["name"].as_str().unwrap_or_default().to_string())?;
244            w(out, v["root"].as_str().unwrap_or_default().to_string())?;
245        }
246    }
247    Ok(0)
248}

250fn releases(app: &App, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
251    let v = call(app, &["distribution", "releases"], json!({}))?;
252    match format {
253        OutputFormat::Json => w(out, pretty(&v))?,
254        OutputFormat::Text => {
255            let latest = v["latest"]["tag"].as_str().unwrap_or("");
256            let rows = v["releases"].as_array().cloned().unwrap_or_default();
257            if rows.is_empty() {
258                w(out, "no release is recorded yet".to_string())?;
259                return Ok(0);
260            }
261            w(
262                out,
263                format!(
264                    "{:<14} {:<22} {:<12} {}",
265                    "TAG", "PUBLISHED", "CHANNEL", "ARTIFACTS"
266                ),
267            )?;
268            for r in &rows {
269                let tag = r["tag"].as_str().unwrap_or("?");
270                w(
271                    out,
272                    format!(
273                        "{:<14} {:<22} {:<12} {}{}",
274                        tag,
275                        r["published_at"].as_str().unwrap_or("?"),
276                        r["channel"].as_str().unwrap_or("?"),
277                        r["artifacts"],
278                        if tag == latest { "   (latest)" } else { "" }
279                    ),
280                )?;
281            }
282        }
283    }
284    Ok(0)
285}

287/// The public metadata one record publishes, rendered from the record and the model alone.
288/// The release pipeline uses this to see what it is about to commit; the test suite uses it
289/// to serve a release that never existed. Neither renders the document itself.
290fn metadata(app: &App, record: &std::path::Path, out: &mut Out<'_>) -> Result<u8> {
291    let model = Model::load(&app.share)?;
292    if !record.is_file() {
293        return Err(Error::InvalidRelease {
294            path: record.display().to_string(),
295            reason: "no such release record; a record is written by the release pipeline under .ai/repo/releases/ and named by its tag".into(),
296        });
297    }
298    let text = std::fs::read_to_string(record).map_err(|e| Error::io(record, e))?;
299    let release =
300        crate::distribution::Release::parse(&text).map_err(|reason| Error::InvalidRelease {
301            path: record.display().to_string(),
302            reason,
303        })?;
304    let findings = release.findings(&model);
305    if let Some(first) = findings.first() {
306        return Err(Error::InvalidRelease {
307            path: record.display().to_string(),
308            reason: first.clone(),
309        });
310    }
311    w(out, release.public_json(&model).trim_end().to_string())?;
312    Ok(0)
313}

315fn build(app: &App, format: OutputFormat, out: &mut Out<'_>) -> Result<u8> {
316    let v = call(app, &["distribution", "build"], json!({}))?;
317    match format {
318        OutputFormat::Json => w(out, pretty(&v))?,
319        OutputFormat::Text => {
320            let s = |k: &str| v[k].as_str().unwrap_or("?").to_string();
321            w(out, format!("version      {}", s("version")))?;
322            w(out, format!("target       {}", s("target")))?;
323            w(out, format!("profile      {}", s("profile")))?;
324            w(out, format!("commit       {}", s("commit")))?;
325            w(
326                out,
327                format!(
328                    "distributed  {}",
329                    v["distribution_target"]
330                        .as_str()
331                        .unwrap_or("no target of the model matches this triple")
332                ),
333            )?;
334        }
335    }
336    Ok(0)
