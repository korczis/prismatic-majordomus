1//! The product model: what this repository's product does for a person, as the features
2//! under the layer's features section declare it, with every fact a surface may say about
3//! a feature derived from the registries that own those facts.
4//!
5//! Nothing here discovers or parses a file. A feature is an ordinary declarative object of
6//! the layer — kind `feature` in `share/kinds.yaml`, discovered by the source class in the
7//! repository's `sources.yaml`, validated against its JSON Schema by the time the index
8//! holds it. This module projects that validated metadata into typed records, resolves
9//! every reference the records make against the registry that owns it, and derives what
10//! nobody authored: which interfaces expose a feature, the tools, routes and command-line
11//! paths behind it, how many objects of its kinds the layer holds, which operational
12//! moments it answers, and what is guaranteed about it.
13//!
14//! Built once, when a [`crate::capability::Context`] is composed, and shared by every
15//! projection: the command line, the HTTP routes, the MCP tools, the derived graph and the
16//! site's dataset are readers of this one value.
17//!
18//! Two directions are kept apart on purpose, exactly as the Why catalogue keeps them.
19//! **Authored** is what a file says: the headline, the summary, the order, whether the
20//! homepage features it, and the typed references to what the feature is made of.
21//! **Derived** is everything else, and none of it may be written into a source file: the
22//! schema refuses a `surfaces` or a `route` key, and this module is where those come from.
23

137/// The interfaces a feature is exposed through, decided from what its references project
138/// and from nothing a file says. Each is a fact with a reason a reader can check: `cli` is
139/// true when a module of the feature has a capability with a command-line path or the
140/// feature names a shell command; `api` when a module has an HTTP route; `mcp` when a module
141/// has an MCP tool or resource, or the feature names a kind, since every object of the layer
142/// is an MCP resource; `cockpit` when the feature names a Cockpit area or a module, since
143/// every capability has a Cockpit page; `docs` when it names a document.
144#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
145pub struct Surfaces {
146    /// The command line: the executable's own, or the shell tool's.
147    pub cli: bool,
148    /// The HTTP routes under `/api/v1/`, and the OpenAPI document over them.
149    pub api: bool,
150    /// MCP: a tool, or a resource.
151    pub mcp: bool,
152    /// The Cockpit.
153    pub cockpit: bool,
154    /// The documentation.
155    pub docs: bool,
156}

158impl Surfaces {
159    /// The surface ids that are true, in the order the vocabulary lists them.
160    pub fn present(&self) -> Vec<&'static str> {
161        SURFACES
162            .iter()
163            .filter(|(id, _, _)| self.has(id))
164            .map(|(id, _, _)| *id)
165            .collect()
166    }
167
168    /// Is this surface exposed?
169    pub fn has(&self, id: &str) -> bool {
170        match id {
171            "cli" => self.cli,
172            "api" => self.api,
173            "mcp" => self.mcp,
174            "cockpit" => self.cockpit,
175            "docs" => self.docs,
176            _ => false,
177        }
178    }
179}

552/// What claims a module: the features that name it, each with the areas it serves.
553type Claims = BTreeMap<String, Vec<(String, Vec<String>)>>;

555/// Resolve one area per module, and name the modules whose claimants disagree.
556///
557/// The rule is the catalogue's own ranking: among the areas the claiming features serve,
558/// the one with the lowest `weight`, ties broken by id. The areas are already ordered by
559/// what the operator cares about most, and a second ranking invented here would be a second
560/// opinion about a question the layer has already answered.
561///
562/// Two features that name one module and share no area disagree about what the module is
563/// for. The resolution stays deterministic and the disagreement is still returned, so that
564/// it is settled in the feature files where the semantics live rather than by a tiebreak in
565/// this function.
566fn resolve_module_areas(
567    claims: &Claims,
568    weight: &BTreeMap<&str, u32>,
569) -> (BTreeMap<String, String>, Vec<(String, String)>) {
570    let mut areas: BTreeMap<String, String> = BTreeMap::new();
571    let mut contested: Vec<(String, String)> = Vec::new();
572
573    for (module, claimants) in claims {
574        // A set, not a sort and a dedup: unique and ordered is what a BTreeSet is, and the
575        // canonical-order gate is right to ask why a comparator appeared here.
576        let union: std::collections::BTreeSet<&str> = claimants
577            .iter()
578            .flat_map(|(_, areas)| areas.iter().map(String::as_str))
579            .collect();
580
581        let disjoint = claimants.len() > 1
582            && claimants.iter().any(|(_, a)| {
583                claimants
584                    .iter()
585                    .any(|(_, b)| !a.iter().any(|x| b.contains(x)))
586            });
587        if disjoint {
588            let who: Vec<String> = claimants
589                .iter()
590                .map(|(id, a)| format!("{id} ({})", a.join(", ")))
591                .collect();
592            contested.push((module.clone(), who.join(" / ")));
593        }
594
595        if let Some(best) = union.iter().min_by(|a, b| {
596            weight
597                .get(*a)
598                .unwrap_or(&u32::MAX)
599                .cmp(weight.get(*b).unwrap_or(&u32::MAX))
600                .then_with(|| a.cmp(b))
601        }) {
602            areas.insert(module.clone(), (*best).to_string());
603        }
604    }
605    (areas, contested)
606}

608/// Levenshtein distance, for the nearest-candidate hint on an unresolved reference.
609fn distance(a: &str, b: &str) -> usize {
610    let (a, b): (Vec<char>, Vec<char>) = (a.chars().collect(), b.chars().collect());
611    let mut prev: Vec<usize> = (0..=b.len()).collect();
612    let mut cur = vec![0usize; b.len() + 1];
613    for (i, ca) in a.iter().enumerate() {
614        cur[0] = i + 1;
615        for (j, cb) in b.iter().enumerate() {
616            let cost = usize::from(ca != cb);
617            cur[j + 1] = (prev[j] + cost).min(prev[j + 1] + 1).min(cur[j] + 1);
618        }
619        std::mem::swap(&mut prev, &mut cur);
620    }
621    prev[b.len()]
622}

624/// The nearest candidate to `value`, when one is close enough to be a plausible typo.
625fn nearest<'a>(value: &str, candidates: impl Iterator<Item = &'a str>) -> Option<String> {
626    let limit = (value.chars().count() / 3).max(2);
627    candidates
628        .map(|c| (distance(value, c), c))
629        .filter(|(d, _)| *d <= limit)
630        .min_by(|a, b| a.0.cmp(&b.0).then(a.1.cmp(b.1)))
631        .map(|(_, c)| c.to_string())
632}

648/// What the layer holds, read once into the lookups every reference needs.
649struct Lookups<'a> {
650    commands: BTreeMap<String, &'a Object>,
651    rules_by_stem: BTreeMap<String, &'a Object>,
652    docs: BTreeMap<String, &'a Object>,
653    adrs_by_id: BTreeMap<String, &'a Object>,
654    claims: BTreeMap<String, &'a Object>,
655    use_cases: BTreeMap<String, &'a Object>,
656    kinds: BTreeMap<String, usize>,
657    areas: BTreeSet<String>,
658    audiences: BTreeSet<String>,
659}
660
661impl<'a> Lookups<'a> {
662    fn new(index: &'a Index) -> Self {
663        let mut l = Lookups {
664            commands: BTreeMap::new(),
665            rules_by_stem: BTreeMap::new(),
666            docs: BTreeMap::new(),
667            adrs_by_id: BTreeMap::new(),
668            claims: BTreeMap::new(),
669            use_cases: BTreeMap::new(),
670            kinds: BTreeMap::new(),
671            areas: BTreeSet::new(),
672            audiences: BTreeSet::new(),
673        };
674        for o in &index.objects {
675            *l.kinds.entry(o.kind.clone()).or_default() += 1;
676            match o.kind.as_str() {
677                // a public command only: an internal one is dispatched and absent from the
678                // help text, and a feature may not present it
679                "command" => {
680                    if meta_str(o, "visibility").as_deref() == Some("public") {
681                        l.commands.insert(o.identity.clone(), o);
682                    }
683                }
684                "rule" => {
685                    let stem = o
686                        .identity
687                        .split_once('@')
688                        .map(|(s, _)| s.to_string())
689                        .unwrap_or_else(|| o.identity.clone());
690                    l.rules_by_stem.entry(stem).or_insert(o);
691                }
692                "document" => {
693                    l.docs.insert(o.provenance.path.clone(), o);
694                }
695                "adr" => {
696                    if let Some(id) = meta_str(o, "id") {
697                        l.adrs_by_id.insert(id, o);
698                    }
699                }
700                "claim" => {
701                    l.claims.insert(o.identity.clone(), o);
702                }
703                "use-case" => {
704                    l.use_cases.insert(o.identity.clone(), o);
705                }
706                crate::why::AREA => {
707                    l.areas.insert(o.identity.clone());
708                }
709                crate::why::AUDIENCE => {
710                    l.audiences.insert(o.identity.clone());
711                }
712                _ => {}
713            }
714        }
715        l
716    }
717}

719impl ProductModel {
720    /// Build the model from an index, the registry its module references resolve against,
721    /// the catalogue whose moments it answers, and the topology whose surfaces it names.
722    /// Deterministic: the same objects produce the same value, and no derivation reads a
723    /// clock.
724    pub fn build(
725        index: &Index,
726        registry: &CapabilityRegistry,
727        why: &Catalogue,
728        web: &Topology,
729    ) -> Self {
730        let mut m = ProductModel::default();
731        let mut features: Vec<Feature> = Vec::new();
732        let mut seen: BTreeMap<String, String> = BTreeMap::new();
733        for o in index.objects.iter().filter(|o| o.kind == FEATURE) {
734            let path = o.provenance.path.clone();
735            if let Some(first) = seen.insert(o.identity.clone(), path.clone()) {
736                m.findings.push(ProductFinding {
737                    severity: Severity::Error,
738                    code: "duplicate_identity".into(),
739                    path: path.clone(),
740                    id: Some(o.identity.clone()),
741                    field: Some("id".into()),
742                    message: format!(
743                        "a second feature claims the identity '{}' (first: {first})",
744                        o.identity
745                    ),
746                    did_you_mean: None,
747                });
748            }
749            let stem = path
750                .rsplit('/')
751                .next()
752                .and_then(|f| f.strip_suffix(".md"))
753                .unwrap_or_default();
754            if stem != o.identity {
755                m.findings.push(ProductFinding {
756                    severity: Severity::Error,
757                    code: "filename_mismatch".into(),
758                    path: path.clone(),
759                    id: Some(o.identity.clone()),
760                    field: Some("id".into()),
761                    message: format!(
762                        "the file is named '{stem}.md' and the record's id is '{}'; the id is the file name and the route",
763                        o.identity
764                    ),
765                    did_you_mean: Some(format!("{}.md", o.identity)),
766                });
767            }
768            if let Ok(mut f) = serde_json::from_value::<Feature>(o.metadata.clone()) {
769                f.route = format!("{ROUTE}{}/", f.id);
770                f.source = path;
771                f.body = o.body.clone();
772                features.push(f);
773            }
774        }
775        crate::order::canonical(&mut features);
776
777        m.adopt_diagnostics(index);
778        let lookups = Lookups::new(index);
779        let backlinks = backlinks(&features);
780        let areas = nav::areas();
781
782        let mut resolved = Vec::with_capacity(features.len());
783        for f in &features {
784            let (r, findings) = resolve(
785                f, registry, index, why, web, &lookups, areas, &backlinks, &features,
786            );
787            m.findings.extend(findings);
788            resolved.push(r);
789        }
790        m.features = resolved;
791        for (i, r) in m.features.iter().enumerate() {
792            m.by_id.insert(r.feature.id.clone(), i);
793        }
794        m.providers = providers(index);
795        for decl in &index.providers.providers {
796            if !decl.template {
797                m.findings.push(ProductFinding {
798                    severity: Severity::Error,
799                    code: "provider_without_template".into(),
800                    path: "share/providers.yaml".into(),
801                    id: Some(decl.id.clone()),
802                    field: Some("providers".into()),
803                    message: format!(
804                        "provider '{}' is declared and the distribution ships no share/providers/{}.tmpl; a provider with no bootstrap is a name and nothing else",
805                        decl.id, decl.id
806                    ),
807                    did_you_mean: None,
808                });
809            } else if !decl.declared {
810                m.findings.push(ProductFinding {
811                    severity: Severity::Warning,
812                    code: "provider_undeclared".into(),
813                    path: format!("share/providers/{}.tmpl", decl.id),
814                    id: Some(decl.id.clone()),
815                    field: None,
816                    message: format!(
817                        "provider '{}' has a template and no entry in share/providers.yaml; it is named by its file and declares no title, client configuration or scratch root",
818                        decl.id
819                    ),
820                    did_you_mean: None,
821                });
822            }
823        }
824        m.coverage(registry, &lookups);
825        let area_weights: BTreeMap<&str, u32> = why
826            .areas()
827            .iter()
828            .map(|a| (a.id.as_str(), a.weight))
829            .collect();
830        m.derive_module_areas(&area_weights);
831        m.findings.sort_by(|a, b| {
832            b.severity
833                .cmp(&a.severity)
834                .then(a.path.cmp(&b.path))
835                .then(a.code.cmp(&b.code))
836                .then(a.message.cmp(&b.message))
837        });
838        m.fingerprint = m.compute_fingerprint();
839        m
840    }

842    /// The area each capability module serves, derived rather than declared.
843    ///
844    /// A module has no area of its own and is given none: the features that name it in
845    /// `modules:` already say which areas they serve, so the module's area is theirs,
846    /// resolved by [`resolve_module_areas`]. A feature that declares no area claims
847    /// nothing; a module no feature claims is shown under no heading and ordered last.
848    fn derive_module_areas(&mut self, weight: &BTreeMap<&str, u32>) {
849        let mut claims: Claims = BTreeMap::new();
850        for r in &self.features {
851            if r.feature.areas.is_empty() {
852                continue;
853            }
854            for module in &r.feature.modules {
855                claims
856                    .entry(module.clone())
857                    .or_default()
858                    .push((r.feature.id.clone(), r.feature.areas.clone()));
859            }
860        }
861        let (areas, contested) = resolve_module_areas(&claims, weight);
862        self.module_areas = areas;
863        for (module, who) in contested {
864            self.findings.push(ProductFinding {
865                severity: Severity::Warning,
866                code: "contested_area".into(),
867                path: ".ai/repo/features".into(),
868                id: Some(module.clone()),
869                field: Some("areas".into()),
870                message: format!(
871                    "the module '{module}' is claimed by features that share no area: {who}; \
872                     the lowest-weight area wins until a feature file settles it"
873                ),
874                did_you_mean: None,
875            });
876        }
877    }

890    /// Every diagnostic the index raised about a file of the features section: an object the
891    /// index refused never reaches the model, and without this the model would report
892    /// itself valid over the objects excluded from under it.
893    fn adopt_diagnostics(&mut self, index: &Index) {
894        let Some(section) = index.repository.sections.get("features") else {
895            return;
896        };
897        for d in index

899            .iter()
900            .filter(|d| d.severity == Severity::Error)

903                continue;
904            };
905            if !path.starts_with(section.as_str()) {
906                continue;
907            }
908            self.findings.push(ProductFinding {
909                severity: Severity::Error,
910                code: d.code.clone(),
911                path: path.to_string(),
912                id: None,
913                field: None,
914                message: d.message.clone(),
915                did_you_mean: None,
916            });
917        }

920    /// Every module of the executable, every public command and every kind of the layer,
921    /// each with the stable features that name it. A thing no feature names is a warning:
922    /// the product page says nothing about something the product does.
923    fn coverage(&mut self, registry: &CapabilityRegistry, lookups: &Lookups<'_>) {
924        let stable: Vec<&ResolvedRefs> = self
925            .features
926            .iter()
927            .filter(|r| r.feature.status == STABLE)
928            .collect();
929        let naming = |pick: &dyn Fn(&Feature) -> &Vec<String>, id: &str| -> Vec<String> {
930            stable
931                .iter()
932                .filter(|r| pick(&r.feature).iter().any(|x| x == id))
933                .map(|r| r.feature.id.clone())
934                .collect()
935        };
936        self.modules = registry
937            .modules()
938            .filter(|m| m.source != crate::capability::registry::ModuleSource::Declarative)
939            .map(|m| ProductCoverage {
940                id: m.id.to_string(),
941                title: m.title.clone(),
942                features: naming(&|f| &f.modules, m.id.as_str()),
943            })
944            .collect();
945        self.commands = lookups
946            .commands
947            .iter()
948            .map(|(id, o)| ProductCoverage {
949                id: id.clone(),
950                title: o.title.clone().unwrap_or_else(|| id.clone()),
951                features: naming(&|f| &f.commands, id),
952            })
953            .collect();
954        // the kinds a feature can present: those holding an object of the layer under the
955        // manifest's sections; the implementation and test classes index source files so
956        // that a claim's chain resolves, and are not things the product does
957        self.kinds = lookups
958            .kinds
959            .iter()
960            .filter(|(k, _)| !matches!(k.as_str(), "implementation" | "test" | "document"))
961            .map(|(k, n)| ProductCoverage {
962                id: k.clone(),
963                title: format!("{k} ({n})"),
964                features: naming(&|f| &f.kinds, k),
965            })
966            .collect();
967        let mut uncovered = Vec::new();
968        for (what, list) in [
969            ("module", &self.modules),
970            ("command", &self.commands),
971            ("kind", &self.kinds),
972        ] {
973            for c in list.iter().filter(|c| c.features.is_empty()) {
974                uncovered.push(ProductFinding {
975                    severity: Severity::Warning,
976                    code: "uncovered".into(),
977                    path: match what {
978                        "module" => crate::capability::model::CRATE_DIR.to_string(),
979                        "command" => "share/commands.yaml".to_string(),
980                        _ => "share/kinds.yaml".to_string(),
981                    },
982                    id: Some(c.id.clone()),
983                    field: None,
984                    message: format!(
985                        "the {what} '{}' is named by no stable feature; the product page says nothing about it",
986                        c.id
987                    ),
988                    did_you_mean: None,
989                });
990            }
991        }
992        self.findings.extend(uncovered);
993    }

995    /// A hash of the model's own sources, in identity order: the feature files, and the
996    /// facts derived from the registries, so a registry change that moves a feature's
997    /// surfaces moves the fingerprint too.
998    fn compute_fingerprint(&self) -> String {
999        let mut h = Sha256::new();
1000        for r in &self.features {
1001            h.update(r.feature.id.as_bytes());
1002            h.update([0]);
1003            h.update(r.feature.source.as_bytes());
1004            h.update([0]);
1005            h.update(r.feature.body.as_bytes());
1006            h.update([0]);
1007            h.update(
1008                serde_json::to_string(&r.surfaces)
1009                    .unwrap_or_default()
1010                    .as_bytes(),
1011            );
1012            h.update(
1013                serde_json::to_string(&r.counts)
1014                    .unwrap_or_default()
1015                    .as_bytes(),
1016            );
1017            h.update(b"\n");
1018        }
1019        format!("{:x}", h.finalize())
1020    }

1119/// Resolve one feature: every reference against the thing it names, with a finding for
1120/// each that resolves to nothing, then everything derived.
1121#[allow(clippy::too_many_arguments)]
1122fn resolve(
1123    f: &Feature,
1124    registry: &CapabilityRegistry,
1125    index: &Index,
1126    why: &Catalogue,
1127    web: &Topology,
1128    lookups: &Lookups<'_>,
1129    areas: &[nav::AreaInfo],
1130    backlinks: &BTreeMap<String, Vec<String>>,
1131    all: &[Feature],
1132) -> (ResolvedRefs, Vec<ProductFinding>) {
1133    let mut findings = Vec::new();
1134    // the unresolved references, collected apart so the closure holds nothing the other
1135    // findings need, and merged once every reference has been looked at
1136    let mut unknowns = Vec::new();
1137    let mut unknown = |field: &str, value: &str, what: &str, candidates: Vec<&str>| {
1138        unknowns.push(ProductFinding {
1139            severity: Severity::Error,
1140            code: "unknown_reference".into(),
1141            path: f.source.clone(),
1142            id: Some(f.id.clone()),
1143            field: Some(field.into()),
1144            message: format!("unknown {what} reference: \"{value}\""),
1145            did_you_mean: nearest(value, candidates.into_iter()),
1146        });
1147    };
1148
1149    // modules: the builtin ones of the registry, with every capability they compose
1150    let mut module_refs = Vec::new();
1151    for id in &f.modules {
1152        match registry
1153            .module(id)
1154            .filter(|m| m.source != crate::capability::registry::ModuleSource::Declarative)
1155        {
1156            Some(m) => {
1157                let capabilities: Vec<&Capability> = registry
1158                    .iter()
1159                    .filter(|c| {
1160                        c.module.as_str() == m.id.as_str()
1161                            && matches!(c.provenance, Provenance::Builtin { .. })
1162                    })
1163                    .collect();
1164                module_refs.push(ModuleRef {
1165                    id: m.id.to_string(),
1166                    title: m.title.clone(),
1167                    description: m.description.clone(),
1168                    stability: m.stability.map(|s| enum_word(&s)),
1169                    source_path: capabilities
1170                        .first()
1171                        .map(|c| c.provenance.source_path())
1172                        .unwrap_or_default(),
1173                    capabilities: capabilities.iter().map(|c| capability_ref(c)).collect(),
1174                });
1175            }
1176            None => unknown(
1177                "modules",
1178                id,
1179                "module",
1180                registry
1181                    .modules()
1182                    .filter(|m| m.source != crate::capability::registry::ModuleSource::Declarative)
1183                    .map(|m| m.id.as_str())
1184                    .collect(),
1185            ),
1186        }
1187    }
1188
1189    let mut command_refs = Vec::new();
1190    for id in &f.commands {
1191        match lookups.commands.get(id) {
1192            Some(o) => command_refs.push(CommandRef {
1193                id: id.clone(),
1194                summary: o.description.clone().unwrap_or_default(),
1195                stage: meta_str(o, "stage"),
1196                read_only: meta_str(o, "class").as_deref() == Some("read-only"),
1197            }),
1198            None => unknown(
1199                "commands",
1200                id,
1201                "command",
1202                lookups.commands.keys().map(String::as_str).collect(),
1203            ),
1204        }
1205    }
1206
1207    let mut kind_refs = Vec::new();
1208    for k in &f.kinds {
1209        match lookups.kinds.get(k) {
1210            Some(n) => kind_refs.push(KindRef {
1211                name: k.clone(),
1212                objects: *n,
1213            }),
1214            None => unknown(
1215                "kinds",
1216                k,
1217                "kind",
1218                lookups.kinds.keys().map(String::as_str).collect(),
1219            ),
1220        }
1221    }
1222
1223    let mut rule_refs = Vec::new();
1224    for id in &f.rules {
1225        match lookups.rules_by_stem.get(id) {
1226            Some(o) => rule_refs.push(RuleRef {
1227                id: id.clone(),
1228                identity: o.identity.clone(),
1229                title: o.title.clone().unwrap_or_else(|| id.clone()),
1230                class: meta_str(o, "class").unwrap_or_default(),
1231                enforced: o.metadata.get("x-majordomus").is_some(),
1232                path: o.provenance.path.clone(),
1233            }),
1234            None => unknown(
1235                "rules",
1236                id,
1237                "rule",
1238                lookups.rules_by_stem.keys().map(String::as_str).collect(),
1239            ),
1240        }
1241    }
1242
1243    let mut doc_refs = Vec::new();
1244    for p in &f.docs {
1245        match lookups.docs.get(p) {
1246            Some(o) => doc_refs.push(DocRef {
1247                path: p.clone(),
1248                title: o.title.clone(),
1249            }),
1250            None => unknown(
1251                "docs",
1252                p,
1253                "document",
1254                lookups.docs.keys().map(String::as_str).collect(),
1255            ),
1256        }
1257    }
1258
1259    let mut adr_refs = Vec::new();
1260    for id in &f.adrs {
1261        match lookups.adrs_by_id.get(id) {
1262            Some(o) => adr_refs.push(AdrRef {
1263                id: id.clone(),

1466            if m.status == crate::why::STABLE && !moment_ids.contains(&m.id) {

1468            }
1469        }
1470    };
1471    for c in &command_refs {
1472        add(why.naming("command", &c.id));
1473    }
1474    for c in &caps {

1497    });
1498

1511        adrs: adr_refs.len(),
1512        claims: claim_refs.len(),
1513        use_cases: use_case_refs.len(),

1551/// The providers, derived: one per template the distribution ships or declaration it
1552/// carries, decorated with what this repository's policy renders through it, the client
1553/// configuration it carries and the hooks the policy wires. The title, the client
1554/// configuration's name and the scratch roots are the declaration's (`share/providers.yaml`);
1555/// whether the configuration is present is a fact of the tree. Reads the policy as the index holds it — an object of kind
1556/// `policy` whose metadata is the whole file — and the repository root for the client
1557/// configurations, which are tracked files and therefore facts of the tree.
1558fn providers(index: &Index) -> Vec<ProductProvider> {
1559    let policy = index.objects.iter().find(|o| o.kind == "policy");
1560    let projections: Vec<(String, String, String, bool)> = policy
1561        .and_then(|p| p.metadata.get("projections"))
1562        .and_then(Value::as_array)
1563        .map(|a| {
1564            a.iter()
1565                .filter_map(|p| {
1566                    Some((
1567                        p.get("provider")?.as_str()?.to_string(),
1568                        p.get("target")?.as_str()?.to_string(),
1569                        p.get("mode")
1570                            .and_then(Value::as_str)
1571                            .unwrap_or("file")
1572                            .to_string(),
1573                        p.get("always_loaded")
1574                            .map(|v| v == "true" || v == &Value::Bool(true))
1575                            .unwrap_or(false),
1576                    ))
1577                })
1578                .collect()
1579        })
1580        .unwrap_or_default();
1581    let hooks: Vec<(String, String)> = policy
1582        .and_then(|p| p.metadata.get("enforcement"))
1583        .and_then(Value::as_array)
1584        .map(|a| {
1585            a.iter()
1586                .filter_map(|e| {
1587                    let wired = e.get("wired_by")?.as_str()?;
1588                    let provider = wired.strip_prefix("provider-hook:")?;
1589                    let provider = provider.split(':').next().unwrap_or(provider);
1590                    Some((provider.to_string(), e.get("name")?.as_str()?.to_string()))
1591                })
1592                .collect()
1593        })
1594        .unwrap_or_default();
1595    let root = Path::new(&index.repository.root);
1596    let mut out: Vec<ProductProvider> = index
1597        .providers
1598        .providers
1599        .iter()
1600        .map(|decl| {
1601            let id = &decl.id;
1602            let client_config = decl
1603                .client_config
1604                .as_deref()
1605                .filter(|c| root.join(c).is_file())
1606                .map(str::to_string);
1607            ProductProvider {
1608                id: id.clone(),
1609                title: decl.title.clone(),
1610                bootstraps: projections
1611                    .iter()
1612                    .filter(|(p, _, _, _)| p == id)
1613                    .map(|(_, target, mode, always)| ProviderBootstrap {
1614                        target: target.clone(),
1615                        mode: mode.clone(),
1616                        always_loaded: *always,
1617                    })
1618                    .collect(),
1619                client_config,
1620                hooks: hooks
1621                    .iter()
1622                    .filter(|(p, _)| p == id)
1623                    .map(|(_, n)| n.clone())
1624                    .collect(),
1625                scratch_roots: decl.scratch_roots.clone(),
1626            }
1627        })
1628        .collect();
1629    crate::order::canonical(&mut out);
1630    out
1631}
