1//! The "done" invariant: the questions a task must answer before anybody may call it
2//! finished, each answered from the one place that already knows.
3//!
4//! # Why this is a projection and not a checklist
5//!
6//! Nineteen questions get asked of every piece of work in this repository — is it tested,
7//! is it documented, are the projections current, did the version move, is it pushed, is it
8//! on the trunk, is it published, is CI green, was the deployment verified, is a stale
9//! branch left behind. Written as a checklist they would be nineteen new opinions, each
10//! able to disagree with the subsystem that already holds the fact, and a checklist is
11//! exactly the artifact a worker learns to tick.
12//!
13//! So every question here names a **source** and takes its answer from it:
14//!
15//! - an **obligation** of `share/obligations.yaml`, judged by
16//!   [`crate::capability::builtin::obligations`] — which is where `implemented`, `tested`,
17//!   `documented`, `generated`, `committed`, `pushed`, `integrated`, `published`,
18//!   `deployed` and `deployment-verified` come from. Nothing is re-judged here: the state
19//!   word comes back from that capability's own execution;
20//! - a **gate** of the CI model, judged by [`super::judge`] — which is where the CI verdict
21//!   and the release questions come from, when the model declares gates for them;
22//! - the **change set** itself, for the one question it settles: whether a test path is
23//!   among the files this task touched;
24//! - or **nothing here**, in which case the answer is [`super::GateStatus::Unknown`] and
25//!   the question carries the command that would answer it. An honest gap beats a boolean
26//!   conjured by a report, and this is where the gaps are written down rather than left for
27//!   a reader to infer.
28//!
29//! # The rule about absence
30//!
31//! No question here invents a `pass`. A question whose source has not spoken is `queued`
32//! (it was asked and nothing answered) or `unknown` (it could not be asked), never `pass`;
33//! a question the change makes irrelevant is `exempt`. Those three are different findings
34//! and a reader is entitled to all three.

43/// ```
44/// use majordomus_cli::gates::{DoneQuestion, GateStatus};
45///
46/// let q = DoneQuestion {
47///     id: "pushed".into(),
48///     question: "Has the commit reached the remote?".into(),
49///     status: GateStatus::Queued,
50///     evidence: "the obligation is owed".into(),
51///     source: "obligation push (obligations.closure)".into(),
52///     remediation: "git push".into(),
53/// };
54/// // every question names what answered it and what would settle it: a question with
55/// // neither is a checklist line, which is the artifact this type exists instead of
56/// assert!(!q.source.is_empty() && !q.remediation.is_empty());
57/// assert!(q.status.unverified());
58/// ```
59#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
60/// One question of the done invariant, and what answered it.
61pub struct DoneQuestion {
62    /// The question's identity, stable across reports.
63    pub id: String,
64    /// The question, as a person would ask it.
65    pub question: String,
66    /// The answer, in the same vocabulary every gate uses.
67    pub status: GateStatus,
68    /// What was actually read — never a restatement of the status.
69    pub evidence: String,
70    /// What answered it: an obligation token, a gate id, or the capability that would.
71    pub source: String,
72    /// What would settle it, as a command.
73    pub remediation: String,
74}
75
76/// How one question is answered.
77enum Answers {
78    /// An obligation token of `share/obligations.yaml`.
79    Obligation(&'static str),
80    /// The aggregate of the gates the change selects.
81    Gates,
82    /// A gate of the CI model, by id, when the model declares one.
83    Gate(&'static str),
84    /// The change set itself.
85    Change,
86    /// The repository's convergence: whether any work is held where it can be lost.
87    Convergence,
88    /// Nothing reachable from here; the command that would answer it.
89    Elsewhere(&'static str),
90}

216/// ```
217/// use majordomus_cli::gates::ObligationStanding;
218///
219/// // the closure's own word, carried verbatim; nothing here re-judges it
220/// let standing = ObligationStanding {
221///     state: "owed".into(),
222///     detail: "no evidence covers it".into(),
223///     reproduce: "majordomus evidence --covers tests --command 'just test'".into(),
224/// };
225/// assert_eq!(standing.state, "owed");
226/// ```
227/// The state of one obligation as [`crate::capability::builtin::obligations`] judged it:
228/// the token, its state word, and the line the validator would refuse with.
229#[derive(Debug, Clone, Default)]
230pub struct ObligationStanding {
231    /// The state word: `discharged`, `owed`, `stale` or `undeclared`.
232    pub state: String,
233    /// What the closure said about it, verbatim.
234    pub detail: String,
235    /// The command that would discharge it, as the closure gave it.
236    pub reproduce: String,
237}

239/// Answer the invariant.
240///
241/// Every argument is a judgement somebody else already made: `standing` is the obligation
242/// closure's, `gates` is [`super::judge`]'s, `implied` is [`super::implied`]'s,
243/// `changed` is the change set and `convergence` is [`crate::convergence`]'s verdict over
244/// the repository's holdings. Nothing is measured here — this composes.
245pub(crate) fn answer(
246    standing: &std::collections::BTreeMap<String, ObligationStanding>,
247    implied: &[ImpliedObligation],
248    gates: &[Gate],
249    changed: &[String],
250    closure_reachable: bool,
251    convergence: Option<&ConvergenceReport>,
252) -> Vec<DoneQuestion> {
253    invariant()
254        .into_iter()
255        .map(|(id, question, from, remediation)| {
256            let (status, evidence, source) = match from {
257                Answers::Obligation(token) => {
258                    let applicable = implied
259                        .iter()
260                        .find(|o| o.id == token)
261                        .map(|o| (o.applicable, o.declared, o.reason.clone()));
262                    match (standing.get(token), applicable) {
263                        // the closure judged it: that word is the answer, translated once
264                        (Some(s), _) => (
265                            match s.state.as_str() {
266                                "discharged" => GateStatus::Pass,
267                                "stale" => GateStatus::Stale,
268                                "owed" => GateStatus::Queued,
269                                _ => GateStatus::Unknown,
270                            },
271                            s.detail.clone(),
272                            format!("obligation {token} (obligations.closure)"),
273                        ),
274                        // the change implies it and the task never promised it: owed by the
275                        // change and held by nothing, which is a debt and not a pass
276                        (None, Some((true, false, why))) => (
277                            GateStatus::Queued,
278                            format!(
279                                "the change implies this obligation and the task does not declare it, \
280                                 so nothing holds it to it: {why}"
281                            ),
282                            format!("obligation {token} (derived from its own inputs)"),
283                        ),
284                        (None, Some((false, _, why))) => (
285                            GateStatus::Exempt,
286                            why,
287                            format!("obligation {token} (derived from its own inputs)"),
288                        ),
289                        _ if !closure_reachable => (
290                            GateStatus::Unknown,
291                            "the obligation closure could not be read in this checkout".into(),
292                            format!("obligation {token}"),
293                        ),
294                        _ => (
295                            GateStatus::Unknown,
296                            format!("share/obligations.yaml declares no token '{token}' here"),
297                            format!("obligation {token}"),
298                        ),
299                    }
300                }
301                Answers::Gates => {
302                    let required: Vec<&Gate> = gates.iter().filter(|g| g.required).collect();
303                    let refused = required.iter().filter(|g| g.status.refuses()).count();
304                    let silent = required.iter().filter(|g| g.status.unverified()).count();
305                    let passing = required
306                        .iter()
307                        .filter(|g| g.status == GateStatus::Pass)
308                        .count();
309                    let status = if required.is_empty() {
310                        GateStatus::Exempt
311                    } else if refused > 0 {
312                        GateStatus::Fail
313                    } else if silent > 0 {
314                        GateStatus::Queued
315                    } else {
316                        GateStatus::Pass
317                    };
318                    (
319                        status,
320                        format!(
321                            "{} gate(s) selected: {passing} passing, {refused} refusing, \
322                             {silent} never reported",
323                            required.len()
324                        ),
325                        ".ai/repo/ci/gates.yaml".to_string(),
326                    )
327                }
328                Answers::Gate(gate) => match gates.iter().find(|g| g.id == gate) {
329                    Some(g) => (
330                        g.status,
331                        g.reason.clone(),
332                        format!("gate {gate} (.ai/repo/ci/gates.yaml)"),
333                    ),
334                    None => (
335                        GateStatus::Unknown,
336                        format!("this repository's CI model declares no gate '{gate}'"),
337                        format!("gate {gate}"),
338                    ),
339                },
340                Answers::Change => {
341                    let tests: Vec<&String> = changed
342                        .iter()
343                        .filter(|p| is_test_path(p))
344                        .take(3)
345                        .collect();
346                    if changed.is_empty() {
347                        (
348                            GateStatus::Exempt,
349                            "the task has changed nothing".into(),
350                            "the change set".to_string(),
351                        )
352                    } else if tests.is_empty() {
353                        (
354                            GateStatus::Queued,
355                            format!(
356                                "no test path among the {} file(s) this task touched",
357                                changed.len()
358                            ),
359                            "the change set (profile verification.regression_test_required)"
360                                .to_string(),
361                        )
362                    } else {
363                        (
364                            GateStatus::Pass,
365                            format!(
366                                "{} test path(s) touched, including {}",
367                                tests.len(),
368                                tests[0]
369                            ),
370                            "the change set (profile verification.regression_test_required)"
371                                .to_string(),
372                        )
373                    }
374                }
375                Answers::Convergence => match convergence {
376                    // the verdict is one word over every holding; the evidence names the
377                    // holdings themselves, because "3 at risk" is not something a person
378                    // can act on and `feature/x, /tmp/wt` is
379                    Some(report) => {
380                        let at_risk: Vec<&str> = report
381                            .holdings
382                            .iter()
383                            .filter(|h| h.at_risk)
384                            .map(|h| h.identity.as_str())
385                            .take(3)
386                            .collect();
387                        if report.converged {
388                            (
389                                GateStatus::Pass,
390                                report.summary(),
391                                "convergence.report".to_string(),
392                            )
393                        } else {
394                            (
395                                GateStatus::Fail,
396                                format!("{}: {}", report.summary(), at_risk.join(", ")),
397                                "convergence.report".to_string(),
398                            )
399                        }
400                    }
401                    None => (
402                        GateStatus::Unknown,
403                        "the repository's holdings could not be read in this checkout".into(),
404                        "convergence.report".to_string(),
405                    ),
406                },
407                Answers::Elsewhere(command) => (
408                    GateStatus::Unknown,
409                    format!(
410                        "nothing in this report reaches that fact; `{command}` is what answers it"
411                    ),
412                    command.to_string(),
413                ),
414            };
415            DoneQuestion {
416                id: id.to_string(),
417                question: question.to_string(),
418                status,
419                evidence,
420                source,
421                remediation: remediation.to_string(),
422            }
423        })
424        .collect()
425}

427/// Is this a path a test lives at? `mj_validate_profile_requirements`' predicate, which is
428/// the one this repository already refuses over; nothing stricter is invented here.
429fn is_test_path(path: &str) -> bool {
430    let name = path.rsplit('/').next().unwrap_or(path);
431    path.split('/')
432        .any(|seg| matches!(seg, "test" | "tests" | "spec" | "specs"))
433        || name.contains("_test.")
434        || name.contains(".test.")
435        || name.contains("_spec.")
436        || name.contains(".spec.")
437}
