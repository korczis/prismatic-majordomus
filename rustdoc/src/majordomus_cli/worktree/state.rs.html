12/// Count the entries of `git status --porcelain -z` for a work tree. Untracked files are
13/// counted as files (`--untracked-files=all`), because a directory of untracked work is the
14/// thing a move must not lose and "1 untracked" for a directory of forty files would
15/// understate it.
16pub fn dirty_state(worktree: &Path) -> Result<DirtyState> {
17    let bytes = git::status_porcelain_z(worktree, "all")?;
18    let mut state = parse_status_z(&bytes);
19    state.in_progress = operation_in_progress(worktree);
20    Ok(state)
21}

23/// Parse the NUL-separated porcelain v1 status. Renames are disabled by the caller
24/// (`--no-renames`), so every entry is `XY path\0` and no entry is followed by a second
25/// path.
26///
27/// ```
28/// use majordomus_cli::worktree::state::parse_status_z;
29/// let s = parse_status_z(b"M  a\0 M b\0?? c\0UU d\0MM e\0");
30/// assert_eq!((s.staged, s.unstaged, s.untracked, s.conflicted), (2, 2, 1, 1));
31/// assert!(!s.clean);
32/// assert!(parse_status_z(b"").clean);
33/// ```
34pub fn parse_status_z(bytes: &[u8]) -> DirtyState {
35    let mut state = DirtyState::default();
36    for entry in bytes.split(|b| *b == 0).filter(|e| e.len() >= 3) {
37        let x = entry[0] as char;
38        let y = entry[1] as char;
39        match (x, y) {
40            ('?', '?') => state.untracked += 1,
41            ('!', '!') => {}
42            ('U', _) | (_, 'U') | ('A', 'A') | ('D', 'D') => state.conflicted += 1,
43            _ => {
44                if x != ' ' {
45                    state.staged += 1;
46                }
47                if y != ' ' {
48                    state.unstaged += 1;
49                }
50            }
51        }
52    }
53    state.clean =
54        state.staged == 0 && state.unstaged == 0 && state.untracked == 0 && state.conflicted == 0;
55    state
56}

58/// The operation git is in the middle of in this work tree, if any: what `git status`
59/// would report as "rebase in progress" and so on, read from the per-worktree git
60/// directory the way git itself does.
61pub fn operation_in_progress(worktree: &Path) -> Option<String> {
62    let out = git::try_run(worktree, &["rev-parse", "--absolute-git-dir"]).ok()?;
63    if out.status != Some(0) {
64        return None;
65    }
66    let dir = PathBuf::from(out.text().ok()?);
67    let probes: &[(&str, &str)] = &[
68        ("rebase-merge", "rebase"),
69        ("rebase-apply", "rebase"),
70        ("MERGE_HEAD", "merge"),
71        ("CHERRY_PICK_HEAD", "cherry-pick"),
72        ("REVERT_HEAD", "revert"),
73        ("BISECT_LOG", "bisect"),
74    ];
75    probes
76        .iter()
77        .find(|(file, _)| dir.join(file).exists())
78        .map(|(_, op)| (*op).to_string())
79}

81/// One local branch as `for-each-ref` reports it.
82#[derive(Debug, Clone, PartialEq, Eq)]
83pub struct BranchRef {
84    /// Short name.
85    pub name: String,
86    /// The commit.
87    pub head: String,
88    /// The upstream, when configured.
89    pub upstream: Option<UpstreamState>,
90    /// The work tree holding it, when one does.
91    pub worktree: Option<PathBuf>,
92}

94/// Every local branch, in one subprocess: name, commit, upstream with its ahead/behind
95/// distance, and the work tree holding it.
96pub fn branches(primary: &Path) -> Result<Vec<BranchRef>> {
97    let out = git::run(
98        primary,
99        &[
100            "for-each-ref",
101            "refs/heads",
102            "--format=%(refname:short)%00%(objectname)%00%(upstream:short)%00%(upstream:track,nobracket)%00%(worktreepath)%00",
103        ],
104    )?;
105    Ok(parse_branches(&String::from_utf8_lossy(&out.stdout)))
106}

108/// Parse the format above: five NUL-terminated fields per branch, one branch per line.
109///
110/// ```
111/// use majordomus_cli::worktree::state::parse_branches;
112/// let b = parse_branches("feature/x\0abc\0origin/feature/x\0ahead 2, behind 1\0/a/foo-wt/feature/x\0\nmain\0def\0\0\0/a/foo\0\n");
113/// assert_eq!(b.len(), 2);
114/// let u = b[0].upstream.as_ref().unwrap();
115/// assert_eq!((u.ahead, u.behind, u.gone), (Some(2), Some(1), false));
116/// assert!(b[1].upstream.is_none());
117/// ```
118pub fn parse_branches(text: &str) -> Vec<BranchRef> {
119    let mut out = Vec::new();
120    for line in text.split('\n') {
121        let fields: Vec<&str> = line.split('\0').collect();
122        if fields.len() < 5 || fields[0].is_empty() {
123            continue;
124        }
125        let upstream = if fields[2].is_empty() {
126            None
127        } else {
128            let track = fields[3].trim();
129            let mut ahead = None;
130            let mut behind = None;
131            let mut gone = false;
132            for part in track.split(',').map(str::trim) {
133                if part == "gone" {
134                    gone = true;
135                } else if let Some(n) = part.strip_prefix("ahead ") {
136                    ahead = n.trim().parse().ok();
137                } else if let Some(n) = part.strip_prefix("behind ") {
138                    behind = n.trim().parse().ok();
139                }
140            }
141            if !gone {
142                ahead = ahead.or(Some(0));
143                behind = behind.or(Some(0));
144            }
145            Some(UpstreamState {
146                name: fields[2].to_string(),
147                ahead,
148                behind,
149                gone,
150            })
151        };
152        out.push(BranchRef {
153            name: fields[0].to_string(),
154            head: fields[1].to_string(),
155            upstream,
156            worktree: if fields[4].is_empty() {
157                None
158            } else {
159                Some(PathBuf::from(fields[4]))
160            },
161        });
162    }
163    out
164}

166/// Every local branch reachable from `trunk`, in one subprocess.
167pub fn merged_into(primary: &Path, trunk: &str) -> Result<BTreeSet<String>> {
168    let out = git::run(
169        primary,
170        &[
171            "branch",
172            "--list",
173            "--merged",
174            trunk,
175            "--format=%(refname:short)",
176        ],
177    )?;
178    Ok(String::from_utf8_lossy(&out.stdout)
179        .lines()
180        .map(str::trim)
181        .filter(|l| !l.is_empty())
182        .map(str::to_string)
183        .collect())
184}

186/// The issue ids this repository's project model declares: the file names under
187/// `.ai/repo/project/issues/`, read from the primary checkout. A filesystem read, not an
188/// index build, so the topology stays cheap.
189pub fn issue_ids(primary: &Path) -> Vec<String> {
190    let dir = primary.join(".ai/repo/project/issues");
191    let Ok(entries) = std::fs::read_dir(dir) else {
192        return Vec::new();
193    };
194    let mut ids: Vec<String> = entries
195        .filter_map(|e| e.ok())
196        .filter_map(|e| {
197            let name = e.file_name().to_string_lossy().to_string();
198            name.strip_suffix(".yaml").map(str::to_string)
199        })
200        .filter(|id| !id.eq_ignore_ascii_case("readme"))
201        .collect();
202    crate::order::canonical(&mut ids);
203    ids
204}

206/// The issue a branch provably names: a path component equal to an issue id, or beginning
207/// with the id followed by `-`. Nothing else counts.
208///
209/// ```
210/// use majordomus_cli::worktree::state::issue_of;
211/// let ids = vec!["I0042".to_string(), "I1003".to_string()];
212/// assert_eq!(issue_of("feature/I1003-graph-kinds", &ids).as_deref(), Some("I1003"));
213/// assert_eq!(issue_of("fix/I0042", &ids).as_deref(), Some("I0042"));
214/// assert_eq!(issue_of("feature/graph-kinds", &ids), None);
215/// assert_eq!(issue_of("feature/I10030", &ids), None);
216/// ```
217pub fn issue_of(branch: &str, ids: &[String]) -> Option<String> {
218    for component in branch.split('/') {
219        for id in ids {
220            if component == id
221                || component
222                    .strip_prefix(id.as_str())
223                    .is_some_and(|rest| rest.starts_with('-'))
224            {
225                return Some(id.clone());
226            }
227        }
228    }
229    None
230}

238        let b = parse_branches("x\0abc\0origin/x\0gone\0\0\n");
