1//! Editors and agents as projections of the layer: the files Zed and Google Antigravity
2//! read from a project, derived from the canonical registries and from nothing else.
3//!
4//! # One core, two adapters
5//!
6//! [`EditorEnvironment`] is what the repository says that an editor or an agent can use —
7//! its skills, from the index, and its workflows, from the tracked justfile — each with the
8//! source it was read from and that source's fingerprint. An adapter ([`zed`],
9//! [`antigravity`]) declares what its product reads, with the evidence for each claim, and
10//! maps the environment onto the files it reads. Neither adapter discovers anything: adding
11//! a skill to `.ai/repo/skills/` or a recipe to the justfile changes what both project, and
12//! no list here names either.
13//!
14//! Where two products read the same file the projection is written once. Zed and
15//! Antigravity both discover `.agents/skills/<name>/SKILL.md`, so one artifact serves both
16//! and names both as its consumers.
17//!
18//! # What is never projected
19//!
20//! Personal settings, credentials, model choice, approval and permission modes, and anything
21//! whose project-level format could not be verified against the installed product. Those
22//! are reported as [`EditorSupport::RequiresUserAction`] or [`EditorSupport::Unknown`] with the evidence,
23//! never written as a guess (ADR 0140).
24//!
25//! # Ownership
26//!
27//! Every artifact carries the generated-file banner, and [`conflicts`] refuses to overwrite a
28//! file at a projected path that does not: a hand-written `.zed/tasks.json` or skill is the
29//! person's, and `majordomus generate` stops rather than taking it over. A banner-carrying
30//! file under a projected directory that the plan no longer contains is an [`orphans`] entry:
31//! generator-owned and stale, removed by `generate` and reported by `generate --check`.
32//!
33//! # Determinism and cost
34//!
35//! The projection is a pure function of the environment, which is ordered canonically on the
36//! way in, so enumeration order never reaches a byte. It is cheap enough — one index walk and
37//! one `just --dump` over a copy of the tracked justfile — that no persisted cache is kept;
38//! each artifact's [`EditorDerivation::key`] is the content address of its inputs, which is what
39//! `editors.status` explains and what a cache would be keyed by if one ever earned its place.

280/// The environment of the repository at `root`, as the index and the tracked justfile state it.
281pub fn environment(index: &Index, root: &Path) -> EditorEnvironment {
282    let mut env = EditorEnvironment::default();
283    for o in index.objects.iter().filter(|o| o.kind == SKILL_KIND) {
284        let status = o.metadata.get("status").and_then(Value::as_str);
285        if status.is_some_and(|s| s != "active") {
286            continue;
287        }
288        let id = o
289            .metadata
290            .get("id")
291            .and_then(Value::as_str)
292            .unwrap_or(&o.identity)
293            .to_string();
294        let sha256 = sha256_hex(&o.content);
295        env.inputs.push(EditorInput {
296            path: o.provenance.path.clone(),
297            mutability: EditorInputClass::Repository,
298            sha256: sha256.clone(),
299        });
300        env.skills.push(SkillRef {
301            title: o.title.clone().unwrap_or_else(|| id.clone()),
302            description: o.description.clone().unwrap_or_default(),
303            path: o.provenance.path.clone(),
304            id,
305            sha256,
306        });
307    }
308    crate::order::canonical(&mut env.skills);
309
310    match tracked_workflows(root) {
311        Some((sources, workflows, groups)) => {
312            env.workflows_resolved = Some(true);
313            env.inputs.extend(sources);
314            env.tasks = tasks_of(&workflows, &groups, &mut env.diagnostics);
315        }
316        None if root.join("justfile").is_file() => {
317            env.workflows_resolved = Some(false);
318            env.diagnostics.push(EditorDiagnostic {
319                severity: Severity::Warning,
320                code: "workflows-unreadable".into(),
321                adapter: Some("zed".into()),
322                path: Some("justfile".into()),
323                message: "the justfile exists and `just --dump` could not describe it, so no task is projected".into(),
324                remedy: "install `just` (the CI runner has it) and run `just --dump` to see why it refuses".into(),
325            });
326        }
327        None => env.workflows_resolved = None,
328    }
329    crate::order::canonical(&mut env.inputs);
330    env.inputs.dedup_by(|a, b| a.path == b.path);
331    env
332}

334/// The recipes of the *tracked* justfile: `justfile` and `.just/*.just`, read through `just`
335/// from a private copy. The root justfile also imports `.ai/local/cache/command-graph/
336/// bridge.just`, which each checkout materialises for itself; read in place, a checkout that
337/// has the bridge would project a different file than one that has not, and a committed
338/// projection would be stale on every machine but the one that wrote it.
339fn tracked_workflows(
340    root: &Path,
341) -> Option<(Vec<EditorInput>, Vec<WorkflowDescriptor>, Vec<String>)> {
342    let justfile = root.join("justfile");
343    if !justfile.is_file() {
344        return None;
345    }
346    let mut files: Vec<PathBuf> = vec![PathBuf::from("justfile")];
347    if let Ok(entries) = std::fs::read_dir(root.join(".just")) {
348        let mut more: Vec<String> = entries
349            .filter_map(|e| e.ok())
350            .map(|e| e.path())
351            .filter(|p| p.extension().is_some_and(|x| x == "just") && p.is_file())
352            .filter_map(|p| {
353                p.strip_prefix(root)
354                    .ok()
355                    .map(|r| r.to_string_lossy().replace('\\', "/"))
356            })
357            .collect();
358        crate::order::canonical_strings(&mut more);
359        files.extend(more.into_iter().map(PathBuf::from));
360    }
361    let scratch = Scratch::new()?;
362    let mut inputs = Vec::with_capacity(files.len());
363    for rel in &files {
364        let text = std::fs::read_to_string(root.join(rel)).ok()?;
365        let dest = scratch.path.join(rel);
366        if let Some(dir) = dest.parent() {
367            std::fs::create_dir_all(dir).ok()?;
368        }
369        std::fs::write(&dest, &text).ok()?;
370        inputs.push(EditorInput {
371            path: rel.to_string_lossy().replace('\\', "/"),
372            mutability: EditorInputClass::Repository,
373            sha256: sha256_hex(&text),
374        });
375    }
376    let catalogue = workflows::resolve(&scratch.path);
377    if catalogue.state != crate::environment::TierState::Resolved {
378        return None;
379    }
380    let groups = workflows::groups(&scratch.path).unwrap_or_default();
381    Some((inputs, catalogue.workflows, groups))
382}

384/// A directory of the process's own under the system's temporary directory, removed when
385/// dropped.
386struct Scratch {
387    path: PathBuf,
388}
389
390impl Scratch {
391    fn new() -> Option<Scratch> {
392        let nanos = std::time::SystemTime::now()
393            .duration_since(std::time::UNIX_EPOCH)
394            .map(|d| d.as_nanos())
395            .unwrap_or(0);
396        let path =
397            std::env::temp_dir().join(format!("majordomus-editors-{}-{nanos}", std::process::id()));
398        std::fs::create_dir_all(&path).ok()?;
399        Some(Scratch { path })
400    }
401}
402
403impl Drop for Scratch {
404    fn drop(&mut self) {
405        let _ = std::fs::remove_dir_all(&self.path);
406    }
407}

409/// A recipe name `just` accepts and no shell reads as anything but a word.
410fn is_plain_recipe(name: &str) -> bool {
411    !name.is_empty()
412        && name
413            .chars()
414            .all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '_' || c == ':')
415}
416
417/// Every public recipe a person may run with one keystroke: not one that asks before it runs
418/// (`[confirm]` is the justfile's own statement that it is dangerous), not one whose name is
419/// anything but a plain word, and not one that needs an argument it has no default for.
420fn tasks_of(
421    workflows: &[WorkflowDescriptor],
422    groups: &[String],
423    diagnostics: &mut Vec<EditorDiagnostic>,
424) -> Vec<TaskRef> {
425    let mut out = Vec::new();
426    for w in workflows {
427        let recipe = match &w.namespace {
428            Some(ns) if !ns.is_empty() => format!("{ns}::{}", w.name),
429            _ => w.name.clone(),
430        };
431        if w.confirm || w.parameters.iter().any(|p| p.required) {
432            continue;
433        }
434        if !is_plain_recipe(&recipe) {
435            diagnostics.push(EditorDiagnostic {
436                severity: Severity::Warning,
437                code: "task-name-refused".into(),
438                adapter: Some("zed".into()),
439                path: Some(".zed/tasks.json".into()),
440                message: format!(
441                    "the recipe {recipe:?} is not a plain name and is not projected as a task"
442                ),
443                remedy: "name the recipe with letters, digits, `-` and `_` only".into(),
444            });
445            continue;
446        }
447        let group_rank = w
448            .group
449            .as_ref()
450            .and_then(|g| groups.iter().position(|x| x == g))
451            .unwrap_or(usize::MAX);
452        out.push(TaskRef {
453            recipe,
454            group: w.group.clone(),
455            group_rank,
456        });
457    }
458    crate::order::canonical(&mut out);
459    out
460}

464/// The content address of a derivation: the projection version, the document, and the
465/// fingerprints of its sources in path order. Independent of the generator's own version,
466/// which the banner states and which changes no rule.
467pub fn derivation_key(document: &str, sources: &[&EditorInput]) -> String {
468    let mut sorted: Vec<&EditorInput> = sources.to_vec();
469    crate::order::canonical(&mut sorted);
470    let mut text = format!("editors/v{PROJECTION_VERSION}\n{document}\n");
471    for i in sorted {
472        text.push_str(&format!("{}\t{}\n", i.path, i.sha256));
473    }
474    sha256_hex(&text)
475}

477/// Every artifact both adapters read, every finding, every capability — as data.
478pub fn plan(index: &Index, root: &Path, version: &str) -> Plan {
479    let env = environment(index, root);
480    let mut diagnostics = env.diagnostics.clone();
481    let mut artifacts: Vec<(Artifact, Vec<String>, Vec<String>)> = Vec::new();
482
483    for skill in &env.skills {
484        match skill_artifact(skill, version) {
485            Ok(a) => artifacts.push((
486                a,
487                vec!["antigravity".into(), "zed".into()],
488                vec![skill.path.clone()],
489            )),
490            Err(d) => diagnostics.push(*d),
491        }
492    }
493    if let Some(a) = zed::tasks_artifact(&env, version) {
494        let sources = env
495            .inputs
496            .iter()
497            .filter(|i| i.path == "justfile" || i.path.starts_with(".just/"))
498            .map(|i| i.path.clone())
499            .collect();
500        artifacts.push((a, vec!["zed".into()], sources));
501    }
502    // in path order by construction: every skill (canonical by identity) under
503    // .agents/, then the task list under .zed/
504
505    diagnostics.extend(description_findings(&env.skills));
506    diagnostics.extend(checkout_findings(root));
507    sort_diagnostics(&mut diagnostics);
508
509    let derivations = artifacts
510        .iter()
511        .map(|(a, adapters, sources)| {
512            let inputs: Vec<&EditorInput> = env
513                .inputs
514                .iter()
515                .filter(|i| sources.contains(&i.path))
516                .collect();
517            EditorDerivation {
518                path: a.path.clone(),
519                document: a.document.clone(),
520                adapters: adapters.clone(),
521                sources: sources.clone(),
522                key: derivation_key(&a.document, &inputs),
523                sha256: sha256_hex(&a.content),
524                state: state_of(root, a),
525            }
526        })
527        .collect();
528    let adapters = vec![antigravity::adapter(&env), zed::adapter(&env)];
529    Plan {
530        artifacts: artifacts.into_iter().map(|(a, _, _)| a).collect(),
531        derivations,
532        adapters,
533        inputs: env.inputs,
534        diagnostics,
535    }
536}

538/// The report `editors.status` answers with.
539pub fn report(index: &Index, root: &Path, version: &str) -> EditorProjection {
540    let plan = plan(index, root, version);
541    let orphans = orphans(root, &plan.artifacts);
542    EditorProjection {
543        schema: SCHEMA.into(),
544        projection_version: PROJECTION_VERSION,
545        adapters: plan.adapters,
546        inputs: plan.inputs,
547        artifacts: plan.derivations,
548        orphans,
549        diagnostics: plan.diagnostics,
550    }
551}

571// By the position of the recipe's group in the justfile's declaration, then by name: the
572// order the justfile already shows a person. A recipe in no declared group comes last.
573impl crate::order::Ordered for TaskRef {
574    fn order_key(&self) -> crate::order::OrderKey<'_> {
575        let rank = i64::try_from(self.group_rank).unwrap_or(i64::MAX);
576        crate::order::OrderKey::plain(&self.recipe, &self.recipe).ranked(rank)
577    }
578}

580// Errors first, then warnings, then information; inside a severity by code, then path.
581impl crate::order::Ordered for EditorDiagnostic {
582    fn order_key(&self) -> crate::order::OrderKey<'_> {
583        let rank = match self.severity {
584            Severity::Error => 0,
585            Severity::Warning => 1,
586            Severity::Info => 2,
587        };
588        crate::order::OrderKey::plain(&self.code, self.path.as_deref().unwrap_or("")).ranked(rank)
589    }
590}

592/// A skill name both products accept: lowercase letters, digits and single hyphens, at most
593/// 64 characters (Zed's limit; Antigravity documents the same shape and no length).
594pub fn is_skill_name(name: &str) -> bool {
595    name.len() <= 64
596        && !name.is_empty()
597        && !name.starts_with('-')
598        && !name.ends_with('-')
599        && !name.contains("--")
600        && name
601            .chars()
602            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-')
603}

608/// The shared skill projection: the front matter both products require — `name` and
609/// `description` — the banner, and a pointer to the one definition. The procedure itself is
610/// never copied: it lives in the canonical file, which the agent is told to read.
611pub fn skill_artifact(skill: &SkillRef, version: &str) -> Result<Artifact, Box<EditorDiagnostic>> {
612    let path = format!("{SKILLS_DIR}/{}/SKILL.md", skill.id);
613    if !is_skill_name(&skill.id) {
614        return Err(Box::new(EditorDiagnostic {
615            severity: Severity::Warning,
616            code: "skill-name-refused".into(),
617            adapter: None,
618            path: Some(skill.path.clone()),
619            message: format!(
620                "the skill {:?} is not lowercase letters, digits and hyphens (at most 64), which both products require; it is not projected",
621                skill.id
622            ),
623            remedy: "rename the skill's `id` to a lowercase, hyphenated name".into(),
624        }));
625    }
626    let description = if skill.description.trim().is_empty() {
627        skill.title.clone()
628    } else {
629        skill.description.trim().to_string()
630    };
631    let one_line = description.split_whitespace().collect::<Vec<_>>().join(" ");
632    let quoted = serde_json::to_string(&one_line).unwrap_or_else(|_| "\"\"".into());
633    let [a, b, c] = banner_lines(&skill.path, version);
634    let depth = path.matches('/').count();
635    let up = "../".repeat(depth);
636    let content = format!(
637        "---\nname: {name}\ndescription: {quoted}\n---\n<!-- {a}\n     {b}\n     {c} -->\n\n\
638         # {title}\n\n\
639         This skill is defined once, in [`{src}`]({up}{src}). Read that file and follow it: it is \
640         the procedure, its inputs and what it must produce. This file only makes the skill \
641         discoverable to agents that load `{SKILLS_DIR}/`, and says nothing the definition does \
642         not.\n\n\
643         The same definition is served as `majordomus skills explain {name}` and as the MCP \
644         resource `majordomus://skill/{name}`.\n",
645        name = skill.id,
646        title = skill.title,
647        src = skill.path,
648    );
649    Ok(Artifact::verbatim(
650        path,
651        format!("{DOCUMENT_PREFIX}skill"),
652        ArtifactFormat::Markdown,
653        None,
654        skill.path.clone(),
655        content,
656    ))
657}

679/// The `//` banner a JSONC artifact opens with: Zed reads every `.zed/**/*.json` as JSON
680/// with comments, and a task list is an array, which has no member to carry provenance.
681pub fn jsonc_banner(source: &str, version: &str) -> String {
682    banner_lines(source, version)
683        .iter()
684        .map(|l| format!("// {l}\n"))
685        .collect()
686}
687
688/// The JSON below a JSONC banner, when the content opens with one.
689pub fn jsonc_body(content: &str) -> Option<&str> {
690    if !content.starts_with(&format!("// {HEADER}")) {
691        return None;
692    }
693    let mut rest = content;
694    while rest.starts_with("//") {
695        rest = rest.split_once('\n').map(|(_, r)| r).unwrap_or("");
696    }
697    Some(rest)
698}
699
700/// Whether an editor artifact carries its banner in the form its file allows: after the
701/// front matter for a skill, as `//` lines for JSONC, first for anything else.
702pub fn banner_is_present(content: &str) -> bool {
703    if jsonc_body(content).is_some() {
704        return true;
705    }
706    let after_front_matter = content
707        .strip_prefix("---\n")
708        .and_then(|r| r.split_once("\n---\n").map(|(_, rest)| rest))
709        .unwrap_or(content);
710    after_front_matter.starts_with(&format!("<!-- {HEADER}"))
711}

713/// The first lines of a file, enough to find a banner after a front matter.
714fn head(path: &Path) -> Option<String> {
715    use std::io::Read;
716    let mut buf = vec![0u8; 4096];
717    let mut f = std::fs::File::open(path).ok()?;
718    let n = f.read(&mut buf).ok()?;
719    buf.truncate(n);
720    Some(String::from_utf8_lossy(&buf).into_owned())
721}
722
723fn state_of(root: &Path, a: &Artifact) -> EditorFileState {
724    let path = root.join(&a.path);
725    match std::fs::symlink_metadata(&path) {
726        Err(_) => EditorFileState::Absent,
727        Ok(m) if !m.file_type().is_file() => EditorFileState::Conflict,
728        Ok(_) => match std::fs::read_to_string(&path) {
729            Ok(on_disk) if on_disk == a.content => EditorFileState::Current,
730            Ok(on_disk) if banner_is_present(&on_disk) => EditorFileState::Stale,
731            _ => EditorFileState::Conflict,
732        },
733    }
734}

736/// Every projected path where a file sits that this generator does not own: one without the
737/// banner, a symbolic link, a directory. `majordomus generate` refuses to write while any
738/// remains, and names each with what to do.
739pub fn conflicts(root: &Path, artifacts: &[Artifact]) -> Vec<String> {
740    let mut out = Vec::new();
741    for a in artifacts
742        .iter()
743        .filter(|a| a.document.starts_with(DOCUMENT_PREFIX))
744    {
745        if escapes(root, &a.path) {
746            out.push(format!("{} (outside the repository)", a.path));
747            continue;
748        }
749        if state_of(root, a) == EditorFileState::Conflict {
750            out.push(a.path.clone());
751        }
752    }
753    out
754}
755
756/// Whether any existing ancestor of `rel` under `root` is a symbolic link, which would carry
757/// a write somewhere the repository does not decide.
758fn escapes(root: &Path, rel: &str) -> bool {
759    if rel.starts_with('/') || rel.split('/').any(|s| s == ".." || s.is_empty()) {
760        return true;
761    }
762    let mut at = root.to_path_buf();
763    let parts: Vec<&str> = rel.split('/').collect();
764    for part in &parts[..parts.len().saturating_sub(1)] {
765        at.push(part);
766        match std::fs::symlink_metadata(&at) {
767            Ok(m) if m.file_type().is_symlink() => return true,
768            Ok(_) => {}
769            Err(_) => break,
770        }
771    }
772    false
773}

775/// Generator-owned skill projections the plan no longer contains: a skill was retired or
776/// renamed and its projection remained. Only a `SKILL.md` that carries the banner is ours.
777pub fn orphans(root: &Path, artifacts: &[Artifact]) -> Vec<String> {
778    let planned: std::collections::BTreeSet<&str> =
779        artifacts.iter().map(|a| a.path.as_str()).collect();
780    let mut out = Vec::new();
781    let Ok(entries) = std::fs::read_dir(root.join(SKILLS_DIR)) else {
782        return out;
783    };
784    for e in entries.filter_map(|e| e.ok()) {
785        let rel = format!("{SKILLS_DIR}/{}/SKILL.md", e.file_name().to_string_lossy());
786        if planned.contains(rel.as_str()) {
787            continue;
788        }
789        let path = root.join(&rel);
790        let is_file = std::fs::symlink_metadata(&path).is_ok_and(|m| m.file_type().is_file());
791        if is_file && head(&path).is_some_and(|h| banner_is_present(&h)) {
792            out.push(rel);
793        }
794    }
795    crate::order::canonical_strings(&mut out);
796    out
797}
798
799/// Remove every orphan and the directory it leaves empty. Returns what was removed.
800pub fn remove_orphans(root: &Path, orphans: &[String]) -> std::io::Result<Vec<String>> {
801    for rel in orphans {
802        let path = root.join(rel);
803        std::fs::remove_file(&path)?;
804        if let Some(dir) = path.parent() {
805            // only when empty: anything else in it is someone's
806            let _ = std::fs::remove_dir(dir);
807        }
808    }
809    Ok(orphans.to_vec())
810}

812/// Facts of this checkout an adapter acts on differently. None changes a projected byte.
813fn checkout_findings(root: &Path) -> Vec<EditorDiagnostic> {
814    let mut out = Vec::new();
815    out.extend(zed::instruction_findings(root));
816    out.extend(antigravity::legacy_findings(root));
817    out
