1//! Toolchains: the ones this repository declares, and what is installed for them.
2//!
3//! Relevance is decided by the repository, not by the machine. A toolchain appears in the
4//! snapshot when a file in the checkout declares it — `Cargo.toml` for Rust, `package.json`
5//! for Node — and never because the executable happens to be on the path: a laptop with
6//! nine language runtimes installed would otherwise report all nine for every repository,
7//! and the one line that matters would be lost among them.
8//!
9//! Declared, installed and available are three different facts and are kept apart. A
10//! repository can ask for a version nobody has; a machine can have a version nobody asked
11//! for; and the interesting case — they disagree — is only visible when both are reported.
12//!
13//! Asking a version costs a process spawn each, so it happens in a full resolution only;
14//! a fast resolution takes the answers from the cache and says `unknown` when it has none.
15

22/// How a declared version is read out of the file that declares it. Small on purpose:
23/// three shapes cover every manifest here, and a fourth is a new variant rather than a
24/// parser.
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26pub enum Declares {
27    /// The whole file is the version, comments and blank lines skipped (`.nvmrc`).
28    WholeFile,
29    /// A `key = "value"` or `key: value` line at any indentation (`rust-version`).
30    KeyValue(&'static str),
31    /// A member of a JSON object, one level deep (`engines` → `node`).
32    JsonMember(&'static str, &'static str),
33    /// A `go 1.22` style directive at the start of a line.
34    Directive(&'static str),
35    /// The file marks the toolchain relevant but declares no version.
36    Nothing,
37}
38
39/// One place a toolchain may be declared.
40#[derive(Debug, Clone, Copy)]
41pub struct Marker {
42    /// Repository-relative path. Exactly one segment may be `*`, resolved with a single
43    /// directory read; nothing here walks a tree.
44    pub path: &'static str,
45    /// How to read the version out of it.
46    pub declares: Declares,
47}

49/// One toolchain this executable knows how to recognise.
50#[derive(Debug, Clone, Copy)]
51pub struct Detector {
52    /// A stable id, `[a-z][a-z0-9-]*`.
53    pub id: &'static str,
54    /// The short name a person reads.
55    pub title: &'static str,
56    /// The executable whose version is asked, in a full resolution.
57    pub executable: &'static str,
58    /// The arguments that make it print its version.
59    pub version_args: &'static [&'static str],
60    /// Where it may be declared, most specific first; the first that exists decides.
61    pub markers: &'static [Marker],
62}

178/// How long a version subprocess may take before it is abandoned. A toolchain shim that
179/// downloads a release on first use has been seen to take minutes; a snapshot reports
180/// `unknown` rather than waiting for it.
181pub const VERSION_TIMEOUT: Duration = Duration::from_millis(1500);
182
183/// The toolchains this repository declares, without asking any of them their version.
184/// Cheap enough for the hot path: one `exists()` per marker, one directory read for a
185/// marker with a `*`, and one small file read for each that matched.
186pub fn declared(root: &Path) -> Vec<ToolchainState> {
187    let mut out = Vec::new();
188    for detector in DETECTORS {
189        let Some((path, declares)) = first_marker(root, detector) else {
190            continue;
191        };
192        let declared = read_declared(&root.join(&path), declares);
193        out.push(ToolchainState {
194            id: detector.id.into(),
195            title: detector.title.into(),
196            declared,
197            declared_by: path,
198            installed: None,
199            availability: ToolchainAvailability::Unknown,
200        });
201    }
202    out
203}

205/// Ask every declared toolchain what is installed. One bounded subprocess each; a
206/// toolchain that is not on the path is [`ToolchainAvailability::Missing`], and one that
207/// does not answer within [`VERSION_TIMEOUT`] stays `Unknown`.
208pub fn with_installed(mut toolchains: Vec<ToolchainState>) -> Vec<ToolchainState> {
209    for state in &mut toolchains {
210        let Some(detector) = DETECTORS.iter().find(|d| d.id == state.id) else {
211            continue;
212        };
213        match installed_version(detector) {
214            Some(version) => {
215                state.installed = Some(version);
216                state.availability = ToolchainAvailability::Installed;
217            }
218            None => {
219                state.availability = ToolchainAvailability::Missing;
220            }
221        }
222    }
223    toolchains
224}

226/// The version one toolchain reports, or `None` when it is not there or did not answer.
227fn installed_version(detector: &Detector) -> Option<String> {
228    let out = crate::environment::probe::bounded_output(
229        Command::new(detector.executable).args(detector.version_args),
230        VERSION_TIMEOUT,
231    )?;
232    if !out.status.success() {
233        return None;
234    }
235    let text = String::from_utf8_lossy(&out.stdout);
236    let line = text.lines().find(|l| !l.trim().is_empty())?.trim();
237    // `rustc 1.90.0 (1159e78c4 2025-09-14)` -> `1.90.0`; `v22.20.0` -> `22.20.0`;
238    // anything without a recognisable version number is reported as it came.
239    Some(
240        line.split_whitespace()
241            .find_map(|word| {
242                let candidate = word.trim_start_matches('v');
243                let looks_like_a_version = candidate.split('.').all(|part| {
244                    !part.is_empty() && part.chars().next().is_some_and(|c| c.is_ascii_digit())
245                }) && candidate.contains('.');
246                looks_like_a_version.then(|| candidate.to_string())
247            })
248            .unwrap_or_else(|| line.to_string()),
249    )
250}

252/// The first marker of a detector that exists, with the path it resolved to.
253fn first_marker(root: &Path, detector: &Detector) -> Option<(String, Declares)> {
254    for marker in detector.markers {
255        match marker.path.split_once('*') {
256            None => {
257                if root.join(marker.path).is_file() {
258                    return Some((marker.path.to_string(), marker.declares));
259                }
260            }
261            Some((before, after)) => {
262                // exactly one `*`, standing for one directory name: one read_dir, sorted,
263                // so the answer does not depend on the order the filesystem hands them back
264                let dir = root.join(before.trim_end_matches('/'));
265                let mut names: Vec<String> = std::fs::read_dir(&dir)
266                    .into_iter()
267                    .flatten()
268                    .flatten()
269                    .filter_map(|e| e.file_name().into_string().ok())
270                    .collect();
271                names.sort();
272                for name in names {
273                    let rel = format!("{before}{name}{after}");
274                    if root.join(&rel).is_file() {
275                        return Some((rel, marker.declares));
276                    }
277                }
278            }
279        }
280    }
281    None
282}

284/// Read the declared version out of one marker file.
285///
286/// ```
287/// use majordomus_cli::environment::toolchain::{read_text, Declares};
288/// assert_eq!(read_text("# a comment\n\n22.20.0\n", Declares::WholeFile).as_deref(), Some("22.20.0"));
289/// assert_eq!(read_text("rust-version = \"1.85\"\n", Declares::KeyValue("rust-version")).as_deref(), Some("1.85"));
290/// assert_eq!(read_text("{\"engines\":{\"node\":\">=22\"}}", Declares::JsonMember("engines", "node")).as_deref(), Some(">=22"));
291/// assert_eq!(read_text("module x\n\ngo 1.22\n", Declares::Directive("go")).as_deref(), Some("1.22"));
292/// assert_eq!(read_text("anything", Declares::Nothing), None);
293/// ```
294pub fn read_text(text: &str, declares: Declares) -> Option<String> {
295    match declares {
296        Declares::Nothing => None,
297        Declares::WholeFile => text
298            .lines()
299            .map(str::trim)
300            .find(|l| !l.is_empty() && !l.starts_with('#'))
301            .map(str::to_string),
302        Declares::KeyValue(key) => text.lines().find_map(|line| {
303            let line = line.trim();
304            let rest = line.strip_prefix(key)?;
305            let rest = rest.trim_start();
306            let rest = rest.strip_prefix('=').or_else(|| rest.strip_prefix(':'))?;
307            let value = rest.trim().trim_matches(['"', '\'']).trim();
308            (!value.is_empty()).then(|| value.to_string())
309        }),
310        Declares::JsonMember(object, member) => {
311            let value: serde_json::Value = serde_json::from_str(text).ok()?;
312            value.get(object)?.get(member)?.as_str().map(str::to_string)
313        }
314        Declares::Directive(word) => text.lines().find_map(|line| {
315            let rest = line.strip_prefix(word)?;
316            let value = rest.trim();
317            (!value.is_empty() && value.chars().next().is_some_and(|c| c.is_ascii_digit()))
318                .then(|| value.to_string())
319        }),
320    }
321}
322
323fn read_declared(path: &Path, declares: Declares) -> Option<String> {
324    if matches!(declares, Declares::Nothing) {
325        return None;
326    }
327    let text = std::fs::read_to_string(path).ok()?;
328    read_text(&text, declares)
329}

335    fn tree(files: &[(&str, &str)]) -> tempfile::TempDir {
336        let dir = tempfile::tempdir().expect("a temporary directory");
337        for (path, content) in files {
338            let full = dir.path().join(path);
339            std::fs::create_dir_all(full.parent().expect("a parent")).expect("a directory");
340            std::fs::write(full, content).expect("a file");
341        }
342        dir
343    }

345    /// The point of the whole module: a toolchain the repository does not declare does not
346    /// appear, however many of them the machine has installed.
347    #[test]
348    fn only_declared_toolchains_appear() {
349        let dir = tree(&[("package.json", "{}")]);
350        let found = declared(dir.path());
351        let ids: Vec<&str> = found.iter().map(|t| t.id.as_str()).collect();
352        assert_eq!(ids, ["node"], "a repository with only a package.json");
353        assert!(
354            !ids.contains(&"rust"),
355            "rustc is installed on this machine and must still not appear"
356        );
357    }

361        let dir = tree(&[
362            ("rust-toolchain.toml", "[toolchain]\nchannel = \"1.90.0\"\n"),
363            ("Cargo.toml", "[package]\nrust-version = \"1.85\"\n"),
364        ]);
365        let found = declared(dir.path());
366        assert_eq!(found[0].declared.as_deref(), Some("1.90.0"));
