1//! The `served` module: whether a deployment serves the commit it was meant to, observed
2//! from outside and recorded as evidence a later reader can re-judge.
3//!
4//! Three operations, deliberately unequal:
5//!
6//! - `served.observe` reaches the network and appends a record, so it is a command of the
7//!   trusted command line only. A browser or an MCP client that could make this process
8//!   probe an arbitrary URL would have been handed a request forger.
9//! - `served.forge` asks the forge which deployments and releases it recorded, through the
10//!   forge's own tool, and keeps the answer for `deploy.observed` to read. It reaches the
11//!   network and spends the caller's credential, so it too is a command line and nothing
12//!   else.
13//! - `served.show` is a read: it asks no network, writes nothing and re-judges what was
14//!   recorded against the commit asked about, which is what makes an old observation go
15//!   stale without anyone deciding it has (see [`crate::served`]).
16//!
17//! ```
18//! use majordomus_cli::capability::builtin::served;
19//! use majordomus_cli::capability::model::CapabilityKind;
20//!
21//! let m = served::module();
22//! let observe = m.capabilities.iter().find(|c| c.capability.id.as_str() == "served.observe").unwrap();
23//! // the operation that reaches the network is on no surface a client can call
24//! assert_eq!(observe.capability.kind, CapabilityKind::Command);
25//! assert!(observe.capability.exposure.http.is_none());
26//! assert!(observe.capability.exposure.mcp.is_none());
27//!
28//! let show = m.capabilities.iter().find(|c| c.capability.id.as_str() == "served.show").unwrap();
29//! assert_eq!(show.capability.exposure.http.as_ref().unwrap().path, "/api/v1/served");
30//! ```

57/// The input of `served.observe`: which deployment, where it is, which commit is expected
58/// and how long the probe may take. Everything is optional — the defaults are this
59/// repository's own site, its `build.json` and `HEAD`.
60///
61/// ```
62/// use majordomus_cli::capability::builtin::served::ObserveInput;
63/// let input: ObserveInput = serde_json::from_str(r#"{"commit":"HEAD"}"#).unwrap();
64/// assert_eq!(input.commit.as_deref(), Some("HEAD"));
65/// assert!(serde_json::from_str::<ObserveInput>(r#"{"shell":"rm"}"#).is_err());
66/// ```
67#[derive(Debug, Default, Serialize, Deserialize, JsonSchema)]
68#[serde(deny_unknown_fields)]
69pub struct ObserveInput {
70    /// The name the observation is recorded under; `pages` when absent.
71    #[serde(default, skip_serializing_if = "Option::is_none")]
72    pub deployment: Option<String>,
73    /// The base URL; when absent, the `provider.url` of the deployment object `deployment`
74    /// names, else `base_url` of `site/config.toml` (I3822).
75    #[serde(default, skip_serializing_if = "Option::is_none")]
76    pub url: Option<String>,
77    /// The identity under the base; when absent, the `health.liveness` route of the
78    /// deployment object `deployment` names (which answers `commit` and `dirty`), else
79    /// `build.json`.
80    #[serde(default, skip_serializing_if = "Option::is_none")]
81    pub identity: Option<String>,
82    /// The commit expected to be served, any revision git resolves; `HEAD` when absent.
83    #[serde(default, skip_serializing_if = "Option::is_none")]
84    pub commit: Option<String>,
85    /// The bound on the probe in seconds; ten when absent.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub timeout_seconds: Option<u64>,
88    /// Judge without recording. Recording is the default, because an observation nobody
89    /// kept is a measurement nobody can read.
90    #[serde(default)]
91    pub dry_run: bool,
92}

94/// What `served.observe` did: the observation, where it was recorded, and the exit status
95/// the command line reports the verdict with.
96///
97/// ```
98/// use majordomus_cli::capability::builtin::served::Observed;
99/// // the shape a caller reads: a dry run judges and records nothing
100/// let json = r#"{"observation":{"schema":"majordomus.served-observation/v1",
101///   "deployment":"pages","url":"https://majordomus.dev/build.json",
102///   "expected":"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","verdict":"unreachable",
103///   "reason":"nothing was received","at":"1970-01-01T00:00:00Z"},"exit_code":12}"#;
104/// let got: Observed = serde_json::from_str(json).unwrap();
105/// assert_eq!(got.exit_code, 12);
106/// assert!(got.recorded.is_none());
107/// ```
108#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
109pub struct Observed {
110    /// The observation, with its judgement.
111    pub observation: Observation,
112    /// Where it was recorded, repository-relative; absent on a dry run.
113    #[serde(default, skip_serializing_if = "Option::is_none")]
114    pub recorded: Option<String>,
115    /// The exit status the command line reports: 0 served, 10 a measured no, 12 unanswered.
116    pub exit_code: i32,
117}

119/// What `served.forge` came to: whether the forge answered, and what its answer held. The
120/// answer itself is read through `deploy.observed`; this is the receipt of having asked.
121///
122/// ```
123/// use majordomus_cli::capability::builtin::served::ForgeObserved;
124/// // a forge that could not be asked is said, with the reason, and is exit 12: not answered
125/// let json = r#"{"observed":false,"reason":"gh repo view failed: no network",
126///   "deployments":0,"releases":0,"environments":0,"exit_code":12}"#;
127/// let got: ForgeObserved = serde_json::from_str(json).unwrap();
128/// assert!(!got.observed);
129/// assert_eq!(got.exit_code, 12);
130/// assert!(got.repository.is_none());
131/// ```
132#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
133pub struct ForgeObserved {
134    /// Whether the forge answered and the answer was recorded.
135    pub observed: bool,
136    /// Why not, in the words of whatever refused; absent when it answered.
137    #[serde(default, skip_serializing_if = "Option::is_none")]
138    pub reason: Option<String>,
139    /// `owner/name`, as the forge named the repository.
140    #[serde(default, skip_serializing_if = "Option::is_none")]
141    pub repository: Option<String>,
142    /// When the forge was asked, RFC 3339.
143    #[serde(default, skip_serializing_if = "Option::is_none")]
144    pub observed_at: Option<String>,
145    /// How many deployments the answer held.
146    pub deployments: usize,
147    /// How many releases the answer held.
148    pub releases: usize,
149    /// How many environments those deployments went to.
150    pub environments: usize,
151    /// The exit status the command line reports: 0 observed, 12 not answered.
152    pub exit_code: i32,
153}

178/// One deployment's newest record and what it proves about the commit asked. Both are
179/// carried: the recorded judgement is what was seen, `now` is what it still proves.
180///
181/// ```
182/// use majordomus_cli::capability::builtin::served::DeploymentStanding;
183/// use majordomus_cli::served::ServedVerdict;
184/// let json = r#"{"deployment":"pages","observation":{"schema":"majordomus.served-observation/v1",
185///   "deployment":"pages","url":"https://majordomus.dev/build.json",
186///   "expected":"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa","verdict":"served",
187///   "reason":"the deployment serves aaaaaaaaaaaa","at":"1970-01-01T00:00:00Z"},
188///   "now":{"verdict":"stale","reason":"it does not contain bbbbbbbbbbbb"}}"#;
189/// let s: DeploymentStanding = serde_json::from_str(json).unwrap();
190/// // seen as served, and no longer proving the commit asked about
191/// assert_eq!(s.observation.judgement.verdict, ServedVerdict::Served);
192/// assert_eq!(s.now.verdict, ServedVerdict::Stale);
193/// ```
194#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
195pub struct DeploymentStanding {
196    /// The deployment name.
197    pub deployment: String,
198    /// The newest record of it.
199    pub observation: Observation,
200    /// That record re-judged against the commit asked about. This, not the recorded
201    /// judgement, is what a reader acts on.
202    pub now: Judgement,
203    /// For a deployment the layer declares as an object: `active` while `now` is served,
204    /// `declared` otherwise, `retired` when the object says so. Derived from the newest
205    /// observation, never set by hand (I3822); absent for a name no object carries.
206    #[serde(default, skip_serializing_if = "Option::is_none")]
207    pub status: Option<crate::deploy::Status>,
208}

210/// Every deployment's standing against one commit: the answer `served.show` gives on the
211/// command line, over HTTP and to an MCP client, from one derivation.
212///
213/// ```
214/// use majordomus_cli::capability::builtin::served::{standing, ServedStanding};
215/// use majordomus_cli::served::{Ancestry, OBSERVATIONS_PATH};
216/// # struct A; impl Ancestry for A {
217/// #     fn has(&self, _: &str) -> Option<bool> { Some(true) }
218/// #     fn contains(&self, _: &str, _: &str) -> Option<bool> { Some(true) } }
219/// let dir = tempfile::tempdir().unwrap();
220/// let s: ServedStanding = standing(dir.path(), &"a".repeat(40), None, &A);
221/// assert_eq!(s.observations, OBSERVATIONS_PATH);
222/// assert!(s.deployments.is_empty(), "nothing observed yet");
223/// assert_eq!(s.unreadable, 0);
224/// ```
225#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
226pub struct ServedStanding {
227    /// The full commit every record was re-judged against.
228    pub expected: String,
229    /// The checkout-local file the records were read from.
230    pub observations: String,
231    /// Lines of that file that were not readable observations.
232    pub unreadable: usize,
233    /// One entry per deployment observed, in name order.
234    pub deployments: Vec<DeploymentStanding>,
235}

241/// Resolves `rev` to a full commit sha, refusing what git does not resolve to a commit.
242fn resolve(root: &Path, rev: &str) -> Result<String, CapabilityError> {
243    if rev.starts_with('-') {
244        return Err(CapabilityError::InvalidInput(format!(
245            "`{rev}` is not a revision"
246        )));
247    }
248    let out = crate::git::read_only(root)
249        .args(["rev-parse", "--verify", "--quiet", "--end-of-options"])
250        .arg(format!("{rev}^{{commit}}"))
251        .output()
252        .map_err(|e| CapabilityError::Internal(format!("cannot run git: {e}")))?;
253    let sha = String::from_utf8_lossy(&out.stdout).trim().to_string();
254    if !out.status.success() || sha.len() != 40 {
255        return Err(CapabilityError::InvalidInput(format!(
256            "`{rev}` does not name a commit this clone holds"
257        )));
258    }
259    Ok(sha)
260}

262/// The `base_url` of the site configuration, when the checkout has one.
263fn configured_base(root: &Path) -> Option<String> {
264    let text = std::fs::read_to_string(root.join("site/config.toml")).ok()?;
265    text.lines().find_map(|l| {
266        let rest = l
267            .trim()
268            .strip_prefix("base_url")?
269            .trim_start()
270            .strip_prefix('=')?;
271        Some(rest.trim().trim_matches('"').to_string())
272    })
273}

275/// The deployment objects the layer declares, by id. One that does not parse is left out
276/// here: `deploy.check` is where it is refused, with the key and the correction.
277fn declared_deployments(ctx: &Context) -> Vec<crate::deploy::Deployment> {
278    ctx.index
279        .objects
280        .iter()
281        .filter(|o| o.kind == crate::deploy::KIND)
282        .filter_map(|o| crate::deploy::Deployment::parse(o).ok())
283        .collect()
284}

286/// Where a deployment object says it answers: its provider's URL and its liveness route,
287/// which reports the commit the running executable was built from. `None` for an object
288/// that names no provider, whose address is the platform's to know.
289fn provider_address(deployment: &crate::deploy::Deployment) -> Option<(String, String)> {
290    let provider = deployment.provider.as_ref()?;
291    let route = deployment.health.liveness.as_str();
292    Some((
293        provider.url.clone(),
294        route.trim_start_matches('/').to_string(),
295    ))
296}

298/// The status a deployment object stands at, judged from what it was last seen serving:
299/// `active` only while that is the commit asked about, and a retired object stays retired.
300fn derived_status(
301    declared: Option<crate::deploy::Status>,
302    now: &Judgement,
303) -> crate::deploy::Status {
304    use crate::deploy::Status;
305    match declared {
306        Some(Status::Retired) => Status::Retired,
307        _ if now.verdict.passes() => Status::Active,
308        _ => Status::Declared,
309    }
310}

312fn observe(ctx: &Context, input: ObserveInput) -> Result<Observed, CapabilityError> {
313    let root = root_of(ctx);
314    let expected = resolve(&root, input.commit.as_deref().unwrap_or("HEAD"))?;
315    let object = input.deployment.as_deref().and_then(|id| {
316        declared_deployments(ctx)
317            .into_iter()
318            .find(|d| d.id == id)
319            .and_then(|d| provider_address(&d))
320    });
321    let base = match (input.url, &object) {
322        (Some(u), _) => u,
323        (None, Some((url, _))) => url.clone(),
324        (None, None) => configured_base(&root).ok_or_else(|| {
325            CapabilityError::InvalidInput(
326                "no `url` was given, no deployment object with a provider is named, and site/config.toml names no base_url".into(),
327            )
328        })?,
329    };
330    served::check_base(&base).map_err(CapabilityError::InvalidInput)?;
331    let identity = match (input.identity, object) {
332        (Some(i), _) => i,
333        (None, Some((_, route))) => route,
334        (None, None) => served::DEFAULT_IDENTITY.into(),
335    };
336    if identity.contains("..") || identity.contains('?') || identity.contains('#') {
337        return Err(CapabilityError::InvalidInput(format!(
338            "`{identity}` is not a plain path under the site"
339        )));
340    }
341    let deployment = input
342        .deployment
343        .unwrap_or_else(|| DEFAULT_DEPLOYMENT.into());
344    let fetch = Curl {
345        max_seconds: input
346            .timeout_seconds
347            .unwrap_or(served::DEFAULT_PROBE_SECONDS)
348            .max(1),
349    };
350    let obs = served::observe(
351        &deployment,
352        &base,
353        &identity,
354        &expected,
355        &fetch,
356        &GitAncestry { root: &root },
357        SystemTime::now(),
358    );
359    let recorded = if input.dry_run {
360        None
361    } else {
362        served::append(&root, &obs).map_err(|e| {
363            CapabilityError::Internal(format!("cannot record the observation: {e}"))
364        })?;
365        Some(served::OBSERVATIONS_PATH.to_string())
366    };
367    Ok(Observed {
368        exit_code: obs.judgement.verdict.exit_code(),
369        observation: obs,
370        recorded,
371    })
372}

378/// Ask the forge through `program`, record what it said, and say what that came to. A forge
379/// that could not be asked is an answer of its own, never an error: the caller is a job that
380/// goes on to read whatever was last recorded.
381fn forge_with(program: &str, root: &Path) -> ForgeObserved {
382    match crate::deploy::observed::refresh_with(program, root) {
383        Ok(o) => ForgeObserved {
384            observed: true,
385            reason: None,
386            environments: crate::deploy::observed::environments(&o).len(),
387            deployments: o.deployments.len(),
388            releases: o.releases.len(),
389            repository: Some(o.repository),
390            observed_at: Some(o.observed_at),
391            exit_code: 0,
392        },
393        Err(reason) => ForgeObserved {
394            observed: false,
395            reason: Some(reason),
396            repository: None,
397            observed_at: None,
398            deployments: 0,
399            releases: 0,
400            environments: 0,
401            exit_code: 12,
402        },
403    }
404}

406fn show(ctx: &Context, input: ShowInput) -> Result<ServedStanding, CapabilityError> {
407    let root = root_of(ctx);
408    let expected = resolve(&root, input.commit.as_deref().unwrap_or("HEAD"))?;
409    let mut answer = standing(
410        &root,
411        &expected,
412        input.deployment.as_deref(),
413        &GitAncestry { root: &root },
414    );
415    let objects = declared_deployments(ctx);
416    for d in &mut answer.deployments {
417        if let Some(object) = objects.iter().find(|o| o.id == d.deployment) {
418            d.status = Some(derived_status(object.status, &d.now));
419        }
420    }
421    Ok(answer)
422}

424/// The standing of every recorded deployment against `expected`. Public so that a consumer
425/// that judges an intent's deployment criterion reads the same derivation as this read.
426///
427/// ```
428/// use majordomus_cli::capability::builtin::served::standing;
429/// use majordomus_cli::served::{append, observe, Ancestry, Fetch, Fetched, ServedVerdict};
430/// use std::time::SystemTime;
431/// struct Linear;  // a <- b
432/// impl Ancestry for Linear {
433///     fn has(&self, c: &str) -> Option<bool> { Some(c.starts_with('a') || c.starts_with('b')) }
434///     fn contains(&self, d: &str, a: &str) -> Option<bool> { Some(d >= a) }
435/// }
436/// struct Serves(String);
437/// impl Fetch for Serves {
438///     fn fetch(&self, _: &str) -> Fetched {
439///         Fetched::Body(format!(r#"{{"commit":"{}","dirty":false}}"#, self.0).into_bytes())
440///     }
441/// }
442/// let (a, b) = ("a".repeat(40), "b".repeat(40));
443/// let dir = tempfile::tempdir().unwrap();
444/// let o = observe("pages", "https://s", "build.json", &a, &Serves(a.clone()), &Linear,
445///                 SystemTime::UNIX_EPOCH);
446/// append(dir.path(), &o).unwrap();
447///
448/// // the record still proves the commit it was taken against ...
449/// let now = standing(dir.path(), &a, None, &Linear);
450/// assert_eq!(now.deployments[0].now.verdict, ServedVerdict::Served);
451/// // ... and stops proving once the commit asked about moves past it
452/// let later = standing(dir.path(), &b, None, &Linear);
453/// assert_eq!(later.deployments[0].now.verdict, ServedVerdict::Stale);
454/// ```
455pub fn standing(
456    root: &Path,
457    expected: &str,
458    only: Option<&str>,
459    git: &dyn served::Ancestry,
460) -> ServedStanding {
461    let (all, unreadable) = served::read_all(root);
462    let mut names: Vec<&str> = all.iter().map(|o| o.deployment.as_str()).collect();
463    // the canonical order every other collection is shown in; natural_cmp is total, so
464    // equal names are adjacent and dedup still removes every repeat
465    crate::order::canonical_strings(&mut names);
466    names.dedup();
467    let deployments = names
468        .into_iter()
469        .filter(|n| only.is_none_or(|o| o == *n))
470        .filter_map(|n| served::latest(&all, n))
471        .map(|o| DeploymentStanding {
472            deployment: o.deployment.clone(),
473            now: o.rejudge(expected, git),
474            observation: o.clone(),
475            status: None,
476        })
477        .collect();
478    ServedStanding {
479        expected: expected.into(),
480        observations: served::OBSERVATIONS_PATH.into(),
481        unreadable,
482        deployments,
483    }
484}
