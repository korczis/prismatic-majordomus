1//! The semantics of the native command line that its declaration cannot carry.
2//!
3//! clap declares everything about *shape*: the commands, their arguments, defaults, value
4//! sets, help. It cannot declare what running one does to the repository, whether it holds
5//! a terminal, or where the values of `<BRANCH>` come from. Those are decisions, and this
6//! is where they are made — once, beside the declaration, in the same crate, never again
7//! per surface.
8//!
9//! # Why this is not a second registry
10//!
11//! Two properties keep it from becoming one:
12//!
13//! - **The default is the answer for most commands.** This executable is read-only and
14//!   non-interactive; [`Semantics::DEFAULT`] says so, and a command that is nothing more
15//!   than that appears nowhere below. Only a command that *departs* from the default is
16//!   named — which is exactly the information nothing else holds.
17//! - **Completeness is enforced in both directions.** `validate` fails when an entry
18//!   names a command that does not exist (a rename that left this file behind) and when a
19//!   command's effect could not be decided. The table cannot silently go stale, which is
20//!   the property that distinguishes a declaration from a mirror.
21//!
22//! This is the same arrangement the examples beside the command line already use, and it
23//! is checked by the same kind of test.
24

27/// What a command does, beyond what its declaration says.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub struct Semantics {
30    /// The command path this describes, the executable's own name excluded:
31    /// `&["worktree", "remove"]`.
32    pub path: &'static [&'static str],
33    /// What running it changes.
34    pub effect: Effect,
35    /// How it behaves towards a terminal.
36    pub interactivity: Interactivity,
37    /// What must be true for it to mean anything.
38    pub requires: &'static [Requirement],
39}
40
41impl Semantics {
42    /// What a command is when it says nothing: this executable reads and answers.
43    pub const DEFAULT: Semantics = Semantics {
44        path: &[],
45        effect: Effect::ReadOnly,
46        interactivity: Interactivity::NonInteractive,
47        requires: &[Requirement::Repository, Requirement::Layer],
48    };
49}

242/// A name the workflow runner answered to before the bridge was derived, kept so that no
243/// recipe a person or a script used disappears.
244///
245/// An alias carries no description, no group and no semantics of its own: it names a
246/// command of the graph and, where the older spelling meant a command *with* an argument,
247/// the arguments it always passed. Everything a projection shows about it comes from the
248/// command it names. [`unused_aliases`] fails the build when an alias names a command that
249/// no longer exists, so a rename cannot leave a recipe pointing at nothing.
250#[derive(Debug, Clone, Copy, PartialEq, Eq)]
251pub struct WorkflowAlias {
252    /// The recipe name, as it was.
253    pub alias: &'static str,
254    /// The command it resolves to, by canonical identity.
255    pub command: &'static str,
256    /// The arguments the older spelling always passed, before the caller's own.
257    pub arguments: &'static [&'static str],
258    /// Why it exists, in one line, for the generated file and the reference to print.
259    pub reason: &'static str,
260}

373/// Every alias that names a command the graph does not carry.
374pub fn unused_aliases(existing: &[String]) -> Vec<&'static WorkflowAlias> {
375    WORKFLOW_ALIASES
376        .iter()
377        .filter(|a| !existing.iter().any(|id| id == a.command))
378        .collect()
379}

381/// The semantics of one command: the most specific entry that names it, or the most
382/// specific entry that names a prefix of it, or the default.
383///
384/// Inheritance down the tree is what keeps the table short: `worktree` says the whole
385/// topology needs no layer once, and every command under it is answered by that.
386pub fn of(path: &[String]) -> Semantics {
387    let mut best: Option<&Semantics> = None;
388    for entry in SEMANTICS {
389        if entry.path.len() <= path.len()
390            && entry
391                .path
392                .iter()
393                .zip(path.iter())
394                .all(|(a, b)| *a == b.as_str())
395            && best.is_none_or(|b| b.path.len() < entry.path.len())
396        {
397            best = Some(entry);
398        }
399    }
400    best.copied().unwrap_or(Semantics::DEFAULT)
401}

403/// Where an argument's values come from, when a binding names it.
404pub fn binding(path: &[String], argument: &str) -> Option<ValueSource> {
405    VALUE_BINDINGS
406        .iter()
407        .find(|b| {
408            b.argument == argument
409                && b.path.len() == path.len()
410                && b.path
411                    .iter()
412                    .zip(path.iter())
413                    .all(|(a, c)| *a == c.as_str())
414        })
415        .map(|b| b.source)
416}

418/// How openly an argument's value may be handled, inferred from what it is called.
419///
420/// Nothing in this executable takes a credential today. The classification exists so that
421/// the day one does, no completion, log or cache has to be taught to leave it alone.
422pub fn secrecy(argument: &str, value_name: Option<&str>) -> Secrecy {
423    let hay = format!(
424        "{argument} {}",
425        value_name.unwrap_or_default().to_lowercase()
426    );
427    const SECRET: &[&str] = &["token", "secret", "password", "passwd", "api-key", "apikey"];
428    const SENSITIVE: &[&str] = &["credential", "auth", "key"];
429    if SECRET.iter().any(|w| hay.contains(w)) {
430        Secrecy::Secret
431    } else if SENSITIVE.iter().any(|w| hay.contains(w)) {
432        Secrecy::Sensitive
433    } else {
434        Secrecy::Public
435    }
436}

438/// Every entry names a command that exists.
439///
440/// The failure this catches is the one that makes a table like this dangerous: a command
441/// is renamed, the entry beside it is not, and the command silently falls back to the
442/// default — which for `worktree remove` would mean a destructive command advertised to
443/// every machine surface. Returns the offending paths.
444pub fn unused(existing: &[Vec<String>]) -> Vec<String> {
445    let mut out = Vec::new();
446    for entry in SEMANTICS {
447        if !existing
448            .iter()
449            .any(|p| p.len() == entry.path.len() && p.iter().zip(entry.path).all(|(a, b)| a == b))
450        {
451            out.push(entry.path.join(" "));
452        }
453    }
454    for binding in VALUE_BINDINGS {
455        if !existing.iter().any(|p| {
456            p.len() == binding.path.len() && p.iter().zip(binding.path).all(|(a, b)| a == b)
457        }) {
458            out.push(format!("{} <{}>", binding.path.join(" "), binding.argument));
459        }
460    }
461    out
462}

464#[cfg(test)]
465mod tests {
466    use super::*;
467
468    fn p(words: &[&str]) -> Vec<String> {
469        words.iter().map(|w| w.to_string()).collect()
470    }
471
472    #[test]
473    fn the_default_answers_an_unnamed_command() {
474        let s = of(&p(&["scope"]));
475        assert_eq!(s.effect, Effect::ReadOnly);
476        assert_eq!(s.interactivity, Interactivity::NonInteractive);
477    }
478
479    #[test]
480    fn the_most_specific_entry_wins() {
481        assert_eq!(of(&p(&["bench"])).effect, Effect::RepositoryMutation);
482        assert_eq!(of(&p(&["bench", "coverage"])).effect, Effect::ReadOnly);
483        assert_eq!(
484            of(&p(&["bench", "baseline", "update"])).effect,
485            Effect::RepositoryMutation
486        );
487    }
488
489    #[test]
490    fn semantics_are_inherited_down_the_tree() {
491        // `worktree` declares that the topology needs no layer; `worktree list` inherits it.
492        let s = of(&p(&["worktree", "list"]));
493        assert_eq!(s.requires, &[Requirement::Repository]);
494        assert_eq!(s.effect, Effect::ReadOnly);
495        assert_eq!(of(&p(&["worktree", "remove"])).effect, Effect::Destructive);
496    }
497
498    #[test]
499    fn a_binding_is_exact_not_inherited() {
500        assert_eq!(
501            binding(&p(&["worktree", "path"]), "branch"),
502            Some(ValueSource::Branch)
503        );
504        assert_eq!(binding(&p(&["worktree"]), "branch"), None);
505    }
506
507    #[test]
508    fn secrets_are_recognised_by_name() {
509        assert_eq!(secrecy("token", None), Secrecy::Secret);
510        assert_eq!(secrecy("api_key", Some("API-KEY")), Secrecy::Secret);
511        assert_eq!(secrecy("branch", Some("BRANCH")), Secrecy::Public);
512    }
513
514    #[test]
515    fn an_entry_naming_nothing_is_reported() {
516        let existing = vec![p(&["mcp"])];
517        let stale = unused(&existing);
518        assert!(stale.contains(&"serve".to_string()));
519    }
