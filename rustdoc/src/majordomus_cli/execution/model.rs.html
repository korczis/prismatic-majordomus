1//! What an execution is: its identity, the states it may be in, and the snapshot every
2//! transport answers with. Plain data with no transport type in it and no handle to a
3//! running thread: a snapshot is taken under the store's lock and read afterwards.
4

9use serde::{Deserialize, Serialize};

12/// The identity of one execution: `x-<UTC timestamp>-<8 hex>`.
13///
14/// The shape is the layer's own — a task is `t-20260905034523-a9f1` — so that an id read
15/// in a log, a URL or a browser tab is recognisable as this repository's without a
16/// dependency on a UUID crate. The timestamp orders ids by creation for a reader; the
17/// suffix, not the timestamp, is what makes two ids created in the same second differ.
18#[derive(
19    Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
20)]
21#[serde(transparent)]
22pub struct ExecutionId(String);
23
24impl ExecutionId {
25    /// A fresh id for now.
26    ///
27    /// ```
28    /// use majordomus_cli::execution::ExecutionId;
29    /// let a = ExecutionId::fresh();
30    /// let b = ExecutionId::fresh();
31    /// assert_ne!(a, b, "two ids from the same second still differ");
32    /// assert!(ExecutionId::parse(a.as_str()).is_some());
33    /// ```
34    pub fn fresh() -> Self {
35        static COUNTER: AtomicU64 = AtomicU64::new(0);
36        let now = SystemTime::now()
37            .duration_since(UNIX_EPOCH)
38            .unwrap_or_default();
39        let stamp = compact_utc(now.as_secs());
40        // the suffix is not a secret and does not need to be unguessable: the server
41        // refuses a state-changing request from another origin, and an id is not a
42        // credential. It needs to be unique within a process, which a counter guarantees,
43        // and unlikely to repeat across two processes of one repository, which the
44        // sub-second part gives it.
45        let seq = COUNTER.fetch_add(1, Ordering::Relaxed);
46        let mix = (now.subsec_nanos() as u64) ^ (seq << 20) ^ (seq.wrapping_mul(0x9e37_79b9));
47        ExecutionId(format!("x-{stamp}-{:08x}", mix & 0xffff_ffff))
48    }
49
50    /// Parse an id a client sent. `None` when it is not one this executable would mint,
51    /// which is what keeps a path, a traversal or a wildcard out of a store lookup.
52    ///
53    /// ```
54    /// use majordomus_cli::execution::ExecutionId;
55    /// assert!(ExecutionId::parse("x-20260908T010203Z-0a1b2c3d").is_some());
56    /// assert!(ExecutionId::parse("../../etc/passwd").is_none());
57    /// assert!(ExecutionId::parse("x-20260908T010203Z-0a1b2c3").is_none(), "eight hex digits");
58    /// ```
59    pub fn parse(text: &str) -> Option<Self> {
60        let rest = text.strip_prefix("x-")?;
61        let (stamp, suffix) = rest.split_once('-')?;
62        let stamp_ok = stamp.len() == 16
63            && stamp.ends_with('Z')
64            && stamp.as_bytes()[8] == b'T'
65            && stamp[..8].bytes().all(|b| b.is_ascii_digit())
66            && stamp[9..15].bytes().all(|b| b.is_ascii_digit());
67        let suffix_ok = suffix.len() == 8 && suffix.bytes().all(|b| b.is_ascii_hexdigit());
68        (stamp_ok && suffix_ok).then(|| ExecutionId(text.to_string()))
69    }
70
71    /// The id as text.
72    pub fn as_str(&self) -> &str {
73        &self.0
74    }
75}
76
77impl std::fmt::Display for ExecutionId {
78    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
79        f.write_str(&self.0)
80    }
81}

83/// `YYYYMMDDTHHMMSSZ` in UTC from a Unix second, the compact form of the layer's own
84/// timestamps.
85///
86/// ```
87/// use majordomus_cli::execution::model::compact_utc;
88/// assert_eq!(compact_utc(0), "19700101T000000Z");
89/// assert_eq!(compact_utc(1_788_000_000), "20260829T104000Z");
90/// ```
91pub fn compact_utc(secs: u64) -> String {
92    let iso = crate::peers::rfc3339(UNIX_EPOCH + std::time::Duration::from_secs(secs));
93    iso.chars().filter(|c| !matches!(c, '-' | ':')).collect()
94}

96/// Where an execution is in its life.
97///
98/// The transitions are the whole contract, and [`ExecutionState::may_move_to`] is the one
99/// place they are written down: the engine asks before every change, the store refuses a
100/// move it did not allow, and a client that reads a final state never sees it move again.
101///
102/// ```text
103///   queued ──► running ──┬──► succeeded
104///      │                 ├──► failed
105///      │                 └──► cancelling ──┬──► cancelled
106///      │                                   ├──► succeeded
107///      └──► cancelled                      └──► failed
108/// ```
109///
110/// There is no `starting`. An in-process engine claims an execution and enters its handler
111/// in the same instant, so a state between the two would be one no client could ever
112/// observe and every client would have to handle.
113#[derive(
114    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
115)]
116#[serde(rename_all = "snake_case")]
117pub enum ExecutionState {
118    /// Accepted, and no worker has picked it up.
119    Queued,
120    /// The handler is running.
121    Running,
122    /// Cancellation was asked for and the handler has not stopped yet.
123    Cancelling,
124    /// The handler returned an output.
125    Succeeded,
126    /// The handler returned an error, or the worker could not run it.
127    Failed,
128    /// It stopped because it was asked to.
129    Cancelled,
130}
131
132impl ExecutionState {
133    /// The word as serialised.
134    pub fn as_str(self) -> &'static str {
135        match self {
136            ExecutionState::Queued => "queued",
137            ExecutionState::Running => "running",
138            ExecutionState::Cancelling => "cancelling",
139            ExecutionState::Succeeded => "succeeded",
140            ExecutionState::Failed => "failed",
141            ExecutionState::Cancelled => "cancelled",
142        }
143    }
144
145    /// Is this the last thing that will be said about the execution?
146    ///
147    /// ```
148    /// use majordomus_cli::execution::ExecutionState;
149    /// assert!(ExecutionState::Succeeded.is_final());
150    /// assert!(!ExecutionState::Cancelling.is_final());
151    /// ```
152    pub fn is_final(self) -> bool {
153        matches!(
154            self,
155            ExecutionState::Succeeded | ExecutionState::Failed | ExecutionState::Cancelled
156        )
157    }
158
159    /// Is the execution doing work, or about to?
160    pub fn is_active(self) -> bool {
161        !self.is_final()
162    }
163
164    /// May the execution move from here to `next`?
165    ///
166    /// ```
167    /// use majordomus_cli::execution::ExecutionState::*;
168    /// assert!(Queued.may_move_to(Running));
169    /// assert!(Running.may_move_to(Cancelling));
170    /// assert!(Queued.may_move_to(Cancelled), "a queued execution is cancelled without running");
171    /// assert!(!Succeeded.may_move_to(Running), "a final state is final");
172    /// assert!(!Running.may_move_to(Queued), "nothing goes back");
173    /// ```
174    pub fn may_move_to(self, next: ExecutionState) -> bool {
175        use ExecutionState::*;
176        matches!(
177            (self, next),
178            (Queued, Running)
179                | (Queued, Cancelled)
180                | (Queued, Failed)
181                | (Running, Succeeded)
182                | (Running, Failed)
183                | (Running, Cancelling)
184                | (Cancelling, Cancelled)
185                | (Cancelling, Succeeded)
186                | (Cancelling, Failed)
187        )
188    }
189}

222/// How far along an execution is.
223#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
224pub struct ProgressView {
225    /// Units done.
226    pub current: u64,
227    /// Units in total, when the handler knows how many.
228    #[serde(skip_serializing_if = "Option::is_none")]
229    pub total: Option<u64>,
230    /// What is being done, for a reader.
231    #[serde(skip_serializing_if = "Option::is_none")]
232    pub message: Option<String>,
233}
234
235impl ProgressView {
236    /// The fraction done, 0..=100, when a total is known.
237    ///
238    /// ```
239    /// use majordomus_cli::execution::ProgressView;
240    /// let p = ProgressView { current: 3, total: Some(4), message: None };
241    /// assert_eq!(p.percent(), Some(75));
242    /// let open = ProgressView { current: 3, total: None, message: None };
243    /// assert_eq!(open.percent(), None);
244    /// ```
245    pub fn percent(&self) -> Option<u64> {
246        match self.total {
247            Some(total) if total > 0 => Some((self.current.min(total) * 100) / total),
248            _ => None,
249        }
250    }
251}

253/// Why an execution ended badly, in the repository's diagnostic vocabulary rather than in
254/// Rust's: a code a program branches on, a sentence a person reads, and where to look.
255///
256/// A panic message and a backtrace are never in here. The engine catches a panicking
257/// handler and reports `internal` with the execution's correlation id; the panic itself is
258/// on the process's own error stream, where the operator running the server can read it.
259#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
260pub struct ExecutionError {
261    /// `invalid_input`, `not_found`, `refused`, `internal`, `cancelled`, `unavailable`.
262    pub code: String,
263    /// What went wrong, for a person.
264    pub message: String,
265    /// What to do about it, when there is something.
266    #[serde(skip_serializing_if = "Option::is_none")]
267    pub suggestion: Option<String>,
268    /// The execution this failure belongs to, so a log line and a UI agree.
269    pub correlation_id: String,
270}

333/// Which repository, and which checkout of it, an execution ran against.
334///
335/// An execution is never run against "wherever this process happens to be": the engine
336/// stamps the repository the index was read from, and a request that names a different one
337/// is refused. The path itself is not here — this value is served to whoever can reach the
338/// socket, and where the checkout sits on the host is of no use to them.
339#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
340pub struct RepositoryRef {
341    /// The repository's name: the last component of its root.
342    pub name: String,
343    /// The stable identity two processes over one checkout both compute.
344    pub id: String,
345    /// The branch checked out, when git can say.
346    #[serde(skip_serializing_if = "Option::is_none")]
347    pub branch: Option<String>,
348}

350/// One execution, as every transport answers it.
351///
352/// This is the whole of what a client may know. It is produced under the store's lock, so
353/// a reader never sees a state and an output that disagree: an execution that says
354/// `succeeded` carries its output in the same snapshot, and one that says `failed` carries
355/// its error.
356#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
357pub struct Execution {
358    /// The identity.
359    pub id: ExecutionId,
360    /// The capability that ran, by its canonical id.
361    pub capability: String,
362    /// That capability's title, so a list needs no second lookup.
363    pub title: String,
364    /// Where it is.
365    pub state: ExecutionState,
366    /// When it was accepted, RFC 3339 in UTC.
367    pub created_at: String,
368    /// When the handler was entered.
369    #[serde(skip_serializing_if = "Option::is_none")]
370    pub started_at: Option<String>,
371    /// When it reached a final state.
372    #[serde(skip_serializing_if = "Option::is_none")]
373    pub finished_at: Option<String>,
374    /// How long it ran, in milliseconds, once it has finished.
375    #[serde(skip_serializing_if = "Option::is_none")]
376    pub duration_ms: Option<u64>,
377    /// The input it was given, with every value the input schema marks sensitive replaced.
378    pub input: Value,
379    /// What the handler returned, for a succeeded execution and never before.
380    #[serde(skip_serializing_if = "Option::is_none")]
381    pub output: Option<Value>,
382    /// Why it failed, for a failed execution and never before.
383    #[serde(skip_serializing_if = "Option::is_none")]
384    pub error: Option<ExecutionError>,
385    /// The last progress the handler reported.
386    #[serde(skip_serializing_if = "Option::is_none")]
387    pub progress: Option<ProgressView>,
388    /// The steps it entered, in order.
389    #[serde(default, skip_serializing_if = "Vec::is_empty")]
390    pub steps: Vec<StepView>,
391    /// The findings it reported.
392    #[serde(default, skip_serializing_if = "Vec::is_empty")]
393    pub diagnostics: Vec<ExecutionDiagnostic>,
394    /// The sequence number of the last event: the cursor a client subscribes from.
395    pub last_sequence: u64,
396    /// How many events it has produced, including the ones no longer retained.
397    pub event_count: u64,
398    /// Whether the retained event history has lost its oldest entries to the bound.
399    pub events_truncated: bool,
400    /// Whether asking to cancel this execution will do anything: the capability's policy.
401    pub cancellable: bool,
402    /// Who asked for it.
403    pub actor: Actor,
404    /// Which repository it ran against.
405    pub repository: RepositoryRef,
406    /// The id used in this process's logs and tracing spans for this execution; the
407    /// execution id itself, carried under the name a reader of a log expects.
408    pub correlation_id: String,
409}
410
411impl Execution {
412    /// The percentage a progress bar shows, when the handler said enough to compute one.
413    pub fn percent(&self) -> Option<u64> {
414        self.progress.as_ref().and_then(ProgressView::percent)
415    }
416}

422    #[test]
423    fn an_id_round_trips_and_a_path_does_not() {
424        let id = ExecutionId::fresh();
425        assert_eq!(ExecutionId::parse(id.as_str()).as_ref(), Some(&id));
426        assert_eq!(id.to_string(), id.as_str());
427        for bad in [
428            "",
429            "x-",
430            "x--0a1b2c3d",
431            "../x-20260908T010203Z-0a1b2c3d",
432            "x-20260908T010203Z-0a1b2c3d/..",
433            "y-20260908T010203Z-0a1b2c3d",
434            "x-2026090801020Z-0a1b2c3d",
435            "x-20260908T010203Z-0a1b2c3g",
436        ] {
437            assert!(ExecutionId::parse(bad).is_none(), "{bad:?} accepted");
438        }
439    }

441    #[test]
442    fn ids_do_not_repeat_within_a_process() {
443        let ids: std::collections::BTreeSet<_> = (0..2000).map(|_| ExecutionId::fresh()).collect();
444        assert_eq!(ids.len(), 2000);
445    }

447    #[test]
448    fn every_state_is_reachable_and_no_final_state_moves() {
449        use ExecutionState::*;
450        let all = [Queued, Running, Cancelling, Succeeded, Failed, Cancelled];
451        for state in all {
452            if state.is_final() {
453                assert!(
454                    all.iter().all(|next| !state.may_move_to(*next)),
455                    "{state:?} moved"
456                );
457            } else {
458                assert!(
459                    all.iter().any(|next| state.may_move_to(*next)),
460                    "{state:?} is a dead end"
461                );
462            }
463            // every state is reachable from queued by some path
464            assert_eq!(state.is_active(), !state.is_final());
465        }
466        // the words are the wire format and are asserted rather than assumed
467        assert_eq!(
468            all.map(|s| s.as_str()),
469            [
470                "queued",
471                "running",
472                "cancelling",
473                "succeeded",
474                "failed",
475                "cancelled"
476            ]
477        );
478        for state in all {
479            let text = serde_json::to_string(&state).unwrap();
480            assert_eq!(text, format!("\"{}\"", state.as_str()));
481        }
482    }

485    fn a_compact_timestamp_is_the_layers_own_shape() {
486        assert_eq!(compact_utc(0), "19700101T000000Z");
487        assert_eq!(compact_utc(1_788_000_000).len(), 16);
