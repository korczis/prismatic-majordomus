1//! Knowledge: what the deriver left for review and whether it is still writing. Read from
2//! `knowledge_base.status` and `knowledge_base.candidates`.
3//!
4//! The screen shows the queue and names the commands a reviewer runs. It promotes and rejects
5//! nothing: a screen reads and never executes (ADR 0142), and promotion is a person's act.
6//! The table is bounded, because the queue is not: what is left out is counted and said.

25/// How many candidates, and how many subjects, the screen draws. The queue has no bound of
26/// its own beyond the policy's cap, and a screen answers within its budget whatever the queue
27/// holds; the newest are drawn and the rest are counted.
28pub const SHOWN: usize = 50;

59/// The Knowledge screen.
60pub fn screen(ctx: &Context) -> View {
61    let answers = ask_all(ctx, &[STATUS, CANDIDATES]);
62    let status = &answers[&STATUS];
63    let queue = &answers[&CANDIDATES];
64
65    let top = hero(
66        "sparkles",
67        "orange",
68        "Knowledge",
69        "What the work left behind, awaiting a reviewer.",
70        "Every record the deriver wrote from a closed episode and nobody has promoted or rejected yet, what it is about, and whether the deriver is still writing.",
71        vec![
72            kpi("Awaiting review", &queue.read("/total")),
73            kpi("On this branch", &queue.read("/on_this_branch")),
74            kpi("Unattributed", &queue.read("/unattributed")),
75            ledger_kpi(
76                "Closed, underived",
77                status,
78                kpi(
79                    "Closed, underived",
80                    &status.read("/closed_without_derivation"),
81                ),
82            ),
83            ledger_kpi(
84                "Newest episode",
85                status,
86                kpi_state("Newest episode", &status.read("/freshness")),
87            ),
88        ],
89    );
90
91    let grid = el("div")
92        .class("mj-ui-grid")
93        .child(panel(
94            "The deriver",
95            "Whether a derivation still follows every closed episode, as the status judges it.",
96            unanswered(status).unwrap_or_else(|| deriver(status)),
97            &[status],
98        ))
99        .child(panel(
100            "Review",
101            "What a reviewer runs. This screen reads the queue; it promotes and rejects nothing.",
102            review(),
103            &[],
104        ))
105        .child(wide(panel(
106            "By subject",
107            "The records derived from observations, grouped by what they are about.",
108            unanswered(queue).unwrap_or_else(|| subjects(queue)),
109            &[queue],
110        )))
111        .child(wide(panel(
112            "Awaiting review",
113            "The newest candidates, each leading to its record.",
114            unanswered(queue).unwrap_or_else(|| candidates(queue)),
115            &[queue],
116        )));
117
118    View::new(
119        "Knowledge",
120        el("div").class("mj-ui-stack").child(top).child(grid),
121    )
122}

124/// The judgement of the deriver: when it last wrote, when an episode last closed, why the
125/// newest episode is judged as it is, and every finding the status reports.
126fn deriver(status: &Answer) -> El {
127    let Some(answer) = status.ok() else {
128        return nothing("The status did not answer.");
129    };
130    if !ledger_recorded(answer) {
131        return no_ledger();
132    }
133    let mark = |key: &str| -> String {
134        answer
135            .get(key)
136            .map(|m| {
137                let at = text(m, "ts");
138                let episode = text(m, "episode");
139                if episode.is_empty() {
140                    at
141                } else {
142                    format!("{at}, episode {episode}")
143                }
144            })
145            .filter(|line| !line.is_empty())
146            .unwrap_or_else(|| "never, in this checkout".to_string())
147    };
148    let judged = answer.get("judged").and_then(Value::as_bool);
149    let stopped = answer.get("stopped_writer").and_then(Value::as_bool);
150    let verdict = match (judged, stopped) {
151        (Some(true), Some(true)) => "episodes close and no derivation followed the newest one",
152        (Some(true), _) => "a derivation follows the newest closed episode",
153        (Some(false), _) => "not judged; the findings say why",
154        (None, _) => "the status carries no judgement",
155    };
156    let fact = |name: &str, value: String| {
157        el("div")
158            .class("mj-ui-fact")
159            .child(el("dt").text(name))
160            .child(el("dd").text(value))
161    };
162    let facts = el("dl")
163        .class("mj-ui-facts")
164        .child(fact("Last derived", mark("last_derived")))
165        .child(fact("Last closed", mark("last_closed")))
166        .child(fact("Judgement", verdict.to_string()))
167        .child(fact("Why", text(answer, "freshness_reason")));
168    let findings = list(status, "/findings");
169    let body = el("div").class("mj-ui-stack").child(facts);
170    if findings.is_empty() {
171        return body;
172    }
173    body.child(
174        el("ul").class("mj-ui-prose-list").children(
175            findings
176                .iter()
177                .filter_map(Value::as_str)
178                .map(|finding| el("li").text(finding)),
179        ),
180    )
181}

183/// What a checkout without a session ledger is told instead of a judgement: the published
184/// snapshot and a fresh clone carry none, and a value read from a ledger that is not there
185/// would read as "never" or "0" about the repository when it is only about this checkout.
186const NO_LEDGER: &str = "not recorded here";
187
188/// Whether the status read any line of the session ledger: a close, a derivation, or a
189/// close no derivation names. The status's own `present` also counts candidates, which are
190/// tracked records and are read in any checkout; the ledger is local, so it is asked apart.
191fn ledger_recorded(answer: &Value) -> bool {
192    answer.get("last_closed").is_some_and(|m| !m.is_null())
193        || answer.get("last_derived").is_some_and(|m| !m.is_null())
194        || answer
195            .get("closed_without_derivation")
196            .and_then(Value::as_u64)
197            .is_some_and(|n| n > 0)
198}
199
200/// A tile whose value is read from the session ledger: as drawn when the status read a
201/// ledger, and as [`NO_LEDGER`] when it read none, so a snapshot never states a count or a
202/// freshness it could not observe.
203fn ledger_kpi(label: &str, status: &Answer, drawn: El) -> El {
204    match status.ok() {
205        Some(answer) if !ledger_recorded(answer) => kpi(
206            label,
207            &Reading {
208                source: STATUS,
209                pointer: "/last_closed",
210                state: Availability::Available(Value::String(NO_LEDGER.to_string())),
211            },
212        ),
213        _ => drawn,
214    }
215}

217/// The deriver's panel where no session ledger is here: what that means, and where the
218/// judgement is made instead. Prose, not facts: there is nothing to tabulate.
219fn no_ledger() -> El {
220    el("p")
221        .class("mj-ui-prose")
222        .text("No session ledger is recorded in this checkout, so whether the deriver keeps up is not judged here. The published snapshot and a fresh clone carry none; in a working checkout, ")
223        .child(el("code").class("mj-ui-code").text("majordomus knowledge status"))
224        .text(" judges it, and this panel shows when the deriver last wrote.")
225}

227/// The review, as the commands it is made of.
228fn review() -> El {
229    el("dl")
230        .class("mj-ui-facts")
231        .children(REVIEW.iter().map(|(command, does)| {
232            el("div")
233                .class("mj-ui-fact")

235                .child(el("dd").text(*does))

239/// The subjects, the first [`SHOWN`] of them, each with how many records await review, how
240/// many were rejected and kept, and how many episodes observed it.
241fn subjects(queue: &Answer) -> El {
242    let groups = list(queue, "/groups");
243    let size = |group: &Value, key: &str| -> String {
244        group
245            .get(key)
246            .and_then(Value::as_array)
247            .map_or(0, Vec::len)
248            .to_string()
249    };
250    let rows = groups
251        .iter()
252        .take(SHOWN)
253        .map(|group| {
254            el("tr")
255                .child(el("td").text(text(group, "about")))
256                .child(el("td").class("mj-ui-dim").text(text(group, "route")))
257                .child(el("td").text(size(group, "candidates")))
258                .child(el("td").text(size(group, "rejected")))
259                .child(el("td").text(size(group, "episodes")))
260        })
261        .collect::<Vec<_>>();
262    let shown = rows.len();
263    let body = el("div").class("mj-ui-stack").child(table(
264        &["About", "Route", "Awaiting", "Rejected", "Episodes"],
265        rows,
266        "No record derived from an observation awaits review.",
267    ));
268    match more(shown, groups.len(), None) {
269        Some(line) => body.child(line),
270        None => body,
271    }
272}

274/// The candidates, the newest [`SHOWN`] of them first. The source orders by id, which begins
275/// with the episode, so the last of its list are the newest; the order is the source's and is
276/// only read from its other end.
277fn candidates(queue: &Answer) -> El {
278    let all = list(queue, "/candidates");
279    let rows = all
280        .iter()
281        .rev()
282        .take(SHOWN)
283        .map(|record| {
284            let id = text(record, "id");
285            let uri = text(record, "uri");
286            let from = record
287                .get("branch")
288                .and_then(Value::as_str)
289                .filter(|branch| !branch.is_empty())
290                .map_or_else(|| "not known in this checkout".to_string(), str::to_string);
291            el("tr")
292                .child(
293                    el("td").child(object(&uri, id)).child(
294                        el("span")
295                            .class("mj-ui-cell-title")
296                            .text(text(record, "title")),
297                    ),
298                )
299                .child(el("td").class("mj-ui-dim").text(text(record, "class")))
300                .child(el("td").class("mj-ui-dim").text(from))
301                .child(
302                    el("td")
303                        .child(state(record, "freshness"))
304                        .child(el("span").class("mj-ui-dim").text(waited(record))),
305                )
306        })
307        .collect::<Vec<_>>();
308    let shown = rows.len();
309    let body = el("div").class("mj-ui-stack").child(table(
310        &["Record", "Class", "From", "Waiting"],
311        rows,
312        "No candidate awaits review.",
313    ));
314    match more(shown, all.len(), None) {
315        Some(line) => body.child(line),
316        None => body,
317    }
318}

320/// How long a record has waited, in the largest whole unit: the source measured it in
321/// minutes, and says in `freshness_reason` when it could not.
322fn waited(record: &Value) -> String {
323    match record.get("age_minutes").and_then(Value::as_i64) {
324        Some(minutes) if minutes >= 2 * 24 * 60 => format!(" {} d", minutes / (24 * 60)),
325        Some(minutes) if minutes >= 2 * 60 => format!(" {} h", minutes / 60),
326        Some(minutes) => format!(" {minutes} min"),
327        None => format!(" {}", text(record, "freshness_reason")),
328    }
329}

350        json!({
351            "id": format!("s-20261013{n:06}-aaaa-{n:012x}"),

523    /// Each line of the session ledger counts on its own, and a null or a zero is no line:
