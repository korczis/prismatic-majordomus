1//! The tool distribution's data directory: `share/kinds.yaml` and
2//! `share/schemas/<kind>.schema.json`, read at run time. Nothing about kinds or keys is
3//! compiled into the executable; the directory is located per invocation, explicitly or
4//! by convention, and named in every error when it is not.
5//!
6//! # Why nothing is compiled in
7//!
8//! A kind is a fact about a repository's layer, not about the executable that reads it. If
9//! the kinds were compiled in, adding one would mean shipping a binary, and a repository
10//! that added its own would be adding it to somebody else's program. Reading them at run
11//! time is what makes `.ai/repo/knowledge/` able to declare a kind that the same binary
12//! then serves.
13//!
14//! ```
15//! use majordomus_cli::share::Share;
16//!
17//! // located explicitly, and the paths under it are derived rather than joined by callers
18//! let dir = tempfile::tempdir().unwrap();
19//! std::fs::write(dir.path().join("kinds.yaml"), "version: 1\nkinds: []\n").unwrap();
20//! std::fs::create_dir(dir.path().join("schemas")).unwrap();
21//!
22//! let share = Share::locate(Some(dir.path()), dir.path()).unwrap();
23//! assert_eq!(share.origin, "--share", "it says how it was found");
24//! assert!(share.kinds_path().ends_with("kinds.yaml"));
25//! assert!(share.schemas_dir().ends_with("schemas"));
26//! assert_eq!(share.kinds_path().parent(), Some(share.dir()));
27//!
28//! // and when there is none, the error names where it looked
29//! let nowhere = tempfile::tempdir().unwrap();
30//! let err = Share::locate(None, nowhere.path()).unwrap_err();
31//! assert_eq!(err.exit_code(), 12);
32//! ```

68/// A located distribution directory.
69#[derive(Debug, Clone, PartialEq, Eq)]
70pub struct Share {
71    dir: PathBuf,
72    /// How it was found, for diagnostics: `--share`, the environment, the repository, or
73    /// the executable's own installation.
74    pub origin: &'static str,
75}
76
77impl Share {
78    /// Locate the share directory. In order: `explicit` (from `--share`), `MAJORDOMUS_SHARE`,
79    /// `<repository root>/share` when it holds a kinds file (this repository supervising
80    /// itself), and `<directory of the executable>/../share` (an installation laid out as
81    /// `bin/` beside `share/`). A candidate that exists but holds no kinds file is skipped
82    /// only when it came from a convention; an explicit path that lacks one is an error.
83    pub fn locate(explicit: Option<&Path>, repo_root: &Path) -> Result<Self> {
84        let mut tried = Vec::new();
85        if let Some(p) = explicit {
86            return Self::open(p, "--share");
87        }
88        if let Some(p) = std::env::var_os(SHARE_ENV) {
89            return Self::open(Path::new(&p), SHARE_ENV);
90        }
91        let in_repo = repo_root.join("share");
92        if in_repo.join(KINDS_FILE).is_file() {
93            return Self::open(&in_repo, "repository");
94        }
95        tried.push(in_repo);
96        if let Ok(exe) = std::env::current_exe() {
97            if let Some(dir) = exe.parent() {
98                let beside = dir.join("../share");
99                if beside.join(KINDS_FILE).is_file() {
100                    return Self::open(&beside, "installation");
101                }
102                tried.push(beside);
103            }
104        }
105        Err(Error::ShareNotFound { tried })
106    }
107
108    fn open(dir: &Path, origin: &'static str) -> Result<Self> {
109        let kinds = dir.join(KINDS_FILE);
110        if !kinds.is_file() {
111            return Err(Error::ShareNotFound {
112                tried: vec![dir.to_path_buf()],
113            });
114        }
115        let dir = dir.canonicalize().map_err(|e| Error::io(dir, e))?;
116        Ok(Share { dir, origin })
117    }
118
119    /// The directory, canonical.
120    pub fn dir(&self) -> &Path {
121        &self.dir
122    }

160    /// The providers the tool has a template for, by id (the file stem), sorted. A
161    /// distribution without the directory has none; that is not an error, it is a
162    /// distribution that ships no adapter.
163    pub fn provider_templates(&self) -> Vec<String> {
164        let Ok(entries) = std::fs::read_dir(self.providers_dir()) else {
165            return Vec::new();
166        };
167        let mut out: Vec<String> = entries
168            .filter_map(|e| e.ok())
169            .filter_map(|e| e.file_name().to_str().map(str::to_string))
170            .filter_map(|n| n.strip_suffix(".tmpl").map(str::to_string))
171            .collect();
172        crate::order::canonical(&mut out);
173        out
174    }

176    /// `<share>/providers.yaml`.
177    pub fn providers_path(&self) -> PathBuf {
178        self.dir.join(PROVIDERS_FILE)
179    }
180
181    /// The provider declarations the distribution ships, joined with its templates: one
182    /// [`ProviderDeclaration`] per id that has either, sorted by id. A distribution without
183    /// the file declares nothing and every template is a provider named by its file; a
184    /// file that does not parse is an error naming it, because a declaration half-read is
185    /// worse than none.
186    pub fn providers(&self) -> Result<ProviderDeclarations> {
187        let path = self.providers_path();
188        let file: ProvidersFile = match std::fs::read_to_string(&path) {
189            Ok(text) => {
190                crate::metadata::yaml::parse_into(&text).map_err(|reason| Error::KindSchema {
191                    reason: format!("{}: {reason}", path.display()),
192                })?
193            }
194            Err(e) if e.kind() == std::io::ErrorKind::NotFound => ProvidersFile::default(),
195            Err(e) => return Err(Error::io(&path, e)),
196        };
197        let templates = self.provider_templates();
198        let mut ids: Vec<String> = templates.clone();
199        ids.extend(file.providers.keys().cloned());
200        crate::order::canonical(&mut ids);
201        ids.dedup();
202        let providers = ids
203            .into_iter()
204            .map(|id| {
205                let decl = file.providers.get(&id);
206                ProviderDeclaration {
207                    title: decl
208                        .and_then(|d| d.title.clone())
209                        .unwrap_or_else(|| id.clone()),
210                    client_config: decl.and_then(|d| d.client_config.clone()),
211                    scratch_roots: decl.map(|d| d.scratch_roots.clone()).unwrap_or_default(),
212                    lifecycle: decl.map(|d| d.lifecycle.clone()).unwrap_or_default(),
213                    prompt_capture: decl.map(|d| d.prompt_capture).unwrap_or(false),
214                    offers: decl
215                        .and_then(|d| d.offers.as_ref())
216                        .map(offers_of)
217                        .unwrap_or_default(),
218                    template: templates.contains(&id),
219                    declared: decl.is_some(),
220                    id,
221                }
222            })
223            .collect();
224        Ok(ProviderDeclarations {
225            scratch_roots: file.scratch_roots,
226            providers,
227        })
228    }
229}

298    ProviderOffers {

301        prompts: e.prompts.unwrap_or_default(),
302        prompts_evidence: e.prompts_evidence.clone().unwrap_or_default(),
303        browser: e.browser.unwrap_or_default(),
304        browser_evidence: e.browser_evidence.clone().unwrap_or_default(),
305        browser_host: e.browser_host.clone(),
306        browser_setup: e.browser_setup.clone(),
307        usage: e.usage.clone().unwrap_or_default(),
308        usage_evidence: e.usage_evidence.clone().unwrap_or_default(),
309    }

312/// How an episode boundary can be drawn for one provider.
313///
314/// Three words, because three things are true of different providers and collapsing them
315/// into "supported / not supported" is what let this tool report one provider and stay
316/// silent about five others. A provider that fires its own events is not the same as one
317/// that fires none but speaks MCP, and neither is the same as one that can do nothing.
318///
319/// The words are the declaration's own, and a word outside them is refused rather than read
320/// as `none`:
321///
322/// ```
323/// use majordomus_cli::share::EpisodeSource;
324/// let read: EpisodeSource = serde_json::from_str("\"connection\"").unwrap();
325/// assert_eq!(read, EpisodeSource::Connection);
326/// assert_eq!(EpisodeSource::default(), EpisodeSource::None, "undeclared reads as none");
327/// assert!(serde_json::from_str::<EpisodeSource>("\"hook\"").is_err(), "hooks, plural");
328/// ```
329#[derive(
330    Debug,
331    Clone,
332    Copy,
333    Default,
334    PartialEq,
335    Eq,
336    serde::Serialize,
337    serde::Deserialize,
338    schemars::JsonSchema,
339)]
340#[serde(rename_all = "snake_case")]
341pub enum EpisodeSource {
342    /// The provider fires its own session events and a hook this tool installs draws the
343    /// boundary from them. The strongest form: it sees a window somebody closed.
344    Hooks,
345    /// The provider fires no session event, but its client attaches to this repository's
346    /// shared MCP server, and the connection is the boundary (ADR 0103).
347    Connection,
348    /// Neither. Nothing can say when this provider's episode began or ended, and the tool
349    /// reports that rather than implying a capability nobody has.
350    #[default]
351    None,
352}
353
354impl EpisodeSource {
355    /// The word as the declaration writes it, which is also how it serialises; `capture
356    /// status` prints it beside the evidence that backs it.
357    ///
358    /// ```
359    /// use majordomus_cli::share::EpisodeSource;
360    /// for s in [EpisodeSource::Hooks, EpisodeSource::Connection, EpisodeSource::None] {
361    ///     assert_eq!(serde_json::to_string(&s).unwrap(), format!("\"{}\"", s.as_str()));
362    /// }
363    /// ```
364    pub fn as_str(self) -> &'static str {
365        match self {
366            EpisodeSource::Hooks => "hooks",
367            EpisodeSource::Connection => "connection",
368            EpisodeSource::None => "none",
369        }
370    }
371}

398pub enum PromptSource {

422/// Whether one provider's client drives a browser in the same session as this repository's
423/// MCP server.
424///
425/// A bridge is the one way a single session both works on a page and reads and changes the
426/// live mesh (`docs/MCP.md`, "Claude in the browser"). It is a vendor capability like the two
427/// above, so it is declared with its citation; whether *this machine* has the bridge is a
428/// different fact, which preflight reads from the machine and never from here. A browser
429/// agent of the vendor's own — a side panel that reaches only the vendor's cloud — is not a
430/// bridge: nothing it does arrives in a session that holds this repository's server.
431///
432/// ```
433/// use majordomus_cli::share::BrowserSource;
434/// assert_eq!(serde_json::from_str::<BrowserSource>("\"bridge\"").unwrap(), BrowserSource::Bridge);
435/// assert!(serde_json::from_str::<BrowserSource>("\"extension\"").is_err(), "one word for it");
436/// assert_eq!(BrowserSource::default(), BrowserSource::None, "undeclared reads as none");
437/// assert_eq!(BrowserSource::Bridge.as_str(), "bridge");
438/// ```
439#[derive(
440    Debug,
441    Clone,
442    Copy,
443    Default,
444    PartialEq,
445    Eq,
446    serde::Serialize,
447    serde::Deserialize,
448    schemars::JsonSchema,
449)]
450#[serde(rename_all = "snake_case")]
451pub enum BrowserSource {
452    /// The client drives a browser through a vendor extension whose tools arrive in the same
453    /// session as this repository's MCP server.
454    Bridge,
455    /// It does not, or not in a session that holds this repository's server.
456    #[default]
457    None,
458}

476/// What one provider can do about the session lifecycle, and where that was verified.
477///
478/// **Every cell carries its evidence.** A capability asserted with no citation is the defect
479/// this type exists to make impossible: before it, a provider absent from a shell table was
480/// silently unsupported and a provider present in one was silently capable, and no surface
481/// could tell a reader which vendor page either answer came from — `capture status`
482/// enumerated the one provider that had an adapter and said nothing whatever about the
483/// other five this distribution declares. The rule
484/// `project.a-provider-capability-cites-its-evidence` refuses a declaration without one.
485///
486/// A provider that declares nothing is `none` on both axes with empty evidence — the state
487/// that rule refuses, so the gap is a finding rather than a silent "unsupported":
488///
489/// ```
490/// use majordomus_cli::share::{EpisodeSource, PromptSource, ProviderOffers};
491/// let undeclared = ProviderOffers::default();
492/// assert_eq!((undeclared.episode, undeclared.prompts), (EpisodeSource::None, PromptSource::None));
493/// assert!(undeclared.episode_evidence.is_empty() && undeclared.prompts_evidence.is_empty());
494/// let generic: ProviderOffers = serde_json::from_value(serde_json::json!({
495///     "episode": "connection", "episode_evidence": "ADR 0103",
496///     "prompts": "none", "prompts_evidence": "MCP carries no prompt",
497/// })).unwrap();
498/// assert_eq!(generic.episode, EpisodeSource::Connection);
499/// ```
500#[derive(
501    Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
502)]
503pub struct ProviderOffers {
504    /// How the episode boundary is drawn for this provider.
505    pub episode: EpisodeSource,
506    /// Where `episode` was verified: a vendor documentation URL, or the statement that no
507    /// such documentation exists and what was searched for it.
508    pub episode_evidence: String,
509    /// Whether the raw prompt can be captured below the model.
510    pub prompts: PromptSource,
511    /// Where `prompts` was verified, on the same terms as `episode_evidence`.
512    pub prompts_evidence: String,
513    /// Whether the client drives a browser beside this repository's server. Absent in a
514    /// value written before the axis existed, and read as `none` there.
515    #[serde(default)]
516    pub browser: BrowserSource,
517    /// Where `browser` was verified, on the same terms as `episode_evidence`.
518    #[serde(default)]
519    pub browser_evidence: String,
520    /// The native messaging host a bridge registers with the browser, which is how a
521    /// machine is seen to have it. Declared for a `bridge` only.
522    #[serde(default, skip_serializing_if = "Option::is_none")]
523    pub browser_host: Option<String>,
524    /// What a person does, once, to install the bridge on a machine, in the vendor's terms.
525    /// Declared for a `bridge` only.
526    #[serde(default, skip_serializing_if = "Option::is_none")]
527    pub browser_setup: Option<String>,
528    /// The official seams through which the provider reports what a session consumed (I3287,
529    /// ADR 0213), every one it offers: `none` alone when it offers nothing a receiver could
530    /// read. Empty in a value written before the axis existed, which declares nothing.
531    #[serde(default)]
532    pub usage: Vec<UsageSource>,
533    /// Where `usage` was verified, on the same terms as `episode_evidence`.
534    #[serde(default)]
535    pub usage_evidence: String,
536}

538/// One official seam through which a provider reports what a session consumed.
539///
540/// Usage is read only where the provider itself reports it (ADR 0213): an official
541/// OpenTelemetry export, a protocol notification of the client's own server, or the usage a
542/// non-interactive run prints in its structured output. Never a transcript, a proxy, or a
543/// patched client. `none` is the declared answer for a provider that offers no such seam,
544/// and it stands alone.
545///
546/// ```
547/// use majordomus_cli::share::UsageSource;
548/// assert_eq!(serde_json::from_str::<UsageSource>("\"otel\"").unwrap(), UsageSource::Otel);
549/// assert_eq!(serde_json::from_str::<UsageSource>("\"app-server\"").unwrap(), UsageSource::AppServer);
550/// assert!(serde_json::from_str::<UsageSource>("\"transcript\"").is_err(), "no transcript seam");
551/// assert_eq!(UsageSource::StreamJson.as_str(), "stream-json");
552/// ```
553#[derive(
554    Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
555)]
556#[serde(rename_all = "kebab-case")]
557pub enum UsageSource {
558    /// The provider's own OpenTelemetry export, which the person enables.
559    Otel,
560    /// A notification of the client's own app server protocol.
561    AppServer,
562    /// The usage a non-interactive run reports in its structured (`stream-json`) output.
563    StreamJson,
564    /// No official seam reports usage.
565    None,
566}
567
568impl UsageSource {
569    /// The word as the declaration writes it, which is also how it serialises.
570    ///
571    /// ```
572    /// use majordomus_cli::share::UsageSource;
573    /// assert_eq!(UsageSource::None.as_str(), "none");
574    /// assert_eq!(serde_json::to_string(&UsageSource::AppServer).unwrap(), "\"app-server\"");
575    /// ```
576    pub fn as_str(self) -> &'static str {
577        match self {
578            UsageSource::Otel => "otel",
579            UsageSource::AppServer => "app-server",
580            UsageSource::StreamJson => "stream-json",
581            UsageSource::None => "none",
582        }
583    }
584}

595/// One provider: the template's file stem, and what the declaration adds to it.
596#[derive(
597    Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
598)]
599pub struct ProviderDeclaration {
600    /// The id: the template's file stem, and the key in the declarations.
601    pub id: String,
602    /// The name a person knows it by; the id when nothing is declared.
603    pub title: String,
604    /// The file it reads a project-scoped MCP client configuration from, when it has one.
605    #[serde(default, skip_serializing_if = "Option::is_none")]
606    pub client_config: Option<String>,
607    /// The scratch roots it creates checkouts under, unexpanded.
608    #[serde(default)]
609    pub scratch_roots: Vec<String>,
610    /// The lifecycle events the tool's adapter for it declares, in the provider's own
611    /// vocabulary (`SessionStart`, `SessionEnd`, `PreCompact`). Empty means the tool ships
612    /// no lifecycle adapter for it, which costs a worker the automation and none of the
613    /// model. Declared, because the two other ways to answer this question are both wrong:
614    /// reading `.claude/hooks/` reports an installation as a capability, and a list written
615    /// into a page is a second source of truth that goes stale the day an adapter changes.
616    #[serde(default)]
617    pub lifecycle: Vec<String>,
618    /// Whether the tool's adapter for it can archive the worker's prompts.
619    #[serde(default)]
620    pub prompt_capture: bool,
621    /// What the provider itself can do about the session lifecycle, and where each answer
622    /// was verified: the vendor's capability, where `lifecycle` is this tool's adapter. A
623    /// provider with no `offers:` block declares nothing, which reads as `none` with no
624    /// evidence — the state the capability rule refuses, so that the gap is a finding
625    /// rather than a silent "unsupported".
626    #[serde(default)]
627    pub offers: ProviderOffers,
628    /// The distribution ships a template for it.
629    pub template: bool,
630    /// The distribution declares it.
631    pub declared: bool,
632}
633
634impl ProviderDeclarations {
635    /// Every scratch root — the tool's own and every provider's — expanded against
636    /// `primary` (the repository's primary checkout) and the environment, in declaration
637    /// order, with the provider that declared each. A root whose expansion needs a variable
638    /// the environment does not hold is skipped, not an error.
639    pub fn scratch_roots(&self, primary: &Path) -> Vec<(Option<String>, PathBuf)> {
640        let mut out = Vec::new();
641        for r in &self.scratch_roots {
642            if let Some(p) = expand_root(r, primary) {
643                out.push((None, p));
644            }
645        }
646        for p in &self.providers {
647            for r in &p.scratch_roots {
648                if let Some(path) = expand_root(r, primary) {
649                    out.push((Some(p.id.clone()), path));
650                }
651            }
652        }
653        out
654    }
655}

657/// Expand one declared root: `<primary>/`, `~/`, `${NAME:-default}` and `$NAME`, nothing
658/// else. `None` when a variable without a default is unset.
659///
660/// ```
661/// # use std::path::Path;
662/// use majordomus_cli::share::expand_root;
663/// std::env::set_var("MJ_DOCTEST_ROOT", "/srv/x");
664/// std::env::remove_var("MJ_DOCTEST_UNSET");
665/// let primary = Path::new("/repo");
666/// assert_eq!(expand_root("<primary>/.claude/worktrees", primary).unwrap(), Path::new("/repo/.claude/worktrees"));
667/// assert_eq!(expand_root("$MJ_DOCTEST_ROOT/wt", primary).unwrap(), Path::new("/srv/x/wt"));
668/// assert_eq!(expand_root("${MJ_DOCTEST_UNSET:-/d}/wt", primary).unwrap(), Path::new("/d/wt"));
669/// assert!(expand_root("$MJ_DOCTEST_UNSET/wt", primary).is_none());
670/// assert_eq!(expand_root("/tmp", primary).unwrap(), Path::new("/tmp"));
671/// ```
672pub fn expand_root(root: &str, primary: &Path) -> Option<PathBuf> {
673    let mut out = String::new();
674    let mut rest = root;
675    while !rest.is_empty() {
676        if let Some(r) = rest.strip_prefix("<primary>") {
677            out.push_str(&primary.to_string_lossy());
678            rest = r;
679        } else if let Some(r) = rest.strip_prefix('~') {
680            let home = std::env::var_os("HOME")?;
681            out.push_str(&home.to_string_lossy());
682            rest = r;
683        } else if let Some(r) = rest.strip_prefix("${") {
684            let end = r.find('}')?;
685            let (name, default) = match r[..end].split_once(":-") {
686                Some((n, d)) => (n, Some(d)),
687                None => (&r[..end], None),
688            };
689            match std::env::var(name) {
690                Ok(v) if !v.is_empty() => {
691                    out.push_str(&expand_root(&v, primary)?.to_string_lossy())
692                }
693                _ => out.push_str(&expand_root(default?, primary)?.to_string_lossy()),
694            }
695            rest = &r[end + 1..];
696        } else if let Some(r) = rest.strip_prefix('$') {
697            let end = r
698                .find(|c: char| !(c.is_ascii_alphanumeric() || c == '_'))
699                .unwrap_or(r.len());
700            let v = std::env::var(&r[..end]).ok().filter(|v| !v.is_empty())?;
701            out.push_str(&v);
702            rest = &r[end..];
703        } else {
704            let end = rest.find(['<', '~', '$']).unwrap_or(rest.len());
705            let end = if end == 0 { 1 } else { end };
706            out.push_str(&rest[..end]);
707            rest = &rest[end..];
708        }
709    }
710    Some(PathBuf::from(out))
711}

713/// Every `<vendor>/<name>/<name>.v<n>.schema.json` under a schema root, parsed, keyed by
714/// the schema identity its path derives (`majordomus.adr/v1`) and sorted by it. A root that
715/// does not exist yields nothing; that is the repository's case when it adds no schema of
716/// its own.
717///
718/// The identity and the path fix each other, so a schema can be found from a `kinds.yaml`
719/// entry without a directory scan and a file cannot quietly answer to a name it does not
720/// declare. A file whose path does not have the shape is refused by name rather than
721/// skipped, because a schema nobody loads validates nothing and says so nowhere.
722pub fn read_schema_dir(dir: &Path) -> Result<Vec<(String, serde_json::Value)>> {
723    let mut schemas = Vec::new();
724    read_schema_tree(dir, dir, &mut schemas)?;
725    schemas.sort_by(|a, b| a.0.cmp(&b.0));
726    Ok(schemas)
727}
728
729fn read_schema_tree(
730    root: &Path,
731    dir: &Path,
732    out: &mut Vec<(String, serde_json::Value)>,
733) -> Result<()> {
734    if !dir.is_dir() {
735        return Ok(());
736    }
737    for entry in std::fs::read_dir(dir).map_err(|e| Error::io(dir, e))? {
738        let path = entry.map_err(|e| Error::io(dir, e))?.path();
739        if path.is_dir() {
740            // `<schemas>/generated/` holds the contracts of the *generated documents* —
741            // the manifest, the registry, the benchmark matrix. Those are not kinds of the
742            // layer, carry no `<vendor>/<name>/` identity, and are read by
743            // [`read_generated_schema_dir`] instead.
744            if dir == root
745                && path.file_name().and_then(|f| f.to_str()) == Some(GENERATED_SCHEMAS_DIR)
746            {
747                continue;
748            }
749            read_schema_tree(root, &path, out)?;
750            continue;
751        }
752        let Some(file) = path.file_name().and_then(|f| f.to_str()) else {
753            continue;
754        };
755        if !file.ends_with(SCHEMA_SUFFIX) {
756            continue;
757        }
758        let relative = path
759            .strip_prefix(root)
760            .map(|p| p.to_string_lossy().replace('\\', "/"))
761            .unwrap_or_else(|_| file.to_string());
762        let identity = schema_identity(&relative).ok_or_else(|| Error::KindSchema {
763            reason: format!(
764                "{}: a schema's path is <vendor>/<name>/<name>.v<n>{SCHEMA_SUFFIX}, which this is not",
765                path.display()
766            ),
767        })?;
768        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
769        let json: serde_json::Value =
770            serde_json::from_str(&text).map_err(|e| Error::KindSchema {
771                reason: format!("{}: not JSON: {e}", path.display()),
772            })?;
773        out.push((identity, json));
774    }
775    Ok(())
776}

778/// Every `<name>.schema.json` directly under a directory, parsed, keyed by that `<name>`
779/// and sorted by it. A directory that does not exist yields nothing.
780///
781/// This is the reader for `<share>/schemas/generated/`, whose files are the contracts of
782/// the *generated documents* rather than of the layer's kinds: they answer to the document
783/// id they pin in `properties.schema.const`, not to a `<vendor>/<name>` path, so the shape
784/// [`read_schema_dir`] insists on does not apply to them.
785pub fn read_generated_schema_dir(dir: &Path) -> Result<Vec<(String, serde_json::Value)>> {
786    let mut out = Vec::new();
787    if !dir.is_dir() {
788        return Ok(out);
789    }
790    for entry in std::fs::read_dir(dir).map_err(|e| Error::io(dir, e))? {
791        let path = entry.map_err(|e| Error::io(dir, e))?.path();
792        let Some(file) = path.file_name().and_then(|f| f.to_str()) else {
793            continue;
794        };
795        if path.is_dir() || !file.ends_with(SCHEMA_SUFFIX) {
796            continue;
797        }
798        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
799        let json: serde_json::Value =
800            serde_json::from_str(&text).map_err(|e| Error::KindSchema {
801                reason: format!("{}: not JSON: {e}", path.display()),
802            })?;
803        out.push((file[..file.len() - SCHEMA_SUFFIX.len()].to_string(), json));
804    }
805    out.sort_by(|a, b| a.0.cmp(&b.0));
806    Ok(out)
807}

809/// The identity a schema path carries: `majordomus/adr/adr.v1.schema.json` is
810/// `majordomus.adr/v1`. A path of another shape yields nothing.
811///
812/// ```
813/// use majordomus_cli::share::schema_identity;
814/// assert_eq!(
815///     schema_identity("majordomus/adr/adr.v1.schema.json").as_deref(),
816///     Some("majordomus.adr/v1")
817/// );
818/// assert_eq!(schema_identity("adr.schema.json"), None);
819/// assert_eq!(schema_identity("majordomus/adr/other.v1.schema.json"), None);
820/// ```
821pub fn schema_identity(relative: &str) -> Option<String> {
822    let stem = relative.strip_suffix(SCHEMA_SUFFIX)?;
823    let mut parts = stem.split('/');
824    let vendor = parts.next()?;
825    let name = parts.next()?;
826    let file = parts.next()?;
827    if parts.next().is_some() || vendor.is_empty() || name.is_empty() {
828        return None;
829    }
830    let version = file.strip_prefix(&format!("{name}."))?;
831    if !version.starts_with('v') || version.len() < 2 {
832        return None;
833    }
834    Some(format!("{vendor}.{name}/{version}"))
835}
