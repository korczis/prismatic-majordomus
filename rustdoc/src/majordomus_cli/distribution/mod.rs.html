1//! The distribution model: how Majordomus is packaged, published and installed.
2//!
3//! One file, `share/distribution.yaml`, declares the binary's name, the repository it is
4//! published from, the installer's canonical URL and defaults, how an archive is named,
5//! and every target that may be built. Everything else in this repository that has an
6//! opinion about a platform, an artifact name or an installation URL is a projection of
7//! this module: the release build matrix, the installer's target table, the installation
8//! guide, the site's install commands and the public release metadata.
9//!
10//! The rule the projections exist to keep is in `docs/DISTRIBUTION.md` and in the project
11//! rule `project.distribution-canonical`: adding a target is one edit to the model and a
12//! regeneration, and a tree in which any projection disagrees with the model is refused.
13//!
14//! Note on the word: `share/` is the tool's data directory (see [`crate::share`]), and
15//! "the distribution model" here is about *shipping the tool*, not about that directory.
16

107/// What the project promises about a target.
108///
109/// See [`crate::deploy::Status`]: both are `Status` in their own module and neither is in
110/// the one component namespace the schema document has.
111#[derive(
112    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
113)]
114#[serde(rename_all = "lowercase")]
115#[schemars(rename = "TargetStatus")]
116pub enum Status {
117    /// Built by every release, offered by the installer, listed as supported. A release
118    /// that is missing this artifact is not published.
119    Supported,
120    /// Built and offered, and documented as not yet proven.
121    Experimental,
122    /// Documented, never built, refused by the installer with its recorded reason.
123    Unavailable,
124}
125
126impl Status {
127    /// Is an artifact of this target built and published?
128    pub fn is_published(self) -> bool {
129        matches!(self, Status::Supported | Status::Experimental)
130    }
131}

277impl Target {
278    /// The packaging format, the model's default unless this target overrides it.
279    pub fn format(&self, archive: &Archive) -> Format {
280        self.format.unwrap_or(archive.format)
281    }
282
283    /// The file extension of this target's artifact.
284    pub fn extension(&self, archive: &Archive) -> String {
285        match self.format(archive) {
286            f if f == archive.format => archive.extension.clone(),
287            Format::TarGz => "tar.gz".to_string(),
288            Format::Zip => "zip".to_string(),
289        }
290    }
291
292    /// The artifact's file name: the one naming function, `{binary}-{tag}-{target}.{extension}`.
293    /// `tag` is the git tag, `v` and the version.
294    pub fn artifact_name(&self, project: &Project, archive: &Archive, tag: &str) -> String {
295        format!(
296            "{}-{}-{}.{}",
297            project.binary,
298            tag,
299            self.rust_target,
300            self.extension(archive)
301        )
302    }
303
304    /// The directory the archive unpacks into: the artifact's name without its extension.
305    pub fn archive_root(&self, project: &Project, tag: &str) -> String {
306        format!("{}-{}-{}", project.binary, tag, self.rust_target)
307    }
308
309    /// How this target is written in prose: `macOS ARM64`, `Linux x86_64 musl`.
310    pub fn title(&self, titles: &Titles) -> String {
311        let os = match self.os {
312            Os::Macos => &titles.os.macos,
313            Os::Linux => &titles.os.linux,
314            Os::Windows => &titles.os.windows,
315        };
316        let arch = match self.arch {
317            Arch::X86_64 => &titles.arch.x86_64,
318            Arch::Aarch64 => &titles.arch.aarch64,
319        };
320        match self.libc {
321            Some(Libc::Gnu) => format!("{os} {arch} {}", titles.libc.gnu),
322            Some(Libc::Musl) => format!("{os} {arch} {}", titles.libc.musl),
323            None => format!("{os} {arch}"),
324        }
325    }
326}

328/// The whole model.
329#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
330#[serde(deny_unknown_fields)]
331pub struct Model {
332    /// The contract this file follows.
333    pub schema: String,
334    /// What is installed and where it comes from.
335    pub project: Project,
336    /// The installer's canonical URL and defaults.
337    pub installer: Installer,
338    /// How an artifact is packaged, named and verified.
339    pub archive: Archive,
340    /// How each vocabulary term is written in prose.
341    pub titles: Titles,
342    /// Every target, in declaration order.
343    pub targets: Vec<Target>,
344}
345
346impl Model {
347    /// Read the model from the tool's data directory: the file an installed copy carries.
348    pub fn load(share: &Share) -> Result<Self> {
349        let path = share.dir().join(FILE);
350        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
351        Self::parse(&text).map_err(|reason| Error::InvalidDistribution {
352            path: path.display().to_string(),
353            reason,
354        })
355    }
356
357    /// The model, when the tool's data directory carries one. A directory without the file
358    /// is a tool that declares no distribution, which is not an error; a file that does not
359    /// parse or breaks an invariant is.
360    pub fn locate(share: &Share) -> Result<Option<Self>> {
361        if !share.dir().join(FILE).is_file() {
362            return Ok(None);
363        }
364        Self::load(share).map(Some)
365    }
366
367    /// Read the model from the repository's index, where it is an object of kind
368    /// `distribution`. This is what a capability sees; [`Model::load`] is what the command
369    /// line and the generator see, and `distribution::tests` proves the two agree.
370    pub fn from_index(index: &Index) -> std::result::Result<Self, String> {
371        let object = index
372            .objects
373            .iter()
374            .find(|o| o.kind == KIND)
375            .ok_or_else(|| format!("this repository indexes no object of kind `{KIND}`"))?;
376        let model: Model = serde_json::from_value(object.metadata.clone())
377            .map_err(|e| format!("{}: {e}", object.provenance.path))?;
378        model.check().map(|()| model)
379    }
380
381    /// Parse the model's YAML subset and check its invariants.
382    pub fn parse(text: &str) -> std::result::Result<Self, String> {
383        let model: Model = yaml::parse_into(text)?;
384        model.check().map(|()| model)
385    }
386
387    /// Every invariant the types cannot state. The first failure is returned, named.
388    pub fn check(&self) -> std::result::Result<(), String> {
389        let findings = self.findings();
390        match findings.first() {
391            Some(first) => Err(if findings.len() == 1 {
392                first.clone()
393            } else {
394                format!("{first} (and {} more)", findings.len() - 1)
395            }),
396            None => Ok(()),
397        }
398    }
399
400    /// Every invariant violation, in a stable order. This is what
401    /// `majordomus distribution validate` reports and what the drift gate refuses.
402    pub fn findings(&self) -> Vec<String> {
403        let mut out = Vec::new();
404        if self.schema != SCHEMA {
405            out.push(format!(
406                "schema is `{}`, not `{SCHEMA}`; a tool that reads an older format must refuse rather than guess",
407                self.schema
408            ));
409        }
410        if self.archive.name != NAME_TEMPLATE {
411            out.push(format!(
412                "archive.name is `{}`; the one naming function is `{NAME_TEMPLATE}`",
413                self.archive.name
414            ));
415        }
416        if self.archive.checksum != "sha256" {
417            out.push(format!(
418                "archive.checksum is `{}`; the installer verifies sha256 and nothing else",
419                self.archive.checksum
420            ));
421        }
422        if !self.installer.base_url.starts_with("https://") {
423            out.push(format!(
424                "installer.base_url `{}` is not https; the installer refuses any other scheme",
425                self.installer.base_url
426            ));
427        }
428        if self.installer.base_url.ends_with('/') {
429            out.push(format!(
430                "installer.base_url `{}` ends with a slash; every URL is built by appending one",
431                self.installer.base_url
432            ));
433        }
434        if self.project.repository.split('/').count() != 2 {
435            out.push(format!(
436                "project.repository `{}` is not owner/name",
437                self.project.repository
438            ));
439        }
440        if self.targets.is_empty() {
441            out.push("targets is empty; the installer would offer nothing".to_string());
442        }
443        let mut ids: Vec<&str> = Vec::new();
444        let mut triples: Vec<&str> = Vec::new();
445        let mut names: Vec<String> = Vec::new();
446        for t in &self.targets {
447            if ids.contains(&t.id.as_str()) {
448                out.push(format!("two targets claim the id `{}`", t.id));
449            }
450            ids.push(&t.id);
451            if triples.contains(&t.rust_target.as_str()) {
452                out.push(format!(
453                    "two targets claim the Rust target `{}`",
454                    t.rust_target
455                ));
456            }
457            triples.push(&t.rust_target);
458            let name = t.artifact_name(&self.project, &self.archive, "v0.0.0");
459            if names.contains(&name) {
460                out.push(format!(
461                    "target `{}` derives the artifact name `{name}`, which another target already derives",
462                    t.id
463                ));
464            }
465            names.push(name);
466            match (t.os, t.libc) {
467                (Os::Linux, None) => out.push(format!(
468                    "target `{}` is linux and declares no libc; the installer cannot resolve it",
469                    t.id
470                )),
471                (Os::Linux, Some(_)) | (_, None) => {}
472                (_, Some(_)) => out.push(format!(
473                    "target `{}` is not linux and declares a libc",
474                    t.id
475                )),
476            }
477            if t.status.is_published() && t.build.is_none() {
478                out.push(format!(
479                    "target `{}` is {:?} and declares no build; nothing would build it",
480                    t.id, t.status
481                ));
482            }
483            if !t.status.is_published() && t.reason.is_none() {
484                out.push(format!(
485                    "target `{}` is not published and records no reason",
486                    t.id
487                ));
488            }
489            if !t.status.is_published() && t.build.is_some() {
490                out.push(format!(
491                    "target `{}` is not published and declares a build",
492                    t.id
493                ));
494            }
495        }
496        out
497    }
498
499    /// Every target a release builds and the installer offers.
500    pub fn published(&self) -> impl Iterator<Item = &Target> {
501        self.targets.iter().filter(|t| t.status.is_published())
502    }
503
504    /// A target by id.
505    pub fn target(&self, id: &str) -> Option<&Target> {
506        self.targets.iter().find(|t| t.id == id)
507    }
508
509    /// The installer's canonical URL.
510    pub fn installer_url(&self) -> String {
511        format!("{}/{}", self.installer.base_url, self.installer.script)
512    }
513
514    /// The one-line install command, composed from its parts and written nowhere else.
515    pub fn install_command(&self) -> String {
516        format!(
517            "{} {} | {}",
518            self.installer.download_command,
519            self.installer_url(),
520            self.installer.shell
521        )
522    }
523
524    /// The one-line install command that also initialises the repository afterwards.
525    pub fn install_and_init_command(&self) -> String {
526        format!("{} -s -- --init", self.install_command())
527    }
528
529    /// The URL of the release metadata for a tag; `latest` is the stable pointer.
530    pub fn release_url(&self, tag: &str) -> String {
531        format!("{}/releases/{tag}.json", self.installer.base_url)
532    }
533}
