1//! The one rule set that makes text a run printed fit to publish.
2//!
3//! Evidence carries text nobody wrote for a reader outside the machine it ran on: the reason
4//! a case skipped, the tail of a failure, a command's captured output. Any of it can hold a
5//! credential the command echoed or a path that names the account it ran under, and once it
6//! is on a public page it cannot be taken back. So every such text goes through
7//! [`public_text`], and nothing else decides what is fit to publish.
8//!
9//! The credential shapes are not a second table. They are a port of `MJ_CAPTURE_SECRETS` and
10//! `MJ_CAPTURE_SECRET_ASSIGN` in `lib/capture.sh`, the table the prompt archive has redacted
11//! with from the start, applied the way `mj_capture_redact` applies it: each shape in table
12//! order over the whole text, every leftmost-longest match replaced by a marker naming the
13//! shape, and then the assignment rule, which keeps the name a value was assigned to. Two
14//! tests hold the port to the table: one reads `lib/capture.sh` and compares the shapes, and
15//! one fixture, `test/fixtures/redaction/shapes.tsv`, is run through both implementations.
16//! The crate has no regular-expression engine, so the shapes are matched by the small
17//! hand-written matcher below, which knows exactly the constructs the table uses.
18//!
19//! Two neighbours are different concerns and stay apart: `execution::redact` removes the
20//! fields a capability's input schema marks sensitive, and `generate::forbidden_in` is the
21//! last word on a published artifact, which [`public_text`] asks after it has done its work.
22//! Before asking, [`public_text`] refuses a text that still carries a secret no shape covers:
23//! one whose announcing shape fired, in this call or in an earlier redaction whose marker
24//! the text carries, and one whose armour or name is left once the shapes have run. Redacting
25//! the announcement is what would let the rest past that check.
26//!
27//! ```
28//! use majordomus_cli::redaction::redact_secrets;
29//!
30//! // assembled at run time, so that no committed file carries a credential's shape
31//! let token = format!("{}{}", "ghp_", "a".repeat(24));
32//! let out = redact_secrets(&format!("pushing with {token} failed"));
33//! assert_eq!(out.text, "pushing with [redacted:github-token] failed");
34//! assert_eq!(out.kinds, ["github-token"]);
35//! ```
36

39/// Text after [`redact_secrets`] or [`public_text`], with the shapes that fired in it.
40///
41/// `kinds` is what `mj_capture_redacted_kinds` reports for the same text: the names its
42/// `[redacted:<name>]` markers carry, each once, in the crate's canonical order, which for
43/// these lower-case hyphenated names is the byte order the shell's `LC_ALL=C sort -u` gives.
44/// They are read off the text, as the shell reads them, so a marker the input already
45/// carried counts as much as one this call wrote: text redacted once and redacted again
46/// still says what was taken out of it. A marker that names no shape of the table is not a
47/// kind here, since a kind is one of [`shape_names`]; the shell would list that name too.
48///
49/// ```
50/// use majordomus_cli::redaction::{redact_secrets, Redacted};
51///
52/// let key = format!("{}{}", "sk-ant-", "b".repeat(20));
53/// let pat = format!("{}{}", "github_pat_", "c".repeat(20));
54/// let out: Redacted = redact_secrets(&format!("{pat} then {key} then {pat}"));
55/// assert_eq!(out.kinds, ["anthropic-key", "github-pat"]);
56/// assert!(!out.text.contains("sk-ant-"));
57///
58/// // redacting again finds nothing new, and still names what the text had taken out
59/// assert_eq!(redact_secrets(&out.text).kinds, out.kinds);
60/// ```
61#[derive(Debug, Clone, PartialEq, Eq)]
62pub struct Redacted {
63    /// The text with every credential replaced by `[redacted:<name>]`.
64    pub text: String,
65    /// The shapes the text's markers name, sorted and unique.
66    pub kinds: Vec<&'static str>,
67}

69/// The names of the credential shapes, in the order they are applied, then `assignment`.
70///
71/// The order is `MJ_CAPTURE_SECRETS`' own, and it matters: the shapes are applied one after
72/// another, so an Anthropic key is named as one before the broader OpenAI shape could claim
73/// it.
74///
75/// ```
76/// use majordomus_cli::redaction::shape_names;
77///
78/// let names = shape_names();
79/// assert_eq!(names.first(), Some(&"anthropic-key"));
80/// assert_eq!(names.last(), Some(&"assignment"));
81/// let anthropic = names.iter().position(|n| *n == "anthropic-key");
82/// let openai = names.iter().position(|n| *n == "openai-key");
83/// assert!(anthropic < openai);
84/// ```
85pub fn shape_names() -> Vec<&'static str> {
86    SHAPES
87        .iter()
88        .chain(std::iter::once(&ASSIGNMENT))
89        .map(|shape| shape.name)
90        .collect()
91}
92
93/// Every credential in `text` replaced by a marker that names its shape.
94///
95/// Byte for byte what `mj_capture_redact` makes of the same text: each shape in table order
96/// over the whole text, every non-overlapping leftmost-longest match replaced as `sed -E`
97/// with `g` replaces it, then the assignment rule, which keeps the name and the separator
98/// and replaces only the value. Text with nothing credential-shaped in it comes back
99/// unchanged, prose that merely talks about a password included.
100///
101/// ```
102/// use majordomus_cli::redaction::redact_secrets;
103///
104/// let value = "d".repeat(16);
105/// let out = redact_secrets(&format!("export API_KEY={value}"));
106/// assert_eq!(out.text, "export API_KEY=[redacted:assignment]");
107/// assert_eq!(out.kinds, ["assignment"]);
108///
109/// let prose = "the password is in the manual";
110/// assert_eq!(redact_secrets(prose).text, prose);
111/// assert!(redact_secrets(prose).kinds.is_empty());
112/// ```
113pub fn redact_secrets(text: &str) -> Redacted {
114    let mut text = text.to_string();
115    for shape in SHAPES.iter().chain(std::iter::once(&ASSIGNMENT)) {
116        if let Some(replaced) = shape.replace_all(&text) {
117            text = replaced;
118        }
119    }
120    let kinds = kinds_in(&text);
121    Redacted { text, kinds }
122}
123
124/// How every marker opens; the shape's name and `]` follow.
125const MARKER: &str = "[redacted:";
126
127/// The shapes named by the markers in `text`, each once and in canonical order: what
128/// `mj_capture_redacted_kinds` reads with `grep -oE '\[redacted:[a-z-]+\]'`, for the names of
129/// the table. No marker can hold another's opening, so every opening is a candidate.
130fn kinds_in(text: &str) -> Vec<&'static str> {
131    let names = shape_names();
132    let mut kinds: Vec<&'static str> = text
133        .match_indices(MARKER)
134        .filter_map(|(at, _)| {
135            let rest = &text[at + MARKER.len()..];
136            let len = rest
137                .find(|c: char| !(c.is_ascii_lowercase() || c == '-'))
138                .unwrap_or(rest.len());
139            rest[len..].starts_with(']').then_some(&rest[..len])
140        })
141        .filter_map(|name| names.iter().copied().find(|shape| *shape == name))
142        .collect();
143    crate::order::canonical_strings(&mut kinds);
144    kinds.dedup();
145    kinds
146}

148/// The repository root written as `<repo>` and the home directory as `<home>`.
149///
150/// These are `mj_uc_normalise`'s first rules, in its order: the canonical (symlinks
151/// resolved) spelling of the root, then the spelling the caller gave, then the home
152/// directory. The canonical spelling goes first because the given one can be a suffix of it
153/// (a temporary directory reached through a symlink), and the root before the home because
154/// a checkout usually lives inside one. A trailing separator is not part of a spelling, and
155/// a spelling that is empty without it, the filesystem root, replaces nothing.
156///
157/// ```
158/// use majordomus_cli::redaction::normalise_machine_paths;
159///
160/// let dir = tempfile::tempdir().unwrap();
161/// let home = dir.path().join("home");
162/// let root = home.join("checkout");
163/// std::fs::create_dir_all(&root).unwrap();
164///
165/// let text = format!("{}/src/lib.rs and {}/.cache", root.display(), home.display());
166/// let out = normalise_machine_paths(&text, &root, Some(&home));
167/// assert_eq!(out, "<repo>/src/lib.rs and <home>/.cache");
168/// ```
169pub fn normalise_machine_paths(text: &str, root: &Path, home: Option<&Path>) -> String {
170    let canonical = std::fs::canonicalize(root).ok();
171    let mut out = text.to_string();
172    for spelling in [canonical.as_deref(), Some(root)]
173        .into_iter()
174        .flatten()
175        .filter_map(spelling_of)
176    {
177        out = out.replace(&spelling, "<repo>");
178    }
179    if let Some(home) = home.and_then(spelling_of) {
180        out = out.replace(&home, "<home>");
181    }
182    out
183}

185/// Text fit to publish: machine paths normalised, then credentials redacted, then checked.
186///
187/// A redaction covers what its shape matched, and two shapes match less than the secret
188/// they announce. So the text is refused, with what it carries and never the text, when:
189///
190/// - its markers name a private key's header, whose body follows on the lines after it, or
191///   an access key id, whose secret access key no shape matches. The markers are read off
192///   the text, so a text redacted once already is refused as the original would be.
193///   Redacting the announcement removes the very marker `generate::forbidden_in` refuses
194///   on, and would publish the rest;
195/// - it still holds a private key's armour line, such as the footer a failure's tail keeps
196///   after cutting the header off, or a secret access key assigned a value: a secret whose
197///   announcement no longer is in the text, or never was;
198/// - `generate::forbidden_in`, the check every published artifact already passes, still
199///   finds a marker: something neither rule can make safe, such as another account's home
200///   directory or a bearer header too short to be a credential's shape.
201///
202/// ```
203/// use majordomus_cli::redaction::public_text;
204///
205/// let dir = tempfile::tempdir().unwrap();
206/// let key = format!("{}{}", "xoxb-", "e".repeat(12));
207/// let text = format!("{}/target: {key}", dir.path().display());
208/// let out = public_text(&text, dir.path(), None).unwrap();
209/// assert_eq!(out.text, "<repo>/target: [redacted:slack-token]");
210///
211/// let elsewhere = format!("{}{}", "/ho", "me/someone/.cache");
212/// let refused = public_text(&elsewhere, dir.path(), None).unwrap_err();
213/// assert!(refused.contains("an absolute path on the machine"));
214///
215/// let pem = format!("{}{}\n{}", "-----BEGIN ", "PRIVATE KEY-----", "f".repeat(64));
216/// let refused = public_text(&pem, dir.path(), None).unwrap_err();
217/// assert!(refused.contains("`private-key-header`"));
218///
219/// let tail = format!("{}\n{}{}", "g".repeat(64), "-----END ", "PRIVATE KEY-----");
220/// let refused = public_text(&tail, dir.path(), None).unwrap_err();
221/// assert!(refused.contains("`PRIVATE KEY-----`"));
222/// ```
223pub fn public_text(text: &str, root: &Path, home: Option<&Path>) -> Result<Redacted, String> {
224    let redacted = redact_secrets(&normalise_machine_paths(text, root, home));
225    if let Some((marker, what)) = uncovered_secret(&redacted) {
226        return Err(format!(
227            "the text carries {what} (`{marker}`), which redacting the shapes of the table \
228             does not cover, so it is not published"
229        ));
230    }
231    match crate::generate::forbidden_in(&redacted.text) {
232        None => Ok(redacted),
233        Some((marker, what)) => Err(format!(
234            "the text still carries {what} (`{marker}`) once its machine paths are normalised \
235             and its credentials redacted, so it is not published"
236        )),
237    }
238}

240/// The first secret `redacted` still carries that no shape covers, as the name it is
241/// refused under and what it is.
242fn uncovered_secret(redacted: &Redacted) -> Option<(&'static str, &'static str)> {
243    let announced = ANNOUNCES_MORE
244        .iter()
245        .find(|(kind, _)| redacted.kinds.contains(kind));
246    let armour = ARMOUR
247        .iter()
248        .find(|(line, _)| contains_ignoring_case(&redacted.text, line));
249    announced.or(armour).copied().or_else(|| {
250        SECRET_ACCESS_KEY.found_in(&redacted.text).then_some((
251            SECRET_ACCESS_KEY.name,
252            "a secret access key assigned a value, which no shape of the table matches",
253        ))
254    })
255}

257/// The shapes whose match announces a secret it does not cover, with what that secret is.
258const ANNOUNCES_MORE: &[(&str, &str)] = &[
259    (
260        "aws-access-key-id",
261        "an access key id, whose secret access key no shape matches",
262    ),
263    (
264        "private-key-header",
265        "a private key's header, whose body no shape matches",

269/// The ends of a private key's armour lines, in any case: every header the shape misses
270/// and every footer, which a tail keeps when it has cut the header off.
271const ARMOUR: &[(&str, &str)] = &[
272    (
273        "PRIVATE KEY-----",
274        "a private key's armour line, whose body no shape matches",
275    ),
276    (
277        "PRIVATE KEY BLOCK-----",
278        "a PGP private key's armour line, whose body no shape matches",

282/// Does `text` contain `needle`, ASCII letters compared without their case?
283fn contains_ignoring_case(text: &str, needle: &str) -> bool {
284    text.as_bytes()
285        .windows(needle.len())
286        .any(|window| window.eq_ignore_ascii_case(needle.as_bytes()))
287}

289/// A secret access key assigned a long opaque value, in the assignment rule's own form:
290/// `aws_secret_access_key = ...`, `AWS_SECRET_ACCESS_KEY=...`, `"SecretAccessKey": "..."`.
291/// It is not a shape of the table and redacts nothing: [`public_text`] refuses a text it is
292/// found in, since the id it pairs with may be on a line the text no longer has. A name with
293/// no value after it, as a skip reason says it, is not a match.
294const SECRET_ACCESS_KEY: Shape = Shape {
295    name: "secret-access-key",
296    kept: &[
297        Atom::Caseless("secret"),
298        run("-_", 0, Some(1)),
299        Atom::Caseless("access"),
300        run("-_", 0, Some(1)),
301        Atom::Caseless("key"),
302        ASSIGNED,
303    ],
304    secret: &[OPAQUE_VALUE],
305};

307/// A path as the text it appears in, without a trailing separator; `None` when nothing is
308/// left, so that the filesystem root never replaces every separator in the text.
309fn spelling_of(path: &Path) -> Option<String> {
310    let lossy = path.to_string_lossy();
311    let trimmed = lossy.trim_end_matches('/');
312    (!trimmed.is_empty()).then(|| trimmed.to_string())
313}

317/// The inside of a POSIX bracket expression, as `lib/capture.sh` spells it: single bytes and
318/// `a-z` ranges. A backslash is an ordinary member, as POSIX has it and as both BSD and GNU
319/// `sed` read the assignment rule's `[\"']`.
320#[derive(Debug, Clone, Copy)]
321struct Class(&'static str);
322
323impl Class {
324    fn has(self, byte: u8) -> bool {
325        let spelling = self.0.as_bytes();
326        let mut i = 0;
327        while i < spelling.len() {
328            if i + 2 < spelling.len() && spelling[i + 1] == b'-' {
329                if (spelling[i]..=spelling[i + 2]).contains(&byte) {
330                    return true;
331                }
332                i += 3;
333            } else {
334                if spelling[i] == byte {
335                    return true;
336                }
337                i += 1;
338            }
339        }
340        false
341    }
342}

344/// One construct of the table's extended regular expressions.
345#[derive(Debug, Clone, Copy)]
346enum Atom {
347    /// These bytes, exactly.
348    Lit(&'static str),
349    /// These letters in either case, which the table spells `[Aa][Pp][Ii]`.
350    Caseless(&'static str),
351    /// One byte of the class.
352    One(Class),
353    /// At least `min` and at most `max` bytes of the class: `?`, `*`, `+`, `{n}`, `{n,}`.
354    Run {
355        class: Class,
356        min: usize,
357        max: Option<usize>,
358    },
359    /// One of the alternatives, or nothing as well when `optional`: `(a|b)` and `(a)?`.
360    Group {
361        alternatives: &'static [&'static [Atom]],
362        optional: bool,
363    },
364}

366/// A credential shape: `kept` is what the replacement keeps (the `\1\2` of the assignment
367/// rule, and nothing for a table shape), `secret` what the marker replaces.
368struct Shape {
369    name: &'static str,
370    kept: &'static [Atom],
371    secret: &'static [Atom],
372}
373
374const fn run(class: &'static str, min: usize, max: Option<usize>) -> Atom {
375    Atom::Run {
376        class: Class(class),
377        min,
378        max,
379    }
380}

521/// Every end at which `atoms` can match `text` from `at`, handed to `found`. A POSIX match is
522/// the longest one, so the callers keep the largest end rather than the first.
523fn ends(atoms: &[Atom], text: &[u8], at: usize, found: &mut dyn FnMut(usize)) {
524    let Some((first, rest)) = atoms.split_first() else {
525        found(at);
526        return;
527    };
528    match *first {
529        Atom::Lit(lit) => {
530            if text[at..].starts_with(lit.as_bytes()) {
531                ends(rest, text, at + lit.len(), found);
532            }
533        }
534        Atom::Caseless(word) => {
535            let end = at + word.len();
536            if end <= text.len() && text[at..end].eq_ignore_ascii_case(word.as_bytes()) {
537                ends(rest, text, end, found);
538            }
539        }
540        Atom::One(class) => {
541            if text.get(at).is_some_and(|b| class.has(*b)) {
542                ends(rest, text, at + 1, found);
543            }
544        }
545        Atom::Run { class, min, max } => {
546            let cap = max.unwrap_or(usize::MAX);
547            let mut n = 0;
548            while n < cap && text.get(at + n).is_some_and(|b| class.has(*b)) {
549                n += 1;
550            }
551            for len in (min..=n).rev() {
552                ends(rest, text, at + len, found);
553            }
554        }
555        Atom::Group {
556            alternatives,
557            optional,
558        } => {
559            if optional {
560                ends(rest, text, at, found);
561            }
562            for alternative in alternatives {
563                ends(alternative, text, at, &mut |end| {
564                    ends(rest, text, end, found)
565                });
566            }
567        }
568    }
569}

571impl Shape {
572    /// The longest match starting at `at`, as the end of the kept part and the end of the
573    /// whole; among equally long matches, the one whose kept part is longest, as POSIX
574    /// assigns subexpressions.
575    fn match_at(&self, text: &[u8], at: usize) -> Option<(usize, usize)> {
576        let mut best: Option<(usize, usize)> = None;
577        ends(self.kept, text, at, &mut |kept_end| {
578            ends(self.secret, text, kept_end, &mut |end| {
579                if best.is_none_or(|(k, e)| (end, kept_end) > (e, k)) {
580                    best = Some((kept_end, end));
581                }
582            });
583        });
584        best
585    }
586
587    /// Does the shape match anywhere in `text`?
588    fn found_in(&self, text: &str) -> bool {
589        let bytes = text.as_bytes();
590        (0..bytes.len()).any(|at| self.match_at(bytes, at).is_some_and(|(_, end)| end > at))
591    }
592
593    /// `s%<shape>%<kept>[redacted:<name>]%g` over the whole text, or `None` when the shape
594    /// matched nothing. Every construct of the table starts with an ASCII byte and matches
595    /// only ASCII bytes, so every cut below falls on a character boundary.
596    fn replace_all(&self, text: &str) -> Option<String> {
597        let bytes = text.as_bytes();
598        let mut out = String::with_capacity(text.len());
599        let (mut copied, mut at) = (0, 0);
600        while at < bytes.len() {
601            match self.match_at(bytes, at) {
602                Some((kept_end, end)) if end > at => {
603                    out.push_str(&text[copied..kept_end]);
604                    out.push_str(MARKER);
605                    out.push_str(self.name);
606                    out.push(']');
607                    copied = end;
608                    at = end;
609                }
610                _ => at += 1,
611            }
612        }
613        if copied == 0 {
614            return None;
615        }
616        out.push_str(&text[copied..]);
617        Some(out)
618    }
619}

630    /// An atom as the extended regular expression `lib/capture.sh` spells it.
631    fn ere(atoms: &[Atom]) -> String {
632        atoms.iter().map(atom_ere).collect()
633    }
634
635    fn atom_ere(atom: &Atom) -> String {
636        match *atom {
637            Atom::Lit(lit) => lit.to_string(),
638            Atom::Caseless(word) => word
639                .chars()
640                .map(|c| format!("[{}{}]", c.to_ascii_uppercase(), c.to_ascii_lowercase()))
641                .collect(),
642            Atom::One(class) => format!("[{}]", class.0),
643            Atom::Run { class, min, max } => {
644                let times = match (min, max) {
645                    (0, Some(1)) => "?".to_string(),
646                    (0, None) => "*".to_string(),
647                    (1, None) => "+".to_string(),
648                    (n, None) => format!("{{{n},}}"),
649                    (n, Some(m)) if n == m => format!("{{{n}}}"),
650                    (n, Some(m)) => format!("{{{n},{m}}}"),
651                };
652                format!("[{}]{times}", class.0)
653            }
654            Atom::Group {
655                alternatives,
656                optional,
657            } => {
658                let inner: Vec<String> = alternatives.iter().map(|a| ere(a)).collect();
659                format!("({}){}", inner.join("|"), if optional { "?" } else { "" })
660            }
661        }
662    }

664    /// The value of a single-quoted shell assignment `NAME='...'` in `lib/capture.sh`, with
665    /// the `'\''` idiom decoded, as the shell itself reads it.
666    fn shell_value(source: &str, name: &str) -> String {
667        let opening = format!("\n{name}='");
668        let start = source
669            .find(&opening)
670            .expect("the assignment is in lib/capture.sh");
671        let mut rest = &source[start + opening.len()..];
672        let mut value = String::new();
673        loop {
674            let close = rest.find('\'').expect("the quoted value is closed");
675            value.push_str(&rest[..close]);
676            rest = &rest[close + 1..];
677            match rest.strip_prefix("\\''") {
678                Some(more) => {
679                    value.push('\'');
680                    rest = more;
681                }
682                None => return value,
683            }
684        }
685    }
