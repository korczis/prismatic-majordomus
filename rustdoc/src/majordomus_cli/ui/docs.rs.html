1//! Documentation: what the product does and where it is written down. Read from
2//! `objects.list` (the product features and the documents of the layer) and `web.surfaces`
3//! (where each surface is served or published).
4//!
5//! A document is named by its path, which is where it lives; the published site renders the
6//! same files, and the surface that serves them is listed beside them.
7
8use crate::capability::Context;
9use crate::cockpit::html::el;
10
11use super::dashboard::ROWS;
12use super::shell::View;
13use super::source::{ask_all, Source};
14use super::view::{
15    hero, kpi, kpi_count, list, more, object_or, panel, table, text, unanswered, value, wide,
16};
17
18/// The route of the screen.
19pub const ROUTE: &str = "/ui/docs";
20
21const FEATURES: Source = Source::with("objects.list", r#"{"kind":"feature"}"#);
22const DOCUMENTS: Source = Source::with("objects.list", r#"{"kind":"document"}"#);
23const SURFACES: Source = Source::of("web.surfaces");
24
25/// The Documentation screen.
26pub fn screen(ctx: &Context) -> View {
27    let answers = ask_all(ctx, &[FEATURES, DOCUMENTS, SURFACES]);
28    let (features, documents, surfaces) = (
29        &answers[&FEATURES],
30        &answers[&DOCUMENTS],
31        &answers[&SURFACES],
32    );
33
34    let top = hero(
35        "book",
36        "cyan",
37        "Documentation",
38        "What the product does, and where it is written down.",
39        "Every product feature as the layer declares it, every document of the repository, and every surface that serves or publishes them.",
40        vec![
41            kpi("Features", &features.read("/count")),
42            kpi("Documents", &documents.read("/count")),
43            kpi_count("Surfaces", &surfaces.read("/surfaces")),
44        ],
45    );
46
47    let feature_list = unanswered(features).unwrap_or_else(|| {
48        el("ul")
49            .class("mj-ui-cards")
50            .children(list(features, "/objects").iter().map(|f| {
51                // a feature objects.list names is an object of the index: its technical view
52                let identity = text(f, "identity");
53                el("li")
54                    .class("mj-ui-card")
55                    .attr("id", format!("feature-{}", super::view::anchor(&identity)))
56                    .child(object_or(&text(f, "uri"), &identity).class("mj-ui-card-kicker"))
57                    .child(
58                        el("strong")
59                            .class("mj-ui-card-title")
60                            .text(value(f, "title")),
61                    )
62                    .child(
63                        el("span")
64                            .class("mj-ui-card-text")
65                            .text(value(f, "description")),
66                    )
67            }))
68    });
69
70    let documents_body = unanswered(documents).unwrap_or_else(|| {
71        let under_docs: Vec<serde_json::Value> = list(documents, "/objects")
72            .into_iter()
73            .filter(|d| text(d, "path").starts_with("docs/"))
74            .collect();
75        let rows = under_docs
76            .iter()
77            .take(ROWS * 6)
78            .map(|d| {
79                // a document objects.list names is an object of the index: its technical view
80                el("tr")
81                    .child(el("td").class("mj-ui-cell-title").text(value(d, "title")))
82                    .child(el("td").child(object_or(&text(d, "uri"), &text(d, "path"))))
83            })
84            .collect::<Vec<_>>();
85        el("div")
86            .child(table(
87                &["Document", "Where it lives"],
88                rows,
89                "The repository holds no document under docs/.",
90            ))
91            .children(more(
92                (ROWS * 6).min(under_docs.len()),
93                under_docs.len(),
94                None,
95            ))
96    });
97
98    let surfaces_body = unanswered(surfaces).unwrap_or_else(|| {
99        let rows = list(surfaces, "/surfaces")
100            .iter()
101            .filter(|s| text(s, "visibility") == "public")
102            .map(|s| {
103                let mount = text(s, "mount");
104                el("tr")
105                    .child(
106                        el("td").child(
107                            el("a")
108                                .class("mj-ui-link")
109                                .attr("href", mount.clone())
110                                .text(text(s, "title")),
111                        ),
112                    )
113                    .child(el("td").class("mj-ui-mono").text(mount))
114                    .child(el("td").class("mj-ui-dim").text(value(s, "availability")))
115            })
116            .collect::<Vec<_>>();
117        table(
118            &["Surface", "Mount", "Where"],
119            rows,
120            "No public surface is declared.",
121        )
122    });
123
124    let grid = el("div")
125        .class("mj-ui-grid")
126        .child(wide(panel(
127            "What the product does",
128            "Each feature as one file of the layer declares it.",
129            feature_list,
130            &[features],
131        )))
132        .child(panel(
133            "Documents",
134            "The project's documents under docs/, by title.",
135            documents_body,
136            &[documents],
137        ))
138        .child(panel(
139            "Surfaces",
140            "Every public surface of the topology and where it is served.",
141            surfaces_body,
142            &[surfaces],
143        ));
144
145    View::new(
146        "Documentation",
147        el("div").class("mj-ui-stack").child(top).child(grid),
148    )
149}

151#[cfg(test)]
152mod tests {
153    use super::*;
154    use crate::ui::deployments::tests::furnished;
155
156    /// Every feature the layer declares is a card with its identity and its title, as the
157    /// one file that declares it says.
158    #[test]
159    fn every_declared_feature_is_a_card() {
160        let repo = furnished();
161        let ctx = repo.context().expect("a context");
162        let html = screen(&ctx).main.render();
163        assert!(html.contains("mj-ui-card"), "{html}");
164        assert!(html.contains(">synthetic<"), "{html}");
165        assert!(html.contains("A synthetic feature"), "{html}");
166        assert!(
167            !html.contains("objects.list could not answer"),
168            "the furnished repository has the kind: {html}"
169        );
170    }
171
172    /// A feature and a document lead to their technical views by the URI the index gives
173    /// them, and a feature's card is a place a link can lead to.
174    #[test]
175    fn a_feature_and_a_document_lead_to_their_technical_views() {
176        let repo = furnished();
177        let ctx = repo.context().expect("a context");
178        let html = screen(&ctx).main.render();
179        assert!(html.contains("id=\"feature-synthetic\""), "{html}");
180        assert!(
181            html.contains("href=\"/cockpit/object?uri=majordomus%3A%2F%2Ffeature%2Fsynthetic\""),
182            "{html}"
183        );
184        assert!(
185            html.contains("href=\"/cockpit/object?uri=majordomus%3A%2F%2Fdocument%2Fdocs%2F"),
186            "a document under docs/ leads to its technical view: {html}"
187        );
188    }
189}
