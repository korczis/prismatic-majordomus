1//! The `shell` module: the repository's shell automation against the inventory that tracks
2//! its migration.
3//!
4//! One capability, `shell.check`, and it is a gate: every shell unit under the governed
5//! directories must be declared in `.ai/repo/automation/inventory.jsonl` with an exemption,
6//! and every record there must name something the tree still has. The judgement lives in
7//! [`crate::automation`]; this module declares it once and every projection — the command
8//! line, MCP, HTTP, the benchmark — is derived from the declaration.
9//!
10//! ```
11//! use majordomus_cli::capability::builtin::modules;
12//! let m = modules().into_iter().find(|m| m.id.as_str() == "shell").unwrap();
13//! let c = &m.capabilities[0].capability;
14//! assert_eq!(c.id.as_str(), "shell.check");
15//! assert_eq!(c.exposure.cli.as_ref().unwrap().path, ["shell", "check"]);
16//! ```
17
18use crate::automation::{self, ShellReport};
19use crate::capability::handler::{CapabilityError, Context};
20use crate::capability::model::{CliExposure, Exposure, Stability};
21use crate::capability::module::ModuleDescriptor;
22use crate::{capability, module};
23
24use super::{get, mcp, Empty};
25
26fn check(ctx: &Context, _: Empty) -> Result<ShellReport, CapabilityError> {
27    Ok(automation::check(std::path::Path::new(
28        &ctx.index.repository.root,
29    )))
30}
31
32/// The module.
33pub fn module() -> ModuleDescriptor {
34    module! {
35        id: "shell",
36        title: "Shell automation",
37        description: "The repository's shell automation measured against the tracked migration inventory: new shell is refused unless an exemption declares why it has to be shell and when it stops being, and an exemption whose file is gone is refused too, so the list only shrinks as units move into typed and scripted capabilities (ADR 0069).",
38        stability: Stability::BehaviorallyVerified,
39        capabilities: [
40            capability! {
41                id: "shell.check",
42                title: "Refuse undeclared shell",
43                description: "Every shell unit under bin/, lib/, scripts/, share/, .githooks/ and .claude/hooks/ — by interpreter line or by a .sh name, as git would commit it, links not followed — against .ai/repo/automation/inventory.jsonl: an undeclared unit, a record whose unit is gone, an exemption missing its reason or its removal condition, a disposition outside A-F, a duplicate and a record out of canonical order are each a finding that names the file and the remedy. Reads the tree and git's index only; no build and no network.",
44                input: Empty,
45                output: ShellReport,
46                stability: Stability::BehaviorallyVerified,
47                exposure: Exposure {
48                    mcp: mcp("majordomus_shell_check"),
49                    http: get("/api/v1/shell/check"),
50                    cli: Some(CliExposure { path: vec!["shell".into(), "check".into()] }),
51                },
52                tags: ["shell", "governance", "migration"],
53                handler: check,
54            },
55        ],
56    }
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62
63    /// The declaration is the only place these names exist; a refactor that dropped an
64    /// exposure would still compile, and this is the assertion that would not.
65    #[test]
66    fn the_declaration_yields_the_projections_it_claims() {
67        let m = module();
68        assert_eq!(m.id.as_str(), "shell");
69        let ids: Vec<&str> = m
70            .capabilities
71            .iter()
72            .map(|e| e.capability.id.as_str())
73            .collect();
74        assert_eq!(ids, ["shell.check"]);
75        let exposure = &m.capabilities[0].capability.exposure;
76        assert_eq!(
77            exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
78            Some("majordomus_shell_check")
79        );
80        assert_eq!(exposure.http.as_ref().unwrap().path, "/api/v1/shell/check");
81        assert_eq!(exposure.cli.as_ref().unwrap().path, ["shell", "check"]);
82    }
83}
