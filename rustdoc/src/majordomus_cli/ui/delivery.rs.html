1//! CI/CD & Releases: what was released, where it is deployed, and what the CI model runs.
2//! Read from `distribution.releases`, `release.version`, `deploy.list` and `gates.model`.
3//!
4//! How long CI takes is `integration.timings`, measured from the recorded forge observation
5//! (`super::timings`): the CI model says which gates run and when, the timings how long the
6//! runs on the open pull requests took.
7
8use crate::capability::Context;
9use crate::cockpit::html::el;
10use serde_json::Value;

12use super::dashboard::releases_panel;

15use super::view::{
16    anchor, code_link, hero, kpi, list, none, panel, table, text, unanswered, value, wide,
17};

19/// The route of the screen.
20pub const ROUTE: &str = "/ui/delivery";
21
22/// Where one gate of the CI model is: its row on this screen. A rule's proof and a pull
23/// request's checks name gates by id, and lead here.
24pub fn gate_href(id: &str) -> String {
25    format!("{ROUTE}#gate-{}", anchor(id))
26}
27
28const RELEASES: Source = Source::of("distribution.releases");
29const VERSION: Source = Source::of("release.version");
30const DEPLOYMENTS: Source = Source::of("deploy.list");
31const GATES: Source = Source::of("gates.model");
32
33/// The CI/CD & Releases screen.
34pub fn screen(ctx: &Context) -> View {
35    let answers = ask_all(
36        ctx,
37        &[
38            RELEASES,
39            VERSION,
40            DEPLOYMENTS,
41            GATES,
42            super::timings::TIMINGS,
43        ],
44    );
45    let (releases, version, deployments, gates) = (
46        &answers[&RELEASES],
47        &answers[&VERSION],
48        &answers[&DEPLOYMENTS],
49        &answers[&GATES],
50    );
51
52    let top = hero(
53        "layers",
54        "orange",
55        "CI/CD & Releases",
56        "From a green check to a published release.",
57        "What was released, what the tree declares and what its commits imply next, where it is deployed, and every gate the CI model can run.",
58        vec![
59            kpi("Released", &releases.read("/latest/version")),
60            kpi("Declared", &version.read("/declared")),
61            kpi("Next", &version.read("/next")),
62            kpi("Gates", &gates.read("/count")),
63        ],
64    );
65
66    let deployed = unanswered(deployments).unwrap_or_else(|| {
67        let rows = list(deployments, "/deployments")
68            .iter()
69            .map(|d| super::deployments::summary_row(ctx, d))
70            .collect::<Vec<_>>();
71        table(
72            &["Deployment", "Application", "State"],
73            rows,
74            "No deployment is declared.",
75        )
76    });
77
78    let model = unanswered(gates).unwrap_or_else(|| {
79        let rows = list(gates, "/gates")
80            .iter()
81            .map(|g| {
82                let when = if g.get("always").and_then(Value::as_bool) == Some(true) {
83                    "every plan"
84                } else if g.get("on-demand").and_then(Value::as_bool) == Some(true) {
85                    "on demand"
86                } else {
87                    "when its paths change"
88                };
89                let id = text(g, "id");
90                el("tr")
91                    .attr("id", format!("gate-{}", anchor(&id)))
92                    .child(el("td").child(code_link(gate_href(&id), value(g, "id"))))
93                    .child(el("td").text(value(g, "job")))
94                    .child(el("td").class("mj-ui-dim").text(when))
95                    .child(el("td").class("mj-ui-dim").text(value(g, "summary")))
96                    .child(el("td").child(gates_named(g, "implies")))
97            })
98            .collect::<Vec<_>>();
99        table(
100            &["Gate", "Job", "Runs", "What it holds", "Implies"],
101            rows,
102            "The CI model declares no gate.",
103        )
104    });
105
106    let grid = el("div")
107        .class("mj-ui-grid")
108        .child(releases_panel(releases, version, None))
109        .child(panel(
110            "Deployments",
111            "Where the product is declared to run.",
112            deployed,
113            &[deployments],
114        ))
115        .child(wide(panel(
116            "CI model",
117            "Every gate the model declares, the job it runs in and when it is selected.",
118            model,
119            &[gates],
120        )))
121        .child(wide(super::timings::panel_of(
122            &answers[&super::timings::TIMINGS],
123        )));
124
125    View::new(
126        "CI/CD & Releases",
127        el("div").class("mj-ui-stack").child(top).child(grid),
128    )
129}

131/// The gates an entry names under `key`, each a link to its own row; `none` when it names none.
132fn gates_named(g: &Value, key: &str) -> crate::cockpit::html::El {
133    let ids: Vec<String> = g
134        .get(key)
135        .and_then(Value::as_array)
136        .map(|a| {
137            a.iter()
138                .filter_map(Value::as_str)
139                .map(str::to_string)
140                .collect()
141        })
142        .unwrap_or_default();
143    if ids.is_empty() {
144        return none();
145    }
146    el("span")
147        .class("mj-ui-badges")
148        .children(ids.iter().map(|id| code_link(gate_href(id), id.clone())))
149}

151#[cfg(test)]
152mod tests {
153    use super::*;
154    use crate::ui::deployments::tests::furnished;
155
156    /// Every declared deployment is a row, and every gate of the CI model says when it
157    /// runs: in every plan, on demand, or when its paths change.
158    #[test]
159    fn every_deployment_and_every_gate_is_a_row_that_says_when_it_runs() {
160        let repo = furnished();
161        let ctx = repo.context().expect("a context");
162        let html = screen(&ctx).main.render();
163        assert!(
164            html.contains("The Majordomus layer, served from its container image"),
165            "{html}"
166        );
167        assert!(!html.contains("No deployment is declared."), "{html}");
168        for (gate, when) in [
169            ("every-plan", "every plan"),
170            ("when-asked", "on demand"),
171            ("by-path", "when its paths change"),
172        ] {
173            assert!(html.contains(gate), "{gate} is missing: {html}");
174            assert!(html.contains(when), "{gate} does not say {when}: {html}");
175        }
176        assert!(
177            html.contains("runs in every plan"),
178            "the gate's own summary: {html}"
179        );
180        assert!(!html.contains("The CI model declares no gate."), "{html}");
181    }
182
183    /// A gate is a row a link can lead to, and the gates it implies lead to theirs.
184    #[test]
185    fn a_gate_is_a_row_links_lead_to() {
186        assert_eq!(gate_href("shell-suite"), "/ui/delivery#gate-shell-suite");
187        let implied = gates_named(
188            &serde_json::json!({ "implies": ["rust-integration", "rust-quality"] }),
189            "implies",
190        )
191        .render();
192        assert!(
193            implied.contains("href=\"/ui/delivery#gate-rust-integration\""),
194            "{implied}"
195        );
196        assert!(
197            implied.contains("href=\"/ui/delivery#gate-rust-quality\""),
198            "{implied}"
199        );
200        assert!(gates_named(&serde_json::json!({}), "implies")
201            .render()
202            .contains(">none<"));
203
204        let repo = furnished();
205        let ctx = repo.context().expect("a context");
206        let html = screen(&ctx).main.render();
207        assert!(html.contains("id=\"gate-every-plan\""), "{html}");
208        assert!(
209            html.contains("href=\"/ui/delivery#gate-by-path\""),
210            "{html}"
211        );
212        assert!(
213            html.contains(&format!(
214                "href=\"{}\"",
215                super::super::deployments::href("majordomus")
216            )),
217            "a deployment leads to its row on Deployments: {html}"
218        );
219    }
