1//! The benchmark report: the run's own results, rendered.
2//!
3//! The evidence is a result document `majordomus bench` already writes — the accepted
4//! baselines under the layer's benchmarks section have the same shape — and this module
5//! renders it. It measures nothing: a figure on the page is a figure a run produced, which
6//! is the whole of `project.performance-evidence` applied to the page as well as the prose.
7
8use std::path::Path;
9
10use serde::{Deserialize, Serialize};
11
12use super::{html, Origin};
13use crate::error::{Error, Result};
14
15/// A results document, read loosely: the fields this report needs, and nothing else, so a
16/// document that grows a field does not stop rendering.
17#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
18pub struct Document {
19    /// When the run finished, as the run recorded it.
20    #[serde(default)]
21    pub finished_at: Option<String>,
22    /// The build profile the run measured.
23    #[serde(default)]
24    pub profile: Option<String>,
25    /// One entry per measured target.
26    #[serde(default)]
27    pub results: Vec<Measurement>,
28}
29
30/// One measured target.
31#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
32pub struct Measurement {
33    /// The target's key: what was measured, over which transport, with which case.
34    pub key: String,
35    /// Whether the run went through a cache.
36    #[serde(default)]
37    pub cache_mode: Option<String>,
38    /// The distribution the run observed.
39    pub stats: Stats,
40}
41
42/// The distribution of one target's samples, in microseconds.
43#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
44pub struct Stats {
45    /// How many samples the figure rests on.
46    pub samples: u64,
47    /// The median.
48    pub p50_us: f64,
49    /// The 95th percentile.
50    pub p95_us: f64,
51    /// The 99th percentile.
52    pub p99_us: f64,
53    /// The slowest sample.
54    pub max_us: f64,
55}
56
57/// Read a results document from disk.
58pub fn read(path: &Path) -> Result<Document> {
59    let text =
60        std::fs::read_to_string(path).map_err(|e| Error::io(path.display().to_string(), e))?;
61    serde_json::from_str(&text).map_err(|e| Error::InvalidSurface {
62        surface: "benchmarks".into(),
63        reason: format!(
64            "{} does not parse as a results document: {e}",
65            path.display()
66        ),
67    })
68}
69
70/// Render the document into its own directory under the generated web root, and declare it.
71pub fn render(root: &Path, document: &Document, source: &str) -> Result<std::path::PathBuf> {
72    let dir = super::declare(
73        root,
74        "benchmarks",
75        "Benchmark results",
76        "majordomus bench, rendered by majordomus web report benchmarks",
77    )?;
78    let origin = Origin::read(root);
79    super::write_json(&dir, "results.json", document)?;
80
81    let mut ordered = document.results.clone();
82    ordered.sort_by(|a, b| {
83        b.stats
84            .p50_us
85            .partial_cmp(&a.stats.p50_us)
86            .unwrap_or(std::cmp::Ordering::Equal)
87            .then_with(|| a.key.cmp(&b.key))
88    });
89    let rows: Vec<Vec<String>> = ordered
90        .iter()
91        .map(|m| {
92            vec![
93                format!("<span class=\"mono\">{}</span>", html::escape(&m.key)),
94                html::escape(m.cache_mode.as_deref().unwrap_or("—")),
95                format!("<span class=\"num\">{}</span>", m.stats.samples),
96                micros(m.stats.p50_us),
97                micros(m.stats.p95_us),
98                micros(m.stats.p99_us),
99                micros(m.stats.max_us),
100            ]
101        })
102        .collect();
103
104    let slowest = ordered
105        .first()
106        .map(|m| m.key.clone())
107        .unwrap_or_else(|| "—".into());
108    let summary = html::summary(&[
109        ("targets", document.results.len().to_string()),
110        (
111            "profile",
112            document.profile.clone().unwrap_or_else(|| "—".into()),
113        ),
114        (
115            "finished",
116            document.finished_at.clone().unwrap_or_else(|| "—".into()),
117        ),
118        ("slowest median", slowest),
119    ]);
120    let body = format!(
121        "{summary}<h2>Every measured target</h2>{}{}",
122        html::table(
123            &["target", "cache", "samples", "p50 µs", "p95 µs", "p99 µs", "max µs"],
124            &rows
125        ),
126        html::origin(&origin, &[("results.json", "results.json")])
127    );
128    let page = html::page(
129        "Benchmark results",
130        &format!(
131            "Every externally callable operation, timed. Read from {}.",
132            html::escape(source)
133        ),
134        &body,
135    );
136    super::write(&dir, "index.html", &page)?;
137    Ok(dir)
138}
139
140/// Microseconds, at the precision a reader can act on.
141fn micros(value: f64) -> String {
142    format!("<span class=\"num\">{value:.1}</span>")
143}

145#[cfg(test)]
146mod unit {
147    use super::*;
148
149    fn document() -> Document {
150        Document {
151            finished_at: Some("2026-09-06T00:00:00Z".into()),
152            profile: Some("debug".into()),
153            results: vec![
154                Measurement {
155                    key: "capabilities.describe|direct|repository-info".into(),
156                    cache_mode: Some("uncached".into()),
157                    stats: Stats {
158                        samples: 200,
159                        p50_us: 41.5,
160                        p95_us: 76.5,
161                        p99_us: 112.7,
162                        max_us: 180.0,
163                    },
164                },
165                Measurement {
166                    key: "objects.search|http|one-word".into(),
167                    cache_mode: None,
168                    stats: Stats {
169                        samples: 200,
170                        p50_us: 900.0,
171                        p95_us: 1200.0,
172                        p99_us: 1500.0,
173                        max_us: 2000.0,
174                    },
175                },
176            ],
177        }
178    }
179
180    #[test]
181    fn a_real_baseline_document_parses() {
182        let text = r#"{"schema":1,"finished_at":"x","profile":"debug","provenance":{"a":1},
183            "results":[{"key":"k","kind":{"kind":"capability"},"cache_mode":"uncached",
184            "stats":{"samples":200,"min_us":1.0,"p50_us":2.0,"p90_us":3.0,"p95_us":4.0,
185                     "p99_us":5.0,"max_us":6.0,"mean_us":2.5}}]}"#;
186        let doc: Document = serde_json::from_str(text).unwrap();
187        assert_eq!(doc.results.len(), 1);
188        assert_eq!(doc.results[0].stats.p95_us, 4.0);
189    }
190
191    #[test]
192    fn the_report_declares_itself_and_orders_by_median() {
193        let tmp = tempfile::tempdir().unwrap();
194        let dir = render(
195            tmp.path(),
196            &document(),
197            ".ai/repo/benchmarks/rust/baseline.json",
198        )
199        .unwrap();
200        let page = std::fs::read_to_string(dir.join("index.html")).unwrap();
201        let search = page.find("objects.search").unwrap();
202        let describe = page.find("capabilities.describe").unwrap();
203        assert!(search < describe, "the slowest target is listed first");
204        let surfaces = crate::web::discover::generated(tmp.path()).unwrap();
205        assert_eq!(surfaces[0].mount.as_str(), "/benchmarks");
206    }
207
208    #[test]
209    fn a_document_that_is_not_one_is_refused_by_name() {
210        let tmp = tempfile::tempdir().unwrap();
211        let path = tmp.path().join("not-results.json");
212        std::fs::write(&path, "{\"results\": 3}").unwrap();
213        let err = read(&path).unwrap_err().to_string();
214        assert!(err.contains("results document"), "{err}");
215    }
