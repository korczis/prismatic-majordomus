1//! The preflight: whether Majordomus is actually in force in this checkout, and what proves
2//! each part of that claim.
3//!
4//! The [`super::RepositoryEnvironment`] says what this checkout *is* — its branch, its
5//! toolchains, where a server published its address. It does not say what is *proven*: an
6//! address that accepts a connection is not a server of this version serving this checkout,
7//! a rule file on disk is not a rule anything enforces, and a recorded test run is not a run
8//! of the tree in front of you. Until this module, the entry banner drew a `✓` beside the
9//! Cockpit on the strength of a TCP connection while the server behind it was two versions
10//! old. The preflight is the value that refuses that.
11//!
12//! # One verdict per claim, and no verdict without evidence
13//!
14//! Every [`Check`] carries a [`Verdict`] and the [`Evidence`] behind it. The three verdicts
15//! that assert something is in force — [`Verdict::Verified`], [`Verdict::Active`] and
16//! [`Verdict::Fresh`] — cannot be constructed without evidence: [`Check::new`] turns such a
17//! claim into [`Verdict::Unknown`] and says why. That is the whole rule
18//! `project.entry-reports-only-evidence` enforces, and it is enforced in the type rather
19//! than in each renderer.
20//!
21//! # Two halves: observing and deriving
22//!
23//! [`observe`] reads — the lease, one loopback request, the session and task records, the
24//! evidence ledger, one `git log` of the deployment ref — and returns [`Observations`], plain
25//! data. [`derive()`] turns observations into a [`Preflight`] and reads nothing, so every
26//! verdict in this file is a test over a value rather than over a machine. The command line,
27//! the HTTP route, the MCP tool and resource, the Cockpit and the entry banner all render
28//! the value [`derive()`] returns.
29//!
30//! # What it never does
31//!
32//! It never builds the index on the entry path (the rule tally comes from the cache a full
33//! resolution wrote, and says when it was counted), never contacts anything but the loopback
34//! address a server of this checkout published, never runs a test, a build or a generator,
35//! and never writes outside `.ai/local/`.
36//!
37//! ```
38//! use majordomus_cli::environment::preflight::{derive, Observations, Verdict};
39//!
40//! // a checkout nobody has observed anything about
41//! let preflight = derive(&Observations::empty("demo", 0));
42//! // nothing is claimed in force, and nothing absent is dressed up as present
43//! assert!(preflight.sections.iter().flat_map(|s| &s.checks).all(|c| !c.verdict.proves()));
44//! assert_eq!(preflight.check("integration.server").unwrap().verdict, Verdict::Unknown);
45//! assert_eq!(preflight.check("verification.deployment").unwrap().verdict, Verdict::Unavailable);
46//! ```

94/// What the repository can show about one claim.
95///
96/// Nine words, not a boolean, because the actions they call for differ: `stale` is re-run
97/// the proof, `unavailable` is nothing is there, `unknown` is nobody looked, and
98/// `not_applicable` is there is nothing to look for.
99///
100/// ```
101/// use majordomus_cli::environment::preflight::Verdict;
102/// assert!(Verdict::Verified.proves() && Verdict::Active.proves() && Verdict::Fresh.proves());
103/// assert!(!Verdict::Stale.proves() && !Verdict::NotApplicable.proves());
104/// assert_eq!(Verdict::NotApplicable.as_str(), "not_applicable");
105/// // the most urgent first: a failure outranks a stale proof, which outranks an absence
106/// assert!(Verdict::Failed.urgency() < Verdict::Stale.urgency());
107/// assert_eq!(Verdict::Verified.urgency(), None);
108/// ```
109#[derive(
110    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
111)]
112#[serde(rename_all = "snake_case")]
113#[schemars(rename = "PreflightVerdict")]
114pub enum Verdict {
115    /// Proven against this tree by a recorded result or a direct answer.
116    Verified,
117    /// In force now, observed directly, with nothing recorded to verify.
118    Active,
119    /// Judged recent enough to act on, against a declared threshold or the current commit.
120    Fresh,
121    /// Evidence exists, but it is about another commit, tree or moment.
122    Stale,
123    /// In force, but in a form that is not the one this executable would serve.
124    Degraded,
125    /// Nothing is there to observe.
126    Unavailable,
127    /// Observed, and it did not hold.
128    Failed,
129    /// Nobody looked, or the looking did not finish.
130    Unknown,
131    /// There is nothing here to judge.
132    NotApplicable,
133}
134
135impl Verdict {
136    /// The word as serialised and printed.
137    ///
138    /// ```
139    /// use majordomus_cli::environment::preflight::Verdict;
140    /// assert_eq!(Verdict::Degraded.as_str(), "degraded");
141    /// ```
142    pub fn as_str(self) -> &'static str {
143        match self {
144            Verdict::Verified => "verified",
145            Verdict::Active => "active",
146            Verdict::Fresh => "fresh",
147            Verdict::Stale => "stale",
148            Verdict::Degraded => "degraded",
149            Verdict::Unavailable => "unavailable",
150            Verdict::Failed => "failed",
151            Verdict::Unknown => "unknown",
152            Verdict::NotApplicable => "not_applicable",
153        }
154    }
155
156    /// Does this verdict assert that something is in force? Only these three need evidence
157    /// to be constructed, and only these three may be drawn as a success.
158    ///
159    /// ```
160    /// use majordomus_cli::environment::preflight::Verdict;
161    /// assert!(Verdict::Fresh.proves());
162    /// assert!(!Verdict::Unknown.proves());
163    /// ```
164    pub fn proves(self) -> bool {
165        matches!(self, Verdict::Verified | Verdict::Active | Verdict::Fresh)
166    }
167
168    /// How urgently a person should look, lowest first; `None` for a verdict that needs no
169    /// attention.
170    ///
171    /// ```
172    /// use majordomus_cli::environment::preflight::Verdict;
173    /// assert_eq!(Verdict::Failed.urgency(), Some(0));
174    /// assert_eq!(Verdict::NotApplicable.urgency(), None);
175    /// ```
176    pub fn urgency(self) -> Option<u8> {
177        match self {
178            Verdict::Failed => Some(0),
179            Verdict::Degraded => Some(1),
180            Verdict::Stale => Some(2),
181            Verdict::Unavailable => Some(3),
182            Verdict::Unknown => Some(4),
183            Verdict::Verified | Verdict::Active | Verdict::Fresh | Verdict::NotApplicable => None,
184        }
185    }
186
187    /// The status word the Cockpit's design system files this verdict under.
188    ///
189    /// ```
190    /// use majordomus_cli::environment::preflight::Verdict;
191    /// assert_eq!(Verdict::Verified.status(), "ok");
192    /// assert_eq!(Verdict::Failed.status(), "fail");
193    /// ```
194    pub fn status(self) -> &'static str {
195        match self {
196            Verdict::Verified | Verdict::Active | Verdict::Fresh => "ok",
197            Verdict::Stale | Verdict::Degraded | Verdict::Unknown => "warn",
198            Verdict::Failed => "fail",
199            Verdict::Unavailable | Verdict::NotApplicable => "info",
200        }
201    }
202}

236/// One claim, its verdict, and what the verdict rests on.
237///
238/// ```
239/// use majordomus_cli::environment::preflight::{Check, Evidence, Verdict};
240///
241/// // a claim of something in force without evidence is not accepted as one
242/// let bare = Check::new("integration.mcp", "MCP", Verdict::Verified, "healthy", vec![]);
243/// assert_eq!(bare.verdict, Verdict::Unknown);
244/// assert!(bare.summary.contains("no evidence"));
245///
246/// let backed = Check::new(
247///     "integration.mcp", "MCP", Verdict::Verified, "served",
248///     vec![Evidence::new("GET /", "surface mcp ready")],
249/// );
250/// assert_eq!(backed.verdict, Verdict::Verified);
251/// ```
252#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
253#[schemars(rename = "PreflightCheck")]
254pub struct Check {
255    /// A stable dotted id, `<section>.<claim>`.
256    pub id: String,
257    /// The short name a person reads.
258    pub title: String,
259    /// What the repository can show.
260    pub verdict: Verdict,
261    /// The verdict in one sentence, with the numbers that decided it.
262    pub summary: String,
263    /// Everything the verdict rests on. Never empty for a verdict that proves something.
264    #[serde(default, skip_serializing_if = "Vec::is_empty")]
265    pub evidence: Vec<Evidence>,
266    /// The command that shows more, or that changes the verdict.
267    #[serde(default, skip_serializing_if = "Option::is_none")]
268    pub next: Option<String>,
269}
270
271impl Check {
272    /// A check. A verdict that [`Verdict::proves`] something and arrives without evidence
273    /// becomes [`Verdict::Unknown`], with the summary saying so.
274    ///
275    /// ```
276    /// use majordomus_cli::environment::preflight::{Check, Verdict};
277    /// let c = Check::new("x.y", "Y", Verdict::Stale, "counted at another commit", vec![]);
278    /// assert_eq!(c.verdict, Verdict::Stale, "only a claim of force needs evidence");
279    /// ```
280    pub fn new(
281        id: impl Into<String>,
282        title: impl Into<String>,
283        verdict: Verdict,
284        summary: impl Into<String>,
285        evidence: Vec<Evidence>,
286    ) -> Self {
287        let summary = summary.into();
288        let (verdict, summary) = if verdict.proves() && evidence.is_empty() {
289            (
290                Verdict::Unknown,
291                format!(
292                    "claimed {} with no evidence, so not claimed: {summary}",
293                    verdict.as_str()
294                ),
295            )
296        } else {
297            (verdict, summary)
298        };
299        Check {
300            id: id.into(),
301            title: title.into(),
302            verdict,
303            summary,
304            evidence,
305            next: None,
306        }
307    }
308
309    /// The same check, naming the command that shows more.
310    ///
311    /// ```
312    /// use majordomus_cli::environment::preflight::{Check, Verdict};
313    /// let c = Check::new("x.y", "Y", Verdict::Unknown, "not probed", vec![])
314    ///     .next("majordomus serve status");
315    /// assert_eq!(c.next.as_deref(), Some("majordomus serve status"));
316    /// ```
317    pub fn next(mut self, command: impl Into<String>) -> Self {
318        self.next = Some(command.into());
319        self
320    }
321}

345/// Whether Majordomus is in force in this checkout, claim by claim, with the evidence.
346///
347/// ```
348/// use majordomus_cli::environment::preflight::{derive, Observations, Preflight};
349/// let p = derive(&Observations::empty("demo", 0));
350/// assert_eq!(p.schema, Preflight::schema_id());
351/// assert_eq!(p.verdicts().len(), p.sections.iter().map(|s| s.checks.len()).sum::<usize>());
352/// ```
353#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
354pub struct Preflight {
355    /// `majordomus/preflight/v1`.
356    pub schema: String,
357    /// When the observations were taken, RFC 3339 in UTC.
358    pub generated_at: String,
359    /// How the environment behind it was resolved.
360    pub resolution: Resolution,
361    /// The repository's name.
362    pub repository: String,
363    /// The branch, when git named one.
364    #[serde(default, skip_serializing_if = "Option::is_none")]
365    pub branch: Option<String>,
366    /// The commit HEAD names.
367    #[serde(default, skip_serializing_if = "Option::is_none")]
368    pub head: Option<String>,
369    /// The groups of checks.
370    pub sections: Vec<Section>,
371    /// The ids of every check that needs a look, most urgent first.
372    pub attention: Vec<String>,
373}

375impl Preflight {
376    /// The schema identity a document of this shape carries.
377    ///
378    /// ```
379    /// use majordomus_cli::environment::preflight::Preflight;
380    /// assert_eq!(Preflight::schema_id(), "majordomus/preflight/v1");
381    /// ```
382    pub fn schema_id() -> String {
383        format!("{SCHEMA}/v{SCHEMA_VERSION}")
384    }
385
386    /// One check by its dotted id, or `None` for an id no section declares — which a caller
387    /// must not read as a check that passed.
388    ///
389    /// ```
390    /// use majordomus_cli::environment::preflight::{derive, Observations};
391    /// let p = derive(&Observations::empty("demo", 0));
392    /// assert!(p.check("session.episode").is_some());
393    /// assert!(p.check("nothing.here").is_none());
394    /// ```
395    pub fn check(&self, id: &str) -> Option<&Check> {
396        self.sections
397            .iter()
398            .flat_map(|s| &s.checks)
399            .find(|c| c.id == id)
400    }
401
402    /// Every check's verdict by id: the part two surfaces must agree on.
403    ///
404    /// ```
405    /// use majordomus_cli::environment::preflight::{derive, Observations, Verdict};
406    /// let v = derive(&Observations::empty("demo", 0)).verdicts();
407    /// assert_eq!(v["verification.coverage"], Verdict::Unavailable);
408    /// ```
409    pub fn verdicts(&self) -> BTreeMap<String, Verdict> {
410        self.sections
411            .iter()
412            .flat_map(|s| &s.checks)
413            .map(|c| (c.id.clone(), c.verdict))
414            .collect()
415    }
416}

586    ///     coverage: Default::default(), findings: vec![], satisfied: true };
587    /// assert_eq!(RulesTally::of(&empty).rules, 0);
588    /// ```
589    pub fn of(report: &RulesReport) -> Self {
590        let count = |s: RuleState| report.states.get(s.label()).copied().unwrap_or(0);
591        RulesTally {
592            head: report.head.clone(),
593            working_tree: report.working_tree.clone(),
594            rules: report.coverage.rules,
595            blocking: report.coverage.blocking,
596            project: report
597                .rules
598                .iter()
599                .filter(|r| r.rule.id.starts_with("project."))
600                .count(),
601            gated: report.coverage.gated,
602            states: report.states.clone(),
603            proven: count(RuleState::Proven),
604            failing: count(RuleState::Failing),
605            reviewed: count(RuleState::Reviewed),
606            unproven: count(RuleState::Unproven),
607            findings: report.findings.len(),
608        }
609    }
610}

920/// The preflight these observations support. Reads nothing.
921///
922/// ```
923/// use majordomus_cli::environment::preflight::{derive, DeploymentObservation, GitObservation, Observations, Verdict};
924/// let head = "b".repeat(40);
925/// let mut o = Observations::empty("demo", 0);
926/// o.git = Some(GitObservation { branch: None, head: Some(head.clone()), clean: true, changed: 0 });
927/// o.deployment = DeploymentObservation::Published { commit: "d".into(), at: "t".into(), source: Some(head) };
928/// assert_eq!(derive(&o).check("verification.deployment").unwrap().verdict, Verdict::Verified);
929/// ```
930pub fn derive(o: &Observations) -> Preflight {
931    let head = o.git.as_ref().and_then(|g| g.head.clone());
932    let clean = o.git.as_ref().map(|g| g.clean);
933    let sections = vec![
934        Section {
935            id: "repository".into(),
936            title: "Repository".into(),
937            checks: vec![git_check(o)],
938        },
939        Section {
940            id: "session".into(),
941            title: "Session".into(),
942            checks: vec![
943                episode_check(o),
944                task_check(o),
945                context_check(o, head.as_deref(), clean),
946                handover_check(o),
947            ],
948        },
949        Section {
950            id: "governance".into(),
951            title: "Governance".into(),
952            checks: vec![
953                policy_check(o),
954                rules_check(o, head.as_deref()),
955                adrs_check(o),
956            ],
957        },
958        Section {
959            id: "integration".into(),
960            title: "Integration".into(),
961            checks: vec![
962                server_check(o),
963                surface_check(o, "mcp", "integration.mcp", "MCP"),
964                surface_check(o, "api", "integration.api", "API"),
965                surface_check(o, "cockpit", "integration.cockpit", "Cockpit"),
966                peers_check(o),
967            ],
968        },
969        Section {
970            id: "verification".into(),
971            title: "Verification".into(),
972            checks: vec![
973                tests_check(o),
974                coverage_check(o, head.as_deref(), clean),
975                enforcement_check(o, head.as_deref(), clean),
976                projections_check(o),
977                docs_check(),
978                deployment_check(o, head.as_deref()),
979            ],
980        },
981    ];
982    // Most urgent first, and within one urgency in the order the sections declare their
983    // checks: one pass per level over a list already in canonical order, so no sort is
984    // needed and none can disagree with the declaration order.
985    let attention: Vec<String> = (0..=Verdict::Unknown.urgency().unwrap_or(0))
986        .flat_map(|level| {
987            sections
988                .iter()
989                .flat_map(|s| &s.checks)
990                .filter(move |c| c.verdict.urgency() == Some(level))
991                .map(|c| c.id.clone())
992        })
993        .collect();
994    Preflight {
995        schema: Preflight::schema_id(),
996        generated_at: o.generated_at.clone(),
997        resolution: o.resolution,
998        repository: o.repository.clone(),
999        branch: o.git.as_ref().and_then(|g| g.branch.clone()),
1000        head,
1001        sections,
1002        attention,
1003    }
1004}

1126/// The briefing an episode was given is exact at the commit and tree it was written at. It
1127/// goes stale when HEAD moves, when the tree changes under it, or when a task starts after
1128/// it — the three things a briefing is about.
1129fn context_check(o: &Observations, head: Option<&str>, clean: Option<bool>) -> Check {
1130    let Some(e) = o.episode.as_ref().filter(|e| e.this_checkout) else {
1131        return Check::new(
1132            "session.context",
1133            "context",
1134            Verdict::Unknown,
1135            "no episode of this checkout, so no briefing to judge",
1136            vec![],
1137        );
1138    };
1139    let evidence = vec![Evidence::new(
1140        SESSION_RECORD,
1141        format!(
1142            "briefing written at {} ({})",
1143            short(&e.start_head),
1144            if e.start_working_tree.is_empty() {
1145                "tree unrecorded"
1146            } else {
1147                &e.start_working_tree
1148            }
1149        ),
1150    )];
1151    let Some(head) = head else {
1152        return Check::new(
1153            "session.context",
1154            "context",
1155            Verdict::Unknown,
1156            "git named no HEAD to compare the briefing with",
1157            evidence,
1158        );
1159    };
1160    let mut why: Vec<String> = Vec::new();
1161    if e.start_head.is_empty() {
1162        why.push("the episode recorded no commit".into());
1163    } else if e.start_head != head {
1164        why.push(format!(
1165            "HEAD moved from {} to {}",
1166            short(&e.start_head),
1167            short(head)
1168        ));
1169    }
1170    if e.start_working_tree == "clean" && clean == Some(false) {
1171        why.push("the tree changed since".into());
1172    }
1173    // A briefing written over a dirty tree cannot be shown exact later: the record keeps the
1174    // word `dirty`, not which changes, so an edit or a revert since is invisible. It is not
1175    // stale either — nothing is known to have moved — so it is reported as what it is.
1176    if why.is_empty() && e.start_working_tree != "clean" {
1177        return Check::new(
1178            "session.context",
1179            "context",
1180            Verdict::Unknown,
1181            format!(
1182                "the briefing was written over a {} tree at {}, so whether the tree changed since cannot be judged",
1183                if e.start_working_tree.is_empty() {
1184                    "an unrecorded"
1185                } else {
1186                    e.start_working_tree.as_str()
1187                },
1188                short(head)
1189            ),
1190            evidence,
1191        )
1192        .next("majordomus context");
1193    }
1194    if let Some(t) = &o.task {
1195        if in_progress(t) && !t.started_at.is_empty() && t.started_at > e.started_at {
1196            why.push(format!("task {} started after it", t.id));
1197        }
1198    }
1199    if why.is_empty() {
1200        Check::new(
1201            "session.context",
1202            "context",
1203            Verdict::Fresh,
1204            format!("the briefing is exact at {}", short(head)),
1205            evidence,
1206        )
1207    } else {
1208        Check::new(
1209            "session.context",
1210            "context",
1211            Verdict::Stale,
1212            format!("the briefing is stale: {}", why.join("; ")),
1213            evidence,
1214        )

1740        return Check::new(
1741            "verification.enforcement",
1742            "enforcement",

1746        )

1748    };

1750    let summary = format!(
1751        "{} of {owed} rule(s) with an executable proof are proven against the tree · {} gated with no recorded verdict · {} finding(s)",
1752        t.proven,
1753        t.states.get(RuleState::Gated.label()).copied().unwrap_or(0),

1761                .iter()
1762                .map(|(k, v)| format!("{k} {v}"))

1764                .join(", ")
1765        ),
1766    )];

1768    // Currency first: a tally about another commit, or counted over a dirty tree (whose
1769    // changes the tally cannot name), says nothing about this tree — neither that a rule
1770    // fails here nor that every rule is proven.
1771    if !at_head || t.working_tree != "clean" || clean == Some(false) {

1789            "verification.enforcement",

1792            format!("{} rule(s) failing · {summary}", t.failing),
1793            evidence,
1794        )
1795        .next("majordomus-cli rules report")

1799            "verification.enforcement",
1800            "enforcement",
1801            Verdict::NotApplicable,
1802            format!("no rule owes an executable proof · {summary}"),
1803            evidence,
1804        )

1806        Check::new(
1807            "verification.enforcement",
1808            "enforcement",
1809            Verdict::Verified,
1810            summary,
1811            evidence,
1812        )
1813    } else {
1814        Check::new(
1815            "verification.enforcement",
1816            "enforcement",
1817            Verdict::Degraded,
1818            summary,
1819            evidence,
1820        )

1822    }

2032/// Read everything a preflight is decided from.
2033///
2034/// `environment` is the snapshot already resolved for this call (its version control, layer
2035/// counts and provider projections are used as they are, never re-read); `policy` is the
2036/// policy the caller read, or why it could not; `rules` is a tally counted in this process,

2040/// use majordomus_cli::environment::preflight::{derive, observe, Probe, Verdict};
2041/// use majordomus_cli::environment::{resolve, EnvironmentQuery, Inputs};
2042/// use majordomus_cli::synthetic::SyntheticRepository;

2046/// let env = resolve(
2047///     &Inputs { repository: &repository, share: None, index: None, registry: None, policy: None },
2048///     &EnvironmentQuery::fast().sealed(),
2049/// );
2050/// let p = derive(&observe(repo.root(), &env, Err("not read".into()), None, Probe::sealed()));
2051/// // no lease: nothing serves this checkout, and that needs no probe to know
2052/// assert_eq!(p.check("integration.server").unwrap().verdict, Verdict::Unavailable);
2053/// ```
2054pub fn observe(
2055    root: &Path,
2056    environment: &RepositoryEnvironment,
2057    policy: Result<&LoadedPolicy, String>,
2058    rules: Option<RulesTally>,
2059    probe: Probe,
2060) -> Observations {
2061    let local = environment.repository.local_path.as_str();
2062    let state = root.join(local).join("state");
2063    let now = std::time::SystemTime::now()
2064        .duration_since(std::time::UNIX_EPOCH)
2065        .map(|d| d.as_secs() as i64)
2066        .unwrap_or(0);
2067
2068    let git = environment.vcs.tree().map(|t| GitObservation {
2069        branch: t.branch.clone(),
2070        head: t.head.clone(),
2071        clean: t.clean,
2072        changed: t.dirty_files() + t.untracked,
2073    });
2074    let head = git.as_ref().and_then(|g| g.head.clone());
2075    let branch = git
2076        .as_ref()
2077        .and_then(|g| g.branch.clone())
2078        .unwrap_or_else(|| "DETACHED".into());
2079
2080    let episode = read_scalars(&state.join("session-current.yaml")).and_then(|f| {
2081        let id = f.get("session_id")?.clone();
2082        let worktree = f.get("worktree").cloned().unwrap_or_default();
2083        Some(EpisodeObservation {
2084            id,
2085            started_at: f.get("started_at").cloned().unwrap_or_default(),
2086            provider: f.get("provider").cloned().unwrap_or_default(),
2087            start_head: f.get("start_head").cloned().unwrap_or_default(),
2088            start_working_tree: f.get("start_working_tree").cloned().unwrap_or_default(),
2089            this_checkout: worktree.is_empty() || Path::new(&worktree) == root,
2090        })
2091    });
2092    let task = continuity::read_task(&state.join("current.yaml")).map(|t| TaskObservation {
2093        id: t.id,
2094        task: t.task,
2095        outcome: t.outcome,
2096        started_at: t.started_at,
2097    });
2098
2099    let (policy_observation, thresholds) = match &policy {
2100        Ok(p) => (
2101            PolicyObservation::Parsed,
2102            Thresholds {
2103                fresh_minutes: p.policy.session.freshness.fresh_minutes,
2104                stale_minutes: p.policy.session.freshness.stale_minutes,
2105            },
2106        ),
2107        Err(reason) if reason.is_empty() => (PolicyObservation::NotRead, Thresholds::default()),
2108        Err(reason) => (
2109            PolicyObservation::Refused(reason.clone()),
2110            Thresholds::default(),
2111        ),
2112    };
2113    let handover = continuity::resolve_record(
2114        root,
2115        &state.join("handovers"),
2116        &branch,
2117        head.as_deref(),
2118        thresholds,
2119        now,
2120    )
2121    .0
2122    .map(|r| HandoverObservation {
2123        path: r.path,
2124        freshness: r.freshness,
2125        reason: r.freshness_reason,
2126        divergence: r.divergence.as_str().to_string(),
2127    });
2128
2129    let rules = match rules {
2130        Some(t) => RulesObservation::Counted(t),
2131        None => match cached_rules(root, local) {
2132            Some(t) => RulesObservation::Cached(t),
2133            None => RulesObservation::Absent,
2134        },
2135    };
2136
2137    let server = observe_server(root, local, probe.server);
2138    let peers = if !probe.peers {
2139        Err("not asked on entry: the board gathers every checkout's and costs a round trip".into())

2141        match (&server.standing, &server.url) {
2142            (Some(ServerStanding::Ready | ServerStanding::Outdated), Some(url)) => ask_peers(url),
2143            (Some(ServerStanding::Absent), _) => {
2144                Err("no server, so no board: the board lives in the shared server".into())
2145            }
2146            _ => Err("the server was not answering, so the board was not asked".into()),
2147        }
2148    };
2149
2150    Observations {
2151        repository: environment.repository.name.clone(),
2152        generated_at: crate::peers::rfc3339(std::time::SystemTime::now()),
2153        now,
2154        resolution: environment.resolution,
2155        git,
2156        episode,
2157        task,
2158        handover,
2159        policy: policy_observation,
2160        rules,
2161        adrs: environment.layer.kind("adr"),
2162        server,
2163        peers,
2164        ledger: observe_ledger(root, local, environment.vcs.tree(), probe.cache),
2165        coverage: observe_coverage(root),
2166        projections: environment
2167            .providers
2168            .iter()
2169            .map(|p| (p.id.clone(), p.state))
2170            .collect(),
2171        deployment: observe_deployment(root),
2172    }
2173}

2192fn observe_server(root: &Path, local: &str, probe: bool) -> ServerObservation {
2193    // A served request is its own evidence: this process holds the lease, so the surfaces it
2194    // is serving the request from are the checkout's, at this executable's version.
2195    //
2196    // Holding the lease proves *which* process serves; it does not prove the code. A binary
2197    // rebuilt under a running server leaves it holding the lease with code no longer on disk,
2198    // so the standing is still `standing_of`'s — the one decision every other reader uses —
2199    // over this process's own document, with the answer already known to be yes.
2200    if let Some(doc) = lease::held().filter(|d| d.root.as_path() == root) {
2201        let (standing, reason) = standing_of(
2202            &LeaseFile::Document(doc.clone()),
2203            Duration::ZERO,
2204            |_| true,
2205            crate::VERSION,
2206        );
2207        return ServerObservation {
2208            standing: Some(standing),
2209            reason,
2210            url: doc.url.clone(),
2211            pid: Some(doc.pid),
2212            version: doc.version.clone(),
2213            surfaces: served_surfaces(),
2214            this_process: true,
2215        };
2216    }
2217    let path = lease::lease_file(root, local);
2218    let file = LeaseFile::read(&path);
2219    let doc = file.document().cloned();
2220    let mut observation = ServerObservation {
2221        url: doc.as_ref().and_then(|d| d.url.clone()),
2222        pid: doc.as_ref().map(|d| d.pid),
2223        version: doc.as_ref().and_then(|d| d.version.clone()),
2224        ..Default::default()
2225    };
2226    if matches!(file, LeaseFile::Absent) {
2227        observation.standing = Some(ServerStanding::Absent);
2228        return observation;
2229    }
2230    if !probe {
2231        return observation;
2232    }
2233    let reply = doc
2234        .as_ref()
2235        .and_then(|d| d.url.as_deref())
2236        .and_then(|url| lease::probe_reply(url, root, ANSWER_BUDGET));
2237    let (standing, reason) = standing_of(
2238        &file,
2239        lease::file_age(&path),
2240        |_| reply.is_some(),
2241        crate::VERSION,
2242    );
2243    observation.standing = Some(standing);
2244    observation.reason = reason;
2245    if let Some(body) = reply {
2246        observation.surfaces = body["surfaces"]
2247            .as_array()
2248            .map(|a| {
2249                a.iter()
2250                    .filter_map(|s| {
2251                        Some((
2252                            s["id"].as_str()?.to_string(),
2253                            s["path"].as_str().unwrap_or_default().to_string(),
2254                            s["ready"].as_bool().unwrap_or(false),
2255                        ))
2256                    })
2257                    .collect()
2258            })
2259            .unwrap_or_default();
2260    }
2261    observation
2262}

2280fn ask_peers(url: &str) -> Result<PeersObservation, String> {
2281    // `checkouts=this`: this checkout's board only, which is the board a served preflight
2282    // reads from its own process. The default gathers every worktree's board over HTTP, which
2283    // would make the command line and the Cockpit count different peers for one state.
2284    let reply = crate::mcp::bridge::request(
2285        url,
2286        "GET",
2287        "/api/v1/peers?checkouts=this",
2288        &[],
2289        None,
2290        BOARD_BUDGET,
2291    )
2292    .map_err(|e| {
2293        format!(
2294            "the board did not answer within {} ms: {e}",
2295            BOARD_BUDGET.as_millis()
2296        )
2297    })?;
2298    if reply.status != 200 {
2299        return Err(format!("the board answered {}", reply.status));
2300    }
2301    let mut v: Value = serde_json::from_str(&reply.body).map_err(|e| e.to_string())?;
2302    let peers: Vec<crate::peers::Peer> =
2303        serde_json::from_value(v["peers"].take()).map_err(|e| e.to_string())?;
2304    Ok(PeersObservation::of(&peers))
2305}

2333fn observe_ledger(
2334    root: &Path,
2335    local: &str,
2336    tree: Option<&super::GitWorkingTree>,
2337    use_cache: bool,
2338) -> LedgerObservation {
2339    if !Ledger::present(root) {
2340        return LedgerObservation::Absent;
2341    }
2342    let ledger = match Ledger::load(root) {
2343        Ok(l) => l,
2344        Err(e) => return LedgerObservation::Unreadable(e.to_string()),
2345    };
2346    // What `evidence.report` calls proven, decided by its own comparison: the tree in front
2347    // of you is byte for byte the tree the run measured, the ledger's own row excluded —
2348    // recording dirties the tree and committing the record moves HEAD, so without that
2349    // exclusion no run could ever be current. One comparison per distinct recorded commit,
2350    // two git processes each — which is most of what entry would otherwise cost, so the
2351    // answers are cached under everything they depend on: HEAD, every path git reports
2352    // changed, and the ledger file.
2353    // Every changed path is stat'ed, not only named: an edit to a path already changed keeps
2354    // the set of names and moves the file's size or modification time.
2355    let fingerprint = tree.map(|t| {
2356        let mut paths = t.changed_paths.clone();
2357        paths.push(crate::evidence::ledger::LEDGER_PATH.to_string());
2358        super::cache::fingerprint_of(root, &paths, &[t.head.as_deref().unwrap_or("")])
2359    });
2360    let cache = use_cache.then(|| Cache::load(root, local));
2361    let cached = match (&cache, &fingerprint) {
2362        (Some(c), Some(f)) => c
2363            .tiers
2364            .ledger_trees
2365            .as_ref()
2366            .and_then(|e| e.fresh(f, None))
2367            .cloned(),
2368        _ => None,
2369    };
2370    let unchanged: BTreeMap<String, bool> = match cached {
2371        Some(answers) => answers,
2372        None => {
2373            let mut answers: BTreeMap<String, bool> = BTreeMap::new();
2374            for e in &ledger.executions {
2375                answers.entry(e.commit.clone()).or_insert_with(|| {
2376                    crate::evidence::changed_since(root, &e.commit).is_some_and(|changed| {
2377                        changed
2378                            .iter()
2379                            .all(|p| p == crate::evidence::ledger::LEDGER_PATH)
2380                    })
2381                });
2382            }
2383            if let (true, Some(f)) = (use_cache, fingerprint) {
2384                // Loaded again just before the write, with only this tier changed: the reading
2385                // above is minutes older than this under load, and writing it back whole would
2386                // put back a rule tally a `--full` run replaced meanwhile.
2387                let mut c = Cache::load(root, local);
2388                c.tiers.ledger_trees = Some(Cache::entry(f, answers.clone()));
2389                if let Err(e) = c.store(root, local) {
2390                    tracing::debug!(error = %e, "the ledger comparison could not be cached");
2391                }
2392            }
2393            answers
2394        }
2395    };
2396    let current: Vec<_> = ledger
2397        .executions
2398        .iter()
2399        .filter(|e| {
2400            e.working_tree == "clean"
2401                && unchanged.get(e.commit.as_str()) == Some(&true)
2402                && e.digest_matches(root) == Some(true)
2403        })
2404        .collect();
2405    let newest = ledger.executions.iter().max_by(|a, b| a.at.cmp(&b.at));
2406    LedgerObservation::Read {
2407        total: ledger.executions.len(),
2408        current: current.len(),
2409        current_failing: current.iter().filter(|e| !e.outcome.proves()).count(),
2410        newest_commit: newest.map(|e| e.commit.clone()).unwrap_or_default(),
2411        newest_at: newest.map(|e| e.at.clone()).unwrap_or_default(),
2412    }
2413}

2500/// The mark a terminal draws beside a verdict. A success mark only for a verdict that
2501/// proves something, which is the one guarantee a person reading a banner needs.
2502///
2503/// ```
2504/// use majordomus_cli::environment::preflight::{mark, Verdict};
2505/// assert_eq!(mark(Verdict::Verified, true), "✓");
2506/// assert_eq!(mark(Verdict::Stale, true), "◐");
2507/// assert_eq!(mark(Verdict::Failed, false), "x");
2508/// ```
2509pub fn mark(verdict: Verdict, unicode: bool) -> &'static str {
2510    match (verdict, unicode) {
2511        (v, true) if v.proves() => "✓",
2512        (v, false) if v.proves() => "ok",
2513        (Verdict::Stale | Verdict::Degraded, true) => "◐",
2514        (Verdict::Stale | Verdict::Degraded, false) => "~",
2515        (Verdict::Failed, true) => "✗",
2516        (Verdict::Failed, false) => "x",
2517        (Verdict::Unavailable, true) => "○",
2518        (Verdict::Unavailable, false) => "o",
2519        (Verdict::NotApplicable, _) => "-",
2520        (_, _) => "?",
2521    }
2522}

2524/// The short form an entry prints: who and where, one line of the claims that matter most,
2525/// and the most urgent thing to look at.
2526///
2527/// ```
2528/// use majordomus_cli::environment::preflight::{compact, derive, Observations};
2529/// let text = compact(&derive(&Observations::empty("demo", 0)), true);
2530/// assert!(text.starts_with("◆ demo"));
2531/// assert!(text.contains("server ?"), "{text}");
2532/// assert!(!text.contains('✓'), "nothing observed, nothing marked proven: {text}");
2533/// ```
2534pub fn compact(p: &Preflight, unicode: bool) -> String {
2535    let bullet = if unicode { "·" } else { "-" };
2536    let episode = p
2537        .check("session.episode")
2538        .filter(|c| c.verdict.proves())
2539        .and_then(|c| c.evidence.first())
2540        .and_then(|e| e.observed.strip_prefix("session_id "))
2541        .map(|id| format!("episode {id}"))
2542        .unwrap_or_else(|| "no episode".into());
2543    let mut out = format!(
2544        "{} {} {bullet} {} {bullet} {episode}\n ",
2545        if unicode { "◆" } else { ">" },
2546        p.repository,
2547        p.branch.as_deref().unwrap_or("detached"),
2548    );
2549    let shown = [
2550        ("governance.rules", "rules"),
2551        ("session.context", "context"),
2552        ("integration.server", "server"),
2553        ("integration.mcp", "mcp"),
2554        ("integration.cockpit", "cockpit"),
2555        ("verification.tests", "tests"),
2556        ("verification.enforcement", "enforced"),
2557        ("verification.deployment", "deploy"),
2558    ];
2559    let parts: Vec<String> = shown
2560        .iter()
2561        .filter_map(|(id, label)| {
2562            p.check(id)
2563                .map(|c| format!("{label} {}", mark(c.verdict, unicode)))
2564        })
2565        .collect();
2566    let _ = write!(out, " {}", parts.join(&format!(" {bullet} ")));
2567    if let Some(first) = p.attention.first().and_then(|id| p.check(id)) {
2568        let _ = write!(
2569            out,
2570            "\n  {} {} {}: {}",
2571            mark(first.verdict, unicode),
2572            first.title,
2573            first.verdict.as_str(),
2574            first.summary
2575        );
2576        if p.attention.len() > 1 {
2577            let _ = write!(
2578                out,
2579                " (+{} more: majordomus env preflight)",
2580                p.attention.len() - 1
2581            );
2582        }
2583    }
2584    out.push('\n');
2585    out
2586}

2588/// The whole preflight, a section per heading and a line per claim, with its evidence.
2589///
2590/// ```
2591/// use majordomus_cli::environment::preflight::{full, derive, Observations};
2592/// let text = full(&derive(&Observations::empty("demo", 0)), true);
2593/// assert!(text.starts_with("MAJORDOMUS PREFLIGHT"));
2594/// assert!(text.contains("Verification"));
2595/// ```
2596pub fn full(p: &Preflight, unicode: bool) -> String {
2597    let mut out = format!(
2598        "MAJORDOMUS PREFLIGHT  {} {} {}\n",
2599        p.repository,
2600        p.branch.as_deref().unwrap_or("detached"),
2601        p.head.as_deref().map(short).unwrap_or("")
2602    );
2603    for section in &p.sections {
2604        let _ = write!(out, "\n{}\n", section.title);
2605        for c in &section.checks {
2606            let _ = writeln!(
2607                out,
2608                "  {:<13} {} {:<14} {}",
2609                c.title,
2610                mark(c.verdict, unicode),
2611                c.verdict.as_str(),
2612                c.summary
2613            );
2614            for e in &c.evidence {
2615                let _ = writeln!(
2616                    out,
2617                    "  {:<13}   {:<14} {}: {}",
2618                    "", "", e.source, e.observed
2619                );
2620            }
2621            if let Some(next) = &c.next {
2622                if c.verdict.urgency().is_some() {
2623                    let _ = writeln!(out, "  {:<13}   {:<14} next: {next}", "", "");
2624                }
2625            }
2626        }
2627    }
2628    if p.attention.is_empty() {
2629        out.push_str("\nNothing needs a look.\n");
2630    } else {
2631        let _ = writeln!(out, "\nNeeds a look: {}", p.attention.join(", "));
2632    }
2633    out
2634}
