1//! What the public site serves, observed once per report.
2//!
3//! The publication model is `.ai/repo/ci/pages.yaml`, and nothing here is a second one: the
4//! served identity document is the file its `deploy.identity` names, the published branch
5//! is its `deploy.branch`, and the site is the `base_url` of `site/config.toml` (or, for a
6//! repository without a site configuration, the installer's `base_url` of
7//! `share/distribution.yaml`, which is the same origin).
8//!
9//! Two readings, and one of them is not this module's. The identity document is fetched
10//! here, once, because `DEPLOYED` is an *ancestry* question — does the served commit contain
11//! the feature's revision — and `scripts/pages verify` answers an *equality* question: does
12//! the site serve exactly this commit. Verification is `scripts/pages`' own and is reused,
13//! not re-derived: `scripts/pages verify` probes the site again from outside with the cache
14//! defeated, and `scripts/pages built` reads GitHub's own build of the published branch.
15//! Their exit codes are the vocabulary — 0 yes, 10 no, 12 not determinable — and 12 is
16//! unknown here, never a pass.
17//!
18//! The network is injectable, so a test never reaches it: `MAJORDOMUS_DELIVERY_SITE_URL`
19//! replaces the site's address, and a `file://` directory holding an identity document is a
20//! site that both this module and `scripts/pages verify` read through the same `curl`.
21

31/// The variable that replaces the public site's address, for a fixture or a mirror.
32pub(crate) const SITE_URL_ENV: &str = "MAJORDOMUS_DELIVERY_SITE_URL";
33/// The publication model.
34pub(crate) const PAGES_MODEL: &str = ".ai/repo/ci/pages.yaml";
35/// The script that owns publication and its verification.
36pub(crate) const PAGES_SCRIPT: &str = "scripts/pages";
37/// Seconds one probe of the site may take, the bound `scripts/pages verify` uses too.
38const PROBE_SECONDS: u64 = 10;

40/// What reading the public identity document gave.
41#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
42#[serde(tag = "state", rename_all = "snake_case")]
43pub(crate) enum IdentityRead {
44    /// The site answered with an identity naming the commit it was built from.
45    Served {
46        /// The commit, as the site states it.
47        commit: String,
48        /// Whether the tree it was built from was dirty, when the identity says.
49        #[serde(default, skip_serializing_if = "Option::is_none")]
50        dirty: Option<bool>,
51    },
52    /// The site was asked and gave no identity: no network, no route, not a document.
53    Unreachable {
54        /// Why, in one clause.
55        reason: String,
56    },
57    /// There was nothing to ask: no site address or no publication model.
58    Unconfigured {
59        /// Why, in one clause.
60        reason: String,
61    },
62}
63
64/// Whether the publication was verified from outside.
65#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
66#[serde(tag = "state", rename_all = "snake_case")]
67pub(crate) enum SiteVerification {
68    /// Clean identity, a second probe agrees, GitHub's own build succeeded.
69    Verified {
70        /// What was observed.
71        detail: String,
72    },
73    /// Measured, and it does not hold.
74    Refused {
75        /// What was observed.
76        reason: String,
77    },
78    /// Could not be measured.
79    Unmeasured {
80        /// Why.
81        reason: String,
82    },
83}

85/// The publication as observed: where the site is, what it serves, and whether that was
86/// verified.
87#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
88pub(crate) struct Publication {
89    /// The site's address, when one was found.
90    #[serde(default, skip_serializing_if = "Option::is_none")]
91    pub url: Option<String>,
92    /// The identity document's path under it, from the publication model.
93    #[serde(default, skip_serializing_if = "Option::is_none")]
94    pub identity_path: Option<String>,
95    /// What reading it gave.
96    pub identity: IdentityRead,
97    /// Whether the publication was verified.
98    pub verification: SiteVerification,
99}

101/// The observer: where to look and with what.
102#[derive(Debug, Clone)]
103pub(crate) struct Probe {
104    root: PathBuf,
105    url: Result<String, String>,
106    model: Result<Map<String, Value>, String>,
107    curl: String,
108    pages: PathBuf,
109}
110
111impl Probe {
112    /// The probe of the repository at `root`, with the site address the environment may
113    /// replace.
114    pub(crate) fn for_repository(root: &Path) -> Self {
115        let url = match std::env::var(SITE_URL_ENV) {
116            Ok(v) if !v.trim().is_empty() => Ok(v.trim().to_string()),
117            _ => base_url(root),
118        };
119        Probe {
120            root: root.to_path_buf(),
121            url,
122            model: read_model(root),
123            curl: "curl".into(),
124            pages: root.join(PAGES_SCRIPT),
125        }
126    }
127
128    /// Observe the publication: read the identity, then verify it.
129    pub(crate) fn observe(&self) -> Publication {
130        let identity_path = self
131            .model
132            .as_ref()
133            .ok()
134            .and_then(|m| string_at(m, &["deploy", "identity"]));
135        let identity = match (&self.url, &self.model, &identity_path) {
136            (Err(reason), _, _) | (_, Err(reason), _) => IdentityRead::Unconfigured {
137                reason: reason.clone(),
138            },
139            (_, _, None) => IdentityRead::Unconfigured {
140                reason: format!("{PAGES_MODEL} names no deploy.identity"),
141            },
142            (Ok(url), Ok(_), Some(path)) => self.read_identity(url, path),
143        };
144        let verification = self.verify(&identity);
145        Publication {
146            url: self.url.as_ref().ok().cloned(),
147            identity_path,
148            identity,
149            verification,
150        }
151    }
152
153    fn read_identity(&self, url: &str, path: &str) -> IdentityRead {
154        let now = std::time::SystemTime::now()
155            .duration_since(std::time::UNIX_EPOCH)
156            .map(|d| d.as_secs())
157            .unwrap_or(0);
158        let target = format!("{}/{}", url.trim_end_matches('/'), path);
159        let out = Command::new(&self.curl)
160            .args(["-fsSL", "--max-time", &PROBE_SECONDS.to_string()])
161            .args(["-H", "Cache-Control: no-cache"])
162            .arg(format!("{target}?t={now}"))
163            .output();
164        let body = match out {
165            Ok(o) if o.status.success() => o.stdout,
166            Ok(o) => {
167                return IdentityRead::Unreachable {
168                    reason: format!(
169                        "{target} did not answer (curl exit {})",
170                        code(o.status.code())
171                    ),
172                }
173            }
174            Err(e) => {
175                return IdentityRead::Unreachable {
176                    reason: format!("{} could not run: {e}", self.curl),
177                }
178            }
179        };
180        let doc: Value = serde_json::from_slice(&body).unwrap_or(Value::Null);
181        match doc.get("commit").and_then(Value::as_str) {
182            Some(c) if c.len() >= 7 && c.chars().all(|ch| ch.is_ascii_hexdigit()) => {
183                IdentityRead::Served {
184                    commit: c.to_string(),
185                    dirty: doc.get("dirty").and_then(Value::as_bool),
186                }
187            }
188            _ => IdentityRead::Unreachable {
189                reason: format!("{target} answered without an identity naming a commit"),
190            },
191        }
192    }
193
194    fn verify(&self, identity: &IdentityRead) -> SiteVerification {
195        let unmeasured = |reason: String| SiteVerification::Unmeasured { reason };
196        let (commit, dirty) = match identity {
197            IdentityRead::Served { commit, dirty } => (commit, *dirty),
198            _ => return unmeasured("the public identity was not read".into()),
199        };
200        let short = commit.get(..12).unwrap_or(commit);
201        match dirty {
202            Some(true) => {
203                return SiteVerification::Refused {
204                    reason: format!("the public site was built from a dirty tree at {short}"),
205                }
206            }
207            None => {
208                return unmeasured(
209                    "the public identity does not say whether its tree was clean".into(),
210                )
211            }
212            Some(false) => {}
213        }
214        if !self.pages.is_file() {
215            return unmeasured(format!(
216                "this repository has no {PAGES_SCRIPT} to verify the publication with"
217            ));
218        }
219        let url = self.url.as_deref().unwrap_or_default();
220        match self.run_pages(&[
221            "verify",
222            "--url",
223            url,
224            "--commit",
225            commit,
226            "--timeout",
227            "0",
228            "--quiet",
229        ]) {
230            Some(0) => {}
231            Some(10) => {
232                return SiteVerification::Refused {
233                    reason: format!("a second probe of the site did not serve {short}"),
234                }
235            }
236            other => {
237                return unmeasured(format!(
238                    "{PAGES_SCRIPT} verify could not measure the site (exit {})",
239                    code(other)
240                ))
241            }
242        }
243        let branch = self
244            .model
245            .as_ref()
246            .ok()
247            .and_then(|m| string_at(m, &["deploy", "branch"]))
248            .unwrap_or_else(|| "gh-pages".into());
249        let published = crate::git::read_only(&self.root)
250            .args(["rev-parse", "--verify", "--quiet"])
251            .arg(format!("refs/remotes/origin/{branch}^{{commit}}"))
252            .output()
253            .ok()
254            .filter(|o| o.status.success())
255            .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string());
256        let Some(published) = published else {
257            return unmeasured(format!(
258                "this clone has no origin/{branch} to ask GitHub's build about"
259            ));
260        };
261        let pshort = published.get(..12).unwrap_or(&published).to_string();
262        match self.run_pages(&["built", "--commit", &published, "--porcelain"]) {
263            Some(0) => SiteVerification::Verified {
264                detail: format!(
265                    "the site serves {short} from a clean tree, a second probe agrees, and GitHub's own build of {branch} {pshort} is built"
266                ),
267            },
268            Some(10) => SiteVerification::Refused {
269                reason: format!("GitHub's own build of {branch} {pshort} errored"),
270            },
271            other => unmeasured(format!(
272                "GitHub's own build of {branch} {pshort} could not be read (exit {})",
273                code(other)
274            )),
275        }
276    }
277
278    fn run_pages(&self, args: &[&str]) -> Option<i32> {
279        Command::new(&self.pages)
280            .args(args)
281            .current_dir(&self.root)
282            .stdout(std::process::Stdio::null())
283            .stderr(std::process::Stdio::null())
284            .status()
285            .ok()
286            .and_then(|s| s.code())
287    }
288}

290fn code(c: Option<i32>) -> String {
291    c.map(|c| c.to_string()).unwrap_or_else(|| "none".into())
292}

294fn string_at(map: &Map<String, Value>, path: &[&str]) -> Option<String> {
295    let (last, parents) = path.split_last()?;
296    let mut node = map;
297    for key in parents {
298        node = node.get(*key)?.as_object()?;
299    }
300    node.get(*last)
301        .and_then(Value::as_str)
302        .map(str::trim)
303        .filter(|s| !s.is_empty())
304        .map(str::to_string)
305}

307fn read_model(root: &Path) -> Result<Map<String, Value>, String> {
308    let text = std::fs::read_to_string(root.join(PAGES_MODEL))
309        .map_err(|_| format!("this repository has no publication model at {PAGES_MODEL}"))?;
310    parse_mapping(&text).map_err(|e| format!("{PAGES_MODEL} does not parse: {e}"))
311}

313/// The site's address: `site/config.toml`'s `base_url`, else the installer's.
314fn base_url(root: &Path) -> Result<String, String> {
315    let from_site = std::fs::read_to_string(root.join(crate::web::discover::SITE_CONFIG))
316        .ok()
317        .and_then(|text| {
318            text.lines().find_map(|line| {
319                let (key, value) = line.split_once('=')?;
320                (key.trim() == "base_url").then(|| value.trim().trim_matches('"').to_string())
321            })
322        });
323    let from_installer = || {
324        std::fs::read_to_string(root.join("share/distribution.yaml"))
325            .ok()
326            .and_then(|text| parse_mapping(&text).ok())
327            .and_then(|m| string_at(&m, &["installer", "base_url"]))
328    };
329    from_site
330        .filter(|u| u.contains("://"))
331        .or_else(from_installer)
332        .ok_or_else(|| {
333            "no public site address (site/config.toml base_url, share/distribution.yaml installer.base_url)"
334                .to_string()
335        })
336}

349    impl Site {
350        /// A repository with a publication model, a site directory served as `file://`, and
351        /// a `scripts/pages` that answers `verify` and `built` with the given exits.
352        fn new(identity: Option<&str>, verify: i32, built: i32) -> Self {
353            let s = Site {
354                dir: tempfile::tempdir().unwrap(),
355            };
356            let root = s.root();
357            std::fs::create_dir_all(root.join(".ai/repo/ci")).unwrap();
358            std::fs::write(
359                root.join(PAGES_MODEL),
360                "version: 1\ndeploy:\n  branch: gh-pages\n  identity: build.json\n",
361            )
362            .unwrap();
363            std::fs::create_dir_all(root.join("public")).unwrap();
364            if let Some(body) = identity {
365                std::fs::write(root.join("public/build.json"), body).unwrap();
366            }
367            std::fs::create_dir_all(root.join("scripts")).unwrap();
368            let script = root.join(PAGES_SCRIPT);
369            std::fs::write(
370                &script,
371                format!("#!/bin/sh\ncase \"$1\" in verify) exit {verify};; built) exit {built};; esac\nexit 2\n"),
372            )
373            .unwrap();
374            std::fs::set_permissions(&script, std::fs::Permissions::from_mode(0o755)).unwrap();
375            let git = |args: &[&str]| {
376                let ok = Command::new("git")
377                    .arg("-C")
378                    .arg(&root)
379                    .args(args)
380                    .output()
381                    .unwrap()
382                    .status
383                    .success();
384                assert!(ok, "git {args:?}");
385            };
386            git(&["init", "-q", "."]);
387            git(&[
388                "-c",
389                "user.email=t@e",
390                "-c",
391                "user.name=t",
392                "commit",
393                "-q",
394                "--allow-empty",
395                "-m",
396                "p",
397            ]);
398            git(&["update-ref", "refs/remotes/origin/gh-pages", "HEAD"]);
399            s
400        }
401
402        fn root(&self) -> PathBuf {
403            self.dir.path().canonicalize().unwrap()
404        }
405
406        fn probe(&self) -> Probe {
407            let mut p = Probe::for_repository(&self.root());
408            p.url = Ok(format!("file://{}", self.root().join("public").display()));
409            p
410        }
411    }

521    #[test]
522    fn the_address_and_the_model_come_from_the_repository() {
523        let dir = tempfile::tempdir().unwrap();
524        let root = dir.path();
525        // nothing configured: nothing to ask
526        let p = Probe::for_repository(root);
527        let observed = p.observe();
528        assert!(matches!(
529            observed.identity,
530            IdentityRead::Unconfigured { .. }
531        ));
532        assert!(observed.url.is_none() || std::env::var(SITE_URL_ENV).is_ok());
533        assert!(base_url(root).is_err());
534        // the installer's address stands in for a repository with no site configuration
535        std::fs::create_dir_all(root.join("share")).unwrap();
536        std::fs::write(
537            root.join("share/distribution.yaml"),
538            "installer:\n  base_url: https://installer.example\n",
539        )
540        .unwrap();
541        assert_eq!(base_url(root).as_deref(), Ok("https://installer.example"));
542        // and the site's own configuration wins
543        std::fs::create_dir_all(root.join("site")).unwrap();
544        std::fs::write(
545            root.join("site/config.toml"),
546            "title = \"x\"\nbase_url = \"https://site.example\"\n",
547        )
548        .unwrap();
549        assert_eq!(base_url(root).as_deref(), Ok("https://site.example"));
550        // a model without an identity names nothing to read
551        std::fs::create_dir_all(root.join(".ai/repo/ci")).unwrap();
552        std::fs::write(
553            root.join(PAGES_MODEL),
554            "version: 1\ndeploy:\n  branch: gh-pages\n",
555        )
556        .unwrap();
557        let mut p = Probe::for_repository(root);
558        p.url = Ok("https://site.example".into());
559        assert!(matches!(
560            p.observe().identity,
561            IdentityRead::Unconfigured { reason } if reason.contains("deploy.identity")
562        ));
563        assert_eq!(code(None), "none");
564    }
