1//! The `integration` module: the pull-request integration queue, projected (ADR 0101), and
2//! the trains of stacked pull requests in it (ADR 0138).
3//!
4//! Six questions, all read-only and all offline but the dry-run proof. Each renders the queue
5//! [`crate::integration::queue_of`] builds from the last recorded forge observation and
6//! this clone's fetched master — the value `majordomus prs status` prints — so the HTTP
7//! route, the MCP tool, the Cockpit and the command line cannot disagree. None reaches the
8//! network: observing the forge is `majordomus prs refresh`, and merging is
9//! `majordomus prs drain`, both command-line operations of the same module, because a
10//! capability that writes the repository or a remote must be declared as one and the
11//! shared server does not take that on.

35/// The queue, or why there is none.
36#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
37pub struct IntegrationStatus {
38    /// Whether a forge observation is recorded in this checkout.
39    pub observed: bool,
40    /// Why there is no queue, when there is none.
41    pub reason: Option<String>,
42    /// The queue.
43    pub queue: Option<IntegrationQueue>,
44    /// The base branch's integration lease as an observer reads it: `None` when nobody
45    /// holds it (or nothing is observed, so no base is known).
46    #[serde(default)]
47    pub lease: Option<IntegrationLeaseState>,
48    /// The last merge the executor recorded, from the audit trail.
49    #[serde(default)]
50    pub last_merge: Option<IntegrationEvent>,
51    /// How fast the executor has turned work into master over the last
52    /// [`THROUGHPUT_WINDOW_DAYS`] days, folded from the same trail.
53    pub throughput: IntegrationThroughput,
54}

67/// What cleanup would do, decided offline: the pull requests it would close or leave for a
68/// person, from the recorded observation, and the last branch report `prs cleanup` recorded.
69#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
70pub struct IntegrationCleanup {
71    /// Whether a forge observation is recorded in this checkout.
72    pub observed: bool,
73    /// Why there is no plan, when there is none.
74    pub reason: Option<String>,
75    /// When the forge was observed, when it was.
76    pub observed_at: Option<String>,
77    /// The plan, in rank order: `would_close` for what is provably on master or superseded by
78    /// a successor that landed, `left_for_a_person` for weak evidence and for what a person
79    /// marked obsolete. Nothing here closes anything.
80    pub items: Vec<CleanupItem>,
81    /// The branches merged pull requests left on origin, as `prs cleanup` last read them;
82    /// `None` when it never ran in this checkout. Never read from the network here.
83    pub branches: Option<LeftBranchReport>,
84    /// How old that report is, in seconds, when its time reads.
85    pub branches_age_seconds: Option<u64>,
86}
87
88/// [`IntegrationCleanup`] for the checkout at `root`, at `now` (seconds since the epoch):
89/// [`crate::integration::drain::cleanup_plan`] over [`crate::integration::queue_of`], which
90/// writes nothing, and the recorded branch report with its age. No network, no write.
91pub fn cleanup_status(root: &Path, now: u64) -> IntegrationCleanup {
92    let branches = crate::integration::drain::recorded_left_branches(root);
93    let branches_age_seconds = branches
94        .as_ref()
95        .and_then(|b| crate::peers::epoch_seconds(&b.read_at))
96        .and_then(|at| u64::try_from(at).ok())
97        .map(|at| now.saturating_sub(at));
98    match crate::integration::queue_of(root) {
99        Ok(q) => IntegrationCleanup {
100            observed: true,
101            reason: None,
102            observed_at: Some(q.observed_at.clone()),
103            items: crate::integration::drain::cleanup_plan(&q),
104            branches,
105            branches_age_seconds,
106        },
107        Err(reason) => IntegrationCleanup {
108            observed: false,
109            reason: Some(reason),
110            observed_at: None,
111            items: Vec::new(),
112            branches,
113            branches_age_seconds,
114        },
115    }
116}

166fn integration_queue(ctx: &Context, _: Empty) -> Result<IntegrationStatus, CapabilityError> {
167    let root = root(ctx);
168    let trail = crate::integration::drain::events(root);
169    let last_merge = trail
170        .iter()
171        .rev()
172        .find(|e| e.action == crate::integration::drain::IntegrationAction::MergeSucceeded)
173        .cloned();
174    let now = std::time::SystemTime::now()
175        .duration_since(std::time::UNIX_EPOCH)
176        .map(|d| d.as_secs())
177        .unwrap_or(0);
178    let throughput = metrics::throughput(&trail, now, THROUGHPUT_WINDOW_DAYS);
179    Ok(match crate::integration::queue_of(root) {
180        Ok(q) => IntegrationStatus {
181            observed: true,
182            reason: None,
183            lease: IntegrationLease::read(root, &q.base).ok().flatten(),
184            queue: Some(q),
185            last_merge,
186            throughput,
187        },
188        Err(reason) => IntegrationStatus {
189            observed: false,
190            reason: Some(reason),
191            queue: None,
192            lease: None,
193            last_merge,
194            throughput,
195        },
196    })
197}

199fn integration_explain(
200    ctx: &Context,
201    input: PullRequestExplainInput,
202) -> Result<IntegrationExplanation, CapabilityError> {
203    let q = match crate::integration::queue_of(root(ctx)) {
204        Ok(q) => q,
205        Err(reason) => {
206            return Ok(IntegrationExplanation {
207                found: false,
208                reason: Some(reason),
209                rank: None,
210                of: 0,
211                observed_at: None,
212                assessment: None,
213            })
214        }
215    };
216    let rank = q
217        .assessments
218        .iter()
219        .position(|a| a.number == input.number)
220        .map(|i| i + 1);
221    Ok(IntegrationExplanation {
222        found: rank.is_some(),
223        reason: rank.is_none().then(|| {
224            format!(
225                "#{} is not among the {} open pull request(s) observed",
226                input.number, q.tallies.open
227            )
228        }),
229        rank,
230        of: q.assessments.len(),
231        observed_at: Some(q.observed_at.clone()),
232        assessment: q.get(input.number).cloned(),
233    })
234}

242fn integration_cleanup(ctx: &Context, _: Empty) -> Result<IntegrationCleanup, CapabilityError> {
243    let now = std::time::SystemTime::now()
244        .duration_since(std::time::UNIX_EPOCH)
245        .map(|d| d.as_secs())
246        .unwrap_or(0);
247    Ok(cleanup_status(root(ctx), now))
248}

250fn integration_timings(ctx: &Context, _: Empty) -> Result<IntegrationTimings, CapabilityError> {
251    Ok(match crate::integration::load_observation(root(ctx)) {
252        Ok(Some(obs)) => IntegrationTimings {
253            observed: true,
254            reason: None,
255            timings: Some(crate::integration::timings::timings(&obs)),
256        },
257        Ok(None) => IntegrationTimings {
258            observed: false,
259            reason: Some(
260                "no forge observation is recorded in this checkout; majordomus prs refresh".into(),
261            ),
262            timings: None,
263        },
264        Err(e) => IntegrationTimings {
265            observed: false,
266            reason: Some(e),
267            timings: None,
268        },
269    })
270}

449    /// The timings are measured from the recorded observation when there is one, and an
450    /// observation that cannot be read is said as the reason, never as no observation.
451    #[test]
452    fn the_timings_are_measured_from_the_observation_or_say_why_not() {
453        use crate::integration::{
454            state_path, store_observation, ForgeObservation, OBSERVATION_FILE, OBSERVATION_SCHEMA,
455        };
456        let repo = crate::synthetic::SyntheticRepository::small().expect("a repository");
457        let ctx = repo.context().expect("a context");
458        let root = Path::new(&ctx.index.repository.root).to_path_buf();
459
460        let none = integration_timings(&ctx, Empty {}).unwrap();
461        assert!(!none.observed && none.timings.is_none());
462        assert!(none.reason.unwrap().contains("prs refresh"));
463
464        store_observation(
465            &root,
466            &ForgeObservation {
467                schema: OBSERVATION_SCHEMA,
468                repository: "owner/repo".into(),
469                base: "master".into(),
470                base_sha: "0".repeat(40),
471                observed_at: "2026-10-01T00:00:00Z".into(),
472                required_checks: Some(vec!["ci".into()]),
473                review_policy: None,
474                up_to_date_required: None,
475                merge_methods: vec!["merge".into()],
476                pull_requests: Vec::new(),
477                resolved: Default::default(),
478                delete_branch_on_merge: None,
479            },
480        )
481        .unwrap();
482        let read = integration_timings(&ctx, Empty {}).unwrap();
483        assert!(read.observed && read.reason.is_none(), "{read:?}");
484        assert!(read.timings.is_some());
485
486        std::fs::write(state_path(&root, OBSERVATION_FILE), "not an observation").unwrap();
487        let unreadable = integration_timings(&ctx, Empty {}).unwrap();
488        assert!(!unreadable.observed && unreadable.timings.is_none());
489        let reason = unreadable.reason.unwrap();
490        assert!(!reason.contains("prs refresh"), "{reason}");
491    }
