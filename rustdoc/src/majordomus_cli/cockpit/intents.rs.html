1//! The intents area: what must become true, how far reality is from it, which work is making
2//! it true, and the test behind every criterion.
3//!
4//! Two pages, both projections. The list is `intent_realization.work`; one intent is
5//! `intent_realization.explain`. Nothing here derives a stage, grades a link or decides
6//! whether a criterion is met: every word on the page is a word a capability answered, and
7//! the links go to the objects the index already holds — the test a criterion names, the
8//! issues serving it, the milestones realising the intent (ADR 0075).

27/// The status colour a stage is read in, the stage itself being the label.
28fn stage_badge(stage: IntentStage) -> El {
29    let status = match stage {
30        IntentStage::Satisfied => "ok",
31        IntentStage::Verifying | IntentStage::Cancelled => "warn",
32        IntentStage::Executing => "info",
33        IntentStage::Superseded => "bad",
34        IntentStage::Declared | IntentStage::Planned => "neutral",
35    };
36    badge(status, stage.as_str())
37}

39/// The status colour a verdict is read in, the verdict itself being the label.
40fn verdict_badge(verdict: IntentVerdictState) -> El {
41    let status = match verdict {
42        IntentVerdictState::Satisfied => "ok",
43        IntentVerdictState::Unsatisfied => "bad",
44        IntentVerdictState::Unknown => "neutral",
45    };
46    badge(status, verdict.as_str())
47}

49/// A criterion's state, and for a met one the verdict it rests on: a proof at this revision
50/// and a pass whose named inputs are unchanged both meet it, and never wear the same badge.
51fn evidence_badge(state: IntentEvidenceState, proof: Option<ProofState>) -> El {
52    let (status, label) = match state {
53        IntentEvidenceState::Current => match proof {
54            Some(ProofState::Proven) => ("ok", "current · proven"),
55            Some(ProofState::InputsUnchanged) => ("info", "current · inputs unchanged"),
56            _ => ("ok", "current"),
57        },
58        IntentEvidenceState::Stale => ("warn", "stale"),
59        IntentEvidenceState::Failing => ("bad", "failing"),
60        IntentEvidenceState::NotRun => ("neutral", "not run"),
61        IntentEvidenceState::NotDerivable => ("neutral", "not derivable"),
62        IntentEvidenceState::Unresolved => ("bad", "unresolved"),
63    };
64    badge(status, label)
65}

67fn provenance_badge(p: IntentLinkProvenance) -> El {
68    let status = match p {
69        IntentLinkProvenance::Declared => "ok",
70        IntentLinkProvenance::Observed => "info",
71        IntentLinkProvenance::Derived => "neutral",
72        IntentLinkProvenance::Inferred => "warn",
73    };
74    badge(status, p.as_str())
75}

77fn object_href(uri: &str) -> String {
78    format!("/cockpit/object?uri={}", percent_encode(uri))
79}
80
81/// The object page of an index object, when the index holds it; otherwise the reference as
82/// text, so a link never points at a page that would answer 404. Whether it is held is asked
83/// of `objects.get`, the capability every surface resolves a URI with, rather than read from
84/// the index beside it (ADR 0012).
85fn object_link(ctx: &Context, kind: &str, identity: &str, label: &str) -> El {
86    let uri = format!("majordomus://{kind}/{identity}");
87    match ctx.execute("objects.get", json!({ "uri": uri })) {
88        Ok(_) => link(object_href(&uri), label),
89        Err(_) => mono(label.to_string()),
90    }
91}

93/// How many criteria have current evidence, as the Cockpit's progress bar.
94fn criteria_bar(met: usize, total: usize) -> El {
95    let percent = (met * 100).checked_div(total).unwrap_or(0);
96    el("div")
97        .class("mj-progress")
98        .attr("role", "progressbar")
99        .attr("aria-valuenow", percent.to_string())
100        .attr("aria-valuemin", "0")
101        .attr("aria-valuemax", "100")
102        .attr(
103            "aria-label",
104            format!("{met} of {total} criteria have current evidence"),
105        )
106        .child(
107            el("div")
108                .class("mj-progress-bar")
109                .attr("style", format!("width:{percent}%")),
110        )
111        .child(
112            el("span")
113                .class("mj-progress-text")
114                .text(format!("{met}/{total}")),
115        )
116}

118/// One unit of work as one intent sees it: its strongest link to that intent and the issues it
119/// links through. A unit with no link to the intent shows a dash rather than a grade.
120fn work_row(intent: &str, w: &IntentRealizedWork) -> El {
121    let links: Vec<&crate::intent_realization::IntentLink> =
122        w.links.iter().filter(|l| l.intent == intent).collect();
123    let strongest = links.iter().map(|l| l.provenance).min();
124    let issues: Vec<String> = links
125        .iter()
126        .map(|l| format!("{} ({})", l.issue, l.via.as_str()))
127        .collect();
128    row(vec![
129        text_cell(w.work.kind.as_str()),
130        cell(mono(w.work.id.clone())),
131        text_cell(w.work.outcome.clone()),
132        cell(match strongest {
133            Some(p) => provenance_badge(p),
134            None => el("span").text("—"),
135        }),
136        text_cell(issues.join(", ")),
137        text_cell(w.work.providers().join(", ")),
138        text_cell(w.work.handovers.len().to_string()),
139    ])
140}

148/// Every intent with how far reality is from it and who is realising it.
149pub fn list(ctx: &Context) -> Page {
150    let r: IntentRealization = match ask(ctx, "intent_realization.work", json!({})) {
151        Ok(r) => r,
152        Err(e) => return failed(Area::Intents, "Intents", e),
153    };
154    let met: usize = r.intents.iter().map(|i| i.met).sum();
155    let criteria: usize = r.intents.iter().map(|i| i.criteria).sum();
156    let rows: Vec<El> = r
157        .intents
158        .iter()
159        .map(|i| {
160            row(vec![
161                id_cell(intent_href(&i.intent), i.intent.clone()),
162                cell(stage_badge(i.stage)),
163                text_cell(format!("{}/{}", i.met, i.criteria)),
164                text_cell(i.work.len().to_string()),
165                text_cell(if i.providers.is_empty() {
166                    "—".to_string()
167                } else {
168                    i.providers.join(", ")
169                }),
170                text_cell(i.findings.len().to_string()),
171                text_cell(i.title.clone()),
172            ])
173        })
174        .collect();
175    let body = el("div")
176        .class("mj-grid")
177        .child(
178            el("div")
179                .class("mj-stats")
180                .child(statistic(r.intents.len().to_string(), "intents", "intent_realization.work"))
181                .child(statistic(format!("{met}/{criteria}"), "criteria met", "intent_realization.work"))
182                .child(statistic(
183                    r.orphans.to_string(),
184                    "units of work serving no intent",
185                    "intent_realization.work",
186                )),
187        )
188        .child(card(
189            "Intents",
190            if rows.is_empty() {
191                nothing("This repository declares no intent. One is a YAML record of the intent kind; `majordomus intent --help` says how to declare one.")
192            } else {
193                table(
194                    &["Intent", "Stage", "Met", "Work", "Providers", "Findings", "Title"],
195                    rows,
196                )
197            },
198        ))
199        .when(!r.findings.is_empty(), |d| d.child(findings_card(&r.findings)));
200    Page::new(Area::Intents, "Intents", body)
201        .subtitle(
202            "What must become true, derived from the plan and the recorded evidence on every read",
203        )
204        .trail(vec![("Cockpit", Some("/cockpit")), ("Intents", None)])
205}

207fn findings_card(findings: &[crate::intent::IntentFinding]) -> El {
208    let mut list = el("div");
209    for f in findings {
210        let level = if f.level == crate::intent::FAIL {
211            "fail"
212        } else {
213            "warn"
214        };
215        list = list.child(alert(
216            level,
217            format!("{} — {}: {}", f.code, f.subject, f.message),
218        ));
219    }
220    card("Findings", list)
221}

225/// One intent: the statement, the distance to it, each criterion with its test and evidence,
226/// the milestones and issues realising it, the work and providers, and why it stands where it
227/// stands.
228pub fn intent(ctx: &Context, id: &str) -> Page {
229    let e: IntentExplanation = match ask(ctx, "intent_realization.explain", json!({ "id": id })) {
230        Ok(e) => e,
231        Err(reason) => {
232            return Page::new(
233                Area::Intents,
234                "No such intent",
235                el("div")
236                    .child(alert("fail", reason))
237                    .child(link("/cockpit/intents", "Every intent")),
238            )
239            .status(404)
240            .trail(vec![
241                ("Cockpit", Some("/cockpit")),
242                ("Intents", Some("/cockpit/intents")),
243                (id, None),
244            ])
245        }
246    };
247    let i = &e.intent;
248    let total = i.satisfaction.len();
249
250    let summary = card(
251        "How far reality is from it",
252        el("div")
253            .child(el("p").class("mj-prose").text(&i.statement))
254            .child(
255                el("div")
256                    .class("mj-stats")
257                    .child(statistic(
258                        i.stage.as_str().to_string(),
259                        "stage",
260                        "intent_realization.explain",
261                    ))
262                    .child(statistic(
263                        format!("{}/{total}", i.met),
264                        "criteria with current evidence",
265                        "intent_realization.explain",
266                    ))
267                    .child(statistic(
268                        e.realization.work.len().to_string(),
269                        "units of work",
270                        "intent_realization.explain",
271                    )),
272            )
273            .child(criteria_bar(i.met, total))
274            .child(facts(vec![
275                ("Stage", Node::Element(stage_badge(i.stage))),
276                ("Verdict", Node::Element(verdict_badge(i.verdict.state))),
277                (
278                    "Record",
279                    Node::Element(object_link(ctx, "intent", &i.id, &i.source)),
280                ),
281            ])),
282    );
283
284    // each criterion with the test it names and the issues declaring they serve it
285    let criteria_rows: Vec<El> = i
286        .satisfaction
287        .iter()
288        .map(|c| {
289            let evidence = if c.evidence == "test" {
290                object_link(ctx, "test", &c.reference, &c.reference)
291            } else {
292                mono(format!("{} {}", c.evidence, c.reference))
293            };
294            let issues: Vec<String> = e
295                .coverage
296                .iter()
297                .find(|k| k.criterion == c.id)
298                .map(|k| k.issues.iter().map(|x| x.id.clone()).collect())
299                .unwrap_or_default();
300            let mut served = el("span");
301            if issues.is_empty() {
302                served = served.text("—");
303            }
304            for (n, issue) in issues.iter().enumerate() {
305                if n > 0 {
306                    served = served.text(" ");
307                }
308                served = served.child(object_link(ctx, "issue", issue, issue));
309            }
310            row(vec![
311                text_cell(c.id.clone()),
312                text_cell(c.criterion.clone()),
313                cell(evidence_badge(c.state, c.proof)),
314                cell(evidence),
315                cell(served),
316                cell(match &c.reproduce {
317                    Some(r) => mono(r.clone()),
318                    None => el("span").text("—"),
319                }),
320            ])
321        })
322        .collect();
323    let criteria = card(
324        "Criteria and the evidence behind each",
325        table(
326            &[
327                "Criterion",
328                "What must be true",
329                "Evidence",
330                "Test",
331                "Served by",
332                "Reproduce",
333            ],
334            criteria_rows,
335        ),
336    );
337
338    let milestone_rows: Vec<El> = i
339        .milestones
340        .iter()
341        .map(|m| {
342            row(vec![
343                cell(object_link(ctx, "milestone", &m.id, &m.id)),
344                text_cell(
345                    m.status
346                        .clone()
347                        .unwrap_or_else(|| "does not resolve".into()),
348                ),
349            ])
350        })
351        .collect();
352    let milestones = card(
353        "Milestones realising it",
354        table(&["Milestone", "Status"], milestone_rows),
355    );
356
357    let work_rows: Vec<El> = e.work.iter().map(|w| work_row(&i.id, w)).collect();
358    let work = card(
359        "Work realising it",
360        if work_rows.is_empty() {
361            nothing("No recorded task, session or live claim realises this intent.")
362        } else {
363            table(
364                &[
365                    "Kind",
366                    "Work",
367                    "Outcome",
368                    "Link",
369                    "Via issue",
370                    "Providers",
371                    "Handovers",
372                ],
373                work_rows,
374            )
375        },
376    );
377
378    let mut because = el("ul").class("mj-list");
379    for b in &e.because {
380        because = because.child(el("li").text(b));
381    }
382
383    let body = el("div")
384        .class("mj-grid")
385        .child(summary)
386        .child(criteria)
387        .child(milestones)
388        .child(work)
389        .child(card("Why it stands where it stands", because))
390        .when(!e.realization.findings.is_empty(), |d| {
391            d.child(findings_card(&e.realization.findings))
392        });
393    Page::new(Area::Intents, i.title.clone(), body)
394        .subtitle(format!("intent {}", i.id))
395        .trail(vec![
396            ("Cockpit", Some("/cockpit")),
397            ("Intents", Some("/cockpit/intents")),
398            (&i.id, None),
399        ])
400}

409    #[test]
410    fn every_stage_state_and_provenance_is_read_in_its_own_colour() {
411        for (stage, status) in [
412            (IntentStage::Declared, "neutral"),
413            (IntentStage::Planned, "neutral"),
414            (IntentStage::Executing, "info"),
415            (IntentStage::Verifying, "warn"),
416            (IntentStage::Satisfied, "ok"),
417            (IntentStage::Cancelled, "warn"),
418            (IntentStage::Superseded, "bad"),
419        ] {
420            let html = stage_badge(stage).render();
421            assert_eq!(html, badge(status, stage.as_str()).render(), "{stage:?}");
422        }
423        for (verdict, status) in [
424            (IntentVerdictState::Satisfied, "ok"),
425            (IntentVerdictState::Unsatisfied, "bad"),
426            (IntentVerdictState::Unknown, "neutral"),
427        ] {
428            let html = verdict_badge(verdict).render();
429            assert_eq!(
430                html,
431                badge(status, verdict.as_str()).render(),
432                "{verdict:?}"
433            );
434        }
435        for (state, status, label) in [
436            (IntentEvidenceState::Current, "ok", "current"),
437            (IntentEvidenceState::Stale, "warn", "stale"),
438            (IntentEvidenceState::Failing, "bad", "failing"),
439            (IntentEvidenceState::NotRun, "neutral", "not run"),
440            (
441                IntentEvidenceState::NotDerivable,
442                "neutral",
443                "not derivable",
444            ),
445            (IntentEvidenceState::Unresolved, "bad", "unresolved"),
446        ] {
447            assert_eq!(
448                evidence_badge(state, None).render(),
449                badge(status, label).render(),
450                "{state:?}"
451            );
452        }
453        // a met criterion says which verdict it rests on, and the two never render alike
454        let proven = evidence_badge(IntentEvidenceState::Current, Some(ProofState::Proven));
455        let unchanged = evidence_badge(
456            IntentEvidenceState::Current,
457            Some(ProofState::InputsUnchanged),
458        );
459        assert_eq!(proven.render(), badge("ok", "current · proven").render());
460        assert_eq!(
461            unchanged.render(),
462            badge("info", "current · inputs unchanged").render()
463        );
464        assert_ne!(proven.render(), unchanged.render());
465        for (p, status) in [
466            (IntentLinkProvenance::Declared, "ok"),
467            (IntentLinkProvenance::Observed, "info"),
468            (IntentLinkProvenance::Derived, "neutral"),
469            (IntentLinkProvenance::Inferred, "warn"),
470        ] {
471            assert_eq!(
472                provenance_badge(p).render(),
473                badge(status, p.as_str()).render(),
474                "{p:?}"
475            );
476        }
477    }
