1//! Loading the graph, at two speeds.
2//!
3//! A completion runs on every TAB and must feel like nothing at all; `commands graph` runs
4//! when a person asks and may take a moment. The difference is not a different graph — it
5//! is which inputs are read:
6//!
7//! | | clap tree | shell tool registry | capability registry | workflows | cache |
8//! |---|---|---|---|---|---|
9//! | [`fast`] | walked | read | composed | from the cache, when there is one | read |
10//! | [`full`] | walked | read | composed | asked of the runner | written |
11//!
12//! Neither loads the repository index. The index takes seconds to build and answers a
13//! question the command graph does not ask: nothing about *what commands exist* depends on
14//! what is in `.ai/`. That is the whole reason the completion can be synchronous.
15//!
16//! The workflows are the one input that costs a subprocess, so they are cached: [`full`]
17//! writes the graph beside the bridge it materialises, and [`fast`] reads it back. A cache
18//! that is missing or from another version of the executable is not an error — the fast
19//! path answers from the two declarations it can read directly, which is every command of
20//! both programs and none of the repository's own workflows.
21

34/// The file the full load writes and the fast load reads.
35pub const CACHE: &str = "graph.json";
36
37/// A loaded graph and where it came from.
38pub struct Loaded {
39    /// The graph.
40    pub graph: CommandGraph,
41    /// The repository root it describes.
42    pub root: PathBuf,
43    /// Whether the workflows in it came from the cache rather than from the runner.
44    pub cached: bool,
45}
46
47/// The capability registry, composed from the modules and nothing else.
48///
49/// No index, so no declarative capabilities and no filesystem walk. Every command-line
50/// exposure is declared by a builtin module, so the join is complete without one. Public
51/// because the completion answers `ValueSource::Capability` from it: composing the modules
52/// is arithmetic over compiled-in descriptors, which is why it is affordable on the hot
53/// path when building the index is not.
54pub fn registry() -> Option<CapabilityRegistry> {
55    CapabilityRegistry::builder()
56        .with_modules(builtin::modules())
57        .build()
58        .ok()
59}
60
61/// The whole graph, with the workflows asked of the runner, and the cache refreshed.
62pub fn full(repo: &RepoArgs) -> Result<Loaded> {
63    let root = root_of(repo)?;
64    let share = Share::locate(repo.share.as_deref(), &root).ok();
65    let registry = registry();
66    let catalogue = workflows::resolve(&root);
67    let graph = build(&Inputs {
68        registry: registry.as_ref(),
69        share: share.as_ref().map(|s| s.dir()),
70        workflows: Some(&catalogue),
71        bridged: bridge::bridged(&root),
72    });
73    let _ = write_cache(&root, &graph);
74    Ok(Loaded {
75        graph,
76        root,
77        cached: false,
78    })
79}
80
81/// The graph as fast as it can be had: no subprocess, no index, no network.
82pub fn fast(repo: &RepoArgs) -> Result<Loaded> {
83    let root = root_of(repo)?;
84    if let Some(graph) = read_cache(&root) {
85        return Ok(Loaded {
86            graph,
87            root,
88            cached: true,
89        });
90    }
91    let share = Share::locate(repo.share.as_deref(), &root).ok();
92    let registry = registry();
93    let graph = build(&Inputs {
94        registry: registry.as_ref(),
95        share: share.as_ref().map(|s| s.dir()),
96        workflows: None,
97        bridged: Default::default(),
98    });
99    Ok(Loaded {
100        graph,
101        root,
102        cached: false,
103    })
104}

106/// The graph of a repository whose root is already known, without asking the runner.
107///
108/// What a server answers from: it holds an index and therefore a root, it must not spawn a
109/// subprocess per request, and the cache the last materialisation wrote is exactly the
110/// answer it wants.
111pub fn fast_at(root: &Path, share: Option<&Path>) -> CommandGraph {
112    if let Some(graph) = read_cache(root) {
113        return graph;
114    }
115    let registry = registry();
116    build(&Inputs {
117        registry: registry.as_ref(),
118        share,
119        workflows: None,
120        bridged: Default::default(),
121    })
122}
123
124/// The whole graph of a repository whose root is already known.
125pub fn full_at(root: &Path, share: Option<&Path>) -> CommandGraph {
126    let registry = registry();
127    let catalogue = workflows::resolve(root);
128    build(&Inputs {
129        registry: registry.as_ref(),
130        share,
131        workflows: Some(&catalogue),
132        bridged: bridge::bridged(root),
133    })
134}

136/// Where the repository is.
137pub fn root(repo: &RepoArgs) -> Result<PathBuf> {
138    root_of(repo)
139}

141/// Where the repository is.
142fn root_of(repo: &RepoArgs) -> Result<PathBuf> {
143    let start = repo
144        .repo
145        .clone()
146        .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| PathBuf::from(".")));
147    Ok(Repository::discover(&start)?.root().to_path_buf())
148}

150/// The cached graph, when there is one this executable wrote.
151///
152/// A cache from another schema is ignored rather than migrated: it costs one subprocess to
153/// produce a correct one, and reading a document whose shape is not known is how a cache
154/// becomes a source of wrong answers.
155pub fn read_cache(root: &Path) -> Option<CommandGraph> {
156    let text = std::fs::read_to_string(root.join(bridge::DIRECTORY).join(CACHE)).ok()?;
157    let graph: CommandGraph = serde_json::from_str(&text).ok()?;
158    (graph.schema == super::SCHEMA).then_some(graph)
159}
160
161/// Write the cache, beside the bridge, atomically.
162pub fn write_cache(root: &Path, graph: &CommandGraph) -> std::io::Result<()> {
163    let dir = root.join(bridge::DIRECTORY);
164    std::fs::create_dir_all(&dir)?;
165    let path = dir.join(CACHE);
166    let tmp = path.with_extension(format!("tmp.{}", std::process::id()));
167    std::fs::write(
168        &tmp,
169        serde_json::to_vec(graph).map_err(std::io::Error::other)?,
170    )?;
171    std::fs::rename(&tmp, &path)
172}
173
174/// The repository root, for a caller that has already discovered it.
175pub fn cache_path(root: &Path) -> PathBuf {
176    root.join(bridge::DIRECTORY).join(CACHE)
177}

184#[cfg(test)]
185mod tests {
186    use super::*;
187
188    #[test]
189    fn a_cache_round_trips_and_a_foreign_one_is_ignored() {
190        let dir = tempfile::tempdir().expect("a temporary directory");
191        let graph = build(&Inputs::default());
192        write_cache(dir.path(), &graph).expect("the cache is written");
193        let read = read_cache(dir.path()).expect("the cache is read back");
194        assert_eq!(read.fingerprint, graph.fingerprint);
195        assert_eq!(read.commands.len(), graph.commands.len());
196
197        std::fs::write(cache_path(dir.path()), br#"{"schema":"other/v9"}"#).unwrap();
198        assert!(read_cache(dir.path()).is_none());
199    }
