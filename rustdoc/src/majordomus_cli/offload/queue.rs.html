1//! The FIFO queue on a host: who derives next, decided in Rust over what the host reports.
2//!
3//! The v1 script waited with `until mkdir lock; do sleep 15; done`. That is a race, not a
4//! queue: on 2026-10-11 five sessions waited on lundra, the oldest for 2 h 12 min, while
5//! later arrivals won the lock. Here every waiter writes a ticket named
6//! `<arrival>-<pid>-<tag>`, where the pid is its [`Holder`](super::remote::Holder) on the
7//! host, and takes the lock only when its ticket is the oldest one whose holder is alive.
8//!
9//! - A ticket whose holder is dead is removed by whoever sees it, so a waiter that was
10//!   killed or lost its connection blocks nobody.
11//! - The lock is a directory made by `mkdir` holding one empty file per process of the
12//!   derive that holds it, named by its pid. When every one of those processes is dead, the
13//!   lock is reclaimed, by any waiter, and the reclaim names the dead pids.
14//! - A lock directory with no pid file in it was taken by a v1 or v2 waiter (or is being
15//!   taken this instant) and cannot be checked: it is waited for, never reclaimed.
16//!
17//! The paths and the ticket names are the v2 script's, so waiters of both kinds see each
18//! other's tickets while sessions move over.
19

27/// One waiter's ticket as the host reported it.
28#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
29pub struct Ticket {
30    /// The file name: `<arrival>-<pid>-<tag>`.
31    pub name: String,
32    /// Whether the pid in the name is a running process on the host.
33    pub alive: bool,
34}
35
36/// The lock as the host reported it.
37#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize)]
38pub struct LockState {
39    /// Whether the lock directory exists.
40    pub present: bool,
41    /// The pid files in it, each with whether that process runs.
42    pub holders: Vec<(u32, bool)>,
43}
44
45impl LockState {
46    /// Held by somebody who may still be working: present, and either a holder is alive or
47    /// there is no holder file to check.
48    pub fn held(&self) -> bool {
49        self.present && (self.holders.is_empty() || self.holders.iter().any(|(_, a)| *a))
50    }
51
52    /// Present, with holder files, every one of them dead: safe to reclaim.
53    pub fn stale(&self) -> bool {
54        self.present && !self.holders.is_empty() && self.holders.iter().all(|(_, a)| !*a)
55    }
56}

58/// One look at a host: its free space, its lock and its queue.
59#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize)]
60pub struct Observation {
61    /// KiB free under the host's home directory, when `df` answered.
62    pub free_kib: Option<u64>,
63    /// The lock.
64    pub lock: LockState,
65    /// The tickets, oldest first (sorted by name, which is how v2 sorts them).
66    pub tickets: Vec<Ticket>,
67}
68
69impl Observation {
70    /// Parse what [`OBSERVE`] prints. A line this module does not write is an error, so a
71    /// login banner or a changed script is reported, never misread as an empty queue.
72    pub fn parse(text: &str) -> Result<Observation, String> {
73        let mut o = Observation::default();
74        for line in text.lines().map(str::trim).filter(|l| !l.is_empty()) {
75            let w: Vec<&str> = line.split_whitespace().collect();
76            match w.as_slice() {
77                ["free", n] => o.free_kib = n.parse().ok(),
78                ["lock"] => o.lock.present = true,
79                ["holder", pid, state] => {
80                    let pid = pid
81                        .parse()
82                        .map_err(|_| format!("holder pid {pid:?} is not a number"))?;
83                    o.lock.holders.push((pid, *state == "alive"));
84                }
85                ["ticket", name, state] => o.tickets.push(Ticket {
86                    name: name.to_string(),
87                    alive: *state == "alive",
88                }),
89                _ => {
90                    return Err(format!(
91                        "the host printed a line the queue does not write: {line:?}"
92                    ))
93                }
94            }
95        }
96        o.tickets.sort_by(|a, b| a.name.cmp(&b.name));
97        Ok(o)
98    }
99
100    /// How many derives are running or waiting: every live ticket, and the lock when held.
101    pub fn load(&self) -> usize {
102        self.tickets.iter().filter(|t| t.alive).count() + usize::from(self.lock.held())
103    }
104
105    /// GiB free, rounded down, as v2 computed it.
106    pub fn free_gib(&self) -> Option<u64> {
107        self.free_kib.map(|k| k / 1_048_576)
108    }
109}

111/// What a waiter does next.
112#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
113#[serde(tag = "step", rename_all = "snake_case")]
114pub enum Step {
115    /// Its ticket is the oldest live one and the lock is free: take it.
116    Take,
117    /// Every holder of the lock is dead: reclaim it, then look again.
118    Reclaim {
119        /// The dead holders' pids.
120        dead: Vec<u32>,
121    },
122    /// Wait: this many live tickets are ahead, and the lock is or is not held.
123    Wait {
124        /// Live tickets before this one.
125        ahead: usize,
126        /// Whether the lock is held.
127        locked: bool,
128    },
129    /// The waiter's own ticket is not on the host (removed as dead, or never written).
130    Lost,
131}
132
133/// The decision, and the dead tickets to remove first.
134#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
135pub struct Decision {
136    /// Tickets whose holder is dead, oldest first.
137    pub remove: Vec<String>,
138    /// What to do.
139    pub step: Step,
140}
141
142/// Decide one waiter's next step. Pure: the same observation always gives the same answer.
143pub fn decide(obs: &Observation, mine: &str) -> Decision {
144    let remove: Vec<String> = obs
145        .tickets
146        .iter()
147        .filter(|t| !t.alive && t.name != mine)
148        .map(|t| t.name.clone())
149        .collect();
150    let step = match obs.tickets.iter().find(|t| t.name == mine) {
151        None | Some(Ticket { alive: false, .. }) => Step::Lost,
152        Some(_) if obs.lock.stale() => Step::Reclaim {
153            dead: obs.lock.holders.iter().map(|(p, _)| *p).collect(),
154        },
155        Some(_) => {
156            let ahead = obs
157                .tickets
158                .iter()
159                .take_while(|t| t.name != mine)
160                .filter(|t| t.alive)
161                .count();
162            if ahead == 0 && !obs.lock.present {
163                Step::Take
164            } else {
165                Step::Wait {
166                    ahead,
167                    locked: obs.lock.present,
168                }
169            }
170        }
171    };
172    Decision { remove, step }
173}

175/// The name of a ticket: the arrival time in nanoseconds since the epoch, nineteen digits as
176/// v2's `date +%s%N` writes them, then the holder's pid and the derive's tag. The time is
177/// the waiter's own clock read once: waiters on one machine are ordered exactly, and the
178/// machines of the cluster keep their clocks within milliseconds of each other.
179pub fn ticket_name(arrival_nanos: u128, pid: u32, tag: &str) -> String {
180    format!("{arrival_nanos:019}-{pid}-{tag}")
181}

183/// Report the host: free KiB under `$HOME`, the lock and each holder file, each ticket.
184/// Arguments: the queue directory, the lock directory.
185pub const OBSERVE: &str = r#"q=$1; l=$2
186f=$(df -Pk "$HOME" 2>/dev/null | awk 'NR==2{print $4}')
187echo "free ${f:-unknown}"
188if [ -d "$l" ]; then
189  echo lock
190  for p in "$l"/*; do
191    [ -e "$p" ] || continue
192    n=${p##*/}
193    case $n in ''|*[!0-9]*) continue ;; esac
194    if kill -0 "$n" 2>/dev/null; then echo "holder $n alive"; else echo "holder $n dead"; fi
195  done
196fi
197if [ -d "$q" ]; then
198  for t in "$q"/*; do
199    [ -e "$t" ] || continue
200    n=${t##*/}
201    pid=$(printf '%s' "$n" | cut -d- -f2)
202    case $pid in ''|*[!0-9]*) echo "ticket $n dead"; continue ;; esac
203    if kill -0 "$pid" 2>/dev/null; then echo "ticket $n alive"; else echo "ticket $n dead"; fi
204  done
205fi"#;

207/// Write a ticket. Arguments: the queue directory, the ticket's name.
208const ENQUEUE: &str = r#"mkdir -p "$1" && : > "$1/$2""#;
209
210/// Remove tickets. Arguments: the queue directory, then the names.
211const REMOVE: &str = r#"d=$1; shift; for n in "$@"; do rm -f "$d/$n"; done"#;
212
213/// Take the lock, atomically, naming the holder. Exit 3 when somebody else has it.
214/// Arguments: the lock directory, the holder's pid.
215const TAKE: &str = r#"mkdir "$1" 2>/dev/null || exit 3; : > "$1/$2""#;
216
217/// Reclaim a lock whose holders were all dead: check each again, remove its file, then the
218/// directory, which fails — exit 3 — if anybody else's file appeared. Arguments: the lock
219/// directory, then the dead pids.
220const RECLAIM: &str = r#"l=$1; shift
221for p in "$@"; do kill -0 "$p" 2>/dev/null && exit 3; rm -f "$l/$p"; done
222rmdir "$l" 2>/dev/null || exit 3"#;
223
224/// Release the lock: remove this derive's pid files, then the directory when it is empty.
225/// Arguments: the lock directory, then the pids.
226const RELEASE: &str =
227    r#"l=$1; shift; for p in "$@"; do rm -f "$l/$p"; done; rmdir "$l" 2>/dev/null; exit 0"#;

229/// Why waiting ended without the lock.
230#[derive(Debug)]
231pub enum QueueError {
232    /// The host could not be asked, or answered in a way the queue does not read.
233    Host(String),
234    /// The waiter's ticket disappeared: its holder was taken for dead.
235    Lost,
236    /// The declared wait ran out; the ticket is withdrawn.
237    TimedOut(Duration),
238}
239
240impl std::fmt::Display for QueueError {
241    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
242        match self {
243            QueueError::Host(e) => write!(f, "{e}"),
244            QueueError::Lost => write!(
245                f,
246                "this waiter's ticket is gone from the host's queue: its holder was taken for dead"
247            ),
248            QueueError::TimedOut(d) => write!(f, "gave up after {} s in the queue", d.as_secs()),
249        }
250    }
251}

253/// The queue of one host, as the declaration places it.
254#[derive(Debug, Clone)]
255pub struct Queue<'a> {
256    /// The host.
257    pub remote: &'a Remote,
258    /// The ticket directory.
259    pub directory: &'a str,
260    /// The lock directory.
261    pub lock: &'a str,
262    /// Between looks.
263    pub poll: Duration,
264    /// In all.
265    pub wait_max: Duration,
266}
267
268fn host_err(what: &str, e: impl std::fmt::Display) -> QueueError {
269    QueueError::Host(format!("{what}: {e}"))
270}
271
272impl Queue<'_> {
273    /// Look at the host once.
274    pub fn observe(&self) -> Result<Observation, QueueError> {
275        let out = self
276            .remote
277            .run(OBSERVE, &[self.directory, self.lock], None)
278            .map_err(|e| host_err(&format!("{} could not be asked", self.remote.label), e))?;
279        if !out.ok() {
280            return Err(QueueError::Host(format!(
281                "{} did not answer the observation (exit {:?}): {}",
282                self.remote.label,
283                out.code,
284                out.stderr.trim()
285            )));
286        }
287        Observation::parse(&out.stdout).map_err(QueueError::Host)
288    }
289
290    /// Join the queue under the holder's pid; returns the ticket's name.
291    pub fn enqueue(&self, holder_pid: u32, tag: &str) -> Result<String, QueueError> {
292        let arrival = std::time::SystemTime::now()
293            .duration_since(std::time::UNIX_EPOCH)
294            .map(|d| d.as_nanos())
295            .unwrap_or(0);
296        let name = ticket_name(arrival, holder_pid, tag);
297        let out = self
298            .remote
299            .run(ENQUEUE, &[self.directory, &name], None)
300            .map_err(|e| host_err("the ticket could not be written", e))?;
301        if !out.ok() {
302            return Err(QueueError::Host(format!(
303                "the ticket could not be written: {}",
304                out.stderr.trim()
305            )));
306        }
307        Ok(name)
308    }
309
310    /// Withdraw a ticket; a ticket already gone is not an error.
311    pub fn withdraw(&self, ticket: &str) {
312        let _ = self.remote.run(REMOVE, &[self.directory, ticket], None);
313    }
314
315    /// Release the lock held under these pids.
316    pub fn release(&self, pids: &[u32]) {
317        let pids: Vec<String> = pids.iter().map(u32::to_string).collect();
318        let mut args = vec![self.lock];
319        args.extend(pids.iter().map(String::as_str));
320        let _ = self.remote.run(RELEASE, &args, None);
321    }
322
323    /// Wait in the queue until this ticket takes the lock under `holder_pid`. Every change
324    /// of position, every removed ticket and every reclaim is written to `log`. On success
325    /// the ticket is withdrawn and the lock is held; on failure the ticket is withdrawn.
326    pub fn wait(
327        &self,
328        ticket: &str,
329        holder_pid: u32,
330        log: &mut dyn Write,
331    ) -> Result<(), QueueError> {
332        let started = Instant::now();
333        let mut last: Option<Step> = None;
334        let result = loop {
335            let obs = match self.observe() {
336                Ok(o) => o,
337                Err(e) => break Err(e),
338            };
339            let d = decide(&obs, ticket);
340            if !d.remove.is_empty() {
341                let _ = writeln!(
342                    log,
343                    "derive-offload: removed {} dead waiter ticket(s) on {}: {}",
344                    d.remove.len(),
345                    self.remote.label,
346                    d.remove.join(", ")
347                );
348                let mut args = vec![self.directory];
349                args.extend(d.remove.iter().map(String::as_str));
350                let _ = self.remote.run(REMOVE, &args, None);
351            }
352            match &d.step {
353                Step::Lost => break Err(QueueError::Lost),
354                Step::Take => {
355                    let pid = holder_pid.to_string();
356                    match self.remote.run(TAKE, &[self.lock, &pid], None) {
357                        Ok(o) if o.ok() => break Ok(()),
358                        Ok(_) => {} // somebody outside the queue took it first: look again
359                        Err(e) => break Err(host_err("the lock could not be taken", e)),
360                    }
361                }
362                Step::Reclaim { dead } => {
363                    let pids: Vec<String> = dead.iter().map(u32::to_string).collect();
364                    let mut args = vec![self.lock];
365                    args.extend(pids.iter().map(String::as_str));
366                    if let Ok(o) = self.remote.run(RECLAIM, &args, None) {
367                        if o.ok() {
368                            let _ = writeln!(
369                                log,
370                                "derive-offload: reclaimed the lock on {} from dead holder(s) {}",
371                                self.remote.label,
372                                pids.join(", ")
373                            );
374                            continue;
375                        }
376                    }
377                }
378                Step::Wait { ahead, locked } => {
379                    if last.as_ref() != Some(&d.step) {
380                        let _ = writeln!(
381                            log,
382                            "derive-offload: waiting on {}: {ahead} ahead, lock {}",
383                            self.remote.label,
384                            if *locked { "held" } else { "free" }
385                        );
386                    }
387                }
388            }
389            last = Some(d.step);
390            if started.elapsed() >= self.wait_max {
391                break Err(QueueError::TimedOut(started.elapsed()));
392            }
393            std::thread::sleep(self.poll);
394        };
395        self.withdraw(ticket);
396        result
397    }
398}

429    /// Two waiters on one host take the lock in the order they arrived, although the later one
430    /// started waiting first: with the old `until mkdir` loop it would simply have won.
431    #[test]
432    fn two_waiters_are_served_in_arrival_order() {
433        let tmp = tempfile::tempdir().unwrap();
434        let (dir, lock) = (tmp.path().join("queue"), tmp.path().join("lock"));
435        let r = Remote::local();
436        let q = on_disk(&r, &dir, &lock);
437        let first = Holder::start(&r).unwrap();
438        let first_ticket = q.enqueue(first.pid, "aaaaaaaaaa").unwrap();
439        let second = Holder::start(&r).unwrap();
440        let second_ticket = q.enqueue(second.pid, "bbbbbbbbbb").unwrap();
441        assert!(
442            first_ticket < second_ticket,
443            "{first_ticket} does not sort before {second_ticket}"
444        );
445
446        let order = Arc::new(Mutex::new(Vec::new()));
447        std::thread::scope(|s| {
448            let late_but_eager = s.spawn(|| {
449                let q = on_disk(&r, &dir, &lock);
450                q.wait(&second_ticket, second.pid, &mut std::io::sink())
451                    .unwrap();
452                order.lock().unwrap().push("second");
453                q.release(&[second.pid]);
454            });
455            std::thread::sleep(Duration::from_millis(300));
456            assert!(
457                order.lock().unwrap().is_empty(),
458                "the second waiter took the lock ahead of the first"
459            );
460            let q = on_disk(&r, &dir, &lock);
461            q.wait(&first_ticket, first.pid, &mut std::io::sink())
462                .unwrap();
463            order.lock().unwrap().push("first");
464            std::thread::sleep(Duration::from_millis(200));
465            q.release(&[first.pid]);
466            late_but_eager.join().unwrap();
467        });
468        assert_eq!(*order.lock().unwrap(), ["first", "second"]);
469        assert!(!lock.exists(), "the lock outlived its last holder");
470        assert_eq!(
471            std::fs::read_dir(&dir).unwrap().count(),
472            0,
473            "a ticket outlived its waiter"
474        );
475    }

680    /// A stand-in for ssh that runs the remote command here, after `before` — shell text that
681    /// sees the command as `$1` and may fail it, or remove the stand-in itself so that the
682    /// next command cannot even be started.
683    fn stand_in(dir: &Path, before: &str) -> std::path::PathBuf {
684        use std::os::unix::fs::PermissionsExt;
685        let p = dir.join("ssh");
686        std::fs::write(&p, format!("#!/bin/sh\n{before}\nexec sh -c \"$1\"\n")).unwrap();
687        std::fs::set_permissions(&p, std::fs::Permissions::from_mode(0o755)).unwrap();
688        p
689    }

692    /// error of the queue, never an empty queue or a ticket that was not written.
693    #[test]
694    fn a_host_that_cannot_be_asked_is_a_queue_error() {
695        let tmp = tempfile::tempdir().unwrap();
696        let (dir, lock) = (tmp.path().join("queue"), tmp.path().join("lock"));
697
698        let gone = Remote::through(&tmp.path().join("no-such-ssh"));
699        let q = on_disk(&gone, &dir, &lock);

702            other => panic!("{other:?}"),

748    /// Somebody outside the queue takes the lock between the look and the take, or a stale
749    /// lock cannot be reclaimed this time: the waiter looks again rather than giving up.
750    #[test]
751    fn a_take_or_reclaim_that_fails_once_is_tried_again() {
752        let tmp = tempfile::tempdir().unwrap();
753        let (dir, lock) = (tmp.path().join("queue"), tmp.path().join("lock"));
754        let marks = tmp.path().join("marks");
755        std::fs::create_dir(&marks).unwrap();
756        let ssh = stand_in(
757            tmp.path(),
758            &format!(
759                r#"case "$1" in
760  *'|| exit 3; :'*) [ -e {m}/take ] || {{ : > {m}/take; exit 3; }} ;;
761  *'&& exit 3'*) [ -e {m}/reclaim ] || {{ : > {m}/reclaim; exit 3; }} ;;
762esac"#,
763                m = marks.display()
764            ),
765        );
766        let r = Remote::through(&ssh);
767        let q = on_disk(&r, &dir, &lock);
768
769        let me = Holder::start(&Remote::local()).unwrap();
770        let ticket = q.enqueue(me.pid, "tttttttttt").unwrap();
771        q.wait(&ticket, me.pid, &mut std::io::sink()).unwrap();
772        assert!(marks.join("take").exists(), "the take never failed");
773        assert!(lock.join(me.pid.to_string()).exists());
774        q.release(&[me.pid]);
775
776        let gone = Holder::start(&Remote::local()).unwrap();
777        let gone_pid = gone.pid;
778        drop(gone);
779        std::fs::create_dir(&lock).unwrap();
780        std::fs::write(lock.join(gone_pid.to_string()), "").unwrap();
781        let ticket = q.enqueue(me.pid, "rrrrrrrrrr").unwrap();
782        let mut log = Vec::new();
783        q.wait(&ticket, me.pid, &mut log).unwrap();
784        assert!(marks.join("reclaim").exists(), "the reclaim never failed");
785        let log = String::from_utf8(log).unwrap();
786        assert!(
787            log.contains(&format!(
788                "reclaimed the lock on stand-in from dead holder(s) {gone_pid}"
789            )),
790            "{log}"
791        );
792        q.release(&[me.pid]);
793        assert!(!lock.exists());
794    }
