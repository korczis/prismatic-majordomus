1//! The `evidence` module: from a claim to the run that proves it, and back.
2//!
3//! `docs/CLAIMS.yaml` binds a claim to a test by path. [`crate::evidence`] binds that path
4//! to the runs actually recorded against it. These capabilities are the one declaration of
5//! that join, and every surface — the command line, the HTTP API and its OpenAPI operation,
6//! the MCP tool a client reads — is derived from it. There is no second evidence model
7//! behind any of them.
8//!
9//! # Both directions
10//!
11//! A relation that can only be walked one way is half a relation. `evidence.claim` answers
12//! "what proves this claim"; `evidence.test` answers "what does this test prove", listing
13//! every claim that names it. They read the same derivation, so the two answers cannot
14//! disagree.
15//!
16//! # Why the recorder is here and not exposed over the network
17//!
18//! `evidence.record` writes a tracked file, and this executable's MCP and HTTP surfaces are
19//! read-only — that is a property of the server, not an accident of what has been built.
20//! So it declares a command line and nothing else, which
21//! [`Visibility::Developer`](crate::capability::Visibility) is exactly the word for: offered
22//! to whoever runs the executable and to nobody over a network. Declaring it as a capability
23//! rather than as a hand-written command is what keeps it inside the registry, the command
24//! graph and the generated reference instead of in the inventory of commands nothing derives.
25//!
26//! ```
27//! use majordomus_cli::capability::builtin::evidence;
28//!
29//! let m = evidence::module();
30//! assert_eq!(m.id.as_str(), "evidence");
31//! let ids: Vec<&str> = m
32//!     .capabilities
33//!     .iter()
34//!     .map(|e| e.capability.id.as_str())
35//!     .collect();
36//! assert_eq!(
37//!     ids,
38//!     ["evidence.report", "evidence.claim", "evidence.test", "evidence.record"]
39//! );
40//!
41//! // the recorder writes a tracked file, so it is offered to whoever runs the executable
42//! // and to nobody over a network; that is what keeps the server read-only
43//! let record = m
44//!     .capabilities
45//!     .iter()
46//!     .find(|e| e.capability.id.as_str() == "evidence.record")
47//!     .unwrap();
48//! assert!(record.capability.exposure.mcp.is_none());
49//! assert!(record.capability.exposure.http.is_none());
50//! assert_eq!(
51//!     record.capability.exposure.cli.as_ref().unwrap().path,
52//!     ["evidence".to_string(), "record".to_string()]
53//! );
54//! ```
55
56//! # Example
57//!
58//! Four capabilities, and every projection — the command line, the HTTP operation, the MCP
59//! tool — is derived from that one declaration.
60//!
61//! ```
62//! use majordomus_cli::capability::builtin::evidence::module;
63//! let m = module();
64//! assert_eq!(m.id.as_str(), "evidence");
65//! assert_eq!(m.capabilities.len(), 4);
66//! ```

93/// Which part of the matrix to answer for, and at which revision.
94///
95/// Every filter narrows, and they compose; saying nothing asks for the whole matrix at the
96/// working tree. A key nothing recognises is refused rather than dropped, because a
97/// misspelled filter that read as "no filter" would answer a question nobody asked and look
98/// like a clean one.
99///
100/// ```
101/// use majordomus_cli::capability::builtin::evidence::EvidenceReportInput;
102/// use majordomus_cli::evidence::{ProofState, TreeState};
103///
104/// let all: EvidenceReportInput = serde_json::from_str("{}").unwrap();
105/// assert!(all.state.is_none() && all.status.is_none() && !all.findings_only);
106/// assert!(all.presented.is_none(), "the working tree unless a commit is named");
107///
108/// let narrowed: EvidenceReportInput =
109///     serde_json::from_str(r#"{"state": "stale", "findings_only": true}"#).unwrap();
110/// assert_eq!(narrowed.state, Some(ProofState::Stale));
111/// assert!(narrowed.findings_only);
112///
113/// let at: EvidenceReportInput =
114///     serde_json::from_str(r#"{"presented": "HEAD", "presented_tree": "dirty"}"#).unwrap();
115/// assert_eq!(at.presented.as_deref(), Some("HEAD"));
116/// assert_eq!(at.presented_tree, Some(TreeState::Dirty));
117///
118/// assert!(serde_json::from_str::<EvidenceReportInput>(r#"{"states": "stale"}"#).is_err());
119/// ```
120pub struct EvidenceReportInput {
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    /// Only claims in this proof state (`proven`, `inputs_unchanged`, `stale`, `failing`,
123    /// `not_run`, `unrunnable`, `no_test`). Absent: every claim.
124    pub state: Option<ProofState>,
125    #[serde(default, skip_serializing_if = "Option::is_none")]
126    /// Only claims declaring this status (`guaranteed`, `advisory`, `planned`, `rejected`).
127    /// Absent: every claim.
128    pub status: Option<String>,
129    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
130    /// Only the claims whose declared status the evidence does not support, with the
131    /// findings. The tallies still count the whole matrix, so a filtered answer never
132    /// misreports how much of it was examined.
133    pub findings_only: bool,
134    #[serde(default, skip_serializing_if = "Option::is_none")]
135    /// Judge at this revision, as committed, rather than at the working tree: `HEAD`, or
136    /// any name of the checked-out commit. It must be the checked-out commit; the ledger is
137    /// read as that commit holds it, and a run the working ledger holds that the commit's
138    /// does not can only withhold `proven`. Absent: the working tree.
139    pub presented: Option<String>,
140    #[serde(default, skip_serializing_if = "Option::is_none")]
141    /// What the caller knows about the presented commit's tree (`clean`, `dirty`,
142    /// `unknown`). It can only weaken the measured state, never assert a clean tree over a
143    /// dirty checkout, and it means nothing without `presented`.
144    pub presented_tree: Option<TreeState>,
145}

206/// One test, by its stable identity or by the path a claim names it with.
207///
208/// Both spellings reach the same test. `docs/CLAIMS.yaml` names a path, the ledger keys on
209/// an identity, and a caller holding one of them should not have to know the other; the
210/// handler resolves a path to the identity and reads the same derivation either way.
211///
212/// ```
213/// use majordomus_cli::capability::builtin::evidence::EvidenceTestInput;
214/// use majordomus_cli::evidence::{Runner, TestId};
215///
216/// let by_path: EvidenceTestInput =
217///     serde_json::from_str(r#"{"test": "test/cases/84_distribution_model.sh"}"#).unwrap();
218/// let id = TestId::of(&by_path.test).unwrap();
219/// assert_eq!(id.as_string(), "suite:84_distribution_model");
220/// assert_eq!(id.runner, Runner::Suite);
221///
222/// // an identity is not a path: `TestId::of` reads paths, and the handler parses the
223/// // `<runner>:<name>` form itself
224/// let by_id: EvidenceTestInput = serde_json::from_str(r#"{"test": "crate:why"}"#).unwrap();
225/// assert!(TestId::of(&by_id.test).is_none());
226/// ```
227pub struct EvidenceTestInput {
228    /// `suite:84_distribution_model`, `crate:why`, or the path itself
229    /// (`test/cases/84_distribution_model.sh`), which is resolved to the same identity.
230    pub test: String,
231}
232
233impl BenchmarkCases for EvidenceTestInput {
234    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
235        let test = ctx
236            .index
237            .objects
238            .iter()
239            .filter(|o| o.kind == "claim")
240            .find_map(|o| {
241                o.metadata
242                    .get("test")
243                    .and_then(|v| v.as_str())
244                    .and_then(evidence::TestId::of)
245            })
246            .map(|t| t.as_string())
247            .unwrap_or_else(|| "suite:00_yaml_flatten".to_string());
248        vec![NamedCase::new("first-test", EvidenceTestInput { test })]
249    }

254/// A run to record into the ledger.
255///
256/// Paths to what the runners already wrote, not a run to perform: the recorder reads a
257/// report, stamps each result with the provenance the run did not carry, and merges it.

260/// ```
261/// use majordomus_cli::capability::builtin::evidence::EvidenceRecordInput;

264///     serde_json::from_str(r#"{"suite": "tmp/report.tsv"}"#).unwrap();

266/// assert!(input.crate_output.is_none() && input.origin.is_none());
267///
268/// // what was not given is not written back as null: the shape a client reads is the
269/// // shape it sent
270/// assert_eq!(
271///     serde_json::to_string(&input).unwrap(),
272///     r#"{"suite":"tmp/report.tsv"}"#
273/// );
274/// ```
275pub struct EvidenceRecordInput {
276    #[serde(default, skip_serializing_if = "Option::is_none")]
277    /// The runner's TSV report: `MJ_TEST_REPORT=<file> bash test/run.sh`.
278    pub suite: Option<String>,
279    #[serde(default, skip_serializing_if = "Option::is_none")]
280    /// A file holding `cargo test`'s output, for the crate's own integration tests.
281    pub crate_output: Option<String>,
282    #[serde(default, skip_serializing_if = "Option::is_none")]
283    /// Where the run happened: `local` (the default), `ci` or `release`.
284    pub origin: Option<String>,
285}

297#[derive(Debug, Clone, Serialize, JsonSchema)]
298/// One claim with everything the repository can say about its proof, and the way back out
299/// to the test that carries it.
300///
301/// The proof is flattened in, so a client reads one claim rather than an evidence wrapper
302/// around one; the executions and the other claims the same test proves sit beside it.
303///
304/// ```
305/// use majordomus_cli::capability::builtin::evidence::ClaimEvidence;
306/// use majordomus_cli::evidence::ClaimProof;
307///
308/// let proof: ClaimProof = serde_json::from_value(serde_json::json!({
309///     "id": "evidence-both-directions",
310///     "claim": "a claim and its test answer for each other",
311///     "status": "guaranteed",
312///     "source": null,
313///     "implementation": null,
314///     "test_path": "test/cases/84_distribution_model.sh",
315///     "test": "suite:84_distribution_model",
316///     "state": "not_run",
317///     "meaning": "no run of this test has been recorded",
318///     "execution": null,
319///     "changed": [],
320///     "reproduce": "bash test/run.sh 84_distribution_model"
321/// }))
322/// .unwrap();
323///
324/// let answer = ClaimEvidence {
325///     proof,
326///     executions: Vec::new(),
327///     also_proves: vec!["evidence-record-is-not-networked".to_string()],
328/// };
329/// let json = serde_json::to_value(&answer).unwrap();
330/// assert_eq!(json["id"], "evidence-both-directions");
331/// assert!(json.get("proof").is_none(), "the claim is flattened in, not nested");
332/// assert_eq!(json["also_proves"], serde_json::json!(["evidence-record-is-not-networked"]));
333/// // a list even where the ledger keeps one: retention is a decision that may change,
334/// // and a client that read a bare object would break when it did
335/// assert_eq!(json["executions"], serde_json::json!([]));
336/// ```
337pub struct ClaimEvidence {
338    /// The claim, joined to its evidence.
339    #[serde(flatten)]
340    pub proof: ClaimProof,
341    /// Every execution the ledger holds for this claim's test — today one, the latest, and
342    /// a list because the ledger's retention is a decision that may change and a client
343    /// that read a bare object would break when it did.
344    pub executions: Vec<Execution>,
345    /// The other claims the same test proves. A test shared by twelve claims is a test
346    /// whose failure is twelve findings, and a reader of one of them should be able to see
347    /// the other eleven without searching.
348    pub also_proves: Vec<String>,
349}

352/// One test, and what it proves.
353///
354/// The other direction of [`ClaimEvidence`], read from the test's end: its identity, its
355/// source and whether that source is still the one that ran, the command that runs it
356/// again, and every claim that names it.
357///
358/// ```
359/// use majordomus_cli::capability::builtin::evidence::TestEvidence;
360/// use majordomus_cli::evidence::TestId;
361///
362/// let id = TestId::of("apps/majordomus-cli/tests/why.rs").unwrap();
363/// let answer = TestEvidence {
364///     test: id.as_string(),
365///     runner: id.runner,
366///     source: id.source(),
367///     present: true,
368///     reproduce: id.reproduce(),
369///     execution: None,
370///     digest_matches: None,
371///     proves: Vec::new(),
372/// };
373/// assert_eq!(answer.test, "crate:why");
374/// assert_eq!(answer.reproduce, "cargo test --test why");
375///
376/// // nothing recorded: the two provenance keys are absent rather than null, so "no
377/// // execution" cannot be read as an execution that said nothing
378/// let json = serde_json::to_value(&answer).unwrap();
379/// assert!(json.get("execution").is_none() && json.get("digest_matches").is_none());
380/// assert_eq!(json["proves"], serde_json::json!([]));
381/// ```
382pub struct TestEvidence {
383    /// The stable identity: `suite:<case>` or `crate:<binary>`.
384    pub test: String,
385    /// Which runner owns it.
386    pub runner: evidence::Runner,
387    /// Its own source, repository-relative.
388    pub source: String,
389    /// Whether that source is in this checkout.
390    pub present: bool,
391    /// The command that runs this one test.
392    pub reproduce: String,
393    /// The latest recorded execution, when there is one.
394    #[serde(skip_serializing_if = "Option::is_none")]
395    pub execution: Option<Execution>,
396    /// Whether the test's source still hashes to what the execution recorded. `None` when
397    /// there is no execution, or the source is not there to hash.
398    #[serde(skip_serializing_if = "Option::is_none")]
399    pub digest_matches: Option<bool>,
400    /// Every claim that names this test, with the state each is in.
401    pub proves: Vec<ClaimProof>,
402}

404#[derive(Debug, Clone, Serialize, JsonSchema)]
405/// What a recording did: how much of the run reached the ledger, and under what provenance.
406///
407/// The recorder decides nothing, and this report says so. A result no runner in this
408/// repository owns is named in `unknown` rather than written under a guessed identity, and
409/// a failure is recorded like a pass — the ledger holds what happened, so `passed` is a
410/// count and never a filter.
411///
412/// ```
413/// use majordomus_cli::capability::builtin::evidence::RecordReport;
414///
415/// let report = RecordReport {
416///     recorded: 3,
417///     passed: 2,
418///     commit: "06fa258913a1b2c3d4e5f60718293a4b5c6d7e8f".to_string(),
419///     working_tree: "clean".to_string(),
420///     unknown: vec!["docs/CLAIMS.yaml".to_string()],
421///     ledger: majordomus_cli::evidence::LEDGER_PATH.to_string(),
422/// };
423/// // the failing test is one of the three recorded, not a fourth thing that was dropped
424/// assert_eq!(report.recorded - report.passed, 1);
425/// assert!(report.ledger.starts_with(".ai/repo/"));
426///
427/// let json = serde_json::to_value(&report).unwrap();
428/// assert_eq!(json["unknown"].as_array().unwrap().len(), 1);
429/// assert_eq!(json["working_tree"], "clean");
430/// ```
431pub struct RecordReport {
432    /// How many executions were written.
433    pub recorded: usize,
434    /// How many of them passed.
435    pub passed: usize,
436    /// The commit they were recorded against.
437    pub commit: String,
438    /// The tree's state at the time.
439    pub working_tree: String,
440    /// Results the run named that no runner in this repository owns.
441    pub unknown: Vec<String>,
442    /// Where the ledger was written.
443    pub ledger: String,
444}

456/// The whole joined report, read fresh: the ledger is a file that changes outside this
457/// process, and a cached answer would be the stale evidence this module exists to name.
458fn full_report(ctx: &Context) -> Result<EvidenceReport, CapabilityError> {
459    let root = root_of(ctx);
460    let ledger = Ledger::load(&root).map_err(internal)?;
461    Ok(evidence::report(&ctx.index, &ledger))
462}

464/// The joined report at the checked-out commit, as committed: the ledger that commit holds,
465/// and the runs the working ledger holds beside it, which can only withhold `proven`.
466fn report_at_commit(
467    ctx: &Context,
468    rev: &str,
469    given: Option<TreeState>,
470) -> Result<EvidenceReport, CapabilityError> {
471    let root = root_of(ctx);
472    let presented =
473        freshness::presented_commit(&root, rev, given).map_err(CapabilityError::InvalidInput)?;
474    let commit = presented.commit().unwrap_or_default().to_string();
475    let ledger = freshness::ledger_at(&root, &commit).map_err(internal)?;
476    let held = freshness::uncommitted(&root, &ledger).map_err(internal)?;
477    Ok(evidence::report_at(&ctx.index, &ledger, &presented, &held))
478}
479
480fn report(ctx: &Context, input: EvidenceReportInput) -> Result<EvidenceReport, CapabilityError> {
481    let mut r = match (&input.presented, input.presented_tree) {
482        (Some(rev), given) => report_at_commit(ctx, rev, given)?,
483        (None, Some(_)) => {
484            return Err(CapabilityError::InvalidInput(
485                "`presented_tree` weakens the measured tree of a presented commit, and no \
486                 commit was presented: name one with `presented`"
487                    .to_string(),
488            ))
489        }
490        (None, None) => full_report(ctx)?,
491    };
492    // the tallies are taken before the filter: a filtered answer says how much of the
493    // matrix it looked at, never how much it returned
494    if input.findings_only {
495        let named: std::collections::BTreeSet<&str> =
496            r.findings.iter().map(|f| f.claim.as_str()).collect();
497        r.claims.retain(|c| named.contains(c.id.as_str()));
498    }
499    if let Some(state) = input.state {
500        r.claims.retain(|c| c.state == state);
501    }
502    if let Some(status) = &input.status {
503        r.claims.retain(|c| &c.status == status);
504    }
505    Ok(r)
506}

508fn claim(ctx: &Context, input: EvidenceClaimInput) -> Result<ClaimEvidence, CapabilityError> {
509    let r = full_report(ctx)?;
510    let Some(proof) = r.claims.iter().find(|c| c.id == input.claim).cloned() else {
511        return Err(CapabilityError::NotFound(format!(
512            "`{}` is not a claim of docs/CLAIMS.yaml",
513            input.claim
514        )));
515    };
516    let also_proves = proof
517        .test
518        .as_ref()
519        .map(|t| {
520            r.claims
521                .iter()
522                .filter(|c| c.test.as_ref() == Some(t) && c.id != proof.id)
523                .map(|c| c.id.clone())
524                .collect()
525        })
526        .unwrap_or_default();
527    let executions = proof.execution.clone().into_iter().collect();
528    Ok(ClaimEvidence {
529        proof,
530        executions,
531        also_proves,
532    })
533}

535fn test(ctx: &Context, input: EvidenceTestInput) -> Result<TestEvidence, CapabilityError> {
536    // a path and an identity both name the same test; a client that has one should not
537    // have to know the other's spelling
538    let id = evidence::TestId::of(&input.test)
539        .or_else(|| {
540            let (prefix, name) = input.test.split_once(':')?;
541            let runner = match prefix {
542                "suite" => evidence::Runner::Suite,
543                "crate" => evidence::Runner::Crate,
544                _ => return None,
545            };
546            // the same rule the path form applies: one grammar, not two that disagree
547            evidence::TestId::named(runner, name)
548        })
549        .ok_or_else(|| {
550            CapabilityError::InvalidInput(format!(
551                "`{}` names no test: give `suite:<case>`, `crate:<binary>`, or the path of a \
552                 case under test/cases/ or an integration test under apps/majordomus-cli/tests/",
553                input.test
554            ))
555        })?;
556
557    let root = root_of(ctx);
558    let r = full_report(ctx)?;
559    let key = id.as_string();
560    let proves: Vec<ClaimProof> = r
561        .claims
562        .iter()
563        .filter(|c| c.test.as_deref() == Some(key.as_str()))
564        .cloned()
565        .collect();
566    let ledger = Ledger::load(&root).map_err(internal)?;
567    let execution = ledger.latest(&key).cloned();
568    let digest_matches = execution.as_ref().and_then(|e| e.digest_matches(&root));
569    Ok(TestEvidence {
570        test: key,
571        runner: id.runner,
572        source: id.source(),
573        present: root.join(id.source()).exists(),
574        reproduce: id.reproduce(),
575        execution,
576        digest_matches,
577        proves,
578    })
579}

581fn record(ctx: &Context, input: EvidenceRecordInput) -> Result<RecordReport, CapabilityError> {
582    let root = root_of(ctx);
583    let origin = match input.origin.as_deref() {
584        None => Origin::Local,
585        Some(w) => Origin::parse(w).ok_or_else(|| {
586            CapabilityError::InvalidInput(format!(
587                "`{w}` is not an origin: it is `local`, `ci` or `release`"
588            ))
589        })?,
590    };
591    // A CI run names itself from its environment; a local run has no run to name, and a
592    // local recording inside a CI job's shell is still not that job's evidence.
593    let run = match origin {
594        Origin::Ci => evidence::RunRef::from_env(|k| std::env::var(k).ok()),
595        Origin::Local | Origin::Release => None,
596    };
597    let req = RecordRequest {
598        suite: input.suite.map(std::path::PathBuf::from),
599        crate_output: input.crate_output.map(std::path::PathBuf::from),
600        origin,
601        run,
602    };
603    let got = evidence::record(&root, &req).map_err(|e| match e {
604        crate::error::Error::InvalidSurface { reason, .. } => CapabilityError::Refused(reason),
605        other => internal(other),
606    })?;
607    Ok(RecordReport {
608        recorded: got.recorded,
609        passed: got.passed,
610        commit: got.commit,
611        working_tree: got.working_tree,
612        unknown: got.unknown,
613        ledger: evidence::LEDGER_PATH.to_string(),
614    })
615}

882    /// Each ledger a report reads — the one a presented commit holds, the working copy
883    /// beside it, and the working tree's own — is refused when it cannot be read, rather
884    /// than read as a ledger that recorded nothing.
885    #[test]
886    fn an_unreadable_ledger_is_refused_at_every_revision() {
887        use crate::evidence::LEDGER_PATH;
888        let repo = crate::synthetic::SyntheticRepository::small().unwrap();
889        let git = |args: &[&str]| {
890            let out = std::process::Command::new("git")
891                .arg("-C")
892                .arg(repo.root())
893                .args(args)
894                .output()
895                .unwrap();
896            String::from_utf8_lossy(&out.stdout).trim().to_string()
897        };
898        git(&["init", "-q"]);
899        git(&["config", "user.email", "t@example.com"]);
900        git(&["config", "user.name", "t"]);
901        git(&["add", "-A"]);
902        git(&["commit", "-q", "-m", "fixture"]);
903        let ctx = repo.context().unwrap();
904        let at_head = serde_json::json!({ "presented": "HEAD" });
905
906        let v = ctx.execute("evidence.report", at_head.clone()).unwrap();
907        assert_eq!(v["presented"]["revision"], git(&["rev-parse", "HEAD"]));
908        assert_eq!(v["presented"]["tree"], "clean");
909
910        let refused = |input: serde_json::Value| match ctx.execute("evidence.report", input) {
911            Err(CapabilityError::Internal(m)) => m,
912            other => panic!("an unreadable ledger was judged past: {other:?}"),
913        };
914        // the working copy cannot be read: a run it may hold cannot be ruled out
915        std::fs::create_dir_all(repo.root().join(".ai/repo/evidence")).unwrap();
916        std::fs::write(repo.root().join(LEDGER_PATH), "not a ledger").unwrap();
917        assert!(refused(at_head.clone()).contains("cannot be ruled out"));
918        assert!(refused(serde_json::json!({})).contains(LEDGER_PATH));
919        // and the ledger the commit holds cannot be read either
920        git(&["add", "-A"]);
921        git(&["commit", "-q", "-m", "an unreadable ledger"]);
922        assert!(refused(at_head).contains(LEDGER_PATH));
923    }
