1//! `majordomus ui`: the primary surface, a product of its own, server-rendered (ADR 0135).
2//!
3//! Two surfaces of this process are for a person. The UI (`/ui`) is the product: the
4//! delivery in the project's own words — how the work stands, what is planned and blocked,
5//! what is in the queue, which release is live, who is on the mesh, what holds as evidence
6//! and which rules are proven. The Cockpit (`/cockpit`) is the technical layer beneath it,
7//! the registry rendered, and the UI refers to it only as the low-level reference of a value
8//! ("technical view" on a source line).
9//!
10//! It is a projection, never an application. Every screen composes capabilities through the
11//! executor every transport calls, and every value carries the capability and pointer it
12//! was read from (`source`): inferred, derived and projected, never kept. It holds no data,
13//! no status vocabulary and no route list beyond `SCREENS` and `PANELS`, which a test
14//! holds against the dispatcher. It shares mechanics rather than copying them:
15//! - the element builder and the asset server, pointed at `share/ui/`;
16//! - the content-security policy, with its one hash-pinned inline script;
17//! - the design tokens, the kit and the icon set every surface draws from.
18//!
19//! ```mermaid
20//! flowchart LR
21//!   declaration["capability!"] --> registry["registry"] --> executor["executor"]
22//!   executor --> api["MCP, HTTP, command line"]
23//!   executor --> ui["UI · /ui<br/>(the product)"]
24//!   executor --> cockpit["Cockpit · /cockpit<br/>(its technical reference)"]
25//! ```
26//!
27//! With no JavaScript, every screen is complete HTML and the menu still opens. The one
28//! script adds two things: it pulls in the panels whose sources are slow, which are
29//! otherwise reached by their own links, and it reveals the theme toggle.
30//!
31//! The UI is mounted on its own prefix, named in the web topology and among the local
32//! services by one id, and is a surface apart from the Cockpit:
33//!
34//! ```
35//! use majordomus_cli::ui::{PREFIX, SURFACE};
36//! assert_eq!(PREFIX, "/ui");
37//! assert_eq!(SURFACE, "ui");
38//! assert_ne!(PREFIX, majordomus_cli::cockpit::PREFIX, "the Cockpit is another surface");
39//! ```

267/// The UI over the repository as it is now.
268pub(crate) struct Ui {
269    live: Arc<Live>,
270    version: &'static str,
271    assets: Assets,
272    /// The Cockpit's files, for the modules the UI borrows from it ([`MOVES_LIB`],
273    /// [`SOCKET_LIB`]).
274    cockpit_assets: Assets,
275    /// Whether this UI's screens have been read once behind its first reader.
276    warmed: std::sync::atomic::AtomicBool,
277}

279impl Ui {
280    /// A UI over a view of the repository, serving its assets from `share_dir/ui`.
281    pub(crate) fn new(
282        live: impl IntoLive,
283        version: &'static str,
284        share_dir: Option<&std::path::Path>,
285    ) -> Self {
286        let assets = match share_dir {
287            Some(dir) => Assets::under(dir, ASSETS_DIR, ASSETS_PREFIX),
288            None => Assets::none_under(ASSETS_DIR, ASSETS_PREFIX),
289        };
290        let cockpit_assets = match share_dir {
291            Some(dir) => Assets::new(dir),
292            None => Assets::none_under(crate::cockpit::assets::DIR, crate::cockpit::assets::PREFIX),
293        };
294        if !assets.present() {
295            tracing::warn!(
296                "the distribution has no share/ui/ directory: the UI will serve unstyled markup; run scripts/cockpit-assets to build it"
297            );
298        }
299        Ui {
300            live: live.into_live(),
301            version,
302            assets,
303            cockpit_assets,
304            warmed: std::sync::atomic::AtomicBool::new(false),
305        }
306    }
307
308    /// Read every screen and every panel once, behind the first reader of this UI, so that
309    /// the screen they open next has its sources read before they ask for it.
310    ///
311    /// The routes are [`SCREENS`] and [`PANELS`], the lists the navigation and the
312    /// dispatcher are held to, so a screen added tomorrow is warmed with no edit here. What
313    /// each request renders is discarded: its effect is that the sources it names are now
314    /// remembered and kept current (`crate::capability::readings`). A process that runs
315    /// every capability without its cache does not warm, and neither do the crate's own
316    /// unit tests, whose fixtures are gone before a background reader would finish.
317    pub(crate) fn warm_once(ui: &Arc<Ui>) {
318        use std::sync::atomic::Ordering;
319        if cfg!(test) || crate::capability::executor::bypassed() {
320            return;
321        }
322        if ui.warmed.swap(true, Ordering::SeqCst) {
323            return;
324        }
325        let ui = Arc::clone(ui);
326        let warm = move || {
327            let routes = SCREENS
328                .iter()
329                .map(|s| s.route)
330                .chain(PANELS.iter().copied());
331            for route in routes {
332                let _ = ui.handle(&Request::parse_target("GET", route, Vec::new()));
333            }
334        };
335        if let Err(refused) = std::thread::Builder::new()
336            .name("mj-ui-warm".into())
337            .spawn(warm)
338        {
339            tracing::warn!("no thread to warm the UI's screens: {refused}");
340        }
341    }

343    /// Does this request belong to the UI? The prefix itself and every path below it, and
344    /// nothing that merely starts with the same letters.
345    #[cfg(test)]
346    pub(crate) fn owns(path: &str) -> bool {
347        path == PREFIX || path.starts_with(&format!("{PREFIX}/"))
348    }

350    /// Answer one request under [`PREFIX`].
351    pub(crate) fn handle(&self, req: &Request) -> Response {
352        if req.method != "GET" {
353            return Response::error(
354                405,
355                "method_not_allowed",
356                "the UI's screens are read with GET; a capability that changes something is called on its own route",
357            );
358        }
359        if let Some(name) = req.path.strip_prefix(ASSETS_PREFIX) {
360            return self.assets.respond(name, req);
361        }
362        // one reading of the repository for the whole screen: the generation there is, never
363        // a wait for the next one (ADR 0142); a screen drawn from a generation the repository
364        // has moved past says so, and the page's script reads it again
365        let (reading, behind) = self.live.view_without_waiting();
366        let ctx = reading.ctx;
367        let fragment = req.query.iter().any(|(k, _)| k == FRAGMENT);
368        let path = req.path.trim_end_matches('/');
369        // an earlier or mistyped name leads to the route it meant, query kept
370        if let Some((_, target)) = observatory::REDIRECTS.iter().find(|(old, _)| *old == path) {
371            let query = req
372                .query
373                .iter()
374                .map(|(k, v)| format!("{k}={v}"))
375                .collect::<Vec<_>>()
376                .join("&");
377            let to = match query.is_empty() {
378                true => target.to_string(),
379                false => format!("{target}?{query}"),
380            };
381            return Response::new(308, "text/plain; charset=utf-8", format!("{to}\n"))
382                .with_header("Location", to);
383        }
384        // the kiosk is the Observatory's screen under a second name, never a second screen
385        let kiosk = path == observatory::KIOSK;
386        let path = if kiosk { observatory::ROUTE } else { path };
387        let mut screen = self.view(&ctx, path, &req.query, kiosk);
388        // A snapshot has no script to fetch a slow panel after the screen has painted, so the
389        // panel is read now and stands where its placeholder stood, as the script would have
390        // put it: a published screen carries what the live one shows (ADR 0141).
391        if req.header(SNAPSHOT_HEADER).is_some() {
392            screen.main = screen.main.replacing("data-panel-src", &mut |src| {
393                let panel = Request::parse_target("GET", src, Vec::new());
394                let route = panel.path.trim_end_matches('/');
395                PANELS
396                    .contains(&route)
397                    .then(|| self.view(&ctx, route, &panel.query, false).main)
398            });
399        }
400        if behind {
401            screen.main = crate::cockpit::html::el("div")
402                .class("mj-ui-stack")
403                .child(
404                    crate::cockpit::html::el("p")
405                        .class("mj-ui-dim")
406                        .attr(view::CATCHING_UP, "")
407                        .text("The repository changed: this screen shows it as it was, and is being read again."),
408                )
409                .child(screen.main);
410        }
411        // A panel read as a page of its own has no hero, so it carries the page's one heading
412        // itself; the fragment a screen pulls in does not, or the screen would have two.
413        if !fragment && PANELS.contains(&path) {
414            screen.main = crate::cockpit::html::el("div")
415                .class("mj-ui-stack")
416                .child(
417                    crate::cockpit::html::el("h1")
418                        .class("mj-ui-title")
419                        .text(&screen.title),
420                )
421                .child(screen.main);
422        }
423        let status = screen.status;
424        let body = if fragment {
425            screen.main.render()
426        } else {
427            shell::page(
428                &ctx,
429                &shell::Frame {
430                    current: path,
431                    stylesheet: self.assets.url(STYLESHEET),
432                    favicon: self.assets.url("favicon.svg"),
433                    script: self.assets.url(SCRIPT),
434                    bar_scripts: BAR_SCRIPTS.iter().map(|s| self.assets.url(s)).collect(),
435                    // named only when the distribution holds it: a checkout that never ran
436                    // scripts/cockpit-assets has none, and asking for it would be a 404 and a
437                    // console error on every screen with a graph; the table stands instead (I2900)
438                    graph_lib: if self.assets.has(GRAPH_LIB) {
439                        self.assets.url(GRAPH_LIB)
440                    } else {
441                        String::new()
442                    },
443                    moves_lib: self.cockpit_assets.url(MOVES_LIB),
444                    socket_lib: self.cockpit_assets.url(SOCKET_LIB),
445                    assets_present: self.assets.present(),
446                    version: self.version,
447                },
448                screen,
449            )
450        };
451        let mut response = Response::new(status, "text/html; charset=utf-8", body);
452        for (name, value) in [
453            ("Cache-Control", "no-cache"),
454            ("Content-Security-Policy", crate::cockpit::csp()),
455            ("X-Content-Type-Options", "nosniff"),
456            ("Referrer-Policy", "no-referrer"),
457        ] {
458            response.headers.push((name.into(), value.into()));
459        }
460        response
461    }

463    /// The view a route answers with: a screen, a panel, or the page that says the route is
464    /// not one.
465    fn view(
466        &self,
467        ctx: &crate::capability::Context,
468        path: &str,
469        query: &[(String, String)],
470        kiosk: bool,
471    ) -> shell::View {
472        match path {
473            "" | PREFIX => dashboard::screen(ctx),
474            idd::ROUTE => idd::screen(ctx, query),
475            roadmap::ROUTE => roadmap::screen(ctx, query),
476            work::ROUTE => work::screen(ctx, query),
477            issues::ROUTE => issues::screen(ctx, query),
478            prs::ROUTE => prs::screen(ctx, query),
479            mesh::ROUTE => mesh::screen(ctx, query),
480            peers::ROUTE => peers::screen(ctx, &self.assets.url(peers::LIB)),
481            delivery::ROUTE => delivery::screen(ctx),
482            deployments::ROUTE => deployments::screen(ctx),
483            command::ROUTE => command::screen(ctx, query),
484            capabilities::ROUTE => capabilities::screen(ctx, query),
485            skills::ROUTE => skills::screen(ctx),
486            docs::ROUTE => docs::screen(ctx),
487            trends::ROUTE => trends::screen(ctx),
488            evidence::ROUTE => evidence::screen(ctx, query),
489            knowledge::ROUTE => knowledge::screen(ctx),
490            governance::ROUTE => governance::screen(ctx, query),
491            observatory::ROUTE => {
492                let libs = observatory::Libs {
493                    director: self.assets.url(observatory::LIB),
494                    scenes: self.assets.url(observatory::SCENES_LIB),
495                    p5: self.assets.url(observatory::P5_LIB),
496                };
497                observatory::screen(ctx, query, &libs, kiosk)
498            }
499            dashboard::PULL_REQUESTS_PANEL => dashboard::pull_requests_panel(ctx, query),
500            deployments::DELIVERY_PANEL => deployments::delivery_panel(ctx),
501            prs::PANEL => prs::panel_view(ctx, query),
502            idd::STANDING_PANEL => idd::standing_panel(ctx, query),
503            idd::SHARED_SCOPE_PANEL => idd::shared_scope_panel(ctx, query),
504            other => shell::not_found(other),
505        }
506    }

513    /// The navigation and the dispatcher describe one UI. The dispatcher's arms are read
514    /// from this file's own source, so an arm added without a screen or a panel, or a
515    /// screen with no arm, fails here.
516    #[test]
517    fn every_screen_is_answered_and_every_answered_route_is_a_screen_or_a_panel() {
518        let source = include_str!("mod.rs");
519        let dispatcher = source
520            .split("fn view(")
521            .nth(1)
522            .and_then(|rest| rest.split("other =>").next())
523            .expect("the dispatcher's match");
524        let arms: Vec<&str> = dispatcher
525            .lines()
526            .filter_map(|l| l.trim().split_once(" =>").map(|(arm, _)| arm.trim()))
527            .collect();
528        assert_eq!(
529            arms.len(),
530            SCREENS.len() + PANELS.len(),
531            "an arm that is neither a screen nor a panel: {arms:?}"
532        );
533        for screen in SCREENS {
534            assert!(Ui::owns(screen.route), "{} is outside the UI", screen.route);
535        }
536        for panel in PANELS {
537            assert!(Ui::owns(panel), "{panel} is outside the UI");
538        }
539        // the prefix and what is below it, and nothing that only shares its letters
540        assert!(Ui::owns("/ui") && Ui::owns("/ui/panels/pull-requests"));
541        assert!(!Ui::owns("/uix") && !Ui::owns("/cockpit"));
542        let mut routes: Vec<&str> = SCREENS.iter().map(|s| s.route).collect();
543        routes.extend(PANELS);
544        routes.sort_unstable();
545        routes.dedup();
546        assert_eq!(
547            routes.len(),
548            SCREENS.len() + PANELS.len(),
549            "a route is declared twice"
550        );
551    }

685    /// Every class a screen's source gives an element is one the UI's stylesheet defines, or
686    /// a block whose parts it defines (`mj-ui-graph` beside `.mj-ui-graph-canvas`): a class
687    /// nothing styles is a promise the page does not keep, as the search labels were visible
688    /// while `mj-ui-sr-only` was meant to hide them (I2852). Read from the sources, so a
689    /// screen added later is held too.
690    #[test]
691    fn every_class_a_screen_emits_is_one_the_stylesheet_defines() {
692        let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR"));
693        // the stylesheet the UI ships, compiled from its source and the design files it imports
694        let css = std::fs::read_to_string(root.join("../../share/ui/ui.css"))
695            .expect("the UI's stylesheet");
696        let defined: std::collections::BTreeSet<&str> = css
697            .split('.')
698            .skip(1)
699            .filter_map(|s| {
700                let end = s
701                    .find(|c: char| !(c.is_ascii_alphanumeric() || c == '-' || c == '_'))
702                    .unwrap_or(s.len());
703                s[..end].starts_with("mj-").then(|| &s[..end])
704            })
705            .collect();
706        assert!(
707            defined.len() > 50,
708            "the stylesheet was not read: {defined:?}"
709        );
710        let accounted = |c: &str| {
711            defined.contains(c)
712                || defined
713                    .iter()
714                    .any(|d| d.starts_with(&format!("{c}-")) || d.starts_with(&format!("{c}--")))
715        };
716        let mut undefined: Vec<String> = Vec::new();
717        let mut seen = 0;
718        for entry in std::fs::read_dir(root.join("src/ui")).expect("the UI's sources") {
719            let path = entry.expect("a source").path();
720            if path.extension().and_then(|e| e.to_str()) != Some("rs") {
721                continue;
722            }
723            let source = std::fs::read_to_string(&path).expect("a source file");
724            for (n, line) in source.lines().enumerate() {
725                for call in line.split(".class(\"").skip(1) {
726                    let value = call.split('"').next().unwrap_or_default();
727                    for class in value.split_whitespace().filter(|c| c.starts_with("mj-")) {
728                        seen += 1;
729                        if !class.contains('{') && !accounted(class) {
730                            let file = path.file_name().and_then(|f| f.to_str()).unwrap_or("?");
731                            undefined.push(format!("{file}:{} {class}", n + 1));
732                        }
733                    }
734                }
735            }
736        }
737        assert!(
738            seen > 100,
739            "only {seen} class(es) were read from the sources"
740        );
741        assert!(
742            undefined.is_empty(),
743            "classes the stylesheet does not define: {undefined:?}"
744        );
745    }
