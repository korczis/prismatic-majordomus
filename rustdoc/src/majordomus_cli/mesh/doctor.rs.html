1//! The mesh self-check: every prerequisite proved on this machine alone, no second node
2//! required. Deterministic checks in a fixed order, each with its own verdict, so "why
3//! is the mesh not working" has an answer that names the broken link instead of a
4//! shrug — and, when a check fails or limits cooperation, what that means and what to do.
5//! Read-only toward the repository; the sockets it probes are ephemeral and closed before
6//! it answers.
7//!
8//! One check judges a decision rather than the machine. `runtime` reads what the mesh
9//! runtime of the process it runs in decided: in a shared server, whether an enabled
10//! declaration actually activated, and why not when it did not — the failure a declaration
11//! that says on and a server that is off would otherwise be, silently (ADR 0059). In any
12//! other process nothing activates the mesh, and the check reports that absence instead of
13//! judging it; `majordomus mesh doctor` therefore asks the running server for the report
14//! when one serves the checkout, so that the verdict is the server's.
15//!
16//! What this cannot see is whether the server's cooperation thread beats and whether its
17//! peers answer. That is `mesh verify`, which asks the server for a live round.
18//!
19//! ```
20//! use majordomus_cli::mesh::doctor::doctor;
21//!
22//! // No declaration is the default posture, and the self-check says so and runs on.
23//! let report = doctor(None);
24//! let declaration = report.checks.iter().find(|c| c.check == "declaration").unwrap();
25//! assert!(declaration.ok);
26//! assert!(declaration.detail.contains("default posture"));
27//! ```

43/// One check's verdict.
44#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
45pub struct DoctorCheck {
46    /// What was checked.
47    pub check: String,
48    /// Whether it holds.
49    pub ok: bool,
50    /// The evidence: a path, an address, an error.
51    pub detail: String,
52    /// What a failure or a limitation means for the mesh.
53    #[serde(default, skip_serializing_if = "Option::is_none")]
54    pub impact: Option<String>,
55    /// What to do about it.
56    #[serde(default, skip_serializing_if = "Option::is_none")]
57    pub remediation: Option<String>,
58}
59
60impl DoctorCheck {
61    fn pass(check: &str, detail: impl Into<String>) -> Self {
62        DoctorCheck {
63            check: check.into(),
64            ok: true,
65            detail: detail.into(),
66            impact: None,
67            remediation: None,
68        }
69    }
70
71    fn fail(check: &str, detail: impl Into<String>, impact: &str, remediation: &str) -> Self {
72        DoctorCheck {
73            check: check.into(),
74            ok: false,
75            detail: detail.into(),
76            impact: Some(impact.into()),
77            remediation: Some(remediation.into()),
78        }
79    }
80}

82/// The `trust-root` check: which copy of the declaration is obeyed, and every line of the
83/// working tree's copy this runtime ignores because the trunk's copy does not say it
84/// ([`super::root`]). `None` when there is no declaration to judge.
85///
86/// ```
87/// use majordomus_cli::mesh::doctor::declaration_source;
88/// use majordomus_cli::mesh::root::Resolved;
89/// use majordomus_cli::mesh::MeshConfig;
90/// let config = MeshConfig::from_metadata(
91///     &serde_json::json!({"schema": "mesh/v1", "kind": "mesh-declaration", "id": "x"}), "x").unwrap();
92/// let ignored = Resolved {
93///     declaration: Some(Ok(config)),
94///     trunk: Some("master".into()),
95///     committed: true,
96///     narrowed: vec!["trust.allow: bb is not on the trunk".into()],
97/// };
98/// let check = declaration_source(&ignored).unwrap();
99/// assert!(!check.ok && check.detail.contains("bb is not on the trunk"));
100/// let nothing = Resolved { declaration: None, trunk: None, committed: false, narrowed: vec![] };
101/// assert!(declaration_source(&nothing).is_none());
102/// ```
103pub fn declaration_source(resolved: &super::root::Resolved) -> Option<DoctorCheck> {
104    resolved.declaration.as_ref()?;
105    let trunk = resolved.trunk.as_deref().unwrap_or("the trunk");
106    if resolved.narrowed.is_empty() {
107        let detail = if resolved.committed {
108            format!("obeyed as {trunk} holds it; the working tree widens nothing")
109        } else {
110            format!("not committed on {trunk}; obeyed as written, since it widens nothing a committed copy would have to")
111        };
112        return Some(DoctorCheck::pass("trust-root", detail));
113    }
114    Some(DoctorCheck::fail(
115        "trust-root",
116        format!(
117            "this checkout's runtime ignores {} line(s) of the working tree's declaration: {}",
118            resolved.narrowed.len(),
119            resolved.narrowed.join("; ")
120        ),
121        "a working tree may narrow trust and never widen it: a key, an address or a looser policy takes effect only once the trunk's copy says it",
122        &format!("commit the change on {trunk} (merge it), or revert it here"),
123    ))
124}

126/// The `journal` check: whether the cooperation journal's file holds what the journal
127/// holds. A failed append keeps the event in memory and in replication, loses it from this
128/// machine's disk and is not acknowledged as durable; an unreadable line is one the last
129/// reload could not use; unread bytes lie beyond the reload bound.
130///
131/// ```
132/// use majordomus_cli::mesh::doctor::journal_check;
133/// use majordomus_cli::mesh::journal::JournalTallies;
134/// assert!(journal_check(&JournalTallies::default()).ok);
135/// let lost = JournalTallies { write_failures: 2, ..Default::default() };
136/// let check = journal_check(&lost);
137/// assert!(!check.ok && check.detail.contains("2 event(s)"));
138/// let beyond = JournalTallies { unread_bytes: 4096, ..Default::default() };
139/// let check = journal_check(&beyond);
140/// assert!(!check.ok && check.detail.contains("4096 byte(s)"));
141/// ```
142pub fn journal_check(tallies: &super::journal::JournalTallies) -> DoctorCheck {
143    if tallies.write_failures == 0 && tallies.unreadable == 0 && tallies.unread_bytes == 0 {
144        return DoctorCheck::pass(
145            "journal",
146            format!(
147                "{} event(s) held, every append synced to disk, the file reloaded whole with no unreadable line",
148                tallies.events
149            ),
150        );
151    }
152    DoctorCheck::fail(
153        "journal",
154        format!(
155            "{} event(s) did not reach the journal file and were not acknowledged as durable; {} line(s) of it did not read on reload; {} byte(s) of it lie beyond the {}-byte reload bound and were left on disk unread",
156            tallies.write_failures,
157            tallies.unreadable,
158            tallies.unread_bytes,
159            super::journal::MAX_RELOAD_BYTES
160        ),
161        "an event that is not on disk is lost to this machine when the runtime restarts, unless a peer still holds it; the events beyond the reload bound are not held, so this runtime neither folds nor relays them",
162        "free disk space or fix the permissions of the mesh state directory; a torn line is left as one unreadable line, and the events after it read; a journal beyond the bound was not written by this executable's compaction — move it aside and let the peers refill it",
163    )
164}

166/// The `transport` check: how each declared rendezvous endpoint and seed would be reached —
167/// over TLS (`https://`), or in cleartext to an address that is loopback, private or an
168/// overlay's — and every one the client refuses to dial, because it would carry the mesh in
169/// cleartext across a public network (ADR 0220). Nothing is dialed; an http host name is
170/// resolved, and one that does not resolve here is reported and judged again at every dial.
171///
172/// ```
173/// use majordomus_cli::mesh::doctor::transport_check;
174/// use majordomus_cli::mesh::MeshConfig;
175/// let declare = |endpoints: serde_json::Value| -> MeshConfig {
176///     serde_json::from_value(serde_json::json!({ "schema": "mesh/v1", "kind": "mesh-declaration",
177///         "id": "x", "rendezvous": { "endpoints": endpoints } })).unwrap()
178/// };
179/// let held = transport_check(&declare(serde_json::json!(["https://hub.example", "http://10.0.0.2:8791"])));
180/// assert!(held.ok && held.detail.starts_with("1 endpoint(s) over TLS, 1 in cleartext"), "{}", held.detail);
181/// let refused = transport_check(&declare(serde_json::json!(["http://203.0.113.5:8791"])));
182/// assert!(!refused.ok && refused.detail.contains("203.0.113.5 is a public address"));
183/// ```
184pub fn transport_check(config: &MeshConfig) -> DoctorCheck {
185    use crate::mcp::bridge::{transport, Transport};
186    let (mut tls, mut cleartext) = (0usize, 0usize);
187    let (mut refused, mut unresolved) = (Vec::new(), Vec::new());
188    for endpoint in config
189        .rendezvous
190        .endpoints
191        .iter()
192        .chain(&config.cooperation.seeds)
193    {
194        match transport(endpoint) {
195            Ok(Transport::Tls) => tls += 1,
196            Ok(Transport::Cleartext) => cleartext += 1,
197            Err(e) if e.kind() == std::io::ErrorKind::PermissionDenied => {
198                refused.push(e.to_string())
199            }
200            Err(e) => unresolved.push(format!("{endpoint}: {e}")),
201        }
202    }
203    if refused.is_empty() {
204        let mut detail = format!(
205            "{tls} endpoint(s) over TLS, {cleartext} in cleartext to loopback, private or overlay addresses"
206        );
207        if !unresolved.is_empty() {
208            detail.push_str(&format!(
209                "; not resolved here, judged again at every dial: {}",
210                unresolved.join("; ")
211            ));
212        }
213        return DoctorCheck::pass("transport", detail);
214    }
215    DoctorCheck::fail(
216        "transport",
217        refused.join("; "),
218        "the client refuses to dial these: no registration and no link is sent to them, so they neither discover nor cooperate with this node",
219        "serve the hub over TLS and declare it as https://, or reach it at a private or overlay address (a tailnet)",
220    )
221}

232/// Run the self-check against the declaration as parsed (or its absence, or its error),
233/// in a process no shared server runs in. Deterministic order, no second node required, no

235pub fn doctor(declaration: Option<Result<MeshConfig, MeshError>>) -> MeshDoctorReport {
236    doctor_at(declaration, None, None)

239/// [`doctor`] for the repository at `root`, in the process whose mesh runtime decided
240/// `runtime`: adds whether the repository has a mesh identity cooperation can match on, and
241/// judges what the runtime decided about the declaration.
242///
243/// Discovery works without a repository identity — two nodes can hear each other while
244/// belonging to different repositories — but a link does not, so a report that omits this
245/// check can say every prerequisite holds while cooperation is unreachable. `root` is
246/// optional because the caller may have no repository at hand, and then the check is not
247/// run rather than failed.
248///
249/// `runtime` is the status of a runtime a shared server decided on — activated, or declined
250/// with a reason — and `None` in any process no server runs in
251/// ([`MeshRuntime::decided`](super::manager::MeshRuntime::decided)). The `runtime` check
252/// fails exactly when the declaration is enabled and a server's mesh is not active.
253///
254/// ```
255/// use majordomus_cli::mesh::doctor::doctor_at;
256/// use majordomus_cli::mesh::manager::MeshRuntime;
257/// use majordomus_cli::mesh::MeshConfig;
258///
259/// // a directory holding no history has no identity to match on, and the check says so
260/// let nowhere = tempfile::tempdir().unwrap();
261/// let report = doctor_at(None, Some(nowhere.path()), None);
262/// let repository = report.checks.iter().find(|c| c.check == "repository").unwrap();
263/// assert!(!repository.ok);
264/// assert!(repository.remediation.as_deref().unwrap().contains("cooperation.repository"));
265///
266/// // with no root to ask about, the question is not asked at all
267/// assert!(doctor_at(None, None, None).checks.iter().all(|c| c.check != "repository"));
268///
269/// // an enabled declaration a server declined is the runtime's failure, with the reason
270/// let enabled: MeshConfig = serde_json::from_value(serde_json::json!({
271///     "schema": "mesh/v1", "kind": "mesh-declaration", "id": "doc", "enabled": true,
272/// })).unwrap();
273/// let server = MeshRuntime::new();
274/// server.decline("the node identity did not load");
275/// let report = doctor_at(Some(Ok(enabled)), None, Some(&server.status()));
276/// let runtime = report.checks.iter().find(|c| c.check == "runtime").unwrap();
277/// assert!(!runtime.ok && !report.ok);
278/// assert!(runtime.detail.ends_with("the node identity did not load"));
279/// ```
280pub fn doctor_at(
281    declaration: Option<Result<MeshConfig, MeshError>>,
282    root: Option<&Path>,
283    runtime: Option<&MeshStatus>,
284) -> MeshDoctorReport {
285    let mut checks = Vec::new();
286
287    // The declaration: present, parseable, and what it says.
288    match &declaration {
289        None => checks.push(DoctorCheck::pass(
290            "declaration",
291            "absent: the mesh is off, which is the default posture",
292        )),
293        Some(Err(e)) => checks.push(DoctorCheck::fail(
294            "declaration",
295            e.to_string(),
296            "the mesh stays off: no discovery socket opens and no link is made",
297            "fix the declaration under .ai/repo/mesh/ (majordomus check names the schema violation), then restart the server",
298        )),
299        Some(Ok(config)) => checks.push(DoctorCheck::pass(
300            "declaration",
301            format!(
302                "{}: enabled={}, multicast={}, broadcast={}, rendezvous endpoints={}, trust={} ({} allowed key(s)), cooperation={} (heartbeat {}s, expiry {}s, {} seed(s))",
303                config.id,
304                config.enabled,
305                config.multicast.enabled,
306                config.broadcast.mode != super::config::BroadcastMode::Disabled,
307                config.rendezvous.endpoints.len(),
308                config.trust.policy.as_str(),
309                config.trust.allow.len(),
310                config.cooperation.enabled,
311                config.cooperation.heartbeat_seconds,
312                config.cooperation.expiry_seconds,
313                config.cooperation.seeds.len()
314            ),
315        )),
316    }
317
318    // The transport: how each declared hub and seed would be reached, and which the client
319    // refuses to dial.
320    if let Some(Ok(config)) = &declaration {
321        checks.push(transport_check(config));
322    }
323
324    // The identity: where it lives, and whether what is there loads.
325    let identity = match default_identity_path() {
326        None => {
327            checks.push(DoctorCheck::fail(
328                "identity",
329                "no HOME and no XDG_STATE_HOME: this process has nowhere to keep a node identity",
330                "the mesh cannot activate: a node without a key cannot be told apart from an impostor",
331                "run the server with HOME or XDG_STATE_HOME set",
332            ));
333            None
334        }
335        Some(path) if path.is_file() => match NodeIdentity::load_or_create(&path) {
336            Ok(identity) => {
337                checks.push(DoctorCheck::pass(
338                    "identity",
339                    format!("{} at {}", identity.public.node_id, path.display()),
340                ));
341                Some(identity)
342            }
343            Err(e) => {
344                checks.push(DoctorCheck::fail(
345                    "identity",
346                    e.to_string(),
347                    "the mesh cannot activate with a broken identity, and replacing it silently would make this machine a new node every peer distrusts",
348                    "restore the file from a backup, or remove it deliberately — a new key is a new node, and every allowlist naming the old key must be updated",
349                ));
350                None
351            }
352        },
353        Some(path) => {
354            checks.push(DoctorCheck::pass(
355                "identity",
356                format!("absent; created at first activation at {}", path.display()),
357            ));
358            None
359        }
360    };
361
362    // Trust: under an allowlist-shaped declaration, is this machine's own key listed? An
363    // unlisted machine keeps the mesh closed (`manager::closed_to`), so this is the check
364    // that names why and what to do.
365    if let (Some(Ok(config)), Some(identity)) = (&declaration, &identity) {
366        if config.enabled && config.trust.policy != super::trust::TrustPolicy::Tofu {
367            let listed = config.trust.allow.contains(&identity.public.public_key);
368            checks.push(if listed {
369                DoctorCheck::pass("trust", "this machine's key is on the allowlist")
370            } else {
371                DoctorCheck::fail(
372                    "trust",
373                    format!(
374                        "this machine's key {} is not among the {} allowed",
375                        identity.public.public_key,
376                        config.trust.allow.len()
377                    ),
378                    "the mesh stays closed on this machine: it neither announces nor dials, and no other machine links to it",
379                    "add the key above to trust.allow in the mesh declaration and commit it",
380                )
381            });
382        }
383    }
384
385    // The repository's mesh identity: what links are matched on.
386    if let Some(root) = root {
387        let declared = declaration
388            .as_ref()
389            .and_then(|d| d.as_ref().ok())
390            .and_then(|c| c.cooperation.repository.clone());
391        checks.push(match super::repository::resolve(root, declared.as_deref()) {
392            Ok(identity) => DoctorCheck::pass(
393                "repository",
394                format!(
395                    "{} from {:?}: {}",
396                    identity.id, identity.basis, identity.detail
397                ),
398            ),
399            Err(e) => DoctorCheck::fail(
400                "repository",
401                e.to_string(),
402                "cooperation stays off: without a repository identity no peer can be matched to this repository",
403                "declare cooperation.repository in the mesh declaration, or fetch the full history (git fetch --unshallow)",
404            ),
405        });
406    }
407
408    // A UDP socket at all.
409    checks.push(match UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)) {
410        Ok(socket) => DoctorCheck::pass(
411            "udp",
412            format!(
413                "ephemeral bind ok ({})",
414                socket
415                    .local_addr()
416                    .map(|a| a.to_string())
417                    .unwrap_or_else(|_| "?".into())
418            ),
419        ),
420        Err(e) => DoctorCheck::fail(
421            "udp",
422            format!("cannot bind an ephemeral UDP socket: {e}"),
423            "multicast and broadcast discovery cannot run; seeds and rendezvous still can",
424            "allow UDP for this process (a sandbox or a firewall is refusing it), or declare cooperation.seeds",
425        ),
426    });
427
428    // Multicast capability, against the declared (or default) group.
429    let group_text = declaration
430        .as_ref()
431        .and_then(|d| d.as_ref().ok())
432        .map(|c| c.multicast.group.clone())
433        .unwrap_or_else(|| super::config::DEFAULT_GROUP.into());
434    checks.push(match multicast_probe(&group_text) {
435        Ok(()) => DoctorCheck::pass(
436            "multicast",
437            format!("joined and left {group_text} on an ephemeral socket"),
438        ),
439        Err(e) => DoctorCheck::fail(
440            "multicast",
441            e.to_string(),
442            "runtimes on this segment will not hear each other by multicast",
443            "check the interface's multicast route, or declare cooperation.seeds or rendezvous endpoints instead",
444        ),
445    });
446
447    // Broadcast capability.
448    checks.push(
449        match UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)).and_then(|s| s.set_broadcast(true)) {
450            Ok(()) => DoctorCheck::pass("broadcast", "the broadcast flag sets"),
451            Err(e) => DoctorCheck::fail(
452                "broadcast",
453                format!("cannot enable broadcast: {e}"),
454                "the broadcast fallback cannot run",
455                "leave broadcast disabled and rely on multicast, seeds or rendezvous",
456            ),
457        },
458    );
459
460    // The discovery protocol, end to end in memory: sign, encode, parse, verify.
461    checks.push(match protocol_probe() {
462        Ok(detail) => DoctorCheck::pass("protocol", detail),
463        Err(e) => DoctorCheck::fail(
464            "protocol",
465            e,
466            "no advertisement this executable writes would verify anywhere",
467            "this is a defect of the executable: reinstall it and report the detail",
468        ),
469    });
470
471    // The link protocol, end to end in memory: a hello signed, verified, refused when
472    // tampered — the handshake's own cryptography, with no peer.
473    checks.push(match link_probe() {
474        Ok(detail) => DoctorCheck::pass("link", detail),
475        Err(e) => DoctorCheck::fail(
476            "link",
477            e,
478            "no link this executable opens would be admitted anywhere",
479            "this is a defect of the executable: reinstall it and report the detail",
480        ),
481    });
482
483    // The host firewall: what the declaration implies this machine must admit, and whether
484    // the host does. Every check above is a fact of the process; this one is a fact of the
485    // host, and the one that was missing when a declared hub dropped every registration
486    // at the kernel while the doctor said every check held.
487    checks.push(firewall_check(&declaration));
488
489    // The runtime: what the process's mesh runtime decided about the declaration, when
490    // anything decided. Every check above says the machine could run a mesh; this one says
491    // whether the server did.
492    checks.push(runtime_check(declared(&declaration), runtime));
493
494    MeshDoctorReport {
495        ok: checks.iter().all(|c| c.ok),
496        checks,
497    }
498}

500/// The `firewall` verdict: the admissions the declaration implies on this host, derived
501/// by [`firewall::plan`] from the declaration and this machine's addresses, against what
502/// the host's firewall says and what the kernel logged it dropping. It fails when a rule
503/// is observed missing or when the kernel logged drops toward the mesh's ports in the
504/// last five minutes; a firewall that cannot be asked without root, with nothing logged, holds —
505/// an absence, not a verdict — and says how to ask as root.
506fn firewall_check(declaration: &Option<Result<MeshConfig, MeshError>>) -> DoctorCheck {
507    let config = declaration.as_ref().and_then(|d| d.as_ref().ok());
508    let local: Vec<std::net::Ipv4Addr> = super::address::interfaces()
509        .into_iter()
510        .map(|(_, ip)| ip)
511        .collect();
512    firewall_verdict(&firewall::report(config, &local, None))
513}
514
515/// The `firewall` check of a report already built. Apart from [`firewall_check`] so that
516/// the verdict is a function of the report alone: which firewall a machine runs, and
517/// whether it admits the mesh, is a fact of that machine, and the failing verdict has to
518/// be held on every machine, not only on one whose firewall happens to drop the mesh.
519fn firewall_verdict(report: &firewall::FirewallReport) -> DoctorCheck {
520    let detail = if report.plan.rules.is_empty() {
521        report.verdict.clone()
522    } else {
523        format!(
524            "{} ({}): needs {}; {}",
525            report.backend.as_str(),
526            report.platform,
527            report
528                .plan
529                .rules
530                .iter()
531                .map(firewall::FirewallRule::label)
532                .collect::<Vec<_>>()
533                .join(", "),
534            report.verdict
535        )
536    };
537    if report.ok {
538        DoctorCheck::pass("firewall", detail)
539    } else {
540        DoctorCheck::fail(
541            "firewall",
542            detail,
543            "what the firewall drops never reaches the socket: a declared hub that drops registrations is a hub nobody can register with, and a machine that drops the multicast group hears no peer on its segment, while every other check holds",
544            "run `majordomus mesh firewall` for the rules and the commands that admit them, and `sudo majordomus mesh firewall apply` to admit them",
545        )
546    }
547}

549/// What the declaration says about the mesh, as far as the `runtime` verdict needs it.
550#[derive(Debug, Clone, Copy, PartialEq, Eq)]
551enum Declared {
552    /// No mesh-declaration object in the layer.
553    Absent,
554    /// An object that does not parse.
555    Broken,
556    /// `enabled: false`.
557    Disabled,
558    /// `enabled: true`.
559    Enabled,
560}
561
562fn declared(declaration: &Option<Result<MeshConfig, MeshError>>) -> Declared {
563    match declaration {
564        None => Declared::Absent,
565        Some(Err(_)) => Declared::Broken,
566        Some(Ok(config)) if config.enabled => Declared::Enabled,
567        Some(Ok(_)) => Declared::Disabled,
568    }
569}

571/// The `runtime` verdict. `None` is a runtime nobody decided on — the command line's, a
572/// test's — whose inactivity is an absence and never a failure. A decided runtime is
573/// active (a degraded one names its failed providers, and fails only when no provider
574/// runs at all), off as declared, off because there is nothing to activate, or — the one
575/// failure — off under an enabled declaration, carrying the server's reason.
576/// The `runtime` verdict alone: what a server decided about the declaration, judged as
577/// `mesh doctor` judges it. Health reads this one check rather than running the doctor's
578/// probes, which open sockets.
579///
580/// ```
581/// use majordomus_cli::mesh::doctor::runtime_verdict;
582/// use majordomus_cli::mesh::manager::MeshRuntime;
583/// use majordomus_cli::mesh::MeshConfig;
584/// let enabled: MeshConfig = serde_json::from_value(serde_json::json!({
585///     "schema": "mesh/v1", "kind": "mesh-declaration", "id": "x", "enabled": true })).unwrap();
586/// let server = MeshRuntime::new();
587/// server.decline("the node identity did not load");
588/// assert!(!runtime_verdict(Some(Ok(enabled)), Some(&server.status())).ok);
589/// assert!(runtime_verdict(None, Some(&server.status())).ok, "no declaration, nothing owed");
590/// ```
591pub fn runtime_verdict(
592    declaration: Option<Result<MeshConfig, MeshError>>,
593    runtime: Option<&MeshStatus>,
594) -> DoctorCheck {
595    runtime_check(declared(&declaration), runtime)
596}
597
598fn runtime_check(declared: Declared, runtime: Option<&MeshStatus>) -> DoctorCheck {
599    let Some(status) = runtime else {
600        return DoctorCheck::pass(
601            "runtime",
602            match declared {
603                Declared::Enabled => {
604                    "not decided in this process: the mesh lives in the shared server, and `majordomus mesh doctor` asks the server when one serves this checkout"
605                }
606                Declared::Disabled => {
607                    "not decided in this process, and nothing to decide: the declaration is disabled"
608                }
609                Declared::Broken => {
610                    "not decided in this process, and nothing to decide: the declaration does not parse"
611                }
612                Declared::Absent => {
613                    "not decided in this process, and nothing to decide: the repository declares no mesh"
614                }
615            },
616        );
617    };
618    let reason = status.reason.as_deref().unwrap_or("no reason recorded");
619    let why = reason.strip_prefix("not active: ").unwrap_or(reason);
620    if !status.active {
621        return match declared {
622            // closed by the trust list: the trust check carries the failure and the remedy
623            Declared::Enabled if why.starts_with("closed: ") => {
624                DoctorCheck::pass("runtime", why.to_string())
625            }
626            Declared::Enabled => DoctorCheck::fail(
627                "runtime",
628                format!("the declaration is enabled and this server's mesh is not active — {why}"),
629                "this checkout's server announces nothing, hears nobody and links to no peer, while the committed declaration says the fleet sees it",
630                "fix what the reason names, then restart the server (`majordomus serve stop && majordomus serve ensure`)",
631            ),
632            Declared::Disabled => DoctorCheck::pass("runtime", format!("off, as declared — {why}")),
633            Declared::Broken => DoctorCheck::pass(
634                "runtime",
635                format!("off: the declaration does not parse, so nothing was activated — {why}"),
636            ),
637            Declared::Absent => DoctorCheck::pass(
638                "runtime",
639                format!("off: the repository declares no mesh — {why}"),
640            ),
641        };
642    }
643    let running = status
644        .providers
645        .iter()
646        .filter(|p| matches!(p.state, MeshProviderState::Running))
647        .count();
648    let failed: Vec<String> = status
649        .providers
650        .iter()
651        .filter(|p| matches!(p.state, MeshProviderState::Failed))
652        .map(|p| format!("{} ({})", p.id, p.detail.as_deref().unwrap_or("no detail")))
653        .collect();
654    let node = status
655        .identity
656        .as_ref()
657        .map(|i| i.node_id.to_string())
658        .unwrap_or_else(|| "?".into());
659    let mut detail = format!(
660        "active as {node} since {}: {running} of {} provider(s) running; {} node(s) known, {} trusted, {} present",
661        status.started_at.as_deref().unwrap_or("?"),
662        status.providers.len(),
663        status.tallies.nodes,
664        status.tallies.trusted,
665        status.tallies.present
666    );
667    if !failed.is_empty() {
668        detail.push_str(&format!("; failed: {}", failed.join(", ")));
669    }
670    // A provider that failed beside one that runs is a degraded mesh, named and holding;
671    // every declared transport failed is a mesh that hears nothing and is heard by nobody.
672    if failed.is_empty() || running > 0 {
673        DoctorCheck::pass("runtime", detail)
674    } else {
675        DoctorCheck::fail(
676            "runtime",
677            detail,
678            "the mesh is active and every discovery transport it declares has failed: it hears nothing and is heard by nobody",
679            "read each failed provider's detail (`majordomus mesh status`), fix what it names, then restart the server",
680        )
681    }
682}

684fn multicast_probe(group_text: &str) -> Result<(), MeshError> {
685    let group: Ipv4Addr = group_text
686        .parse()
687        .map_err(|_| MeshError::Provider(format!("'{group_text}' is not an IPv4 group")))?;
688    if !group.is_multicast() {
689        return Err(MeshError::Provider(format!(
690            "{group} is not a multicast address"
691        )));
692    }
693    let socket = UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0))
694        .map_err(|e| MeshError::Provider(format!("cannot bind: {e}")))?;
695    socket
696        .join_multicast_v4(&group, &Ipv4Addr::UNSPECIFIED)
697        .map_err(|e| MeshError::Provider(format!("cannot join {group}: {e}")))?;
698    let _ = socket.leave_multicast_v4(&group, &Ipv4Addr::UNSPECIFIED);
699    Ok(())
700}

702fn protocol_probe() -> Result<String, String> {
703    let identity = NodeIdentity::ephemeral().map_err(|e| e.to_string())?;
704    let envelope = protocol::advertise_as(
705        &identity,
706        "0000000000000001",
707        1,
708        &["127.0.0.1:1".into()],
709        &["http".into()],
710        &[],
711        "self-check",
712    );
713    let bytes = protocol::encode(&envelope).map_err(|e| e.to_string())?;
714    let parsed = protocol::parse(&bytes).map_err(|r| format!("a fresh envelope refused: {r}"))?;
715    if parsed.adv.node_id() != Some(identity.public.node_id.clone()) {
716        return Err("the parsed envelope proves a different node".into());
717    }
718    Ok(format!(
719        "sign → encode ({} bytes) → parse → verify, protocol {}",
720        bytes.len(),
721        protocol::PROTOCOL_VERSION
722    ))
723}

725fn link_probe() -> Result<String, String> {
726    use super::link::{sign, verify_signed, Domain, LINK_PROTOCOL_MAX, LINK_PROTOCOL_MIN};
727    let identity = NodeIdentity::ephemeral().map_err(|e| e.to_string())?;
728    let mut signed = sign(
729        &identity,
730        Domain::Hello,
731        serde_json::json!({"probe": super::link::fresh_token()}),
732    );
733    if !verify_signed(&identity.public.public_key, Domain::Hello, &signed) {
734        return Err("a fresh hello does not verify".into());
735    }
736    signed.body = serde_json::json!({"probe": "tampered"});
737    if verify_signed(&identity.public.public_key, Domain::Hello, &signed) {
738        return Err("a tampered hello verifies".into());
739    }
740    Ok(format!(
741        "hello signed → verified → tampered copy refused, link protocol {LINK_PROTOCOL_MIN}..{LINK_PROTOCOL_MAX}"
742    ))
743}

897    /// A firewall report as [`firewall::report`] builds one, for a host this suite does
898    /// not run on: ufw on Linux, asked `status`, with the multicast group the one rule an
899    /// enabled declaration with no hub implies. The report is assembled from the module's

901    fn firewall_report(declared: bool, status: &str) -> firewall::FirewallReport {
902        let config = config(declared);

904        let observation = firewall::judge_ufw(status, &plan);
905        let ok = observation.state != firewall::FirewallObservationState::Missing;
906        firewall::FirewallReport {
907            ok,
908            platform: "linux".into(),
909            backend: firewall::Backend::Ufw,
910            executable: None,
911            commands: firewall::render(firewall::Backend::Ufw, &plan, None),
912            plan,
913            verdict: format!("what ufw said: {}", observation.detail),
914            observation,
915            window_seconds: firewall::BLOCK_WINDOW_SECONDS,
916            blocked_recently: None,
917        }
918    }
