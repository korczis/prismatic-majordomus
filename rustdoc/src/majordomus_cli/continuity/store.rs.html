1//! Where published handovers are kept: one git ref, `refs/majordomus/continuity`, whose
2//! commit holds a tree of `records/<id>.json` blobs.
3//!
4//! Git is the transport because it is already what moves this repository between machines,
5//! it works offline, and every clone already has it. A ref of its own, rather than files on
6//! a branch, because of what a branch would cost:
7//!
8//! - **No branch is touched.** A record is written with plumbing (`hash-object`,
9//!   `mktree`, `commit-tree`, `update-ref`), so neither the index nor the working tree nor
10//!   HEAD moves, and a dirty tree can be handed over without anything being committed to the
11//!   work it describes.
12//! - **No pull request carries it.** A record on a feature branch would appear in that
13//!   branch's diff, in its merge, and in every derived projection of tracked files.
14//! - **Every worktree sees it.** Refs live in the common git directory.
15//! - **It does not travel by accident.** A plain `git push` or `git pull` leaves it alone;
16//!   `continuity sync` moves it, and only when asked.
17//!
18//! Records are immutable and named by their content digest, so two stores are merged by
19//! taking the union of their files: the same name is the same bytes, and a name holding
20//! different bytes on the two sides is reported, never resolved by picking one. The ref's
21//! commit history is only a transport detail; the lineage of the work is in the records.
22//!
23//! What another machine published arrives under
24//! `refs/majordomus/remotes/<remote>/continuity` when fetched, and is merged into the local
25//! ref only by `sync`.
26

46/// Is `name` a remote name this module will put into a refspec?
47pub fn valid_remote(name: &str) -> bool {
48    !name.is_empty()
49        && name.len() <= 64
50        && !name.contains("..")
51        && !name.starts_with('-')
52        && name
53            .chars()
54            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.'))
55}

57fn git(root: &Path) -> Command {
58    let mut cmd = Command::new("git");
59    cmd.env_remove("GIT_DIR");
60    cmd.env_remove("GIT_WORK_TREE");
61    cmd.env_remove("GIT_INDEX_FILE");
62    // a push or fetch never waits for a password prompt nobody is there to answer
63    cmd.env("GIT_TERMINAL_PROMPT", "0");
64    // the checkout is the working directory rather than `-C`: the same repository, and a
65    // checkout that does not exist is refused by the spawn itself
66    cmd.current_dir(root);
67    cmd
68}

70/// Run git in `root` with `env` added and `input` on its stdin, and answer its stdout.
71///
72/// Every git call of the store goes through here, so a failure has one shape: git could not
73/// be started, its stdin could not be written (it exited without reading what it was
74/// given), or it refused, with what it said on stderr. Stdin is written on a thread of its
75/// own, because git may start answering before it has read everything (`cat-file --batch`
76/// does) and a full stdout pipe would otherwise stall both sides.
77fn exec(root: &Path, args: &[&str], env: &[(&str, &str)], input: &[u8]) -> Result<Vec<u8>, String> {
78    let mut child = git(root)
79        .args(args)
80        .envs(env.iter().copied())
81        .stdin(Stdio::piped())
82        .stdout(Stdio::piped())
83        .stderr(Stdio::piped())
84        .spawn()
85        .map_err(|e| format!("cannot run git: {e}"))?;
86    let mut stdin = child.stdin.take().expect("stdin is piped");
87    let input = input.to_vec();
88    let writer = std::thread::spawn(move || stdin.write_all(&input));
89    let out = child.wait_with_output();
90    let written = writer.join().expect("writing a pipe does not panic");
91    let out = written
92        .and(out)
93        .map_err(|e| format!("git {}: {e}", args.join(" ")))?;
94    if !out.status.success() {
95        return Err(format!(
96            "git {}: {}",
97            args.join(" "),
98            String::from_utf8_lossy(&out.stderr).trim()
99        ));
100    }
101    Ok(out.stdout)
102}

112/// The commit a ref points at, or `None` when the ref does not exist.
113pub fn tip(root: &Path, reference: &str) -> Option<String> {
114    run(
115        root,
116        &[
117            "rev-parse",
118            "--verify",
119            "--quiet",
120            &format!("{reference}^{{commit}}"),
121        ],
122    )
123    .ok()
124    .filter(|s| !s.is_empty())
125}

127/// The files of a store's tree: path → blob id. Paths other than `records/<hex>.json` are
128/// returned too, so that the reader can report them rather than silently skip them.
129pub fn entries(root: &Path, commit: &str) -> Result<BTreeMap<String, String>, String> {
130    let listing = run(root, &["ls-tree", "-r", "-z", "--full-tree", commit])?;
131    let mut out = BTreeMap::new();
132    for entry in listing.split('\0').filter(|e| !e.is_empty()) {
133        // <mode> SP <type> SP <object> TAB <path>
134        let (meta, path) = entry.split_once('\t').unwrap_or((entry, ""));
135        let mut parts = meta.split(' ');
136        // a blob is a file of the store; anything else in its tree (a gitlink) is not
137        if let (_, Some("blob"), Some(object)) = (parts.next(), parts.next(), parts.next()) {
138            out.insert(path.to_string(), object.to_string());
139            if out.len() > MAX_RECORDS {
140                return Err(format!(
141                    "the store holds more than {MAX_RECORDS} files; it is read no further"
142                ));
143            }
144        }
145    }
146    Ok(out)
147}

149/// The bytes of the given blobs, read in one `cat-file --batch`.
150pub fn blobs(root: &Path, ids: &[&str]) -> Result<BTreeMap<String, Vec<u8>>, String> {
151    if ids.is_empty() {
152        return Ok(BTreeMap::new());
153    }
154    let mut request = String::new();
155    for id in ids {
156        request.push_str(id);
157        request.push('\n');
158    }
159    parse_batch(&exec(
160        root,
161        &["cat-file", "--batch"],
162        &[],
163        request.as_bytes(),
164    )?)
165}
166
167/// The answer of `cat-file --batch`: `<id> <type> <size>\n<bytes>\n` per object, or
168/// `<id> missing\n` for one the repository lacks, which is left out. A blob larger than
169/// any record could be is answered empty, so admission refuses it without it being held.
170fn parse_batch(data: &[u8]) -> Result<BTreeMap<String, Vec<u8>>, String> {
171    let mut result = BTreeMap::new();
172    let mut at = 0usize;
173    while at < data.len() {
174        let Some(nl) = data[at..].iter().position(|b| *b == b'\n') else {
175            return Err("git cat-file: a truncated answer".into());
176        };
177        let header = String::from_utf8_lossy(&data[at..at + nl]).to_string();
178        at += nl + 1;
179        let mut parts = header.split(' ');
180        let (id, _kind, size) = (parts.next(), parts.next(), parts.next());
181        let (Some(id), Some(size)) = (id, size.and_then(|s| s.parse::<usize>().ok())) else {
182            // `<id> missing`
183            continue;
184        };
185        if at + size > data.len() {
186            return Err("git cat-file: a truncated answer".into());
187        }
188        let bytes = if size <= MAX_RECORD_BYTES * 2 {
189            data[at..at + size].to_vec()
190        } else {
191            Vec::new()
192        };
193        result.insert(id.to_string(), bytes);
194        at += size + 1;
195    }
196    Ok(result)
197}

199/// Read every file of the store at `reference`: path → bytes. An absent ref is an empty
200/// store.
201pub fn read(root: &Path, reference: &str) -> Result<BTreeMap<String, Vec<u8>>, String> {
202    let Some(commit) = tip(root, reference) else {
203        return Ok(BTreeMap::new());
204    };
205    // one chain: the listing, then the bytes of what it listed, and either failing is the
206    // read failing
207    entries(root, &commit).and_then(|files| {
208        let ids: Vec<&str> = files.values().map(String::as_str).collect();
209        blobs(root, &ids).map(|bytes| {
210            files
211                .iter()
212                .map(|(path, id)| (path.clone(), bytes.get(id).cloned().unwrap_or_default()))
213                .collect()
214        })
215    })
216}

218/// Two stores held different bytes under one name. Records are content-addressed, so this
219/// is corruption or tampering on one side; the local copy is kept and the name reported.
220#[derive(Debug, Clone, PartialEq, Eq)]
221pub struct Collision {
222    /// The path in the tree.
223    pub path: String,
224}

226/// The tree of a store holding `files` (path → blob id), written into the object database.
227fn tree_of(root: &Path, files: &BTreeMap<String, String>) -> Result<String, String> {
228    // `mktree` builds one level; records live one level down, so build `records/` first.
229    let mut nested: BTreeMap<&str, Vec<(&str, &str)>> = BTreeMap::new();
230    let mut top: Vec<(String, String, String)> = Vec::new();
231    for (path, blob) in files {
232        match path.split_once('/') {
233            Some((dir, name)) if !name.contains('/') => {
234                nested.entry(dir).or_default().push((name, blob));
235            }
236            None => top.push(("blob".into(), blob.clone(), path.clone())),
237            Some(_) => return Err(format!("{path}: the store holds one level of directories")),
238        }
239    }
240    for (dir, children) in nested {
241        let mut listing = String::new();
242        for (name, blob) in children {
243            listing.push_str(&format!("100644 blob {blob}\t{name}\0"));
244        }
245        let tree = run_with_input(root, &["mktree", "-z"], listing.as_bytes())?;
246        top.push(("tree".into(), tree, dir.to_string()));
247    }
248    let mut listing = String::new();
249    for (kind, object, name) in top {
250        let mode = if kind == "tree" { "040000" } else { "100644" };
251        listing.push_str(&format!("{mode} {kind} {object}\t{name}\0"));
252    }
253    run_with_input(root, &["mktree", "-z"], listing.as_bytes())
254}

256fn commit(root: &Path, tree: &str, parents: &[&str], message: &str) -> Result<String, String> {
257    let mut args = vec!["commit-tree", tree];
258    for p in parents {
259        args.push("-p");
260        args.push(p);
261    }
262    // The committer is whoever git says it is on this machine; a fixture with no identity
263    // gets a neutral one rather than a refusal, since this commit carries no authored work.
264    let neutral: &[(&str, &str)] = if run(root, &["config", "user.email"]).is_err() {
265        &[
266            ("GIT_AUTHOR_NAME", "majordomus"),
267            ("GIT_AUTHOR_EMAIL", "majordomus@localhost"),
268            ("GIT_COMMITTER_NAME", "majordomus"),
269            ("GIT_COMMITTER_EMAIL", "majordomus@localhost"),
270        ]
271    } else {
272        &[]
273    };
274    exec(root, &args, neutral, message.as_bytes())
275        .map(|out| String::from_utf8_lossy(&out).trim().to_string())
276}

278/// Move `reference` from `old` to `new`, or fail if another writer moved it first.
279fn update(root: &Path, reference: &str, new: &str, old: Option<&str>) -> Result<(), String> {
280    // the zero id says "create; it must not exist yet"
281    let zero = "0".repeat(new.len());
282    let old = old.unwrap_or(&zero);
283    run(
284        root,
285        &[
286            "update-ref",
287            "-m",
288            "majordomus continuity",
289            reference,
290            new,
291            old,
292        ],
293    )
294    .map(|_| ())
295}

297/// Write `files` as the store's tree, commit it on `parents`, and move the local ref from
298/// `old` to that commit — or fail, if another writer moved the ref first.
299fn write_tip(
300    root: &Path,
301    files: &BTreeMap<String, String>,
302    parents: &[&str],
303    message: &str,
304    old: Option<&str>,
305) -> Result<String, String> {
306    let tree = tree_of(root, files)?;
307    let new = commit(root, &tree, parents, message)?;
308    update(root, REF, &new, old)?;
309    Ok(new)
310}

312/// Add records to the local store: `records` is id → stored bytes. Records already present
313/// are left as they are. Returns the new tip, or the old one when nothing was added.
314pub fn add(root: &Path, records: &[(String, Vec<u8>)], message: &str) -> Result<String, String> {
315    let old = tip(root, REF);
316    let mut files = match &old {
317        Some(c) => entries(root, c)?,
318        None => BTreeMap::new(),
319    };
320    let fresh: Vec<&(String, Vec<u8>)> = records
321        .iter()
322        .filter(|(id, _)| !files.contains_key(&path_of(id)))
323        .collect();
324    if let (Some(old), true) = (&old, fresh.is_empty()) {
325        return Ok(old.clone());
326    }
327    for (id, bytes) in fresh {
328        let blob = run_with_input(root, &["hash-object", "-w", "--stdin"], bytes)?;
329        files.insert(path_of(id), blob);
330    }
331    let parents: Vec<&str> = old.iter().map(String::as_str).collect();
332    write_tip(root, &files, &parents, message, old.as_deref())
333}

335/// Merge the store at `other` (a commit) into the local ref: the local files and those of
336/// `other`'s files that `keep` admits, as a commit with both as parents. Returns the new tip
337/// and the names whose bytes differed. When nothing is left out and one side already
338/// contains the other, the ref fast-forwards or stays.
339///
340/// `keep` is how a record that failed admission — forged, from another repository — stays
341/// out of this store's tree: the remote's commit is still a parent, so the next sync with
342/// that remote is a fast-forward rather than a second merge, but no reader ever sees what
343/// was refused.
344pub fn merge(
345    root: &Path,
346    other: &str,
347    keep: &dyn Fn(&str) -> bool,
348) -> Result<(String, Vec<Collision>), String> {
349    let old = tip(root, REF);
350    let theirs = entries(root, other)?;
351    if let Some(local) = old
352        .as_deref()
353        .filter(|l| *l == other || is_ancestor(root, other, l))
354    {
355        return Ok((local.to_string(), Vec::new()));
356    }
357    let filtered = theirs.keys().any(|p| !keep(p));
358    let fast_forward = old.as_deref().is_none_or(|l| is_ancestor(root, l, other));
359    if fast_forward && !filtered {
360        update(root, REF, other, old.as_deref())?;
361        return Ok((other.to_string(), Vec::new()));
362    }
363    let mut files = match &old {
364        Some(local) => entries(root, local)?,
365        None => BTreeMap::new(),
366    };
367    let mut collisions = Vec::new();
368    for (path, blob) in theirs.into_iter().filter(|(p, _)| keep(p)) {
369        match files.get(&path) {
370            Some(mine) if *mine != blob => collisions.push(Collision { path }),
371            Some(_) => {}
372            None => {
373                files.insert(path, blob);
374            }
375        }
376    }
377    let parents: Vec<&str> = old.iter().map(String::as_str).chain([other]).collect();
378    let new = write_tip(
379        root,
380        &files,
381        &parents,
382        "majordomus continuity: merge\n",
383        old.as_deref(),
384    )?;
385    Ok((new, collisions))
386}

399/// What asking a remote produced.
400#[derive(Debug, Clone, PartialEq, Eq)]
401pub enum Fetched {
402    /// The remote holds a store; it is now at [`remote_ref`].
403    Store(String),
404    /// The remote answered and holds no store yet.
405    Empty,
406}
407
408/// Fetch `remote`'s store into [`remote_ref`]. The only network operation besides
409/// [`push`], and only `sync` calls either.
410pub fn fetch(root: &Path, remote: &str) -> Result<Fetched, String> {
411    if !valid_remote(remote) {
412        return Err(format!("`{remote}` is not a remote name"));
413    }
414    let listed = run(root, &["ls-remote", "--refs", remote, REF])?;
415    if listed.is_empty() {
416        return Ok(Fetched::Empty);
417    }
418    let target = remote_ref(remote);
419    run(
420        root,
421        &[
422            "fetch",
423            "--quiet",
424            "--no-tags",
425            "--no-write-fetch-head",
426            remote,
427            &format!("+{REF}:{target}"),
428        ],
429    )?;
430    tip(root, &target)
431        .map(Fetched::Store)
432        .ok_or_else(|| format!("the fetch of {remote} left no {target}"))
433}

435/// Push the local store to `remote`, never forced: a remote that moved since the fetch
436/// refuses, and the caller fetches, merges and pushes again.
437pub fn push(root: &Path, remote: &str) -> Result<(), String> {
438    if !valid_remote(remote) {
439        return Err(format!("`{remote}` is not a remote name"));
440    }
441    run(root, &["push", "--quiet", remote, &format!("{REF}:{REF}")])?;
442    // what the remote now holds, recorded so that the next status needs no network
443    let _ = run(root, &["update-ref", &remote_ref(remote), REF]);
444    Ok(())
445}

447/// The remote `sync` uses when none is named: the current branch's, else `origin` when it
448/// exists, else none.
449pub fn default_remote(root: &Path) -> Option<String> {
450    let branch = run(root, &["symbolic-ref", "--quiet", "--short", "HEAD"]).ok();
451    if let Some(branch) = branch.filter(|b| !b.is_empty()) {
452        if let Ok(remote) = run(root, &["config", &format!("branch.{branch}.remote")]) {
453            if valid_remote(&remote) && remote != "." {
454                return Some(remote);
455            }
456        }
457    }
458    let remotes = run(root, &["remote"]).unwrap_or_default();
459    remotes
460        .lines()
461        .find(|r| *r == "origin")
462        .map(str::to_string)
463        .or_else(|| remotes.lines().next().map(str::to_string))
464        .filter(|r| valid_remote(r))
465}

473    /// A commit holding the tree `mktree -z` builds from `lines`, each `<mode> <type>
474    /// <object>\t<name>`.
475    pub(crate) fn planted(root: &Path, lines: &[String]) -> String {
476        let listing: String = lines.iter().map(|l| format!("{l}\0")).collect();
477        let tree = run_with_input(root, &["mktree", "-z"], listing.as_bytes()).unwrap();
478        commit(root, &tree, &[], "planted\n").unwrap()
479    }

481    pub(crate) fn blob(root: &Path, bytes: &[u8]) -> String {
482        run_with_input(root, &["hash-object", "-w", "--stdin"], bytes).unwrap()
483    }
484
485    /// A store of `count` files that are no records: that many names for one blob.
486    pub(crate) fn filled(root: &Path, count: usize) -> String {
487        let one = blob(root, b"x\n");
488        let listing: String = (0..count)
489            .map(|i| format!("100644 blob {one}\t{i:032x}.json\0"))
490            .collect();
491        let tree = run_with_input(root, &["mktree", "-z"], listing.as_bytes()).unwrap();
492        planted(root, &[format!("040000 tree {tree}\trecords")])
493    }
494
495    /// A store one file over the bound.
496    pub(crate) fn oversized(root: &Path) -> String {
497        filled(root, MAX_RECORDS + 1)
498    }
499
500    /// A store whose tree is deeper than a store is: `a/b/c`.
501    pub(crate) fn deep(root: &Path) -> String {
502        let one = blob(root, b"x\n");
503        let tree = |line: String| {
504            run_with_input(root, &["mktree", "-z"], format!("{line}\0").as_bytes()).unwrap()
505        };
506        let sub = tree(format!("100644 blob {one}\tc"));
507        let mid = tree(format!("040000 tree {sub}\tb"));
508        planted(root, &[format!("040000 tree {mid}\ta")])
509    }
510
511    /// Hold the lock git takes on `reference`, as a writer in the middle of moving it does.
512    pub(crate) fn locked(root: &Path, reference: &str) -> std::path::PathBuf {
513        let lock = root.join(".git").join(format!("{reference}.lock"));
514        std::fs::create_dir_all(lock.parent().unwrap()).unwrap();
515        std::fs::write(&lock, "").unwrap();
516        lock
517    }

586            files.get(&path_of(&"b".repeat(32))).unwrap(),

589        );
590        assert_eq!(
591            collisions,
592            vec![Collision {
593                path: path_of(&"b".repeat(32))
594            }]
595        );
596        // merging what is already contained is a no-op
597        assert_eq!(merge(root, &right, &|_| true).unwrap().0, merged);

797    fn a_remote_is_asked_by_name_and_answers_with_a_store_or_none() {
798        let dir = tempfile::tempdir().unwrap();
799        let root = dir.path().join("here");
800        let remote = dir.path().join("remote.git");
801        std::fs::create_dir_all(&root).unwrap();
802        run(
803            dir.path(),
804            &["init", "-q", "--bare", remote.to_str().unwrap()],
805        )
806        .unwrap();
807        run(&root, &["init", "-q", "-b", "main"]).unwrap();
808
809        // no remote at all, then one that is not `origin`, then both
810        assert_eq!(default_remote(&root), None);
811        assert!(fetch(&root, "a:b")
812            .unwrap_err()
813            .contains("is not a remote name"));
814        assert!(push(&root, "a:b")
815            .unwrap_err()
816            .contains("is not a remote name"));
817        run(
818            &root,
819            &["remote", "add", "upstream", remote.to_str().unwrap()],
820        )
821        .unwrap();
822        assert_eq!(default_remote(&root).as_deref(), Some("upstream"));
823        run(
824            &root,
825            &["remote", "add", "origin", remote.to_str().unwrap()],
826        )
827        .unwrap();
828        assert_eq!(default_remote(&root).as_deref(), Some("origin"));
829        // the branch's own remote wins; `.` (a local upstream) is not a remote
830        run(&root, &["config", "branch.main.remote", "upstream"]).unwrap();
831        assert_eq!(default_remote(&root).as_deref(), Some("upstream"));
832        run(&root, &["config", "branch.main.remote", "."]).unwrap();
833        assert_eq!(default_remote(&root).as_deref(), Some("origin"));
834
835        assert_eq!(fetch(&root, "origin").unwrap(), Fetched::Empty);
836        let first = add(&root, &[("a".repeat(32), b"A\n".to_vec())], "a\n").unwrap();
837        push(&root, "origin").unwrap();
838        assert_eq!(tip(&root, &remote_ref("origin")), Some(first.clone()));
839        assert_eq!(fetch(&root, "origin").unwrap(), Fetched::Store(first));
840
841        // the tracking ref is held by another writer: the fetch is refused, not half done
842        run(&root, &["update-ref", "-d", &remote_ref("origin")]).unwrap();
843        let lock = locked(&root, &remote_ref("origin"));
844        assert!(fetch(&root, "origin").is_err());
845        std::fs::remove_file(&lock).unwrap();
846
847        // a remote whose ref is not a commit holds no store this one can read
848        let stray = run_with_input(&remote, &["hash-object", "-w", "--stdin"], b"x\n").unwrap();
849        run(&remote, &["update-ref", REF, &stray]).unwrap();
850        let err = fetch(&root, "origin").unwrap_err();
851        assert!(err.contains("left no"), "{err}");
852
853        // detached: no branch to ask, so the remotes decide
854        let tree = tree_of(&root, &BTreeMap::new()).unwrap();
855        let head = commit(&root, &tree, &[], "base\n").unwrap();
856        run(&root, &["checkout", "-q", "--detach", &head]).unwrap();
