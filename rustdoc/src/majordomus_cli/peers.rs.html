1//! Peers: the clients attached to one shared server. Every MCP session, the owner's
2//! stdio one and every one that arrived over HTTP, is a peer from the moment it attaches;
3//! `initialize` gives it a name (the client's own `clientInfo`), and `peers.announce`
4//! lets it say what it is working on. The board lives in the server's memory and nowhere
5//! else: it ends with the process, and nothing here touches the repository.
6//!
7//! # Why it is in memory and stays there
8//!
9//! A peer board is about *now*: who is attached, what they said they are doing. Writing it
10//! into the repository would make it a record of something that has already stopped being
11//! true, and the layer's records are for what survives the session. The board ends with the
12//! process, which is exactly right for a fact about the process.
13//!
14//! # What a peer may say
15//!
16//! An announcement is informational and enforces nothing. Its whole purpose is that another
17//! agent working in the same checkout can read it and choose not to collide — the
18//! coordination is between the peers, not imposed by the server.
19//!
20//! A peer may hold **several claims at once**, each under a name of its own. One connection
21//! is not always one piece of work: a session that fans work out to subagents shares its MCP
22//! session with all of them, so they are one peer here while being several workers there.
23//! Before named claims the board could hold only the last thing said, and each announcement
24//! silently erased the one before it — the other peers stopped being told about scope that
25//! was still very much being written. Announcing under the same name updates that claim;
26//! announcing under a new one leaves the others standing; announcing without a name is the
27//! peer's one unnamed claim, which is all a single session ever needs.
28//!
29//! ```
30//! use majordomus_cli::peers::{ClientInfo, PeerBoard, Transport};
31//!
32//! let board = PeerBoard::new();
33//! let id = board.attach(Transport::Stdio);
34//! board.identify(&id, ClientInfo::unknown());
35//!
36//! let announced = board
37//!     .announce(&id, "landing the quality gate", vec!["apps/majordomus-cli".into()])
38//!     .expect("the peer is attached");
39//! assert_eq!(
40//!     announced.peer.announcement.as_ref().map(|a| a.intent.as_str()),
41//!     Some("landing the quality gate")
42//! );
43//! assert_eq!(board.list().len(), 1);
44//!
45//! // a session that ends is no longer counted; what it announced stays readable, so the
46//! // next worker can still see the claim and avoid it
47//! board.detach(&id);
48//! assert_eq!(board.len(), 0);
49//! assert!(!board.list()[0].attached);
50//! ```

91/// What a client said about itself in `initialize`.
92#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
93pub struct ClientInfo {
94    /// `clientInfo.name`: `claude-code`, `codex`, `gemini-cli`, whatever the client sends.
95    pub name: String,
96    /// `clientInfo.version`.
97    pub version: String,
98    // `default` as well as `skip_serializing_if`: a board is now read back off the wire
99    // from a sibling checkout's server, and a field that is skipped when absent must also
100    // be accepted when absent, or the peer that omitted it makes the whole board unreadable.
101    #[serde(default, skip_serializing_if = "Option::is_none")]
102    /// `clientInfo.title`, when the client sends one.
103    pub title: Option<String>,
104}

106impl ClientInfo {
107    /// A session that has attached and not yet initialised.
108    pub fn unknown() -> Self {
109        ClientInfo {
110            name: "(not initialized)".into(),
111            version: String::new(),
112            title: None,
113        }
114    }
115
116    /// Read `clientInfo` out of `initialize` params; `unknown` when it is missing.
117    pub fn from_initialize(params: &serde_json::Value) -> Self {
118        let info = &params["clientInfo"];
119        let name = info["name"].as_str().unwrap_or("").trim();
120        if name.is_empty() {
121            return Self::unknown();
122        }
123        ClientInfo {
124            name: name.to_string(),
125            version: info["version"].as_str().unwrap_or("").to_string(),
126            title: info["title"].as_str().map(str::to_string),
127        }
128    }

154/// One claim a peer has made: what it is working on, and where.
155///
156/// A peer may hold several. The `name` is what tells them apart and what makes announcing
157/// again an update rather than a second claim: announce under the same name and the claim
158/// is replaced, announce under a new one and the peer now holds both. A peer that never
159/// names anything holds exactly one claim, the unnamed one, which is what a single session
160/// announcing about itself has always had.
161#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
162pub struct Announcement {
163    /// What this claim is called, when the peer named it. Absent for a peer's unnamed claim.
164    #[serde(default, skip_serializing_if = "Option::is_none")]
165    pub name: Option<String>,
166    /// One line: the task or intent, in the peer's words.
167    pub intent: String,
168    /// Repository-relative paths the peer expects to touch; informational, never enforced here.
169    pub scope: Vec<String>,
170    /// When it was announced, RFC 3339, UTC.
171    pub at: String,
172}

174/// The key a claim is held under: its name, or the empty string for the unnamed claim.
175///
176/// Private, because the emptiness is an implementation of "unnamed" and not something a
177/// reader of the board should ever see: [`Announcement::name`] is `None` there.
178fn claim_key(name: Option<&str>) -> String {
179    name.map(str::trim)
180        .filter(|n| !n.is_empty())
181        .unwrap_or("")
182        .to_string()
183}

185/// Which checkout of the repository a peer is attached to.
186///
187/// A board is one process's memory and a server serves one checkout, so a peer read from
188/// a sibling checkout's server and a peer of this one are otherwise indistinguishable —
189/// and worse than indistinguishable, because the ids collide: `p1` is the first session
190/// of *every* board, so a repository-wide listing without this field would hold several
191/// peers called `p1` and no way to tell them apart. This is the qualification. The
192/// worktree path is the durable half of a worker's identity: the peer id is a position on
193/// one board and is handed out again after a reconnect (`p3` this morning, `p1` after the
194/// bridge re-attached), while the checkout a worker is working in outlives its connection.
195///
196/// Absent on a peer a server reports about its own board without being asked which
197/// checkout that is — `peers.announce` answers about the announcing session and the
198/// answer is this server's by construction. The repository-wide listing stamps every
199/// peer it gathers, its own included, so that no entry of that answer is unqualified.
200///
201/// ```
202/// use majordomus_cli::peers::PeerCheckout;
203/// let c = PeerCheckout {
204///     id: "b8293f11".into(),
205///     worktree: "/repo-wt/feature/x".into(),
206///     branch: Some("feature/x".into()),
207///     this_checkout: false,
208/// };
209/// let v = serde_json::to_value(&c).unwrap();
210/// assert_eq!(v["branch"], "feature/x");
211/// assert_eq!(v["this_checkout"], false);
212/// ```
213#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
214pub struct PeerCheckout {
215    /// The checkout's identity digest, the same value `server.status` reports as
216    /// `checkout_id`. One per checkout, never a path, and stable across a restart.
217    pub id: String,
218    /// Where the checkout is. The one thing on this board that a human can act on
219    /// directly, and the one thing that survives a reconnect.
220    pub worktree: PathBuf,
221    /// The branch checked out there, when it could be read.
222    #[serde(default, skip_serializing_if = "Option::is_none")]
223    pub branch: Option<String>,
224    /// Whether this is the checkout of the server answering the call.
225    pub this_checkout: bool,
226}

228/// One peer as the board lists it.
229#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
230pub struct Peer {
231    /// `p1`, `p2`, ...
232    pub id: PeerId,
233    /// The client behind it.
234    pub client: ClientInfo,
235    /// How it is attached.
236    pub transport: Transport,
237    /// When it attached, RFC 3339, UTC.
238    pub connected_at: String,
239    /// Seconds since its last message.
240    pub last_seen_seconds_ago: u64,
241    /// Whether the session is still attached. A peer that announced something and then
242    /// went away is kept and listed with `attached: false`: what it said it was working on
243    /// outlives the connection that said it, because the work does.
244    pub attached: bool,
245    /// Every claim this peer holds, in name order, the unnamed one first. Empty for a peer
246    /// that has announced nothing.
247    ///
248    /// One connection may be doing several things at once — a session that fans work out to
249    /// subagents shares its MCP session with all of them — and before this field the board
250    /// could hold only the last thing said, so each announcement silently erased the one
251    /// before it and the other peers stopped seeing the rest of the scope.
252    #[serde(default, skip_serializing_if = "Vec::is_empty")]
253    pub claims: Vec<Announcement>,
254    // `default` as well as `skip_serializing_if`, for the reason on `ClientInfo::title`:
255    // a peer that has attached and not yet announced serializes without this field, and
256    // before the board was gathered off the wire nothing ever read one back.
257    #[serde(default, skip_serializing_if = "Option::is_none")]
258    /// The most recently made of [`Peer::claims`], for a reader that wants one line rather
259    /// than all of them. Derived from `claims` and never a second account of it: a peer
260    /// holding one claim reports the same thing in both, which is what every peer that does
261    /// not name its claims reports.
262    pub announcement: Option<Announcement>,
263    /// Which checkout of the repository this peer is attached to; see [`PeerCheckout`].
264    ///
265    /// The board itself never sets it — a board does not know where it is — and it is
266    /// stamped by the reader that gathered this peer, which is the one thing that knows
267    /// which checkout's server it asked.
268    #[serde(default, skip_serializing_if = "Option::is_none")]
269    pub checkout: Option<PeerCheckout>,
270    /// Where the worker behind this peer can be reached from outside its connection: its
271    /// machine, its process, its provider session and the socket other sessions message it
272    /// on; see [`PeerAddress`]. Derived when the peer attached, never announced by hand.
273    /// Absent for a client that exposes none of it.
274    #[serde(default, skip_serializing_if = "Option::is_none")]
275    pub address: Option<PeerAddress>,
276}

333/// Do two claimed paths meet? Equal, or one inside the other.
334///
335/// Repository-relative, `/`-separated, trailing slashes ignored. `apps` contains
336/// `apps/majordomus-cli`; `app` does not, because a claim is a path and not a prefix of a
337/// string.
338///
339/// ```
340/// use majordomus_cli::peers::claims_meet;
341/// assert!(claims_meet("apps", "apps/majordomus-cli"));
342/// assert!(claims_meet("apps/majordomus-cli", "apps"));
343/// assert!(claims_meet("docs/", "docs"));
344/// assert!(!claims_meet("app", "apps/majordomus-cli"));
345/// assert!(!claims_meet("docs", "site"));
346/// ```
347pub fn claims_meet(a: &str, b: &str) -> bool {
348    let (a, b) = (
349        a.trim().trim_end_matches('/'),
350        b.trim().trim_end_matches('/'),
351    );
352    if a.is_empty() || b.is_empty() {
353        return false;
354    }
355    a == b || b.starts_with(&format!("{a}/")) || a.starts_with(&format!("{b}/"))
356}

360/// Where the worker behind a peer can be reached from outside its MCP connection.
361///
362/// A board row named a client and a version, and nothing a person or another session could
363/// act on: on 2026-10-22 the session running a 33-minute remote `cargo test --doc` could be
364/// found only by walking an ssh client's parent chain to a provider pid, its messaging
365/// socket, and a start time matched by hand. Every one of those facts is in the environment
366/// the provider hands the processes it starts — the MCP server among them — so the address
367/// is read from there when the peer attaches, and never asked of the model.
368///
369/// The provider's messaging *token* is a credential and is never read: the address is where
370/// to knock, not the key.
371///
372/// ```
373/// use majordomus_cli::peers::PeerAddress;
374/// // a client that exposes nothing still has a machine and a parent process
375/// let bare = PeerAddress::from_env(|_| None, Some(4242), "lundra").unwrap();
376/// assert_eq!((bare.host.as_str(), bare.pid), ("lundra", Some(4242)));
377/// assert!(bare.provider.is_none() && bare.mesh_session().is_none());
378/// ```
379#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
380pub struct PeerAddress {
381    /// The provider whose session this is (`claude-code`), when its environment says so.
382    #[serde(default, skip_serializing_if = "Option::is_none")]
383    pub provider: Option<String>,
384    /// The machine the worker runs on.
385    pub host: String,
386    /// The worker's own process: the provider's pid, not the MCP bridge it started.
387    #[serde(default, skip_serializing_if = "Option::is_none")]
388    pub pid: Option<u32>,
389    /// The provider's id for the session, the one its lifecycle hooks are given.
390    #[serde(default, skip_serializing_if = "Option::is_none")]
391    pub session: Option<String>,
392    /// The socket other sessions on the same machine message this one through.
393    #[serde(default, skip_serializing_if = "Option::is_none")]
394    pub socket: Option<String>,
395}

397/// The `_meta` key a bridge carries the address under in the `initialize` it forwards.
398pub(crate) const ADDRESS_META: &str = "dev.majordomus/address";
399
400impl PeerAddress {
401    /// The address an environment describes. `env` answers a variable by name; `parent` is
402    /// the pid of the process that started this one, used when the provider names no pid of
403    /// its own; `host` is the machine. `None` only when there is no host to name.
404    ///
405    /// ```
406    /// use majordomus_cli::peers::PeerAddress;
407    /// let env = |k: &str| match k {
408    ///     "CLAUDE_CODE_SESSION_ID" => Some("s-1".to_string()),
409    ///     "CLAUDE_CODE_MESSAGING_TOKEN" => Some("secret".to_string()),
410    ///     _ => None,
411    /// };
412    /// let a = PeerAddress::from_env(env, Some(7), "mac").unwrap();
413    /// assert_eq!(a.provider.as_deref(), Some("claude-code"));
414    /// assert_eq!(a.pid, Some(7), "no provider pid: the parent is the worker");
415    /// assert!(!serde_json::to_string(&a).unwrap().contains("secret"));
416    /// ```
417    pub fn from_env(
418        env: impl Fn(&str) -> Option<String>,
419        parent: Option<u32>,
420        host: &str,
421    ) -> Option<PeerAddress> {
422        let host = host.trim();
423        if host.is_empty() {
424            return None;
425        }
426        let read = |k: &str| {
427            env(k)
428                .map(|v| v.trim().to_string())
429                .filter(|v| !v.is_empty())
430        };
431        let session = read("CLAUDE_CODE_SESSION_ID");
432        let socket = read("CLAUDE_CODE_MESSAGING_SOCKET");
433        let claude = session.is_some() || socket.is_some() || read("CLAUDECODE").is_some();
434        let pid = read("CLAUDE_PID")
435            .and_then(|p| p.parse::<u32>().ok())
436            .filter(|p| *p > 1)
437            .or(parent.filter(|p| *p > 1));
438        Some(PeerAddress {
439            provider: claude.then(|| "claude-code".to_string()),
440            host: host.to_string(),
441            pid,
442            session,
443            socket,
444        })
445    }
446
447    /// The address of the worker that started this process: its environment, its parent
448    /// and this machine.
449    pub(crate) fn of_this_process() -> Option<PeerAddress> {
450        Self::from_env(
451            |k| std::env::var(k).ok(),
452            Some(std::os::unix::process::parent_id()),
453            &crate::integration::drain::host(),
454        )
455    }
456
457    /// The address a bridge carried in `initialize` params, under
458    /// `_meta["dev.majordomus/address"]`.
459    ///
460    /// ```
461    /// use majordomus_cli::peers::PeerAddress;
462    /// let params = serde_json::json!({ "_meta": { "dev.majordomus/address": { "host": "mac", "pid": 9 } } });
463    /// assert_eq!(PeerAddress::from_initialize(&params).unwrap().pid, Some(9));
464    /// assert!(PeerAddress::from_initialize(&serde_json::json!({})).is_none());
465    /// ```
466    pub fn from_initialize(params: &serde_json::Value) -> Option<PeerAddress> {
467        serde_json::from_value(params.get("_meta")?.get(ADDRESS_META)?.clone()).ok()
468    }
469
470    /// The id this provider session has on the mesh, the one the edit guard judges its
471    /// claims under (`<provider>-<session>`): what joins a board row to a mesh session.
472    /// `None` unless both the provider and its session id are known.
473    ///
474    /// ```
475    /// use majordomus_cli::peers::PeerAddress;
476    /// let env = |k: &str| (k == "CLAUDE_CODE_SESSION_ID").then(|| "s1".to_string());
477    /// let a = PeerAddress::from_env(env, Some(7), "mac").unwrap();
478    /// assert_eq!(a.mesh_session().as_deref(), Some("claude-code-s1"));
479    /// let bare = PeerAddress::from_env(|_| None, Some(7), "mac").unwrap();
480    /// assert_eq!(bare.mesh_session(), None);
481    /// ```
482    pub fn mesh_session(&self) -> Option<String> {
483        Some(format!(
484            "{}-{}",
485            self.provider.as_ref()?,
486            self.session.as_ref()?
487        ))
488    }
489
490    /// One line a person can act on: `claude-code pid 85501 on mac, /tmp/cc-socks/85501.sock`.
491    ///
492    /// ```
493    /// use majordomus_cli::peers::PeerAddress;
494    /// let a = PeerAddress::from_env(|_| None, Some(12), "lundra").unwrap();
495    /// assert_eq!(a.line(), "pid 12 on lundra");
496    /// ```
497    pub fn line(&self) -> String {
498        let mut out = String::new();
499        if let Some(p) = &self.provider {
500            out.push_str(p);
501            out.push(' ');
502        }
503        if let Some(pid) = self.pid {
504            out.push_str(&format!("pid {pid} "));
505        }
506        out.push_str(&format!("on {}", self.host));
507        if let Some(socket) = &self.socket {
508            out.push_str(&format!(", {socket}"));
509        }
510        out
511    }
512}

514/// Put the address of the worker that started this process into an `initialize` message's
515/// params, under `_meta["dev.majordomus/address"]`, unless the client already put one there.
516///
517/// ```
518/// use majordomus_cli::peers::{with_address, PeerAddress};
519/// let a = PeerAddress::from_env(|_| None, Some(3), "mac").unwrap();
520/// let mut init = serde_json::json!({ "method": "initialize", "params": { "clientInfo": { "name": "x" } } });
521/// with_address(&mut init, &a);
522/// assert_eq!(PeerAddress::from_initialize(&init["params"]), Some(a));
523/// ```
524pub fn with_address(message: &mut serde_json::Value, address: &PeerAddress) {
525    let Some(params) = message.get_mut("params").and_then(|p| p.as_object_mut()) else {
526        return;
527    };
528    let meta = params
529        .entry("_meta")
530        .or_insert_with(|| serde_json::Value::Object(Default::default()));
531    // an address always serialises (string keys, plain values); a `_meta` that is not an
532    // object is left as the client sent it
533    if let (Some(meta), Ok(v)) = (meta.as_object_mut(), serde_json::to_value(address)) {
534        meta.entry(ADDRESS_META).or_insert(v);
535    }
536}

713    /// assert_eq!(named.peer.claims[1].name.as_deref(), Some("worker-a"));
714    /// ```

716        &self,
717        id: &PeerId,
718        claim: Option<&str>,
719        intent: &str,
720        scope: Vec<String>,
721    ) -> Option<Announced> {
722        let announced = self.announce_locked(id, claim, intent, scope);
723        if announced.is_some() {
724            crate::change::BLACKBOARD.advance();
725        }

729    fn announce_locked(
730        &self,
731        id: &PeerId,
732        claim: Option<&str>,
733        intent: &str,
734        scope: Vec<String>,
735    ) -> Option<Announced> {
736        let mut slots = self.lock();
737        if !slots.contains_key(&id.seq()) {
738            return None;
739        }
740        let overlaps = overlaps_against(&slots, id.seq(), &scope);
741        let key = claim_key(claim);
742        let s = slots.get_mut(&id.seq())?;
743        s.last_seen = Instant::now();
744        s.departed = false;
745        let released = s
746            .claims
747            .get(&key)
748            .map(|old| {
749                old.scope
750                    .iter()
751                    .filter(|held| !scope.iter().any(|now| claims_meet(held, now)))
752                    .cloned()
753                    .collect()
754            })
755            .unwrap_or_default();
756        s.claims.insert(
757            key.clone(),
758            Announcement {
759                name: (!key.is_empty()).then(|| key.clone()),
760                intent: intent.to_string(),
761                scope,
762                at: rfc3339(SystemTime::now()),
763            },
764        );
765        Some(Announced {
766            peer: peer(id.seq(), s),
767            overlaps,
768            released,
769        })
770    }

789    /// A peer that announced something is kept and marked departed rather than removed:

791    /// not stop being true because a socket closed. A peer that never announced leaves

804    /// assert_eq!(listed.len(), 1, "the one that said something is still on the board");
805    /// assert!(!listed[0].attached);
806    /// assert_eq!(listed[0].announcement.as_ref().unwrap().intent, "the ordering rule");
807    /// ```

813    fn detach_locked(&self, id: &PeerId) {
814        let mut slots = self.lock();
815        match slots.get_mut(&id.seq()) {
816            Some(s) if !s.claims.is_empty() => s.departed = true,
817            _ => {
818                slots.remove(&id.seq());
819                return;
820            }
821        }
822        let departed: Vec<u64> = slots
823            .iter()
824            .filter(|(_, s)| s.departed)
825            .map(|(seq, _)| *seq)
826            .collect();
827        for seq in departed
828            .iter()
829            .take(departed.len().saturating_sub(RETAINED))
830        {
831            slots.remove(seq);
832        }
833    }

852    /// The peers, one bounded line, for a log or the `initialize` instructions.
853    ///
854    /// Bounded because a client cuts the instructions it is handed: on 2026-10-22 five
855    /// peers with twenty claims made this line about 6 kB, Claude Code cut the
856    /// instructions inside it, and the sentences after it — the ones about the caller
857    /// itself — never reached the caller. Each claim keeps its name and the first
858    /// [`SUMMARY_INTENT`] characters of its intent, each peer at most [`SUMMARY_CLAIMS`]
859    /// claims, and the line at most [`SUMMARY_LINE`] characters; what is past a bound is
860    /// counted, and `majordomus_peers` carries every word of it.
861    pub fn summary(&self) -> String {
862        use crate::environment::text::truncate;
863        let peers = self.list();
864        if peers.is_empty() {
865            return "none".into();
866        }
867        let mut line = String::new();
868        for (i, p) in peers.iter().enumerate() {
869            // a client and a claim name are the peer's own words too, and cut like them
870            let client = truncate(&p.client.name, SUMMARY_NAME);
871            let entry = if p.claims.is_empty() {
872                format!("{} {client} ({:?})", p.id, p.transport)
873            } else {
874                let mut claims = p
875                    .claims
876                    .iter()
877                    .take(SUMMARY_CLAIMS)
878                    .map(|a| {
879                        let intent = truncate(a.intent.trim(), SUMMARY_INTENT);
880                        match &a.name {
881                            Some(n) => format!("{}: {intent}", truncate(n, SUMMARY_NAME)),
882                            None => intent,
883                        }
884                    })
885                    .collect::<Vec<_>>();
886                if p.claims.len() > SUMMARY_CLAIMS {
887                    claims.push(format!("{} more", p.claims.len() - SUMMARY_CLAIMS));
888                }
889                format!(
890                    "{} {client} ({:?}: {})",
891                    p.id,
892                    p.transport,
893                    claims.join(" | ")
894                )
895            };
896            let sep = if line.is_empty() { "" } else { "; " };
897            if line.chars().count() + sep.len() + entry.chars().count() > SUMMARY_LINE {
898                line.push_str(&format!(
899                    "{sep}{} more peer(s), every claim in full: majordomus_peers",
900                    peers.len() - i
901                ));
902                break;
903            }
904            line.push_str(sep);
905            line.push_str(&entry);
906        }
907        line
908    }

932/// Every pair of claims that meet among an arbitrary set of peers, each pair once.
933///
934/// The listing side of the overlap question, and the only account of it: [`PeerBoard::overlaps`]
935/// is this function over its own peers, and the repository-wide listing is this function
936/// over the peers of every checkout's board. They cannot drift into two answers, and a
937/// collision between two worktrees is found by exactly the algorithm that found a
938/// collision between two sessions of one — which is the whole point, because until the
939/// board became repository-wide the second kind was the only kind anyone could see.
940///
941/// A pair is reported once, against the later peer of the two, naming the earlier one:
942/// `yours` is the later peer's path and `theirs` the earlier's, and [`Overlap::peer`]
943/// with [`Overlap::checkout`] says who and where the earlier one is. Peers are paired by
944/// position, never by [`PeerId`], because ids are unique to one board and a gathered list
945/// holds a `p1` per checkout.
946///
947/// ```
948/// use majordomus_cli::peers::{overlaps_among, PeerBoard, Transport};
949/// let board = PeerBoard::new();
950/// let a = board.attach(Transport::Stdio);
951/// let b = board.attach(Transport::Http);
952/// board.announce(&a, "the ordering rule", vec!["apps/majordomus-cli/src".into()]);
953/// board.announce(&b, "the release model", vec!["apps/majordomus-cli/src/release".into()]);
954/// let found = overlaps_among(&board.list());
955/// assert_eq!(found.len(), 1);
956/// assert_eq!(found[0].peer.as_str(), "p1", "reported against the later peer, naming the earlier");
957/// assert_eq!(found[0].paths[0].yours, "apps/majordomus-cli/src/release");
958/// assert_eq!(found[0].paths[0].theirs, "apps/majordomus-cli/src");
959/// // and it is the same answer the board gives about itself
960/// assert_eq!(board.overlaps(), found);
961/// ```
962pub fn overlaps_among(peers: &[Peer]) -> Vec<Overlap> {
963    let mut out = Vec::new();
964    for (i, later) in peers.iter().enumerate() {
965        for mine in &later.claims {
966            for earlier in peers.iter().take(i) {
967                for theirs in &earlier.claims {
968                    let mut paths: Vec<OverlapPath> = Vec::new();
969                    for yours in &mine.scope {
970                        for their in &theirs.scope {
971                            if claims_meet(yours, their) {
972                                paths.push(OverlapPath {
973                                    yours: yours.clone(),
974                                    theirs: their.clone(),
975                                });
976                            }
977                        }
978                    }
979                    if !paths.is_empty() {
980                        out.push(Overlap {
981                            peer: earlier.id.clone(),
982                            attached: earlier.attached,
983                            intent: theirs.intent.clone(),
984                            paths,
985                            checkout: earlier.checkout.clone(),
986                        });
987                    }
988                }
989            }
990        }
991    }
992    out
993}

995/// Every claim of every peer other than `seq` whose scope meets `scope`.
996///
997/// One entry per claim, not per peer: a connection holding two claims that both meet this
998/// scope is two entries, because they are two pieces of work and the reader needs the
999/// intent of each to know which one to talk about.
1000fn overlaps_against(slots: &BTreeMap<u64, Slot>, seq: u64, scope: &[String]) -> Vec<Overlap> {
1001    let mut out = Vec::new();
1002    for (other, s) in slots.iter() {
1003        if *other == seq {
1004            continue;
1005        }
1006        for theirs in s.claims.values() {
1007            let mut paths: Vec<OverlapPath> = Vec::new();
1008            for mine in scope {
1009                for path in &theirs.scope {
1010                    if claims_meet(mine, path) {
1011                        paths.push(OverlapPath {
1012                            yours: mine.clone(),
1013                            theirs: path.clone(),
1014                        });
1015                    }
1016                }
1017            }
1018            if !paths.is_empty() {
1019                out.push(Overlap {
1020                    peer: PeerId::new(*other),
1021                    attached: !s.departed,
1022                    intent: theirs.intent.clone(),
1023                    paths,
1024                    // this board's own peers: the caller is on this checkout too, and the
1025                    // one answer that gathers several boards stamps them all itself
1026                    checkout: None,
1027                });
1028            }
1029        }
1030    }
1031    out
1032}

1034/// Seconds since the epoch of `YYYY-MM-DDTHH:MM:SSZ`, the one shape [`rfc3339`] writes, and
1035/// its inverse. Anything else is `None`: a reader that guessed at other shapes would turn a
1036/// malformed time into a plausible one.
1037///
1038/// ```
1039/// use std::time::{Duration, UNIX_EPOCH};
1040/// use majordomus_cli::peers::{parse_rfc3339, rfc3339};
1041/// assert_eq!(parse_rfc3339("1970-01-01T00:00:00Z"), Some(0));
1042/// let t = UNIX_EPOCH + Duration::from_secs(1_788_000_000);
1043/// assert_eq!(parse_rfc3339(&rfc3339(t)), Some(1_788_000_000));
1044/// assert_eq!(parse_rfc3339("2026-08-29T10:40:00+02:00"), None);
1045/// ```
1046pub fn parse_rfc3339(s: &str) -> Option<u64> {
1047    let b = s.as_bytes();
1048    if b.len() != 20 || b[19] != b'Z' {
1049        return None;
1050    }
1051    let n = |r: std::ops::Range<usize>| s.get(r)?.parse::<i64>().ok();
1052    let (y, mo, d) = (n(0..4)?, n(5..7)?, n(8..10)?);
1053    let (h, mi, se) = (n(11..13)?, n(14..16)?, n(17..19)?);
1054    // days-from-civil, Howard Hinnant's algorithm: the inverse of the one rfc3339 uses
1055    let y = if mo <= 2 { y - 1 } else { y };
1056    let era = y.div_euclid(400);
1057    let yoe = y - era * 400;
1058    let doy = (153 * (if mo > 2 { mo - 3 } else { mo + 9 }) + 2) / 5 + d - 1;
1059    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
1060    let days = era * 146_097 + doe - 719_468;
1061    u64::try_from(days * 86_400 + h * 3600 + mi * 60 + se).ok()
1062}

1064/// A `SystemTime` as RFC 3339 in UTC, to the second (`2026-09-05T12:34:56Z`).
1065///
1066/// ```
1067/// use std::time::{Duration, UNIX_EPOCH};
1068/// use majordomus_cli::peers::rfc3339;
1069/// assert_eq!(rfc3339(UNIX_EPOCH), "1970-01-01T00:00:00Z");
1070/// assert_eq!(rfc3339(UNIX_EPOCH + Duration::from_secs(1_788_000_000)), "2026-08-29T10:40:00Z");
1071/// ```
1072pub fn rfc3339(t: SystemTime) -> String {
1073    let secs = t
1074        .duration_since(UNIX_EPOCH)
1075        .map(|d| d.as_secs())
1076        .unwrap_or(0);
1077    let days = secs / 86_400;
1078    let rem = secs % 86_400;
1079    let (h, m, s) = (rem / 3600, (rem % 3600) / 60, rem % 60);
1080    // civil-from-days, Howard Hinnant's algorithm
1081    let z = days as i64 + 719_468;
1082    let era = z.div_euclid(146_097);
1083    let doe = z.rem_euclid(146_097);
1084    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
1085    let y = yoe + era * 400;
1086    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
1087    let mp = (5 * doy + 2) / 153;
1088    let d = doy - (153 * mp + 2) / 5 + 1;
1089    let mth = if mp < 10 { mp + 3 } else { mp - 9 };
1090    let y = if mth <= 2 { y + 1 } else { y };
1091    format!("{y:04}-{mth:02}-{d:02}T{h:02}:{m:02}:{s:02}Z")
1092}

1094/// The inverse of [`rfc3339`]: `2026-08-29T10:40:00Z` back to Unix seconds.
1095///
1096/// It lives beside the formatter because the two are one thing, and a repository that had
1097/// the one without the other grew a second parser every time somebody needed to compare two
1098/// recorded instants. Strict about the shape it accepts — exactly `YYYY-MM-DDTHH:MM:SSZ`,
1099/// which is the only shape this tool writes — because a lenient parser that guessed at a
1100/// half-recognised string would answer `None` for a *malformed* timestamp and a plausible
1101/// number for a *different* one, and only the first of those is safe.
1102///
1103/// `None` is the honest answer for anything else, and the caller's job is to report it as
1104/// unjudgeable rather than as fresh: an episode whose evidence cannot be read as a time is
1105/// one a recovery sweep must skip and count, never one it may sweep.
1106///
1107/// ```
1108/// use majordomus_cli::peers::{epoch_seconds, rfc3339};
1109/// use std::time::{Duration, UNIX_EPOCH};
1110///
1111/// assert_eq!(epoch_seconds("1970-01-01T00:00:00Z"), Some(0));
1112/// assert_eq!(epoch_seconds("2026-08-29T10:40:00Z"), Some(1_788_000_000));
1113/// assert_eq!(epoch_seconds("not a timestamp"), None);
1114/// assert_eq!(epoch_seconds("2026-08-29 10:40:00Z"), None, "the T is not optional");
1115/// assert_eq!(epoch_seconds("2026-08-29T10:40:00+02:00"), None, "UTC or nothing");
1116///
1117/// // and it round-trips with the formatter it is the inverse of
1118/// for secs in [0u64, 1, 1_788_000_000, 2_000_000_000] {
1119///     let text = rfc3339(UNIX_EPOCH + Duration::from_secs(secs));
1120///     assert_eq!(epoch_seconds(&text), Some(secs as i64), "{text}");
1121/// }
1122/// ```
1123pub fn epoch_seconds(ts: &str) -> Option<i64> {
1124    let b = ts.as_bytes();
1125    if b.len() != 20
1126        || b[4] != b'-'
1127        || b[7] != b'-'
1128        || b[10] != b'T'
1129        || b[13] != b':'
1130        || b[16] != b':'
1131        || b[19] != b'Z'
1132    {
1133        return None;
1134    }
1135    let n = |from: usize, to: usize| -> Option<i64> {
1136        let part = ts.get(from..to)?;
1137        part.bytes().all(|c| c.is_ascii_digit()).then_some(())?;
1138        part.parse::<i64>().ok()
1139    };
1140    let (y, m, d) = (n(0, 4)?, n(5, 7)?, n(8, 10)?);
1141    let (hh, mm, ss) = (n(11, 13)?, n(14, 16)?, n(17, 19)?);
1142    if !(1..=12).contains(&m) || !(1..=31).contains(&d) || hh > 23 || mm > 59 || ss > 60 {
1143        return None;
1144    }
1145    // days-from-civil, Howard Hinnant's algorithm — the inverse of the civil-from-days
1146    // above, so the two agree by construction rather than by two tables being kept level.
1147    let y = if m <= 2 { y - 1 } else { y };
1148    let era = y.div_euclid(400);
1149    let yoe = y - era * 400;
1150    let mp = if m > 2 { m - 3 } else { m + 9 };
1151    let doy = (153 * mp + 2) / 5 + d - 1;
1152    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
1153    let days = era * 146_097 + doe - 719_468;
1154    Some(days * 86_400 + hh * 3600 + mm * 60 + ss)
1155}
