1//! The tool distribution's data directory: `share/kinds.yaml` and
2//! `share/schemas/<kind>.schema.json`, read at run time. Nothing about kinds or keys is
3//! compiled into the executable; the directory is located per invocation, explicitly or
4//! by convention, and named in every error when it is not.
5//!
6//! # Why nothing is compiled in
7//!
8//! A kind is a fact about a repository's layer, not about the executable that reads it. If
9//! the kinds were compiled in, adding one would mean shipping a binary, and a repository
10//! that added its own would be adding it to somebody else's program. Reading them at run
11//! time is what makes `.ai/repo/knowledge/` able to declare a kind that the same binary
12//! then serves.
13//!
14//! ```
15//! use majordomus_cli::share::Share;
16//!
17//! // located explicitly, and the paths under it are derived rather than joined by callers
18//! let dir = tempfile::tempdir().unwrap();
19//! std::fs::write(dir.path().join("kinds.yaml"), "version: 1\nkinds: []\n").unwrap();
20//! std::fs::create_dir(dir.path().join("schemas")).unwrap();
21//!
22//! let share = Share::locate(Some(dir.path()), dir.path()).unwrap();
23//! assert_eq!(share.origin, "--share", "it says how it was found");
24//! assert!(share.kinds_path().ends_with("kinds.yaml"));
25//! assert!(share.schemas_dir().ends_with("schemas"));
26//! assert_eq!(share.kinds_path().parent(), Some(share.dir()));
27//!
28//! // and when there is none, the error names where it looked
29//! let nowhere = tempfile::tempdir().unwrap();
30//! let err = Share::locate(None, nowhere.path()).unwrap_err();
31//! assert_eq!(err.exit_code(), 12);
32//! ```

68/// A located distribution directory.
69#[derive(Debug, Clone, PartialEq, Eq)]
70pub struct Share {
71    dir: PathBuf,
72    /// How it was found, for diagnostics: `--share`, the environment, the repository, or
73    /// the executable's own installation.
74    pub origin: &'static str,
75}
76
77impl Share {
78    /// Locate the share directory. In order: `explicit` (from `--share`), `MAJORDOMUS_SHARE`,
79    /// `<repository root>/share` when it holds a kinds file (this repository supervising
80    /// itself), and `<directory of the executable>/../share` (an installation laid out as
81    /// `bin/` beside `share/`). A candidate that exists but holds no kinds file is skipped
82    /// only when it came from a convention; an explicit path that lacks one is an error.
83    pub fn locate(explicit: Option<&Path>, repo_root: &Path) -> Result<Self> {
84        let mut tried = Vec::new();
85        if let Some(p) = explicit {
86            return Self::open(p, "--share");
87        }
88        if let Some(p) = std::env::var_os(SHARE_ENV) {
89            return Self::open(Path::new(&p), SHARE_ENV);
90        }
91        let in_repo = repo_root.join("share");
92        if in_repo.join(KINDS_FILE).is_file() {
93            return Self::open(&in_repo, "repository");
94        }
95        tried.push(in_repo);
96        if let Ok(exe) = std::env::current_exe() {
97            if let Some(dir) = exe.parent() {
98                let beside = dir.join("../share");
99                if beside.join(KINDS_FILE).is_file() {
100                    return Self::open(&beside, "installation");
101                }
102                tried.push(beside);
103            }
104        }
105        Err(Error::ShareNotFound { tried })
106    }
107
108    fn open(dir: &Path, origin: &'static str) -> Result<Self> {
109        let kinds = dir.join(KINDS_FILE);
110        if !kinds.is_file() {
111            return Err(Error::ShareNotFound {
112                tried: vec![dir.to_path_buf()],
113            });
114        }
115        let dir = dir.canonicalize().map_err(|e| Error::io(dir, e))?;
116        Ok(Share { dir, origin })
117    }
118

160    /// The providers the tool has a template for, by id (the file stem), sorted. A
161    /// distribution without the directory has none; that is not an error, it is a
162    /// distribution that ships no adapter.
163    pub fn provider_templates(&self) -> Vec<String> {
164        let Ok(entries) = std::fs::read_dir(self.providers_dir()) else {
165            return Vec::new();
166        };
167        let mut out: Vec<String> = entries
168            .filter_map(|e| e.ok())
169            .filter_map(|e| e.file_name().to_str().map(str::to_string))
170            .filter_map(|n| n.strip_suffix(".tmpl").map(str::to_string))
171            .collect();
172        crate::order::canonical(&mut out);
173        out
174    }

176    /// `<share>/providers.yaml`.
177    pub fn providers_path(&self) -> PathBuf {
178        self.dir.join(PROVIDERS_FILE)
179    }

181    /// The provider declarations the distribution ships, joined with its templates: one
182    /// [`ProviderDeclaration`] per id that has either, sorted by id. A distribution without
183    /// the file declares nothing and every template is a provider named by its file; a
184    /// file that does not parse is an error naming it, because a declaration half-read is
185    /// worse than none.
186    pub fn providers(&self) -> Result<ProviderDeclarations> {
187        let path = self.providers_path();
188        let file: ProvidersFile = match std::fs::read_to_string(&path) {
189            Ok(text) => {
190                crate::metadata::yaml::parse_into(&text).map_err(|reason| Error::KindSchema {
191                    reason: format!("{}: {reason}", path.display()),
192                })?
193            }
194            Err(e) if e.kind() == std::io::ErrorKind::NotFound => ProvidersFile::default(),
195            Err(e) => return Err(Error::io(&path, e)),
196        };
197        let templates = self.provider_templates();
198        let mut ids: Vec<String> = templates.clone();
199        ids.extend(file.providers.keys().cloned());
200        crate::order::canonical(&mut ids);
201        ids.dedup();
202        let providers = ids
203            .into_iter()
204            .map(|id| {
205                let decl = file.providers.get(&id);
206                ProviderDeclaration {
207                    title: decl
208                        .and_then(|d| d.title.clone())
209                        .unwrap_or_else(|| id.clone()),
210                    client_config: decl.and_then(|d| d.client_config.clone()),
211                    scratch_roots: decl.map(|d| d.scratch_roots.clone()).unwrap_or_default(),
212                    lifecycle: decl.map(|d| d.lifecycle.clone()).unwrap_or_default(),
213                    prompt_capture: decl.map(|d| d.prompt_capture).unwrap_or(false),
214                    offers: decl
215                        .and_then(|d| d.offers.as_ref())
216                        .map(offers_of)
217                        .unwrap_or_default(),
218                    template: templates.contains(&id),
219                    declared: decl.is_some(),
220                    id,
221                }
222            })
223            .collect();
224        Ok(ProviderDeclarations {
225            scratch_roots: file.scratch_roots,
226            providers,
227        })
228    }
229}

289/// How an episode boundary can be drawn for one provider.
290///
291/// Three words, because three things are true of different providers and collapsing them
292/// into "supported / not supported" is what let this tool report one provider and stay
293/// silent about five others. A provider that fires its own events is not the same as one
294/// that fires none but speaks MCP, and neither is the same as one that can do nothing.
295///
296/// The words are the declaration's own, and a word outside them is refused rather than read
297/// as `none`:
298///
299/// ```
300/// use majordomus_cli::share::EpisodeSource;
301/// let read: EpisodeSource = serde_json::from_str("\"connection\"").unwrap();
302/// assert_eq!(read, EpisodeSource::Connection);
303/// assert_eq!(EpisodeSource::default(), EpisodeSource::None, "undeclared reads as none");
304/// assert!(serde_json::from_str::<EpisodeSource>("\"hook\"").is_err(), "hooks, plural");
305/// ```
306#[derive(
307    Debug,
308    Clone,
309    Copy,
310    Default,
311    PartialEq,
312    Eq,
313    serde::Serialize,
314    serde::Deserialize,
315    schemars::JsonSchema,
316)]
317#[serde(rename_all = "snake_case")]
318pub enum EpisodeSource {
319    /// The provider fires its own session events and a hook this tool installs draws the
320    /// boundary from them. The strongest form: it sees a window somebody closed.
321    Hooks,
322    /// The provider fires no session event, but its client attaches to this repository's
323    /// shared MCP server, and the connection is the boundary (ADR 0103).
324    Connection,
325    /// Neither. Nothing can say when this provider's episode began or ended, and the tool
326    /// reports that rather than implying a capability nobody has.
327    #[default]
328    None,
329}
330
331impl EpisodeSource {
332    /// The word as the declaration writes it, which is also how it serialises; `capture
333    /// status` prints it beside the evidence that backs it.
334    ///
335    /// ```
336    /// use majordomus_cli::share::EpisodeSource;
337    /// for s in [EpisodeSource::Hooks, EpisodeSource::Connection, EpisodeSource::None] {
338    ///     assert_eq!(serde_json::to_string(&s).unwrap(), format!("\"{}\"", s.as_str()));
339    /// }
340    /// ```
341    pub fn as_str(self) -> &'static str {
342        match self {
343            EpisodeSource::Hooks => "hooks",
344            EpisodeSource::Connection => "connection",
345            EpisodeSource::None => "none",
346        }
347    }
348}

350/// Whether the person's raw prompt can be captured below the model for one provider.
351///
352/// There is no `connection` here and there must not be: an MCP server is handed tool calls,
353/// never the prompt that produced them. Prompt capture is provider-specific and the
354/// declaration says so rather than pretending the connection can stand in for it.
355///
356/// ```
357/// use majordomus_cli::share::PromptSource;
358/// assert_eq!(serde_json::from_str::<PromptSource>("\"hook\"").unwrap(), PromptSource::Hook);
359/// assert!(serde_json::from_str::<PromptSource>("\"connection\"").is_err(),
360///         "no connection can observe a prompt");
361/// assert_eq!(PromptSource::default(), PromptSource::None);
362/// ```
363#[derive(
364    Debug,
365    Clone,
366    Copy,
367    Default,
368    PartialEq,
369    Eq,
370    serde::Serialize,
371    serde::Deserialize,
372    schemars::JsonSchema,
373)]
374#[serde(rename_all = "snake_case")]
375pub enum PromptSource {
376    /// The provider hands the raw prompt to a command before the model runs.
377    Hook,
378    /// It does not. Nothing this tool can do makes the prompt observable.
379    #[default]
380    None,
381}
382

384    /// The word as the declaration writes it, which is also how it serialises.
385    ///
386    /// ```
387    /// use majordomus_cli::share::PromptSource;
388    /// assert_eq!(PromptSource::Hook.as_str(), "hook");
389    /// assert_eq!(serde_json::to_string(&PromptSource::None).unwrap(), "\"none\"");
390    /// ```
391    pub fn as_str(self) -> &'static str {
392        match self {
393            PromptSource::Hook => "hook",
394            PromptSource::None => "none",
395        }
396    }

399/// What one provider can do about the session lifecycle, and where that was verified.
400///
401/// **Every cell carries its evidence.** A capability asserted with no citation is the defect
402/// this type exists to make impossible: before it, a provider absent from a shell table was
403/// silently unsupported and a provider present in one was silently capable, and no surface
404/// could tell a reader which vendor page either answer came from — `capture status`
405/// enumerated the one provider that had an adapter and said nothing whatever about the
406/// other five this distribution declares. The rule
407/// `project.a-provider-capability-cites-its-evidence` refuses a declaration without one.
408///
409/// A provider that declares nothing is `none` on both axes with empty evidence — the state
410/// that rule refuses, so the gap is a finding rather than a silent "unsupported":
411///
412/// ```
413/// use majordomus_cli::share::{EpisodeSource, PromptSource, ProviderOffers};
414/// let undeclared = ProviderOffers::default();
415/// assert_eq!((undeclared.episode, undeclared.prompts), (EpisodeSource::None, PromptSource::None));
416/// assert!(undeclared.episode_evidence.is_empty() && undeclared.prompts_evidence.is_empty());
417/// let generic: ProviderOffers = serde_json::from_value(serde_json::json!({
418///     "episode": "connection", "episode_evidence": "ADR 0103",
419///     "prompts": "none", "prompts_evidence": "MCP carries no prompt",
420/// })).unwrap();
421/// assert_eq!(generic.episode, EpisodeSource::Connection);
422/// ```
423#[derive(
424    Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
425)]
426pub struct ProviderOffers {
427    /// How the episode boundary is drawn for this provider.
428    pub episode: EpisodeSource,
429    /// Where `episode` was verified: a vendor documentation URL, or the statement that no
430    /// such documentation exists and what was searched for it.
431    pub episode_evidence: String,
432    /// Whether the raw prompt can be captured below the model.
433    pub prompts: PromptSource,
434    /// Where `prompts` was verified, on the same terms as `episode_evidence`.
435    pub prompts_evidence: String,
436}

447/// One provider: the template's file stem, and what the declaration adds to it.
448#[derive(
449    Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
450)]
451pub struct ProviderDeclaration {
452    /// The id: the template's file stem, and the key in the declarations.
453    pub id: String,
454    /// The name a person knows it by; the id when nothing is declared.
455    pub title: String,
456    /// The file it reads a project-scoped MCP client configuration from, when it has one.

470    /// Whether the tool's adapter for it can archive the worker's prompts.
471    #[serde(default)]
472    pub prompt_capture: bool,
473    /// What the provider itself can do about the session lifecycle, and where each answer
474    /// was verified: the vendor's capability, where `lifecycle` is this tool's adapter. A
475    /// provider with no `offers:` block declares nothing, which reads as `none` with no
476    /// evidence — the state the capability rule refuses, so that the gap is a finding
477    /// rather than a silent "unsupported".
478    #[serde(default)]
479    pub offers: ProviderOffers,
480    /// The distribution ships a template for it.
481    pub template: bool,
482    /// The distribution declares it.
483    pub declared: bool,

486impl ProviderDeclarations {
487    /// Every scratch root — the tool's own and every provider's — expanded against
488    /// `primary` (the repository's primary checkout) and the environment, in declaration
489    /// order, with the provider that declared each. A root whose expansion needs a variable
490    /// the environment does not hold is skipped, not an error.
491    pub fn scratch_roots(&self, primary: &Path) -> Vec<(Option<String>, PathBuf)> {
492        let mut out = Vec::new();
493        for r in &self.scratch_roots {
494            if let Some(p) = expand_root(r, primary) {
495                out.push((None, p));
496            }
497        }
498        for p in &self.providers {
499            for r in &p.scratch_roots {
500                if let Some(path) = expand_root(r, primary) {
501                    out.push((Some(p.id.clone()), path));
502                }
503            }
504        }
505        out
506    }
507}

509/// Expand one declared root: `<primary>/`, `~/`, `${NAME:-default}` and `$NAME`, nothing
510/// else. `None` when a variable without a default is unset.
511///
512/// ```
513/// # use std::path::Path;
514/// use majordomus_cli::share::expand_root;
515/// std::env::set_var("MJ_DOCTEST_ROOT", "/srv/x");
516/// std::env::remove_var("MJ_DOCTEST_UNSET");
517/// let primary = Path::new("/repo");
518/// assert_eq!(expand_root("<primary>/.claude/worktrees", primary).unwrap(), Path::new("/repo/.claude/worktrees"));
519/// assert_eq!(expand_root("$MJ_DOCTEST_ROOT/wt", primary).unwrap(), Path::new("/srv/x/wt"));
520/// assert_eq!(expand_root("${MJ_DOCTEST_UNSET:-/d}/wt", primary).unwrap(), Path::new("/d/wt"));
521/// assert!(expand_root("$MJ_DOCTEST_UNSET/wt", primary).is_none());
522/// assert_eq!(expand_root("/tmp", primary).unwrap(), Path::new("/tmp"));
523/// ```
524pub fn expand_root(root: &str, primary: &Path) -> Option<PathBuf> {
525    let mut out = String::new();
526    let mut rest = root;
527    while !rest.is_empty() {
528        if let Some(r) = rest.strip_prefix("<primary>") {
529            out.push_str(&primary.to_string_lossy());
530            rest = r;
531        } else if let Some(r) = rest.strip_prefix('~') {
532            let home = std::env::var_os("HOME")?;
533            out.push_str(&home.to_string_lossy());
534            rest = r;
535        } else if let Some(r) = rest.strip_prefix("${") {
536            let end = r.find('}')?;
537            let (name, default) = match r[..end].split_once(":-") {
538                Some((n, d)) => (n, Some(d)),
539                None => (&r[..end], None),
540            };
541            match std::env::var(name) {
542                Ok(v) if !v.is_empty() => {
543                    out.push_str(&expand_root(&v, primary)?.to_string_lossy())
544                }
545                _ => out.push_str(&expand_root(default?, primary)?.to_string_lossy()),
546            }
547            rest = &r[end + 1..];
548        } else if let Some(r) = rest.strip_prefix('$') {
549            let end = r
550                .find(|c: char| !(c.is_ascii_alphanumeric() || c == '_'))
551                .unwrap_or(r.len());
552            let v = std::env::var(&r[..end]).ok().filter(|v| !v.is_empty())?;
553            out.push_str(&v);
554            rest = &r[end..];
555        } else {
556            let end = rest.find(['<', '~', '$']).unwrap_or(rest.len());
557            let end = if end == 0 { 1 } else { end };
558            out.push_str(&rest[..end]);
559            rest = &rest[end..];
560        }
561    }
562    Some(PathBuf::from(out))
563}

565/// Every `<vendor>/<name>/<name>.v<n>.schema.json` under a schema root, parsed, keyed by
566/// the schema identity its path derives (`majordomus.adr/v1`) and sorted by it. A root that
567/// does not exist yields nothing; that is the repository's case when it adds no schema of
568/// its own.
569///
570/// The identity and the path fix each other, so a schema can be found from a `kinds.yaml`
571/// entry without a directory scan and a file cannot quietly answer to a name it does not
572/// declare. A file whose path does not have the shape is refused by name rather than
573/// skipped, because a schema nobody loads validates nothing and says so nowhere.
574pub fn read_schema_dir(dir: &Path) -> Result<Vec<(String, serde_json::Value)>> {
575    let mut schemas = Vec::new();
576    read_schema_tree(dir, dir, &mut schemas)?;
577    schemas.sort_by(|a, b| a.0.cmp(&b.0));
578    Ok(schemas)
579}

581fn read_schema_tree(
582    root: &Path,
583    dir: &Path,
584    out: &mut Vec<(String, serde_json::Value)>,
585) -> Result<()> {
586    if !dir.is_dir() {
587        return Ok(());
588    }
589    for entry in std::fs::read_dir(dir).map_err(|e| Error::io(dir, e))? {
590        let path = entry.map_err(|e| Error::io(dir, e))?.path();
591        if path.is_dir() {
592            // `<schemas>/generated/` holds the contracts of the *generated documents* —
593            // the manifest, the registry, the benchmark matrix. Those are not kinds of the
594            // layer, carry no `<vendor>/<name>/` identity, and are read by
595            // [`read_generated_schema_dir`] instead.
596            if dir == root
597                && path.file_name().and_then(|f| f.to_str()) == Some(GENERATED_SCHEMAS_DIR)
598            {
599                continue;
600            }
601            read_schema_tree(root, &path, out)?;
602            continue;
603        }
604        let Some(file) = path.file_name().and_then(|f| f.to_str()) else {
605            continue;
606        };
607        if !file.ends_with(SCHEMA_SUFFIX) {
608            continue;
609        }
610        let relative = path
611            .strip_prefix(root)
612            .map(|p| p.to_string_lossy().replace('\\', "/"))
613            .unwrap_or_else(|_| file.to_string());
614        let identity = schema_identity(&relative).ok_or_else(|| Error::KindSchema {
615            reason: format!(
616                "{}: a schema's path is <vendor>/<name>/<name>.v<n>{SCHEMA_SUFFIX}, which this is not",
617                path.display()
618            ),
619        })?;
620        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
621        let json: serde_json::Value =
622            serde_json::from_str(&text).map_err(|e| Error::KindSchema {
623                reason: format!("{}: not JSON: {e}", path.display()),
624            })?;
625        out.push((identity, json));
626    }
627    Ok(())
628}

630/// Every `<name>.schema.json` directly under a directory, parsed, keyed by that `<name>`
631/// and sorted by it. A directory that does not exist yields nothing.
632///
633/// This is the reader for `<share>/schemas/generated/`, whose files are the contracts of
634/// the *generated documents* rather than of the layer's kinds: they answer to the document
635/// id they pin in `properties.schema.const`, not to a `<vendor>/<name>` path, so the shape
636/// [`read_schema_dir`] insists on does not apply to them.
637pub fn read_generated_schema_dir(dir: &Path) -> Result<Vec<(String, serde_json::Value)>> {
638    let mut out = Vec::new();
639    if !dir.is_dir() {
640        return Ok(out);
641    }
642    for entry in std::fs::read_dir(dir).map_err(|e| Error::io(dir, e))? {
643        let path = entry.map_err(|e| Error::io(dir, e))?.path();
644        let Some(file) = path.file_name().and_then(|f| f.to_str()) else {
645            continue;
646        };
647        if path.is_dir() || !file.ends_with(SCHEMA_SUFFIX) {
648            continue;
649        }
650        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
651        let json: serde_json::Value =
652            serde_json::from_str(&text).map_err(|e| Error::KindSchema {
653                reason: format!("{}: not JSON: {e}", path.display()),
654            })?;
655        out.push((file[..file.len() - SCHEMA_SUFFIX.len()].to_string(), json));
656    }
657    out.sort_by(|a, b| a.0.cmp(&b.0));
658    Ok(out)
659}

661/// The identity a schema path carries: `majordomus/adr/adr.v1.schema.json` is
662/// `majordomus.adr/v1`. A path of another shape yields nothing.
663///
664/// ```
665/// use majordomus_cli::share::schema_identity;
666/// assert_eq!(
667///     schema_identity("majordomus/adr/adr.v1.schema.json").as_deref(),
668///     Some("majordomus.adr/v1")
669/// );
670/// assert_eq!(schema_identity("adr.schema.json"), None);
671/// assert_eq!(schema_identity("majordomus/adr/other.v1.schema.json"), None);
672/// ```
673pub fn schema_identity(relative: &str) -> Option<String> {
674    let stem = relative.strip_suffix(SCHEMA_SUFFIX)?;
675    let mut parts = stem.split('/');
676    let vendor = parts.next()?;
677    let name = parts.next()?;
678    let file = parts.next()?;
679    if parts.next().is_some() || vendor.is_empty() || name.is_empty() {
680        return None;
681    }
682    let version = file.strip_prefix(&format!("{name}."))?;
683    if !version.starts_with('v') || version.len() < 2 {
684        return None;
685    }
686    Some(format!("{vendor}.{name}/{version}"))
687}
