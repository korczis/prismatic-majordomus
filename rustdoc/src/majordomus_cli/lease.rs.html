1//! One shared server per checkout: the lease that decides who it is. The first
2//! `majordomus mcp` (or `serve`) to create `state/mcp/server.json` under the checkout's
3//! local half owns the server and publishes its URL there; every later process reads the
4//! file, checks that the server answers for this root, and attaches to it. A lease whose
5//! server does not answer is stale, and the next process takes it over; so is a file that
6//! is not a lease document, an empty one, or one whose owner never published a URL. A
7//! server that is slow to answer while its process is alive is busy, not stale: the
8//! election waits for it for [`BUSY_GRACE`] before it takes anything over. The
9//! file is the only thing the server writes anywhere, it lives under `.ai/local/` (never
10//! tracked, by the layer's contract), and it is removed when the server stops, or when
11//! the server dies of `SIGTERM`, `SIGINT` or `SIGHUP`.
12//!
13//! Per *checkout*, and this file said "per repository" until ADR 0044: a linked worktree
14//! is a checkout, so it has a manifest, a root, a lease and a server of its own, and on
15//! 2026-09-11 this repository had seven of them at once. What is repository-wide is the
16//! board, which `peers.list` gathers over these very leases; the election is unchanged.
17

33/// The lease file, relative to the checkout-local half (`.ai/local/`).

36/// The lease file's `schema`.
37pub const SCHEMA: &str = "majordomus-mcp-lease/v1";

39/// How long a lease without a URL may be (the owner is still binding) before it counts as
40/// abandoned.
41pub const BIND_GRACE: Duration = Duration::from_secs(15);
42
43/// How long the probe of a published URL waits.
44pub const PROBE_TIMEOUT: Duration = Duration::from_secs(2);
45
46/// How long a live server that does not answer its probe is waited for before its lease is
47/// taken over.
48///
49/// A probe that times out is not a dead server. On 2026-09-15 ten MCP clients attached at
50/// once to a server a shell entry had started; nine were bridged to it, and the tenth
51/// probed while the server was answering the other nine, heard nothing within
52/// [`PROBE_TIMEOUT`], took the lease over and started a second server for one checkout
53/// (test/cases/192). So a silent probe of a lease whose process is still alive is waited
54/// on, and asked again; only a server that stays silent for this long, or whose process is
55/// gone, loses its lease. The bound keeps the recovery from a hung server, and it sits
56/// well inside [`JOIN_TIMEOUT`] so that the take-over still happens within one election.
57pub const BUSY_GRACE: Duration = Duration::from_secs(10);
58
59/// How long a process keeps trying to acquire or join the lease before it gives up and
60/// says so: the bind grace with a margin for the probes. The caller then serves its
61/// client alone.
62pub const JOIN_TIMEOUT: Duration = Duration::from_secs(20);

64/// The identity of the file an executable was started from: where it is, and the mtime
65/// and size of the file at that path. A server outlives its own binary — a rebuild replaces
66/// the file under a process that keeps serving the code it loaded hours ago — and nothing
67/// about the process itself says so. This is what makes that visible.
68///
69/// ```
70/// use majordomus_cli::lease::ExecutableIdentity;
71/// let recorded = ExecutableIdentity { path: "/opt/majordomus".into(), mtime: 1_700_000_000, size: 42 };
72/// let text = serde_json::to_string(&recorded).unwrap();
73/// assert_eq!(serde_json::from_str::<ExecutableIdentity>(&text).unwrap(), recorded);
74/// ```
75#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
76pub struct ExecutableIdentity {
77    /// The path the process was started from.
78    pub path: PathBuf,
79    /// The file's modification time, seconds since the epoch.
80    pub mtime: u64,
81    /// The file's size in bytes.
82    pub size: u64,
83}
84
85impl ExecutableIdentity {
86    /// Has the file at the recorded path been replaced or removed since it was recorded?
87    /// The reason, for a reader; `None` when the same file is still there.
88    ///
89    /// ```
90    /// use majordomus_cli::lease::ExecutableIdentity;
91    /// let gone = ExecutableIdentity { path: "/nowhere/majordomus".into(), mtime: 1, size: 1 };
92    /// assert!(gone.replaced().unwrap().contains("no longer on disk"));
93    /// ```
94    pub fn replaced(&self) -> Option<String> {
95        let Ok(meta) = fs::metadata(&self.path) else {
96            return Some(format!(
97                "the executable it was started from, {}, is no longer on disk",
98                self.path.display()
99            ));
100        };
101        let mtime = meta
102            .modified()
103            .ok()
104            .and_then(|m| m.duration_since(SystemTime::UNIX_EPOCH).ok())
105            .map(|d| d.as_secs())
106            .unwrap_or(0);
107        if mtime == self.mtime && meta.len() == self.size {
108            return None;
109        }
110        Some(format!(
111            "the executable it was started from, {}, has been replaced since (it is serving \
112             code that is no longer on disk)",
113            self.path.display()
114        ))
115    }
116}

118/// The lease file, typed: what the process serving a checkout wrote down about itself.
119/// Every reader of the file — the election, a command asking which server is running, the
120/// environment snapshot, the server's own status — reads it through [`LeaseFile::read`],
121/// so that a field added here is added once and a file that is not a lease is refused in
122/// one place. A key this version does not know is ignored, so a lease written by a newer
123/// executable still reads; a key it needs that an older lease lacks takes its default.
124///
125/// ```
126/// use majordomus_cli::lease::{LeaseDocument, SCHEMA};
127/// // a lease an older server wrote: no pid, no version, a key this version has never heard of
128/// let older = format!(r#"{{"schema":"{SCHEMA}","token":"t","root":"/r","url":"http://127.0.0.1:1","later":1}}"#);
129/// let doc: LeaseDocument = serde_json::from_str(&older).unwrap();
130/// assert_eq!(doc.pid, 0);
131/// assert!(doc.version.is_none(), "older than any executable that reads this");
132/// assert_eq!(doc.url.as_deref(), Some("http://127.0.0.1:1"));
133/// // and what this version writes reads back as itself
134/// let text = serde_json::to_string(&doc).unwrap();
135/// assert_eq!(serde_json::from_str::<LeaseDocument>(&text).unwrap(), doc);
136/// ```
137#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
138pub struct LeaseDocument {
139    /// [`SCHEMA`].
140    pub schema: String,
141    /// The process id of the server. Informational: a live pid is not a live server, and
142    /// nothing decides liveness from it.
143    #[serde(default)]
144    pub pid: u32,
145    /// What makes the file this process's: only the process holding this token removes it.
146    #[serde(default)]
147    pub token: String,
148    /// The checkout root the server serves.
149    #[serde(default)]
150    pub root: PathBuf,
151    /// The address, once bound; `null` while the owner is still binding.
152    pub url: Option<String>,
153    /// When the server took the lease, RFC 3339.
154    #[serde(default)]
155    pub started_at: String,
156    /// The executable the server was started from, when it could be located.
157    #[serde(default)]
158    pub executable: Option<ExecutableIdentity>,
159    /// The executable's version. Absent in a lease written before this field existed,
160    /// which is itself a fact: that server is older than any executable that reads this.
161    #[serde(default, skip_serializing_if = "Option::is_none")]
162    pub version: Option<String>,
163}

165/// What a lease file holds, read once. Every state of the file is a variant, so that a
166/// reader decides what to do about each rather than learning it from an error.
167///
168/// ```
169/// use majordomus_cli::lease::LeaseFile;
170/// let dir = tempfile::tempdir().unwrap();
171/// let path = dir.path().join("server.json");
172/// assert_eq!(LeaseFile::read(&path), LeaseFile::Absent);
173/// assert!(LeaseFile::read(&path).document().is_none(), "no document to hand back");
174/// ```
175#[derive(Debug, Clone, PartialEq, Eq)]
176pub enum LeaseFile {
177    /// There is no file.
178    Absent,
179    /// A file with nothing in it: its owner created it and has not written it yet.
180    Empty,
181    /// A file that is not a lease document; the reason, for the log.
182    Corrupt(String),
183    /// A lease document.
184    Document(LeaseDocument),
185}
186
187impl LeaseFile {
188    /// Read and type the file at `path`. Never fails: every state of the file is one of the
189    /// variants, and what to do about each is the reader's decision.
190    ///
191    /// ```
192    /// use majordomus_cli::lease::{LeaseFile, SCHEMA};
193    /// let dir = tempfile::tempdir().unwrap();
194    /// let path = dir.path().join("server.json");
195    /// assert_eq!(LeaseFile::read(&path), LeaseFile::Absent);
196    /// std::fs::write(&path, "").unwrap();
197    /// assert_eq!(LeaseFile::read(&path), LeaseFile::Empty);
198    /// std::fs::write(&path, "{not json").unwrap();
199    /// assert!(matches!(LeaseFile::read(&path), LeaseFile::Corrupt(r) if r.starts_with("not JSON")));
200    /// std::fs::write(&path, r#"{"schema":"other/v1"}"#).unwrap();
201    /// assert!(matches!(LeaseFile::read(&path), LeaseFile::Corrupt(r) if r.contains(SCHEMA)));
202    /// std::fs::write(&path, format!(r#"{{"schema":"{SCHEMA}","token":"x","root":"/r","url":null}}"#)).unwrap();
203    /// match LeaseFile::read(&path) {
204    ///     LeaseFile::Document(d) => { assert_eq!(d.token, "x"); assert!(d.url.is_none()); assert!(d.version.is_none()); }
205    ///     other => panic!("{other:?}"),
206    /// }
207    /// ```
208    pub fn read(path: &Path) -> LeaseFile {
209        let text = match fs::read_to_string(path) {
210            Ok(text) => text,
211            Err(e) if e.kind() == std::io::ErrorKind::NotFound => return LeaseFile::Absent,
212            Err(e) => return LeaseFile::Corrupt(format!("cannot be read ({e})")),
213        };
214        if text.trim().is_empty() {
215            return LeaseFile::Empty;
216        }
217        let value: Value = match serde_json::from_str(&text) {
218            Ok(value) => value,
219            Err(e) => return LeaseFile::Corrupt(format!("not JSON ({e})")),
220        };
221        if !value.is_object() || value["schema"] != SCHEMA {
222            return LeaseFile::Corrupt(format!("not a {SCHEMA} document"));
223        }
224        match serde_json::from_value::<LeaseDocument>(value) {
225            Ok(document) => LeaseFile::Document(document),
226            Err(e) => LeaseFile::Corrupt(format!("not a {SCHEMA} document ({e})")),
227        }
228    }
229
230    /// The document, when the file holds one; `None` for an absent, empty or corrupt file,
231    /// for a reader that has no different answer for those three.
232    ///
233    /// ```
234    /// use majordomus_cli::lease::{LeaseDocument, LeaseFile, SCHEMA};
235    /// let doc = LeaseDocument {
236    ///     schema: SCHEMA.into(), pid: 7, token: "t".into(), root: "/r".into(), url: None,
237    ///     started_at: "2026-09-10T00:00:00Z".into(), executable: None, version: None,
238    /// };
239    /// assert_eq!(LeaseFile::Document(doc.clone()).document(), Some(&doc));
240    /// assert_eq!(LeaseFile::Corrupt("not JSON".into()).document(), None);
241    /// ```
242    pub fn document(&self) -> Option<&LeaseDocument> {
243        match self {
244            LeaseFile::Document(d) => Some(d),
245            _ => None,
246        }
247    }
248}

250/// The lease this process holds and has published, when it is a server. Set by
251/// [`Lease::publish`], cleared when the lease is released, so that the server can say what
252/// it wrote down about itself without reading its own file back.
253static PUBLISHED: OnceLock<Mutex<Option<LeaseDocument>>> = OnceLock::new();
254
255fn published() -> &'static Mutex<Option<LeaseDocument>> {
256    PUBLISHED.get_or_init(|| Mutex::new(None))
257}

259/// The lease this process holds and has published: `None` in a process that serves nothing.
260///
261/// ```
262/// // a process that has published no lease holds none
263/// assert!(majordomus_cli::lease::held().is_none());
264/// ```
265pub fn held() -> Option<LeaseDocument> {
266    published().lock().ok()?.clone()
267}

269/// Is `doc` the lease this very process holds, published and not lost?
270///
271/// A server asked where its own checkout stands must answer from memory. Probing the
272/// address its lease names is a request to itself, served by the same small pool of HTTP
273/// workers the question arrived on: with every worker answering such a question at once,
274/// none is left to answer the probes, each waits out [`PROBE_TIMEOUT`], the server calls
275/// itself stale — and a client electing in that window times out on the same queue and
276/// takes a live lease over. Measured on 2026-09-15: ten MCP clients calling
277/// `majordomus_peers` at once, and one of them started a second server of the checkout.
278///
279/// ```
280/// use majordomus_cli::lease::{is_own, LeaseDocument};
281/// let doc: LeaseDocument = serde_json::from_str(
282///     r#"{"schema":"lease/v1","pid":1,"token":"t","url":"http://127.0.0.1:1"}"#,
283/// )
284/// .unwrap();
285/// // a process that holds no lease owns no document
286/// assert!(!is_own(&doc));
287/// ```
288pub fn is_own(doc: &LeaseDocument) -> bool {
289    !was_lost()
290        && held().is_some_and(|h| h.token == doc.token && h.pid == doc.pid && h.url == doc.url)
291}

293/// The lease this process holds. Dropping it removes the file (when the file is still
294/// this process's), so a failed start never leaves a stale lease behind.
295#[derive(Debug)]
296pub struct Lease {
297    path: PathBuf,
298    token: String,
299    root: PathBuf,
300    released: bool,
301    /// When this process took the lease, RFC 3339. Fixed here rather than at publish time
302    /// because it is the identity of this server's generation.
303    started_at: String,
304    /// `true` from the election until [`Lease::publish`]: the window in which
305    /// [`Lease::keep_alive`] rewrites the file so that it never looks abandoned, and the
306    /// lock under which it and `publish` write, so that a touch can never land on top of
307    /// the URL.
308    binding: Arc<Mutex<bool>>,
309}

311/// What the election decided for this process.
312#[derive(Debug)]
313pub enum Role {
314    /// Nobody serves this repository: this process does, holding the lease.
315    Server(Lease),
316    /// A server answers at this URL: attach to it.
317    Peer {
318        /// `http://host:port` of the running server.
319        url: String,
320    },
321}

328/// Where the lease of the checkout at `root` lives, given the repository-relative path of
329/// its local half (`.ai/local`). The one composition of that path; every reader of a lease
330/// that is not this process's own repository — another checkout's, in the server's status
331/// or the environment snapshot — goes through here.
332///
333/// ```
334/// use majordomus_cli::lease::lease_file;
335/// use std::path::Path;

337///     lease_file(Path::new("/r"), ".ai/local"),
338///     Path::new("/r/.ai/local/state/mcp/server.json")
339/// );
340/// ```
341pub fn lease_file(root: &Path, local_half: &str) -> PathBuf {
342    root.join(local_half).join(LEASE_PATH)
343}

345/// Which server is serving this repository right now, if any.
346///
347/// Read-only, unlike [`elect`]: it takes no lease, creates no file and waits for nobody, so
348/// a command that only wants to *ask* the running server something cannot accidentally
349/// become it. `None` means no lease, no URL in it, or a lease whose server does not answer
350/// for this root.
351pub fn serving(repo: &Repository) -> Option<String> {
352    let url = LeaseFile::read(&lease_path(repo)).document()?.url.clone()?;
353    probe(&url, repo.root()).then_some(url)
354}

356/// Decide whether this process serves the repository or attaches to the process that does.
357///
358/// An existing file is read on every attempt and classified: the lease of a live server
359/// (it answers for this root) is attached to; a stale one (its server does not answer),
360/// a corrupt one (not a lease document), an empty one, or an abandoned one (no URL after
361/// [`BIND_GRACE`]) is taken over; a fresh lease without a URL means its owner is still
362/// binding, and this process waits for it. Nothing a client leaves behind can lock the
363/// others out; when the file can neither be created nor removed within [`JOIN_TIMEOUT`],
364/// the error names the path and the caller serves its client alone.
365pub fn elect(repo: &Repository) -> Result<Role> {
366    let path = lease_path(repo);
367    let root = repo.root().to_path_buf();
368    if let Some(dir) = path.parent() {
369        fs::create_dir_all(dir).map_err(|e| Error::io(dir, e))?;
370    }
371    let waited_since = Instant::now();
372    // when a live server first failed to answer this election; cleared whenever it answers,
373    // or the file stops naming a busy server
374    let mut busy_since: Option<Instant> = None;
375    loop {
376        match fs::OpenOptions::new()
377            .write(true)
378            .create_new(true)
379            .open(&path)
380        {
381            Ok(mut file) => {
382                let token = format!(
383                    "{}-{:x}",
384                    std::process::id(),
385                    SystemTime::now()
386                        .duration_since(SystemTime::UNIX_EPOCH)
387                        .map(|d| d.as_nanos())
388                        .unwrap_or(0)
389                );
390                let lease = Lease {
391                    path: path.clone(),
392                    token,
393                    root,
394                    released: false,
395                    started_at: crate::peers::rfc3339(SystemTime::now()),
396                    binding: Arc::new(Mutex::new(true)),
397                };
398                let text =
399                    serde_json::to_string(&lease.document(None)).map_err(|e| Error::Lease {
400                        reason: format!("cannot render the lease: {e}"),
401                    })?;
402                file.write_all(text.as_bytes())
403                    .map_err(|e| Error::io(&path, e))?;
404                signals::hold(&path);
405                return Ok(Role::Server(lease));
406            }
407            Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
408                let (found, seen) = inspect(&path, &root);
409                if !matches!(found, Found::Busy(_)) {
410                    busy_since = None;
411                }
412                match found {
413                    Found::Live(url) => return Ok(Role::Peer { url }),
414                    Found::Busy(url) => {
415                        let since = *busy_since.get_or_insert_with(Instant::now);
416                        if since.elapsed() < BUSY_GRACE {
417                            std::thread::sleep(Duration::from_millis(200));
418                        } else {
419                            tracing::warn!(
420                                lease = %path.display(),
421                                "the server at {url} is alive and has not answered for {} seconds; taking it over",
422                                BUSY_GRACE.as_secs()
423                            );
424                            take_over(&path, &seen)?;
425                            busy_since = None;
426                        }
427                    }
428                    Found::Stale(reason) => {
429                        tracing::warn!(lease = %path.display(), "{reason}; taking it over");
430                        take_over(&path, &seen)?;
431                    }
432                    // an owner that is still binding keeps its file young (`keep_alive`), so
433                    // a file that is old enough to be abandoned is classified so by `inspect`
434                    // and this arm only ever waits; the whole attempt is bounded below
435                    Found::Binding => std::thread::sleep(Duration::from_millis(100)),
436                }
437            }
438            Err(e) => return Err(Error::io(&path, e)),
439        }
440        if waited_since.elapsed() > JOIN_TIMEOUT {
441            return Err(Error::Lease {
442                reason: format!(
443                    "could not acquire or join the lease at {} within {} seconds",
444                    path.display(),
445                    JOIN_TIMEOUT.as_secs()
446                ),
447            });
448        }
449    }
450}

452/// What a lease file that already exists says.
453enum Found {
454    /// A server answers at this URL for this root.
455    Live(String),
456    /// The server at this URL did not answer in time, and the process that holds the lease
457    /// is alive: busy rather than gone, and waited on for [`BUSY_GRACE`].
458    Busy(String),
459    /// The file is not a usable lease; the reason says why, for the log.
460    Stale(String),
461    /// A lease without a URL, young enough that its owner may still be binding.
462    Binding,
463}

465/// What the running executable is, for the lease to record: the path it was started from
466/// and the identity of the file at that path. A server outlives its own binary — a rebuild
467/// replaces the file under a process that keeps serving the code it loaded hours ago — and
468/// nothing about the process itself says so. This is what makes that visible.
469///
470/// `None` when the executable cannot be located or stat'ed; the lease then carries no
471/// claim, and the election's `superseded` check makes none either.
472///
473/// ```
474/// use majordomus_cli::lease::executable_identity;
475/// // the process running this example can locate itself, and the file it names is there
476/// let mine = executable_identity().expect("a test binary knows where it is");
477/// assert!(mine.path.is_file());
478/// assert!(mine.replaced().is_none(), "the file that is running has not been replaced");
479/// ```
480pub fn executable_identity() -> Option<ExecutableIdentity> {
481    let path = std::env::current_exe().ok()?;
482    let meta = fs::metadata(&path).ok()?;
483    let mtime = meta
484        .modified()
485        .ok()?
486        .duration_since(SystemTime::UNIX_EPOCH)
487        .ok()?
488        .as_secs();
489    Some(ExecutableIdentity {
490        path,
491        mtime,
492        size: meta.len(),
493    })
494}

496/// Has the executable behind a lease been replaced since that server started?
497///
498/// Only a lease naming *this process's own* executable path can answer: same path, a file
499/// that is now a different file, and the server on the other end is provably running code
500/// that no longer exists on disk. A lease naming some other path — a release install next
501/// to a debug build — makes no claim either way and is left alone, so two legitimate
502/// binaries never fight over the lease.
503fn superseded(doc: &LeaseDocument) -> Option<String> {
504    let recorded = doc.executable.as_ref()?;
505    let mine = executable_identity()?;
506    if recorded.path != mine.path {
507        return None;
508    }
509    if recorded.mtime == mine.mtime && recorded.size == mine.size {
510        return None;
511    }
512    Some(format!(
513        "superseded lease: the server was started from {} and that file has been replaced since \
514         (it is serving code that is no longer on disk)",
515        mine.path.display()
516    ))
517}

519/// Read and classify an existing lease file. What was read comes back beside the verdict,
520/// so that a take-over can insist on removing the file it judged and not one that arrived
521/// in the meantime.
522fn inspect(path: &Path, root: &Path) -> (Found, LeaseFile) {
523    let age = file_age(path);
524    let seen = LeaseFile::read(path);
525    let doc = match &seen {
526        // gone between the failed create and this read: the next attempt creates it
527        LeaseFile::Absent => return (Found::Binding, seen),
528        LeaseFile::Empty if age > BIND_GRACE => {
529            return (
530                Found::Stale("empty lease: its owner never wrote it".into()),
531                seen,
532            )
533        }
534        LeaseFile::Empty => return (Found::Binding, seen),
535        LeaseFile::Corrupt(reason) => {
536            return (Found::Stale(format!("corrupt lease: {reason}")), seen)
537        }
538        LeaseFile::Document(doc) => doc.clone(),
539    };
540    // before asking whether it answers: a server that answers from a binary that has been
541    // replaced answers with yesterday's code, which is the harder failure to see
542    if let Some(reason) = superseded(&doc) {
543        return (Found::Stale(reason), seen);
544    }
545    let found = match doc.url.as_deref() {
546        Some(url) => match ask(url, root, PROBE_TIMEOUT) {
547            Answer::Ours(_) => Found::Live(url.to_string()),
548            Answer::Silent if alive(doc.pid) => Found::Busy(url.to_string()),
549            Answer::Silent | Answer::NotOurs => Found::Stale(format!(
550                "stale lease: the server it names at {url} does not answer for this repository"
551            )),
552        },
553        None if age > BIND_GRACE => {
554            Found::Stale("abandoned lease: its owner never published a URL".into())
555        }
556        None => Found::Binding,
557    };
558    (found, seen)
559}

561/// How long ago the file was last written; zero when that cannot be read, so that a file
562/// whose age is unknown is never taken for an old one.
563///
564/// ```
565/// use majordomus_cli::lease::file_age;
566/// use std::time::Duration;
567/// let dir = tempfile::tempdir().unwrap();
568/// let path = dir.path().join("server.json");
569/// assert_eq!(file_age(&path), Duration::ZERO, "no file: no age");
570/// std::fs::write(&path, "").unwrap();
571/// assert!(file_age(&path) < Duration::from_secs(60), "just written");
572/// ```
573pub fn file_age(path: &Path) -> Duration {
574    fs::metadata(path)
575        .and_then(|m| m.modified())
576        .ok()
577        .and_then(|m| m.elapsed().ok())
578        .unwrap_or(Duration::ZERO)
579}

581/// Remove a lease that cannot be used, so that the next attempt creates a fresh one — but
582/// only the file that was judged: when another process has replaced it since, the file on
583/// disk is somebody's fresh lease, and the next round of the election reads that one.
584fn take_over(path: &Path, seen: &LeaseFile) -> Result<()> {
585    if LeaseFile::read(path) != *seen {
586        tracing::debug!(lease = %path.display(), "the lease changed under the take-over; reading it again");
587        return Ok(());
588    }
589    match fs::remove_file(path) {
590        Ok(()) => Ok(()),
591        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
592        Err(e) => Err(Error::Lease {
593            reason: format!(
594                "cannot remove the unusable lease at {}: {e}",
595                path.display()
596            ),
597        }),
598    }
599}

601/// Set by [`lost`], never cleared: a lease this process was given and no longer has.
602///
603/// Distinct from `held().is_none()`, which is also true of a process that never took a
604/// lease at all — a test, a one-off command, a bridged client. Only a process that *had*
605/// the lease and lost it must change how it behaves, so only that one is recorded here.
606static LOST: AtomicBool = AtomicBool::new(false);
607
608/// The key the index answers [`was_lost`]'s converse under: whether the server answering
609/// still holds the lease of the checkout it serves. Named once, read by [`probe`] and
610/// written by the index route, so the two cannot drift apart.
611pub const LEASEHOLDER_KEY: &str = "leaseholder";
612
613/// This process's lease was taken over by another process: from now on a signal must not
614/// unlink the file, which is somebody else's, and this process stops claiming to be the
615/// checkout's server — [`held`] answers `None` and [`was_lost`] answers `true` from here
616/// on. It serves the peers it has and ends with them; what it must not do is take on new
617/// ones, or answer a stranger's probe as though it were still the one.
618///
619/// ```
620/// // a process that holds no lease has nothing to lose; saying so twice changes nothing
621/// majordomus_cli::lease::lost();
622/// majordomus_cli::lease::lost();
623/// assert!(majordomus_cli::lease::held().is_none());
624/// assert!(majordomus_cli::lease::was_lost(), "it was told the lease is gone");
625/// ```
626pub fn lost() {
627    LOST.store(true, Ordering::SeqCst);
628    if let Ok(mut published) = published().lock() {
629        *published = None;
630    }
631    signals::release();
632}

634/// Did this process hold the checkout's lease and lose it?
635///
636/// The one question a server asks before taking on work that outlives the request. A
637/// process that never held a lease answers `false` and goes on serving whoever asked it —
638/// a test, a bridged client, a one-off command. One that was superseded answers `true`,
639/// and from then on refuses to become anybody's server again. It never resets: a lease
640/// that was taken over is not given back.
641///
642/// ```
643/// // nothing has taken a lease away from the process running this example; `lost()` in
644/// // the doctest above runs in a process of its own
645/// assert!(!majordomus_cli::lease::was_lost());
646/// ```
647pub fn was_lost() -> bool {
648    LOST.load(Ordering::SeqCst)
649}

651/// Does a Majordomus server answer at `url` for the repository at `root`, *as* that
652/// repository's server?
653///
654/// Three questions, and the third is the one a remembered address needs: it is a
655/// Majordomus server, it serves this root, and it still holds this checkout's lease.
656/// Without the third, a process whose lease was taken over answers every probe exactly as
657/// the current server does — same name, same `repository_id`, its own peer board, its own
658/// generation of the layer — and a client holding an address from before the takeover is
659/// served plausible answers by a server nobody else is talking to. That was measured on
660/// 2026-09-10: two servers of one checkout, one lease, and the older one indistinguishable
661/// from the current one on this endpoint.
662///
663/// A server too old to answer the question is accepted. It cannot be told from a current
664/// one here, and refusing it would be the worse failure: a live server taken for dead is
665/// taken over, which is how one checkout comes to have two.
666pub fn probe(url: &str, root: &Path) -> bool {
667    matches!(ask(url, root, PROBE_TIMEOUT), Answer::Ours(_))
668}
669
670/// [`probe`], keeping what the server answered: its index document when it answers as the
671/// leaseholder of this checkout, `None` otherwise. One decision, so a caller that also wants
672/// the surfaces the server lists makes neither a second request nor a second judgement.
673///
674/// ```
675/// use majordomus_cli::lease::probe_reply;
676/// use std::time::Duration;
677/// // nothing listens on port 1 of the loopback address
678/// let root = std::path::Path::new("/r");
679/// assert!(probe_reply("http://127.0.0.1:1", root, Duration::from_millis(50)).is_none());
680/// ```
681pub fn probe_reply(url: &str, root: &Path, timeout: Duration) -> Option<Value> {
682    match ask(url, root, timeout) {
683        Answer::Ours(index) => Some(index),
684        Answer::NotOurs | Answer::Silent => None,
685    }
686}
687
688/// What a probe of a published URL heard.
689enum Answer {
690    /// A Majordomus server that serves this root and holds its lease, with the index
691    /// document it answered.
692    Ours(Value),
693    /// Something answered that is not this checkout's server, or nothing is listening.
694    NotOurs,
695    /// The connection was made and no answer came within the timeout: a server that is
696    /// busy looks exactly like this, and so does a hung one.
697    Silent,
698}
699
700/// Ask a published URL the probe's three questions within `timeout`, telling a server that
701/// said no apart from one that said nothing in time.
702fn ask(url: &str, root: &Path, timeout: Duration) -> Answer {
703    match bridge::request(url, "GET", "/", &[], None, timeout) {
704        Ok(reply) if reply.status == 200 => {
705            let v: Value = serde_json::from_str(&reply.body).unwrap_or(Value::Null);
706            // the identity and not the path: the index names the repository it serves
707            // without telling every caller where the checkout sits
708            let ours = v["name"] == "majordomus"
709                && v["repository_id"].as_str() == Some(crate::repository::identity(root).as_str())
710                && v[LEASEHOLDER_KEY] != Value::Bool(false);
711            if ours {
712                Answer::Ours(v)
713            } else {
714                Answer::NotOurs
715            }
716        }
717        Ok(_) => Answer::NotOurs,
718        Err(e)
719            if matches!(
720                e.kind(),
721                std::io::ErrorKind::TimedOut | std::io::ErrorKind::WouldBlock
722            ) =>
723        {
724            Answer::Silent
725        }
726        Err(_) => Answer::NotOurs,
727    }
728}

730/// Is the process with this id alive on this machine? A lease is a file of this checkout,
731/// so its pid is a pid of this host. `kill(pid, 0)` delivers nothing and asks only whether
732/// the process exists; a process owned by somebody else answers `EPERM`, which is alive.
733fn alive(pid: u32) -> bool {
734    if pid == 0 {
735        return false;
736    }
737    let Ok(pid) = libc::pid_t::try_from(pid) else {
738        return false;
739    };
740    // SAFETY: signal 0 performs the existence and permission check and sends nothing.
741    let rc = unsafe { libc::kill(pid, 0) };
742    rc == 0 || std::io::Error::last_os_error().raw_os_error() == Some(libc::EPERM)
743}

745impl Lease {
746    /// The file.
747    pub fn path(&self) -> &Path {
748        &self.path
749    }
750
751    /// The repository root the lease is for.
752    pub fn root(&self) -> &Path {
753        &self.root
754    }
755
756    /// When this process took the lease, RFC 3339: the identity of this server's
757    /// generation, the same value the lease file carries as `started_at`.
758    pub fn started_at(&self) -> &str {
759        &self.started_at
760    }
761
762    /// What makes the file this process's. For the server's own reader, which checks from
763    /// another thread whether the lease is still its own.
764    pub(crate) fn token(&self) -> &str {
765        &self.token
766    }
767
768    /// Keep the lease young while the layer loads. A peer that finds a lease without a URL
769    /// waits for it only as long as the file is younger than [`BIND_GRACE`]; a cold start
770    /// that takes longer than that — a large layer, a machine under load — would otherwise
771    /// be taken for an abandoned one and taken over while its owner is still binding. So
772    /// until [`Lease::publish`], a thread rewrites the same document every third of the
773    /// grace, under the lock `publish` also takes, so that a touch can never land after the
774    /// URL. It writes only while the file is still this process's: a lease that was taken
775    /// over anyway is somebody else's to write.
776    ///
777    /// ```
778    /// use majordomus_cli::lease::{elect, held, LeaseFile, Role};
779    /// use majordomus_cli::Repository;
780    /// let dir = tempfile::tempdir().unwrap();
781    /// std::fs::create_dir_all(dir.path().join(".ai/repo")).unwrap();
782    /// std::fs::write(dir.path().join(".ai/manifest.yaml"),
783    ///     "schema: ai-repository/v1\nrepo:\n  path: repo\nlocal:\n  path: local\n  tracked: false\n  implicit_context: false\nsections:\n  policy: repo/policy.yaml\n").unwrap();
784    /// let repo = Repository::discover(dir.path()).unwrap();
785    /// let Role::Server(lease) = elect(&repo).unwrap() else { panic!("nobody else serves a fresh directory") };
786    /// lease.keep_alive();                         // returns at once; the touching runs beside the load
787    /// assert!(held().is_none(), "nothing is published yet");
788    /// lease.publish("http://127.0.0.1:1").unwrap();  // stops the touching, under the same lock
789    /// assert_eq!(held().unwrap().url.as_deref(), Some("http://127.0.0.1:1"));
790    /// assert!(matches!(LeaseFile::read(lease.path()), LeaseFile::Document(d) if d.url.is_some()));
791    /// lease.release();
792    /// assert_eq!(LeaseFile::read(&dir.path().join(".ai/local/state/mcp/server.json")), LeaseFile::Absent);
793    /// ```
794    pub fn keep_alive(&self) {
795        let path = self.path.clone();
796        let token = self.token.clone();
797        let binding = Arc::clone(&self.binding);
798        let document = serde_json::to_string(&self.document(None)).unwrap_or_default();
799        let tick = BIND_GRACE / 3;
800        let _ = std::thread::Builder::new()
801            .name("majordomus-lease-keep-alive".into())
802            .spawn(move || loop {
803                std::thread::sleep(tick);
804                let Ok(guard) = binding.lock() else { return };
805                if !*guard {
806                    return;
807                }
808                let mine = LeaseFile::read(&path)
809                    .document()
810                    .is_some_and(|d| d.token == token);
811                if !mine {
812                    return;
813                }
814                let tmp = path.with_extension("json.tmp");
815                if fs::write(&tmp, &document).is_ok() {
816                    let _ = fs::rename(&tmp, &path);
817                }
818            });
819    }
820
821    fn document(&self, url: Option<&str>) -> LeaseDocument {
822        LeaseDocument {
823            schema: SCHEMA.into(),
824            pid: std::process::id(),
825            token: self.token.clone(),
826            root: self.root.clone(),
827            url: url.map(str::to_string),
828            started_at: self.started_at.clone(),
829            executable: executable_identity(),
830            version: Some(crate::VERSION.into()),
831        }
832    }
833
834    /// Record the URL the server is listening on, atomically, so that a peer reading the
835    /// file sees either no URL or the whole one. Refused when the file is no longer this
836    /// process's: a lease taken over while its owner was binding belongs to whoever took it,
837    /// and writing over it would leave two servers claiming one address.
838    pub fn publish(&self, url: &str) -> Result<()> {
839        let mut binding = self.binding.lock().map_err(|_| Error::Lease {
840            reason: "the lease's binding lock is poisoned".into(),
841        })?;
842        *binding = false;
843        if !self.is_mine() {
844            return Err(Error::Lease {
845                reason: format!(
846                    "the lease at {} was taken over while this server was starting; another process serves this checkout",
847                    self.path.display()
848                ),
849            });
850        }
851        let tmp = self.path.with_extension("json.tmp");
852        let document = self.document(Some(url));
853        let text = serde_json::to_string(&document).map_err(|e| Error::Lease {
854            reason: format!("cannot render the lease: {e}"),
855        })?;
856        fs::write(&tmp, text).map_err(|e| Error::io(&tmp, e))?;
857        fs::rename(&tmp, &self.path).map_err(|e| Error::io(&self.path, e))?;
858        if let Ok(mut slot) = published().lock() {
859            *slot = Some(document);
860        }
861        Ok(())
862    }
863
864    /// Is the file on disk still this process's lease?
865    fn is_mine(&self) -> bool {
866        LeaseFile::read(&self.path)
867            .document()
868            .is_some_and(|d| d.token == self.token)
869    }
870
871    /// Remove the lease: the server has stopped.
872    pub fn release(mut self) {
873        self.release_now();
874    }
875
876    fn release_now(&mut self) {
877        if let Ok(mut binding) = self.binding.lock() {
878            *binding = false;
879        }
880        if !self.released && self.is_mine() {
881            let _ = fs::remove_file(&self.path);
882        }
883        self.released = true;
884        if let Ok(mut slot) = published().lock() {
885            if slot.as_ref().is_some_and(|d| d.token == self.token) {
886                *slot = None;
887            }
888        }
889        signals::release();
890    }
891}

893impl Drop for Lease {
894    fn drop(&mut self) {
895        self.release_now();
896    }
897}

899/// The lease is removed when the process dies of `SIGTERM`, `SIGINT` or `SIGHUP`: a client
900/// killing its server, a person pressing Ctrl-C, a terminal closing. The handler does only
901/// what is safe inside a signal handler: `unlink` the path recorded when the lease was
902/// taken, restore the default disposition, and raise the signal again so that the exit
903/// status still says which signal it was. The path is recorded once per process (a
904/// repository's lease never moves) and is never freed, because a handler may be reading
905/// it. A `kill -9` cannot be caught: the next process finds the stale lease and takes it
906/// over.
907#[cfg(unix)]
908mod signals {
909    use std::ffi::CString;
910    use std::os::unix::ffi::OsStrExt;
911    use std::path::Path;
912    use std::sync::atomic::{AtomicBool, AtomicPtr, Ordering};
913    use std::sync::Once;
914
915    static HELD: AtomicBool = AtomicBool::new(false);
916    static PATH: AtomicPtr<libc::c_char> = AtomicPtr::new(std::ptr::null_mut());
917    static INSTALL: Once = Once::new();
918
919    /// This process now holds the lease at `path`; install the handlers the first time.
920    pub fn hold(path: &Path) {
921        let Ok(c) = CString::new(path.as_os_str().as_bytes()) else {
922            return;
923        };
924        let raw = c.into_raw();
925        if PATH
926            .compare_exchange(
927                std::ptr::null_mut(),
928                raw,
929                Ordering::SeqCst,
930                Ordering::SeqCst,
931            )
932            .is_err()
933        {
934            // the path is already recorded (the same repository, taken over again): this
935            // copy is not needed, and the recorded one stays reachable for the handler
936            drop(unsafe { CString::from_raw(raw) });
937        }
938        HELD.store(true, Ordering::SeqCst);
939        INSTALL.call_once(|| {
940            let handler = on_signal as extern "C" fn(libc::c_int) as libc::sighandler_t;
941            for signal in [libc::SIGTERM, libc::SIGINT, libc::SIGHUP] {
942                // SAFETY: installing a handler that only calls async-signal-safe functions
943                unsafe { libc::signal(signal, handler) };
944            }
945        });
946    }
947
948    /// The lease is released (or was never this process's any more): stop removing it.
949    pub fn release() {
950        HELD.store(false, Ordering::SeqCst);
951    }
952
953    extern "C" fn on_signal(signal: libc::c_int) {
954        if HELD.load(Ordering::SeqCst) {
955            let path = PATH.load(Ordering::SeqCst);
956            if !path.is_null() {
957                // SAFETY: a valid NUL-terminated path that is never freed; unlink is
958                // async-signal-safe
959                unsafe { libc::unlink(path) };
960            }
961        }
962        // SAFETY: restoring the default disposition and re-raising are async-signal-safe
963        unsafe {
964            libc::signal(signal, libc::SIG_DFL);
965            libc::raise(signal);
966        }
967    }
968}

971mod signals {
972    use std::path::Path;
973
974    pub fn hold(_: &Path) {}
975
976    pub fn release() {}
