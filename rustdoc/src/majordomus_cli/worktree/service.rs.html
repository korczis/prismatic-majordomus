1//! The worktree service: every decision about where a worktree belongs, what stands where,
2//! whether a call is where it should be, and what may be created or removed.
3//!
4//! This is the only implementation. The command line renders what it answers, the
5//! capability registry projects its read-only questions to MCP, HTTP, OpenAPI and the
6//! Cockpit, the git hooks ask its guard, and the migration in [`super::migrate`] moves what
7//! it judged misplaced. None of them decides anything; a second decision somewhere else is
8//! the defect this file exists to prevent.
9//!
10//! One operation touches the network: creating a branch that does not exist here fetches
11//! the trunk, and that branch, from the trunk's remote, so that new work starts from the
12//! remote's trunk as it is now and a branch already pushed is taken up rather than started
13//! again (see [`StartPoint`]). The fetch is bounded and quiet, and a remote that cannot be
14//! read is said, with how far behind the local trunk stands. Nothing else fetches: a ref
15//! that does not resolve locally does not resolve, and every other operation works with
16//! the machine offline.

58/// Where a new branch started, and why there.
59///
60/// The local trunk of a long-lived checkout is routinely hundreds of commits behind its
61/// remote, because nothing advances it; a branch cut from it carries none of that work and
62/// every worker had to reset it by hand. So the default is the remote's trunk, fetched at
63/// the moment of creation, and the local trunk only when the remote cannot be read — said,
64/// never silent.
65///
66/// ```
67/// use majordomus_cli::worktree::StartPoint;
68/// let p = StartPoint::RemoteTrunk { remote: "origin".into(), branch: "master".into() };
69/// assert_eq!(p.reference(), "origin/master");
70/// assert!(p.fallback().is_none());
71/// let fell = StartPoint::LocalTrunk {
72///     branch: "master".into(),
73///     unreachable: Some("origin".into()),
74///     reason: Some("no answer within 60s".into()),
75///     behind: Some(779),
76/// };
77/// assert_eq!(fell.reference(), "master");
78/// assert!(fell.fallback().unwrap().contains("779 commits behind origin/master"));
79/// ```
80#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
81#[serde(tag = "from", rename_all = "snake_case")]
82pub enum StartPoint {
83    /// The ref `--base` named, as given; never fetched.
84    Given {
85        /// The ref.
86        reference: String,
87    },
88    /// The trunk on its remote, fetched by this command.
89    RemoteTrunk {
90        /// The remote.
91        remote: String,
92        /// The trunk branch.
93        branch: String,
94    },
95    /// The branch already on the remote: the local branch is created at it and tracks it.
96    RemoteBranch {
97        /// The remote.
98        remote: String,
99        /// The branch, the same name on both sides.
100        branch: String,
101        /// Whether it was fetched now; `false` is the remote-tracking ref as last fetched,
102        /// because the remote could not be read.
103        fetched: bool,
104    },
105    /// The local trunk: the trunk has no remote, or its remote could not be read.
106    LocalTrunk {
107        /// The trunk branch.
108        branch: String,
109        /// The remote that could not be read; `None` when the trunk has no remote.
110        #[serde(default, skip_serializing_if = "Option::is_none")]
111        unreachable: Option<String>,
112        /// Why it could not be read.
113        #[serde(default, skip_serializing_if = "Option::is_none")]
114        reason: Option<String>,
115        /// How many commits the remote's trunk, as last fetched, has that the local trunk
116        /// lacks; `None` when it was never fetched.
117        #[serde(default, skip_serializing_if = "Option::is_none")]
118        behind: Option<u64>,
119    },
120    /// HEAD, because no trunk is known.
121    Head,
122}
123
124impl StartPoint {
125    /// The ref git is given to start the branch from.
126    ///
127    /// ```
128    /// use majordomus_cli::worktree::StartPoint;
129    ///
130    /// let remote = StartPoint::RemoteTrunk { remote: "origin".into(), branch: "master".into() };
131    /// assert_eq!(remote.reference(), "origin/master");
132    /// assert_eq!(StartPoint::Head.reference(), "HEAD");
133    /// ```
134    pub fn reference(&self) -> String {
135        match self {
136            StartPoint::Given { reference } => reference.clone(),
137            StartPoint::RemoteTrunk { remote, branch }
138            | StartPoint::RemoteBranch { remote, branch, .. } => format!("{remote}/{branch}"),
139            StartPoint::LocalTrunk { branch, .. } => branch.clone(),
140            StartPoint::Head => "HEAD".into(),
141        }
142    }
143
144    /// The sentence a person must read when the start is not the remote's state as it is
145    /// now; `None` when it is, or when nothing better could have been asked.
146    ///
147    /// ```
148    /// use majordomus_cli::worktree::StartPoint;
149    ///
150    /// let stale = StartPoint::LocalTrunk {
151    ///     branch: "master".into(),
152    ///     unreachable: Some("origin".into()),
153    ///     reason: Some("offline".into()),
154    ///     behind: Some(3),
155    /// };
156    /// let said = stale.fallback().expect("a stale start is said");
157    /// assert!(said.contains("3 commits behind origin/master"), "{said}");
158    /// let remote = StartPoint::RemoteTrunk { remote: "origin".into(), branch: "master".into() };
159    /// assert_eq!(remote.fallback(), None);
160    /// ```
161    pub fn fallback(&self) -> Option<String> {
162        match self {
163            StartPoint::LocalTrunk {
164                branch,
165                unreachable: Some(remote),
166                reason,
167                behind,
168            } => {
169                let why = reason.as_deref().unwrap_or("it did not answer");
170                let distance = match behind {
171                    Some(0) => format!("level with {remote}/{branch} as last fetched"),
172                    Some(1) => format!("1 commit behind {remote}/{branch} as last fetched"),
173                    Some(n) => format!("{n} commits behind {remote}/{branch} as last fetched"),
174                    None => format!("{remote}/{branch} was never fetched, so how far behind is unknown"),
175                };
176                Some(format!(
177                    "could not fetch {branch} from {remote} ({why}): started from the local {branch}, {distance}; \
178                     once {remote} answers, `git fetch {remote} && git reset --hard {remote}/{branch}` in the worktree before the first commit"
179                ))
180            }
181            StartPoint::RemoteBranch {
182                remote,
183                branch,
184                fetched: false,
185            } => Some(format!(
186                "could not fetch {branch} from {remote}: tracking {remote}/{branch} as last fetched, which may be behind"
187            )),
188            _ => None,
189        }
190    }
191}

231/// The remote the trunk belongs to: the one whose HEAD names it, else the one the trunk's
232/// configuration tracks. `None` when the trunk lives only here.
233fn trunk_remote(primary: &Path, trunk: &str) -> Result<Option<String>> {
234    let remotes = git::try_run(primary, &["remote"])?;
235    if remotes.status != Some(0) {
236        return Ok(None);
237    }
238    let remotes: Vec<String> = remotes
239        .text()?
240        .lines()
241        .map(str::trim)
242        .filter(|r| !r.is_empty())
243        .map(str::to_string)
244        .collect();
245    for remote in &remotes {
246        // A remote with no HEAD, or one git cannot be asked about, names no trunk.
247        let head = git::try_run(
248            primary,
249            &["symbolic-ref", "-q", &format!("refs/remotes/{remote}/HEAD")],
250        )
251        .ok()
252        .filter(|head| head.status == Some(0));
253        if let Some(head) = head {
254            if head.text()? == format!("refs/remotes/{remote}/{trunk}") {
255                return Ok(Some(remote.clone()));
256            }
257        }
258    }
259    let configured = git::try_run(
260        primary,
261        &["config", "--get", &format!("branch.{trunk}.remote")],
262    )?;
263    if configured.status == Some(0) {
264        let remote = configured.text()?;
265        if remotes.contains(&remote) {
266            return Ok(Some(remote));
267        }
268    }
269    Ok(None)
270}

399    /// Everything the topology says about one registered work tree. The one place a
400    /// standing is decided.
401    pub fn judge(
402        &self,
403        record: &WorktreeRecord,
404        detail: Detail,
405        branches: &BTreeMap<String, BranchRef>,
406    ) -> WorktreeState {
407        let dirty = Self::measures_dirty(record, detail)
408            .then(|| state::dirty_state(&record.path).ok())
409            .flatten();
410        self.judge_with(record, branches, dirty)
411    }
412
413    /// Does the topology take a `git status` of this work tree at this detail? The one
414    /// place the condition is written, because the whole topology takes these measurements
415    /// at once ([`state::dirty_states`]) and the batch must select exactly the work trees
416    /// [`Self::judge`] would have measured one at a time.
417    fn measures_dirty(record: &WorktreeRecord, detail: Detail) -> bool {
418        detail == Detail::Full && record.path.is_dir() && !record.bare
419    }
420
421    /// [`Self::judge`] with the uncommitted work already measured — or deliberately not
422    /// measured, which is what `None` means here, exactly as it does in the answer.
423    fn judge_with(
424        &self,
425        record: &WorktreeRecord,
426        branches: &BTreeMap<String, BranchRef>,
427        dirty: Option<DirtyState>,
428    ) -> WorktreeState {
429        let resolved = ResolvedPath::of(&record.path);
430        let is_primary = self.identity.is_primary(record);
431        let exists = record.path.is_dir();
432        let trunk = self.identity.trunk();
433        let mut diagnostics = Vec::new();
434        let path_text = display(&record.path);
435
436        let label = match (&record.branch, &record.head) {
437            (Some(b), _) => b.clone(),
438            (None, Some(h)) => path::detached_label(h),
439            (None, None) => "(unborn)".to_string(),
440        };
441
442        // where the branch belongs, if it has one and the name derives
443        let expected: Option<PathBuf> = match record.branch.as_deref() {
444            Some(b) => match BranchName::parse(b)
445                .and_then(|n| path::expected_path(&self.container.path, &n))
446            {
447                Ok(p) => Some(p),
448                Err(e) => {
449                    diagnostics.push(TopologyDiagnostic {
450                        code: match e {
451                            WorktreeError::PathEscape { .. } => DiagnosticCode::PathEscape,
452                            _ => DiagnosticCode::InvalidBranchName,
453                        },
454                        severity: Severity::Error,
455                        path: Some(path_text.clone()),
456                        branch: Some(b.to_string()),
457                        expected: None,
458                        message: e.to_string(),
459                        remedy: "rename the branch (git branch -m) to a name git and the filesystem both accept".into(),
460                    });
461                    None
462                }
463            },
464            None => None,
465        };
466        let expected_text = expected.as_deref().map(display);
467
468        let standing = if is_primary {
469            if let Some(b) = record.branch.as_deref() {
470                if let Some(t) = trunk.branch.as_deref() {
471                    if b != t {
472                        diagnostics.push(TopologyDiagnostic {
473                            code: DiagnosticCode::PrimaryOnNonTrunk,
474                            severity: Severity::Error,
475                            path: Some(path_text.clone()),
476                            branch: Some(b.to_string()),
477                            expected: expected_text.clone(),
478                            message: format!(
479                                "the primary checkout holds '{b}', and the primary checkout hosts the trunk '{t}'"
480                            ),
481                            remedy: format!(
482                                "when the primary checkout is clean: git switch {t}; then majordomus worktree create {b}"
483                            ),
484                        });
485                    }
486                }
487            }
488            Standing::Primary
489        } else if record.prunable.is_some() || !exists {
490            diagnostics.push(TopologyDiagnostic {
491                code: if record.prunable.is_some() {
492                    DiagnosticCode::StaleRegistration
493                } else {
494                    DiagnosticCode::Missing
495                },
496                severity: Severity::Warning,
497                path: Some(path_text.clone()),
498                branch: record.branch.clone(),
499                expected: expected_text.clone(),
500                message: match &record.prunable {
501                    Some(reason) if !reason.is_empty() => {
502                        format!("git reports it prunable: {reason}")
503                    }
504                    _ => "the directory git registered does not exist".into(),
505                },
506                remedy: "majordomus worktree repair".into(),
507            });
508            Standing::Missing
509        } else if record.branch.is_none() {
510            diagnostics.push(TopologyDiagnostic {
511                code: DiagnosticCode::Detached,
512                severity: Severity::Info,
513                path: Some(path_text.clone()),
514                branch: None,
515                expected: None,
516                message: "HEAD is detached: no branch, so no canonical path; nothing moves it"
517                    .into(),
518                remedy: "git switch -c <branch> there, then majordomus worktree migrate".into(),
519            });
520            Standing::Detached
521        } else if let Some(root) = self.scratch_root_of(&resolved) {
522            diagnostics.push(TopologyDiagnostic {
523                code: DiagnosticCode::Ephemeral,
524                severity: Severity::Warning,
525                path: Some(path_text.clone()),
526                branch: record.branch.clone(),
527                expected: expected_text.clone(),
528                message: format!(
529                    "a session's scratch checkout under {}{}, holding branch '{}' which belongs at {}; nothing moves it while the session that made it may be running",
530                    display(&root.path),
531                    root.provider
532                        .as_deref()
533                        .map(|p| format!(" (created by {p})"))
534                        .unwrap_or_default(),
535                    label,
536                    expected_text.as_deref().unwrap_or("-")
537                ),
538                remedy: format!(
539                    "detach it (git switch --detach) and continue in the canonical worktree (majordomus worktree create {}); or majordomus worktree migrate --include-ephemeral --only {}",
540                    label, label
541                ),
542            });
543            Standing::Ephemeral
544        } else if resolved.same_as(&self.container) {
545            diagnostics.push(TopologyDiagnostic {
546                code: DiagnosticCode::ContainerOccupied,
547                severity: Severity::Error,
548                path: Some(path_text.clone()),
549                branch: record.branch.clone(),
550                expected: expected_text.clone(),
551                message: format!(
552                    "it occupies the container {} itself; the container holds worktrees and cannot be one",
553                    display(&self.container.path)
554                ),
555                remedy: "majordomus worktree migrate".into(),
556            });
557            Standing::Misplaced
558        } else if expected.as_deref().is_some_and(|e| {
559            ResolvedPath::of(e).same_as(&resolved) && !symlink_below(&self.container.path, e)
560        }) {
561            Standing::Canonical
562        } else if expected.is_some() {
563            diagnostics.push(TopologyDiagnostic {
564                code: DiagnosticCode::PathMismatch,
565                severity: Severity::Error,
566                path: Some(path_text.clone()),
567                branch: record.branch.clone(),
568                expected: expected_text.clone(),
569                message: format!(
570                    "branch '{}' belongs at {}",
571                    label,
572                    expected_text.as_deref().unwrap_or("-")
573                ),
574                remedy: "majordomus worktree migrate".into(),
575            });
576            let inside_primary = resolved.is_inside(self.identity.primary_worktree());
577            let inside_other = self
578                .identity
579                .registered_worktrees()
580                .iter()
581                .filter(|r| r.path != record.path)
582                .any(|r| resolved.is_inside(&ResolvedPath::of(&r.path)));
583            if inside_primary || inside_other {
584                diagnostics.push(TopologyDiagnostic {
585                    code: DiagnosticCode::Nested,
586                    severity: Severity::Warning,
587                    path: Some(path_text.clone()),
588                    branch: record.branch.clone(),
589                    expected: expected_text.clone(),
590                    message: if inside_primary {
591                        "it sits inside the primary checkout".into()
592                    } else {
593                        "it sits inside another worktree of this repository".into()
594                    },
595                    remedy: "majordomus worktree migrate".into(),
596                });
597            }
598            Standing::Misplaced
599        } else {
600            // a branch whose name did not derive: already diagnosed above
601            Standing::Misplaced
602        };
603
604        if standing == Standing::Misplaced {
605            if let Some(reason) = &record.locked {
606                diagnostics.push(TopologyDiagnostic {
607                    code: DiagnosticCode::Locked,
608                    severity: Severity::Warning,
609                    path: Some(path_text.clone()),
610                    branch: record.branch.clone(),
611                    expected: expected_text.clone(),
612                    message: if reason.is_empty() {
613                        "it is locked, so git will not move it".into()
614                    } else {
615                        format!("it is locked ({reason}), so git will not move it")
616                    },
617                    remedy: format!("git worktree unlock {path_text}"),
618                });
619            }
620            if let Some(e) = expected.as_deref() {
621                if let Some(d) = self.destination_conflict(e, Some(record)) {
622                    diagnostics.push(d);
623                }
624            }
625        }
626
627        if !is_primary && record.branch.as_deref().is_some_and(|b| trunk.is(b)) {
628            diagnostics.push(TopologyDiagnostic {
629                code: DiagnosticCode::TrunkInLinkedWorktree,
630                severity: Severity::Warning,
631                path: Some(path_text.clone()),
632                branch: record.branch.clone(),
633                expected: Some(display(&self.identity.primary_worktree().path)),
634                message: "the trunk is checked out in a linked worktree; the primary checkout hosts it".into(),
635                remedy: "switch the primary checkout to the trunk and remove this worktree when it is clean".into(),
636            });
637        }
638
639        WorktreeState {
640            path: path_text,
641            kind: if is_primary {
642                WorktreeKind::Primary
643            } else {
644                WorktreeKind::Linked
645            },
646            standing,
647            branch: record.branch.clone(),
648            label,
649            head: record.head.clone(),
650            detached: record.detached,
651            expected_path: if is_primary && record.branch.as_deref().is_some_and(|b| trunk.is(b)) {
652                None
653            } else {
654                expected_text
655            },
656            exists,
657            current: self.identity.is_current(record),
658            locked: record.locked.clone(),
659            prunable: record.prunable.clone(),
660            dirty,
661            upstream: record
662                .branch
663                .as_deref()
664                .and_then(|b| branches.get(b))
665                .and_then(|b| b.upstream.clone()),
666            issue: record
667                .branch
668                .as_deref()
669                .and_then(|b| state::issue_of(b, &self.issue_ids)),
670            diagnostics,
671        }
672    }

674    /// The scratch root a path lies under, when it does. The roots are the declarations'
675    /// (`share/providers.yaml`): the tool's own temporary directories and every provider's,
676    /// such as `.claude/worktrees/` under the primary checkout, which Claude Code creates and
677    /// removes with the session that asked for it, or the thread directory an orchestrator
678    /// keeps under its data directory. A root the primary checkout itself lives under is
679    /// skipped, as a test fixture's temporary directory is: nothing beside the repository
680    /// is a scratch checkout of it.
681    pub fn scratch_root_of(&self, path: &ResolvedPath) -> Option<ScratchRoot> {
682        let primary = self.identity.primary_worktree();
683        for root in &self.scratch_roots {
684            let resolved = ResolvedPath::of(root.path.clone());

686                continue;
687            }
688            if path.is_inside(&resolved) {
689                return Some(ScratchRoot {
690                    provider: root.provider.clone(),
691                    path: resolved.path,
692                });
693            }
694        }
695        None
696    }

703    /// What occupies `expected`, when something other than `own` does.
704    pub(crate) fn destination_conflict(
705        &self,
706        expected: &Path,
707        own: Option<&WorktreeRecord>,
708    ) -> Option<TopologyDiagnostic> {
709        let meta = std::fs::symlink_metadata(expected).ok()?;
710        let resolved = ResolvedPath::of(expected);
711        // A link is a conflict whatever it points at — including this very worktree: the
712        // canonical path has to be the directory, not a name for it somewhere else.
713        if !meta.file_type().is_symlink() {
714            if let Some(r) = self.identity.record_at(&resolved) {
715                if own.is_some_and(|o| o.path == r.path) {
716                    return None;
717                }
718                return Some(TopologyDiagnostic {
719                code: DiagnosticCode::DestinationConflict,
720                severity: Severity::Error,
721                path: own.map(|o| display(&o.path)),
722                branch: own.and_then(|o| o.branch.clone()),
723                expected: Some(display(expected)),
724                message: format!(
725                    "the canonical path is occupied by a registered worktree of this repository holding {}",
726                    r.branch.clone().unwrap_or_else(|| "a detached HEAD".into())
727                ),
728                remedy: "move that worktree to its own canonical path first (majordomus worktree migrate)".into(),
729            });
730            }
731        }
732        let what = if meta.file_type().is_symlink() {
733            "a symbolic link".to_string()
734        } else if meta.is_dir() {
735            if expected.join(".git").exists() {
736                "a git checkout this repository has not registered".to_string()
737            } else if std::fs::read_dir(expected)
738                .map(|mut d| d.next().is_none())
739                .unwrap_or(false)
740            {
741                "an empty directory".to_string()
742            } else {
743                "an unrelated directory with content".to_string()
744            }
745        } else {
746            "a file".to_string()
747        };
748        Some(TopologyDiagnostic {
749            code: DiagnosticCode::DestinationConflict,
750            severity: Severity::Error,
751            path: own.map(|o| display(&o.path)),
752            branch: own.and_then(|o| o.branch.clone()),
753            expected: Some(display(expected)),
754            message: format!("the canonical path is occupied by {what}; nothing overwrites it"),
755            remedy: format!(
756                "inspect and move it aside by hand: ls -la {}",
757                display(expected)
758            ),
759        })
760    }

764    /// The whole topology.
765    pub fn topology(&self, detail: Detail) -> Result<RepositoryTopology> {
766        let primary = &self.identity.primary_worktree().path;
767        let trunk = self.identity.trunk();
768        let branch_refs = state::branches(primary)?;
769        let branches: BTreeMap<String, BranchRef> = branch_refs
770            .iter()
771            .map(|b| (b.name.clone(), b.clone()))
772            .collect();
773        let merged: Option<BTreeSet<String>> = match trunk.branch.as_deref() {
774            Some(t) => Some(state::merged_into_trunk(primary, t)?),
775            None => None,
776        };
777
778        // Every `git status` this topology needs, taken at once rather than one work tree
779        // after the next: the measurements do not depend on one another, and on a
780        // repository with a hundred-odd registered work trees the sequence was most of
781        // what a topology cost. Judging itself runs no subprocess.
782        let records = self.identity.registered_worktrees();
783        let probes: Vec<Option<&Path>> = records
784            .iter()
785            .map(|r| Self::measures_dirty(r, detail).then_some(r.path.as_path()))
786            .collect();
787        let mut worktrees: Vec<WorktreeState> = records
788            .iter()
789            .zip(state::dirty_states(&probes))
790            .map(|(r, d)| self.judge_with(r, &branches, d))
791            .collect();
792
793        // repository-wide facts
794        let mut repository_diagnostics = Vec::new();
795        if trunk.branch.is_none() {
796            repository_diagnostics.push(TopologyDiagnostic {
797                code: DiagnosticCode::TrunkUnknown,
798                severity: Severity::Warning,
799                path: None,
800                branch: None,
801                expected: None,
802                message: "the trunk could not be determined: no remote HEAD, no init.defaultBranch, neither main nor master alone, and the primary checkout is detached".into(),
803                remedy: "git remote set-head origin --auto, or git config init.defaultBranch <branch>".into(),
804            });
805        }
806        let mut by_folded: BTreeMap<String, Vec<String>> = BTreeMap::new();
807        for w in &worktrees {
808            if let (Some(e), Some(b)) = (&w.expected_path, &w.branch) {
809                by_folded
810                    .entry(e.to_lowercase())
811                    .or_default()
812                    .push(b.clone());
813            }
814        }
815        for (_, names) in by_folded.iter().filter(|(_, n)| n.len() > 1) {
816            repository_diagnostics.push(TopologyDiagnostic {
817                code: DiagnosticCode::CaseCollision,
818                severity: Severity::Warning,
819                path: None,
820                branch: Some(names.join(", ")),
821                expected: None,
822                message: "these branch names derive paths that differ only by case, which one directory on a case-insensitive filesystem".into(),
823                remedy: "rename one of the branches (git branch -m)".into(),
824            });
825        }
826
827        let by_path: BTreeMap<String, &WorktreeState> =
828            worktrees.iter().map(|w| (w.path.clone(), w)).collect();
829        let mut branch_states: Vec<BranchState> = branch_refs
830            .iter()
831            .map(|b| {
832                let is_trunk = trunk.is(&b.name);
833                let worktree_path = b.worktree.as_deref().map(display);
834                let holder = worktree_path.as_deref().and_then(|p| by_path.get(p));
835                let merged_into_trunk = merged.as_ref().map(|m| m.contains(&b.name));
836                // A session's scratch checkout is never cleanup-eligible however merged and
837                // clean: the harness that made it removes it, and a listing that offered it
838                // would offer another session's floor.
839                let cleanup_eligible = !is_trunk
840                    && merged_into_trunk == Some(true)
841                    && match holder {
842                        None => true,
843                        Some(h) => {
844                            h.standing != Standing::Ephemeral
845                                && h.dirty.as_ref().is_some_and(|d| d.clean)
846                        }
847                    };
848                BranchState {
849                    name: b.name.clone(),
850                    head: b.head.clone(),
851                    trunk: is_trunk,
852                    expected_path: if is_trunk {
853                        None
854                    } else {
855                        self.expected_path_of(&b.name).ok().map(|p| display(&p))
856                    },
857                    worktree: worktree_path,
858                    upstream: b.upstream.clone(),
859                    merged_into_trunk,
860                    cleanup_eligible,
861                    issue: state::issue_of(&b.name, &self.issue_ids),
862                }
863            })
864            .collect();
865        crate::order::canonical(&mut branch_states);
866
867        let mut tallies = TopologyTallies {
868            worktrees: worktrees.len(),
869            branches: branch_states.len(),
870            branches_without_worktree: branch_states
871                .iter()
872                .filter(|b| b.worktree.is_none())
873                .count(),
874            cleanup_eligible: branch_states.iter().filter(|b| b.cleanup_eligible).count(),
875            ..TopologyTallies::default()
876        };
877        for w in &worktrees {
878            match w.standing {
879                Standing::Canonical => tallies.canonical += 1,
880                Standing::Misplaced => tallies.misplaced += 1,
881                Standing::Detached => tallies.detached += 1,
882                Standing::Ephemeral => tallies.ephemeral += 1,
883                Standing::Missing => tallies.missing += 1,
884                Standing::Primary => {}
885            }
886            if w.locked.is_some() {
887                tallies.locked += 1;
888            }
889            if w.dirty.as_ref().is_some_and(|d| !d.clean) {
890                tallies.dirty += 1;
891            }
892        }
893        let mut diagnostics: Vec<TopologyDiagnostic> = worktrees
894            .iter_mut()
895            .flat_map(|w| w.diagnostics.clone())
896            .collect();
897        diagnostics.extend(repository_diagnostics);
898        tallies.errors = diagnostics
899            .iter()
900            .filter(|d| d.severity == Severity::Error)
901            .count();
902        tallies.warnings = diagnostics
903            .iter()
904            .filter(|d| d.severity == Severity::Warning)
905            .count();
906
907        Ok(RepositoryTopology {
908            schema: SCHEMA.to_string(),
909            repository: self.repository_view(),
910            container: self.container_view(),
911            trunk: self.trunk_view(),
912            observed_from: display(&self.identity.current_worktree().path),
913            valid: tallies.errors == 0,
914            worktrees,
915            branches: branch_states,
916            diagnostics,
917            tallies,
918        })
919    }

1039    fn create_inner(&self, request: &CreateRequest, ensure: bool) -> Result<CreateReport> {
1040        let name = BranchName::parse(&request.branch)?;
1041        // a branch with no place in the container is refused before the network is asked
1042        let target = path::expected_path(&self.container.path, &name)?;
1043        let primary = self.identity.primary_worktree().path.clone();
1044
1045        // The network is asked before the lock is taken, so a slow remote never holds every
1046        // other create in this repository behind it; the decision is re-checked under it.
1047        // A branch git cannot be asked about here is asked about again under the lock.
1048        let unknown_here = matches!(git::branch_exists(&primary, name.as_str()), Ok(false));
1049        let planned = if request.base.is_none() && unknown_here {
1050            Some(self.plan_start(&primary, name.as_str())?)
1051        } else {
1052            None
1053        };
1054        self.create_locked(request, ensure, name, target, planned)
1055    }

1057    /// The create under the lock, given the branch and its place as `create_inner` checked
1058    /// them and what was planned before it: `planned` is `None` when
1059    /// the branch existed (or could not be asked about) before the lock, and the start is
1060    /// then planned here if the branch is gone by the time the lock is held.
1061    fn create_locked(
1062        &self,
1063        request: &CreateRequest,
1064        ensure: bool,
1065        name: BranchName,
1066        target: PathBuf,
1067        planned: Option<StartPoint>,
1068    ) -> Result<CreateReport> {
1069        let resolved_target = ResolvedPath::of(&target);
1070        let primary = self.identity.primary_worktree().path.clone();
1071
1072        // One lock around the whole check-then-act, and the topology re-read under it.
1073        let _lock = WorktreeLock::acquire(&self.identity.git_common_dir().path)?;
1074        let mut identity = self.identity.clone();
1075        identity.refresh()?;
1076
1077        if let Some(existing) = identity.record_of_branch(name.as_str()) {
1078            if ResolvedPath::of(&existing.path).same_as(&resolved_target) {
1079                if ensure {
1080                    // Found rather than made, and approved all the same: a worktree that
1081                    // exists and is blocked is what `ensure` is most often asked about.
1082                    let envrc = direnv::approve(&primary, &existing.path);
1083                    let envrc_local = direnv::link_local(&primary, &existing.path);
1084                    let hooks = hooks::carry_hooks_path(&primary, &existing.path);
1085                    return Ok(CreateReport {
1086                        path: display(&existing.path),
1087                        branch: name.as_str().to_string(),
1088                        branch_created: false,
1089                        base: None,
1090                        start: None,
1091                        start_commit: None,
1092                        container: display(&self.container.path),
1093                        container_created: false,
1094                        existed: true,
1095                        envrc,
1096                        envrc_local,
1097                        hooks,
1098                    });
1099                }
1100                return Err(WorktreeError::WorktreeAlreadyExists {
1101                    path: existing.path.clone(),
1102                    branch: name.as_str().to_string(),
1103                });
1104            }
1105            return Err(WorktreeError::BranchAlreadyCheckedOut {
1106                branch: name.as_str().to_string(),
1107                path: existing.path.clone(),
1108                expected: target,
1109            });
1110        }
1111        if let Some(d) = self.destination_conflict(&target, None) {
1112            return Err(WorktreeError::DestinationConflict {
1113                path: target,
1114                what: d.message,
1115            });
1116        }
1117        if !git::branch_name_is_valid(&primary, name.as_str())? {
1118            return Err(WorktreeError::InvalidBranchName {
1119                given: name.as_str().to_string(),
1120                reason: "git does not accept it as a reference name".into(),
1121            });
1122        }
1123        let branch_exists = git::branch_exists(&primary, name.as_str())?;
1124        let start = if branch_exists {
1125            None
1126        } else {
1127            let start = match (&request.base, planned) {
1128                (Some(given), _) => StartPoint::Given {
1129                    reference: given.clone(),
1130                },
1131                (None, Some(planned)) => planned,
1132                (None, None) => self.plan_start(&primary, name.as_str())?,
1133            };
1134            let base = start.reference();
1135            if !git::rev_exists(&primary, &base)? {
1136                return Err(WorktreeError::BaseDoesNotExist { base });
1137            }
1138            Some(start)
1139        };
1140        let base = start.as_ref().map(StartPoint::reference);
1141
1142        let container_created = self.ensure_container()?;
1143        if let Some(parent) = target.parent() {
1144            std::fs::create_dir_all(parent).map_err(|e| WorktreeError::io(parent, &e))?;
1145        }
1146        let target_text = display(&target);
1147        let mut args: Vec<String> = vec!["worktree".into(), "add".into()];
1148        match (&start, &base) {
1149            (None, _) | (_, None) => {
1150                args.push("--".into());
1151                args.push(target_text.clone());
1152                args.push(name.as_str().to_string());
1153            }
1154            (Some(start), Some(base)) => {
1155                match start {
1156                    // The same branch on both sides: it is the remote's, and tracks it.
1157                    StartPoint::RemoteBranch { .. } => args.push("--track".into()),
1158                    // New work cut from the remote's trunk does not track the trunk; its
1159                    // upstream is its own branch, set when it is first pushed.
1160                    StartPoint::RemoteTrunk { .. } => args.push("--no-track".into()),
1161                    _ => {}
1162                }
1163                args.push("-b".into());
1164                args.push(name.as_str().to_string());
1165                args.push("--".into());
1166                args.push(target_text.clone());
1167                args.push(base.clone());
1168            }
1169        }
1170        git::run(&primary, &args)?;
1171
1172        // Verify rather than assume: the answer is what the topology says.
1173        identity.refresh()?;
1174        let registered = identity
1175            .record_at(&resolved_target)
1176            .cloned()
1177            .ok_or_else(|| WorktreeError::MigrationUnsafe {
1178                path: target.clone(),
1179                reason: "git reported success and `git worktree list` does not show the worktree; \
1180                         nothing was cleaned up because that would risk deleting real work"
1181                    .into(),
1182            })?;
1183        // The path exists now; this is the moment direnv's approval is carried to it.
1184        let envrc = direnv::approve(&primary, &registered.path);
1185        let envrc_local = direnv::link_local(&primary, &registered.path);
1186        // And the moment it is bound to the hooks of its own tree, before its first commit.
1187        let hooks = hooks::carry_hooks_path(&primary, &registered.path);
1188        let start_commit = match start {
1189            // The worktree is made; a commit git cannot read back is not a reason to say not.
1190            Some(_) => git::head_of(&registered.path).ok().flatten(),
1191            None => None,
1192        };
1193        Ok(CreateReport {
1194            path: display(&registered.path),
1195            branch: name.as_str().to_string(),
1196            branch_created: base.is_some(),
1197            base,
1198            start,
1199            start_commit,
1200            container: display(&self.container.path),
1201            container_created,
1202            existed: false,
1203            envrc,
1204            envrc_local,
1205            hooks,
1206        })
1207    }
1208
1209    /// Decide where a branch that does not exist here starts, asking the trunk's remote.
1210    ///
1211    /// The trunk and the branch are fetched from the trunk's remote. A branch the remote
1212    /// has is taken up where it stands; any other starts at the remote's trunk. A remote
1213    /// that cannot be read leaves the remote-tracking refs as last fetched: a branch known
1214    /// there is still taken up, and a new one starts from the local trunk, with how far
1215    /// behind it is. A trunk with no remote is the local trunk, and no trunk at all is HEAD.
1216    fn plan_start(&self, primary: &Path, branch: &str) -> Result<StartPoint> {
1217        let Some(trunk) = self.identity.trunk().branch.clone() else {
1218            return Ok(StartPoint::Head);
1219        };
1220        let Some(remote) = trunk_remote(primary, &trunk)? else {
1221            return Ok(StartPoint::LocalTrunk {
1222                branch: trunk,
1223                unreachable: None,
1224                reason: None,
1225                behind: None,
1226            });
1227        };
1228        let tracking = |b: &str| format!("refs/remotes/{remote}/{b}");
1229        let refspec = |b: &str| format!("+refs/heads/{b}:{}", tracking(b));
1230        let trunk_fetch =
1231            git::fetch_bounded(primary, &remote, &[refspec(&trunk)], git::FETCH_TIMEOUT);
1232        if trunk_fetch.is_ok() {
1233            match git::fetch_bounded(primary, &remote, &[refspec(branch)], git::FETCH_TIMEOUT) {
1234                Ok(()) => {
1235                    return Ok(StartPoint::RemoteBranch {
1236                        remote,
1237                        branch: branch.to_string(),
1238                        fetched: true,
1239                    })
1240                }
1241                // A remote-tracking ref left behind by a branch since deleted there is
1242                // not taken up: the remote has just said it is gone.
1243                Err(git::FetchFailure::NoSuchRemoteRef) => {
1244                    return Ok(StartPoint::RemoteTrunk {
1245                        remote,
1246                        branch: trunk,
1247                    })
1248                }
1249                Err(git::FetchFailure::Unreachable(_)) => {}
1250            }
1251        }
1252        // Either fetch failed: a branch known here as last fetched is still taken up.
1253        if git::rev_exists(primary, &tracking(branch))? {
1254            return Ok(StartPoint::RemoteBranch {
1255                remote,
1256                branch: branch.to_string(),
1257                fetched: false,
1258            });
1259        }
1260        let reason = match trunk_fetch {
1261            // The trunk arrived and only the branch could not be asked about.
1262            Ok(()) => {
1263                return Ok(StartPoint::RemoteTrunk {
1264                    remote,
1265                    branch: trunk,
1266                })
1267            }
1268            Err(git::FetchFailure::NoSuchRemoteRef) => format!("{remote} has no branch {trunk}"),
1269            Err(git::FetchFailure::Unreachable(why)) => why,
1270        };
1271        let behind = git::commits_between(primary, &trunk, &tracking(&trunk));
1272        Ok(StartPoint::LocalTrunk {
1273            branch: trunk,
1274            unreachable: Some(remote),
1275            reason: Some(reason),
1276            behind,
1277        })
1278    }

1281    /// beside the repository and is never committed, so it is created lazily, by the first
1282    /// worktree that needs it.
1283    pub(crate) fn ensure_container(&self) -> Result<bool> {
1284        let p = &self.container.path;
1285        if p.is_dir() {
1286            return Ok(false);
1287        }

1289            return Err(WorktreeError::ContainerNotADirectory { path: p.clone() });

1291        std::fs::create_dir_all(p).map_err(|e| WorktreeError::io(p, &e))?;
1292        Ok(true)

1297    /// Remove one linked worktree. Never a branch, never the primary checkout, never
1298    /// uncommitted work without `force`.
1299    pub fn remove(&self, selector: &str, force: bool) -> Result<RemoveReport> {
1300        let _lock = WorktreeLock::acquire(&self.identity.git_common_dir().path)?;
1301        let mut identity = self.identity.clone();
1302        identity.refresh()?;
1303        let record = self.resolve(selector)?;
1304        if identity.is_primary(&record) {
1305            return Err(WorktreeError::PrimaryWorktreeProtected {
1306                path: record.path.clone(),
1307                operation: "remove".into(),
1308            });
1309        }
1310        let locked = record.locked.is_some();
1311        if locked && !force {
1312            return Err(WorktreeError::LockedWorktree {
1313                path: record.path.clone(),
1314                reason: record.locked.clone(),
1315            });
1316        }
1317        let mut forced = false;
1318        if record.path.is_dir() {
1319            let dirty = state::dirty_state(&record.path)?;
1320            if !dirty.clean {
1321                if !force {
1322                    return Err(WorktreeError::DirtyWorktree {
1323                        path: record.path.clone(),
1324                        summary: dirty.summary(),
1325                        operation: "remove".into(),
1326                    });
1327                }
1328                forced = true;
1329            }
1330        }
1331        let mut args: Vec<String> = vec!["worktree".into(), "remove".into()];
1332        if forced || locked {
1333            args.push("--force".into());
1334        }
1335        if locked {
1336            args.push("--force".into());
1337        }
1338        args.push("--".into());
1339        args.push(display(&record.path));
1340        git::run(&identity.primary_worktree().path, &args)?;
1341        Ok(RemoveReport {
1342            path: display(&record.path),
1343            branch: record.branch.clone(),
1344            forced: forced || locked,
1345        })
1346    }

1350    /// Drop the registrations of work trees whose directories are gone, and let git repair
1351    /// the administrative links of the ones that exist. Deletes no directory.
1352    pub fn repair(&self, dry_run: bool) -> Result<RepairReport> {
1353        let _lock = WorktreeLock::acquire(&self.identity.git_common_dir().path)?;
1354        let primary = &self.identity.primary_worktree().path;
1355        let mut args: Vec<String> = vec!["worktree".into(), "prune".into(), "--verbose".into()];
1356        if dry_run {
1357            args.push("--dry-run".into());
1358        }
1359        let pruned = lines_of(&git::run(primary, &args)?.stdout);
1360        let repaired = if dry_run {
1361            Vec::new()
1362        } else {
1363            let mut args: Vec<String> = vec!["worktree".into(), "repair".into()];
1364            for r in self.identity.registered_worktrees().iter().skip(1) {
1365                if r.path.is_dir() {
1366                    args.push(display(&r.path));
1367                }
1368            }
1369            let out = git::run(primary, &args)?;
1370            let mut lines = lines_of(&out.stdout);
1371            lines.extend(lines_of(out.stderr.as_bytes()));
1372            lines
1373        };
1374        // With the links repaired, each linked worktree is bound to the hooks of its own
1375        // tree: the repair is how a worktree moved or made by hand rejoins the topology.
1376        let hooks = if dry_run {
1377            BTreeMap::new()
1378        } else {
1379            self.identity
1380                .registered_worktrees()
1381                .iter()
1382                .skip(1)
1383                .filter(|r| r.path.is_dir())
1384                .map(|r| (display(&r.path), hooks::carry_hooks_path(primary, &r.path)))
1385                .collect()
1386        };
1387        Ok(RepairReport {
1388            pruned,
1389            repaired,
1390            hooks,
1391            applied: !dry_run,
1392        })
1393    }

1396/// Is any component of `path` below `container` a symbolic link? A link inside the container
1397/// pointing at a worktree elsewhere would make that worktree canonicalise to its expected
1398/// path; the expected path has to be a real directory, not a name for one somewhere else.
1399fn symlink_below(container: &Path, path: &Path) -> bool {
1400    let Ok(rest) = path.strip_prefix(container) else {
1401        return false;
1402    };
1403    let mut cursor = container.to_path_buf();
1404    for c in rest.components() {
1405        cursor.push(c);
1406        if std::fs::symlink_metadata(&cursor)
1407            .map(|m| m.file_type().is_symlink())
1408            .unwrap_or(false)
1409        {
1410            return true;
1411        }
1412    }
1413    false
1414}

1416/// Whether a judged work tree is where it belongs: canonical, or the primary checkout on
1417/// the trunk (or with no trunk to be held to), or detached.
1418fn is_in_place(w: &WorktreeState, trunk_source: TrunkSource) -> bool {
1419    match w.standing {
1420        Standing::Canonical | Standing::Detached => true,
1421        Standing::Primary => {
1422            trunk_source == TrunkSource::Unknown
1423                || !w
1424                    .diagnostics
1425                    .iter()
1426                    .any(|d| d.code == DiagnosticCode::PrimaryOnNonTrunk)
1427        }
1428        Standing::Misplaced | Standing::Missing | Standing::Ephemeral => false,
1429    }

1432/// The scratch roots the distribution declares, expanded against `primary`, in declaration
1433/// order: the tool's own first, then each provider's. Located the way every command locates
1434/// the share (`--share` excepted, which this path does not carry), from `current` — the
1435/// checkout the command runs in, so that a repository supervising itself reads the
1436/// declarations of the branch it is on rather than the primary checkout's; a checkout with
1437/// no share in reach declares none. The standard library's answer for the temporary
1438/// directory is added when the declarations name `$TMPDIR` and the environment does not, so
1439/// that what the tool itself creates under `std::env::temp_dir()` is recognised on every
1440/// platform.
1441pub fn declared_scratch_roots(current: &Path, primary: &Path) -> Vec<ScratchRoot> {
1442    let Ok(share) = crate::share::Share::locate(None, current) else {
1443        return Vec::new();
1444    };
1445    let Ok(decls) = share.providers() else {
1446        return Vec::new();
1447    };
1448    scratch_roots_from(&decls, primary)
1449}
1450
1451/// [`declared_scratch_roots`] over declarations already read.
1452pub fn scratch_roots_from(
1453    decls: &crate::share::ProviderDeclarations,
1454    primary: &Path,
1455) -> Vec<ScratchRoot> {
1456    let mut out: Vec<ScratchRoot> = decls
1457        .scratch_roots(primary)
1458        .into_iter()
1459        .map(|(provider, path)| ScratchRoot { provider, path })
1460        .collect();
1461    if decls.scratch_roots.iter().any(|r| r == "$TMPDIR") {
1462        let temp = std::env::temp_dir();
1463        if !out.iter().any(|r| r.provider.is_none() && r.path == temp) {
1464            out.push(ScratchRoot {
1465                provider: None,
1466                path: temp,
1467            });
1468        }
1469    }
1470    out
1471}

1998        let p = repo.primary.clone();
1999        let svc = repo.service();
2000        let plan = |b: &str| svc.plan_start(&p, b).expect("a plan");
2001        let remote_trunk = StartPoint::RemoteTrunk {
2002            remote: "origin".into(),
2003            branch: "master".into(),
2004        };
2005
2006        // the remote has it: taken up as it is there

2008        assert_eq!(

2018
