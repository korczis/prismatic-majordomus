1//! Navigation, derived. The sidebar's catalogues are not a list in this file: the
2//! capability modules come from the registry, the object kinds come from `repository.info`,
3//! and the graphs come from the derivations. A module added to `compose_modules!`, a kind
4//! added to `share/kinds.yaml`, a graph added to the derivation table — each appears here
5//! with no edit to the Cockpit.
6//!
7//! The kinds are *asked for* rather than read: ADR 0012 says no page reads the index
8//! directly, because a reader that steps around the executor is outside the cache, the
9//! counters and the validation every other caller passes through.
10//!
11//! What *is* written here is the areas: Overview, Capabilities, Commands, Executions,
12//! Objects, Directories, Graphs, Continuity, Plan, Worktrees, Integration, Health, Quality,
13//! Artifacts, Design, API. Those are concepts rather than
14//! entities, they change when the Cockpit's own shape changes, and deriving them from
15//! anything would be deriving them from a list of exactly themselves.
16//!
17//! Every section is presented in the canonical order (`crate::order`). The order a section
18//! is *built* in is an accident of its source — the order the modules were composed in, the
19//! order the graph derivations are declared in — and an accident is not a reading order.
20//! The order is applied to the sections as a whole, so a section added later is ordered
21//! without being told to be.
22//!
23//! One section is grouped rather than flat: the capability modules sit under the
24//! operational area they serve. That area is not written here either. A feature declares
25//! the modules it is built from and the areas it serves; the product model resolves the two
26//! into an area per module, and this file asks for it. A module no feature places is shown
27//! under no heading, which the canonical order puts last, and the product model reports it.
28

34/// One of the Cockpit's areas. A page declares the area it belongs to and the navigation

268/// One entry.
269#[derive(Debug, Clone)]
270pub struct Item {
271    /// What the reader sees.
272    pub label: String,
273    /// Where it goes.
274    pub href: String,
275    /// The area it belongs to.
276    pub area: Area,
277    /// The heading it sits under inside its section, when the section is grouped. Derived,
278    /// never written here: the capability modules are grouped by the operational area the
279    /// features that name them serve.
280    pub group: Option<String>,
281    /// How many things are behind it, when the number is a fact and not decoration: `None`
282    /// for an entry that carries no count, and [`Count::Unknown`] for one whose count the
283    /// capability asked for it did not answer.
284    pub count: Option<Count>,
285    /// Whether this is the page being shown.
286    pub current: bool,
287}

289/// A count an entry shows: a number the capability answered, or the fact that it did not.
290/// A capability that failed is never a count of zero.

292pub enum Count {
293    /// The number.
294    Known(usize),
295    /// The capability behind it did not answer.
296    Unknown,
297}

321/// What the layer holds, as `repository.info` answers it: the object count and the kinds
322/// with theirs.
323///
324/// A capability that cannot answer leaves the navigation without its object counts rather
325/// than without a navigation: the sidebar is how a person reaches the page that would say
326/// what went wrong, so it is the last thing that should fail with the thing it reports on.
327/// The two fields of `repository.info` this file reads; the rest of the report is the
328/// health page's subject, and taking only these keeps the navigation's dependency on the
329/// capability to what it actually shows.
330#[derive(serde::Deserialize)]
331struct Held {
332    objects: usize,
333    kinds: std::collections::BTreeMap<String, usize>,
334}
335
336/// `None` when `repository.info` did not answer, or answered something this file cannot
337/// read: the object count is then unknown, never zero.
338fn held(ctx: &Context) -> Option<Held> {
339    let value = ctx.execute("repository.info", serde_json::json!({})).ok()?;
340    serde_json::from_value::<Held>(value).ok()
341}

343/// Build the navigation for a request: the areas, then the catalogues derived from the
344/// registry, the index and the graph derivations. `here` is the request path, so the
345/// current entry can be marked without a page saying which it is.
346pub fn build(ctx: &Context, here: &str) -> Navigation {
347    let summary = ctx.registry.summary();
348    // What the layer holds is asked for, not opened. `repository.info` answers how many
349    // objects there are and which kinds they have, and asking it is what keeps the
350    // navigation a projection instead of a second reader of the index (ADR 0012). One
351    // call answers both catalogues below.
352    let held = held(ctx);
353    build_with(ctx, here, held.as_ref(), &summary)
354}
355
356/// [`build`], over what `repository.info` answered (`None`: it did not).
357fn build_with(
358    ctx: &Context,
359    here: &str,
360    held: Option<&Held>,
361    summary: &crate::capability::registry::Summary,
362) -> Navigation {
363    // the counts are facts of this context, decided per area: how many things are behind
364    // an entry is not part of what an area is
365    let count = |a: Area| -> Option<Count> {
366        match a {
367            Area::Capabilities => Some(Count::Known(summary.total)),
368            Area::Objects => Some(held.map_or(Count::Unknown, |h| Count::Known(h.objects))),
369            Area::Graphs => Some(Count::Known(graph::ids().len())),
370            Area::Api => Some(Count::Known(summary.http_routes)),
371            Area::Mcp => Some(Count::Known(summary.mcp_tools)),
372            _ => None,
373        }
374    };
375    let areas = Section {
376        title: "Cockpit".into(),
377        items: areas()
378            .iter()
379            .map(|a| item(a.label, a.href, a.area, count(a.area), here))
380            .collect(),
381    };
382
383    // the executable's own modules: what a capability belongs to, from the registry
384    let modules = Section {
385        title: "Capability modules".into(),
386        items: ctx
387            .registry
388            .modules()
389            .filter(|m| m.source != ModuleSource::Declarative)
390            .map(|m| Item {
391                label: m.title.clone(),
392                href: format!(
393                    "/cockpit/capabilities?module={}",
394                    percent_encode(m.id.as_str())
395                ),
396                area: Area::Capabilities,
397                // The module's own area, derived by the product model from the features
398                // that name it, shown under the name the Why catalogue gives it. Neither
399                // the grouping nor the heading is written in this file.
400                group: ctx
401                    .product
402                    .module_area(m.id.as_str())
403                    .and_then(|id| ctx.why.areas().iter().find(|a| a.id == id))
404                    .map(|a| a.title.clone()),
405                count: Some(Count::Known(m.capabilities)),
406                current: false,
407            })
408            .collect(),
409    };
410
411    // the kinds of the layer: what the repository declares, not what this code knows
412    let kinds = Section {
413        title: "Object kinds".into(),
414        items: held
415            .into_iter()
416            .flat_map(|h| &h.kinds)
417            .map(|(kind, count)| Item {
418                label: kind.to_string(),
419                href: crate::entity::kind_route(kind),
420                area: Area::Objects,
421                group: None,
422                count: Some(Count::Known(*count)),
423                current: false,
424            })
425            .collect(),
426    };
427
428    let graphs = Section {
429        title: "Graphs".into(),
430        items: graph::ids()
431            .into_iter()
432            .map(|id| Item {
433                label: id.to_string(),
434                href: format!("/cockpit/graphs/{id}"),
435                area: Area::Graphs,
436                group: None,
437                count: None,
438                current: here == format!("/cockpit/graphs/{id}"),
439            })
440            .collect(),
441    };
442
443    Navigation {
444        sections: [areas, modules, kinds, graphs]
445            .into_iter()
446            .filter(|s| !s.items.is_empty())
447            .map(alphabetical)
448            .collect(),
449    }
450}

452/// One section's entries in the canonical order: by label, case-folded so that `API` sits
453/// with `Artifacts` rather than ahead of every lowercase kind, digit runs by value, and the
454/// href behind it to break a tie between two entries a reader would call the same.
455///
456/// The comparator is `crate::order`'s, not this file's. It was this file's, it was the only
457/// case-folded comparator in the crate, and every other surface sorted by raw bytes instead.
458fn alphabetical(mut section: Section) -> Section {
459    crate::order::canonical(&mut section.items);
460    section
461}
462
463impl crate::order::Ordered for Item {
464    fn order_key(&self) -> crate::order::OrderKey<'_> {
465        match &self.group {
466            Some(group) => crate::order::OrderKey::grouped(group, &self.label, &self.href),
467            None => crate::order::OrderKey::plain(&self.label, &self.href),
468        }
469    }
470}

472fn item(label: &str, href: &str, area: Area, count: Option<Count>, here: &str) -> Item {
473    Item {
474        label: label.into(),
475        href: href.into(),
476        area,
477        group: None,
478        count,
479        // a page under an area's route is in that area: an issue page is the Plan, and a
480        // sidebar that marked nothing there folded itself shut on every detail page. The
481        // overview's route is the prefix of every route, so it marks only itself.
482        current: here == href || (href != super::PREFIX && here.starts_with(&format!("{href}/"))),
483    }
484}

537        let nav = build(&ctx, "/cockpit");

539            let mut seen: Vec<&str> = Vec::new();
540            let mut ungrouped = false;
541            for item in &section.items {

543                    Some(g) => {
544                        assert!(
545                            !ungrouped,
546                            "a grouped entry follows an ungrouped one in '{}': the canonical \

551                            assert!(

553                                "the group '{g}' appears twice in '{}': its members are not \
554                                 contiguous, so one pass cannot render its heading once",
555                                section.title

558                        }
559                    }
