1//! `majordomus board`: the delivery board's documents on the command line. The subcommand
2//! runs the capability its CLI path names in-process and owns the rendering alone; the
3//! lanes, the composition and the digest live in the capability, and `--format json`
4//! prints the answer the HTTP route and the MCP tool serve.

15/// Run `majordomus board`.
16pub fn run(args: BoardArgs) -> Result<u8> {
17    let BoardCommand::Snapshot { by } = args.command;
18    // the repository and the answer fold into one result: a failure leaves by one exit
19    let value = App::load(&args.repo)
20        .and_then(|app| execute(&app.context, &["board", "snapshot"], json!({ "by": by })))?;
21    print!(
22        "{}",
23        match args.format {
24            OutputFormat::Json => format!("{value:#}\n"),
25            OutputFormat::Text => text(&value),
26        }
27    );
28    Ok(0)
29}

31/// Run the capability exposed as `majordomus <words>`, its refusal as this command's.
32fn execute(ctx: &crate::capability::Context, words: &[&str], input: Value) -> Result<Value> {
33    let words: Vec<String> = words.iter().map(|w| w.to_string()).collect();
34    let id = ctx
35        .registry
36        .by_cli(&words)
37        .map(|c| c.id.as_str().to_string())
38        .ok_or_else(|| Error::Protocol {
39            reason: format!(
40                "no capability is exposed as `majordomus {}`",
41                words.join(" ")
42            ),
43        })?;
44    ctx.execute(&id, input).map_err(refusal)
45}

47/// The exit code of a refused snapshot: an open milestone in no lane is a verdict not
48/// satisfied, 10, as `quality report` and `shell check` answer one; anything else is the
49/// executable's own failure.
50pub const EXIT_UNLANED: u8 = 10;
51
52fn refusal(e: CapabilityError) -> Error {
53    match e {
54        CapabilityError::Refused(reason) => Error::Refused {
55            code: EXIT_UNLANED,
56            reason,
57        },
58        other => Error::Protocol {
59            reason: other.to_string(),
60        },
61    }
62}

64/// The snapshot as a person reads it: the digest and commit, then one line per document.
65fn text(v: &Value) -> String {
66    let mut out = String::new();
67    let _ = writeln!(
68        out,
69        "digest     {}",
70        v["digest"].as_str().unwrap_or_default()
71    );
72    let _ = writeln!(
73        out,
74        "master     {} {}",
75        v["master_sha"].as_str().unwrap_or_default(),
76        v["master_committed_at"].as_str().unwrap_or_default()
77    );
78    for (id, doc) in v["documents"].as_object().into_iter().flatten() {
79        let _ = writeln!(
80            out,
81            "{id:<20} {:>4} item(s)  {:>7} bytes",
82            doc["items"].as_array().map_or(0, Vec::len),
83            v["sizes"][id].as_u64().unwrap_or_default()
84        );
85    }
86    for d in v["diagnostics"].as_array().into_iter().flatten() {
87        let _ = writeln!(out, "note       {}", d.as_str().unwrap_or_default());
88    }
89    out
90}

92#[cfg(test)]
93mod tests {
94    use super::*;

96    #[test]
97    fn a_refusal_keeps_its_words_and_exits_10() {
98        let e = refusal(CapabilityError::Refused("milestone m is in no lane".into()));
99        assert_eq!(e.exit_code(), EXIT_UNLANED);
100        assert_eq!(e.to_string(), "milestone m is in no lane");
101        let e = refusal(CapabilityError::Internal("broken".into()));
102        assert_eq!(e.exit_code(), 13);
103    }
104
105    #[test]
106    fn the_text_names_the_digest_and_every_document() {
107        let v = json!({
108            "digest": "sha256:ab", "master_sha": "abc", "master_committed_at": "2026-10-28",
109            "sizes": {"meta": 10, "prs": 20},
110            "documents": {"meta": {}, "prs": {"items": [1, 2]}},
111            "diagnostics": ["the pull request queue is not observed"],
112        });
113        let t = text(&v);
114        assert!(
115            t.starts_with("digest     sha256:ab\nmaster     abc 2026-10-28\n"),
116            "{t}"
117        );
118        assert!(
119            t.contains("meta                    0 item(s)       10 bytes"),
120            "{t}"
121        );
122        assert!(
123            t.contains("prs                     2 item(s)       20 bytes"),
124            "{t}"
125        );
126        assert!(
127            t.ends_with("note       the pull request queue is not observed\n"),
128            "{t}"
129        );
130    }

132    /// The command answers through the capability its words name, and words no capability
133    /// is exposed as, or a directory that holds no repository, are errors before anything runs.
134    #[test]
135    fn the_snapshot_runs_by_its_words_and_what_is_missing_is_an_error() {
136        let repo = crate::synthetic::SyntheticRepository::small().expect("a repository");
137        let ctx = repo.context().expect("a context");
138        let v = execute(&ctx, &["board", "snapshot"], json!({ "by": "t" })).expect("a snapshot");
139        assert_eq!(v["documents"]["meta"]["syncedBy"], "t");
140        let err = execute(&ctx, &["board", "nothing"], json!({})).unwrap_err();
141        assert!(
142            matches!(&err, Error::Protocol { reason }
143                if reason == "no capability is exposed as `majordomus board nothing`"),
144            "{err:?}"
145        );
146        let empty = tempfile::tempdir().expect("a directory");
147        let args = BoardArgs {
148            repo: crate::cli::RepoArgs {
149                repo: Some(empty.path().to_path_buf()),
150                discovery: crate::cli::DiscoveryMode::Filesystem,
151                strict: false,
152                share: Some(crate::synthetic::crate_share()),
153            },
154            command: BoardCommand::Snapshot { by: None },
155            format: OutputFormat::Json,
156        };
157        assert!(run(args).is_err());
158    }
