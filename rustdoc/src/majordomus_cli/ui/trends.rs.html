1//! Trends: how coverage and CI time move from run to run, for the CI/CD family of screens.
2//! Read from `delivery.trends`, which reads `site/data/trends.json` of this checkout as it
3//! lies when the screen is asked: the series the pages publication recorded, oldest run
4//! first. Coverage is drawn as percentage lines (lines, functions, regions), wall and queued
5//! time as minutes per run, both as inline SVG, with the slowest job medians in a table and
6//! every run's figures as text beneath.
7//!
8//! The screen writes nothing and derives nothing: a checkout the publication never wrote
9//! into has no series, and says so in words. A run that measured no coverage is a gap in the
10//! line, never a point at zero; a figure the file does not give reads `unknown`.
11

32/// The chart's drawing area, in the units of its view box.
33const WIDTH: f64 = 600.0;
34const HEIGHT: f64 = 200.0;
35const LEFT: f64 = 44.0;
36const RIGHT: f64 = 12.0;
37const TOP: f64 = 12.0;
38const BOTTOM: f64 = 28.0;
39
40/// One line of a chart: its name, its tone and one value per run, `None` where the run has
41/// none.
42struct Line {
43    name: &'static str,
44    tone: &'static str,
45    values: Vec<Option<f64>>,
46}
47
48fn number(v: &Value, pointer: &str) -> Option<f64> {
49    v.pointer(pointer).and_then(Value::as_f64)
50}
51
52/// A percentage as a reader says it: one place, or `unknown`.
53fn percent(p: Option<f64>) -> String {
54    p.map_or_else(|| "unknown".into(), |p| format!("{p:.1}%"))
55}
56
57/// Seconds as minutes with one place, or `unknown`: never a zero the file did not give.
58fn minutes(s: Option<f64>) -> String {
59    s.map_or_else(|| "unknown".into(), |s| format!("{:.1} min", s / 60.0))
60}
61
62/// The day of an RFC 3339 timestamp, as written.
63fn day(run: &Value) -> String {
64    run.get("started_at")
65        .and_then(Value::as_str)
66        .map_or_else(|| "unknown".into(), super::view::day)
67}

69/// A line chart of `lines` over the runs, as an SVG with the numbers of each line in a
70/// legend beneath it. `ceiling` fixes the top of the scale; `None` scales to the largest
71/// value drawn. A `None` value breaks the line: the run has no point there.
72fn chart(label: &str, unit: &str, lines: &[Line], runs: &[Value], ceiling: Option<f64>) -> El {
73    let drawn = lines.iter().flat_map(|l| l.values.iter().flatten()).count();
74    if drawn == 0 {
75        return nothing(&format!("No run recorded {label}."));
76    }
77    let top = ceiling.unwrap_or_else(|| {
78        lines
79            .iter()
80            .flat_map(|l| l.values.iter().flatten().copied())
81            .fold(0.0_f64, f64::max)
82            .max(1.0)
83    });
84    let n = runs.len();
85    let x = |i: usize| {
86        if n < 2 {
87            LEFT + (WIDTH - LEFT - RIGHT) / 2.0
88        } else {
89            LEFT + (WIDTH - LEFT - RIGHT) * i as f64 / (n - 1) as f64
90        }
91    };
92    let y = |v: f64| TOP + (HEIGHT - TOP - BOTTOM) * (1.0 - v.min(top) / top);
93    let mut svg = el("svg")
94        .attr("viewBox", format!("0 0 {WIDTH} {HEIGHT}"))
95        .attr("width", "100%")
96        .attr("role", "img")
97        .attr(
98            "aria-label",
99            format!("{label}, one point per run, oldest first"),
100        )
101        .attr("focusable", "false");
102    // the axes: a baseline at zero and a rule at the top of the scale, each named
103    for (value, caption) in [(0.0, format!("0{unit}")), (top, format!("{top:.0}{unit}"))] {
104        svg = svg
105            .child(
106                el("line")
107                    .attr("x1", format!("{LEFT}"))
108                    .attr("x2", format!("{}", WIDTH - RIGHT))
109                    .attr("y1", format!("{:.1}", y(value)))
110                    .attr("y2", format!("{:.1}", y(value)))
111                    .attr("stroke", "currentColor")
112                    .attr("opacity", "0.2"),
113            )
114            .child(
115                el("text")
116                    .attr("x", format!("{:.1}", LEFT - 6.0))
117                    .attr("y", format!("{:.1}", y(value) + 4.0))
118                    .attr("text-anchor", "end")
119                    .attr("font-size", "11")
120                    .attr("fill", "currentColor")
121                    .text(caption),
122            );
123    }
124    // one run is captioned once, centred under its point; more runs at both ends of the axis
125    let captions = match (runs.first(), runs.last()) {
126        (Some(only), _) if n == 1 => vec![(0, only, "middle")],
127        (Some(first), Some(last)) => vec![(0, first, "start"), (n - 1, last, "end")],
128        _ => Vec::new(),
129    };
130    for (i, run, anchor) in captions {
131        svg = svg.child(
132            el("text")
133                .attr("x", format!("{:.1}", x(i)))
134                .attr("y", format!("{:.1}", HEIGHT - 8.0))
135                .attr("text-anchor", anchor)
136                .attr("font-size", "11")
137                .attr("fill", "currentColor")
138                .text(day(run)),
139        );
140    }
141    for line in lines {
142        // a path per unbroken stretch; a stretch of one run is its point alone
143        let mut path = String::new();
144        let mut previous = false;
145        for (i, v) in line.values.iter().enumerate() {
146            match v {
147                Some(v) => {
148                    path.push_str(&format!(
149                        "{}{:.1} {:.1} ",
150                        if previous { "L" } else { "M" },
151                        x(i),
152                        y(*v)
153                    ));
154                    previous = true;
155                }
156                None => previous = false,
157            }
158        }
159        let mut group = el("g").class(format!("mj-tone--{}", line.tone));
160        if !path.is_empty() {
161            group = group.child(
162                el("path")
163                    .attr("d", path.trim_end())
164                    .attr("fill", "none")
165                    .attr("stroke", "var(--mj-tone-fg)")
166                    .attr("stroke-width", "2"),
167            );
168        }
169        for (i, v) in line.values.iter().enumerate() {
170            if let Some(v) = v {
171                group = group.child(
172                    el("circle")
173                        .attr("cx", format!("{:.1}", x(i)))
174                        .attr("cy", format!("{:.1}", y(*v)))
175                        .attr("r", "3")
176                        .attr("fill", "var(--mj-tone-fg)"),
177                );
178            }
179        }
180        svg = svg.child(group);
181    }
182    let legend = el("ul")
183        .class("mj-ui-legend")
184        .children(lines.iter().map(|l| {
185            let latest = l.values.iter().rev().flatten().next();
186            el("li")
187                .class(format!("mj-ui-legend-item mj-tone--{}", l.tone))
188                .child(
189                    el("span")
190                        .class("mj-ui-legend-dot")
191                        .attr("aria-hidden", "true"),
192                )
193                .child(el("span").class("mj-ui-legend-name").text(l.name))
194                .child(el("span").class("mj-ui-legend-value").text(match latest {
195                    Some(v) => format!("{v:.1}{unit} latest"),
196                    None => "never recorded".into(),
197                }))
198        }));
199    el("div").class("mj-ui-stack").child(svg).child(legend)
200}

202/// The percentage of one coverage kind of a run, or `None` when the run measured none or
203/// the file gave no total.
204fn coverage(run: &Value, kind: &str) -> Option<f64> {
205    number(run, &format!("/coverage/{kind}/percent"))
206}
207
208fn run_row(run: &Value) -> El {
209    let cell = |s: String| el("td").class("mj-ui-mono").text(s);
210    let sha = text(run, "head_sha");
211    el("tr")
212        .child(cell(text(run, "run_id")))
213        .child(cell(if sha.is_empty() {
214            "unknown".into()
215        } else {
216            sha.chars().take(7).collect()
217        }))
218        .child(cell(day(run)))
219        .child(cell(match text(run, "conclusion").as_str() {
220            "" => "unknown".into(),
221            c => c.to_string(),
222        }))
223        .child(cell(minutes(number(run, "/wall_seconds"))))
224        .child(cell(minutes(number(run, "/queued_seconds_total"))))
225        .child(cell(match run.get("coverage").filter(|c| c.is_object()) {
226            None => "no coverage".into(),
227            Some(_) => format!(
228                "{} · {} · {}",
229                percent(coverage(run, "lines")),
230                percent(coverage(run, "functions")),
231                percent(coverage(run, "regions"))
232            ),
233        }))
234}

236fn medians(trends: &Answer) -> El {
237    let rows = list(trends, "/job_medians")
238        .iter()
239        .map(|m| {
240            let seconds = |k: &str| {
241                m.get(k)
242                    .and_then(Value::as_f64)
243                    .map_or_else(|| "unknown".into(), |s| duration(s.round() as u64))
244            };
245            el("tr")
246                .child(el("td").class("mj-ui-mono").text(text(m, "name")))
247                .child(
248                    el("td")
249                        .class("mj-ui-mono")
250                        .text(seconds("median_run_seconds")),
251                )
252                .child(
253                    el("td")
254                        .class("mj-ui-mono")
255                        .text(seconds("median_queued_seconds")),
256                )
257                .child(
258                    el("td").class("mj-ui-mono").text(
259                        m.get("runs")
260                            .and_then(Value::as_u64)
261                            .map_or_else(|| "unknown".into(), |n| n.to_string()),
262                    ),
263                )
264        })
265        .collect::<Vec<_>>();
266    table(
267        &["Job", "Median run", "Median queued", "Runs"],
268        rows,
269        "The series gives no job median.",
270    )
271}

273/// The Trends screen.
274pub fn screen(ctx: &crate::capability::Context) -> View {
275    let answers = ask_all(ctx, &[TRENDS]);
276    let trends = &answers[&TRENDS];
277    let recorded = trends.read("/recorded").text().as_deref() == Some("true");
278    let reason = trends.read("/reason").text();
279
280    let kpis = if recorded {
281        vec![
282            kpi_count("Runs", &trends.read("/runs")),
283            kpi_count("Jobs measured", &trends.read("/job_medians")),
284            kpi("Series recorded", &trends.read("/recorded")),
285        ]
286    } else {
287        let why = reason.as_deref().unwrap_or("no reason was given");
288        vec![
289            kpi_not_observed("Runs", why, &trends.read("/runs")),
290            kpi_not_observed("Jobs measured", why, &trends.read("/job_medians")),
291        ]
292    };
293    let top = hero(
294        "chart",
295        "violet",
296        "Trends",
297        "How coverage and CI time move, run after run.",
298        "Coverage and time per run of the pages publication's series, read from this checkout when you ask.",
299        kpis,
300    );
301
302    let body = |draw: &dyn Fn(&[Value]) -> El| {
303        unanswered(trends).unwrap_or_else(|| {
304            if !recorded {
305                return el("div")
306                    .class("mj-ui-stack")
307                    .child(nothing(NONE_RECORDED))
308                    .child(
309                        el("p")
310                            .class("mj-ui-dim")
311                            .text(reason.clone().unwrap_or_default()),
312                    );
313            }
314            draw(
315                trends
316                    .ok()
317                    .and_then(|v| v.get("runs"))
318                    .and_then(Value::as_array)
319                    .map_or(&[][..], Vec::as_slice),
320            )
321        })
322    };
323    let kind = |name: &'static str, tone: &'static str, key: &'static str| {
324        move |runs: &[Value]| Line {
325            name,
326            tone,
327            values: runs.iter().map(|r| coverage(r, key)).collect(),
328        }
329    };
330    let coverage_chart = body(&|runs| {
331        chart(
332            "coverage",
333            "%",
334            &[
335                kind("Lines", "blue", "lines")(runs),
336                kind("Functions", "green", "functions")(runs),
337                kind("Regions", "violet", "regions")(runs),
338            ],
339            runs,
340            Some(100.0),
341        )
342    });
343    let time_chart = body(&|runs| {
344        let per = |key: &str| {
345            runs.iter()
346                .map(|r| number(r, key).map(|s| s / 60.0))
347                .collect::<Vec<_>>()
348        };
349        chart(
350            "wall or queued time",
351            " min",
352            &[
353                Line {
354                    name: "Wall time",
355                    tone: "orange",
356                    values: per("/wall_seconds"),
357                },
358                Line {
359                    name: "Queued time",
360                    tone: "cyan",
361                    values: per("/queued_seconds_total"),
362                },
363            ],
364            runs,
365            None,
366        )
367    });
368    let slowest = body(&|_| medians(trends));
369    let runs_table = body(&|runs| {
370        table(
371            &[
372                "Run",
373                "Commit",
374                "Started",
375                "Result",
376                "Wall",
377                "Queued",
378                "Coverage: lines · functions · regions",
379            ],
380            runs.iter()
381                .rev()
382                .take(super::dashboard::ROWS * 2)
383                .map(run_row)
384                .collect(),
385            "The series holds no run.",
386        )
387    });
388
389    let grid = el("div")
390        .class("mj-ui-grid")
391        .child(panel(
392            "Coverage",
393            "Percent covered per run: lines, functions and regions. A run that measured none leaves a gap.",
394            coverage_chart,
395            &[trends],
396        ))
397        .child(panel(
398            "Wall and queued time",
399            "Minutes per run, from its start to its end, and the time its jobs waited for a runner.",
400            time_chart,
401            &[trends],
402        ))
403        .child(panel(
404            "Slowest jobs",
405            "The median run and queue time of each job over the series, the slowest first.",
406            slowest,
407            &[trends],
408        ))
409        .child(wide(panel(
410            "Runs",
411            "The newest runs with the figures the lines are drawn from.",
412            runs_table,
413            &[trends],
414        )));
415
416    View::new(
417        "Trends",
418        el("div").class("mj-ui-stack").child(top).child(grid),
419    )
420}

432    fn repository(trends: Option<&str>) -> SyntheticRepository {
433        let repo = SyntheticRepository::new(Shape::default()).expect("a synthetic repository");
434        if let Some(body) = trends {
435            let path = repo.root().join(crate::delivery::trends::FILE);
436            std::fs::create_dir_all(path.parent().expect("a parent")).expect("a directory");
437            std::fs::write(path, body).expect("a series");
438        }
439        repo
440    }

442    /// The route the dispatcher answers, as a request: its status and its page.
443    fn get(repo: &SyntheticRepository) -> (u16, String) {
444        let ui = crate::ui::Ui::new(repo.context().expect("a context"), "0.0.0-test", None);
445        let r = ui.handle(&crate::http::router::Request::parse_target(
446            "GET",
447            ROUTE,
448            Vec::new(),
449        ));
450        let page = r.body.text().into_owned();
451        // the frame draws icons of its own: the screen is what lies inside its main part
452        let main = page
453            .split("<main")
454            .nth(1)
455            .and_then(|m| m.split("</main>").next())
456            .expect("the page has a main part");
457        (r.status, main.to_string())
458    }
