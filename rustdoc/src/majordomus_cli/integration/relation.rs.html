1//! What a pull request's head is to the current master, decided by git on this machine.
2//!
3//! The forge cannot answer this here: its merge cannot run the per-clone derived-file
4//! driver (`merge=derived`), so it calls nearly every pull request conflicting over files a
5//! generator rewrites anyway. `git merge-tree --write-tree` performs the real merge in the
6//! object database — no work tree, no index, no ref moves — with this clone's drivers, and
7//! `git check-attr merge` says which of the paths it touches are derived, from master's own
8//! `.gitattributes`. Neither list is kept here.
9
10use std::collections::BTreeSet;
11use std::path::Path;
12use std::process::Command;
13
14use super::model::RelationToMaster;
15
16fn git(root: &Path, args: &[&str]) -> Result<(bool, String), String> {
17    let out = Command::new("git")
18        .arg("-C")
19        .arg(root)
20        .args(args)
21        .output()
22        .map_err(|e| format!("git could not run: {e}"))?;
23    Ok((
24        out.status.success(),
25        String::from_utf8_lossy(&out.stdout).into_owned(),
26    ))
27}

29/// Whether `commit` names a commit this clone has.
30pub fn has_commit(root: &Path, commit: &str) -> bool {
31    git(root, &["cat-file", "-e", &format!("{commit}^{{commit}}")])
32        .map(|(ok, _)| ok)
33        .unwrap_or(false)
34}

36/// The subset of `paths` that master's `.gitattributes` marks `merge=derived`.
37pub fn derived_paths(root: &Path, master: &str, paths: &[String]) -> BTreeSet<String> {
38    if paths.is_empty() {
39        return BTreeSet::new();
40    }
41    let mut cmd = Command::new("git");
42    cmd.arg("-C")
43        .arg(root)
44        .args(["check-attr", "--source", master, "-z", "--stdin", "merge"])
45        .stdin(std::process::Stdio::piped())
46        .stdout(std::process::Stdio::piped())
47        .stderr(std::process::Stdio::null());
48    let Ok(mut child) = cmd.spawn() else {
49        return BTreeSet::new();
50    };
51    if let Some(mut stdin) = child.stdin.take() {
52        use std::io::Write;
53        let mut input = Vec::new();
54        for p in paths {
55            input.extend_from_slice(p.as_bytes());
56            input.push(0);
57        }
58        let _ = stdin.write_all(&input);
59    }
60    let Ok(out) = child.wait_with_output() else {
61        return BTreeSet::new();
62    };
63    // -z output: path NUL attribute NUL value NUL, repeated
64    let text = String::from_utf8_lossy(&out.stdout);
65    let fields: Vec<&str> = text.split('\0').collect();
66    fields
67        .chunks(3)
68        .filter(|c| c.len() == 3 && c[2] == "derived")
69        .map(|c| c[0].to_string())
70        .collect()
71}

73/// What `head` is to `master`, with the authored paths it changes.
74///
75/// ```text
76/// use crate::integration::{relation_to_master, RelationToMaster};
77/// let r = relation_to_master(std::path::Path::new("."), "origin/master", "origin/master");
78/// assert_eq!(r, RelationToMaster::Contained);
79/// ```text
80pub fn relation_to_master(root: &Path, master: &str, head: &str) -> RelationToMaster {
81    if !has_commit(root, head) {
82        return RelationToMaster::Unknown {
83            reason: format!(
84                "the head {head} is not in this clone; majordomus prs refresh fetches it"
85            ),
86        };
87    }
88    match git(root, &["merge-base", "--is-ancestor", head, master]) {
89        Ok((true, _)) => return RelationToMaster::Contained,
90        Ok(_) => {}
91        Err(reason) => return RelationToMaster::Unknown { reason },
92    }
93    let (clean, out) = match git(
94        root,
95        &[
96            "merge-tree",
97            "--write-tree",
98            "--name-only",
99            "--no-messages",
100            master,
101            head,
102        ],
103    ) {
104        Ok(v) => v,
105        Err(reason) => return RelationToMaster::Unknown { reason },
106    };
107    let mut lines = out.lines();
108    let Some(tree) = lines.next().map(str::trim).filter(|t| !t.is_empty()) else {
109        return RelationToMaster::Unknown {
110            reason: "git merge-tree printed no tree".into(),
111        };
112    };
113    let conflicted: Vec<String> = if clean {
114        Vec::new()
115    } else {
116        lines
117            .take_while(|l| !l.is_empty())
118            .map(str::to_string)
119            .collect::<BTreeSet<_>>()
120            .into_iter()
121            .collect()
122    };
123    let changed: Vec<String> = match git(root, &["diff", "--name-only", "-z", master, tree]) {
124        Ok((true, s)) => s
125            .split('\0')
126            .filter(|p| !p.is_empty())
127            .map(str::to_string)
128            .collect(),
129        Ok((false, _)) => {
130            return RelationToMaster::Unknown {
131                reason: "git diff of the merge result failed".into(),
132            }
133        }
134        Err(reason) => return RelationToMaster::Unknown { reason },
135    };
136    let mut all: Vec<String> = changed.clone();
137    all.extend(conflicted.iter().cloned());
138    let derived = derived_paths(root, master, &all);
139    // a conflict on a derived path is the regeneration's to resolve, not a person's
140    let authored_conflicts: Vec<String> = conflicted
141        .iter()
142        .filter(|p| !derived.contains(*p))
143        .cloned()
144        .collect();
145    if !authored_conflicts.is_empty() {
146        return RelationToMaster::Conflicting {
147            paths: authored_conflicts,
148        };
149    }
150    let authored: Vec<String> = changed
151        .iter()
152        .filter(|p| !derived.contains(*p))
153        .cloned()
154        .collect();
155    if changed.is_empty() && conflicted.is_empty() {
156        return RelationToMaster::Superseded;
157    }
158    if authored.is_empty() {
159        let paths: BTreeSet<String> = changed.into_iter().chain(conflicted).collect();
160        return RelationToMaster::DerivedOnly {
161            paths: paths.into_iter().collect(),
162        };
163    }
164    match git(root, &["merge-base", "--is-ancestor", master, head]) {
165        Ok((true, _)) => RelationToMaster::UpToDate { authored },
166        Ok(_) => {
167            let behind = git(root, &["rev-list", "--count", &format!("{head}..{master}")])
168                .ok()
169                .and_then(|(_, s)| s.trim().parse().ok())
170                .unwrap_or(0);
171            RelationToMaster::Behind { behind, authored }
172        }
173        Err(reason) => RelationToMaster::Unknown { reason },
174    }
175}
