1//! The `dashboard` module: the Overview of the Dashboard Suite, as a projection of the
2//! capabilities that already answer each of its facts (ADR 0088).
3//!
4//! **A dashboard has no data of its own.** `dashboard.overview` answers four questions —
5//! *is it healthy*, *what changed*, *what is broken*, *what needs action* — and every card
6//! under them is one fact read out of another capability's answer through the same
7//! executor MCP, the HTTP routes and the Cockpit use. Nothing here is stored, counted or
8//! judged a second time: the card carries the capability it asked, the input it asked with,
9//! and the RFC 6901 pointer into that answer its value was read from, so a reader — or a
10//! test — can ask the source the same question and get the same value.
11//!
12//! **The value is the source's, exactly.** A card's [`DashboardSource::measure`] is one of
13//! two readings, and neither is arithmetic on the source's behalf:
14//!
15//! - [`DashboardMeasure::Exact`] — `value == answer.pointer(pointer)`, and `null` when the
16//!   pointer resolves to nothing;
17//! - [`DashboardMeasure::Count`] — the number at the pointer, or the length of the list or
18//!   map there, and `0` when the source left it out: the sources omit an empty list and a
19//!   zero tally (`skip_serializing_if`), so absence *is* their way of saying none.
20//!
21//! `tests/dashboard.rs` and `test/cases/534_the_overview_answers_four_questions.sh` hold
22//! that sentence for every card, naming none: they execute each card's source with its
23//! input and compare. A card that computed its own number fails them.
24//!
25//! **Status words are the sources' verdicts.** `fail` where the source says something
26//! refuses (a failing check, a worktree out of place, a question that refuses `finish`),
27//! `warn` where it names something worth looking at (a pending bump, a blocked issue, two
28//! workers claiming one scope), `ok` otherwise — and `unknown`, never `ok`, when the source
29//! could not answer, with its error as the card's detail. No threshold is invented here.
30//!
31//! **It is cheap.** Each source is asked once per answer, and the sources are the ones that
32//! read what this process already holds: no delivery report, no release analysis, no
33//! evidence judgement — nothing that rebuilds canonical state or refuses on a network.
34//!
35//! ```
36//! use majordomus_cli::capability::builtin::dashboard::module;
37//!
38//! // one capability, reachable from every surface the registry projects
39//! let m = module();
40//! let c = &m.capabilities[0].capability;
41//! assert_eq!(c.id.as_str(), "dashboard.overview");
42//! assert_eq!(c.exposure.http.as_ref().unwrap().path, "/api/v1/dashboard/overview");
43//! assert_eq!(c.exposure.cli.as_ref().unwrap().path, ["dashboard", "overview"]);
44//! ```

62/// How a card's value is read out of its source's answer. Neither reading is arithmetic on
63/// the source's behalf.
64///
65/// ```
66/// use majordomus_cli::capability::builtin::dashboard::DashboardMeasure;
67/// use serde_json::json;
68///
69/// let answer = json!({ "status": "warn", "tallies": { "ok": 6 }, "overlaps": [1, 2] });
70/// // exact: the value at the pointer, null when there is none
71/// assert_eq!(DashboardMeasure::Exact.read(&answer, "/status"), json!("warn"));
72/// assert_eq!(DashboardMeasure::Exact.read(&answer, "/tallies/fail"), json!(null));
73/// // count: a number as it stands, a list or map by its length, and an omitted one is none
74/// assert_eq!(DashboardMeasure::Count.read(&answer, "/tallies/ok"), json!(6));
75/// assert_eq!(DashboardMeasure::Count.read(&answer, "/overlaps"), json!(2));
76/// assert_eq!(DashboardMeasure::Count.read(&answer, "/tallies/fail"), json!(0));
77/// ```
78#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
79#[serde(rename_all = "snake_case")]
80pub enum DashboardMeasure {
81    /// The value at the pointer, exactly; `null` when the pointer resolves to nothing.
82    Exact,
83    /// The number at the pointer, or the length of the list or map there; `0` when the
84    /// source left it out, which is how the sources say an empty list or a zero tally.
85    Count,
86}
87
88impl DashboardMeasure {
89    /// Read the card's value out of the source's answer.
90    ///
91    /// ```
92    /// use majordomus_cli::capability::builtin::dashboard::DashboardMeasure;
93    /// use serde_json::json;
94    /// let answer = json!({ "issue": { "id": "I0001" } });
95    /// assert_eq!(DashboardMeasure::Exact.read(&answer, "/issue/id"), json!("I0001"));
96    /// assert_eq!(DashboardMeasure::Count.read(&answer, "/blockers"), json!(0));
97    /// ```
98    pub fn read(self, answer: &Value, pointer: &str) -> Value {
99        let at = answer.pointer(pointer);
100        match self {
101            DashboardMeasure::Exact => at.cloned().unwrap_or(Value::Null),
102            DashboardMeasure::Count => match at {
103                None | Some(Value::Null) => json!(0),
104                Some(Value::Array(list)) => json!(list.len()),
105                Some(Value::Object(map)) => json!(map.len()),
106                Some(other) => other.clone(),
107            },
108        }
109    }
110}

112/// Where a card's value came from: the capability asked, the input it was asked with, and
113/// the pointer into its answer. Asking the capability the same question and reading the
114/// pointer with the same measure yields the card's value; that is the whole contract.
115///
116/// ```
117/// use majordomus_cli::capability::builtin::dashboard::{DashboardMeasure, DashboardSource};
118/// use serde_json::json;
119///
120/// let s: DashboardSource = serde_json::from_value(json!({
121///     "capability": "health.report",
122///     "input": {},
123///     "pointer": "/status",
124///     "measure": "exact",
125/// })).unwrap();
126/// assert_eq!(s.measure, DashboardMeasure::Exact);
127/// assert_eq!(s.measure.read(&json!({ "status": "ok" }), &s.pointer), json!("ok"));
128/// ```
129#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
130pub struct DashboardSource {
131    /// The canonical id of the capability the value was read from.
132    pub capability: String,
133    /// The input it was asked with, as its input schema takes it.
134    pub input: Value,
135    /// An RFC 6901 JSON pointer into that capability's answer.
136    pub pointer: String,
137    /// How the value is read at the pointer.
138    pub measure: DashboardMeasure,
139}

141/// One fact on a dashboard: a value read out of one capability's answer, the verdict the
142/// source gives it, where a reader drills in, and what acts on it.
143///
144/// ```
145/// use majordomus_cli::capability::builtin::dashboard::DashboardCard;
146/// use serde_json::json;
147///
148/// let card: DashboardCard = serde_json::from_value(json!({
149///     "id": "health-status",
150///     "title": "Health",
151///     "value": "ok",
152///     "status": "ok",
153///     "detail": "the worst of 8 check(s)",
154///     "source": { "capability": "health.report", "input": {}, "pointer": "/status", "measure": "exact" },
155///     "route": "/cockpit/health",
156/// })).unwrap();
157/// // the value is the source's: reading the pointer out of its answer gives it back
158/// assert_eq!(card.source.measure.read(&json!({ "status": "ok" }), &card.source.pointer), card.value);
159/// assert!(card.action.is_none());
160/// ```
161#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
162pub struct DashboardCard {
163    /// A stable id, `[a-z][a-z0-9-]*`, unique across the dashboard.
164    pub id: String,
165    /// What the value is.
166    pub title: String,
167    /// The value, read out of the source's answer by the source's measure: a number, a
168    /// string or a boolean, or `null` when the source holds none.
169    pub value: Value,
170    /// The source's verdict on it; `unknown` when the source could not answer.
171    pub status: HealthStatus,
172    /// One line: what the value means, or why the source could not answer.
173    pub detail: String,
174    /// Where the value came from.
175    pub source: DashboardSource,
176    /// The Cockpit page a reader drills into for the evidence.
177    pub route: String,
178    #[serde(default, skip_serializing_if = "Option::is_none")]
179    /// The command that acts on it, when one does.
180    pub action: Option<String>,
181}

196pub struct DashboardQuestion {
197    /// `healthy`, `changed`, `broken` or `action`.
198    pub id: String,
199    /// The question, as a reader asks it.
200    pub title: String,
201    /// The worst of its cards.
202    pub status: HealthStatus,
203    /// The facts that answer it, each read from a capability.
204    pub cards: Vec<DashboardCard>,
205}

266fn health_status(v: &Value, answer: &Value) -> (HealthStatus, String) {
267    let status = serde_json::from_value::<HealthStatus>(v.clone()).unwrap_or(HealthStatus::Unknown);
268    let checks = answer["checks"].as_array().cloned().unwrap_or_default();
269    let not_ok: Vec<String> = checks
270        .iter()
271        .filter(|c| c["status"] != "ok")
272        .map(|c| {
273            format!(
274                "{} ({})",
275                c["id"].as_str().unwrap_or("?"),
276                c["status"].as_str().unwrap_or("?")
277            )
278        })
279        .collect();
280    let detail = if not_ok.is_empty() {
281        format!("the worst of {} check(s); every one is ok", checks.len())
282    } else {
283        format!(
284            "the worst of {} check(s); not ok: {}",
285            checks.len(),
286            not_ok.join(", ")
287        )
288    };
289    (status, detail)
290}
291
292fn health_failing(v: &Value, answer: &Value) -> (HealthStatus, String) {
293    let failing: Vec<&str> = answer["checks"]
294        .as_array()
295        .map(|a| {
296            a.iter()
297                .filter(|c| c["status"] == "fail")
298                .filter_map(|c| c["id"].as_str())
299                .collect()
300        })
301        .unwrap_or_default();
302    let detail = if failing.is_empty() {
303        "no check's engine reports a failure".to_string()
304    } else {
305        format!("failing: {}", failing.join(", "))
306    };
307    (some_or(v, HealthStatus::Ok, HealthStatus::Fail), detail)
308}

338fn pending_bump(v: &Value, answer: &Value) -> (HealthStatus, String) {
339    let n = answer["changes"].as_array().map(Vec::len).unwrap_or(0);
340    match v.as_str() {
341        Some("none") | None => (
342            HealthStatus::Ok,
343            "no commit since the last release implies a release".to_string(),
344        ),
345        Some(bump) => (
346            HealthStatus::Warn,
347            format!("{n} kind(s) of change since the last release imply a {bump} release"),
348        ),
349    }

352fn next_version(v: &Value, answer: &Value) -> (HealthStatus, String) {
353    let bump = answer["bump"].as_str().unwrap_or("?");
354    match (v.as_str(), answer["decided_by"].as_str()) {
355        (Some(_), Some("contract")) => (
356            HealthStatus::Ok,
357            format!(
358                "the version the public contract requires: the declared one when it already \
359                 satisfies the contract, otherwise the smallest it allows; the commits imply a \
360                 {bump} (evidence)"
361            ),
362        ),
363        (Some(_), Some("contract_and_commits")) => (
364            HealthStatus::Ok,
365            format!(
366                "the contract requires no release over the last one and the commits imply a \
367                 {bump}: the smallest release above it, a patch"
368            ),
369        ),
370        (_, Some("undecided")) => (
371            HealthStatus::Unknown,
372            format!(
373                "undecided: the public contract could not be measured ({}), so no version is \
374                 selected; the commits imply a {bump} (evidence only), and `release bump` needs \
375                 an explicit target",
376                answer["contract_unreadable"]
377                    .as_str()
378                    .unwrap_or("no reason given")
379            ),
380        ),
381        (Some(_), _) => (
382            HealthStatus::Ok,
383            "the version the public contract requires".to_string(),
384        ),
385        (None, _) => (HealthStatus::Ok, "nothing would be released".to_string()),
386    }
387}

620/// The overview: each source asked once, through the executor, and every card read out of
621/// its answer.
622fn overview(ctx: &Context, _: Empty) -> Result<DashboardOverview, CapabilityError> {
623    // one execution per (capability, input): four cards of `release.version` are one call
624    let mut answers: BTreeMap<(&str, &str), Result<Value, String>> = BTreeMap::new();
625    for spec in CARDS {
626        answers
627            .entry((spec.capability, spec.input))
628            .or_insert_with(|| {
629                let input: Value = serde_json::from_str(spec.input)
630                    .map_err(|e| format!("the card's own input does not parse: {e}"))?;
631                let answer = ctx
632                    .execute(spec.capability, input)
633                    .map_err(|e| e.to_string());
634                ctx.progress
635                    .step_done(spec.capability, answer.is_ok(), None);
636                answer
637            });
638    }
639
640    let questions: Vec<DashboardQuestion> = QUESTIONS
641        .iter()
642        .map(|(id, title)| {
643            let cards: Vec<DashboardCard> = CARDS
644                .iter()
645                .filter(|s| s.question == *id)
646                .map(|spec| card(spec, &answers[&(spec.capability, spec.input)]))
647                .collect();
648            DashboardQuestion {
649                id: (*id).into(),
650                title: (*title).into(),
651                status: cards
652                    .iter()
653                    .fold(HealthStatus::Ok, |acc, c| acc.worse(c.status)),
654                cards,
655            }
656        })
657        .collect();
658    Ok(DashboardOverview {
659        status: questions
660            .iter()
661            .fold(HealthStatus::Ok, |acc, q| acc.worse(q.status)),
662        questions,
663    })
664}

666/// One card from its declaration and its source's answer — or, when the source could not
667/// answer, an `unknown` card that says why. Never a pass in place of an answer.
668fn card(spec: &Spec, answer: &Result<Value, String>) -> DashboardCard {
669    let input: Value = serde_json::from_str(spec.input).unwrap_or(Value::Null);
670    let mut source = DashboardSource {
671        capability: spec.capability.into(),
672        input,
673        pointer: spec.pointers[0].into(),
674        measure: spec.measure,
675    };
676    let (value, status, detail) = match answer {
677        Ok(answer) => {
678            if let Some(p) = spec.pointers.iter().find(|p| answer.pointer(p).is_some()) {
679                source.pointer = (*p).into();
680            }
681            let value = spec.measure.read(answer, &source.pointer);
682            let (status, detail) = (spec.judge)(&value, answer);
683            (value, status, detail)
684        }
685        Err(reason) => (
686            Value::Null,
687            HealthStatus::Unknown,
688            format!(
689                "{} could not answer, so this is unknown rather than a pass: {reason}",
690                spec.capability
691            ),
692        ),
693    };
694    DashboardCard {
695        id: spec.id.into(),
696        title: spec.title.into(),
697        value,
698        status,
699        detail,
700        source,
701        route: spec.route.into(),
702        action: spec.action.map(str::to_string),
703    }
704}

708/// The `dashboard` module: the Dashboard Suite's pages, each a projection of the
709/// capabilities that hold its facts. The Overview is the first.
710///
711/// ```
712/// use majordomus_cli::capability::builtin::dashboard::module;
713/// let m = module();
714/// assert_eq!(m.id.as_str(), "dashboard");
715/// let tool = m.capabilities[0].capability.exposure.mcp.as_ref().and_then(|m| m.tool.clone());
716/// assert_eq!(tool.as_deref(), Some("majordomus_dashboard_overview"));
717/// ```
718pub fn module() -> ModuleDescriptor {
719    module! {
720        id: "dashboard",
721        title: "Dashboards",
722        description: "The Dashboard Suite as a projection of the canonical state: every card is one fact read out of an existing capability's answer, carrying the capability it asked, the input, the JSON pointer its value was read from, the Cockpit page that holds the evidence and the command that acts on it. No dashboard stores, counts or judges anything of its own (ADR 0088).",
723        stability: Stability::BehaviorallyVerified,
724        capabilities: [
725            capability! {
726                id: "dashboard.overview",
727                title: "The Overview: four questions",
728                description: "Is it healthy, what changed, what is broken, what needs action — each answered by cards read out of `health.report`, `release.version`, `plan.status`, `worktree.status`, `plan.next`, `continuity.state` and `peers.list` (this checkout's board), asked once each through the executor. Every card carries its source capability, input and RFC 6901 pointer, so asking the source the same question yields the same value; a source that cannot answer makes its cards `unknown`, never `ok`. Nothing here rebuilds canonical state or asks anything beyond this process and its checkout.",
729                input: Empty,
730                output: DashboardOverview,
731                stability: Stability::BehaviorallyVerified,
732                exposure: Exposure {
733                    mcp: mcp("majordomus_dashboard_overview"),
734                    http: get("/api/v1/dashboard/overview"),
735                    cli: Some(CliExposure { path: vec!["dashboard".into(), "overview".into()] }),
736                },
737                tags: ["dashboard", "overview", "introspection"],
738                // every source carries its own cache policy; caching the composition would
739                // serve a card older than the source a reader drills into
740                cache: CachePolicy::Disabled,
741                handler: overview,
742            },
743        ],
744    }

752    /// The next-version card names who answered: the contract, or the contract with the
753    /// commits — and, when the contract cannot be measured, that no one did: the commit
754    /// inference is never passed off as the answer.
755    #[test]
756    fn the_next_version_card_names_who_answered() {
757        let v = Value::from("0.11.0");
758        let by = |decided_by: &str| {
759            next_version(
760                &v,
761                &serde_json::json!({ "bump": "major", "decided_by": decided_by }),
762            )
763        };
764        let (status, contract) = by("contract");
765        assert_eq!(status, HealthStatus::Ok);
766        assert!(contract.starts_with("the version the public contract requires"));
767        assert!(contract.contains("the commits imply a major (evidence)"));
768        let (_, both) = by("contract_and_commits");
769        assert!(both.starts_with("the contract requires no release over the last one"));
770        // unmeasurable: no version is selected, and the card neither passes the commit
771        // inference off as the answer nor reports `ok`
772        let (status, undecided) = next_version(
773            &Value::Null,
774            &serde_json::json!({
775                "bump": "major",
776                "decided_by": "undecided",
777                "contract_unreadable": "no release to compare with",
778            }),
779        );
780        assert_eq!(status, HealthStatus::Unknown);
781        assert!(undecided.starts_with("undecided: the public contract could not be measured"));
782        assert!(undecided.contains("no release to compare with"));
783        assert!(undecided.contains("needs an explicit target"));
784        let (_, nothing) = next_version(
785            &Value::Null,
786            &serde_json::json!({ "decided_by": "contract" }),
787        );
788        assert_eq!(nothing, "nothing would be released");
789        // an answer that names no one, or a word this card does not know (an older or a
790        // newer executable): the version is still the contract's, and the card claims no
791        // more than that — neither the commits nor "undecided" are read into it
792        for answer in [
793            serde_json::json!({ "bump": "minor" }),
794            serde_json::json!({ "bump": "minor", "decided_by": "a-later-word" }),
795        ] {
796            let (status, plain) = next_version(&v, &answer);
797            assert_eq!(status, HealthStatus::Ok);
798            assert_eq!(plain, "the version the public contract requires");
799        }
