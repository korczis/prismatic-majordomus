1//! Skills as proven capabilities: what the repository can show about each skill, derived.
2//!
3//! A skill is data: `SKILL.md` under the layer's skills section, discovered by the source class
4//! `skill` and validated against its schema by the index (ADR 0007). That makes a skill
5//! *exist*. It says nothing about whether the procedure is exercised by anything, reached by
6//! anything, published anywhere or held by any gate — and a catalogue whose entries are
7//! merely present is the hand-maintained registry this repository refuses, one level up.
8//!
9//! So four facts are derived here, per skill, on every read, and none is authored:
10//!
11//! * **tested** — a behavioural case or crate test names the skill with the marker
12//!   `majordomus-skill: <id>` on a comment line, and the evidence ledger holds a run of that
13//!   test. The run is judged by the same judgement a rule's
14//!   named test gets, so a test cannot be current for a rule and stale for a skill;
15//! * **documented** — the site's page projection for the skill is a tracked file;
16//! * **enforced** — a dispatched rule validates skills (its validator is `skills`) and a CI
17//!   gate whose command runs the skills verification is selected by a change to the skill;
18//! * **used** — an invocation surface (a workflow, a prompt, a profile, a provider template,
19//!   a recipe, a CI workflow, a root bootstrap) references the skill by its URI
20//!   `majordomus://skill/<id>` or by the path of its `SKILL.md`.
21//!
22//! An active skill that no test names, or that nothing invokes, is an **orphan**: a failure.
23//! A test that names a skill that does not exist, and an invocation that references one, are
24//! failures too — a binding to nothing reads as proof. Evidence that is not current, a missing
25//! page and a missing gate are warnings: each is a debt with a command that settles it, and
26//! none of them can be settled by the author of the skill alone. A draft or deprecated skill
27//! owes nothing.
28//!
29//! ```
30//! use majordomus_cli::evidence::Ledger;
31//! use majordomus_cli::skill::Skills;
32//! use majordomus_cli::synthetic::SyntheticRepository;
33//!
34//! let repo = SyntheticRepository::small().unwrap();
35//! let index = repo.index().unwrap();
36//! let skills = Skills::build(&index, &Ledger::empty());
37//! // every skill the index holds is answered, and a repository with none owes nothing
38//! assert_eq!(
39//!     skills.skills.len(),
40//!     index.objects.iter().filter(|o| o.kind == "skill").count()
41//! );
42//! assert_eq!(skills.failures() + skills.warnings(), skills.findings.len());
43//! ```

97/// Where a skill came from, when it was not written here first: an opaque marker into a
98/// ledger kept outside the repository. The committed half names no source; the ledger entry
99/// it points at does, locally.
100///
101/// ```
102/// use majordomus_cli::skill::SkillProvenance;
103/// let p = SkillProvenance {
104///     origin: "prior-art".into(),
105///     ledger: "import-2026-09-09#1".into(),
106///     decision: "adapted".into(),
107/// };
108/// assert!(p.ledger.starts_with("import-"));
109/// ```
110#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
111pub struct SkillProvenance {
112    /// `prior-art`: the concept was studied elsewhere before this skill was written.
113    pub origin: String,
114    /// `import-<date>#<n>`: the entry of the local import ledger that holds the mapping.
115    pub ledger: String,
116    /// `adapted`, `reimplemented` or `merged`.
117    pub decision: String,
118}

120/// What the evidence ledger can say about a skill's tests, as a whole: the weakest state of
121/// the tests that name it, or `untested` when none does.
122///
123/// ```
124/// use majordomus_cli::skill::SkillEvidence;
125/// assert!(SkillEvidence::Proven.current());
126/// assert!(SkillEvidence::InputsUnchanged.current());
127/// assert!(!SkillEvidence::Stale.current());
128/// assert!(SkillEvidence::Proven < SkillEvidence::Untested);
129/// ```
130#[derive(
131    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
132)]
133#[serde(rename_all = "snake_case")]
134pub enum SkillEvidence {
135    /// Every naming test has a passing run, and nothing but the ledger changed since.
136    Proven,
137    /// Every naming test passed, and neither the test nor the skill changed since.
138    InputsUnchanged,
139    /// A passing run exists, and the test or the skill changed since.
140    Stale,
141    /// The latest recorded run of a naming test did not pass.
142    Failing,
143    /// A test names the skill and no run of it was ever recorded.
144    NotRun,
145    /// A test names the skill from a path no runner drives.
146    Unrunnable,
147    /// No test names the skill.
148    Untested,
149}
150
151impl SkillEvidence {
152    /// Does this state carry a passing run nothing known has invalidated?
153    ///
154    /// ```
155    /// use majordomus_cli::skill::SkillEvidence;
156    /// assert!(!SkillEvidence::NotRun.current());
157    /// ```
158    pub fn current(self) -> bool {
159        matches!(self, SkillEvidence::Proven | SkillEvidence::InputsUnchanged)
160    }
161
162    /// The word every surface prints for this evidence state: the text listing, the JSON
163    /// answer and the site page all read it here, so they cannot spell a state two ways.
164    ///
165    /// ```
166    /// use majordomus_cli::skill::SkillEvidence;
167    /// assert_eq!(SkillEvidence::InputsUnchanged.label(), "inputs_unchanged");
168    /// ```
169    pub fn label(self) -> &'static str {
170        match self {
171            SkillEvidence::Proven => "proven",
172            SkillEvidence::InputsUnchanged => "inputs_unchanged",
173            SkillEvidence::Stale => "stale",
174            SkillEvidence::Failing => "failing",
175            SkillEvidence::NotRun => "not_run",
176            SkillEvidence::Unrunnable => "unrunnable",
177            SkillEvidence::Untested => "untested",
178        }
179    }
180
181    /// The skill-level word for one test's proof state.
182    ///
183    /// ```
184    /// use majordomus_cli::evidence::ProofState;
185    /// use majordomus_cli::skill::SkillEvidence;
186    /// assert_eq!(SkillEvidence::of(ProofState::NotRun), SkillEvidence::NotRun);
187    /// ```
188    pub fn of(state: ProofState) -> SkillEvidence {
189        match state {
190            ProofState::Proven => SkillEvidence::Proven,
191            ProofState::InputsUnchanged => SkillEvidence::InputsUnchanged,
192            ProofState::Stale => SkillEvidence::Stale,
193            ProofState::Failing => SkillEvidence::Failing,
194            ProofState::NotRun => SkillEvidence::NotRun,
195            ProofState::Unrunnable | ProofState::NoTest => SkillEvidence::Unrunnable,
196        }
197    }
198}

200/// Where a skill stands, derived from the four facts.
201///
202/// ```
203/// use majordomus_cli::skill::SkillStanding;
204/// assert_eq!(SkillStanding::Orphan.label(), "orphan");
205/// ```
206#[derive(
207    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
208)]
209#[serde(rename_all = "snake_case")]
210pub enum SkillStanding {
211    /// Active, and tested with current evidence, documented, enforced and used.
212    Proven,
213    /// Active, named by a test and invoked, with at least one debt a warning names.
214    Partial,
215    /// Active, and no test names it or nothing invokes it.
216    Orphan,
217    /// Its file does not satisfy the skill contract; one the index refused outright is listed
218    /// here too, so a listing never silently shrinks.
219    Invalid,
220    /// Draft or deprecated: nothing is owed.
221    NotRequired,
222}
223
224impl SkillStanding {
225    /// The word every surface prints for this standing, the same snake_case word the JSON
226    /// answer serialises, so a script filtering on it and a person reading it agree.
227    ///
228    /// ```
229    /// use majordomus_cli::skill::SkillStanding;
230    /// assert_eq!(SkillStanding::NotRequired.label(), "not_required");
231    /// ```
232    pub fn label(self) -> &'static str {
233        match self {
234            SkillStanding::Proven => "proven",
235            SkillStanding::Partial => "partial",
236            SkillStanding::Orphan => "orphan",
237            SkillStanding::Invalid => "invalid",
238            SkillStanding::NotRequired => "not_required",
239        }
240    }
241}

455/// Every skill of a repository, derived, and every finding.
456///
457/// ```
458/// use majordomus_cli::evidence::Ledger;
459/// use majordomus_cli::skill::Skills;
460/// use majordomus_cli::synthetic::SyntheticRepository;
461/// let repo = SyntheticRepository::small().unwrap();
462/// let skills = Skills::build(&repo.index().unwrap(), &Ledger::empty());
463/// assert!(skills.skill("no-such-skill").is_none());
464/// ```
465#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
466pub struct Skills {
467    /// Every skill, in index order.
468    pub skills: Vec<SkillStatus>,
469    /// Every finding: each skill's, then those about no one skill (a binding to nothing).
470    pub findings: Vec<SkillFinding>,
471}
472
473impl Skills {
474    /// Derive every skill's standing from the index, the evidence ledger, the tracked tests,
475    /// the tracked invocation surfaces and the CI model.
476    ///
477    /// ```
478    /// use majordomus_cli::evidence::Ledger;
479    /// use majordomus_cli::skill::Skills;
480    /// use majordomus_cli::synthetic::SyntheticRepository;
481    /// let repo = SyntheticRepository::small().unwrap();
482    /// let a = Skills::build(&repo.index().unwrap(), &Ledger::empty());
483    /// let b = Skills::build(&repo.index().unwrap(), &Ledger::empty());
484    /// assert_eq!(a, b, "two reads of one tree derive the same answer");
485    /// ```
486    pub fn build(index: &Index, ledger: &Ledger) -> Skills {
487        let root = PathBuf::from(&index.repository.root);
488        let skills_dir = index
489            .repository
490            .sections
491            .get("skills")
492            .cloned()
493            .unwrap_or_else(|| ".ai/repo/skills".to_string());
494        let tracked: Vec<String> = crate::git::ls_files_all(&root).unwrap_or_default();
495        let tracked_set: BTreeSet<&str> = tracked.iter().map(String::as_str).collect();
496
497        let objects: Vec<&crate::model::Object> =
498            index.objects.iter().filter(|o| o.kind == SKILL).collect();
499        let ids: BTreeSet<String> = objects.iter().map(|o| skill_id(o)).collect();
500
501        let markers = test_markers(&root, &tracked);
502        let invocations = invocation_references(&root, &tracked, index, &skills_dir);
503        let comparisons = crate::rules::ledger_comparisons(&root, ledger);
504        let doctrine = crate::rules::definitions(index)
505            .into_iter()
506            .find(|d| d.enforcement.validator.as_deref() == Some(SKILLS_VALIDATOR))
507            .map(|d| d.id);
508        let model = GateModel::load(&root).ok();
509
510        let mut findings = Vec::new();
511        let mut skills = Vec::new();
512        for o in objects {
513            let id = skill_id(o);
514            let path = o.provenance.path.clone();
515            let meta = &o.metadata;
516            let status = str_field(meta, "status");
517            let active = status == "active";
518            let contract: Vec<&crate::model::Diagnostic> = index
519                .diagnostics
520                .iter()
521                .filter(|d| d.severity == Severity::Error && d.path.as_deref() == Some(&path))
522                .collect();
523            let mut own = Vec::new();
524            let explain = format!("majordomus skills explain {id}");
525            for d in &contract {
526                own.push(SkillFinding {
527                    level: SkillFindingLevel::Fail,
528                    code: "contract".into(),
529                    skill: Some(id.clone()),
530                    subject: path.clone(),
531                    message: format!("{}: {}", d.code, d.message),
532                    reproduce: "majordomus skills check".into(),
533                });
534            }
535
536            // tested
537            let mut tests = Vec::new();
538            for (test_path, named) in &markers {
539                if !named.contains(&id) {
540                    continue;
541                }
542                let tid = TestId::of(test_path);
543                let execution = tid.as_ref().and_then(|t| ledger.latest(&t.as_string()));
544                let state =
545                    crate::rules::test_state(&root, &comparisons, tid.as_ref(), execution, &path);
546                tests.push(SkillTest {
547                    path: test_path.clone(),
548                    test: tid.as_ref().map(TestId::as_string),
549                    state,
550                    reproduce: tid.as_ref().map(TestId::reproduce),
551                    recorded_at: execution.map(|e| e.at.clone()),
552                });
553            }
554            let evidence = tests
555                .iter()
556                .map(|t| SkillEvidence::of(t.state))
557                .max()
558                .unwrap_or(SkillEvidence::Untested);
559            if active {
560                match evidence {
561                    SkillEvidence::Untested => own.push(SkillFinding {
562                        level: SkillFindingLevel::Fail,
563                        code: "untested".into(),
564                        skill: Some(id.clone()),
565                        subject: path.clone(),
566                        message: format!(
567                            "orphan: no test names skill '{id}'; add `{TEST_MARKER} {id}` on a \
568                             comment line of the case or crate test that exercises it"
569                        ),
570                        reproduce: explain.clone(),
571                    }),
572                    SkillEvidence::Failing => own.push(SkillFinding {
573                        level: SkillFindingLevel::Fail,
574                        code: "failing".into(),
575                        skill: Some(id.clone()),
576                        subject: path.clone(),
577                        message: format!(
578                            "the latest recorded run of a test naming skill '{id}' did not pass"
579                        ),
580                        reproduce: weakest_reproduce(&tests, &explain),
581                    }),
582                    e if !e.current() => own.push(SkillFinding {
583                        level: SkillFindingLevel::Warn,
584                        code: "unevidenced".into(),
585                        skill: Some(id.clone()),
586                        subject: path.clone(),
587                        message: format!(
588                            "skill '{id}' is named by {} test(s) and its evidence is {}: run the \
589                             test and record it (majordomus evidence record)",
590                            tests.len(),
591                            e.label()
592                        ),
593                        reproduce: weakest_reproduce(&tests, &explain),
594                    }),
595                    _ => {}
596                }
597            }
598
599            // documented
600            let page = format!("{SKILL_PAGES_DIR}/{id}.md");
601            let documented = tracked_set.contains(page.as_str());
602            if active && !documented {
603                own.push(SkillFinding {
604                    level: SkillFindingLevel::Warn,
605                    code: "undocumented".into(),
606                    skill: Some(id.clone()),
607                    subject: path.clone(),
608                    message: format!("skill '{id}' has no tracked page at {page}"),
609                    reproduce: "scripts/generate-site-data".into(),
610                });
611            }
612
613            // enforced
614            let gates = verifying_gates(model.as_ref(), &path);
615            let enforced = doctrine.is_some() && !gates.is_empty();
616            if active && !enforced {
617                own.push(SkillFinding {
618                    level: SkillFindingLevel::Warn,
619                    code: "unenforced".into(),
620                    skill: Some(id.clone()),
621                    subject: path.clone(),
622                    message: match &doctrine {
623                        None => format!(
624                            "skill '{id}': no dispatched rule validates skills (validator \
625                             `{SKILLS_VALIDATOR}`)"
626                        ),
627                        Some(_) => format!(
628                            "skill '{id}': no CI gate selected by a change to {path} runs \
629                             `{SKILLS_VERIFY_COMMAND}`"
630                        ),
631                    },
632                    reproduce: "scripts/ci-plan".into(),
633                });
634            }
635
636            // used
637            let used: Vec<SkillInvocation> = invocations
638                .iter()
639                .filter(|(target, _)| target == &id)
640                .map(|(_, inv)| inv.clone())
641                .collect();
642            if active && used.is_empty() {
643                own.push(SkillFinding {
644                    level: SkillFindingLevel::Fail,
645                    code: "unused".into(),
646                    skill: Some(id.clone()),
647                    subject: path.clone(),
648                    message: format!(
649                        "orphan: nothing invokes skill '{id}'; reference `{SKILL_URI_PREFIX}{id}` \
650                         from a workflow, prompt, profile, provider template, recipe or CI \
651                         workflow that should follow it"
652                    ),
653                    reproduce: explain.clone(),
654                });
655            }
656
657            let standing = if !contract.is_empty() {
658                SkillStanding::Invalid
659            } else if !active {
660                SkillStanding::NotRequired
661            } else if evidence == SkillEvidence::Untested || used.is_empty() {
662                SkillStanding::Orphan
663            } else if evidence.current() && documented && enforced {
664                SkillStanding::Proven
665            } else {
666                SkillStanding::Partial
667            };
668
669            findings.extend(own.iter().cloned());
670            skills.push(SkillStatus {
671                uri: format!("{SKILL_URI_PREFIX}{id}"),
672                title: o.title.clone().unwrap_or_else(|| id.clone()),
673                description: o.description.clone().unwrap_or_default(),
674                status,
675                version: meta.get("version").and_then(Value::as_u64).unwrap_or(0),
676                provenance: provenance_of(meta),
677                valid: contract.is_empty(),
678                tested: SkillTested {
679                    state: evidence,
680                    tests,
681                },
682                documented: SkillDocumented { documented, page },
683                enforced: SkillEnforced {
684                    enforced,
685                    doctrine: doctrine.clone(),
686                    gates,
687                },
688                used: SkillUsed {
689                    used: !used.is_empty(),
690                    invocations: used,
691                },
692                standing,
693                findings: own,
694                path,
695                id,
696            });
697        }
698
699        // a skill file the index refused holds no object, and a listing that dropped it would
700        // silently shrink: it is answered as invalid, with every diagnostic as a finding
701        let indexed: BTreeSet<&str> = skills.iter().map(|s| s.path.as_str()).collect();
702        let prefix = format!("{}/", skills_dir.trim_end_matches('/'));
703        let mut refused: BTreeMap<String, Vec<&crate::model::Diagnostic>> = BTreeMap::new();
704        for d in index
705            .diagnostics
706            .iter()
707            .filter(|d| d.severity == Severity::Error)
708        {
709            let Some(p) = d.path.as_deref() else { continue };
710            let Some(rest) = p.strip_prefix(&prefix) else {
711                continue;
712            };
713            if rest.ends_with("/SKILL.md") && rest.matches('/').count() == 1 && !indexed.contains(p)
714            {
715                refused.entry(p.to_string()).or_default().push(d);
716            }
717        }
718        for (path, diagnostics) in refused {
719            let id = path
720                .trim_start_matches(&prefix)
721                .trim_end_matches("/SKILL.md")
722                .to_string();
723            let own: Vec<SkillFinding> = diagnostics
724                .iter()
725                .map(|d| SkillFinding {
726                    level: SkillFindingLevel::Fail,
727                    code: "contract".into(),
728                    skill: Some(id.clone()),
729                    subject: path.clone(),
730                    message: format!("{}: {}", d.code, d.message),
731                    reproduce: "majordomus skills check".into(),
732                })
733                .collect();
734            findings.extend(own.iter().cloned());
735            skills.push(SkillStatus {
736                uri: format!("{SKILL_URI_PREFIX}{id}"),
737                title: id.clone(),
738                description: String::new(),
739                status: "unknown".into(),
740                version: 0,
741                provenance: None,
742                valid: false,
743                tested: SkillTested {
744                    state: SkillEvidence::Untested,
745                    tests: Vec::new(),
746                },
747                documented: SkillDocumented {
748                    documented: false,
749                    page: format!("{SKILL_PAGES_DIR}/{id}.md"),
750                },
751                enforced: SkillEnforced {
752                    enforced: false,
753                    doctrine: doctrine.clone(),
754                    gates: Vec::new(),
755                },
756                used: SkillUsed {
757                    used: false,
758                    invocations: Vec::new(),
759                },
760                standing: SkillStanding::Invalid,
761                findings: own,
762                path,
763                id,
764            });
765        }
766
767        // bindings to nothing: a test or an invocation naming a skill that is not there
768        for (test_path, named) in &markers {
769            for n in named.iter().filter(|n| !ids.contains(*n)) {
770                findings.push(SkillFinding {
771                    level: SkillFindingLevel::Fail,
772                    code: "unknown_skill_in_test".into(),
773                    skill: None,
774                    subject: test_path.clone(),
775                    message: format!(
776                        "the test names skill '{n}', which does not exist; a binding to nothing \
777                         reads as proof"
778                    ),
779                    reproduce: "majordomus skills list".into(),
780                });
781            }
782        }
783        for (target, inv) in &invocations {
784            if !ids.contains(target) {
785                findings.push(SkillFinding {
786                    level: SkillFindingLevel::Fail,
787                    code: "unknown_skill_invoked".into(),
788                    skill: None,
789                    subject: inv.path.clone(),
790                    message: format!(
791                        "line {} invokes skill '{target}' ({}), which does not exist",
792                        inv.line, inv.reference
793                    ),
794                    reproduce: "majordomus skills list".into(),
795                });
796            }
797        }
798
799        Skills { skills, findings }
800    }
801
802    /// One skill by id, the directory name under `.ai/repo/skills/`; `None` only when the
803    /// repository has no skill of that name, since one the index refused is still listed,
804    /// standing `invalid`.
805    ///
806    /// ```
807    /// use majordomus_cli::skill::Skills;
808    /// let empty = Skills { skills: vec![], findings: vec![] };
809    /// assert!(empty.skill("implement").is_none());
810    /// ```
811    pub fn skill(&self, id: &str) -> Option<&SkillStatus> {
812        self.skills.iter().find(|s| s.id == id)
813    }
814
815    /// How many findings refuse: `skills verify` is valid only when this is zero. Warnings
816    /// are counted apart, by [`Skills::warnings`], and never make it invalid.
817    ///
818    /// ```
819    /// use majordomus_cli::skill::Skills;
820    /// assert_eq!(Skills { skills: vec![], findings: vec![] }.failures(), 0);
821    /// ```
822    pub fn failures(&self) -> usize {
823        self.findings
824            .iter()
825            .filter(|f| f.level == SkillFindingLevel::Fail)
826            .count()
827    }
828
829    /// How many findings are warnings: debts a skill still owes that do not refuse the
830    /// verdict, such as a partial standing. Refusals are counted by [`Skills::failures`].
831    ///
832    /// ```
833    /// use majordomus_cli::skill::Skills;
834    /// assert_eq!(Skills { skills: vec![], findings: vec![] }.warnings(), 0);
835    /// ```
836    pub fn warnings(&self) -> usize {
837        self.findings
838            .iter()
839            .filter(|f| f.level == SkillFindingLevel::Warn)
840            .count()
841    }

871/// The command that reproduces the weakest of `tests`, or `otherwise` when that test names
872/// no runner that could reproduce it.
873fn weakest_reproduce(tests: &[SkillTest], otherwise: &str) -> String {
874    tests

881fn is_skill_id(word: &str) -> bool {
882    let mut chars = word.chars();
883    matches!(chars.next(), Some(c) if c.is_ascii_lowercase())
884        && chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-')
885}

887/// The skill ids a marker line names, when the line is a comment carrying the marker.
888///
889/// Only a comment line counts — `#`, `//` or `//!` first — so that a test which writes a
890/// marker into a fixture, as data, does not bind itself to the skill it writes.
891///
892/// ```
893/// use majordomus_cli::skill::marker_ids;
894/// let marker = concat!("majordomus-", "skill:");
895/// assert_eq!(marker_ids(&format!("# {marker} implement repo-review")), ["implement", "repo-review"]);
896/// assert_eq!(marker_ids(&format!("//! {marker} implement")), ["implement"]);
897/// assert!(marker_ids(&format!("printf '# {marker} x'")).is_empty());
898/// ```
899pub fn marker_ids(line: &str) -> Vec<String> {
900    let t = line.trim_start();
901    let body = if let Some(r) = t.strip_prefix("//!") {
902        r
903    } else if let Some(r) = t.strip_prefix("//") {
904        r
905    } else if let Some(r) = t.strip_prefix('#') {
906        r
907    } else {
908        return Vec::new();
909    };
910    let Some(rest) = body.trim_start().strip_prefix(TEST_MARKER) else {
911        return Vec::new();
912    };
913    rest.split(|c: char| c.is_whitespace() || c == ',')
914        .filter(|w| is_skill_id(w))
915        .map(str::to_string)
916        .collect()
917}

919/// Every tracked test a runner owns that carries a marker, with the ids it names.
920fn test_markers(root: &Path, tracked: &[String]) -> BTreeMap<String, BTreeSet<String>> {
921    let mut out: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
922    for path in tracked {
923        if !TEST_DIRS.iter().any(|d| path.starts_with(&format!("{d}/"))) {
924            continue;
925        }
926        if TestId::of(path).is_none() {
927            continue;
928        }
929        let Ok(text) = std::fs::read_to_string(root.join(path)) else {
930            continue;
931        };
932        let named: BTreeSet<String> = text.lines().flat_map(marker_ids).collect();
933        if !named.is_empty() {
934            out.insert(path.clone(), named);
935        }
936    }
937    out
938}

940/// The patterns of every invocation surface: the layer's workflows, prompts, profiles and
941/// provider templates (read from the manifest's sections, beside the skills section), and the
942/// recipes, CI workflows and root bootstraps.
943///
944/// ```
945/// use majordomus_cli::skill::invocation_surfaces;
946/// let mut sections = std::collections::BTreeMap::new();
947/// sections.insert("prompts".to_string(), ".ai/repo/prompts".to_string());
948/// let s = invocation_surfaces(&sections, ".ai/repo/skills");
949/// assert!(s.contains(&".ai/repo/prompts/**".to_string()));
950/// assert!(s.contains(&".ai/repo/providers/**".to_string()));
951/// assert!(s.contains(&"justfile".to_string()));
952/// ```
953pub fn invocation_surfaces(sections: &BTreeMap<String, String>, skills_dir: &str) -> Vec<String> {
954    let mut out: Vec<String> = ["workflows", "prompts", "profiles"]
955        .iter()
956        .filter_map(|k| sections.get(*k))
957        .map(|d| format!("{}/**", d.trim_end_matches('/')))
958        .collect();
959    if let Some((layer, _)) = skills_dir.trim_end_matches('/').rsplit_once('/') {
960        out.push(format!("{layer}/providers/**"));
961    }
962    out.extend(ROOT_SURFACES.iter().map(|s| s.to_string()));
963    out
964}

966/// Every reference an invocation surface makes to a skill: `(target id, where)`, in path and
967/// line order.
968fn invocation_references(
969    root: &Path,
970    tracked: &[String],
971    index: &Index,
972    skills_dir: &str,
973) -> Vec<(String, SkillInvocation)> {
974    let globs: Vec<Glob> = invocation_surfaces(&index.repository.sections, skills_dir)
975        .iter()
976        .map(|p| Glob::new(p))
977        .collect();
978    let path_prefix = format!("{}/", skills_dir.trim_end_matches('/'));
979    let mut out = Vec::new();
980    for path in tracked {
981        if !globs.iter().any(|g| g.matches(path)) {
982            continue;
983        }
984        let Ok(text) = std::fs::read_to_string(root.join(path)) else {
985            continue;
986        };
987        for (n, line) in text.lines().enumerate() {
988            for (target, reference) in references_in(line, &path_prefix) {
989                out.push((
990                    target,
991                    SkillInvocation {
992                        path: path.clone(),
993                        line: n + 1,
994                        reference,
995                    },
996                ));
997            }
998        }
999    }
1000    out
1001}

1003/// The skill references one line makes: `majordomus://skill/<id>`, and
1004/// `<skills dir>/<id>/SKILL.md`.
1005///
1006/// ```
1007/// use majordomus_cli::skill::references_in;
1008/// let r = references_in(
1009///     "Follow majordomus://skill/repo-review, then .ai/repo/skills/implement/SKILL.md.",
1010///     ".ai/repo/skills/",
1011/// );
1012/// assert_eq!(r[0].0, "repo-review");
1013/// assert_eq!(r[1], ("implement".to_string(), ".ai/repo/skills/implement/SKILL.md".to_string()));
1014/// ```
1015pub fn references_in(line: &str, skills_prefix: &str) -> Vec<(String, String)> {
1016    let mut out = Vec::new();
1017    let id_end = |s: &str| {
1018        s.find(|c: char| !(c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-'))
1019            .unwrap_or(s.len())
1020    };
1021    let mut rest = line;
1022    while let Some(at) = rest.find(SKILL_URI_PREFIX) {
1023        let after = &rest[at + SKILL_URI_PREFIX.len()..];
1024        let id = after[..id_end(after)].trim_end_matches('-');
1025        if is_skill_id(id) {
1026            out.push((id.to_string(), format!("{SKILL_URI_PREFIX}{id}")));
1027        }
1028        rest = &after[id.len()..];
1029    }
1030    let mut rest = line;
1031    while let Some(at) = rest.find(skills_prefix) {
1032        let after = &rest[at + skills_prefix.len()..];
1033        let end = id_end(after);
1034        let id = &after[..end];
1035        if is_skill_id(id) && after[end..].starts_with("/SKILL.md") {
1036            out.push((id.to_string(), format!("{skills_prefix}{id}/SKILL.md")));
1037        }
1038        rest = &after[end..];
1039    }
1040    out
1041}

1043/// The gates a change to `path` selects whose command runs the skills verification.
1044fn verifying_gates(model: Option<&GateModel>, path: &str) -> Vec<String> {
1045    let Some(model) = model else {
1046        return Vec::new();
1047    };
1048    let verifying: BTreeSet<&str> = model
1049        .gates
1050        .iter()
1051        .filter(|g| g.runs.contains(SKILLS_VERIFY_COMMAND))
1052        .map(|g| g.id.as_str())
1053        .collect();
1054    let mut out: BTreeSet<String> = BTreeSet::new();
1055    for class in model.classes.iter().filter(|c| c.matches(path)) {
1056        if class.escalates() {
1057            out.extend(verifying.iter().map(|g| g.to_string()));
1058        } else {
1059            out.extend(
1060                class
1061                    .named()
1062                    .iter()
1063                    .filter(|g| verifying.contains(g.as_str()))
1064                    .cloned(),
1065            );
1066        }
1067    }
1068    out.into_iter().collect()
1069}

1294    /// A skill the index holds with an error against it is invalid and the error is its
1295    /// finding; its id and title fall back to its identity when the file declares neither.
1296    /// A skill file the index refused is listed as invalid too, under the skills directory
1297    /// the layout implies when the manifest names none; an error elsewhere, or on a file
1298    /// beside a skill, or on no file at all, is no skill.
1299    #[test]
1300    fn a_contract_error_makes_a_skill_invalid_and_nothing_else_is_a_skill() {
1301        use crate::model::Diagnostic;
1302        let repo = crate::synthetic::SyntheticRepository::small().unwrap();
1303        let mut index = repo.index().unwrap();
1304        index.repository.sections.remove("skills");
1305        let plain = ".ai/repo/skills/plain/SKILL.md";
1306        index.objects.push(skill_object(
1307            plain,
1308            "plain",
1309            serde_json::json!({ "status": "active" }),
1310        ));
1311        for (path, message) in [
1312            (Some(plain), "a field is wrong"),
1313            (Some(".ai/repo/skills/refused/SKILL.md"), "refused outright"),
1314            (Some(".ai/repo/skills/refused/notes.md"), "beside a skill"),
1315            (Some(".ai/repo/skills/deep/er/SKILL.md"), "too deep"),
1316            (Some(".ai/repo/rules/x.md"), "elsewhere"),
1317            (None, "no file"),
1318        ] {
1319            index.diagnostics.push(Diagnostic::error(
1320                "schema",
1321                path.map(str::to_string),
1322                message,
1323            ));
1324        }
1325
1326        let skills = Skills::build(&index, &Ledger::empty());
1327        let ids: Vec<&str> = skills.skills.iter().map(|s| s.id.as_str()).collect();
1328        assert_eq!(ids, ["plain", "refused"]);
1329
1330        let s = skills.skill("plain").unwrap();
1331        assert_eq!(s.standing, SkillStanding::Invalid);
1332        assert!(!s.valid);
1333        assert_eq!(s.title, "plain", "no title: the id stands in");
1334        let contract: Vec<&SkillFinding> =
1335            s.findings.iter().filter(|f| f.code == "contract").collect();
1336        assert_eq!(contract.len(), 1);
1337        assert_eq!(contract[0].message, "schema: a field is wrong");
1338        assert_eq!(contract[0].reproduce, "majordomus skills check");
1339        assert_eq!(contract[0].level, SkillFindingLevel::Fail);
1340
1341        assert_eq!(
1342            skills.skill("refused").unwrap().standing,
1343            SkillStanding::Invalid
1344        );
1345    }
