1//! The remote-build guard (I3956): before a provider's shell tool runs a command, does it
2//! start a build on a cluster host outside the offload path?
3//!
4//! On 2026-10-22 a 33-minute `cargo test --doc` ran on lundra, the build host, at load 21. It
5//! had been started with a bare `ssh lundra 'cd …; cargo test'`, so it carried no owner and
6//! waited for no queue. Its rustc outlived the cargo and the shell that were killed above it.
7//! `majordomus offload derive` (I2398) is the path that queues a build, exports its owner
8//! and stops it as a process group (slice 2). This is the check that stops the other path
9//! before it runs.
10//!
11//! The judgement reads the command as a shell would, closely enough to find the program each
12//! simple command runs:
13//!
14//! - quoting (single, double, backslash), `;`, `&&`, `||`, `|`, `&`, newlines, subshells and
15//!   redirections;
16//! - leading assignments and the wrappers that run another program (`env A=B`, `nice -n 10`,
17//!   `nohup`, `timeout 60`, `sudo`, …);
18//! - a script handed to a shell (`bash -c '…'`, `sh -lc '…'`) or to `eval`, read again;
19//! - `ssh`, whose options are skipped the way OpenSSH's `getopt` does (`-o X`, `-p22`, `-tt`,
20//!   options after the destination, `--`), and whose remote command is the rest of its words
21//!   joined by spaces and read again, as the remote shell reads it.
22//!
23//! A command run on a host the offload declaration names (`.ai/repo/offload/*.yaml`, by its
24//! id, its address or the first label of its name) is refused when its program is a build:
25//! `cargo` (unless it only asks for a version or a list), `rustc`, `scripts/derive` or a
26//! `scripts/rust-*` script. Everything else is let through: a read (`ps`, `df`, `uptime`),
27//! `majordomus offload …` on the host, an interactive login, and any command on a host no
28//! declaration names. The offload command's own ssh runs inside the executable, never
29//! through the provider's shell tool, so this never sees it.
30

36/// The rule a refusal names: the offload is the one way a build runs on another machine.
37pub(crate) const RULE: &str = "project.a-derive-offload-waits-its-turn";

39/// How deep a script handed to a shell, to `eval` or to ssh is read again. A command nested
40/// deeper than this is let through rather than judged.
41const DEPTH: u8 = 8;

43/// A build a command would start on a declared cluster host, outside the offload path.
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub(crate) struct RemoteBuild {
46    /// The declared host's id.
47    pub(crate) host: String,
48    /// The ssh destination as the command spelt it.
49    pub(crate) destination: String,
50    /// The build program, as the remote command spelt it.
51    pub(crate) program: String,
52}
53
54impl RemoteBuild {
55    /// The text a provider hands the model: what was refused, under which rule, and the way.
56    pub(crate) fn refusal(&self) -> String {
57        format!(
58            "majordomus guard: `ssh {dest}` reaches the declared offload host {host}, and the \
59             command starts `{program}` there outside the offload path ({RULE}; I3956)\n  \
60             run it as `majordomus offload derive --host {host} <worktree>`: it waits its turn \
61             in the host's queue, carries its owner, and stops as a process group with \
62             `majordomus offload stop <tag>`\n  \
63             a read (ps, df, uptime) and `ssh {dest} majordomus offload …` are not refused",
64            dest = self.destination,
65            host = self.host,
66            program = self.program,
67        )
68    }
69}

71/// The build a provider's pre-tool payload would start on a declared host, when it would.
72///
73/// The payload is Claude Code's (`tool_input.command`, `cwd`). The checkout whose
74/// declaration is read is the one `cwd` lies in, or `base` when the payload names none;
75/// `hosts` reads that checkout's declared hosts and is asked only when the command runs ssh
76/// at all, because this runs before every shell command a provider makes.
77pub(crate) fn from_payload(
78    payload: &str,
79    base: Option<&Path>,
80    hosts: impl FnOnce(&Path) -> Vec<OffloadHost>,
81) -> Option<RemoteBuild> {
82    let v: serde_json::Value = serde_json::from_str(payload).ok()?;
83    let command = v["tool_input"]["command"].as_str()?;
84    if !command.contains("ssh") {
85        return None;
86    }
87    let cwd = v["cwd"].as_str().map(Path::new).or(base)?;
88    let repository = crate::repository::Repository::discover(cwd).ok()?;
89    let declared = hosts(repository.root());
90    remote_build(command, &declared)
91}

93/// Every host the checkout at `root` declares for the offload, read from
94/// `.ai/repo/offload/*.yaml` without building the index. A file that is not a valid
95/// declaration names no host; none at all is an empty list, and nothing is then judged.
96pub(crate) fn declared_hosts(root: &Path) -> Vec<OffloadHost> {
97    let Ok(entries) = std::fs::read_dir(root.join(".ai/repo/offload")) else {
98        return Vec::new();
99    };
100    let files: BTreeSet<_> = entries
101        .flatten()
102        .map(|e| e.path())
103        .filter(|p| p.extension().is_some_and(|x| x == "yaml"))
104        .collect();
105    files
106        .iter()
107        .filter_map(|f| OffloadDeclaration::read(f).ok())
108        .flat_map(|d| d.hosts)
109        .collect()
110}

112/// The build `command` would start on one of `hosts`, read as a shell reads it.
113pub(crate) fn remote_build(command: &str, hosts: &[OffloadHost]) -> Option<RemoteBuild> {
114    scan(command, hosts, None, 0)
115}

117/// Where a script runs: here (`None`), or on a declared host, by its id and the destination
118/// the command spelt.
119type On<'a> = Option<(&'a str, &'a str)>;
120
121fn scan(script: &str, hosts: &[OffloadHost], on: On<'_>, depth: u8) -> Option<RemoteBuild> {
122    if depth > DEPTH {
123        return None;
124    }
125    commands(script)
126        .iter()
127        .find_map(|words| judge(words, hosts, on, depth))
128}
129
130/// The build one simple command starts where it runs, if it starts one.
131fn judge(words: &[String], hosts: &[OffloadHost], on: On<'_>, depth: u8) -> Option<RemoteBuild> {
132    let words = unwrap(words);
133    let (program, args) = words.split_first()?;
134    let name = basename(program);
135    match name {
136        "ssh" => {
137            let (destination, rest) = ssh_target(args)?;
138            let host = declared(hosts, host_of(destination));
139            let remote = rest.join(" ");
140            match host {
141                Some(h) => scan(
142                    &remote,
143                    hosts,
144                    Some((h.id.as_str(), destination)),
145                    depth + 1,
146                ),
147                // an undeclared host is not judged, but a hop it makes to a declared one is
148                None => scan(&remote, hosts, None, depth + 1),
149            }
150        }
151        "bash" | "sh" | "zsh" | "dash" | "ksh" => match shell_script(args) {
152            Script::Inline(s) => scan(s, hosts, on, depth + 1),
153            Script::File(f) => build_on(f, &args[1..], on),
154            Script::None => None,
155        },
156        "eval" => scan(&args.join(" "), hosts, on, depth + 1),
157        _ => build_on(program, args, on),
158    }
159}

161/// `program` as the build it is on the declared host `on`, when it is one there.
162fn build_on(program: &str, args: &[String], on: On<'_>) -> Option<RemoteBuild> {
163    let (host, destination) = on?;
164    is_build(program, args).then(|| RemoteBuild {
165        host: host.to_string(),
166        destination: destination.to_string(),
167        program: program.to_string(),
168    })
169}

171/// Whether `program` with `args` builds: cargo (unless it only asks what it is), rustc,
172/// `scripts/derive` and the `scripts/rust-*` scripts, wherever they are spelt from.
173fn is_build(program: &str, args: &[String]) -> bool {
174    let name = basename(program);
175    match name {
176        "cargo" => {
177            !args.iter().all(|a| {
178                matches!(
179                    a.as_str(),
180                    "--version" | "-V" | "-vV" | "--list" | "--help" | "-h"
181                )
182            }) || args.is_empty()
183        }
184        "rustc" => true,
185        _ => {
186            let parent = Path::new(program)
187                .parent()
188                .and_then(Path::file_name)
189                .and_then(|p| p.to_str());
190            parent == Some("scripts") && (name == "derive" || name.starts_with("rust-"))
191        }
192    }
193}

197    program.rsplit('/').next().unwrap_or(program)

200/// The words of a simple command from the program it runs: leading assignments and the
201/// wrappers that run another program, with their options, taken off.
202fn unwrap(mut words: &[String]) -> &[String] {
203    loop {
204        let Some((first, rest)) = words.split_first() else {
205            return words;
206        };
207        if is_assignment(first) {
208            words = rest;
209            continue;
210        }
211        let Some((with_value, positional)) = wrapper(basename(first)) else {
212            return words;
213        };
214        words = skip_options(rest, with_value);
215        for _ in 0..positional {
216            words = words.split_first().map_or(words, |(_, r)| r);
217        }
218        // `env` takes its own assignments after its options
219        while words.first().is_some_and(|w| is_assignment(w)) {
220            words = &words[1..];
221        }
222    }
223}

225/// A wrapper that runs the program after it: the options of its that take a value, and how
226/// many positional words come before the program.
227fn wrapper(name: &str) -> Option<(&'static [&'static str], usize)> {
228    let pair: (&'static [&'static str], usize) = match name {
229        "env" => (&["-u", "-C", "-S", "--unset", "--chdir"], 0),
230        "nice" => (&["-n", "--adjustment"], 0),
231        "ionice" => (&["-c", "-n", "-p", "--class", "--classdata"], 0),
232        "sudo" => (
233            &[
234                "-u", "-g", "-h", "-p", "-C", "-D", "-U", "--user", "--group",
235            ],
236            0,
237        ),
238        "timeout" | "gtimeout" => (&["-s", "-k", "--signal", "--kill-after"], 1),
239        "stdbuf" => (&["-i", "-o", "-e"], 0),
240        "taskset" => (&[], 1),
241        "nohup" | "time" | "command" | "exec" | "builtin" | "setsid" | "caffeinate" => (&[], 0),
242        _ => return None,
243    };
244    Some(pair)
245}

247/// `words` after the leading options, an option in `with_value` taking the word after it
248/// (`-n 10`) unless it carries its value (`-n10`, `--adjustment=10`). `--` ends them.
249fn skip_options<'w>(mut words: &'w [String], with_value: &[&str]) -> &'w [String] {
250    while let Some((w, rest)) = words.split_first() {
251        if w == "--" {
252            return rest;
253        }
254        if !w.starts_with('-') || w == "-" {
255            return words;
256        }
257        words = rest;
258        if with_value.contains(&w.as_str()) {
259            words = words.split_first().map_or(words, |(_, r)| r);
260        }
261    }
262    words
263}

265/// `NAME=value`, as a shell reads an assignment before a command.
266fn is_assignment(word: &str) -> bool {
267    let Some((name, _)) = word.split_once('=') else {
268        return false;
269    };
270    let mut chars = name.chars();
271    chars
272        .next()
273        .is_some_and(|c| c.is_ascii_alphabetic() || c == '_')
274        && chars.all(|c| c.is_ascii_alphanumeric() || c == '_')
275}

277/// What a shell is asked to run.
278#[derive(Debug, PartialEq, Eq)]
279enum Script<'a> {
280    /// A script given with `-c`.
281    Inline(&'a str),
282    /// A script file.
283    File(&'a str),
284    /// Nothing: an interactive shell.
285    None,
286}
287
288/// The script a shell's `args` hand it: the first word after its options when one of them
289/// carries `c` (`-c`, `-lc`, `-ec`), otherwise the first word, a file.
290fn shell_script(args: &[String]) -> Script<'_> {
291    let mut inline = false;
292    for a in args {
293        if a == "--" {
294            continue;
295        }
296        if let Some(flags) = a.strip_prefix('-').filter(|f| !f.starts_with('-')) {
297            inline |= flags.contains('c');
298            continue;
299        }
300        if a.starts_with("--") {
301            continue;
302        }
303        return if inline {
304            Script::Inline(a)
305        } else {
306            Script::File(a)
307        };
308    }
309    Script::None
310}

312/// The options of OpenSSH's ssh that take a value.
313const SSH_WITH_VALUE: &str = "BbcDEeFIiJLlmOoPpQRSWw";
314
315/// The destination ssh's `args` name and the words of the remote command after it, options
316/// skipped as OpenSSH does: clustered (`-tt`, `-4p22`), with a value attached or in the next
317/// word, before the destination and after it, `--` ending them. `None` when no destination
318/// is named.
319fn ssh_target(args: &[String]) -> Option<(&str, &[String])> {
320    let mut destination = None;
321    let mut i = 0;
322    while i < args.len() {
323        let w = &args[i];
324        i += 1;
325        if w == "--" {
326            if destination.is_none() {
327                destination = Some(args.get(i)?.as_str());
328                i += 1;
329            }
330            break;
331        }
332        match w.strip_prefix('-').filter(|f| !f.is_empty()) {
333            Some(flags) => {
334                if let Some(at) = flags.find(|c| SSH_WITH_VALUE.contains(c)) {
335                    if at + 1 == flags.len() {
336                        i += 1;
337                    }
338                }
339            }
340            None if destination.is_none() => destination = Some(w.as_str()),
341            None => {
342                i -= 1;
343                break;
344            }
345        }
346    }
347    Some((destination?, &args[i.min(args.len())..]))
348}

350/// The host of an ssh destination: `host`, `user@host`, `ssh://user@host:port`, in lower case.
351fn host_of(destination: &str) -> String {
352    let (uri, rest) = match destination.strip_prefix("ssh://") {
353        Some(r) => (true, r),
354        None => (false, destination),
355    };
356    let host = rest.rsplit('@').next().unwrap_or(rest);
357    let host = if uri {
358        host.split([':', '/']).next().unwrap_or(host)
359    } else {
360        host
361    };
362    host.to_ascii_lowercase()
363}

365/// The declared host `host` is, by its id, the host of its address, or a name whose first
366/// label is its id (`lundra.tailnet.ts.net`).
367fn declared(hosts: &[OffloadHost], host: String) -> Option<&OffloadHost> {
368    let label = host.split('.').next().unwrap_or_default();
369    hosts.iter().find(|h| {
370        let id = h.id.to_ascii_lowercase();
371        host == id || label == id || host == host_of(&h.address)
372    })
373}

375/// One lexical piece of a command line.
376#[derive(Debug, PartialEq, Eq)]
377enum Token {
378    /// A word, its quotes removed.
379    Word(String),
380    /// What ends a simple command: `;`, `&`, `|`, a newline, a parenthesis or a backtick.
381    End,
382    /// A redirection: the word after it is its target, not an argument.
383    Redirect,
384}
385
386/// The simple commands of `script`, each its words, a redirection's target left out.
387fn commands(script: &str) -> Vec<Vec<String>> {
388    let mut out = Vec::new();
389    let mut current = Vec::new();
390    let mut target = false;
391    for t in tokens(script) {
392        match t {
393            Token::Word(_) if target => target = false,
394            Token::Word(w) => current.push(w),
395            Token::Redirect => target = true,
396            Token::End => {
397                target = false;
398                out.push(std::mem::take(&mut current));
399            }
400        }
401    }
402    out.push(current);
403    out.retain(|c| !c.is_empty());
404    out
405}

407/// The tokens of `script`, quoted as a POSIX shell quotes: `'…'` literally, `"…"` with `\`
408/// escaping `"`, `\`, `$` and a backtick, `\` outside quotes escaping one character, `#` at
409/// the start of a word opening a comment to the end of the line.
410fn tokens(script: &str) -> Vec<Token> {
411    let mut out = Vec::new();
412    let mut word = String::new();
413    let mut open = false;
414    let mut chars = script.chars().peekable();
415    while let Some(c) = chars.next() {
416        match c {
417            ' ' | '\t' => flush(&mut word, &mut open, &mut out),
418            '\n' | ';' | '|' | '(' | ')' | '`' => {
419                flush(&mut word, &mut open, &mut out);
420                out.push(Token::End);
421            }
422            '&' if chars.peek() == Some(&'>') => {
423                flush(&mut word, &mut open, &mut out);
424                chars.next();
425                out.push(Token::Redirect);
426            }
427            '&' => {
428                flush(&mut word, &mut open, &mut out);
429                out.push(Token::End);
430            }
431            '<' | '>' => {
432                flush(&mut word, &mut open, &mut out);
433                while chars.next_if(|n| matches!(n, '>' | '<' | '&')).is_some() {}
434                out.push(Token::Redirect);
435            }
436            '#' if !open => while chars.next_if(|n| *n != '\n').is_some() {},
437            '\'' => {
438                open = true;
439                word.extend(chars.by_ref().take_while(|n| *n != '\''));
440            }
441            '"' => {
442                open = true;
443                while let Some(n) = chars.next() {
444                    match n {
445                        '"' => break,
446                        '\\' => match chars.next_if(|e| matches!(e, '"' | '\\' | '$' | '`')) {
447                            Some(e) => word.push(e),
448                            None => word.push('\\'),
449                        },
450                        _ => word.push(n),
451                    }
452                }
453            }
454            '\\' => match chars.next() {
455                Some('\n') | None => {}
456                Some(e) => {
457                    open = true;
458                    word.push(e);
459                }
460            },
461            _ => {
462                open = true;
463                word.push(c);
464            }
465        }
466    }
467    flush(&mut word, &mut open, &mut out);
468    out
469}
470
471/// The word being read, as a token, when one is open.
472fn flush(word: &mut String, open: &mut bool, out: &mut Vec<Token>) {
473    if *open {
474        out.push(Token::Word(std::mem::take(word)));
475        *open = false;
476    }
477}

623    fn a_payload_is_judged_in_the_checkout_of_its_cwd() {
624        let dir = tempfile::tempdir().unwrap();
625        std::fs::create_dir_all(dir.path().join(".ai")).unwrap();
626        let skeleton = Path::new(env!("CARGO_MANIFEST_DIR")).join("../../share/skeleton");
627        std::fs::copy(
628            skeleton.join("ai/manifest.yaml"),
629            dir.path().join(".ai/manifest.yaml"),
630        )
631        .unwrap();
632        let root = dir.path().canonicalize().unwrap();
633        let payload = |command: &str, cwd: Option<&Path>| {
634            let mut v =
635                serde_json::json!({ "tool_name": "Bash", "tool_input": { "command": command } });
636            if let Some(c) = cwd {
637                v["cwd"] = serde_json::json!(c);
638            }
639            v.to_string()
640        };
641        let hosts = |asked: &Path| {
642            assert_eq!(asked, root.as_path(), "the checkout of the cwd is read");
643            cluster()
644        };
645        let never = |_: &Path| -> Vec<OffloadHost> { unreachable!("not asked") };
646        let build = payload("ssh lundra cargo test", Some(dir.path()));
647        assert_eq!(from_payload(&build, None, hosts).unwrap().host, "lundra");
648        let no_cwd = payload("ssh lundra cargo test", None);
649        assert!(
650            from_payload(&no_cwd, Some(dir.path()), hosts).is_some(),
651            "base stands in"
652        );
653        assert!(
654            from_payload(&no_cwd, None, never).is_none(),
655            "nowhere to read a declaration"
656        );
657        assert!(from_payload(&payload("ls", Some(dir.path())), None, never).is_none());
658        assert!(from_payload("not json", None, never).is_none());
659        let edit = r#"{"tool_input":{"file_path":"/a"}}"#;
660        assert!(from_payload(edit, None, never).is_none(), "no command");
661        let outside = tempfile::tempdir().unwrap();
662        let bare = payload("ssh lundra cargo test", Some(outside.path()));
663        assert!(
664            from_payload(&bare, None, never).is_none(),
665            "no checkout above it"
666        );

688    #[test]
689    fn the_shell_is_tokenised_as_a_shell_reads_it() {
690        use Token::*;
691        let w = |s: &str| Word(s.to_string());
692        assert_eq!(
693            tokens(
694                r#"a 'b c' "d \"e\" \x" f\ g \
695h"#
696            ),
697            vec![w("a"), w("b c"), w(r#"d "e" \x"#), w("f g"), w("h")]
698        );
699        assert_eq!(tokens("a&&b"), vec![w("a"), End, End, w("b")]);
700        assert_eq!(tokens("a &> f"), vec![w("a"), Redirect, w("f")]);
701        assert_eq!(tokens("a 2>&1"), vec![w("a"), w("2"), Redirect, w("1")]);
702        assert_eq!(
703            tokens("a '' b"),
704            vec![w("a"), w(""), w("b")],
705            "an empty quoted word"
706        );
707        assert_eq!(tokens("a \\"), vec![w("a")], "a trailing backslash");
708        assert_eq!(tokens("a#b"), vec![w("a#b")], "a hash inside a word");
709        assert_eq!(tokens("a # b c\nd"), vec![w("a"), End, w("d")], "a comment");
710        assert_eq!(tokens("$(x)"), vec![w("$"), End, w("x"), End]);
711        assert_eq!(
712            commands("a > f b; c"),
713            vec![words(&["a", "b"]), words(&["c"])]
714        );
715        assert_eq!(commands(">"), Vec::<Vec<String>>::new());
716    }
