1//! The banner: a snapshot rendered for a person at a terminal.
2//!
3//! This is a view and nothing else. It discovers nothing, reads no file, runs no command
4//! and holds no fact of its own — every number, name, URL and command in it comes out of
5//! the [`super::RepositoryEnvironment`] it was handed. That is what keeps the thing a
6//! person sees on `cd` and the thing `GET /api/v1/environment` returns from ever
7//! disagreeing.
8//!
9//! # What it refuses to do
10//!
11//! - Print anything it was not given. A count that was not resolved is shown as `—`, never
12//!   as `0`.
13//! - Print anything unsanitised. Every value goes through [`super::text::sanitise`]; a
14//!   branch name is arbitrary text chosen by whoever made the branch.
15//! - Measure in bytes. Every column is [`super::text::width`], so a box does not fall
16//!   apart on a name that is not ASCII.
17//! - Colour when it was not asked to. `NO_COLOR`, a pipe, and `CI` each turn it off.
18//! - Fail. A render error would take the shell down with it, so there is no path here
19//!   that can produce one.
20

31/// How much of a snapshot to show.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
33pub enum BannerMode {
34    /// Decide from the terminal and from whether this snapshot says anything new.
35    #[default]
36    Auto,
37    /// The whole box.
38    Full,
39    /// Two lines.
40    Compact,
41    /// Nothing.
42    Off,
43}
44
45impl BannerMode {
46    /// Parse a mode by name; `None` for a word that is not one.
47    ///
48    /// ```
49    /// use majordomus_cli::environment::render::BannerMode;
50    /// assert_eq!(BannerMode::parse("compact"), Some(BannerMode::Compact));
51    /// assert_eq!(BannerMode::parse("COMPACT"), Some(BannerMode::Compact));
52    /// assert_eq!(BannerMode::parse("loud"), None);
53    /// ```
54    pub fn parse(word: &str) -> Option<Self> {
55        match word.trim().to_ascii_lowercase().as_str() {
56            "auto" => Some(BannerMode::Auto),
57            "full" => Some(BannerMode::Full),
58            "compact" => Some(BannerMode::Compact),
59            "off" | "none" | "silent" => Some(BannerMode::Off),
60            _ => None,
61        }
62    }
63
64    /// The word as written.
65    pub fn as_str(self) -> &'static str {
66        match self {
67            BannerMode::Auto => "auto",
68            BannerMode::Full => "full",
69            BannerMode::Compact => "compact",
70            BannerMode::Off => "off",
71        }
72    }
73}

75/// What the terminal on the other end can take.
76#[derive(Debug, Clone, Copy, PartialEq, Eq)]
77pub struct Presentation {
78    /// Whether escape sequences may be used.
79    pub colour: bool,
80    /// Whether characters outside ASCII may be used.
81    pub unicode: bool,
82    /// How many columns there are.
83    pub columns: usize,
84    /// Whether the output is going to a terminal at all.
85    pub interactive: bool,
86}
87
88/// The widest a banner is drawn, however wide the terminal is. A box that stretches across
89/// a 200-column window is harder to read than one that does not.
90pub const MAX_WIDTH: usize = 78;
91
92/// The narrowest terminal the box is drawn in; below it, the plain two-line form is used
93/// whatever the mode, because a box that does not fit is worse than no box.
94pub const MIN_BOX_WIDTH: usize = 46;
95
96impl Default for Presentation {
97    fn default() -> Self {
98        Presentation {
99            colour: false,
100            unicode: true,
101            columns: MAX_WIDTH,
102            interactive: false,
103        }
104    }
105}
106
107impl Presentation {
108    /// What this process's standard error can take.
109    ///
110    /// The banner goes to standard error, not standard output: standard output belongs to
111    /// whatever the caller is capturing, and a banner in the middle of `majordomus env
112    /// status --json | jq` would be a bug in every consumer at once.
113    ///
114    /// `NO_COLOR` — set to anything at all, including the empty string, per the
115    /// convention — turns colour off. So does `CI`, and so does a standard error that is
116    /// not a terminal.
117    pub fn detect() -> Self {
118        let interactive = is_terminal(libc::STDERR_FILENO);
119        let ci = std::env::var_os("CI").is_some();
120        Presentation {
121            colour: interactive && !ci && std::env::var_os("NO_COLOR").is_none(),
122            unicode: unicode_locale(),
123            columns: terminal_columns().unwrap_or(MAX_WIDTH),
124            interactive: interactive && !ci,
125        }
126    }
127}

129/// Is this file descriptor a terminal?
130fn is_terminal(fd: i32) -> bool {
131    // SAFETY: isatty takes a file descriptor and reads no memory; any integer is a
132    // defined argument, and an invalid one returns 0.
133    unsafe { libc::isatty(fd) == 1 }
134}

136/// The terminal's width, from `COLUMNS` when a shell exported it, else from the terminal
137/// itself. `None` when neither answers, which is every non-interactive case.
138fn terminal_columns() -> Option<usize> {
139    if let Some(columns) = std::env::var("COLUMNS")
140        .ok()
141        .and_then(|c| c.trim().parse::<usize>().ok())
142    {
143        if columns > 0 {
144            return Some(columns);
145        }
146    }
147    let mut size: libc::winsize = unsafe { std::mem::zeroed() };
148    // SAFETY: TIOCGWINSZ writes a winsize into the pointer given; the struct is owned
149    // here, correctly sized, and the call is a no-op returning -1 on a non-terminal.
150    let ok = unsafe { libc::ioctl(libc::STDERR_FILENO, libc::TIOCGWINSZ, &mut size) } == 0;
151    (ok && size.ws_col > 0).then_some(size.ws_col as usize)
152}

154/// Does the locale say this terminal can render characters outside ASCII?
155fn unicode_locale() -> bool {
156    ["LC_ALL", "LC_CTYPE", "LANG"]
157        .iter()
158        .filter_map(|k| std::env::var(k).ok())
159        .find(|v| !v.is_empty())
160        .map(|v| {
161            v.to_ascii_uppercase().contains("UTF-8") || v.to_ascii_uppercase().contains("UTF8")
162        })
163        // No locale set at all: a terminal that says nothing is assumed to be modern.
164        // Being wrong here costs a few mojibake characters, and being wrong the other way
165        // costs every non-ASCII repository a legible banner.
166        .unwrap_or(true)
167}

235/// Render a banner, or nothing when the mode says nothing.
236///
237/// `seen_before` is the [`RepositoryEnvironment::news_digest`] of the last snapshot shown to
238/// this person, which decides what `auto` does: a repository that has not changed since
239/// they last looked gets the two-line form, and a first look or a changed repository gets
240/// the box. A server that only came up or went away is not a changed repository.
241pub fn banner(
242    environment: &RepositoryEnvironment,
243    mode: BannerMode,
244    presentation: &Presentation,
245    seen_before: Option<&str>,
246) -> Option<String> {
247    let resolved = match mode {
248        BannerMode::Off => return None,
249        BannerMode::Auto => {
250            if !presentation.interactive {
251                return None;
252            }
253            match seen_before {
254                Some(digest) if digest == environment.news_digest() => BannerMode::Compact,
255                _ => BannerMode::Full,
256            }
257        }
258        other => other,
259    };
260    Some(match resolved {
261        BannerMode::Compact => compact(environment, presentation),
262        _ if presentation.columns < MIN_BOX_WIDTH => compact(environment, presentation),
263        _ => full(environment, presentation),
264    })
265}

267/// How wide the label column of an ordinary row is.
268const LABEL_WIDTH: usize = 11;
269
270/// How wide the label column of a command row is: a command is the label there, and
271/// `just cockpit-assets-check` truncated to eleven columns is not a command.
272const COMMAND_WIDTH: usize = 26;
273
274/// One line of the box.
275enum Row {
276    /// A label and a value.
277    Field(String, String),
278    /// A command and what it does; the command gets the wider column.
279    Command(String, String),
280    /// Nothing, for spacing.
281    Blank,
282    /// A horizontal divider.
283    Divider,
284}

286/// The full banner.
287fn full(environment: &RepositoryEnvironment, presentation: &Presentation) -> String {
288    let g = glyphs(presentation);
289    let inner = presentation.columns.clamp(MIN_BOX_WIDTH, MAX_WIDTH) - 2;
290    let mut rows: Vec<Row> = Vec::new();
291
292    // ---- what this is
293    let name = sanitise(&environment.project.name).to_uppercase();
294    let version = format!("v{}", sanitise(&environment.project.version));
295    let title_gap = inner.saturating_sub(2 + width(&name) + width(&version) + 2);
296    rows.push(Row::Field(
297        format!("{name}{}{version}", " ".repeat(title_gap.max(1))),
298        String::new(),
299    ));
300    rows.push(Row::Field(
301        truncate_with(
302            &sanitise(&environment.project.summary),
303            inner - 4,
304            g.ellipsis,
305        ),
306        String::new(),
307    ));
308    rows.push(Row::Divider);
309
310    // ---- the checkout
311    let branch = match &environment.vcs {
312        VcsState::Git(tree) => tree
313            .branch
314            .clone()
315            .map(|b| sanitise(&b))
316            .unwrap_or_else(|| match &tree.head {
317                Some(head) => format!("detached at {}", short(head)),
318                None => "unborn".into(),
319            }),
320        VcsState::Unavailable { .. } => g.unresolved.into(),
321    };
322    let repository_name = sanitise(&environment.repository.name);
323    rows.push(Row::Field(
324        "repo".into(),
325        two_columns(
326            &repository_name,
327            "branch",
328            &branch,
329            inner - LABEL_WIDTH - 2,
330            g,
331        ),
332    ));
333    rows.push(Row::Field("git".into(), git_line(environment, g)));
334    let runtime = toolchain_line(environment, g);
335    if !runtime.is_empty() {
336        rows.push(Row::Field("runtime".into(), runtime));
337    }
338
339    // ---- what the layer holds
340    rows.push(Row::Blank);
341    rows.push(Row::Field(
342        "layer".into(),
343        layer_line(environment, g, inner - LABEL_WIDTH - 2),
344    ));
345    let providers = provider_line(environment, g);
346    if !providers.is_empty() {
347        rows.push(Row::Field("agents".into(), providers));
348    }
349
350    // ---- where things are
351    let running: Vec<&super::ServiceState> = environment
352        .services
353        .iter()
354        .filter(|s| s.url.is_some() && s.id != "index" && s.id != "openapi")
355        .collect();
356    if !running.is_empty() {
357        rows.push(Row::Blank);
358        for service in running {
359            let url = service.url.as_deref().unwrap_or_default();
360            let mark = match service.availability {
361                ServiceAvailability::Available => g.yes,
362                ServiceAvailability::NotRunning => g.no,
363                ServiceAvailability::Unknown => g.unknown,
364            };
365            rows.push(Row::Field(
366                sanitise(&service.id),
367                format!("{} {mark}", sanitise(url)),
368            ));
369        }
370    }
371
372    // ---- what to run
373    let entrypoints = &environment.workflows.entrypoints;
374    if !entrypoints.is_empty() {
375        rows.push(Row::Divider);
376        for entry in entrypoints.iter().take(4) {
377            rows.push(Row::Command(
378                sanitise(&entry.command),
379                truncate_with(
380                    &sanitise(entry.description.as_deref().unwrap_or_default()),
381                    inner.saturating_sub(COMMAND_WIDTH + 2),
382                    g.ellipsis,
383                ),
384            ));
385        }
386    }
387
388    // ---- anything wrong
389    if let Some(worst) = environment.headline_diagnostic() {
390        rows.push(Row::Divider);
391        rows.push(Row::Field(
392            match worst.severity {
393                Severity::Error => g.no.into(),
394                _ => g.unknown.into(),
395            },
396            truncate_with(
397                &sanitise(&worst.message),
398                inner.saturating_sub(6),
399                g.ellipsis,
400            ),
401        ));
402    }
403
404    draw(&rows, inner, g, presentation)
405}

407/// A label column and a value column, both padded, for a row that carries two pairs.
408fn two_columns(first: &str, label: &str, second: &str, budget: usize, g: &Glyphs) -> String {
409    let left = budget.saturating_sub(width(label) + 2).min(budget) / 2 + 4;
410    format!(
411        "{}{label}  {}",
412        pad_with(first, left.min(budget), g.ellipsis),
413        truncate_with(
414            second,
415            budget.saturating_sub(left + width(label) + 2),
416            g.ellipsis
417        )
418    )
419}

421/// `✱ 3 modified · 2 staged · ↑2 ↓1`, or `clean`.
422fn git_line(environment: &RepositoryEnvironment, g: &Glyphs) -> String {
423    let Some(tree) = environment.vcs.tree() else {
424        return match &environment.vcs {
425            VcsState::Unavailable { reason } => truncate_with(&sanitise(reason), 50, g.ellipsis),
426            VcsState::Git(_) => unreachable!("the tree was just taken"),
427        };
428    };
429    let mut parts: Vec<String> = Vec::new();
430    if tree.clean {
431        parts.push("clean".into());
432    } else {
433        let mut dirt: Vec<String> = Vec::new();
434        if tree.staged > 0 {
435            dirt.push(format!("{} staged", tree.staged));
436        }
437        if tree.modified > 0 {
438            dirt.push(format!("{} modified", tree.modified));
439        }
440        if tree.untracked > 0 {
441            dirt.push(format!("{} untracked", tree.untracked));
442        }
443        if tree.conflicted > 0 {
444            dirt.push(format!("{} conflicted", tree.conflicted));
445        }
446        parts.push(format!("{} {}", g.dirty, dirt.join(", ")));
447    }
448    match (tree.ahead, tree.behind) {
449        (Some(0), Some(0)) => parts.push("in step".into()),
450        (Some(a), Some(b)) => parts.push(format!("{}{a} {}{b}", g.ahead, g.behind)),
451        // No upstream is not "in step with nothing"; it is a branch nobody else has.
452        _ => parts.push("no upstream".into()),
453    }
454    parts.join(&format!(" {} ", g.bullet))
455}

457/// `Rust 1.90.0 · Node 22.20.0`, showing only what the repository declares.
458fn toolchain_line(environment: &RepositoryEnvironment, g: &Glyphs) -> String {
459    environment
460        .toolchains
461        .iter()
462        .map(|t| {
463            let version = match (&t.installed, t.availability) {
464                (Some(v), _) => sanitise(v),
465                (None, ToolchainAvailability::Missing) => format!("{} missing", g.no),
466                (None, _) => t
467                    .declared
468                    .as_deref()
469                    .map(|d| format!("{} declared", sanitise(d)))
470                    .unwrap_or_else(|| g.unresolved.into()),
471            };
472            format!("{} {version}", sanitise(&t.title))
473        })
474        .collect::<Vec<_>>()
475        .join(&format!(" {} ", g.bullet))
476}

478/// `rule 87 · use-case 38 · adr 19`, largest first, or a note that nothing counted them.
479fn layer_line(environment: &RepositoryEnvironment, g: &Glyphs, budget: usize) -> String {
480    if environment.layer.state == TierState::Unavailable {
481        return format!("{}  (run `majordomus env status` to count)", g.unresolved);
482    }
483    let mut kinds = environment.layer.kinds.clone();
484    // Largest first: what a repository is mostly made of is what a person wants to see
485    // when the line has to be cut. The rank is the negated count, so the canonical order —
486    // which ranks ascending and ends on an identity — reads the same way here as everywhere.
487    crate::order::canonical(&mut kinds);
488    let mut line = String::new();
489    for kind in &kinds {
490        let part = format!("{} {}", sanitise(&kind.kind), kind.count);
491        let separator = if line.is_empty() {
492            String::new()
493        } else {
494            format!(" {} ", g.bullet)
495        };
496        if width(&line) + width(&separator) + width(&part) > budget {
497            break;
498        }
499        line.push_str(&separator);
500        line.push_str(&part);
501    }
502    if line.is_empty() {
503        return g.unresolved.into();
504    }
505    line
506}

508/// `AGENTS.md ✓ · CLAUDE.md ✗`.
509fn provider_line(environment: &RepositoryEnvironment, g: &Glyphs) -> String {
510    environment
511        .providers
512        .iter()
513        .map(|p| {
514            let mark = match p.state {
515                ProjectionState::Current => g.yes,
516                ProjectionState::Stale | ProjectionState::Absent => g.no,
517                ProjectionState::Unknown => g.unknown,
518            };
519            format!("{} {mark}", sanitise(&p.target))
520        })
521        .collect::<Vec<_>>()
522        .join(&format!(" {} ", g.bullet))
523}

525/// Draw the rows inside a box.
526fn draw(rows: &[Row], inner: usize, g: &Glyphs, presentation: &Presentation) -> String {
527    let dim = |s: &str| -> String {
528        if presentation.colour {
529            format!("\u{1b}[2m{s}\u{1b}[0m")
530        } else {
531            s.to_string()
532        }
533    };
534    let bold = |s: &str| -> String {
535        if presentation.colour {
536            format!("\u{1b}[1m{s}\u{1b}[0m")
537        } else {
538            s.to_string()
539        }
540    };
541    let mut out = String::new();
542    let bar = g.horizontal.repeat(inner);
543    let _ = writeln!(out, "{}{bar}{}", g.top_left, g.top_right);
544    for (i, row) in rows.iter().enumerate() {
545        match row {
546            Row::Divider => {
547                let _ = writeln!(out, "{}{bar}{}", g.tee_left, g.tee_right);
548            }
549            Row::Blank => {
550                let _ = writeln!(out, "{}{}{}", g.vertical, " ".repeat(inner), g.vertical);
551            }
552            Row::Field(label, value) if value.is_empty() => {
553                // A heading row: the label is the whole line.
554                let text = pad_with(&format!("  {label}"), inner, g.ellipsis);
555                let painted = if i == 0 { bold(&text) } else { dim(&text) };
556                let _ = writeln!(out, "{}{painted}{}", g.vertical, g.vertical);
557            }
558            Row::Field(label, value) | Row::Command(label, value) => {
559                let column = match row {
560                    Row::Command(_, _) => COMMAND_WIDTH,
561                    _ => LABEL_WIDTH,
562                };
563                let padded_label = pad_with(&format!("  {label}"), column, g.ellipsis);
564                let padded_value = pad_with(value, inner.saturating_sub(column), g.ellipsis);
565                let _ = writeln!(
566                    out,
567                    "{}{}{padded_value}{}",
568                    g.vertical,
569                    dim(&padded_label),
570                    g.vertical
571                );
572            }
573        }
574    }
575    let _ = writeln!(out, "{}{bar}{}", g.bottom_left, g.bottom_right);
576    out
577}

579/// The two-line form.
580fn compact(environment: &RepositoryEnvironment, presentation: &Presentation) -> String {
581    let g = glyphs(presentation);
582    let budget = presentation.columns.clamp(20, MAX_WIDTH + 20);
583    let mut parts: Vec<String> = vec![sanitise(&environment.repository.name)];
584    if let Some(tree) = environment.vcs.tree() {
585        let mut state = tree
586            .branch
587            .clone()
588            .map(|b| sanitise(&b))
589            .unwrap_or_else(|| "detached".into());
590        if !tree.clean {
591            let _ = write!(state, " {}{}", g.dirty, tree.dirty_files() + tree.untracked);
592        }
593        if let (Some(a), Some(b)) = (tree.ahead, tree.behind) {
594            if a > 0 {
595                let _ = write!(state, " {}{a}", g.ahead);
596            }
597            if b > 0 {
598                let _ = write!(state, " {}{b}", g.behind);
599            }
600        }
601        parts.push(state);
602    }
603    if let Some(rust) = environment
604        .toolchains
605        .iter()
606        .find_map(|t| t.installed.as_ref().map(|v| format!("{} {}", t.title, v)))
607    {
608        parts.push(sanitise(&rust));
609    }
610    if let Some(objects) = environment.layer.objects {
611        parts.push(format!("{objects} objects"));
612    }
613    if let Some(cockpit) = environment.service("cockpit") {
614        if cockpit.availability == ServiceAvailability::Available {
615            parts.push(format!("cockpit {}", g.yes));
616        }
617    }
618    let separator = format!(" {} ", g.bullet);
619    let head = format!("{} {}", g.mark, parts.join(&separator));
620
621    let commands: Vec<String> = environment
622        .workflows
623        .entrypoints
624        .iter()
625        .take(3)
626        .map(|e| sanitise(&e.command))
627        .collect();
628    let mut out = truncate_with(&head, budget, g.ellipsis);
629    if !commands.is_empty() {
630        let _ = write!(
631            out,
632            "\n{}",
633            truncate_with(
634                &format!("  {}", commands.join(&separator)),
635                budget,
636                g.ellipsis
637            )
638        );
639    }
640    if let Some(worst) = environment.headline_diagnostic() {
641        let _ = write!(
642            out,
643            "\n{}",
644            truncate_with(
645                &format!("  {} {}", g.unknown, sanitise(&worst.message)),
646                budget,
647                g.ellipsis
648            )
649        );
650    }
651    out.push('\n');
652    out
653}

655fn glyphs(presentation: &Presentation) -> &'static Glyphs {
656    if presentation.unicode {
657        &UNICODE
658    } else {
659        &ASCII

773    fn plain(columns: usize) -> Presentation {
774        Presentation {

782    /// The box has to be a box: every line exactly as wide as every other, whatever is in
783    /// it. Measured in columns, which is the whole reason `text::width` exists.
784    #[test]
785    fn every_line_of_the_box_is_the_same_width() {
786        for columns in [MIN_BOX_WIDTH, 60, 78, 200] {
787            let drawn =
788                banner(&environment(), BannerMode::Full, &plain(columns), None).expect("a banner");
789            let widths: Vec<usize> = drawn.lines().map(width).collect();
790            let first = widths[0];
791            assert!(
792                widths.iter().all(|w| *w == first),
793                "at {columns} columns the lines are {widths:?}"
794            );
795            assert!(
796                first <= columns.max(MIN_BOX_WIDTH),
797                "wider than the terminal"
798            );
799        }
800    }

802    #[test]
803    fn a_narrow_terminal_gets_the_plain_form_rather_than_a_broken_box() {
804        let drawn = banner(&environment(), BannerMode::Full, &plain(30), None).expect("a banner");
805        assert!(!drawn.contains('╭'), "a box was drawn into 30 columns");
806        for line in drawn.lines() {
807            assert!(width(line) <= 30, "{line:?} overflows 30 columns");
808        }

857    /// The banner prints repository metadata, and repository metadata is not trusted.
858    #[test]
859    fn a_hostile_branch_name_cannot_reach_the_terminal_through_the_banner() {
860        let mut env = environment();
861        if let VcsState::Git(tree) = &mut env.vcs {
862            tree.branch = Some("\u{1b}]0;pwned\u{7}evil".into());
863        }
864        env.repository.name = "\u{1b}[2Jwiped".into();
865        let drawn = banner(&env, BannerMode::Full, &plain(78), None).expect("a banner");
866        assert!(!drawn.contains('\u{1b}'), "an escape reached the output");
867        assert!(drawn.contains("evil"), "the readable part is still shown");
868    }

887    #[test]
888    fn auto_shows_the_box_once_and_the_short_form_on_a_return() {
889        let env = environment();
890        let first = banner(&env, BannerMode::Auto, &plain(78), None).expect("a banner");
891        assert!(first.lines().count() > 5, "a first look gets the box");
892        let again =
893            banner(&env, BannerMode::Auto, &plain(78), Some(&env.news_digest())).expect("a banner");
894        assert!(again.lines().count() <= 3, "a return gets the short form");
895        let changed = banner(
896            &env,
897            BannerMode::Auto,
898            &plain(78),
899            Some("a different digest"),
900        )
901        .expect("a banner");
902        assert!(
903            changed.lines().count() > 5,
904            "a changed repository is news again"
905        );
906    }

908    #[test]
909    fn a_server_coming_up_after_the_box_is_not_a_second_box() {
910        // What direnv did on a cold entry: the box with no server, then the lease moved
911        // under `watch_file` and entry ran again with the address published.
912        let seen = environment();
913        let mut arrived = seen.clone();
914        for service in &mut arrived.services {
915            service.url = None;
916            service.availability = ServiceAvailability::NotRunning;
917        }
918        assert_ne!(seen.digest(), arrived.digest(), "the snapshots do differ");
919        let again = banner(
920            &arrived,
921            BannerMode::Auto,
922            &plain(78),
923            Some(&seen.news_digest()),
924        )
925        .expect("a banner");
926        assert!(
927            again.lines().count() <= 3,
928            "a server coming or going redrew the box:\n{again}"
929        );
930    }
