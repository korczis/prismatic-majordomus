67/// `session.freshness:` of the policy: how old a continuation record may be before it stops
68/// being current.
69///
70/// These are the only numbers that decide it, and they are declared once. Divergence
71/// (`exact`/`advanced`/`diverged`/`different_context`) answers a different question — where
72/// the record's commit sits relative to HEAD — and answers it identically on the day a
73/// record is written and a month later. Keeping a default here would be a second source of
74/// truth for the same thresholds, so absence is carried as absence and reported as
75/// `unknown`, naming the missing key.
76///
77/// Both keys are independently optional, and the pair is what
78/// [`Thresholds::judge`](crate::capability::builtin::continuity::Thresholds::judge) needs:
79/// with one of them missing it has no band to place a record in and answers `unknown`
80/// rather than inventing the other half.
81///
82/// ```
83/// use majordomus_cli::policy::FreshnessPolicy;
84/// use serde_json::json;
85///
86/// // the declared form: the two numbers that draw fresh | aging | stale
87/// let declared: FreshnessPolicy =
88///     serde_json::from_value(json!({"fresh_minutes": 30, "stale_minutes": 240})).unwrap();
89/// assert_eq!(declared.fresh_minutes, Some(30));
90/// assert_eq!(declared.stale_minutes, Some(240));
91///
92/// // a policy that predates the key: absent, not defaulted. Nothing in this file
93/// // supplies a number the repository never declared.
94/// let silent: FreshnessPolicy = serde_json::from_value(json!({})).unwrap();
95/// assert_eq!(silent, FreshnessPolicy::default());
96/// assert!(silent.fresh_minutes.is_none() && silent.stale_minutes.is_none());
97/// ```
98#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Default)]
99pub struct FreshnessPolicy {
100    /// Below this age a record is `fresh`.
101    #[serde(default)]
102    pub fresh_minutes: Option<i64>,
103    /// At or beyond this age a record is `stale`, and is never presented as current.
104    #[serde(default)]
105    pub stale_minutes: Option<i64>,
106}

108/// `knowledge:` of the policy: the bounds on the review queue of derived knowledge records.
109///
110/// A candidate is a record the deriver wrote at an episode boundary and nobody has judged
111/// (ADR 0091). The queue is measured, never emptied by a machine, and both numbers are
112/// declared once in the policy and read from it by the shell validator and by
113/// `knowledge_base.candidates` alike. Absence is carried as absence: a repository whose
114/// policy predates the keys is reported as declaring no cap, never judged against one
115/// nobody wrote down.
116///
117/// ```
118/// use majordomus_cli::policy::KnowledgePolicy;
119/// let declared: KnowledgePolicy =
120///     serde_json::from_str(r#"{"candidates_max_files": 40}"#).expect("a knowledge block");
121/// assert_eq!(declared.candidates_max_files, Some(40));
122/// assert_eq!(declared.candidate_max_age_minutes, None, "absent is absent, not a default");
123/// assert_eq!(KnowledgePolicy::default().candidates_max_files, None);
124/// ```
125#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Default)]
126pub struct KnowledgePolicy {
127    /// More records with status `candidate` than this awaiting review is a finding.
128    #[serde(default)]
129    pub candidates_max_files: Option<usize>,
130    /// A candidate waiting longer than this since it entered the queue is a finding.
131    #[serde(default)]
132    pub candidate_max_age_minutes: Option<i64>,
133}

135/// The policy, typed to what the projections consume. Every other key is carried through
136/// unread: the policy schema under `share/schemas/majordomus/policy/policy.v1.schema.json` owns the full shape.
137#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Default)]
138pub struct Policy {
139    /// `version:`.
140    #[serde(default)]
141    pub version: Option<u64>,
142    /// `context:`.
143    #[serde(default)]
144    pub context: ContextPolicy,
145    /// `profiles:`.
146    #[serde(default)]
147    pub profiles: ProfilesPolicy,
148    /// `session:`.
149    #[serde(default)]
150    pub session: SessionPolicy,
151    /// `knowledge:`.
152    #[serde(default)]
153    pub knowledge: KnowledgePolicy,
154    /// `projections:`.
155    #[serde(default)]
156    pub projections: Vec<Projection>,
157    /// `server:` — the timings a lease contest between two processes is judged by. Absent
158    /// keeps the constants `lease.rs` declares, so a policy that predates the block, or one
159    /// that could not be read, does not silently change how a server is taken over.
160    #[serde(default)]
161    pub server: ServerPolicy,
162    /// `commit:` — what this repository holds a commit message to. Absent is the default,
163    /// which is what this repository's own history already satisfies.
164    #[serde(default)]
165    pub commit: crate::commit::CommitPolicy,
166    /// `intent:` — whether work must be bound to what it serves before it starts, and the
167    /// exemption classes a worker may give instead. Absent is `off`: a repository whose
168    /// policy predates the block starts work exactly as it did (ADR 0111).
169    #[serde(default)]
170    pub intent: IntentPolicy,
171    /// `release:` — who raises the version and by at least how much (ADR 0134). Absent is
172    /// `allocation: off` with no floor: a repository whose policy predates the block raises
173    /// its version exactly as it did.
174    #[serde(default)]
175    pub release: crate::release::landing::ReleasePolicy,
176    /// `checkout:` — what a checkout is held to against the remote trunk. Absent declares
177    /// no threshold, and a distance is then reported as unjudged rather than judged by a
178    /// number nobody wrote down.
179    #[serde(default)]
180    pub checkout: CheckoutPolicy,
181}

194pub struct CheckoutPolicy {

201/// How strictly a start is held to its binding.
202///
203/// ```
204/// use majordomus_cli::policy::BindingMode;
205/// // a policy that says nothing asks for nothing
206/// assert_eq!(BindingMode::default(), BindingMode::Off);
207/// let m: BindingMode = serde_json::from_str("\"required\"").unwrap();
208/// assert_eq!(m, BindingMode::Required);
209/// assert_eq!(m.as_str(), "required");
210/// ```
211#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Deserialize, Serialize, JsonSchema)]
212#[serde(rename_all = "snake_case")]
213pub enum BindingMode {
214    /// Nothing is asked unless the worker names an issue, an intent or an exemption.
215    #[default]
216    Off,
217    /// Every start asks; a refused binding is reported and the work starts.
218    Advisory,
219    /// Every start asks; a refused or unreadable binding does not start.
220    Required,
221}
222
223impl BindingMode {
224    /// The word the policy file and every surface use.
225    ///
226    /// ```
227    /// use majordomus_cli::policy::BindingMode;
228    /// assert_eq!(BindingMode::Advisory.as_str(), "advisory");
229    /// ```
230    pub fn as_str(self) -> &'static str {
231        match self {
232            BindingMode::Off => "off",
233            BindingMode::Advisory => "advisory",
234            BindingMode::Required => "required",
235        }
236    }
237}

258/// `intent:` — the binding a start is held to (ADR 0111).
259///
260/// ```
261/// use majordomus_cli::policy::{BindingMode, IntentPolicy};
262/// // absent: off, and no exemption class exists to give
263/// let absent = IntentPolicy::default();
264/// assert_eq!(absent.binding, BindingMode::Off);
265/// assert!(absent.exemption("maintenance").is_none());
266/// let p: IntentPolicy = serde_json::from_str(
267///     r#"{"binding": "required", "exemptions": [{"id": "maintenance"}]}"#,
268/// )
269/// .unwrap();
270/// assert!(p.exemption("maintenance").is_some());
271/// assert!(p.exemption("because-i-say-so").is_none());
272/// ```
273#[derive(Debug, Clone, PartialEq, Eq, Default, Deserialize, Serialize, JsonSchema)]
274pub struct IntentPolicy {
275    /// `intent.binding:` — `off`, `advisory` or `required`.
276    #[serde(default)]
277    pub binding: BindingMode,
278    /// `intent.exemptions:` — the classes a worker may give instead of naming work.
279    #[serde(default)]
280    pub exemptions: Vec<ExemptionClass>,
281}

295/// `session:` — what the episode boundary does beyond drawing itself.
296///
297/// Only the half two entry paths both read is typed here. The rest of the block is carried
298/// through unread, as every other key is: the policy schema owns the full shape.
299///
300/// ```
301/// use majordomus_cli::policy::SessionPolicy;
302/// // A policy that says nothing about it still converges. The default is what this
303/// // repository's policy and the skeleton a new one is written from both declare, and a
304/// // policy that could not be read must not silently change behaviour.
305/// assert!(SessionPolicy::default().ensure_server_on_start);
306/// assert!(SessionPolicy::default().knowledge_on_end);
307/// assert!(SessionPolicy::default().knowledge_on_compact);
308/// // and a repository that has turned one off is read as having turned it off
309/// let off: SessionPolicy =
310///     serde_json::from_str(r#"{"ensure_server_on_start": false, "knowledge_on_end": false}"#)
311///         .expect("a session block");
312/// assert!(!off.ensure_server_on_start);
313/// assert!(!off.knowledge_on_end);
314/// assert!(off.knowledge_on_compact, "the two switches are independent");
315/// ```
316#[derive(Debug, Clone, PartialEq, Eq, Deserialize)]
317pub struct SessionPolicy {
318    /// `session.freshness:` — how old a continuation record may be before it stops being
319    /// current. A separate question from divergence, which is about git topology and says
320    /// nothing about age.
321    #[serde(default)]
322    pub freshness: FreshnessPolicy,
323    /// Whether entering the repository converges on a ready shared server.
324    ///
325    /// One switch, both entry paths: the provider's start event (ADR 0035) and the file a
326    /// shell evaluates on entry (ADR 0043). It defaults to `true` because that is what
327    /// this repository's policy and the skeleton a new one is written from both declare,
328    /// and because a policy that could not be read must not silently change behaviour —
329    /// an unreadable policy is `doctor`'s finding, not a reason to stop converging.
330    #[serde(default = "yes")]
331    pub ensure_server_on_start: bool,
332    /// Whether closing an episode derives the knowledge it produced into candidate records
333    /// under the tracked knowledge section (ADR 0091). Read here so that
334    /// `knowledge_base.status` judges a silent writer only where the writer is switched on,
335    /// and defaults to `true` for the reason `ensure_server_on_start` does: the policy and
336    /// the skeleton both declare it, and an unreadable policy must not change behaviour.
337    #[serde(default = "yes")]
338    pub knowledge_on_end: bool,
339    /// Whether a provider's compaction event derives the same records. Independent of
340    /// `checkpoint_on_compact`: the two record different things.
341    #[serde(default = "yes")]
342    pub knowledge_on_compact: bool,
343    /// `session.brief_interval_minutes:` — how often a session reports on itself without
344    /// being asked (ADR 0136). Absent is the default `brief.session` carries; the provider
345    /// mod reads the interval from that answer rather than from this file.
346    #[serde(default)]
347    pub brief_interval_minutes: Option<u64>,
348    /// `session.edits_require_claim:` — whether a provider's tool may write a file of the
349    /// checkout only while a claim covers it: the active task's scope, or a live mesh claim
350    /// of the writing session (I2735, [`crate::guard`]). Off unless a repository turns it
351    /// on, because it refuses edits, and a repository that never asked for that must not
352    /// find its workers stopped by an upgrade.
353    #[serde(default)]
354    pub edits_require_claim: bool,
355    /// `session.stranded_after:` — how long an open episode may be silent before its client
356    /// is taken to be gone (`90s`, `30m`, `12h`). `majordomus recover` reads it to judge an
357    /// episode stranded, and `worktree.uncommitted` reads the same value twice over: an
358    /// episode silent this long holds no work, and uncommitted work untouched this long
359    /// that nothing holds is abandoned. Absent is absent: no reader carries a default of
360    /// its own, because a second copy of the number is the drift the policy exists to stop.
361    ///
362    /// ```
363    /// use majordomus_cli::policy::SessionPolicy;
364    /// let declared: SessionPolicy =
365    ///     serde_json::from_str(r#"{"stranded_after": "12h"}"#).expect("a session block");
366    /// assert_eq!(declared.stranded_after.as_deref(), Some("12h"));
367    /// assert_eq!(SessionPolicy::default().stranded_after, None, "absent is not a default");
368    /// // a bare number is seconds, as the shell reader takes it, and never a parse failure
369    /// // that would take the rest of the policy down with it
370    /// let bare: SessionPolicy =
371    ///     serde_json::from_str(r#"{"stranded_after": 3600}"#).expect("a session block");
372    /// assert_eq!(bare.stranded_after.as_deref(), Some("3600"));
373    /// ```
374    #[serde(default, deserialize_with = "scalar_text")]
375    pub stranded_after: Option<String>,
376}

378/// A scalar read as the text it was written as: the subset reader hands `12h` over as a
379/// string and `3600` as a number, and both are the same kind of value to a duration.
380fn scalar_text<'de, D: serde::Deserializer<'de>>(
381    d: D,
382) -> std::result::Result<Option<String>, D::Error> {
383    Ok(match serde_json::Value::deserialize(d)? {
384        serde_json::Value::String(s) => Some(s),
385        serde_json::Value::Number(n) => Some(n.to_string()),
386        _ => None,
387    })
388}

408/// The policy as loaded: the typed value, where it came from, and the hash the stamps
409/// carry.
410#[derive(Debug, Clone)]
411pub struct LoadedPolicy {
412    /// The typed policy.
413    pub policy: Policy,
414    /// Repository-relative path of the policy file.
415    pub path: String,
416    /// SHA-256 (hex) of the [`normal_form`] of the policy file followed by that of every
417    /// profile file, in name order, concatenated: exactly what the shell tool's
418    /// `mj_policy_cat` hashes. A comment or a blank line does not move it (I1991).
419    pub sha256: String,
420}
421
422impl LoadedPolicy {
423    /// Read the policy and profiles of a repository.
424    pub fn load(repository: &Repository) -> Result<Self> {
425        let rel =
426            repository
427                .section_path(POLICY_SECTION)
428                .ok_or_else(|| Error::InvalidManifest {
429                    path: repository.root().join(crate::repository::MANIFEST),
430                    reason: format!("the manifest names no `{POLICY_SECTION}` section"),
431                })?;
432        let path = repository.root().join(&rel);
433        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
434        let policy: Policy = yaml::parse_into(&text).map_err(|reason| Error::InvalidPolicy {
435            path: path.clone(),
436            reason,
437        })?;
438        // The one point a repository's policy is read, and therefore the one place the lease
439        // timings can be declared for this process. `declare_timings` is a OnceLock set, so
440        // reading the policy twice does not move a judgement already in flight.
441        crate::lease::declare_timings(&policy.server);
442        let mut hasher = Sha256::new();
443        hasher.update(normal_form(&text).as_bytes());
444        for profile in profile_files(repository)? {
445            let bytes = std::fs::read(&profile).map_err(|e| Error::io(&profile, e))?;
446            hasher.update(normal_form(&String::from_utf8_lossy(&bytes)).as_bytes());
447        }
448        Ok(LoadedPolicy {
449            policy,
450            path: rel,
451            sha256: format!("{:x}", hasher.finalize()),
452        })
453    }
454}

456/// The normal form of a file in the layer's YAML subset, which the policy hash covers in
457/// place of its bytes: trailing spaces, tabs and carriage returns dropped; blank lines and
458/// whole-line comments dropped; a trailing comment (a `#` after a space or tab) dropped
459/// from a line that holds no quote, since inside a quoted scalar ` #` may be text.
460/// Indentation, order and every value are kept, and every line ends in a newline.
461///
462/// It is the shell tool's `mj_yaml_normal` (lib/common.sh), line for line: the two must
463/// agree, or each renderer would call the other's bootstraps stale. The subset refuses
464/// block scalars, so no `#` line is ever content.
465///
466/// ```
467/// use majordomus_cli::policy::normal_form;
468///
469/// let edited = "# a new comment\nversion: 1   # the schema\n\ncontext:\n  name: \"a #b\"\n";
470/// assert_eq!(normal_form(edited), "version: 1\ncontext:\n  name: \"a #b\"\n");
471/// assert_eq!(normal_form(edited), normal_form("version: 1\ncontext:\n  name: \"a #b\"\n"));
472/// assert_ne!(normal_form("a: 1\n"), normal_form("a: 2\n"));
473/// ```
474pub fn normal_form(text: &str) -> String {
475    let mut out = String::with_capacity(text.len());
476    for raw in text.split('\n') {
477        let mut line = raw.trim_end_matches([' ', '\t', '\r']);
478        let body = line.trim_start_matches([' ', '\t']);
479        if body.is_empty() || body.starts_with('#') {
480            continue;
481        }
482        if !line.contains(['\'', '"']) {
483            let b = line.as_bytes();
484            if let Some(i) =
485                (1..b.len()).find(|&i| b[i] == b'#' && (b[i - 1] == b' ' || b[i - 1] == b'\t'))
486            {
487                line = line[..i].trim_end_matches([' ', '\t']);
488            }
489        }
490        out.push_str(line);
491        out.push('\n');
492    }
493    out
494}

496/// The profile files, `<profiles>/*.yaml`, sorted by name. An absent directory is no
497/// profile, not an error.
498pub fn profile_files(repository: &Repository) -> Result<Vec<PathBuf>> {
499    let Some(rel) = repository.section_path(PROFILES_SECTION) else {
500        return Ok(Vec::new());
501    };
502    let dir = repository.root().join(rel);
503    let entries = match std::fs::read_dir(&dir) {
504        Ok(e) => e,
505        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(Vec::new()),
506        Err(e) => return Err(Error::io(&dir, e)),
507    };
508    let mut files: Vec<PathBuf> = entries
509        .filter_map(|e| e.ok().map(|e| e.path()))
510        .filter(|p| p.extension().is_some_and(|x| x == "yaml") && p.is_file())
511        .collect();
512    files.sort();
513    Ok(files)
514}

524/// Hex SHA-256 of a text.
525pub fn sha256_hex(text: &str) -> String {
526    let mut h = Sha256::new();
527    h.update(text.as_bytes());
528    format!("{:x}", h.finalize())
529}

531/// Hex SHA-256 of arbitrary bytes: the same digest [`sha256_hex`] takes, for content that
532/// is not necessarily text. A file the repository tracks may hold anything, and decoding it
533/// to hash it would be a decision about its encoding that nothing here is entitled to make.
534///
535/// ```
536/// use majordomus_cli::policy::{sha256_bytes_hex, sha256_hex};
537/// assert_eq!(sha256_bytes_hex(b"majordomus"), sha256_hex("majordomus"));
538/// assert_ne!(sha256_bytes_hex(&[0xff, 0xfe]), sha256_bytes_hex(&[0xfe, 0xff]));
539/// ```
540pub fn sha256_bytes_hex(bytes: &[u8]) -> String {
541    let mut h = Sha256::new();
542    h.update(bytes);
543    format!("{:x}", h.finalize())
544}

546/// A path is inside `root` after lexical normalisation: no absolute path, no `..`
547/// component, nothing empty.
548pub fn is_safe_relative(path: &str) -> bool {
549    let p = Path::new(path);
550    !path.is_empty()
551        && !p.is_absolute()
552        && p.components().all(|c| {
553            matches!(
554                c,
555                std::path::Component::Normal(_) | std::path::Component::CurDir
556            )
557        })

650/// `server:` — what a lease contest is judged by.
651///
652/// Every field is optional and absent means "keep the compiled default", which is why there
653/// is no reader-side default *value* here: the defaults live in one place, `lease::Timings`,
654/// and this block only says which of them this repository overrides.
655///
656/// ```
657/// use majordomus_cli::policy::ServerPolicy;
658/// // a policy that predates the block: absent, not defaulted to zero
659/// assert_eq!(ServerPolicy::default().probe_timeout_seconds, None);
660/// let p: ServerPolicy =
661///     serde_json::from_str(r#"{"probe_timeout_seconds": 5}"#).expect("a server block");
662/// assert_eq!(p.probe_timeout_seconds, Some(5));
663/// ```
664#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize, Default)]
665pub struct ServerPolicy {
666    /// `server.probe_timeout_seconds:` — how long a probe waits for the current owner before
667    /// that silence counts as evidence that it is gone.
668    #[serde(default)]
669    pub probe_timeout_seconds: Option<u64>,
670    /// `server.bind_grace_seconds:` — how long a lease naming no URL yet is left alone,
671    /// because a server writes its lease before it can serve.
672    #[serde(default)]
673    pub bind_grace_seconds: Option<u64>,
674    /// `server.join_timeout_seconds:` — how long a process waits to join or create the lease
675    /// file before refusing rather than waiting forever.
676    #[serde(default)]
677    pub join_timeout_seconds: Option<u64>,
678    /// `server.busy_grace_seconds:` — how long a live owner that does not answer is waited
679    /// on, and asked again, before its lease is taken over.
680    #[serde(default)]
681    pub busy_grace_seconds: Option<u64>,
682    /// `server.read_deadline_seconds:` — how long the shared server lets a connection go
683    /// without a request read in full on it before it closes the connection.
684    #[serde(default)]
685    pub read_deadline_seconds: Option<u64>,
686    /// `server.idle_seconds:` — how long a server started for a checkout no client owns
687    /// (`serve ensure`, a shell entering the checkout) outlives its last peer before it stops
688    /// by itself. `--idle` on the command line overrides it.
689    #[serde(default)]
690    pub idle_seconds: Option<u64>,
691}
