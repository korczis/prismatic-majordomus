1//! The derive offload: a branch's `scripts/derive` run on another machine the repository
2//! declares, in that machine's turn, and the derived tree brought back staged.
3//!
4//! A derive costs from ten minutes on an idle machine to an hour on the shared one, and
5//! every session needs one before it can commit. Until this module the offload was a script
6//! in a home directory with its hosts typed into it and a lock that was a race. Now:
7//!
8//! - [`declaration`]: the hosts and their limits, read from the layer (kind
9//!   `derive-offload`, `.ai/repo/offload/*.yaml`) and nowhere else;
10//! - [`host`]: which host a derive goes to — free space, then the fewest derives running or
11//!   waiting, then declaration order;
12//! - [`queue`]: the FIFO queue on the host, decided here over what the host reports;
13//! - [`remote`]: how a script reaches a host, over ssh or directly;
14//! - [`run`]: the round trip — bundle in, derive, bundle back, stage;
15//! - [`owner`]: who a run belongs to, exported into it and recorded on its host (I3956);
16//! - [`stop`]: a run stopped as a process group, on its host, by its owner (I3956).
17//!
18//! `docs/OFFLOAD.md` is the reference; `majordomus offload` is the command.
19
20pub mod declaration;
21pub mod host;
22pub mod owner;
23pub mod queue;
24pub mod remote;
25pub mod run;
26pub mod stop;

30/// The repository's declaration, read from its index: `Ok(None)` when it declares none,
31/// an error naming the object when one is declared and invalid, or when two are.
32pub fn from_objects(
33    objects: &[crate::model::Object],
34) -> Result<Option<OffloadDeclaration>, String> {
35    let mut found: Vec<&crate::model::Object> = objects.iter().filter(|o| o.kind == KIND).collect();
36    match found.len() {
37        0 => Ok(None),
38        1 => OffloadDeclaration::parse(found.remove(0)).map(Some),
39        _ => Err(format!(
40            "{} derive-offload declarations ({}); a repository declares one",
41            found.len(),
42            found
43                .iter()
44                .map(|o| o.uri.as_str())
45                .collect::<Vec<_>>()
46                .join(", ")
47        )),
48    }
49}

51#[cfg(test)]
52mod tests {
53    use super::*;
54    use crate::model::{Object, Provenance};
55
56    fn object(kind: &str, identity: &str, metadata: serde_json::Value) -> Object {
57        Object {
58            kind: kind.into(),
59            identity: identity.into(),
60            uri: format!("majordomus://{kind}/{identity}"),
61            title: None,
62            description: None,
63            metadata,
64            body: String::new(),
65            content: String::new(),
66            media_type: "application/yaml",
67            provenance: Provenance {
68                path: format!(".ai/repo/offload/{identity}.yaml"),
69                directory: ".ai/repo/offload".into(),
70                source_class: "offload".into(),
71                section: None,
72                bytes: 0,
73                member: None,
74            },
75        }
76    }
77
78    fn declared(id: &str) -> Object {
79        object(
80            KIND,
81            id,
82            serde_json::json!({
83                "schema": declaration::SCHEMA_VERSION, "kind": KIND, "id": id,
84                "hosts": [{ "id": "lundra", "address": "u@lundra" }]
85            }),
86        )
87    }
88
89    #[test]
90    fn the_one_declaration_of_the_index_is_found_among_other_objects() {
91        let rule = object("rule", "project.alpha", serde_json::json!({}));
92        assert_eq!(from_objects(std::slice::from_ref(&rule)).unwrap(), None);
93
94        let d = from_objects(&[rule.clone(), declared("derive")])
95            .unwrap()
96            .expect("one declaration is declared");
97        assert_eq!(
98            (d.id.as_str(), d.hosts[0].id.as_str()),
99            ("derive", "lundra")
100        );
101
102        let invalid = object(KIND, "broken", serde_json::json!({ "kind": KIND }));
103        assert!(from_objects(&[invalid])
104            .unwrap_err()
105            .starts_with("majordomus://derive-offload/broken: "));
106    }
107
108    #[test]
109    fn two_declarations_are_refused_naming_both() {
110        assert_eq!(
111            from_objects(&[declared("one"), declared("two")]).unwrap_err(),
112            "2 derive-offload declarations (majordomus://derive-offload/one, \
113             majordomus://derive-offload/two); a repository declares one"
114        );
115    }
