1//! `majordomus commands`: the command graph, read from the command line.
2//!
3//! Every subcommand here is a rendering of [`crate::command_graph`] and decides nothing of
4//! its own: what a command is, where it is projected and why it is withheld are answered by
5//! the graph and the policy beside it, so the terminal, the Cockpit and a machine client
6//! give the same answers to the same questions.
7

19/// The exit code when a graph carries an error, or the materialised bridge is stale.
20pub const EXIT_DRIFT: u8 = 10;
21
22/// The exit code when a command is asked for by an identity nothing carries.
23pub const EXIT_MISSING: u8 = 12;

25/// Run `majordomus commands`.
26pub fn run(args: CommandsArgs) -> Result<u8> {
27    let format = args.format;
28    match args.command {
29        None => list(&args, &CommandsListArgs::none(), format),
30        Some(CommandsCommand::List(ref list_args)) => list(&args, list_args, format),
31        Some(CommandsCommand::Show(ref show)) => show_one(&args, show, format, false),
32        Some(CommandsCommand::Explain(ref show)) => show_one(&args, show, format, true),
33        Some(CommandsCommand::Graph(ref graph_args)) => whole(&args, graph_args, format),
34        Some(CommandsCommand::Bridge(ref bridge_args)) => project(&args, bridge_args, format),
35    }
36}

38impl CommandsListArgs {
39    /// The filters of a bare `majordomus commands`: none.
40    fn none() -> Self {
41        CommandsListArgs {
42            origin: None,
43            effect: None,
44            search: None,
45        }
46    }
47}

49/// `commands list`.
50fn list(args: &CommandsArgs, filters: &CommandsListArgs, format: OutputFormat) -> Result<u8> {
51    let loaded = load::full(&args.repo)?;
52    let matched: Vec<&CommandNode> = loaded
53        .graph
54        .commands
55        .iter()
56        .filter(|c| filters.origin.is_none_or(|o| c.origin == origin_of(o)))
57        .filter(|c| filters.effect.is_none_or(|e| c.effect <= effect_of(e)))
58        .filter(|c| {
59            filters
60                .search
61                .as_ref()
62                .is_none_or(|t| c.haystack().contains(&t.to_lowercase()))
63        })
64        .collect();
65
66    let stdout = std::io::stdout();
67    let mut out = stdout.lock();
68    match format {
69        OutputFormat::Json => {
70            let doc = json!({
71                "schema": crate::command_graph::SCHEMA,
72                "fingerprint": loaded.graph.fingerprint,
73                "commands": matched,
74            });
75            writeln!(out, "{}", pretty(&doc)).map_err(Error::Transport)?;
76        }
77        OutputFormat::Text => {
78            writeln!(
79                out,
80                "commands     {} of {} (graph {})",
81                matched.len(),
82                loaded.graph.commands.len(),
83                loaded.graph.fingerprint
84            )
85            .map_err(Error::Transport)?;
86            for node in &matched {
87                writeln!(
88                    out,
89                    "{:<10} {:<20} {}",
90                    origin_word(node.origin),
91                    effect_word(node.effect),
92                    node.invocation
93                )
94                .map_err(Error::Transport)?;
95            }
96            report(&mut out, &loaded.graph)?;
97        }
98    }
99    Ok(0)
100}

102/// `commands show` and `commands explain`.
103fn show_one(
104    args: &CommandsArgs,
105    show: &CommandsShowArgs,
106    format: OutputFormat,
107    explain: bool,
108) -> Result<u8> {
109    let loaded = load::full(&args.repo)?;
110    let Some(node) = loaded
111        .graph
112        .commands
113        .iter()
114        .find(|c| c.id.as_str() == show.id)
115    else {
116        return Err(Error::Refused {
117            code: EXIT_MISSING,
118            reason: format!("no command carries the identity `{}`", show.id),
119        });
120    };
121
122    let stdout = std::io::stdout();
123    let mut out = stdout.lock();
124    match format {
125        OutputFormat::Json => {
126            writeln!(
127                out,
128                "{}",
129                pretty(&serde_json::to_value(node).unwrap_or(Value::Null))
130            )
131            .map_err(Error::Transport)?;
132        }
133        OutputFormat::Text => {
134            writeln!(out, "command      {}", node.id).map_err(Error::Transport)?;
135            writeln!(out, "run          {}", node.invocation).map_err(Error::Transport)?;
136            writeln!(out, "summary      {}", node.summary).map_err(Error::Transport)?;
137            writeln!(out, "effect       {}", effect_word(node.effect)).map_err(Error::Transport)?;
138            writeln!(out, "runs in      {:?}", node.interactivity).map_err(Error::Transport)?;
139            if !node.arguments.is_empty() {
140                writeln!(out, "arguments").map_err(Error::Transport)?;
141                for arg in &node.arguments {
142                    let spelling = match &arg.long {
143                        Some(long) => format!("--{long}"),
144                        None => format!("<{}>", arg.name.to_uppercase()),
145                    };
146                    writeln!(
147                        out,
148                        "  {:<22} {:<18} {}",
149                        spelling,
150                        source_word(arg.source),
151                        arg.help
152                    )
153                    .map_err(Error::Transport)?;
154                }
155            }
156            writeln!(out, "projections").map_err(Error::Transport)?;
157            let p = &node.projections;
158            for (surface, value) in [
159                ("command line", p.cli.clone()),
160                ("workflow", p.workflow.clone().map(|w| format!("just {w}"))),
161                ("mcp", p.mcp.clone()),
162                ("http", p.http.clone()),
163                ("cockpit", p.cockpit.clone()),
164                ("docs", Some(p.docs.clone())),
165            ] {
166                match value {
167                    Some(v) => writeln!(out, "  {surface:<14} {v}").map_err(Error::Transport)?,
168                    None => writeln!(out, "  {surface:<14} —").map_err(Error::Transport)?,
169                }
170            }
171            if let Some(why) = &p.withheld {
172                writeln!(out, "  withheld       {why}").map_err(Error::Transport)?;
173            }
174            if explain {
175                writeln!(out, "declared in  {}", node.provenance.declared_in)
176                    .map_err(Error::Transport)?;
177                if let Some(read_by) = &node.provenance.read_by {
178                    writeln!(out, "read by      {read_by}").map_err(Error::Transport)?;
179                }
180                if let Some(capability) = &node.provenance.capability {
181                    writeln!(out, "capability   {capability}").map_err(Error::Transport)?;
182                }
183                writeln!(
184                    out,
185                    "requires     {}",
186                    node.availability
187                        .requires
188                        .iter()
189                        .map(|r| format!("{r:?}"))
190                        .collect::<Vec<_>>()
191                        .join(", ")
192                )
193                .map_err(Error::Transport)?;
194                writeln!(
195                    out,
196                    "policy       apps/majordomus-cli/src/command_graph/policy.rs"
197                )
198                .map_err(Error::Transport)?;
199            }
200        }
201    }
202    Ok(0)
203}

205/// `commands graph`.
206fn whole(args: &CommandsArgs, graph_args: &CommandsGraphArgs, format: OutputFormat) -> Result<u8> {
207    let loaded = load::full(&args.repo)?;
208    let stdout = std::io::stdout();
209    let mut out = stdout.lock();
210    match format {
211        OutputFormat::Json => writeln!(
212            out,
213            "{}",
214            pretty(&serde_json::to_value(&loaded.graph).unwrap_or(Value::Null))
215        )
216        .map_err(Error::Transport)?,
217        OutputFormat::Text => {
218            writeln!(out, "schema       {}", loaded.graph.schema).map_err(Error::Transport)?;
219            writeln!(out, "fingerprint  {}", loaded.graph.fingerprint).map_err(Error::Transport)?;
220            writeln!(out, "commands     {}", loaded.graph.commands.len())
221                .map_err(Error::Transport)?;
222            for origin in [Origin::Executable, Origin::Tool, Origin::Workflow] {
223                writeln!(
224                    out,
225                    "  {:<10} {}",
226                    origin_word(origin),
227                    loaded.graph.of_origin(origin).count()
228                )
229                .map_err(Error::Transport)?;
230            }
231            report(&mut out, &loaded.graph)?;
232        }
233    }
234    if graph_args.check && !loaded.graph.errors().is_empty() {
235        return Ok(EXIT_DRIFT);
236    }
237    Ok(0)
238}

240/// `commands bridge`.
241fn project(
242    args: &CommandsArgs,
243    bridge_args: &CommandsBridgeArgs,
244    format: OutputFormat,
245) -> Result<u8> {
246    // The cheap question first, and it is the one asked on every entry into the
247    // repository: have any of the declarations the bridge is derived from changed since it
248    // was written? A few `stat` calls answer it, and when the answer is no there is
249    // nothing to build, nothing to ask the workflow runner, and nothing to write.
250    if !bridge_args.check {
251        if let Ok(root) = load::root(&args.repo) {
252            if bridge::is_current(&root) {
253                if matches!(format, OutputFormat::Text) {
254                    let stdout = std::io::stdout();
255                    let mut out = stdout.lock();
256                    writeln!(out, "bridge       already current").map_err(Error::Transport)?;
257                }
258                return Ok(0);
259            }
260        }
261    }
262    let loaded = load::full(&args.repo)?;
263    if !loaded.graph.errors().is_empty() {
264        // A projection of a graph with an error would write the error into a file every
265        // shell then loads. The refusal is the point.
266        let stderr = std::io::stderr();
267        let mut err = stderr.lock();
268        for d in loaded.graph.errors() {
269            writeln!(err, "commands: {}", d.message).map_err(Error::Transport)?;
270        }
271        return Ok(EXIT_DRIFT);
272    }
273
274    let stdout = std::io::stdout();
275    let mut out = stdout.lock();
276    if bridge_args.check {
277        let current = bridge::manifest(&loaded.root);
278        let fresh = current
279            .as_ref()
280            .is_some_and(|m| m.fingerprint == loaded.graph.fingerprint);
281        match format {
282            OutputFormat::Json => writeln!(
283                out,
284                "{}",
285                pretty(&json!({
286                    "fresh": fresh,
287                    "fingerprint": loaded.graph.fingerprint,
288                    "materialised": current.map(|m| m.fingerprint),
289                }))
290            )
291            .map_err(Error::Transport)?,
292            OutputFormat::Text => writeln!(
293                out,
294                "bridge       {}",
295                if fresh {
296                    "current".to_string()
297                } else {
298                    format!(
299                        "stale; run `majordomus commands bridge` (graph {})",
300                        loaded.graph.fingerprint
301                    )
302                }
303            )
304            .map_err(Error::Transport)?,
305        }
306        return Ok(if fresh { 0 } else { EXIT_DRIFT });
307    }
308
309    let outcome = bridge::materialise(&loaded.root, &loaded.graph)
310        .map_err(|e| Error::io(loaded.root.join(bridge::DIRECTORY), e))?;
311    let _ = load::write_cache(&loaded.root, &loaded.graph);
312    match format {
313        OutputFormat::Json => writeln!(
314            out,
315            "{}",
316            pretty(&json!({
317                "path": outcome.path.to_string_lossy(),
318                "written": outcome.written,
319                "recipes": outcome.recipes,
320                "fingerprint": outcome.fingerprint,
321            }))
322        )
323        .map_err(Error::Transport)?,
324        OutputFormat::Text => writeln!(
325            out,
326            "bridge       {} recipe(s) {} ({})",
327            outcome.recipes,
328            if outcome.written {
329                "written"
330            } else {
331                "already current"
332            },
333            outcome.path.display()
334        )
335        .map_err(Error::Transport)?,
336    }
337    Ok(0)
338}

340/// The diagnostics, when there are any worth printing.
341fn report(out: &mut std::io::StdoutLock<'_>, graph: &CommandGraph) -> Result<()> {
342    for d in graph
343        .diagnostics
344        .iter()
345        .filter(|d| d.severity != Severity::Info)
346    {
347        let word = match d.severity {
348            Severity::Error => "ERROR",
349            Severity::Warning => "WARN ",
350            Severity::Info => "INFO ",
351        };
352        writeln!(out, "{word}        {} [{}]", d.message, d.code).map_err(Error::Transport)?;
353    }
354    Ok(())
355}

357/// The word a person reads for an origin.
358fn origin_word(origin: Origin) -> &'static str {
359    match origin {
360        Origin::Executable => "executable",
361        Origin::Tool => "tool",
362        Origin::Workflow => "workflow",
363    }
364}
365
366/// The word a person reads for an effect.
367fn effect_word(effect: Effect) -> &'static str {
368    match effect {
369        Effect::ReadOnly => "read-only",
370        Effect::LocalMutation => "local-mutation",
371        Effect::RepositoryMutation => "repository-mutation",
372        Effect::NetworkMutation => "network-mutation",
373        Effect::Destructive => "destructive",
374    }
375}
376
377/// The word a person reads for a value source.
378fn source_word(source: crate::command_graph::ValueSource) -> String {
379    format!("{source:?}").to_lowercase()
380}
381
382/// The origin a filter names.
383fn origin_of(o: CommandOrigin) -> Origin {
384    match o {
385        CommandOrigin::Executable => Origin::Executable,
386        CommandOrigin::Tool => Origin::Tool,
387        CommandOrigin::Workflow => Origin::Workflow,
388    }
389}
390
391/// The effect a filter names.
392fn effect_of(e: CommandEffect) -> Effect {
393    match e {
394        CommandEffect::ReadOnly => Effect::ReadOnly,
395        CommandEffect::LocalMutation => Effect::LocalMutation,
396        CommandEffect::RepositoryMutation => Effect::RepositoryMutation,
397        CommandEffect::NetworkMutation => Effect::NetworkMutation,
398        CommandEffect::Destructive => Effect::Destructive,
399    }
400}

402/// One JSON document, deterministic.
403fn pretty(v: &Value) -> String {
404    serde_json::to_string_pretty(v).unwrap_or_else(|_| "{}".into())
