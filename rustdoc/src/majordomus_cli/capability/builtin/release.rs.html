1//! The `release` module: the changelog, the version, and the compatibility analysis behind
2//! it — all three derived.
3//!
4//! Three capabilities, all read-only, all answered by [`crate::release`] — the same code the
5//! command line renders, the generated document is written from, and the site page shows.
6//! Neither reads a file somebody maintains: the changelog composes the layer's own release
7//! records, decisions and the repository's commits, and the version report reads the one
8//! place the version is authored and says whether the projection the shell tool prints is
9//! current.
10//!
11//! Raising the version is not here. It writes tracked files, which makes it a repository
12//! mutation, and the exposure policy keeps repository mutations off every machine surface —
13//! so `release bump` is a command and nothing more, and this module is the read half.
14

34/// Which part of the changelog to answer with.
35#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
36#[serde(deny_unknown_fields)]
37#[schemars(rename = "ReleaseChangelogInput")]
38pub struct ChangelogInput {
39    #[serde(default, skip_serializing_if = "Option::is_none")]
40    /// One version, or `unreleased`; every section when absent.
41    pub version: Option<String>,
42}
43
44impl BenchmarkCases for ChangelogInput {
45    /// One real version from the layer's own records, so the narrow case is timed against
46    /// something that exists rather than against a filter that matches nothing. A
47    /// repository with no records has a narrow case all the same — the coverage gate counts
48    /// targets, not repositories, and the OpenAPI operation's example of `version` *is* this
49    /// case, which case 92 refuses to see missing — so it falls back to `unreleased`, the
50    /// one section a repository has whether it has released anything or not.
51    fn benchmark_cases(ctx: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
52        let version = ctx
53            .index
54            .objects
55            .iter()
56            .find(|o| o.kind == crate::release::changelog::RELEASE_KIND)
57            .and_then(|o| o.metadata.get("version"))
58            .and_then(|v| v.as_str())
59            .map_or_else(|| "unreleased".to_string(), str::to_string);
60        vec![
61            NamedCase::new("all", ChangelogInput::default()),
62            NamedCase::new(
63                "one-version",
64                ChangelogInput {
65                    version: Some(version),
66                },
67            ),
68        ]
69    }
70}

72/// Which release to measure this tree against.
73///
74/// ```
75/// use majordomus_cli::capability::builtin::release::AnalysisInput;
76/// // Absent means the last release the layer records, which is the canonical baseline.
77/// let input = AnalysisInput::default();
78/// assert!(input.since.is_none());
79/// let explicit = AnalysisInput { since: Some("v0.4.0".into()) };
80/// assert_eq!(explicit.since.as_deref(), Some("v0.4.0"));
81/// ```
82#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
83#[serde(deny_unknown_fields)]
84#[schemars(rename = "ReleaseAnalysisInput")]
85pub struct AnalysisInput {
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    /// A ref to compare with instead of the last release: `v0.4.0`, or any commit that
88    /// carries a committed registry.
89    pub since: Option<String>,
90}
91
92impl BenchmarkCases for AnalysisInput {
93    /// None. The capability is waived from the benchmark
94    /// ([`WaiverReason::PublishedHistory`]) because its input is a release this repository
95    /// has published, and a benchmark fixture has published nothing — so every case here
96    /// would time a refusal and be counted as coverage. The trait is still implemented
97    /// because the registry asks every input type for its cases; answering "none" is the
98    /// honest answer rather than a case that measures an error path.
99    fn benchmark_cases(_: &CaseContext<'_>) -> Vec<NamedCase<Self>> {
100        Vec::new()
101    }
102}

104/// The input of `release.next`: the landing it is asked about.
105///
106/// ```
107/// use majordomus_cli::capability::builtin::release::NextInput;
108/// // nothing named is the landing the executor performs: this branch onto the trunk
109/// let input = NextInput::default();
110/// assert_eq!((input.base.as_deref(), input.head()), (None, "HEAD"));
111/// ```
112#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
113#[serde(deny_unknown_fields)]
114#[schemars(rename = "ReleaseNextInput")]
115pub struct NextInput {
116    #[serde(default, skip_serializing_if = "Option::is_none")]
117    /// The trunk the landing raises from; when absent, the remote-tracking ref of the trunk
118    /// this repository names (its remote's default branch, else its own trunk).
119    pub base: Option<String>,
120    #[serde(default, skip_serializing_if = "Option::is_none")]
121    /// What lands; `HEAD` when absent.
122    pub head: Option<String>,
123}
124
125impl NextInput {
126    /// The trunk the landing raises from: the ref named, or the remote-tracking ref of the
127    /// trunk the repository at `root` names — the one rule the delivery report already
128    /// measures features against, never a branch name written here. A repository whose
129    /// trunk cannot be told is refused by name rather than measured against a guess.
130    ///
131    /// ```
132    /// use majordomus_cli::capability::builtin::release::NextInput;
133    /// let here = std::path::Path::new(".");
134    /// assert_eq!(NextInput { base: Some("main".into()), head: None }.base(here), Ok("main".into()));
135    /// let nowhere = std::env::temp_dir().join("majordomus-doc-no-repository");
136    /// assert!(NextInput::default().base(&nowhere).is_err());
137    /// ```
138    pub fn base(&self, root: &std::path::Path) -> Result<String, String> {
139        match self.base.as_deref() {
140            Some(named) => Ok(named.to_string()),
141            None => crate::delivery::revision::trunk_ref(root),
142        }
143    }
144
145    /// What lands: the ref named, or `HEAD`, which is the branch this checkout is on.
146    ///
147    /// ```
148    /// use majordomus_cli::capability::builtin::release::NextInput;
149    /// assert_eq!(NextInput::default().head(), "HEAD");
150    /// ```
151    pub fn head(&self) -> &str {
152        self.head.as_deref().unwrap_or("HEAD")
153    }
154}

332/// The capability's own id. Beside its declaration, so the two cannot drift apart without
333/// the test below noticing; the document carries it so that no page has to enumerate where
334/// this value is served.
335pub const CHANGELOG_ID: &str = "release.changelog";
336
337/// Where else the changelog can be had, read off the declarations that already say so.
338///
339/// One answer for every writer of the document: the handler that serves it and the generator
340/// that commits it both ask here, so the committed artifact and the live answer name the same
341/// surfaces. The route, the tool and the resource are the capability's own declaration; the
342/// command line is the local command that renders this capability, which `cli::LOCAL`
343/// declares once beside the reason it is local. Nothing is spelled out a second time, and a
344/// page that lists these resolves each against the datasets the registry generates rather
345/// than typing a route of its own. `None` only when the registry does not carry the
346/// capability at all, which the module's own test refuses.
347///
348/// ```
349/// use majordomus_cli::capability::builtin::release::{module, produced_by};
350/// use majordomus_cli::capability::registry::CapabilityRegistry;
351///
352/// let registry = CapabilityRegistry::builder()
353///     .with_modules(vec![module()])
354///     .build()
355///     .expect("the release module composes on its own");
356/// let by = produced_by(&registry).expect("the registry carries release.changelog");
357/// assert_eq!(by.capability, "release.changelog");
358/// assert_eq!(by.cli.as_deref(), Some("majordomus release changelog"));
359/// assert_eq!(by.http.as_deref(), Some("/api/v1/changelog"));
360/// assert_eq!(by.mcp_tool.as_deref(), Some("majordomus_changelog"));
361/// assert_eq!(by.mcp_resource.as_deref(), Some("majordomus://changelog"));
362/// ```
363pub fn produced_by(registry: &CapabilityRegistry) -> Option<ProducedBy> {
364    let c = registry.get(CHANGELOG_ID)?;
365    let cli = crate::cli::local::LOCAL
366        .iter()
367        .find(|l| l.reason.renders() == Some(CHANGELOG_ID))
368        .map(|l| format!("majordomus {}", l.command));
369    Some(ProducedBy {
370        capability: c.id.to_string(),
371        cli,
372        http: c.exposure.http.as_ref().map(|h| h.path.clone()),
373        mcp_tool: c.exposure.mcp.as_ref().and_then(|m| m.tool.clone()),
374        mcp_resource: c
375            .exposure
376            .mcp
377            .as_ref()
378            .and_then(|m| m.resource.as_ref().map(|r| r.uri.clone())),
379    })
380}

382fn changelog(ctx: &Context, input: ChangelogInput) -> Result<Changelog, CapabilityError> {
383    let root = std::path::Path::new(&ctx.index.repository.root);
384    let mut log = release::compose(root, &ctx.index.objects);
385    log.produced_by = produced_by(&ctx.registry);
386    if let Some(wanted) = input.version.as_deref() {
387        log.sections.retain(|s| {
388            s.version == wanted
389                || s.tag.as_deref() == Some(wanted)
390                || (wanted == "unreleased" && s.unreleased)
391        });
392        if log.sections.is_empty() {
393            return Err(CapabilityError::NotFound(format!(
394                "no release section carries the version '{wanted}'"
395            )));
396        }
397    }
398    Ok(log)
399}

401/// The version report, whose `next` is the contract's answer (ADR 0051).
402///
403/// [`release::version::select`] — the same selection `release bump` takes its default target
404/// from, so the two never answer differently. When the analysis cannot be made — no published
405/// baseline carries a registry, or a version is not three numbers and the analysis could only
406/// guess — the report is still answered: `next` is absent, `decided_by` is `undecided` and the
407/// reason is carried, beside the commit evidence that does not depend on it.
408fn version(ctx: &Context, _: Empty) -> Result<VersionReport, CapabilityError> {
409    let root = std::path::Path::new(&ctx.index.repository.root);
410    Ok(release::version::select(root, &ctx.registry, &ctx.index.objects).report)
411}

413/// The one analysis, behind every surface that shows a version verdict.
414///
415/// A surface that cannot be read is a refusal and never an empty diff: an empty diff means
416/// "nothing changed", and answering that when the truth is "the baseline could not be read"
417/// is the false negative this whole subsystem exists to stop.
418fn analysis(ctx: &Context, input: AnalysisInput) -> Result<VersionPlan, CapabilityError> {
419    let root = std::path::Path::new(&ctx.index.repository.root);
420    release::compat::analyze(
421        root,
422        &ctx.registry,
423        &ctx.index.objects,
424        input.since.as_deref(),
425    )
426    .map_err(|e| CapabilityError::NotFound(e.to_string()))
427}

429/// What a landing of the input's head onto its base allocates, under this repository's
430/// policy. An unreadable policy is refused: a decision made without its floor would
431/// understate what the repository asks.
432fn next(
433    ctx: &Context,
434    input: NextInput,
435) -> Result<release::landing::LandingDecision, CapabilityError> {
436    let root = std::path::Path::new(&ctx.index.repository.root);
437    let repo = crate::repository::Repository::open(root).map_err(|e| {
438        CapabilityError::NotFound(format!("the repository could not be opened: {e}"))
439    })?;
440    let policy = crate::policy::LoadedPolicy::load(&repo)
441        .map_err(|e| CapabilityError::NotFound(format!("the policy could not be read: {e}")))?
442        .policy
443        .release;
444    let base = input.base(root).map_err(|e| {
445        CapabilityError::NotFound(format!(
446            "the trunk this landing raises from could not be told: {e}; name it with --base"
447        ))
448    })?;
449    release::landing::next_for(root, &base, input.head(), &policy)
450        .map_err(|e| CapabilityError::NotFound(e.to_string()))
451}

453/// Whether the trunk's landed work is released, as of now; the version decided unless the
454/// caller said it does not need it.
455fn currency(
456    ctx: &Context,
457    input: CurrencyInput,
458) -> Result<release::currency::Currency, CapabilityError> {
459    let root = std::path::Path::new(&ctx.index.repository.root);
460    let mut answer =
461        release::currency::measure(root, &ctx.index.objects, std::time::SystemTime::now());
462    if input.next() {
463        release::currency::decide_next(&mut answer, root, &ctx.registry, &ctx.index.objects);
464    }
465    Ok(answer)
466}

598    /// Two capabilities, and every surface that carries them is derived from this
599    /// declaration: the MCP tools, the changelog resource, the HTTP routes, the OpenAPI
600    /// operations. A refactor that renamed a route, dropped the resource or added a third
601    /// would still compile, and the suites that exercise the changelog behind it would
602    /// still pass. This is the assertion that would not.
603    #[test]
604    fn the_declaration_yields_the_projections_it_claims() {
605        let m = module();
606        assert_eq!(m.id.as_str(), "release");
607        assert_eq!(
608            ANALYSIS_ID, "release.analysis",
609            "the id beside the declaration is the declared one"
610        );
611        let expected: &[(&str, &str, &str)] = &[
612            (
613                "release.changelog",
614                "majordomus_changelog",
615                "/api/v1/changelog",
616            ),
617            (
618                "release.version",
619                "majordomus_release_version",
620                "/api/v1/release/version",
621            ),
622            (
623                "release.analysis",
624                "majordomus_release_analysis",
625                "/api/v1/release/analysis",
626            ),
627            (
628                "release.next",
629                "majordomus_release_next",
630                "/api/v1/release/next",
631            ),
632            (
633                "release.currency",
634                "majordomus_release_currency",
635                "/api/v1/release/currency",
636            ),
637        ];
638        let ids: Vec<&str> = m
639            .capabilities
640            .iter()
641            .map(|e| e.capability.id.as_str())
642            .collect();
643        let want: Vec<&str> = expected.iter().map(|(id, _, _)| *id).collect();
644        assert_eq!(
645            ids, want,
646            "the module declares a different set of capabilities"
647        );
648        for (executable, (id, tool, path)) in m.capabilities.iter().zip(expected) {
649            let exposure = &executable.capability.exposure;
650            assert_eq!(
651                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
652                Some(*tool),
653                "{id} lost or renamed its MCP tool"
654            );
655            assert_eq!(
656                exposure.http.as_ref().map(|h| h.path.as_str()),
657                Some(*path),
658                "{id} lost or renamed its HTTP route"
659            );
660        }
661        // the changelog is also a resource, so a client may read it without calling a tool
662        let changelog = &m.capabilities[0].capability.exposure;
663        assert_eq!(
664            changelog
665                .mcp
666                .as_ref()
667                .and_then(|m| m.resource.as_ref())
668                .map(|r| r.uri.as_str()),
669            Some(CHANGELOG_URI),
670            "the changelog stopped being an MCP resource"
671        );
672    }
