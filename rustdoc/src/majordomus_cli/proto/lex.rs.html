1//! The tokenizer for the proto dialect the layer's document schemas are written in.
2//!
3//! Line comments are not discarded: the block of `//` lines immediately above a
4//! declaration is that declaration's documentation, and it becomes the `description` of
5//! the projected JSON Schema. That is the whole reason this is a hand-written lexer rather
6//! than a split on whitespace — the prose beside a field is part of the contract, and a
7//! reader who has to open the proto to learn what a key means has been failed by the
8//! projection.
9
10/// One token, with the comment block that preceded it.
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub struct Token<'a> {
13    /// The token text: an identifier, a number, one punctuation character, or a quoted
14    /// string including its quotes.
15    pub text: &'a str,
16    /// The comment block immediately above, joined with single spaces; empty when none.
17    pub doc: String,
18    /// 1-based line, for diagnostics.
19    pub line: usize,
20}
21
22impl Token<'_> {
23    /// The unquoted content when the token is a string literal, `None` otherwise.
24    pub fn string(&self) -> Option<String> {
25        let inner = self
26            .text
27            .strip_prefix('"')
28            .and_then(|s| s.strip_suffix('"'))?;
29        // The dialect uses only the escapes a pattern needs.
30        let mut out = String::with_capacity(inner.len());
31        let mut chars = inner.chars();
32        while let Some(c) = chars.next() {
33            if c != '\\' {
34                out.push(c);
35                continue;
36            }
37            match chars.next() {
38                Some('n') => out.push('\n'),
39                Some('t') => out.push('\t'),
40                Some(other) => out.push(other),
41                None => out.push('\\'),
42            }
43        }
44        Some(out)
45    }
46}
47
48/// The punctuation this dialect uses. Every one is a token of its own.
49const PUNCTUATION: &[char] = &['{', '}', '[', ']', '(', ')', '=', ';', ',', '.', '<', '>'];
50
51pub struct Tokenizer<'a> {
52    text: &'a str,
53    /// Byte offset of the next character to read.
54    at: usize,
55    line: usize,
56    /// The comment lines gathered since the last token.
57    pending: Vec<String>,
58}
59
60impl<'a> Tokenizer<'a> {
61    pub fn new(text: &'a str) -> Self {
62        Tokenizer {
63            text,
64            at: 0,
65            line: 1,
66            pending: Vec::new(),
67        }
68    }
69
70    fn rest(&self) -> &'a str {
71        &self.text[self.at..]
72    }
73
74    fn bump(&mut self, bytes: usize) -> &'a str {
75        let taken = &self.text[self.at..self.at + bytes];
76        self.line += taken.matches('\n').count();
77        self.at += bytes;
78        taken
79    }
80
81    /// Whitespace and comments, up to the next real token. A blank line ends a comment
82    /// block, so prose that belongs to the file header does not attach to the first
83    /// declaration below it.
84    fn trivia(&mut self) {
85        loop {
86            let rest = self.rest();
87            if rest.is_empty() {
88                return;
89            }
90            if let Some(c) = rest.chars().next() {
91                if c.is_whitespace() {
92                    let newlines_before = self.line;
93                    let len = c.len_utf8();
94                    self.bump(len);
95                    // two newlines in a row: the comment block above is not this
96                    // declaration's
97                    if c == '\n' && self.rest().starts_with('\n') && self.line > newlines_before {
98                        self.pending.clear();
99                    }
100                    continue;
101                }
102            }
103            if let Some(after) = rest.strip_prefix("//") {
104                let len = after.find('\n').unwrap_or(after.len());
105                let comment = &after[..len];
106                self.bump(2 + len);
107                self.pending.push(comment.trim().to_string());
108                continue;
109            }
110            if let Some(after) = rest.strip_prefix("/*") {
111                let len = after.find("*/").map(|i| i + 4).unwrap_or(rest.len());
112                self.bump(len);
113                continue;
114            }
115            return;
116        }
117    }
118
119    fn take_doc(&mut self) -> String {
120        let doc = self
121            .pending
122            .iter()
123            .map(String::as_str)
124            .collect::<Vec<_>>()
125            .join(" ")
126            .trim()
127            .to_string();
128        self.pending.clear();
129        doc
130    }
131}
132
133impl<'a> Iterator for Tokenizer<'a> {
134    type Item = Token<'a>;
135
136    fn next(&mut self) -> Option<Token<'a>> {
137        self.trivia();
138        let rest = self.rest();
139        let first = rest.chars().next()?;
140        let line = self.line;
141
142        if first == '"' {
143            // to the closing quote, honouring a backslash escape
144            let bytes = rest.as_bytes();
145            let mut i = 1;
146            while i < bytes.len() {
147                match bytes[i] {
148                    b'\\' => i += 2,
149                    b'"' => {
150                        i += 1;
151                        break;
152                    }
153                    _ => i += 1,
154                }
155            }
156            let text = self.bump(i.min(rest.len()));
157            let doc = self.take_doc();
158            return Some(Token { text, doc, line });
159        }
160
161        if PUNCTUATION.contains(&first) {
162            let text = self.bump(first.len_utf8());
163            let doc = self.take_doc();
164            return Some(Token { text, doc, line });
165        }
166
167        // an identifier or a number: to the next whitespace or punctuation
168        let len = rest
169            .find(|c: char| c.is_whitespace() || PUNCTUATION.contains(&c))
170            .unwrap_or(rest.len());
171        let text = self.bump(len.max(1));
172        let doc = self.take_doc();
173        Some(Token { text, doc, line })
174    }
175}

177#[cfg(test)]
178mod tests {
179    use super::*;
180
181    #[test]
182    fn splits_punctuation_from_words() {
183        let tokens: Vec<_> = Tokenizer::new("message Header { string id = 1; }")
184            .map(|t| t.text)
185            .collect();
186        assert_eq!(
187            tokens,
188            ["message", "Header", "{", "string", "id", "=", "1", ";", "}"]
189        );
190    }
191
192    #[test]
193    fn a_comment_block_attaches_to_the_declaration_below_it() {
194        let mut it = Tokenizer::new("// one\n// two\nstring id = 1;");
195        let first = it.next().unwrap();
196        assert_eq!(first.text, "string");
197        assert_eq!(first.doc, "one two");
198    }
199
200    #[test]
201    fn a_blank_line_ends_a_comment_block() {
202        let mut it = Tokenizer::new("// a file header\n\nstring id = 1;");
203        assert_eq!(it.next().unwrap().doc, "");
204    }
205
206    #[test]
207    fn reads_a_string_with_escapes_and_keeps_a_pattern_intact() {
208        let mut it = Tokenizer::new(r#""^adr-[0-9]{4}$""#);
209        assert_eq!(it.next().unwrap().string().unwrap(), "^adr-[0-9]{4}$");
210        let mut it = Tokenizer::new(r#""a\"b""#);
211        assert_eq!(it.next().unwrap().string().unwrap(), "a\"b");
212    }
213
214    #[test]
215    fn counts_lines_for_diagnostics() {
216        let mut it = Tokenizer::new("a\n\n\nb");
217        assert_eq!(it.next().unwrap().line, 1);
218        assert_eq!(it.next().unwrap().line, 4);
219    }
220}
