1//! `majordomus mcp`: serve one MCP client over stdio. By default the process joins the
2//! repository's one shared server: it starts it when nobody has (loopback HTTP with
3//! Swagger UI, OpenAPI and MCP over HTTP beside its own stdio session) and bridges its
4//! stdio to it when someone has. `--standalone` serves the client alone, as the first
5//! version of this executable did; `--inspect` prints what would be served and stops.

28/// Run `majordomus mcp`.
29pub fn run(args: McpArgs) -> Result<u8> {
30    if args.inspect {
31        let app = App::load(&args.repo)?;
32        return inspect(&Surface::new(app.context.clone()), args.format);
33    }
34    match args.transport {
35        Transport::Stdio => {}
36    }
37    let stdin = std::io::stdin();
38    let stdout = std::io::stdout();
39    if args.standalone {
40        let mut server = standalone(&args.repo)?;
41        tracing::info!("standalone: no shared server, no HTTP, no peers");
42        stdio::serve(stdin.lock(), stdout.lock(), |m| server.handle(m))?;
43        return Ok(0);
44    }
45    let repo = Repository::discover(&start_dir(&args.repo)?)?;
46    let mut session = Session::open(args, repo)?;
47    stdio::serve(stdin.lock(), stdout.lock(), |m| session.handle(m))?;
48    session.finish();
49    Ok(0)
50}

52/// A server answering one stdio session alone: no shared server, no HTTP, no peers.
53fn standalone(repo: &RepoArgs) -> Result<Server> {
54    let app = App::load(repo)?;
55    // a stdio session outlives commits like any other: it follows the repository too
56    let live = Arc::new(Live::watching(repo.clone(), app.context.clone()));
57    let peer = live.current().peers.attach(PeerTransport::Stdio);
58    Ok(Server::new(
59        Surface::new(Arc::clone(&live)).for_peer(peer),
60        crate::VERSION,
61    ))
62}

64fn start_dir(args: &RepoArgs) -> Result<std::path::PathBuf> {
65    match &args.repo {
66        Some(p) => Ok(p.clone()),
67        None => std::env::current_dir().map_err(|e| Error::io(".", e)),
68    }
69}

71/// The stdio session of this process, answered locally by the shared server this
72/// process runs, or forwarded to the one another process runs.
73struct Session {
74    args: McpArgs,
75    repo: Repository,
76    backend: Backend,
77}
78
79enum Backend {
80    Local(Box<Local>),
81    Remote {
82        bridge: Arc<Mutex<Bridge>>,
83        heartbeat: Heartbeat,
84    },
85    /// The shared server cannot be used (the lease cannot be written or replaced, the
86    /// server cannot start): this client is served alone, as `--standalone` would, and
87    /// the log said why.
88    Alone(Box<Server>),
89}
90
91/// This process is the shared server; its own stdio is answered here.
92struct Local {
93    server: Server,
94    peer: PeerId,
95    shared: SharedServer,
96}
97
98struct Heartbeat {
99    stop: Arc<AtomicBool>,
100    thread: Option<JoinHandle<()>>,
101}
102
103impl Heartbeat {
104    fn start(bridge: Arc<Mutex<Bridge>>) -> Self {
105        let stop = Arc::new(AtomicBool::new(false));
106        let flag = Arc::clone(&stop);
107        let thread = std::thread::Builder::new()
108            .name("mcp-heartbeat".into())
109            .spawn(move || {
110                let tick = Duration::from_millis(500);
111                let mut waited = Duration::ZERO;
112                while !flag.load(Ordering::SeqCst) {
113                    std::thread::sleep(tick);
114                    waited += tick;
115                    if waited >= HEARTBEAT {
116                        waited = Duration::ZERO;
117                        if let Err(e) = lock(&bridge).heartbeat() {
118                            tracing::debug!("heartbeat: {e}");
119                        }
120                    }
121                }
122            })
123            .ok();
124        Heartbeat { stop, thread }
125    }
126
127    fn stop(mut self) {
128        self.stop.store(true, Ordering::SeqCst);
129        if let Some(t) = self.thread.take() {
130            let _ = t.join();
131        }
132    }
133}

135impl Session {
136    fn open(args: McpArgs, repo: Repository) -> Result<Self> {
137        let backend = match lease::elect(&repo) {
138            Ok(Role::Server(lease)) => match Self::serve(&args, lease, None) {
139                Ok(backend) => backend,
140                Err(e) => Self::alone(&args, None, &format!("starting it failed: {e}"))?,
141            },
142            Ok(Role::Peer { url }) => Self::attach(url),
143            Err(e) => Self::alone(&args, None, &e.to_string())?,
144        };
145        Ok(Session {
146            args,
147            repo,
148            backend,
149        })
150    }
151
152    /// The shared server cannot be used: serve this client alone and say why. The layer
153    /// itself still has to load; when it does not, that error is the answer, not this.
154    fn alone(args: &McpArgs, resume: Option<ClientInfo>, why: &str) -> Result<Backend> {
155        let mut server = standalone(&args.repo)?;
156        tracing::warn!(
157            "cannot use the shared server: {why}; serving this client alone (no HTTP, no Swagger UI, no peers), as --standalone would"
158        );
159        if let Some(client) = resume {
160            server.resume(client);
161        }
162        Ok(Backend::Alone(Box::new(server)))
163    }
164
165    /// Become the shared server: load the layer, bind, and answer this stdio locally.
166    fn serve(
167        args: &McpArgs,
168        lease: Lease,
169        resume: Option<crate::peers::ClientInfo>,
170    ) -> Result<Backend> {
171        // the lease is kept young while the layer loads, so that a peer waiting on it
172        // never mistakes a slow start for an abandoned one
173        lease.keep_alive();
174        let app = App::load(&args.repo)?;
175        let live = Arc::new(Live::watching(args.repo.clone(), app.context.clone()));
176        let ctx = live.current();
177        let (http_host, _) = crate::cli::local_http_host(args.http_host.as_deref());
178        let shared = SharedServer::start(
179            Arc::clone(&live),
180            crate::VERSION,
181            &http_host,
182            args.http_port,
183            true,
184            None,
185            lease,
186            Some(app.share.dir()),
187        )?;
188        let peer = ctx.peers.attach(PeerTransport::Stdio);
189        let mut server = Server::new(Surface::new(live).for_peer(peer.clone()), crate::VERSION)
190            .with_endpoint(Some(shared.url()));
191        if let Some(client) = resume {
192            server.resume(client);
193        }
194        Ok(Backend::Local(Box::new(Local {
195            server,
196            peer,
197            shared,
198        })))
199    }
200
201    /// Attach to the shared server another process runs.
202    fn attach(url: String) -> Backend {
203        tracing::info!(
204            url = %url,
205            "a shared server for this repository is already running at {url} (swagger ui {url}{swagger}); bridging this stdio session to it",
206            swagger = crate::http::swagger::SWAGGER_PATH
207        );
208        let bridge = Arc::new(Mutex::new(Bridge::new(url)));
209        let heartbeat = Heartbeat::start(Arc::clone(&bridge));
210        Backend::Remote { bridge, heartbeat }
211    }
212
213    fn handle(&mut self, message: Value) -> Option<Reply> {
214        match &mut self.backend {
215            Backend::Local(local) => local.server.handle(message),
216            Backend::Alone(server) => server.handle(message),
217            Backend::Remote { bridge, .. } => {
218                let answer = lock(bridge).handle(&message);
219                match answer {
220                    Ok(v) => v.map(Reply::Value),
221                    Err(e) if e.calls_for_an_election() => self.failover(message, e),
222                    // the server is there and would not take this request: that is its
223                    // answer, and the client is given it instead of an election
224                    Err(e) => refused(&message, &e),
225                }
226            }
227        }
228    }
229
230    /// The server this session was bridged to is gone: elect again, and either become
231    /// the server (carrying the client's session over) or attach to whoever did.
232    fn failover(&mut self, message: Value, cause: BridgeError) -> Option<Reply> {
233        tracing::warn!("{cause}; electing again");
234        let (client, announcements) = match &self.backend {
235            Backend::Remote { bridge, .. } => {
236                let b = lock(bridge);
237                (b.client().cloned(), b.announcements().cloned().collect())
238            }
239            Backend::Local(_) | Backend::Alone(_) => (None, Vec::new()),
240        };
241        match lease::elect(&self.repo) {
242            Ok(Role::Server(lease)) => {
243                match Self::serve(&self.args, lease, client.clone()) {
244                    Ok(backend) => {
245                        self.replace(backend);
246                        tracing::info!(
247                            "took over as the shared server; this session continues locally"
248                        );
249                        // what the client said it was working on, said again on the board it
250                        // now serves itself: an announcement outlives the server it was made
251                        // to. Every claim it holds, not the newest — a client that carried
252                        // one of three onto its own board would be understating itself to
253                        // every peer that arrives after the takeover.
254                        if let Backend::Local(local) = &self.backend {
255                            for a in &announcements {
256                                let intent = a["intent"].as_str().unwrap_or_default().trim();
257                                if intent.is_empty() {
258                                    continue;
259                                }
260                                let scope: Vec<String> = a["scope"]
261                                    .as_array()
262                                    .map(|s| {
263                                        s.iter()
264                                            .filter_map(|v| v.as_str())
265                                            .map(|v| v.trim().to_string())
266                                            .filter(|v| !v.is_empty())
267                                            .collect()
268                                    })
269                                    .unwrap_or_default();
270                                let claim =
271                                    a["claim"].as_str().map(str::trim).filter(|c| !c.is_empty());
272                                local.server.surface().context().peers.announce_claim(
273                                    &local.peer,
274                                    claim,
275                                    intent,
276                                    scope,
277                                );
278                            }
279                            if !announcements.is_empty() {
280                                tracing::info!(
281                                    claims = announcements.len(),
282                                    "the client's announcement was carried onto this server's board"
283                                );
284                            }
285                        }
286                        self.handle(message)
287                    }
288                    Err(e) => {
289                        tracing::error!("cannot take over as the shared server: {e}");
290                        self.settle_alone(
291                            message,
292                            client,
293                            &format!("{cause}; and taking over failed: {e}"),
294                        )
295                    }
296                }
297            }
298            Ok(Role::Peer { url }) => {
299                let Backend::Remote { bridge, .. } = &self.backend else {
300                    return unavailable(&message, &cause.to_string());
301                };
302                let mut b = lock(bridge);
303                b.move_to(url.clone());
304                let retry = b.reinitialize().and_then(|()| b.handle(&message));
305                match retry {
306                    Ok(v) => {
307                        tracing::info!(url = %url, "re-attached to the shared server");
308                        v.map(Reply::Value)
309                    }
310                    Err(e) => unavailable(&message, &e.to_string()),
311                }
312            }
313            Err(e) => self.settle_alone(
314                message,
315                client,
316                &format!("{cause}; electing again failed: {e}"),
317            ),
318        }
319    }
320
321    /// Replace the backend, stopping the heartbeat of a bridge that is left behind.
322    fn replace(&mut self, backend: Backend) {
323        let old = std::mem::replace(&mut self.backend, backend);
324        if let Backend::Remote { heartbeat, .. } = old {
325            heartbeat.stop();
326        }
327    }
328
329    /// Neither serving nor attaching worked: serve this client alone when the layer
330    /// loads, and answer the request with an error naming every failure when it does not.
331    fn settle_alone(
332        &mut self,
333        message: Value,
334        client: Option<ClientInfo>,
335        reason: &str,
336    ) -> Option<Reply> {
337        match Self::alone(&self.args, client, reason) {
338            Ok(backend) => {
339                self.replace(backend);
340                self.handle(message)
341            }
342            Err(e) => {
343                tracing::error!("cannot serve this client alone either: {e}");
344                unavailable(&message, reason)
345            }
346        }
347    }
348
349    /// The client has gone. A server lingers while other peers are attached; a bridge
350    /// tells the server its session is over.
351    fn finish(self) {
352        match self.backend {
353            Backend::Local(local) => {
354                let Local {
355                    server,
356                    peer,
357                    shared,
358                } = *local;
359                // Detached, not closed: the stdio client that just went may be a client
360                // somebody is restarting, and it comes back to its own episode by name.
361                // `shared.stop()` below closes whatever is still held, which is the moment
362                // at which nothing can come back any more (ADR 0103).
363                server.surface().context().episodes.detach(&peer);
364                server.surface().context().peers.detach(&peer);
365                drop(server);
366                shared.wait_until_peers_leave();
367                shared.stop();
368            }
369            Backend::Remote { bridge, heartbeat } => {
370                heartbeat.stop();
371                lock(&bridge).close();
372                tracing::info!("bridge closed");
373            }
374            Backend::Alone(_) => {}
375        }
376    }
377}

379/// The JSON-RPC answer for a request that no server could take; nothing for a notification.
380fn unavailable(message: &Value, reason: &str) -> Option<Reply> {
381    let id = message.get("id").cloned().filter(|i| !i.is_null())?;
382    Some(Reply::Value(json!({
383        "jsonrpc": "2.0",
384        "id": id,
385        "error": { "code": -32603, "message": format!("shared server unavailable: {reason}") }
386    })))
387}
388
389/// The JSON-RPC answer for a request a serving shared server would not take; nothing for a
390/// notification. An invalid request, in the server's own words: what it refused and why.
391fn refused(message: &Value, cause: &BridgeError) -> Option<Reply> {
392    let id = message.get("id").cloned().filter(|i| !i.is_null())?;
393    Some(Reply::Value(json!({
394        "jsonrpc": "2.0",
395        "id": id,
396        "error": { "code": -32600, "message": cause.to_string() }
397    })))
398}

400fn lock<T>(m: &Mutex<T>) -> std::sync::MutexGuard<'_, T> {
401    m.lock().unwrap_or_else(|e| e.into_inner())
402}

404/// The word an inspection puts before a diagnostic: the verdict words every other report
405/// of this tool begins a line with.
406fn level(severity: Severity) -> &'static str {
407    match severity {
408        Severity::Error => "FAIL",
409        Severity::Warning => "WARN",
410        Severity::Info => "INFO",
411    }
412}

414/// What an inspection says about the projection, one line each: who answers and with which
415/// protocol versions, how many tools carry each effect, the tools that write the
416/// repository, where each declared client's configuration stands, and what that leaves to
417/// be done.
418fn projection_lines(p: &crate::capability::builtin::mcp::McpProjection) -> Vec<String> {
419    // an effect and a standing are written as the words they serialise to, which are the
420    // words every other surface shows
421    fn word<T: serde::Serialize>(value: T) -> String {
422        serde_json::to_value(value)
423            .ok()
424            .and_then(|v| v.as_str().map(String::from))
425            .unwrap_or_default()
426    }
427    let mut lines = vec![
428        format!("server      {} {}", p.server.name, p.server.version),
429        format!("protocol    {}", p.protocol_versions.join(", ")),
430    ];
431    lines.extend(
432        p.effects
433            .iter()
434            .map(|e| format!("effect      {:<20} {} tool(s)", word(e.effect), e.tools)),
435    );
436    lines.extend(p.writers.iter().map(|name| format!("writes      {name}")));
437    lines.extend(
438        p.clients
439            .iter()
440            .map(|c| format!("client      {:<24} {}", c.config, word(c.standing))),
441    );
442    lines.extend(
443        p.findings
444            .iter()
445            .map(|f| format!("WARN {:<22} - — {} ({})", f.code, f.message, f.remedy)),
446    );
447    lines
448}

450fn inspect(surface: &Surface, format: OutputFormat) -> Result<u8> {
451    let stdout = std::io::stdout();
452    let mut out = stdout.lock();
453    let index = surface.index();
454    // the projection as the capability every other surface asks describes it, so that what
455    // this prints about the protocol, the effects and the clients is not a second account
456    let described = crate::capability::builtin::mcp::describe(
457        &surface.context(),
458        crate::capability::builtin::mcp::McpProjectionInput::default(),
459    );
460    match format {
461        OutputFormat::Json => {
462            let v = json!({
463                "repository": surface.repository_info().map_err(|e| Error::Protocol { reason: e.to_string() })?,
464                "mcp": described,
465                "resources": &*surface.resources(),
466                "tools": &*surface.tools(),
467            });
468            writeln!(
469                out,
470                "{}",
471                serde_json::to_string_pretty(&v).map_err(|e| Error::Protocol {
472                    reason: e.to_string()
473                })?
474            )
475            .map_err(Error::Transport)?;
476        }
477        OutputFormat::Text => {
478            // the lines first, then one write: an inspection is printed whole or not at all
479            let mut lines = vec![
480                format!("repository  {}", index.repository.root),
481                format!("discovery   {}", index.repository.discovery),
482                format!(
483                    "state       {}",
484                    match index.state {
485                        crate::index::State::Ok => "ok",
486                        crate::index::State::Degraded => "degraded",
487                    }
488                ),
489            ];
490            lines.extend(
491                index
492                    .kinds()
493                    .into_iter()
494                    .map(|(kind, n)| format!("kind        {kind:<12} {n}")),
495            );
496            let summary = surface.registry().summary();
497            lines.push(format!(
498                "capabilities {} ({} builtin, {} declarative)",
499                summary.total, summary.builtin, summary.declarative
500            ));
501            lines.extend(projection_lines(&described));
502            lines.extend(
503                surface
504                    .resources()
505                    .iter()
506                    .map(|r| format!("resource    {}", r.uri)),
507            );
508            lines.extend(
509                surface
510                    .tools()
511                    .iter()
512                    .map(|t| format!("tool        {}", t.name)),
513            );
514            lines.extend(index.diagnostics.iter().map(|d| {
515                format!(
516                    "{:<4} {:<22} {} — {}",
517                    level(d.severity),
518                    d.code,
519                    d.path.as_deref().unwrap_or("-"),
520                    d.message
521                )
522            }));
523            writeln!(out, "{}", lines.join("\n")).map_err(Error::Transport)?;
524        }
525    }
526    Ok(if index.errors() > 0 { 10 } else { 0 })
527}
