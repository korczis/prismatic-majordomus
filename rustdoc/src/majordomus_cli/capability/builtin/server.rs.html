1//! The `server` module: the shared server of this checkout, and of every other checkout of
2//! the same git repository — what each lease says, whether the server it names answers,
3//! and whether what answers is the code on disk.
4//!
5//! The lease has always carried this — the pid, the address, when the server started, the
6//! executable it runs — and nothing served it: a person read the file with `cat`, a shell
7//! recipe parsed it with `sed`, and a client learned only what the election happened to
8//! log. Two readers had grown their own ideas of "ready" on the way (the environment's
9//! connection attempt, the health route's registry count), and neither asked the question
10//! the lease can answer: is the process behind this address current? This is that answer,
11//! on every surface the registry projects to, from the one typed reading of the file
12//! ([`crate::lease::LeaseFile`]).
13//!
14//! # One repository, every server of it
15//!
16//! A server serves a checkout; a linked worktree is a checkout of its own with a lease of
17//! its own, and until now nothing said that the two belong to one repository. The status
18//! lists every checkout git registers for the repository,
19//! reads the lease of each (through the worktree topology reader), and names the git repository they share
20//! ([`crate::repository::git_identity`]), so that a session in one worktree can see the
21//! servers — and through them the peers — of the others.
22//!
23//! # Two questions, two costs
24//!
25//! Listing every checkout costs a lease read and a probe per checkout, and a machine with
26//! forty worktrees registered pays forty of each. A caller that only wants to know about
27//! the checkout it is in says so — `checkouts: this` — and that answer reads one lease and
28//! probes one server; no other checkout is enumerated, read or probed. The default is
29//! `repository`, the whole list, because that is what every caller of this capability got
30//! before the field existed.
31//!
32//! Read on every call and never cached: the lease is written by other processes.
33//!
34//! ```
35//! use majordomus_cli::capability::builtin::server::{
36//!     module, standing_of, Checkouts, ServerStanding, ServerStatusInput,
37//! };
38//! use majordomus_cli::lease::LeaseFile;
39//! use std::time::Duration;
40//!
41//! // asking nothing asks the wide question, as it always did
42//! assert_eq!(
43//!     serde_json::from_str::<ServerStatusInput>("{}").unwrap().checkouts,
44//!     Checkouts::Repository
45//! );
46//!
47//! // the module composes one capability, and its decision is a pure function
48//! let m = module();
49//! assert_eq!(m.id.as_str(), "server");
50//! assert!(m.capabilities.iter().any(|e| e.capability.id.to_string() == "server.status"));
51//! let (standing, reason) = standing_of(&LeaseFile::Absent, Duration::ZERO, |_| false, "1.0.0");
52//! assert_eq!(standing, ServerStanding::Absent);
53//! assert!(reason.is_none());
54//! ```

78/// Which checkouts a reading of the server status covers.
79///
80/// The two answers cost differently and that is the whole reason the choice exists: the
81/// repository's checkouts are read from git's registry of them and each one costs a lease
82/// read and a probe, while this checkout alone costs one of each and enumerates nothing.
83/// The word each variant serialises to is the word the query parameter, the tool input and
84/// the command-line flag all take.
85///
86/// ```
87/// use majordomus_cli::capability::builtin::server::Checkouts;
88/// assert_eq!(serde_json::to_value(Checkouts::This).unwrap(), "this");
89/// assert_eq!(serde_json::to_value(Checkouts::Repository).unwrap(), "repository");
90/// assert_eq!(Checkouts::default(), Checkouts::Repository, "the wide answer is what a caller who says nothing has always got");
91/// ```
92#[derive(
93    Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema, ValueEnum,
94)]
95#[serde(rename_all = "snake_case")]
96pub enum Checkouts {
97    /// Every checkout git registers for the repository, the primary first — a lease read
98    /// and a probe each. The default, because it is the answer this capability gave before
99    /// there was anything to ask.
100    #[default]
101    Repository,
102    /// This checkout alone. No other checkout is enumerated, no other lease is read and no
103    /// other server is probed; `servers` holds the one entry, and it is this one.
104    This,
105}

107/// What to ask of `server.status`: how wide the answer is.
108///
109/// One field, because the standing of a server is not something a caller filters — it is
110/// something a caller wants for one checkout or for all of them, and the difference is
111/// forty round trips on a machine with forty worktrees.
112///
113/// ```
114/// use majordomus_cli::capability::builtin::server::{Checkouts, ServerStatusInput};
115/// // the field is optional on every surface, and its absence is the wide answer
116/// let asked_nothing: ServerStatusInput = serde_json::from_str("{}").unwrap();
117/// assert_eq!(asked_nothing.checkouts, Checkouts::Repository);
118/// let narrow: ServerStatusInput = serde_json::from_str(r#"{"checkouts":"this"}"#).unwrap();
119/// assert_eq!(narrow.checkouts, Checkouts::This);
120/// ```
121#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
122#[serde(deny_unknown_fields)]
123pub struct ServerStatusInput {
124    /// Which checkouts the answer covers. Absent means `repository`.
125    #[serde(default)]
126    pub checkouts: Checkouts,
127}

150/// Where a checkout's server stands, decided from its lease, whether the server the lease
151/// names answers for that checkout, and whether what answers is this executable's code.
152///
153/// ```
154/// use majordomus_cli::capability::builtin::server::ServerStanding;
155/// let all = [ServerStanding::Absent, ServerStanding::Starting, ServerStanding::Ready,
156///            ServerStanding::Outdated, ServerStanding::Stale];
157/// let words: Vec<&str> = all.iter().map(|s| s.as_str()).collect();
158/// assert_eq!(words, ["absent", "starting", "ready", "outdated", "stale"]);
159/// assert_eq!(serde_json::from_str::<ServerStanding>("\"stale\"").unwrap(), ServerStanding::Stale);
160/// ```
161#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
162#[serde(rename_all = "snake_case")]
163pub enum ServerStanding {
164    /// No lease: nothing serves this checkout.
165    Absent,
166    /// A lease without an address, young enough that its owner is still binding.
167    Starting,
168    /// The server the lease names answers for this checkout, from the code on disk, at
169    /// this executable's version.
170    Ready,
171    /// The server answers, but from older code than this executable, or from a file that
172    /// has been replaced since it started: everything it says is yesterday's.
173    Outdated,
174    /// The lease names a server that does not answer, or is not a lease at all.
175    Stale,
176}
177
178impl ServerStanding {
179    /// The word every surface prints: the same one the JSON carries, so that a shell
180    /// comparing the two never has to know which case each side chose.
181    ///
182    /// ```
183    /// use majordomus_cli::capability::builtin::server::ServerStanding;
184    /// assert_eq!(ServerStanding::Outdated.as_str(), "outdated");
185    /// assert_eq!(serde_json::to_value(ServerStanding::Outdated).unwrap(), "outdated");
186    /// ```
187    pub fn as_str(self) -> &'static str {
188        match self {
189            ServerStanding::Absent => "absent",
190            ServerStanding::Starting => "starting",
191            ServerStanding::Ready => "ready",
192            ServerStanding::Outdated => "outdated",
193            ServerStanding::Stale => "stale",
194        }
195    }
196}

222/// A lease as a reader sees it: what the server wrote about itself, without the token that

244    pub executable: Option<ExecutableIdentity>,

328/// Decide where a server stands from what its lease file holds, how old the file is,
329/// whether the server it names answers for the checkout, and this executable's version.
330///
331/// Pure, so that every branch of the decision is a test rather than a deployment:
332///
333/// ```
334/// use majordomus_cli::capability::builtin::server::{standing_of, ServerStanding};
335/// use majordomus_cli::lease::{LeaseDocument, LeaseFile, SCHEMA};
336/// use std::time::Duration;
337///
338/// let young = Duration::ZERO;
339/// let never = |_: &str| false;
340/// let always = |_: &str| true;
341/// assert_eq!(standing_of(&LeaseFile::Absent, young, never, "1.0.0").0, ServerStanding::Absent);
342/// assert_eq!(standing_of(&LeaseFile::Empty, young, never, "1.0.0").0, ServerStanding::Starting);
343/// assert_eq!(standing_of(&LeaseFile::Corrupt("not JSON".into()), young, never, "1.0.0").0, ServerStanding::Stale);
344///
345/// let lease = |url: Option<&str>, version: Option<&str>| LeaseDocument {
346///     schema: SCHEMA.into(), pid: 1, token: "t".into(), root: "/r".into(),
347///     url: url.map(str::to_string), started_at: "2026-09-10T00:00:00Z".into(),
348///     executable: None, version: version.map(str::to_string),
349/// };
350/// // still binding, then abandoned
351/// assert_eq!(standing_of(&LeaseFile::Document(lease(None, None)), young, never, "1.0.0").0, ServerStanding::Starting);
352/// assert_eq!(standing_of(&LeaseFile::Document(lease(None, None)), Duration::from_secs(3600), never, "1.0.0").0, ServerStanding::Stale);
353/// // names a server that does not answer
354/// assert_eq!(standing_of(&LeaseFile::Document(lease(Some("http://127.0.0.1:1"), Some("1.0.0"))), young, never, "1.0.0").0, ServerStanding::Stale);
355/// // answers from this version
356/// assert_eq!(standing_of(&LeaseFile::Document(lease(Some("http://127.0.0.1:1"), Some("1.0.0"))), young, always, "1.0.0").0, ServerStanding::Ready);
357/// // answers from another version, or from before versions were written down
358/// assert_eq!(standing_of(&LeaseFile::Document(lease(Some("http://127.0.0.1:1"), Some("0.9.0"))), young, always, "1.0.0").0, ServerStanding::Outdated);
359/// assert_eq!(standing_of(&LeaseFile::Document(lease(Some("http://127.0.0.1:1"), None)), young, always, "1.0.0").0, ServerStanding::Outdated);
360/// ```
361pub fn standing_of(
362    file: &LeaseFile,
363    age: Duration,
364    answers: impl Fn(&str) -> bool,
365    version: &str,
366) -> (ServerStanding, Option<String>) {
367    let doc = match file {
368        LeaseFile::Absent => return (ServerStanding::Absent, None),
369        LeaseFile::Empty if age > BIND_GRACE => {
370            return (
371                ServerStanding::Stale,
372                Some("empty lease: its owner never wrote it".into()),
373            )
374        }
375        LeaseFile::Empty => return (ServerStanding::Starting, None),
376        LeaseFile::Corrupt(reason) => {
377            return (
378                ServerStanding::Stale,
379                Some(format!("corrupt lease: {reason}")),
380            )
381        }
382        LeaseFile::Document(doc) => doc,
383    };
384    let Some(url) = doc.url.as_deref() else {
385        return if age > BIND_GRACE {
386            (
387                ServerStanding::Stale,
388                Some("abandoned lease: its owner never published a URL".into()),
389            )
390        } else {
391            (ServerStanding::Starting, None)
392        };
393    };
394    if !answers(url) {
395        return (
396            ServerStanding::Stale,
397            Some(format!(
398                "the server the lease names at {url} does not answer for this checkout"
399            )),
400        );
401    }
402    if let Some(reason) = doc
403        .executable
404        .as_ref()
405        .and_then(ExecutableIdentity::replaced)
406    {
407        return (ServerStanding::Outdated, Some(reason));
408    }
409    match doc.version.as_deref() {
410        None => (
411            ServerStanding::Outdated,
412            Some(format!(
413                "the server published no version, so it is older than this executable ({version})"
414            )),
415        ),
416        Some(theirs) if theirs != version => (
417            ServerStanding::Outdated,
418            Some(format!(
419                "the server is serving version {theirs}; this executable is {version}"
420            )),
421        ),
422        Some(_) => (ServerStanding::Ready, None),
423    }
424}

426/// The repository-relative path of a checkout's local half — the directory its lease is
427/// written into — or the layer's default when no manifest can be read at that checkout.
428///
429/// A directory that is not a repository of the layer still gets an answer, because "does a
430/// server serve this checkout" is a question about a path and not about a manifest.
431///
432/// ```
433/// use majordomus_cli::capability::builtin::server::local_half;
434/// assert_eq!(local_half(std::path::Path::new("/nonexistent/checkout")), ".ai/local");
435/// ```
436pub fn local_half(root: &Path) -> String {
437    Repository::open(root)
438        .map(|r| r.local_path())
439        .unwrap_or_else(|_| ".ai/local".to_string())
440}

442/// Where the server of one checkout stands, with the lease the verdict was read from.
443///
444/// [`standing_of`] is the decision; this is the one reading that feeds it — the lease file
445/// of `worktree` under `local_half`, how old that file is, one probe of the address it
446/// names, against this executable's version. Every surface that asks where a server stands
447/// asks through here, so that `server.status` and the `server` check of `health.report`
448/// cannot drift into two answers.
449///
450/// Costs one small file read, and one HTTP round trip bounded by
451/// [`crate::lease::PROBE_TIMEOUT`] when the lease names an address. Nothing is cached: the
452/// file is written by other processes.
453///
454/// ```
455/// use majordomus_cli::capability::builtin::server::{local_half, standing_at, ServerStanding};
456/// let nowhere = std::path::Path::new("/nonexistent/checkout");
457/// let (standing, reason, lease) = standing_at(nowhere, &local_half(nowhere));
458/// assert_eq!(standing, ServerStanding::Absent);
459/// assert!(reason.is_none(), "there is nothing to explain: no lease, no server");
460/// assert!(lease.document().is_none());
461/// ```
462pub fn standing_at(
463    worktree: &Path,
464    local_half: &str,
465) -> (ServerStanding, Option<String>, LeaseFile) {
466    let file = lease::lease_file(worktree, local_half);
467    let read = LeaseFile::read(&file);
468    // this process's own lease is answered from memory: a probe would be a request to
469    // itself, queued behind the very request asking (lease::is_own)
470    let own = read.document().is_some_and(lease::is_own);
471    let (standing, reason) = standing_of(
472        &read,
473        lease::file_age(&file),
474        |url| own || lease::probe(url, worktree),
475        crate::VERSION,
476    );
477    (standing, reason, read)
478}

480/// How many peers the server at `url` reports, through the route the registry declares
481/// for `peers.list`. `None` when it does not answer or the route is not declared.
482fn peers_of(ctx: &Context, url: &str) -> Option<usize> {
483    let path = ctx
484        .registry
485        .get("peers.list")?
486        .exposure
487        .http
488        .as_ref()?
489        .path
490        .clone();
491    let reply =
492        crate::mcp::bridge::request(url, "GET", &path, &[], None, lease::PROBE_TIMEOUT).ok()?;
493    if reply.status != 200 {
494        return None;
495    }
496    let value: serde_json::Value = serde_json::from_str(&reply.body).ok()?;
497    value["count"].as_u64().map(|n| n as usize)
498}

500/// The checkouts one reading covers: every checkout git registers for the repository, the
501/// primary first, or this one alone.
502///
503/// The narrow answer is the cheap one by construction rather than by filtering — git is
504/// not asked for its registry of work trees, so nothing downstream has another checkout's
505/// path to read a lease at or a server to probe. It spawns no process of its own either:
506/// `primary` comes from the one stat `git_identity` already did (a linked work tree has a
507/// `.git` file, the primary a directory) and the branch from that checkout's own HEAD, so
508/// the narrow entry says everything about this checkout that the wide list would have
509/// said, at the price of two file reads.
510pub(crate) fn checkouts_of(
511    root: &Path,
512    git: Option<&GitIdentity>,
513    which: Checkouts,
514) -> Vec<(PathBuf, Option<String>, bool)> {
515    let here = |primary: bool| vec![(root.to_path_buf(), repository::branch_at(root), primary)];
516    let Some(git) = git else {
517        // not a git repository: this checkout is the only one there is, and it is the
518        // primary of nothing
519        return here(true);
520    };
521    if which == Checkouts::This {
522        return here(!git.linked);
523    }
524    match crate::worktree::topology::read(root) {
525        Ok(records) => records
526            .into_iter()
527            .filter(|r| !r.bare)
528            .enumerate()
529            .map(|(i, r)| (r.path, r.branch, i == 0))
530            .collect(),
531        Err(_) => here(!git.linked),
532    }
533}

535fn server_status(ctx: &Context, input: ServerStatusInput) -> Result<ServerStatus, CapabilityError> {
536    let root = PathBuf::from(&ctx.index.repository.root);
537    let root = root.canonicalize().unwrap_or(root);
538    let local_half = local_half(&root);
539    let git = repository::git_identity(&root);
540    let checkouts = checkouts_of(&root, git.as_ref(), input.checkouts);
541    let mut servers = Vec::with_capacity(checkouts.len());
542    for (path, branch, primary) in checkouts {
543        let worktree = path.canonicalize().unwrap_or(path);
544        let (standing, reason, read) = standing_at(&worktree, &local_half);
545        let peers = match standing {
546            // this process's own board is in memory; asking it over the wire would queue
547            // behind this very request (lease::is_own)
548            ServerStanding::Ready | ServerStanding::Outdated
549                if read.document().is_some_and(lease::is_own) =>
550            {
551                Some(ctx.peers.list().iter().filter(|p| p.attached).count())
552            }
553            ServerStanding::Ready | ServerStanding::Outdated => read
554                .document()
555                .and_then(|d| d.url.as_deref())
556                .and_then(|url| peers_of(ctx, url)),
557            _ => None,
558        };
559        servers.push(ServerView {
560            checkout_id: repository::identity(&worktree),
561            primary,
562            this_checkout: worktree == root,
563            branch,
564            standing,
565            reason,
566            lease: read.document().map(LeaseView::of),
567            peers,
568            worktree,
569        });
570    }
571    let standing = servers
572        .iter()
573        .find(|s| s.this_checkout)
574        .map(|s| s.standing)
575        .unwrap_or(ServerStanding::Absent);
576    Ok(ServerStatus {
577        checkout_id: repository::identity(&root),
578        git,
579        desired: Desired {
580            host: "127.0.0.1".into(),
581            port: crate::cli::DEFAULT_PORT,
582            version: crate::VERSION.into(),
583            executable: lease::executable_identity(),
584        },
585        this_process: lease::held().as_ref().map(LeaseView::of),
586        standing,
587        servers,
588    })
589}
