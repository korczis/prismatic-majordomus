1//! What this process serves, resolved once: the topology narrowed to this router, with a
2//! handler bound to every surface in it.
3//!
4//! The router does not name surfaces. It asks this table who owns a path, and the table was
5//! built from the resolved topology — so a static surface that a producer declared is
6//! served because it was discovered, and nothing is served that was not.
7//!
8//! A native surface is the one thing a topology cannot supply on its own: a path the
9//! executable answers needs code behind it. That binding lives here, in one match, and a
10//! native surface with no arm refuses the router at construction rather than answering 404
11//! at run time. `every_native_surface_has_a_handler` is the test that keeps the two halves
12//! of that match in step.
13

24/// The code behind a route the executable answers itself.
25///
26/// One value per native surface, matched from its declared id. The variants are what this
27/// executable can do, not what a path is called: renaming a mount changes the declaration
28/// and nothing here.
29#[derive(Debug, Clone, Copy, PartialEq, Eq)]
30pub enum Native {
31    /// The page this process is entered through.
32    Home,
33    /// The capability registry's own routes.
34    Api,
35    /// The OpenAPI document.
36    OpenApi,
37    /// The Swagger UI shell.
38    Swagger,
39    /// MCP over HTTP.
40    Mcp,
41    /// The Cockpit.
42    Cockpit,
43    /// The live channel over which executions report what they are doing.
44    Events,
45}
46
47impl Native {
48    /// The handler a native surface's id binds to, when this executable has one.
49    pub fn of(id: &str) -> Option<Native> {
50        Some(match id {
51            discover::HOME => Native::Home,
52            "api" => Native::Api,
53            "openapi" => Native::OpenApi,
54            "swagger" => Native::Swagger,
55            "mcp" => Native::Mcp,
56            "cockpit" => Native::Cockpit,
57            "events" => Native::Events,
58            _ => return None,
59        })
60    }
61}

63/// What a resolved surface is answered by.
64#[derive(Debug)]
65pub enum Bound {
66    /// A route this executable answers.
67    Route(Native),
68    /// A directory a producer generated. Boxed: a static surface carries its resolved root
69    /// and its cache, and a table of them should not pay that for the routes beside them.
70    Directory(Box<Files>),
71}
72
73/// The surfaces one router serves, with what answers each.
74#[derive(Debug)]
75pub struct Served {
76    topology: Topology,
77    bound: BTreeMap<String, Bound>,
78}
79
80impl Served {
81    /// Narrow `topology` to what a process with these runtime capabilities serves, bind a
82    /// handler to every surface in it, and refuse anything that cannot be served coherently.
83    ///
84    /// The refusal is the point: a collision, a nested mount or a native surface with no
85    /// handler stops the router being built, so it is found by whoever changed a
86    /// declaration rather than by whoever sent the first request.
87    pub fn resolve(topology: &Topology, root: &Path, runtime: Runtime) -> Result<Self> {
88        let topology = topology.served(runtime);
89        let findings = validate::validate(&topology, root, Artifacts::Ignore);
90        if validate::blocking(&findings) {
91            let reasons: Vec<String> = findings
92                .iter()
93                .filter(|f| f.severity == validate::Severity::Error)
94                .map(|f| format!("{} ({}): {}", f.rule, f.surface, f.message))
95                .collect();
96            return Err(Error::InvalidSurface {
97                surface: "topology".into(),
98                reason: format!(
99                    "this process cannot serve the surfaces it discovered: {}; `majordomus web validate` reports the same findings with their remedies",
100                    reasons.join("; ")
101                ),
102            });
103        }
104        let mut bound = BTreeMap::new();
105        for surface in &topology.surfaces {
106            let handler = match surface.kind {
107                SurfaceKind::StaticDirectory => {
108                    Bound::Directory(Box::new(Files::new(surface, root)))
109                }
110                SurfaceKind::NativeRoute => {
111                    let Some(native) = Native::of(&surface.id) else {
112                        return Err(Error::InvalidSurface {
113                            surface: surface.id.clone(),
114                            reason: format!(
115                                "is declared as a route this executable answers and no handler is bound to it; add an arm to http::surfaces::Native::of, or declare {} as a generated directory",
116                                surface.id
117                            ),
118                        });
119                    };
120                    Bound::Route(native)
121                }
122                SurfaceKind::Redirect => {
123                    return Err(Error::InvalidSurface {
124                        surface: surface.id.clone(),
125                        reason: "a redirect surface is declared and this executable serves none"
126                            .into(),
127                    })
128                }
129            };
130            bound.insert(surface.id.clone(), handler);
131        }
132        Ok(Served { topology, bound })
133    }
134
135    /// The topology this serves, narrowed and validated.
136    pub fn topology(&self) -> &Topology {
137        &self.topology
138    }
139
140    /// The same topology behind an `Arc`, for a context that hands it to a capability.
141    pub fn shared(&self) -> Arc<Topology> {
142        Arc::new(self.topology.clone())
143    }
144
145    /// The surface that owns a request path, with what answers it.
146    pub fn owner(&self, path: &str) -> Option<(&Surface, &Bound)> {
147        let surface = self.topology.owner(path)?;
148        let bound = self.bound.get(&surface.id)?;
149        Some((surface, bound))
150    }
151
152    /// Is a surface's producer's output there? A route the executable answers always is.
153    pub fn ready(&self, id: &str) -> bool {
154        match self.bound.get(id) {
155            Some(Bound::Directory(files)) => files.available(),
156            Some(Bound::Route(_)) => true,
157            None => false,
158        }
159    }
160
161    /// The one line a server logs when it starts: how many surfaces it serves and where
162    /// each one is, `base` prefixed so the log is clickable.
163    ///
164    /// Every served surface, not only the public ones: this is an operator's diagnostic,
165    /// and the operator attaching a second client needs the MCP mount that the home page
166    /// has no reason to advertise to a browser.
167    ///
168    /// Both halves are read off the resolution. A surface added tomorrow is in this line,
169    /// and a route that moves moves here, because nothing below names one.
170    ///
171    /// ```
172    /// use majordomus_cli::http::Served;
173    /// use majordomus_cli::web::{discover::{self, Runtime}, Topology};
174    /// let topology = Topology::new(discover::native_all());
175    /// let served = Served::resolve(&topology, std::path::Path::new("/nonexistent"), Runtime::full()).unwrap();
176    /// let line = served.summary("http://127.0.0.1:8741");
177    /// assert!(line.contains("http://127.0.0.1:8741/swagger"));
178    /// // the count is the resolution's, not a number written here: a surface added to
179    /// // native_all is in this line, and this assertion does not go stale for saying so
180    /// assert!(line.contains(&format!("{} surface(s)", discover::native_all().len())));
181    /// ```
182    pub fn summary(&self, base: &str) -> String {
183        let mut parts: Vec<String> = self
184            .topology
185            .surfaces
186            .iter()
187            .map(|s| {
188                let mount = s.mount.as_str();
189                let path = if s.mount.is_root() { "/" } else { mount };
190                format!("{} {base}{path}", s.id)
191            })
192            .collect();
193        crate::order::canonical(&mut parts);
194        format!(
195            "{} surface(s): {}",
196            self.topology.surfaces.len(),
197            parts.join(", ")
198        )
199    }
200}

217    #[test]
218    fn a_native_surface_with_no_handler_refuses_the_router() {
219        let mut surfaces = discover::native_all();
220        // a surface at no reserved mount, so what refuses the router is the missing
221        // handler and not a reservation
222        let cockpit = surfaces
223            .iter_mut()
224            .find(|s| s.id == "cockpit")
225            .expect("the cockpit is declared");
226        cockpit.id = "invented".into();
227        let topology = Topology::new(surfaces);
228        let err = Served::resolve(&topology, Path::new("/nonexistent"), Runtime::full())
229            .expect_err("a surface nothing answers is refused")
230            .to_string();
231        assert!(err.contains("invented"), "{err}");
232        assert!(err.contains("Native::of"), "{err}");
233    }

236    fn a_collision_refuses_the_router_and_names_both_surfaces() {
237        let mut surfaces = discover::native_all();
238        let mut clone = surfaces
239            .iter()
240            .find(|s| s.id == "cockpit")

244        surfaces.push(clone);

303    /// The crate's rustdoc, discovered from the crate, bound to the directory its producer
304    /// writes, and answered for every kind of file the reference is made of — through the
305    /// same table the router consults, so what is proved here is what a request gets.
306    #[test]
307    fn the_crates_rustdoc_is_bound_to_its_directory_beside_the_documentation() {
308        let tmp = with_a_site();
309        let krate = tmp.path().join(crate::capability::model::CRATE_DIR);
310        std::fs::create_dir_all(krate.join("src")).expect("mkdir");
311        std::fs::write(krate.join("Cargo.toml"), "").expect("write");
312        std::fs::write(krate.join("src/lib.rs"), "//! x\n").expect("write");
313
314        // declared before it is built: the mount is held and answers what to run
315        let unbuilt = discover::discover(tmp.path(), Runtime::full()).expect("discovers");
316        let served = Served::resolve(&unbuilt, tmp.path(), Runtime::full())
317            .expect("an unbuilt reference is servable");
318        assert!(!served.ready(discover::RUSTDOC));
319        let Some((surface, Bound::Directory(files))) = served.owner("/rustdoc/") else {
320            panic!("the rustdoc mount is bound to a directory");
321        };
322        assert_eq!(surface.id, discover::RUSTDOC);
323        let response = files.respond("/rustdoc/");
324        assert_eq!(response.status, 503);
325        assert!(
326            response.body.text().contains(discover::RUSTDOC_PRODUCER),
327            "{}",
328            response.body
329        );
330
331        let dir = tmp.path().join(discover::RUSTDOC_ARTIFACT);
332        for (file, body) in [
333            ("index.html", "<h1>landing</h1>"),
334            ("static.files/a.css", "body{}"),
335            ("static.files/b.js", "var b;"),
336            ("static.files/c.woff2", "wOF2"),
337            ("majordomus_cli/index.html", "<h1>crate</h1>"),
338            ("majordomus_cli/macro.m!.html", "<p>redirect</p>"),
339        ] {
340            let path = dir.join(file);
341            std::fs::create_dir_all(path.parent().expect("a parent")).expect("mkdir");
342            std::fs::write(path, body).expect("write");
343        }
344        let built = discover::discover(tmp.path(), Runtime::full()).expect("discovers");
345        let served =
346            Served::resolve(&built, tmp.path(), Runtime::full()).expect("the reference serves");
347        assert!(served.ready(discover::RUSTDOC));
348        for (path, media) in [
349            ("/rustdoc/", "text/html; charset=utf-8"),
350            ("/rustdoc/static.files/a.css", "text/css; charset=utf-8"),
351            (
352                "/rustdoc/static.files/b.js",
353                "text/javascript; charset=utf-8",
354            ),
355            ("/rustdoc/static.files/c.woff2", "font/woff2"),
356            (
357                "/rustdoc/majordomus_cli/index.html",
358                "text/html; charset=utf-8",
359            ),
360            (
361                "/rustdoc/majordomus_cli/macro.m!.html",
362                "text/html; charset=utf-8",
363            ),
364        ] {
365            let Some((surface, Bound::Directory(files))) = served.owner(path) else {
366                panic!("{path} is not owned by a directory");
367            };
368            assert_eq!(surface.id, discover::RUSTDOC, "{path}");
369            let response = files.respond(path);
370            assert_eq!(response.status, 200, "{path}: {}", response.body);
371            assert_eq!(response.content_type, media, "{path}");
372        }
373        // the documentation keeps its own mount, and the reference does not swallow it
374        assert_eq!(served.owner("/docs/").unwrap().0.id, discover::DOCS);
375        assert!(served
376            .summary("http://x")
377            .contains("rustdoc http://x/rustdoc"));
378    }
