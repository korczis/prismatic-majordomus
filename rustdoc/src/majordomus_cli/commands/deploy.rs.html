1//! `majordomus deploy`: the command line of `deploy.slug`, through the one executor.
2//!
3//! The text rendering says what was assembled, from which commit and where, then the
4//! top-level entries and the provider the slug is for; `--format json` prints the capability's
5//! answer whole, which is what a workflow reads the provider's application and URL from. A
6//! refusal exits 15, something the request names that does not exist exits 12.
7

17/// Run `majordomus deploy`.
18pub fn run(args: DeployArgs) -> Result<u8> {
19    run_to(args, &mut std::io::stdout().lock())
20}
21
22/// Run `majordomus deploy`, printing to `w`.
23fn run_to(args: DeployArgs, w: &mut dyn Write) -> Result<u8> {
24    let app = App::load(&args.repo)?;
25    let ctx = &app.context;
26    let DeployCommand::Slug {
27        deployment,
28        commit,
29        executable,
30        out,
31        site,
32    } = args.command;
33    let mut input = json!({ "executable": executable, "out": out });
34    if let Some(s) = site {
35        input["site"] = json!(s);
36    }
37    if let Some(d) = deployment {
38        input["deployment"] = json!(d);
39    }
40    if let Some(c) = commit {
41        input["commit"] = json!(c);
42    }
43    let v = execute(ctx, input)?;
44    match args.format {
45        OutputFormat::Json => writeln!(
46            w,
47            "{}",
48            serde_json::to_string_pretty(&v).unwrap_or_default()
49        ),
50        OutputFormat::Text => writeln!(w, "{}", text(&v)),
51    }
52    .map_err(Error::Transport)?;
53    Ok(0)
54}

56/// The slug as a person reads it.
57fn text(v: &Value) -> String {
58    let s = |p: &str| {
59        v.pointer(p)
60            .and_then(Value::as_str)
61            .unwrap_or("?")
62            .to_string()
63    };
64    let entries = v["entries"]
65        .as_array()
66        .map(|a| {
67            a.iter()
68                .filter_map(Value::as_str)
69                .collect::<Vec<_>>()
70                .join(" ")
71        })
72        .unwrap_or_default();
73    format!(
74        "assembled    {} at {} into {}\n\
75         entries      {entries}\n\
76         files        {} file(s), {} byte(s)\n\
77         executable   {} sha256 {}\n\
78         provider     {} {} {}{}",
79        s("/deployment"),
80        s("/commit"),
81        s("/out"),
82        v["files"],
83        v["bytes"],
84        s("/executable"),
85        s("/executable_sha256"),
86        s("/provider/name"),
87        s("/provider/app"),
88        s("/provider/url"),
89        s("/liveness"),
90    )
91}

93fn execute(ctx: &crate::capability::Context, input: Value) -> Result<Value> {
94    let words = ["deploy".to_string(), "slug".to_string()];
95    let id = ctx
96        .registry
97        .by_cli(&words)
98        .map(|c| c.id.as_str())
99        .ok_or_else(|| Error::Protocol {
100            reason: "no capability is exposed as `majordomus deploy slug`".into(),
101        })?;
102    ctx.execute(id, input).map_err(|e| match e {
103        CapabilityError::Refused(reason) => Error::Refused { code: 15, reason },
104        CapabilityError::NotFound(reason) => Error::Refused { code: 12, reason },
105        CapabilityError::InvalidInput(reason) => Error::Refused { code: 2, reason },
106        other => Error::Protocol {
107            reason: other.to_string(),
108        },
109    })
110}

116    #[test]
117    fn the_slug_reads_as_what_where_and_for_whom() {
118        let v = json!({"deployment": "majordomus", "commit": "a".repeat(40), "out": "/tmp/slug",
119            "entries": [".ai", "Procfile"], "files": 2, "bytes": 10,
120            "executable": ".majordomus/bin/majordomus", "executable_sha256": "00",
121            "buildpack": "b", "liveness": "/api/v1/live",
122            "provider": {"name": "gigalixir", "app": "a-b", "stack": "s",
123                         "url": "https://a-b.gigalixirapp.com"}});
124        let t = text(&v);
125        assert!(t.starts_with(&format!(
126            "assembled    majordomus at {} into /tmp/slug",
127            "a".repeat(40)
128        )));
129        assert!(t.contains("entries      .ai Procfile\n"));
130        assert!(t.ends_with("provider     gigalixir a-b https://a-b.gigalixirapp.com/api/v1/live"));
131    }

161    /// The command prints what was assembled as a person reads it, or the capability's
162    /// answer whole as JSON.
163    #[test]
164    fn the_deploy_command_prints_the_slug_as_text_or_json() {
165        let repo = hosted_layer(&["hosted"]);
166        let elsewhere = tempfile::tempdir().unwrap();
167        let exe = elsewhere.path().join("exe");
168        std::fs::write(&exe, b"an executable").unwrap();
169
170        let mut w = Vec::new();
171        let a = args(
172            repo.root(),
173            Some("hosted"),
174            &exe,
175            &elsewhere.path().join("text"),
176            OutputFormat::Text,
177        );
178        assert_eq!(run_to(a, &mut w).unwrap(), 0);
179        let t = String::from_utf8(w).unwrap();
180        assert!(t.starts_with("assembled    hosted at "), "{t}");
181        assert!(t.contains("provider     gigalixir hosted-app"), "{t}");
182
183        let mut w = Vec::new();
184        let a = args(
185            repo.root(),
186            None,
187            &exe,
188            &elsewhere.path().join("json"),
189            OutputFormat::Json,
190        );
191        assert_eq!(run_to(a, &mut w).unwrap(), 0);
192        let v: Value = serde_json::from_slice(&w).unwrap();
193        assert_eq!(v["provider"]["app"], "hosted-app");
194        assert_eq!(v["deployment"], "hosted");
195    }

247    /// Each capability error has its exit: not found 12, invalid input 2, and anything else
248    /// a protocol error — as is a registry that exposes no `deploy slug` at all.
249    #[test]
250    fn every_deploy_slug_error_has_its_exit() {
251        let repo = hosted_layer(&["hosted"]);
252        let ctx = repo.context().unwrap();
253        let elsewhere = tempfile::tempdir().unwrap();
254        let exe = elsewhere.path().join("exe");
255        std::fs::write(&exe, b"an executable").unwrap();
256        let out = elsewhere.path().join("slug").display().to_string();
257        let code = |input: Value| match execute(&ctx, input) {
258            Err(Error::Refused { code, .. }) => Some(code),
259            Err(Error::Protocol { .. }) => None,
260            other => panic!("{other:?}"),
261        };
262        assert_eq!(
263            code(json!({"deployment": "absent", "executable": exe, "out": out})),
264            Some(12)
265        );
266        assert_eq!(code(json!({"out": out})), Some(2));
267        assert_eq!(
268            code(json!({"executable": exe, "out": exe.join("slug")})),
269            None,
270            "a failure under a sound request is not a refusal"
271        );
272
273        let index = repo.index().unwrap();
274        let registry = crate::capability::CapabilityRegistry::builder()
275            .with_index(&index)
276            .build()
277            .unwrap();
278        let bare = crate::capability::Context::new(
279            std::sync::Arc::new(index),
280            std::sync::Arc::new(registry),
281        );
282        assert!(matches!(
283            execute(&bare, json!({})),
284            Err(Error::Protocol { reason }) if reason.contains("majordomus deploy slug")
285        ));
286    }

288    /// A slug assembled and not printed is a transport failure, not a success.
289    #[test]
290    fn a_deploy_slug_that_cannot_be_printed_is_a_transport_failure() {
291        struct Closed;
292        impl Write for Closed {
293            fn write(&mut self, _: &[u8]) -> std::io::Result<usize> {
294                Err(std::io::Error::from(std::io::ErrorKind::BrokenPipe))
295            }
296            fn flush(&mut self) -> std::io::Result<()> {
297                Ok(())
298            }
299        }
300        let repo = hosted_layer(&["hosted"]);
301        let elsewhere = tempfile::tempdir().unwrap();
302        let exe = elsewhere.path().join("exe");
303        std::fs::write(&exe, b"an executable").unwrap();
304        let a = args(
305            repo.root(),
306            None,
307            &exe,
308            &elsewhere.path().join("slug"),
309            OutputFormat::Text,
310        );
311        assert!(matches!(
312            run_to(a, &mut Closed),
313            Err(Error::Transport(e)) if e.kind() == std::io::ErrorKind::BrokenPipe
314        ));
315    }
