1//! The Dashboard: how the delivery stands, on one screen.
2//!
3//! Its first paint asks only sources that answer in well under a second on this
4//! repository, all at once: `repository.info`, `health.report`, `release.version`,
5//! `distribution.releases`, `plan.status`, `plan.roadmap` and `mesh.peers`. Measured one at a
6//! time on 2026-10-11, the slowest was `release.version` at 0.5-0.8 s. The pull-request queue
7//! (`integration.queue`) took 50 s for 1.2 MB, so it is a panel of its own at
8//! [`PULL_REQUESTS_PANEL`]: the script pulls it in once the screen has painted, and a reader
9//! without JavaScript follows its link.

36/// The sources the first paint asks, together.
37pub const FIRST_PAINT: &[Source] = &[REPOSITORY, HEALTH, VERSION, RELEASES, PLAN, ROADMAP, MESH];

39/// How many rows a panel shows before it says how many there are in all.
40pub const ROWS: usize = 8;

42/// The Dashboard screen.
43pub fn screen(ctx: &Context) -> View {
44    let answers = ask_all(ctx, FIRST_PAINT);
45    let a = |s: Source| &answers[&s];
46
47    let git = |p: &'static str| a(REPOSITORY).read(p).text();
48    let lead = match (git("/repository/git/branch"), git("/repository/git/head")) {
49        (Some(branch), Some(head)) => format!(
50            "{branch} at {} · working tree {}",
51            &head[..12.min(head.len())],
52            git("/repository/git/working_tree")
53                .as_deref()
54                .unwrap_or("unknown")
55        ),
56        _ => "The repository's version control could not be read.".into(),
57    };
58
59    let top = hero(
60        "home",
61        "blue",
62        "Dashboard",
63        "How the delivery stands, read from the repository itself.",
64        &lead,
65        vec![
66            kpi_to(kpi_state("Health", &a(HEALTH).read("/status")), "#health"),
67            kpi_to(
68                kpi("Released", &a(RELEASES).read("/latest/version")),
69                super::delivery::ROUTE,
70            ),
71            kpi_to(
72                kpi("Next release", &a(VERSION).read("/next")),
73                super::delivery::ROUTE,
74            ),
75            kpi_to(
76                kpi("Issues ready", &a(PLAN).read("/counts/by_status/READY")),
77                super::work::ROUTE,
78            ),
79            kpi_to(
80                kpi("Issues blocked", &a(PLAN).read("/counts/by_status/BLOCKED")),
81                super::roadmap::ROUTE,
82            ),
83            kpi_to(
84                kpi_count("Mesh machines", &a(MESH).read("/machines")),
85                super::mesh::ROUTE,
86            ),
87        ],
88    );
89
90    let grid = el("div")
91        .class("mj-ui-grid")
92        .child(plan_panel(a(ROADMAP), a(PLAN)))
93        .child(issues_panel(a(PLAN)))
94        .child(wide(queue_placeholder()))
95        .child(releases_panel(a(RELEASES), a(VERSION), Some(ROWS)))
96        .child(mesh_panel(a(MESH)))
97        .child(health_panel(a(HEALTH)));
98
99    View::new(
100        "Dashboard",
101        el("div").class("mj-ui-stack").child(top).child(grid),
102    )
103}

105/// The pull-request queue, as a panel: what the fragment request and the page without

107pub fn pull_requests_panel(ctx: &Context, query: &[(String, String)]) -> View {

118/// The milestones still open, in the plan's own order, each with the issues it has done
119/// out of those it holds.
120pub fn plan_panel(roadmap: &Answer, plan: &Answer) -> El {
121    let body = unanswered(roadmap).unwrap_or_else(|| {
122        let open: Vec<Value> = list(roadmap, "/milestones")
123            .into_iter()
124            .filter(|m| m.get("status").and_then(Value::as_str) != Some("DONE"))
125            .collect();
126        let rows = open
127            .iter()
128            .take(ROWS)
129            .map(milestone_row)
130            .collect::<Vec<_>>();
131        el("div")
132            .child(el("p").class("mj-ui-line").text(format!(
133                "Now {} · next {}",
134                shown(&roadmap.read("/now")),
135                shown(&roadmap.read("/next"))
136            )))
137            .child(table(
138                &["Milestone", "State", "Issues done"],
139                rows,
140                "No milestone is open.",
141            ))
142            .children(more(
143                ROWS.min(open.len()),
144                open.len(),
145                Some(super::roadmap::ROUTE),
146            ))
147    });
148    panel_to(
149        "Roadmap",
150        "The milestones still open, in the plan's own order.",
151        Some(super::roadmap::ROUTE),
152        body,
153        &[roadmap, plan],
154    )
155}

190/// The label of the tile counting merges: the window is the queue's own (`throughput/
191/// window_days`), so the label says what the queue counted rather than a window of its own.
192pub fn merged_label(queue: &Answer) -> String {
193    match queue.read("/throughput/window_days").count() {
194        Some(days) => format!("Merged, {days} days"),
195        None => "Merged, over a window the queue does not state".into(),
196    }
197}

199/// The first two reasons of a pull request, and how many more there are: a row has room for
200/// two, and its inspector holds them all.
201pub fn first_reasons(reasons: &[String]) -> String {
202    match reasons.len() {
203        0 => "none".into(),
204        1 | 2 => reasons.join(", "),
205        n => format!("{} and {} more", reasons[..2].join(", "), n - 2),
206    }
207}

209/// What was published, newest first, as a timeline: the first `limit` of them, saying how
210/// many more there are, or every one of them when there is no limit.
211pub fn releases_panel(releases: &Answer, version: &Answer, limit: Option<usize>) -> El {
212    let body = unanswered(releases).unwrap_or_else(|| {
213        let all = list(releases, "/releases");
214        let limit = limit.unwrap_or(all.len());
215        let entries = all
216            .iter()
217            .take(limit)
218            .map(|r| {
219                let commit = text(r, "commit");
220                let record = text(r, "metadata_url");
221                (
222                    text(r, "published_at"),
223                    format!("v{}", value(r, "version")),
224                    el("span")
225                        .class("mj-ui-dim")
226                        .text(format!(
227                            "{} · {}",
228                            value(r, "channel"),
229                            commit.get(..10).unwrap_or(if commit.is_empty() {
230                                "no commit"
231                            } else {
232                                &commit
233                            })
234                        ))
235                        // the release's own record, at the address the distribution gives
236                        .when(!record.is_empty(), |s| {
237                            s.text(" · ")
238                                .child(super::view::link(record.clone(), "its record"))
239                        }),
240                )
241            })
242            .collect::<Vec<_>>();
243        el("div")
244            .child(el("p").class("mj-ui-line").text(format!(
245                "Declared {} · the commits since the last release imply a {} release",
246                shown(&version.read("/declared")),
247                shown(&version.read("/bump"))
248            )))
249            .child(if entries.is_empty() {
250                nothing("Nothing has been released yet.")
251            } else {
252                timeline(entries)
253            })
254            .children(more(
255                limit.min(all.len()),
256                all.len(),
257                Some(super::delivery::ROUTE),
258            ))
259    });
260    panel_to(
261        "Releases",
262        "What was published, newest first.",
263        Some(super::delivery::ROUTE),
264        body,
265        &[releases, version],
266    )
267}

299/// One machine as a table row: its name, whether it is this one, and the liveness of each
300/// of its runtimes as the mesh runtime observed it.
301pub fn machine_row(m: &Value) -> El {
302    let local = m.get("local").and_then(Value::as_bool);
303    let words: Vec<String> = m
304        .get("runtimes")
305        .and_then(Value::as_array)
306        .map(|rs| {
307            rs.iter()
308                .filter_map(|r| r.get("liveness").and_then(Value::as_str))
309                .map(str::to_string)
310                .collect()
311        })
312        .unwrap_or_default();
313    el("tr")
314        .child(el("td").class("mj-ui-mono").text(text(m, "name")))
315        .child(el("td").text(match local {
316            Some(true) => "this machine",
317            Some(false) => "remote",
318            None => "unknown",
319        }))
320        .child(
321            el("td")
322                .class("mj-ui-badges")
323                .children(words.iter().map(|w| word_badge(w))),
324        )
325}

384/// The queue: its tallies, its throughput and the pull requests in rank order.
385pub fn queue_panel(queue: &Answer, selected: Option<u64>) -> El {
386    let body = unanswered(queue).unwrap_or_else(|| {
387        if queue.read("/observed").state != Availability::Available(Value::Bool(true)) {
388            return el("p").class("mj-ui-empty").text(format!(
389                "The forge was not observed in this checkout: {}",
390                shown(&queue.read("/reason"))
391            ));
392        }
393        let tally = |p: &'static str| queue.read(p);
394        let tallies = strip(vec![
395            kpi("Open", &tally("/queue/tallies/open")),
396            kpi(
397                "Needs refresh",
398                &tally("/queue/tallies/by_disposition/needs_refresh"),
399            ),
400            kpi(
401                "Needs repair",
402                &tally("/queue/tallies/by_disposition/needs_repair"),
403            ),
404            kpi(
405                "Conflicting",
406                &tally("/queue/tallies/by_disposition/conflicting"),
407            ),
408            kpi(&merged_label(queue), &tally("/throughput/merges")),
409            kpi("Merges per day", &tally("/throughput/merges_per_day")),
410        ]);
411        let assessments = list(queue, "/queue/assessments");
412        let rows = assessments
413            .iter()
414            .take(ROWS * 2)
415            .map(|a| {
416                let number = a
417                    .get("number")
418                    .and_then(Value::as_u64)
419                    .map_or_else(|| "unknown".to_string(), |n| format!("#{n}"));
420                let reasons: Vec<String> = a
421                    .get("reasons")
422                    .and_then(Value::as_array)
423                    .map(|rs| {
424                        rs.iter()
425                            .filter_map(Value::as_str)
426                            .map(str::to_string)
427                            .collect()
428                    })
429                    .unwrap_or_default();
430                let n = a.get("number").and_then(Value::as_u64);
431                el("tr")
432                    .attr_if("id", n.map(|n| format!("pr-{n}")))
433                    .child(el("td").class("mj-ui-mono").text(number))
434                    .child(
435                        el("td").child(
436                            el("a")
437                                .class("mj-ui-row-link")
438                                .attr(
439                                    "href",
440                                    n.map_or_else(
441                                        || super::prs::ROUTE.to_string(),
442                                        |n| format!("{}?pr={n}#pr", super::prs::ROUTE),
443                                    ),
444                                )
445                                .child(el("span").class("mj-ui-cell-title").text(text(a, "title"))),
446                        ),
447                    )
448                    .child(el("td").child(state(a, "disposition")))
449                    .child(el("td").text(value(a, "lane")))
450                    .child(
451                        el("td")
452                            .class("mj-ui-mono mj-ui-dim")
453                            .text(first_reasons(&reasons)),
454                    )
455            })
456            .collect::<Vec<_>>();
457        let observed = queue
458            .read("/queue/observed_at")
459            .text()
460            .map(|t| t.replace('T', " "))
461            .unwrap_or_else(|| "at an unknown time".into());
462        let detail = selected.map(|n| {
463            list(queue, "/queue/assessments")
464                .into_iter()
465                .find(|a| a.get("number").and_then(Value::as_u64) == Some(n))
466                .map_or_else(
467                    || {
468                        nothing(&format!(
469                            "#{n} is not among the open pull requests the queue observed."
470                        ))
471                    },
472                    |a| pull_request_detail(&a),
473                )
474        });
475        el("div")
476            .class("mj-ui-stack")
477            .children(detail)
478            .child(tallies)
479            .child(table(
480                &["PR", "Title", "Disposition", "Lane", "Why"],
481                rows,
482                "No pull request is open.",
483            ))
484            .children(more(
485                (ROWS * 2).min(assessments.len()),
486                assessments.len(),
487                Some(super::prs::ROUTE),
488            ))
489            .child(
490                el("p")
491                    .class("mj-ui-line")
492                    .text(format!("Observed {observed} ({})", day(&observed))),
493            )
494    });
495    panel_to(
496        "Pull-request queue",
497        "What the integration lane would merge, repair or wait for, and why.",
498        Some(super::prs::ROUTE),
499        body,
500        &[queue],
501    )
502}

504/// One pull request as the queue assessed it: what it is, every gate with its verdict, the
505/// evidence each verdict was read from, why it stands where it stands, what would move it,
506/// and the factors of its rank. Read from the queue's own answer, so it costs nothing more.
507fn pull_request_detail(a: &Value) -> El {
508    let s = |k: &str| text(a, k);
509    // an assessment without a number is shown as unknown, never as #0
510    let number = a
511        .get("number")
512        .and_then(Value::as_u64)
513        .map_or_else(|| "unknown".to_string(), |n| n.to_string());
514    let strings = |k: &str| -> Vec<String> {
515        a.get(k)
516            .and_then(Value::as_array)
517            .map(|v| {
518                v.iter()
519                    .filter_map(Value::as_str)
520                    .map(str::to_string)
521                    .collect()
522            })
523            .unwrap_or_default()
524    };
525    let gates = el("ul").class("mj-ui-checks").children(
526        a.get("gates")
527            .and_then(Value::as_array)
528            .cloned()
529            .unwrap_or_default()
530            .iter()
531            .map(|g| {
532                let passed = g.get("passed").and_then(Value::as_bool);
533                el("li")
534                    .class("mj-ui-check")
535                    .child(
536                        el("span")
537                            .class("mj-ui-check-title mj-ui-mono")
538                            .text(text(g, "gate")),
539                    )
540                    .child(word_badge(match passed {
541                        Some(true) => "pass",
542                        Some(false) => "fail",
543                        None => "unknown",
544                    }))
545            }),
546    );
547    let evidence = a
548        .get("evidence")
549        .and_then(Value::as_array)
550        .cloned()
551        .unwrap_or_default()
552        .iter()
553        .map(|e| {
554            el("tr")
555                .child(el("td").class("mj-ui-mono").text(text(e, "kind")))
556                .child(el("td").child(state(e, "status")))
557                .child(el("td").class("mj-ui-dim").text(text(e, "detail")))
558        })
559        .collect::<Vec<_>>();
560    let factors = a
561        .get("rank_factors")
562        .and_then(Value::as_object)
563        .map(|m| {
564            m.iter()
565                .map(|(k, v)| {
566                    el("div")
567                        .class("mj-ui-fact")
568                        .child(el("dt").text(k.replace('_', " ")))
569                        .child(el("dd").class("mj-ui-mono").text(match v {
570                            Value::String(s) => s.clone(),
571                            other => other.to_string(),
572                        }))
573                })
574                .collect::<Vec<_>>()
575        })
576        .unwrap_or_default();
577    el("section")
578        .class("mj-ui-detail")
579        .attr("id", "pr")
580        .attr("aria-label", format!("Pull request #{number}"))
581        .child(
582            el("div")
583                .class("mj-ui-detail-head")
584                .child(el("p").class("mj-ui-mono mj-ui-dim").text(format!(
585                    "#{number} · {} · {} → {}",
586                    s("author"),
587                    s("head_ref"),
588                    s("base_ref")
589                )))
590                .child(el("h3").class("mj-ui-detail-title").text(s("title")))
591                .child(
592                    el("p")
593                        .class("mj-ui-badges")
594                        .child(state(a, "disposition"))
595                        .child(state(a, "lane"))
596                        .child(state(a, "risk")),
597                ),
598        )
599        .child(el("p").class("mj-ui-line").text(format!(
600            "Next: {}",
601            if s("next_action").is_empty() {
602                "nothing to do".into()
603            } else {
604                s("next_action")
605            }
606        )))
607        .child(
608            el("p")
609                .class("mj-ui-mono mj-ui-dim")
610                .text(strings("reasons").join(" · ")),
611        )
612        .child(el("h4").class("mj-ui-panel-title").text("Gates"))
613        .child(gates)
614        .child(el("h4").class("mj-ui-panel-title").text("Evidence"))
615        .child(table(
616            &["Kind", "Verdict", "What it says"],
617            evidence,
618            "No evidence was recorded.",
619        ))
620        .child(el("h4").class("mj-ui-panel-title").text("Paths it authors"))
621        .child(el("p").class("mj-ui-mono mj-ui-dim").text({
622            let paths = strings("authored_paths");
623            if paths.is_empty() {
624                "none".into()
625            } else {
626                paths.join(", ")
627            }
628        }))
629        .child(
630            el("h4")
631                .class("mj-ui-panel-title")
632                .text("Why it ranks where it does"),
633        )
634        .child(el("dl").class("mj-ui-facts").children(factors))
635}
