1//! The `convergence` module: is any of this repository's work held where it can be lost?
2//!
3//! One question, read-only, answered by [`crate::convergence`] — which composes the
4//! worktree topology, the commits no remote reaches and the stash list, and gives one
5//! verdict over them. Declaring it here is what puts it on MCP, on HTTP, in the OpenAPI
6//! document, in the Cockpit and in the generated reference without any of those carrying a
7//! route or a schema of its own.
8//!
9//! The completion invariant asks the same question as `no-stale-topology`, and reads this
10//! answer rather than a second one.
11

25/// The URI under which the convergence of the repository is read as an MCP resource.
26pub const CONVERGENCE_URI: &str = "majordomus://convergence";
27
28fn refused(e: WorktreeError) -> CapabilityError {
29    match e.exit_code() {
30        crate::worktree::EXIT_MISSING => CapabilityError::NotFound(e.to_string()),
31        crate::worktree::EXIT_REFUSED => CapabilityError::Refused(e.to_string()),
32        _ => CapabilityError::Internal(e.to_string()),
33    }
34}
35
36fn convergence_report(ctx: &Context, _: Empty) -> Result<ConvergenceReport, CapabilityError> {
37    convergence::report(Path::new(&ctx.index.repository.root)).map_err(refused)
38}

40/// The `convergence` module: where the work is held.
41pub fn module() -> ModuleDescriptor {
42    module! {
43        id: "convergence",
44        title: "Convergence",
45        description: "Whether every unit of work in this repository is somewhere another worker could find it. A holding is anything that can hold work — a work tree with uncommitted files, a branch with commits, a stash — and each carries a disposition read from git: integrated (the trunk reaches it), published (a remote-tracking ref reaches it), local_only (committed on this disk and nowhere else) or uncommitted. The last two are at risk: they are invisible to every other worker, and a worker that stops is not a rollback. Measured offline, from this checkout alone; whether a pull request exists for a branch is a question for the forge and is not asked here.",
46        stability: Stability::BehaviorallyVerified,
47        capabilities: [
48            capability! {
49                id: "convergence.report",
50                title: "Is any work held where it can be lost?",
51                description: "Every holding of this repository with its disposition, the evidence read for it and the command that would move it out of danger, the tallies per disposition, how many holdings exist on one disk only, and the one verdict over them. Read from git on every call: a commit, a push or an editor's save changes the answer between two calls.",
52                input: Empty,
53                output: ConvergenceReport,
54                stability: Stability::BehaviorallyVerified,
55                exposure: Exposure {
56                    mcp: Some(McpExposure {
57                        tool: Some("majordomus_convergence".into()),
58                        resource: Some(McpResource { uri: CONVERGENCE_URI.into(), name: "convergence".into() }),
59                    }),
60                    http: super::get("/api/v1/convergence"),
61                    cli: Some(CliExposure { path: vec!["convergence".into()] }),
62                },
63                tags: ["convergence", "git", "worktree", "integration"],
64                cache: CachePolicy::Disabled,
65                handler: convergence_report,
66            },
67        ],
68    }
69}

75    /// The declaration is the only place these names exist; every projection derives from
76    /// it. This is the assertion a refactor that dropped an exposure would fail.
77    #[test]
78    fn the_declaration_yields_the_projections_it_claims() {
79        let m = module();
80        assert_eq!(m.id.as_str(), "convergence");
81        assert_eq!(m.capabilities.len(), 1);
82        let entry = &m.capabilities[0];
83        assert_eq!(entry.capability.id.as_str(), "convergence.report");
84        let exposure = &entry.capability.exposure;
85        assert_eq!(
86            exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
87            Some("majordomus_convergence")
88        );
89        assert_eq!(
90            exposure
91                .mcp
92                .as_ref()
93                .and_then(|m| m.resource.as_ref())
94                .map(|r| r.uri.as_str()),
95            Some(CONVERGENCE_URI)
96        );
97        assert_eq!(
98            exposure.http.as_ref().map(|h| h.path.as_str()),
99            Some("/api/v1/convergence")
100        );
101        assert_eq!(
102            exposure.cli.as_ref().map(|c| c.path.clone()),
103            Some(vec!["convergence".to_string()])
104        );
105    }

107    /// A worktree refusal keeps its meaning across the capability boundary: what is not
108    /// there is not found, what git could not do is internal, and anything else is a refusal.
109    #[test]
110    fn a_worktree_error_keeps_its_meaning_as_a_capability_error() {
111        let missing = refused(WorktreeError::NoSuchWorktree {
112            selector: "nope".into(),
113        });
114        assert!(
115            matches!(&missing, CapabilityError::NotFound(m) if m.contains("nope")),
116            "{missing:?}"
117        );
118        let internal = refused(WorktreeError::GitCommandFailed {
119            command: "rev-list --branches".into(),
120            status: "128".into(),
121            stderr: "fatal: bad object".into(),
122        });
123        assert!(
124            matches!(&internal, CapabilityError::Internal(m) if m.contains("bad object")),
125            "{internal:?}"
126        );
127        let refusal = refused(WorktreeError::NotInGitRepository {
128            start: std::path::PathBuf::from("/nowhere"),
129        });
130        assert!(
131            matches!(&refusal, CapabilityError::Refused(m) if m.contains("/nowhere")),
132            "{refusal:?}"
133        );
134    }

136    /// The answer changes outside this process — a commit, a push, an editor's save — so
137    /// caching it would serve a verdict about a repository that no longer exists.
138    #[test]
139    fn the_verdict_is_never_cached() {
140        for e in module().capabilities {
141            assert!(
142                !e.capability.cache.is_enabled(),
143                "{} must not cache",
144                e.capability.id
145            );
146            assert!(
147                e.capability.kind.is_read_only(),
148                "{} writes",
149                e.capability.id
150            );
151        }
152    }
