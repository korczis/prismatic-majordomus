1//! How long CI takes, measured from the recorded forge observation: every check run the forge
2//! reported on an open pull request's head, with when it started and when it completed.
3//!
4//! Nothing here asks the forge or estimates: a duration is `completed_at - started_at` of a run
5//! that has both, and a run without either is counted as unmeasured, never as zero. The
6//! observation holds the latest run of each check on each open head, so this is what CI took
7//! on the heads now open, not a history; the history of merges is `integration::metrics`.
8

17use crate::peers::parse_rfc3339;

19/// How one check takes, across the open heads that ran it.
20#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
21pub struct CheckTiming {
22    /// The check's name, as the forge reports it.
23    pub name: String,
24    /// Runs that started and completed, so were measured.
25    pub runs: u64,
26    /// Of those, how many passed.
27    pub passed: u64,
28    /// Of those, how many failed.
29    pub failed: u64,
30    /// The median duration, in seconds.
31    pub median_seconds: u64,
32    /// The 90th percentile duration (nearest rank), in seconds.
33    pub p90_seconds: u64,
34    /// The longest run, in seconds.
35    pub slowest_seconds: u64,
36}
37
38/// How long CI took on one open pull request's head: from its first check's start to its last
39/// check's completion, and the check that took longest.
40#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
41pub struct HeadTiming {
42    /// The pull request.
43    pub number: u64,
44    /// The head the checks ran on.
45    pub head_sha: String,
46    /// When its first measured check started, RFC 3339.
47    pub started_at: String,
48    /// When its last measured check completed, RFC 3339.
49    pub completed_at: String,
50    /// The wall time between them, in seconds: what a reader waited for CI on this head.
51    pub wall_seconds: u64,
52    /// The measured check that took longest.
53    pub slowest_check: String,
54    /// How long it took, in seconds.
55    pub slowest_seconds: u64,
56    /// Whether any measured check on the head failed.
57    pub failed: bool,
58}

60/// CI timings over every open head the observation holds.
61#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
62pub struct CiTimings {
63    /// When the forge was observed, RFC 3339: the moment these timings describe.
64    pub observed_at: String,
65    /// Check runs measured, across every check and head.
66    pub measured_runs: u64,
67    /// Check runs the observation holds without both a start and a completion: running,
68    /// queued, or recorded before starts were read. Counted, never measured as zero.
69    pub unmeasured_runs: u64,
70    /// Every check, the slowest median first.
71    pub checks: Vec<CheckTiming>,
72    /// Every head with a measured check, the longest wall time first.
73    pub heads: Vec<HeadTiming>,
74}
75
76/// Checks are read slowest median first, then by name: the median is the rank, negated so
77/// that the longest wait leads.
78impl Ordered for CheckTiming {
79    fn order_key(&self) -> OrderKey<'_> {
80        OrderKey::plain(&self.name, &self.name).ranked(slowest_first(self.median_seconds))
81    }
82}
83
84/// Heads are read longest wall time first, then by head.
85impl Ordered for HeadTiming {
86    fn order_key(&self) -> OrderKey<'_> {
87        OrderKey::plain(&self.head_sha, &self.head_sha).ranked(slowest_first(self.wall_seconds))
88    }
89}
90
91fn slowest_first(seconds: u64) -> i64 {
92    i64::try_from(seconds).map_or(i64::MIN, |s| -s)
93}
94
95/// The value at the nearest rank `p` (0..=100) of `sorted`, which is ascending and not empty.
96fn rank(sorted: &[u64], p: u64) -> u64 {
97    let n = sorted.len() as u64;
98    let i = (p * n).div_ceil(100).saturating_sub(1).min(n - 1);
99    sorted[i as usize]
100}

102/// CI timings measured from `obs`.
103///
104/// ```text
105/// // a run with a start and a completion is measured; one still running is counted only
106/// let t = timings(&observation_with(&[("ci", "00:00:00", "00:10:00"), ("suite", "00:00:00", "")]));
107/// assert_eq!((t.measured_runs, t.unmeasured_runs), (1, 1));
108/// assert_eq!(t.checks[0].median_seconds, 600);
109/// ```
110pub fn timings(obs: &ForgeObservation) -> CiTimings {
111    let mut by_check: BTreeMap<String, (Vec<u64>, u64, u64)> = BTreeMap::new();
112    let mut heads = Vec::new();
113    let mut unmeasured = 0u64;
114    for pr in &obs.pull_requests {
115        let mut first: Option<(u64, &str)> = None;
116        let mut last: Option<(u64, &str)> = None;
117        let mut slowest: Option<(u64, &str)> = None;
118        let mut failed = false;
119        for c in pr.checks.iter().filter(|c| c.kind == CheckKind::CheckRun) {
120            let measured = parse_rfc3339(&c.started_at).zip(parse_rfc3339(&c.completed_at));
121            let Some((start, end)) = measured.filter(|(s, e)| e >= s) else {
122                unmeasured += 1;
123                continue;
124            };
125            let took = end - start;
126            let entry = by_check.entry(c.name.clone()).or_default();
127            entry.0.push(took);
128            match c.state {
129                CheckRunState::Passed => entry.1 += 1,
130                CheckRunState::Failed => {
131                    entry.2 += 1;
132                    failed = true;
133                }
134                _ => {}
135            }
136            if first.is_none_or(|(s, _)| start < s) {
137                first = Some((start, c.started_at.as_str()));
138            }
139            if last.is_none_or(|(e, _)| end > e) {
140                last = Some((end, c.completed_at.as_str()));
141            }
142            if slowest.is_none_or(|(t, _)| took > t) {
143                slowest = Some((took, c.name.as_str()));
144            }
145        }
146        if let (Some((s, started)), Some((e, completed)), Some((took, name))) =
147            (first, last, slowest)
148        {
149            heads.push(HeadTiming {
150                number: pr.number,
151                head_sha: pr.head_sha.clone(),
152                started_at: started.to_string(),
153                completed_at: completed.to_string(),
154                wall_seconds: e - s,
155                slowest_check: name.to_string(),
156                slowest_seconds: took,
157                failed,
158            });
159        }
160    }
161    let mut checks: Vec<CheckTiming> = by_check
162        .into_iter()
163        .map(|(name, (took, passed, failed))| {
164            // a sample has one order, `order::numeric`'s; whole seconds are exact in an f64
165            let mut sample: Vec<f64> = took.iter().map(|&t| t as f64).collect();
166            numeric(&mut sample);
167            let took: Vec<u64> = sample.into_iter().map(|t| t as u64).collect();
168            CheckTiming {
169                runs: took.len() as u64,
170                passed,
171                failed,
172                median_seconds: rank(&took, 50),
173                p90_seconds: rank(&took, 90),
174                slowest_seconds: *took.last().unwrap_or(&0),
175                name,
176            }
177        })
178        .collect();
179    canonical(&mut checks);
180    canonical(&mut heads);
181    CiTimings {
182        observed_at: obs.observed_at.clone(),
183        measured_runs: checks.iter().map(|c| c.runs).sum(),
184        unmeasured_runs: unmeasured,
185        checks,
186        heads,
187    }
188}

195    fn check(name: &str, state: CheckRunState, started: &str, completed: &str) -> CheckObservation {
196        CheckObservation {
197            name: name.into(),
198            state,
199            kind: CheckKind::CheckRun,
200            started_at: started.into(),
201            completed_at: completed.into(),
202            ..CheckObservation::default()
203        }
204    }
205
206    fn pr(number: u64, checks: Vec<CheckObservation>) -> PullRequestObservation {
207        let mut p: PullRequestObservation = serde_json::from_value(serde_json::json!({
208            "number": number, "title": "t", "author": "a", "head_ref": "h", "head_sha": format!("{number:040}"),
209            "base_ref": "master", "draft": false, "created_at": "2026-10-01T00:00:00Z",
210            "updated_at": "2026-10-01T00:00:00Z"
211        }))
212        .expect("a pull request");
213        p.checks = checks;
214        p
215    }
216
217    fn observation(prs: Vec<PullRequestObservation>) -> ForgeObservation {
218        let mut o: ForgeObservation = serde_json::from_value(serde_json::json!({
219            "schema": crate::integration::OBSERVATION_SCHEMA, "repository": "o/r", "base": "master",
220            "base_sha": "0".repeat(40), "observed_at": "2026-10-11T12:00:00Z",
221            "required_checks": null, "review_policy": null, "merge_methods": ["merge"], "pull_requests": []
222        }))
223        .expect("an observation");
224        o.pull_requests = prs;
225        o
226    }
227
228    const T: &str = "2026-10-11T10:00:00Z";
229    fn at(min: u64) -> String {
230        let s = parse_rfc3339(T).unwrap() + min * 60;
231        crate::peers::rfc3339(std::time::UNIX_EPOCH + std::time::Duration::from_secs(s))
232    }

234    #[test]
235    fn a_check_is_measured_from_its_runs_and_the_slowest_median_comes_first() {
236        let o = observation(vec![
237            pr(
238                1,
239                vec![
240                    check("rust", CheckRunState::Passed, &at(0), &at(10)),
241                    check("site", CheckRunState::Passed, &at(0), &at(40)),
242                ],
243            ),
244            pr(
245                2,
246                vec![
247                    check("rust", CheckRunState::Failed, &at(5), &at(25)),
248                    check("site", CheckRunState::Passed, &at(5), &at(35)),
249                ],
250            ),
251            pr(
252                3,
253                vec![check("rust", CheckRunState::Passed, &at(0), &at(30))],
254            ),
255        ]);
256        let t = timings(&o);
257        assert_eq!(t.measured_runs, 5);
258        assert_eq!(t.checks[0].name, "site", "the slowest median first");
259        let rust = t.checks.iter().find(|c| c.name == "rust").unwrap();
260        assert_eq!((rust.runs, rust.passed, rust.failed), (3, 2, 1));
261        assert_eq!(rust.median_seconds, 20 * 60);
262        assert_eq!(rust.p90_seconds, 30 * 60);
263        assert_eq!(rust.slowest_seconds, 30 * 60);
264    }

313    #[test]
314    fn heads_lead_with_the_longest_wait_and_equal_medians_read_naturally() {
315        let o = observation(vec![
316            pr(
317                1,
318                vec![check("check-10", CheckRunState::Passed, &at(0), &at(5))],
