1//! The `intents` module:
2//!
3//! The module is `intents` and not `intent` because the registry composes builtin modules and
4//! declarative kinds into one namespace, and `intent` is the **kind** of the records under
5//! `.ai/repo/project/intents/`; the registry refuses the collision, as it did for `session`.
6//! The command line keeps the singular word, `majordomus intent <verb>`.
7//!
8//! What it answers: what must become true, derived against the plan and the evidence.
9//!
10//! Every capability here reads [`crate::intent::Intents`], built on demand from the index,
11//! the plan [`crate::plan::Plan::build`] derives from the same index, and the evidence
12//! ledger. All six are read-only: there is no intent status to write, because the stage is
13//! derived, and no transition, because the plan already owns the lifecycle of the work an
14//! intent is realised by (ADR 0070), and the coverage of its criteria by the plan (ADR 0073).
15//!
16//! ```
17//! use majordomus_cli::capability::builtin::intents;
18//! let m = intents::module();
19//! let ids: Vec<&str> = m.capabilities.iter().map(|e| e.capability.id.as_str()).collect();
20//! assert_eq!(ids, ["intents.list", "intents.record", "intents.validate",
21//!                  "intents.coverage", "intents.preflight", "intents.binding"]);
22//! ```

47/// Every intent, with its derived stage and verdict, and how many there are in each: what
48/// `majordomus intent list`, `GET /api/v1/intents` and the `majordomus_intents` tool all answer
49/// with, out of one derivation.
50///
51/// ```
52/// use majordomus_cli::capability::builtin::intents::IntentList;
53/// let empty = IntentList {
54///     count: 0,
55///     stages: std::collections::BTreeMap::new(),
56///     verdicts: std::collections::BTreeMap::new(),
57///     intents: vec![],
58/// };
59/// // a repository that declares no intent answers the route, with nothing in it
60/// assert_eq!(empty.count, 0);
61/// ```
62#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
63pub struct IntentList {
64    /// How many intents.
65    pub count: usize,
66    /// How many in each stage, keyed by the stage word.
67    pub stages: std::collections::BTreeMap<String, usize>,
68    /// How many with each verdict, keyed by the verdict word.
69    pub verdicts: std::collections::BTreeMap<String, usize>,
70    /// Every intent, in identity order.
71    pub intents: Vec<IntentView>,
72}

74/// What the intent model refuses: whether it is valid, how many intents were examined, and
75/// every finding with its level. Warnings are reported and do not make it invalid.
76///
77/// ```
78/// use majordomus_cli::capability::builtin::intents::IntentValidation;
79/// let v = IntentValidation {
80///     valid: true,
81///     intents: 2,
82///     failures: 0,
83///     warnings: 20,
84///     findings: vec![],
85/// };
86/// // twenty warnings and no failure is a valid model
87/// assert!(v.valid);
88/// ```
89#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
90pub struct IntentValidation {
91    /// True when no finding is a failure. Warnings do not make it false.
92    pub valid: bool,
93    /// How many intents were examined.
94    pub intents: usize,
95    /// How many findings are failures.
96    pub failures: usize,
97    /// How many findings are warnings.
98    pub warnings: usize,
99    /// Every finding.
100    pub findings: Vec<IntentFinding>,
101}

143/// The work to preflight: an issue, or the paths it will touch.
144///
145/// `paths` is one comma-separated string rather than a list because a preflight changes
146/// nothing and is therefore a `GET`, and this repository binds a `GET` input to the query
147/// string.
148///
149/// ```
150/// use majordomus_cli::capability::builtin::intents::IntentPreflightInput;
151/// // an issue, when the work executes one
152/// let by_issue = IntentPreflightInput { issue: Some("I1900".into()), paths: String::new() };
153/// assert!(by_issue.path_list().is_empty());
154/// // or the paths it will touch, when it names no issue yet
155/// let by_paths = IntentPreflightInput {
156///     issue: None,
157///     paths: "apps/majordomus-cli/src,docs".into(),
158/// };
159/// assert_eq!(by_paths.path_list(), ["apps/majordomus-cli/src", "docs"]);
160/// ```
161#[derive(Debug, Clone, PartialEq, Eq, Default, Serialize, Deserialize, JsonSchema)]
162#[serde(deny_unknown_fields)]
163pub struct IntentPreflightInput {
164    /// The issue the work executes, by id. When given, the paths are not consulted.
165    #[serde(default, skip_serializing_if = "Option::is_none")]
166    pub issue: Option<String>,
167    /// Repository-relative paths the work will touch, separated by commas.
168    #[serde(default)]
169    pub paths: String,
170}
171
172impl IntentPreflightInput {
173    /// The paths as a list: trimmed, with the empties a trailing or doubled comma leaves
174    /// dropped, so a query string written by hand resolves the same way as one written by a
175    /// program.
176    ///
177    /// ```
178    /// use majordomus_cli::capability::builtin::intents::IntentPreflightInput;
179    /// let i = IntentPreflightInput { issue: None, paths: " lib, ,docs/X.md".into() };
180    /// assert_eq!(i.path_list(), ["lib", "docs/X.md"]);
181    /// ```
182    pub fn path_list(&self) -> Vec<String> {
183        self.paths
184            .split(',')
185            .map(str::trim)
186            .filter(|s| !s.is_empty())
187            .map(str::to_string)
188            .collect()
189    }
190}
191

194        let plan = Plan::build(ctx.index);

201                },
202            )],
203            None => vec![NamedCase::new(
204                "paths",
205                IntentPreflightInput {
206                    issue: None,
207                    paths: "README.md".into(),
208                },
209            )],
210        }
211    }

214/// The input of `intents.binding`: what a worker names when it asks what its work serves.
215/// Flat strings, because the same value arrives as a query string, as MCP arguments and as
216/// the flags of `majordomus intent binding`.
217///
218/// ```
219/// use majordomus_cli::capability::builtin::intents::IntentBindingInput;
220/// let i = IntentBindingInput { issue: Some("I1900".into()), paths: "lib, docs".into(), ..Default::default() };
221/// let r = i.request();
222/// assert_eq!(r.issue.as_deref(), Some("I1900"));
223/// assert_eq!(r.paths, ["lib", "docs"]);
224/// ```
225#[derive(Debug, Clone, PartialEq, Eq, Default, Serialize, Deserialize, JsonSchema)]
226#[serde(deny_unknown_fields)]
227pub struct IntentBindingInput {
228    /// The issue the work executes, by id.
229    #[serde(default, skip_serializing_if = "Option::is_none")]
230    pub issue: Option<String>,
231    /// The intent the work serves, by id, when the worker names it directly.
232    #[serde(default, skip_serializing_if = "Option::is_none")]
233    pub intent: Option<String>,
234    /// Repository-relative paths the work will touch, separated by commas.
235    #[serde(default)]
236    pub paths: String,
237    /// An exemption class the policy declares under `intent.exemptions`.
238    #[serde(default, skip_serializing_if = "Option::is_none")]
239    pub exemption: Option<String>,
240    /// Why the exemption applies.
241    #[serde(default, skip_serializing_if = "Option::is_none")]
242    pub because: Option<String>,
243}
244
245impl IntentBindingInput {
246    /// The typed request: the paths split at commas, everything else as given.
247    ///
248    /// ```
249    /// use majordomus_cli::capability::builtin::intents::IntentBindingInput;
250    /// assert!(IntentBindingInput::default().request().trimmed().names_nothing());
251    /// ```
252    pub fn request(&self) -> BindingRequest {
253        BindingRequest {
254            issue: self.issue.clone(),
255            intent: self.intent.clone(),
256            paths: self
257                .paths
258                .split(',')
259                .map(str::trim)
260                .filter(|p| !p.is_empty())
261                .map(str::to_string)
262                .collect(),
263            exemption: self.exemption.clone(),
264            because: self.because.clone(),
265        }
266    }
267}

283                    paths: "README.md".into(),
284                    ..Default::default()
285                },
286            )],
287        }
288    }

293fn derived(ctx: &Context) -> Result<(Plan, Intents), CapabilityError> {
294    let plan = Plan::build(&ctx.index);
295    let evidence = RepositoryEvidence::load(&ctx.index)
296        .map_err(|e| CapabilityError::Internal(e.to_string()))?;
297    let intents = Intents::build(&ctx.index, &plan, &evidence);
298    Ok((plan, intents))
299}
300
301fn intent_list(ctx: &Context, _: Empty) -> Result<IntentList, CapabilityError> {
302    let (_, intents) = derived(ctx)?;
303    let mut stages = std::collections::BTreeMap::new();
304    let mut verdicts = std::collections::BTreeMap::new();
305    for i in &intents.intents {
306        *stages.entry(i.stage.as_str().to_string()).or_insert(0) += 1;
307        *verdicts
308            .entry(i.verdict.state.as_str().to_string())
309            .or_insert(0) += 1;
310    }
311    Ok(IntentList {
312        count: intents.intents.len(),
313        stages,
314        verdicts,
315        intents: intents.intents,
316    })
317}
318
319fn intent_record(ctx: &Context, input: IntentRecordInput) -> Result<IntentView, CapabilityError> {
320    let (_, intents) = derived(ctx)?;
321    intents.intent(&input.id).cloned().ok_or_else(|| {
322        CapabilityError::NotFound(format!(
323            "no intent '{}'; `intents.list` names every one this repository holds",
324            input.id
325        ))
326    })
327}
328
329fn intent_validate(ctx: &Context, _: Empty) -> Result<IntentValidation, CapabilityError> {
330    let (_, intents) = derived(ctx)?;
331    Ok(IntentValidation {
332        valid: intents.failures() == 0,
333        intents: intents.intents.len(),
334        failures: intents.failures(),
335        warnings: intents.warnings(),
336        findings: intents.findings,
337    })
338}

340fn intent_coverage(ctx: &Context, _: Empty) -> Result<IntentCoverage, CapabilityError> {
341    let plan = Plan::build(&ctx.index);
342    Ok(Intents::coverage(&ctx.index, &plan))
343}

345fn intent_preflight(
346    ctx: &Context,
347    input: IntentPreflightInput,
348) -> Result<IntentPreflight, CapabilityError> {
349    let paths = input.path_list();
350    let issue = input
351        .issue
352        .as_deref()
353        .map(str::trim)
354        .filter(|s| !s.is_empty());
355    if issue.is_none() && paths.is_empty() {
356        return Err(CapabilityError::InvalidInput(
357            "name the issue the work executes, or the paths it will touch".into(),
358        ));
359    }
360    let (plan, intents) = derived(ctx)?;
361    Ok(intents.preflight(
362        &plan,
363        &GapRecord::all(&ctx.index),
364        &CritiqueRecord::all(&ctx.index),
365        issue,
366        &paths,
367    ))
368}

370/// The policy's `intent:` block, or its default when the repository or the policy cannot be
371/// read: no exemption class is then declared, so none can be given, which is the refusal an
372/// unreadable policy owes rather than a class invented for it.
373fn intent_policy(ctx: &Context) -> crate::policy::IntentPolicy {
374    let root = std::path::PathBuf::from(&ctx.index.repository.root);
375    crate::repository::Repository::open(&root)
376        .ok()
377        .and_then(|repo| crate::policy::LoadedPolicy::load(&repo).ok())
378        .map(|loaded| loaded.policy.intent)
379        .unwrap_or_default()
380}
381
382fn intent_binding(
383    ctx: &Context,
384    input: IntentBindingInput,
385) -> Result<IntentBinding, CapabilityError> {
386    derived(ctx).map(|(plan, intents)| {
387        bind(
388            &intents,
389            &plan,
390            &GapRecord::all(&ctx.index),
391            &CritiqueRecord::all(&ctx.index),
392            &intent_policy(ctx),
393            input.request(),
394        )
395    })
396}
