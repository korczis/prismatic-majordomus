1//! Stopping an offloaded run: its whole process group, on its host, through the offload path
2//! and nowhere else (I3956).
3//!
4//! On 2026-10-22 the owner of a runaway remote `cargo test --doc` killed its cargo and its
5//! shell by hand, and a `rustc` under them ran on for 50 minutes with init as its parent,
6//! because nothing stopped the group. A run now starts in a process group of its own and
7//! records it ([`super::owner::RunRecord`]); `majordomus offload stop <tag>` reads that record
8//! on the host, decides here whether this caller may stop it, and signals the group — `TERM`,
9//! then `KILL` to whatever is left after the grace period — never one process of it.
10//!
11//! What it refuses:
12//!
13//! - a run whose record names another session, or names none, unless `--force`: a session
14//!   does not stop another session's work by accident;
15//! - a record without a process group (a host with neither `setsid` nor Python could not
16//!   start one): there is no group to stop, and one process of it is not the run;
17//! - on a host with `/proc`, a group none of whose processes carries this run's tag in its
18//!   environment ([`super::owner::RUN_ENV`]): the host has handed that group id to someone
19//!   else since the record was written, and it is not ours to signal.
20//!
21//! A group that is already empty is reported as gone and its record removed: a record that
22//! outlived its run is evidence until somebody looks, and this is somebody looking.
23

34/// The record names another session (or none) and `--force` was not given.
35pub const EXIT_NOT_OWNER: u8 = 13;

37/// How a stop is asked for.
38#[derive(Debug, Clone, Default)]
39pub struct StopRequest {
40    /// The run's tag, as the derive printed it.
41    pub tag: String,
42    /// A declared host id or address; absent is the declaration's only host.
43    pub host: Option<String>,
44    /// Stop it although its record names another session, or none.
45    pub force: bool,
46    /// Seconds between `TERM` and `KILL`.
47    pub grace_seconds: u64,
48}
49
50/// What a stop did.
51#[derive(Debug, Clone, Serialize)]
52pub struct StopReport {
53    /// The host it ran on.
54    pub host: String,
55    /// The run's record, as the host kept it.
56    pub record: RunRecord,
57    /// How many processes the group had when the stop began.
58    pub members: usize,
59    /// How the group was known to be this run's: `environ` (a member carries the run's tag),
60    /// `record` (a host without `/proc`: the record alone), or `gone` (nothing to verify).
61    pub verified: String,
62    /// The signals sent to the group, in order.
63    pub signals: Vec<String>,
64    /// Whether the record was removed here; otherwise the run's own shell removes it.
65    pub record_removed: bool,
66}

75/// Whether `tag` can name a run: what the derive makes one of, ten to forty letters and
76/// digits, so a tag is never a path on the host.
77pub fn valid_tag(tag: &str) -> bool {
78    (10..=40).contains(&tag.len()) && tag.chars().all(|c| c.is_ascii_alphanumeric())
79}

81/// Whether `caller` may stop the run `record` describes: the record names the caller's own
82/// session, or `force` says to stop it anyway. The refusal names the owner, so the person
83/// knows whom to ask.
84pub fn may_stop(record: &RunRecord, caller: &RunOwner, force: bool) -> Result<(), Refusal> {
85    let owner = record.owner.session.as_deref();
86    if force || (owner.is_some() && owner == caller.session.as_deref()) {
87        return Ok(());
88    }
89    let whose = match (owner, &record.owner.peer, &record.owner.issue) {
90        (Some(s), peer, issue) => format!(
91            "session {s}{}{}",
92            peer.as_deref()
93                .map(|p| format!(", peer {p}"))
94                .unwrap_or_default(),
95            issue
96                .as_deref()
97                .map(|i| format!(", issue {i}"))
98                .unwrap_or_default()
99        ),
100        (None, _, _) => "no session (a run started outside a provider)".to_string(),
101    };
102    Err(refuse(
103        EXIT_NOT_OWNER,
104        format!(
105            "run {} on {} belongs to {whose}, and this caller is {}; ask its owner, or pass \
106             --force to stop it anyway",
107            record.tag,
108            record.host,
109            caller.session.as_deref().unwrap_or("no session")
110        ),
111    ))
112}

114/// The host a stop goes to: the one named, or the declaration's only host.
115fn pick(d: &OffloadDeclaration, named: Option<&str>) -> Result<OffloadHost, Refusal> {
116    let ids = || {
117        d.hosts
118            .iter()
119            .map(|h| h.id.as_str())
120            .collect::<Vec<_>>()
121            .join(", ")
122    };
123    match named {
124        Some(name) => d.host(name).cloned().ok_or_else(|| {
125            refuse(
126                EXIT_UNAVAILABLE,
127                format!("'{name}' is not a declared host; declared: {}", ids()),
128            )
129        }),
130        None if d.hosts.len() == 1 => Ok(d.hosts[0].clone()),
131        None => Err(refuse(
132            EXIT_PRECONDITION,
133            format!(
134                "{} hosts are declared; name the one the run is on with --host ({})",
135                d.hosts.len(),
136                ids()
137            ),
138        )),
139    }
140}

142/// Read a run record on the host. Arguments: runs directory, tag. Exit 3 when there is none.
143const READ: &str = r#"runs=$1; tag=$2
144case $runs in /*) ;; *) runs=$HOME/$runs ;; esac
145[ -f "$runs/$tag.run" ] || exit 3
146cat "$runs/$tag.run""#;
147
148/// Stop a run's process group on the host. Arguments: runs directory, tag, process group,
149/// the run's shell pid, grace seconds, who stops it. Prints `members <n>`, then either
150/// `record removed` (an empty group) or `verified environ|record`, `signalled TERM`,
151/// possibly `signalled KILL`, `stopped`, and `record removed|kept`. Exit 4 when the group is
152/// not this run's (no member carries its tag, on a host with `/proc`), 5 when members
153/// survive `KILL`.
154///
155/// The marker `<runs>/<tag>.stopped` tells the run's own shell, which sees its derive end
156/// with a signal, that the run was stopped rather than failed.
157const STOP: &str = r#"runs=$1; tag=$2; pg=$3; pid=$4; grace=$5; by=$6
158case $runs in /*) ;; *) runs=$HOME/$runs ;; esac
159record="$runs/$tag.run"
160members() { ps -e -o pid= -o pgid= 2>/dev/null | awk -v g="$pg" '$2 == g { print $1 }'; }
161removed() {
162  if kill -0 "$pid" 2>/dev/null; then echo "record kept"
163  else rm -f "$record" "$runs/$tag.stopped"; echo "record removed"; fi
164}
165m=$(members)
166if [ -z "$m" ]; then echo "members 0"; removed; exit 0; fi
167echo "members $(printf '%s\n' $m | wc -l | tr -d ' ')"
168if [ -d /proc/self ]; then
169  ours=""
170  for p in $m; do
171    if tr '\0' '\n' < "/proc/$p/environ" 2>/dev/null | grep -qx "MAJORDOMUS_OWNER_RUN=$tag"; then
172      ours=1; break
173    fi
174  done
175  [ -n "$ours" ] || { echo "reused"; exit 4; }
176  echo "verified environ"
177else
178  echo "verified record"
179fi
180printf '%s\n' "$by" > "$runs/$tag.stopped"
181kill -TERM -"$pg" 2>/dev/null
182echo "signalled TERM"
183i=0
184while [ -n "$(members)" ] && [ "$i" -lt "$grace" ]; do sleep 1; i=$((i+1)); done
185if [ -n "$(members)" ]; then kill -KILL -"$pg" 2>/dev/null; echo "signalled KILL"; sleep 1; fi
186left=$(members)
187if [ -n "$left" ]; then echo "survivors" $left; exit 5; fi
188echo "stopped"
189removed"#;

191/// Stop the run `request` names, on its host, as `caller`. Progress goes to `log`.
192pub fn stop(
193    d: &OffloadDeclaration,
194    request: &StopRequest,
195    caller: &RunOwner,
196    log: &mut dyn Write,
197) -> Result<StopReport, Refusal> {
198    let host = pick(d, request.host.as_deref())?;
199    stop_on(
200        &remote_of(&host),
201        &host.id,
202        &runs_directory(&d.queue.directory),
203        request,
204        caller,
205        log,
206    )
207}

209/// [`stop`] on a host already reached: what a test drives with this machine as the host.
210pub(crate) fn stop_on(
211    remote: &Remote,
212    host: &str,
213    runs: &str,
214    request: &StopRequest,
215    caller: &RunOwner,
216    log: &mut dyn Write,
217) -> Result<StopReport, Refusal> {
218    let tag = request.tag.as_str();
219    if !valid_tag(tag) {
220        return Err(refuse(
221            EXIT_PRECONDITION,
222            format!("'{tag}' is not a run tag (ten to forty letters and digits)"),
223        ));
224    }
225    let unreachable = |e: std::io::Error| refuse(EXIT_UNAVAILABLE, format!("{host}: {e}"));
226    let read = remote.run(READ, &[runs, tag], None).map_err(unreachable)?;
227    match read.code {
228        Some(0) => {}
229        Some(3) => {
230            return Err(refuse(
231                EXIT_UNAVAILABLE,
232                format!(
233                "{host} holds no run record {runs}/{tag}.run: the run ended, or never ran there"
234            ),
235            ))
236        }
237        other => {
238            return Err(refuse(
239                EXIT_UNAVAILABLE,
240                format!(
241                    "{host}: the run record could not be read ({other:?}): {}",
242                    read.stderr.trim()
243                ),
244            ))
245        }
246    }
247    let record = RunRecord::parse(&read.stdout)
248        .map_err(|e| refuse(EXIT_UNAVAILABLE, format!("{host}: {runs}/{tag}.run: {e}")))?;
249    may_stop(&record, caller, request.force)?;
250    let Some(pgid) = record.pgid.filter(|p| *p > 1) else {
251        return Err(refuse(
252            EXIT_UNAVAILABLE,
253            format!(
254                "run {tag} on {host} has no process group of its own (its host had neither \
255                 setsid nor python3); its shell is pid {}",
256                record.pid
257            ),
258        ));
259    };
260    let _ = writeln!(
261        log,
262        "offload stop: {tag} on {host}, process group {pgid}, owner {}",
263        record.owner.session.as_deref().unwrap_or("(unnamed)")
264    );
265    let by = caller.session.clone().unwrap_or_else(|| "unnamed".into());
266    let out = remote
267        .run(
268            STOP,
269            &[
270                runs,
271                tag,
272                &pgid.to_string(),
273                &record.pid.to_string(),
274                &request.grace_seconds.to_string(),
275                &by,
276            ],
277            None,
278        )
279        .map_err(unreachable)?;
280    let _ = log.write_all(out.stderr.as_bytes());
281    let report = report_of(host, record, &out.stdout);
282    match out.code {
283        Some(0) => Ok(report),
284        Some(4) => Err(refuse(
285            EXIT_UNAVAILABLE,
286            format!(
287                "process group {pgid} on {host} is not run {tag}'s any more: none of its \
288                 processes carries {}={tag}; nothing was signalled",
289                super::owner::RUN_ENV
290            ),
291        )),
292        Some(5) => Err(refuse(
293            EXIT_UNAVAILABLE,
294            format!(
295                "process group {pgid} on {host} survived TERM and KILL: {}",
296                out.stdout.trim()
297            ),
298        )),
299        other => Err(refuse(
300            EXIT_UNAVAILABLE,
301            format!(
302                "{host}: the stop ended with {other:?}: {}",
303                out.stderr.trim()
304            ),
305        )),
306    }
307}

309/// The report the stop script's lines make.
310fn report_of(host: &str, record: RunRecord, stdout: &str) -> StopReport {
311    let mut report = StopReport {
312        host: host.to_string(),
313        record,
314        members: 0,
315        verified: "gone".into(),
316        signals: Vec::new(),
317        record_removed: false,
318    };
319    for line in stdout.lines() {
320        if let Some(n) = line.strip_prefix("members ") {
321            report.members = n.trim().parse().unwrap_or(0);
322        } else if let Some(v) = line.strip_prefix("verified ") {
323            report.verified = v.trim().to_string();
324        } else if let Some(s) = line.strip_prefix("signalled ") {
325            report.signals.push(s.trim().to_string());
326        } else if line == "record removed" {
327            report.record_removed = true;
328        }
329    }
330    report
331}

593        let tmp = tempfile::tempdir().unwrap();
594        let at = |p: &str| tmp.path().join(p);
595        let tag = "fedcba9876";
596        for d in ["lock", "wt/offload-fedcba9876", "target"] {
597            std::fs::create_dir_all(at(d)).unwrap();
598        }
599        let derive = at("derive");
600        std::fs::write(
601            &derive,
602            format!(
603                "#!/bin/sh\nsleep 300 &\necho $! > {gc}\nwait\n",
604                gc = at("grandchild").display()
605            ),
606        )
607        .unwrap();
608        use std::os::unix::fs::PermissionsExt;
609        std::fs::set_permissions(&derive, std::fs::Permissions::from_mode(0o755)).unwrap();
610        let runs = at("runs");
611        let run = super::super::run::derive_command(&super::super::run::DeriveArgs {
612            lock: at("lock").to_str().unwrap(),
613            worktrees: at("wt").to_str().unwrap(),
614            tag,
615            shared: at("target").to_str().unwrap(),
616            derive: derive.to_str().unwrap(),
617            spool: at("spool").to_str().unwrap(),
618            prefix: "",
619            observe: "",
620            held: false,
621            runs: runs.to_str().unwrap(),
622            owner: &me("claude-code-a").args(),
623            touch: &[],
624        });
625        let shell = std::process::Command::new("sh")
626            .arg("-c")
627            .arg(run.0)
628            .arg("sh")
629            .args(&run.1)
630            .stdin(std::process::Stdio::null())
631            .stdout(std::process::Stdio::piped())
632            .stderr(std::process::Stdio::piped())
633            .spawn()
634            .unwrap();
635        let record = runs.join(format!("{tag}.run"));
636        let deadline = Instant::now() + Duration::from_secs(30);
637        while !(record.exists() && at("grandchild").exists()) {
638            assert!(Instant::now() < deadline, "the run never recorded itself");
639            std::thread::sleep(Duration::from_millis(50));
640        }
641        let grandchild: i32 = std::fs::read_to_string(at("grandchild"))
642            .unwrap()
643            .trim()
644            .parse()
645            .unwrap();
646        let req = StopRequest {
647            tag: tag.into(),
648            grace_seconds: 5,
649            ..StopRequest::default()
650        };
651        let stopped = stop_on(
652            &Remote::local(),
653            "local",
654            runs.to_str().unwrap(),
655            &req,
656            &me("claude-code-a"),
657            &mut Vec::new(),
658        );
659        let out = shell.wait_with_output().unwrap();
660        let text = String::from_utf8_lossy(&out.stdout);
661        let r = match stopped {
662            Ok(r) => r,
663            Err(e) => {
664                // SAFETY: the pid was written by this test's own derive a moment ago
665                unsafe { libc::kill(grandchild, libc::SIGKILL) };
666                panic!("{}: {text}", e.reason);
667            }
668        };
669        assert!(r.members >= 2, "the derive and its grandchild: {r:?}");
670        assert_eq!(r.signals.first().map(String::as_str), Some("TERM"));
671        assert!(
672            ["environ", "record"].contains(&r.verified.as_str()),
673            "{r:?}"
674        );
675        assert_eq!(out.status.code(), Some(14), "{text}");
676        assert!(String::from_utf8_lossy(&out.stderr).contains("by session claude-code-a"));
677        // SAFETY: kill with signal 0 only asks whether the pid exists
678        assert_ne!(
679            unsafe { libc::kill(grandchild, 0) },
680            0,
681            "the grandchild outlived its group"
682        );
683        assert!(!record.exists(), "the record ends with the run");
684        assert!(!runs.join(format!("{tag}.stopped")).exists());
685    }

687    /// A host stand-in that answers the record read with a record of `claude-code-a`, and
688    /// the stop with `stdout`, `host: said` on stderr and exit `code`, keeping its command in
689    /// `asked`; with `vanish`, it removes itself after the read, so the stop cannot start.
690    fn host(dir: &std::path::Path, stdout: &str, code: i32, vanish: bool) -> std::path::PathBuf {
691        use std::os::unix::fs::PermissionsExt;
692        let p = dir.join("host");
693        let asked = dir.join("asked");
694        let gone = if vanish { "rm -f \"$0\"" } else { ":" };
695        let body = format!(
696            "#!/bin/sh\ncase \"$1\" in\n  *members*) printf '%s\\n' \"$1\" > '{asked}'; \
697             printf '{stdout}'; echo 'host: said' >&2; exit {code} ;;\nesac\n{gone}\n\
698             printf 'majordomus-offload-run/v1\\ntag 0123456789\\npid 7\\npgid 8\\nhost h\\n\
699             owner_session claude-code-a\\n'\n",
700            asked = asked.display()
701        );
702        std::fs::write(&p, body).unwrap();
703        std::fs::set_permissions(&p, std::fs::Permissions::from_mode(0o755)).unwrap();
704        p
705    }

707    /// What the host's stop says is reported as it is: a group that is no longer the run's
708    /// (4), members that survived KILL (5), any other end, and a host that cannot be reached
709    /// for the stop after it answered the read. A forced stop by a caller with no session
710    /// is made by `unnamed`.
711    #[test]
712    fn a_stop_the_host_refuses_or_cannot_finish_is_reported_as_the_host_said() {
713        let tmp = tempfile::tempdir().unwrap();
714        let req = StopRequest {
715            tag: "0123456789".into(),
716            force: true,
717            grace_seconds: 1,
718            ..StopRequest::default()
719        };
720        let stop_through = |stdout: &str, code: i32, vanish: bool| {
721            let remote = Remote::through(&host(tmp.path(), stdout, code, vanish));
722            let mut log = Vec::new();
723            let r = stop_on(
724                &remote,
725                "lundra",
726                "/q.runs",
727                &req,
728                &RunOwner::default(),
729                &mut log,
730            );
731            (r, String::from_utf8(log).unwrap())
732        };
733        let (r, log) = stop_through("members 2\\nreused\\n", 4, false);
734        let e = r.unwrap_err();
735        assert_eq!(e.code, EXIT_UNAVAILABLE);
736        assert!(
737            e.reason
738                .contains("process group 8 on lundra is not run 0123456789's any more")
739                && e.reason.contains("MAJORDOMUS_OWNER_RUN=0123456789")
740                && e.reason.contains("nothing was signalled"),
741            "{}",
742            e.reason
743        );
744        assert!(
745            log.contains("owner claude-code-a") && log.contains("host: said"),
746            "{log}"
747        );
748        let asked = std::fs::read_to_string(tmp.path().join("asked")).unwrap();
749        assert!(
750            asked.contains("unnamed"),
751            "a caller with no session: {asked}"
752        );
753
754        let (r, _) = stop_through("members 2\\nsurvivors 8 9\\n", 5, false);
755        let e = r.unwrap_err();
756        assert_eq!(e.code, EXIT_UNAVAILABLE);
757        assert!(
758            e.reason.contains(
759                "process group 8 on lundra survived TERM and KILL: members 2\nsurvivors 8 9"
760            ),
761            "{}",
762            e.reason
763        );
764
765        let (r, _) = stop_through("", 9, false);
766        let e = r.unwrap_err();
767        assert_eq!(e.code, EXIT_UNAVAILABLE);
768        assert_eq!(e.reason, "lundra: the stop ended with Some(9): host: said");
769
770        let (r, _) = stop_through("", 0, true);
771        let e = r.unwrap_err();
772        assert_eq!(e.code, EXIT_UNAVAILABLE);
773        assert!(e.reason.starts_with("lundra: "), "{}", e.reason);
774        assert!(
775            !e.reason.contains("ended with"),
776            "the stop never ran: {}",
777            e.reason
778        );
779    }
