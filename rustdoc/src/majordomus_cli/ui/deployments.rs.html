17/// The route of the screen.
18pub const ROUTE: &str = "/ui/deployments";
19
20/// Where one declared deployment is: its row on this screen.
21pub fn href(id: &str) -> String {
22    format!("{ROUTE}#deployment-{}", anchor(id))
23}
24
25/// One declared deployment as a row: its id leading to its row here, its declaration's
26/// technical view, the application it runs and its state. The CI/CD screen draws the same row.
27pub fn summary_row(ctx: &Context, d: &Value) -> El {
28    let id = text(d, "id");
29    el("tr")
30        .child(
31            el("td")
32                .child(
33                    el("a")
34                        .class("mj-ui-row-link")
35                        .attr("href", href(&id))
36                        .child(
37                            el("span")
38                                .class("mj-ui-mono mj-ui-dim")
39                                .text(value(d, "id")),
40                        )
41                        .child(el("span").class("mj-ui-cell-title").text(value(d, "title"))),
42                )
43                .child(object_if(
44                    ctx,
45                    &format!("majordomus://deployment/{id}"),
46                    &text(d, "file"),
47                )),
48        )
49        .child(el("td").class("mj-ui-mono").text(value(d, "application")))
50        .child(el("td").child(state(d, "status")))
51}

53/// The route of the delivery panel.
54pub const DELIVERY_PANEL: &str = "/ui/panels/delivery";
55
56const DEPLOYMENTS: Source = Source::of("deploy.list");
57const DELIVERY: Source = Source::of("delivery.report");
58
59/// The Deployments screen.
60pub fn screen(ctx: &Context) -> View {
61    let answers = ask_all(ctx, &[DEPLOYMENTS]);
62    let deployments = &answers[&DEPLOYMENTS];
63
64    let top = hero(
65        "server",
66        "green",
67        "Deployments",
68        "What is live, and whether it is what was built.",
69        "Every declared deployment, and — asked on its own — whether the public site serves the trunk and how far each feature has been delivered.",
70        vec![
71            kpi("Deployments", &deployments.read("/count")),
72            kpi_count("Declared", &deployments.read("/deployments")),
73        ],
74    );
75
76    let declared = unanswered(deployments).unwrap_or_else(|| {
77        let rows = list(deployments, "/deployments")
78            .iter()
79            .map(|d| {
80                summary_row(ctx, d).attr("id", format!("deployment-{}", anchor(&text(d, "id"))))
81            })
82            .collect::<Vec<_>>();
83        table(
84            &["Deployment", "Application", "State"],
85            rows,
86            "No deployment is declared.",
87        )
88    });
89
90    let placeholder = el("section")
91        .class("mj-ui-panel")
92        .attr("aria-label", "Delivery")
93        .attr("data-panel-src", DELIVERY_PANEL)
94        .attr("aria-busy", "false")
95        .child(
96            el("header").class("mj-ui-panel-head").child(
97                el("div")
98                    .class("mj-ui-panel-heading")
99                    .child(el("h2").class("mj-ui-panel-title").text("Delivery"))
100                    .child(el("p").class("mj-ui-panel-sub").text(
101                        "Asked on its own: delivery.report probes the public site and judges every feature, which would hold this screen back.",
102                    )),
103            ),
104        )
105        .child(
106            el("p").class("mj-ui-panel-body").child(
107                el("a").class("mj-ui-link").attr("href", DELIVERY_PANEL).text("Open the delivery report"),
108            ),
109        );
110
111    let grid = el("div")
112        .class("mj-ui-grid")
113        .child(wide(panel(
114            "Declared deployments",
115            "Where the product is declared to run.",
116            declared,
117            &[deployments],
118        )))
119        .child(wide(placeholder));
120
121    View::new(
122        "Deployments",
123        el("div").class("mj-ui-stack").child(top).child(grid),
124    )
125}

127/// The delivery report as a panel: the publication's verification and every feature's stage.
128pub fn delivery_panel(ctx: &Context) -> View {
129    View::new("Delivery", delivery(ctx, &ask(ctx, DELIVERY)))
130}
131
132fn delivery(ctx: &Context, report: &Answer) -> El {
133    let body = unanswered(report).unwrap_or_else(|| {
134        let tallies = strip(vec![
135            kpi("Features", &report.read("/tallies/features")),
136            kpi("On the trunk", &report.read("/tallies/on_master")),
137            kpi("Deployed", &report.read("/tallies/deployed")),
138            kpi(
139                "Publicly verified",
140                &report.read("/tallies/publicly_verified"),
141            ),
142        ]);
143        let url = report.read("/publication/url").text();
144        let publication = el("p")
145            .class("mj-ui-line")
146            .child(state(
147                report
148                    .ok()
149                    .and_then(|v| v.pointer("/publication/verification"))
150                    .unwrap_or(&Value::Null),
151                "state",
152            ))
153            .text(" ")
154            .child(match &url {
155                // the address the report probed, as it gave it
156                Some(u) => link(u.clone(), u.clone()),
157                None => el("span").text("no public address"),
158            })
159            .text(format!(
160                " — {}",
161                report
162                    .read("/publication/verification/detail")
163                    .text()
164                    .unwrap_or_else(|| "none".into())
165            ));
166        let rows = list(report, "/features")
167            .iter()
168            .map(|f| {
169                let blocking: Vec<String> = f
170                    .pointer("/delivery/blocking")
171                    .and_then(Value::as_array)
172                    .map(|b| {
173                        b.iter()
174                            .filter_map(Value::as_str)
175                            .map(str::to_string)
176                            .collect()
177                    })
178                    .unwrap_or_default();
179                let id = text(f, "id");
180                el("tr")
181                    .child(
182                        el("td")
183                            .child(object_if(ctx, &format!("majordomus://feature/{id}"), &id))
184                            .child(el("span").class("mj-ui-cell-title").text(value(f, "title"))),
185                    )
186                    .child(
187                        el("td").child(state(f.get("delivery").unwrap_or(&Value::Null), "stage")),
188                    )
189                    .child(
190                        el("td").child(state(f.get("delivery").unwrap_or(&Value::Null), "state")),
191                    )
192                    .child(el("td").child(if blocking.is_empty() {
193                        none()
194                    } else {
195                        el("span")
196                            .class("mj-ui-mono mj-ui-dim")
197                            .text(blocking.join(", "))
198                    }))
199            })
200            .collect::<Vec<_>>();
201        el("div")
202            .class("mj-ui-stack")
203            .child(tallies)
204            .child(publication)
205            .child(table(
206                &["Feature", "Stage", "Delivery", "What blocks it"],
207                rows,
208                "No feature is declared.",
209            ))
210    });
211    panel_to(
212        "Delivery",
213        "Whether the public site serves the trunk, and how far each feature has been delivered.",
214        None,
215        body,
216        &[report],
217    )
218}

227    /// The sources a furnished repository adds to the synthetic one's: the objects the
228    /// screens draw rows from.
229    const FURNISHED_SOURCES: &str = "
230  - id: deployment
231    kind: deployment
232    discovery: vcs
233    pathspec: ':(glob).ai/repo/deployments/*.yaml'
234    required: false
235  - id: feature
236    kind: feature
237    discovery: vcs
238    pathspec: ':(glob).ai/repo/features/*.md'
239    required: false
240  - id: claims
241    kind: claim
242    discovery: vcs
243    pathspec: ':(glob)docs/CLAIMS.yaml'
244    required: false
245";

247    /// A synthetic repository furnished with what the screens draw rows from: this
248    /// repository's own deployment, one feature, two claims (one of them a guarantee no
249    /// test supports, so the report has a finding) and a CI model with a gate of each kind
250    /// of selection.
251    pub(crate) fn furnished() -> SyntheticRepository {
252        let repo = SyntheticRepository::small().expect("a synthetic repository");
253        let write = |rel: &str, body: &str| {
254            let path = repo.root().join(rel);
255            std::fs::create_dir_all(path.parent().expect("a parent")).expect("a directory");
256            std::fs::write(path, body).expect("a file");
257        };
258        let sources = repo.root().join(".ai/repo/knowledge/sources.yaml");
259        let mut text = std::fs::read_to_string(&sources).expect("the sources");
260        text.push_str(FURNISHED_SOURCES);
261        std::fs::write(&sources, text).expect("the sources");
262        let deployment = std::fs::read_to_string(concat!(
263            env!("CARGO_MANIFEST_DIR"),
264            "/../../.ai/repo/deployments/majordomus.yaml"
265        ))
266        .expect("this repository's deployment");
267        write(".ai/repo/deployments/majordomus.yaml", &deployment);
268        write(
269            ".ai/repo/features/synthetic.md",
270            "---\nschema: feature/v1\nid: synthetic\nkind: feature\ntitle: A synthetic feature\n\
271             headline: A feature that exists for a test.\n\
272             summary: What a synthetic feature does.\nstatus: stable\n---\n\n## What it does\n\nIt is a fixture.\n",
273        );
274        write(
275            "docs/CLAIMS.yaml",
276            "version: 1\nclaims:\n  - id: synthetic-guarantee\n    claim: A guarantee no test supports\n\
277             \x20   source: docs/DOC_0.md\n    status: guaranteed\n\
278             \x20 - id: synthetic-advice\n    claim: An advisory claim\n    source: docs/DOC_0.md\n\
279             \x20   test: test/cases/1_synthetic.sh\n    status: advisory\n",
280        );
281        write(
282            ".ai/repo/ci/gates.yaml",
283            "version: 1\ngates:\n  - id: every-plan\n    job: structure\n    always: true\n\
284             \x20   runs: scripts/ci/every-plan\n    summary: runs in every plan\n\
285             \x20 - id: when-asked\n    job: bench\n    on-demand: true\n    runs: scripts/ci/when-asked\n\
286             \x20   summary: runs when asked\n  - id: by-path\n    job: rust\n    runs: scripts/ci/by-path\n\
287             \x20   summary: runs when its paths change\nclasses:\n  - id: rust\n    paths: [lib/**]\n\
288             \x20   gates: [by-path]\n",
289        );
290        repo
291    }

293    /// An answer as a capability would give it, built here so every branch of a panel can
294    /// be drawn from rows the fixture repository does not hold.
295    pub(crate) fn answer(source: Source, value: Result<Value, String>) -> Answer {
296        Answer {
297            source,
298            route: None,
299            exposure: Default::default(),
300            value,
301            took: Duration::ZERO,
302        }
303    }

305    /// Every declared deployment is a row: its identity, its title, the application it
306    /// ships and its own status word.
307    #[test]
308    fn every_declared_deployment_is_a_row_with_its_own_status() {
309        let repo = furnished();
310        let ctx = repo.context().expect("a context");
311        let view = screen(&ctx);
312        let html = view.main.render();
313        assert_eq!(view.status, 200);
314        assert!(
315            html.contains("The Majordomus layer, served from its container image"),
316            "{html}"
317        );
318        assert!(html.contains(">majordomus<"), "{html}");
319        assert!(html.contains("declared"), "{html}");
320        // the row is where a link to the deployment leads, and its declaration is the
321        // technical view of the object the index holds
322        assert!(html.contains("id=\"deployment-majordomus\""), "{html}");
323        assert!(
324            html.contains(&format!("href=\"{}\"", href("majordomus"))),
325            "{html}"
326        );
327        assert!(
328            html.contains("/cockpit/object?uri=majordomus%3A%2F%2Fdeployment%2Fmajordomus"),
329            "{html}"
330        );
331        assert!(!html.contains("No deployment is declared."), "{html}");
332        assert!(
333            html.contains(&format!("data-panel-src=\"{DELIVERY_PANEL}\"")),
334            "the slow report is a panel of its own: {html}"
335        );
336    }

355    /// A report with a publication and features draws the tallies, the verification with
356    /// its detail, and every feature with its stage, its state and what blocks it; a
357    /// feature that carries no delivery is `unknown`, never delivered.
358    #[test]
359    fn a_report_draws_every_feature_with_its_stage_state_and_blockers() {
360        let report = answer(
361            DELIVERY,
362            Ok(json!({
363                "publication": {
364                    "url": "https://example.test/",
365                    "verification": { "state": "verified", "detail": "the site serves the trunk" }
366                },
367                "tallies": { "features": 2, "on_master": 2, "deployed": 1, "publicly_verified": 1 },
368                "features": [
369                    { "id": "ci", "title": "CI from one model",
370                      "delivery": { "stage": "deployed", "state": "partial", "blocking": ["publication", 7, "site"] } },
371                    { "id": "bare", "title": "A feature with no delivery" }
372                ]
373            })),
374        );
375        let repo = furnished();
376        let ctx = repo.context().expect("a context");
377        let html = delivery(&ctx, &report).render();
378        assert!(
379            html.contains(
380                "href=\"https://example.test/\">https://example.test/</a> — the site serves the trunk"
381            ),
382            "the address the report probed is a link: {html}"
383        );
384        assert!(html.contains("verified"), "{html}");
385        assert!(
386            html.contains("CI from one model") && html.contains("deployed"),
387            "{html}"
388        );
389        assert!(
390            html.contains("publication, site"),
391            "only the words block: {html}"
392        );
393        assert!(html.contains("A feature with no delivery"), "{html}");
394        assert!(html.contains("unknown"), "{html}");
395        // a feature with nothing blocking it says so; the fixture's index holds neither
396        // feature, so each id is text rather than a link to a page that would answer 404
397        assert!(
398            html.contains("<span class=\"mj-ui-dim\">none</span>"),
399            "{html}"
400        );
401        assert!(
402            !html.contains("/cockpit/object?uri=majordomus%3A%2F%2Ffeature%2Fci"),
403            "{html}"
404        );
405        assert!(!html.contains("No feature is declared."), "{html}");
406    }
