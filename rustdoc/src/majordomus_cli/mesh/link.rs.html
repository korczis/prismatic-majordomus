1//! The link protocol: what two runtimes say to become linked, and what they exchange once
2//! they are. Discovery ([`super::protocol`]) proves that a key exists somewhere; a link
3//! proves that the key is answering *now*, serves the *same repository*, speaks a
4//! *compatible protocol*, and is *trusted* — and only then does anything cooperative
5//! travel.
6//!
7//! Two exchanges, both request–answer, both signed end to end:
8//!
9//! 1. **Hello → Welcome.** The dialer presents its [`RuntimeCard`] (key, runtime,
10//!    instance, repository, version, features, endpoints), a fresh nonce and its supported
11//!    protocol range, signed. The answerer verifies the signature, freshness and nonce
12//!    (a replayed hello is refused), negotiates the protocol, refuses another repository,
13//!    itself, or an untrusted key — each with a typed [`RefusalCode`] — and otherwise
14//!    answers with its own card, a link id, its marks, and the dialer's nonce, signed. The
15//!    dialer verifies all of it and applies the same checks the other way. Mutual:
16//!    neither side is linked on the other's word alone.
17//! 2. **Sync ↔ Sync.** The dialer sends its marks and the events the answerer lacks,
18//!    under the link id with a rising counter, signed; the answerer ingests, answers with
19//!    its marks and the events the dialer lacks, signed. One round is one heartbeat and
20//!    one full replication step in both directions, so a link works even when only one
21//!    side can reach the other.
22//!
23//! Every message is a [`Signed`] body: canonical JSON under a per-message domain
24//! separator, so a signature over one message kind can never be replayed as another.
25//! Discovery never grants any of this: a refused hello is recorded, visible, and links
26//! nothing (`project.mesh-cooperation-is-authenticated`).
27//!
28//! ```
29//! use majordomus_cli::mesh::identity::NodeIdentity;
30//! use majordomus_cli::mesh::link::{negotiate, sign, verify_signed, Domain};
31//!
32//! let node = NodeIdentity::ephemeral().unwrap();
33//! let signed = sign(&node, Domain::Hello, serde_json::json!({"hello": 1}));
34//! assert!(verify_signed(&node.public.public_key, Domain::Hello, &signed));
35//! assert!(!verify_signed(&node.public.public_key, Domain::Welcome, &signed), "domains do not cross");
36//! assert_eq!(negotiate(1, 3), Some(1));
37//! assert_eq!(negotiate(2, 3), None);
38//! ```

78/// The message kinds, each its own signing domain. The domain is mixed into the bytes a
79/// signature covers, so a signature made over one kind of message cannot be presented as
80/// another: a captured welcome cannot be replayed as a sync answer even though both are
81/// signed by the same key.
82///
83/// ```
84/// use majordomus_cli::mesh::identity::NodeIdentity;
85/// use majordomus_cli::mesh::link::{sign, verify_signed, Domain};
86///
87/// let node = NodeIdentity::ephemeral().unwrap();
88/// let hello = sign(&node, Domain::Hello, serde_json::json!({ "proto_min": 1 }));
89/// assert!(verify_signed(&node.public.public_key, Domain::Hello, &hello));
90/// for other in [Domain::Welcome, Domain::SyncRequest, Domain::SyncAnswer] {
91///     assert!(!verify_signed(&node.public.public_key, other, &hello), "domains do not cross");
92/// }
93/// ```
94#[derive(Debug, Clone, Copy, PartialEq, Eq)]
95pub enum Domain {
96    /// A dialer's hello.
97    Hello,
98    /// An answerer's welcome.
99    Welcome,
100    /// A dialer's sync round.
101    SyncRequest,
102    /// An answerer's sync answer.
103    SyncAnswer,
104}

106impl Domain {
107    fn prefix(self) -> &'static [u8] {
108        match self {
109            Domain::Hello => b"majordomus-mesh-link-hello/v1\n",
110            Domain::Welcome => b"majordomus-mesh-link-welcome/v1\n",
111            Domain::SyncRequest => b"majordomus-mesh-link-sync-request/v1\n",
112            Domain::SyncAnswer => b"majordomus-mesh-link-sync-answer/v1\n",
113        }
114    }

117/// A signed message: a JSON body and the Ed25519 signature over its canonical bytes under
118/// the message's domain. The body stays JSON so that fields a newer peer adds survive
119/// verification here unchanged.
120///
121/// Keeping the body as JSON rather than as the bytes that were signed is what makes the
122/// protocol extensible: a field a newer peer added is carried through verification
123/// untouched, because the signature is checked over the canonical form of whatever is
124/// there rather than over a shape this version knows.
125///
126/// ```
127/// use majordomus_cli::mesh::identity::NodeIdentity;
128/// use majordomus_cli::mesh::link::{sign, verify_signed, Domain, Signed};
129///
130/// let node = NodeIdentity::ephemeral().unwrap();
131/// let signed: Signed = sign(&node, Domain::SyncRequest, serde_json::json!({ "b": 2, "a": 1 }));
132/// assert_eq!(signed.sig.len(), 128, "hex Ed25519");
133///
134/// // key order is not content, so a round trip through JSON still verifies
135/// let wire = serde_json::to_string(&signed).unwrap();
136/// let back: Signed = serde_json::from_str(&wire).unwrap();
137/// assert!(verify_signed(&node.public.public_key, Domain::SyncRequest, &back));
138/// ```
139#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
140pub struct Signed {
141    /// The message.
142    pub body: Value,
143    /// Hex Ed25519 over the domain and the canonical JSON of `body`.
144    pub sig: String,
145}
146
147/// Sign `body` as a message of `domain`. The signature is taken over the domain separator
148/// followed by the canonical JSON of the body, so two peers that serialize the same facts
149/// in different key orders still agree about what was signed, and nothing signed for one
150/// exchange verifies as part of another.
151///
152/// ```
153/// use majordomus_cli::mesh::identity::NodeIdentity;
154/// use majordomus_cli::mesh::link::{sign, verify_signed, Domain};
155///
156/// let node = NodeIdentity::ephemeral().unwrap();
157/// let signed = sign(&node, Domain::Welcome, serde_json::json!({ "link": "abc" }));
158/// assert!(verify_signed(&node.public.public_key, Domain::Welcome, &signed));
159///
160/// // the body travels as given; the signature is what is derived
161/// assert_eq!(signed.body["link"], "abc");
162/// ```
163pub fn sign(identity: &NodeIdentity, domain: Domain, body: Value) -> Signed {
164    let mut bytes = domain.prefix().to_vec();
165    bytes.extend(canonical_json(&body));
166    Signed {
167        sig: identity.sign(&bytes),
168        body,
169    }
170}

172/// Whether `signed` is a message of `domain` signed by `public_key`. Three things have to
173/// hold at once — the right key, the right domain, and an untouched body — and this
174/// answers about all three with one boolean, because a link has no use for the difference:
175/// any of them failing means the message does not come from the peer it claims to.
176///
177/// ```
178/// use majordomus_cli::mesh::identity::NodeIdentity;
179/// use majordomus_cli::mesh::link::{sign, verify_signed, Domain};
180///
181/// let node = NodeIdentity::ephemeral().unwrap();
182/// let impostor = NodeIdentity::ephemeral().unwrap();
183/// let mut signed = sign(&node, Domain::Hello, serde_json::json!({ "a": 1, "b": 2 }));
184/// assert!(verify_signed(&node.public.public_key, Domain::Hello, &signed));
185///
186/// // another key did not sign it, and a reordered body is the same body
187/// assert!(!verify_signed(&impostor.public.public_key, Domain::Hello, &signed));
188/// signed.body = serde_json::json!({ "b": 2, "a": 1 });
189/// assert!(verify_signed(&node.public.public_key, Domain::Hello, &signed));
190///
191/// // an edited one is not
192/// signed.body = serde_json::json!({ "b": 3, "a": 1 });
193/// assert!(!verify_signed(&node.public.public_key, Domain::Hello, &signed));
194/// ```
195pub fn verify_signed(public_key: &str, domain: Domain, signed: &Signed) -> bool {
196    let mut bytes = domain.prefix().to_vec();
197    bytes.extend(canonical_json(&signed.body));
198    verify(public_key, &bytes, &signed.sig)
199}

201/// The version both sides speak: the highest in the intersection of their ranges, or
202/// `None` when the ranges do not meet.
203///
204/// Taking the highest common version is what lets two executables of different ages link
205/// without either being told about the other: each states the range it speaks and the
206/// newest both understand wins. An inverted range is no range, so a peer that states
207/// nonsense is refused rather than accommodated.
208///
209/// ```
210/// use majordomus_cli::mesh::link::{negotiate, LINK_PROTOCOL_MAX, LINK_PROTOCOL_MIN};
211///
212/// assert_eq!(negotiate(LINK_PROTOCOL_MIN, LINK_PROTOCOL_MAX), Some(LINK_PROTOCOL_MAX));
213/// assert_eq!(negotiate(0, 99), Some(LINK_PROTOCOL_MAX), "the newest both speak");
214/// assert_eq!(negotiate(LINK_PROTOCOL_MAX + 1, LINK_PROTOCOL_MAX + 3), None, "no overlap");
215/// assert_eq!(negotiate(3, 2), None, "an inverted range is no range");
216/// ```
217pub fn negotiate(theirs_min: u32, theirs_max: u32) -> Option<u32> {
218    let low = theirs_min.max(LINK_PROTOCOL_MIN);
219    let high = theirs_max.min(LINK_PROTOCOL_MAX);
220    (theirs_min <= theirs_max && low <= high).then_some(high)
221}

223/// What a runtime says about itself on a link: public facts only. The card carries no
224/// claim a peer has to take on faith — the node id is derived from the key rather than
225/// stated, so a card cannot name a node it cannot sign for, and everything else is either
226/// checkable (the repository, the protocol) or advisory (the name, the endpoints).
227///
228/// ```
229/// use majordomus_cli::mesh::identity::NodeIdentity;
230/// use majordomus_cli::mesh::link::{RuntimeCard, FEATURES};
231///
232/// let node = NodeIdentity::ephemeral().unwrap();
233/// let card = RuntimeCard {
234///     pk: node.public.public_key.clone(),
235///     runtime: "00000000000000aa".into(),
236///     instance: node.public.instance_id.as_str().into(),
237///     repo: "r".repeat(32),
238///     name: "laptop".into(),
239///     version: "0.7.0".into(),
240///     features: FEATURES.iter().map(|f| f.to_string()).collect(),
241///     endpoints: vec!["10.0.0.2:8742".into()],
242/// };
243/// assert!(card.check().is_ok());
244/// assert_eq!(card.runtime_key().unwrap(), format!("{}-00000000000000aa", node.public.node_id));
245/// ```
246#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
247pub struct RuntimeCard {
248    /// The node's Ed25519 public key, hex.
249    pub pk: String,
250    /// The runtime slot, 16 hex.
251    pub runtime: String,
252    /// This run of the runtime, 16 hex.
253    pub instance: String,
254    /// The mesh repository id.
255    pub repo: String,
256    /// The node's display name.
257    #[serde(default)]
258    pub name: String,
259    /// The executable version.
260    #[serde(default)]
261    pub version: String,
262    /// The link features this runtime carries.
263    #[serde(default)]
264    pub features: Vec<String>,
265    /// Where this runtime answers, `host:port`.
266    #[serde(default)]
267    pub endpoints: Vec<String>,
268}

270impl RuntimeCard {
271    /// The stream this card's runtime writes, when the card is well-formed. The node half
272    /// is derived from the key the card carries rather than read from a field, which is
273    /// what makes a stream id unforgeable: to write under a stream a runtime must hold the
274    /// key that stream is derived from. `None` is the answer for a card whose key, runtime
275    /// or instance is not of the right shape — there is no stream for such a card to write.
276    ///
277    /// ```
278    /// # use majordomus_cli::mesh::identity::NodeIdentity;
279    /// # use majordomus_cli::mesh::link::RuntimeCard;
280    /// # let node = NodeIdentity::ephemeral().unwrap();
281    /// # let card = RuntimeCard {
282    /// #     pk: node.public.public_key.clone(),
283    /// #     runtime: "00000000000000aa".into(),
284    /// #     instance: node.public.instance_id.as_str().into(),
285    /// #     repo: "r".repeat(32),
286    /// #     name: String::new(),
287    /// #     version: String::new(),
288    /// #     features: vec![],
289    /// #     endpoints: vec![],
290    /// # };
291    /// let stream = card.stream().unwrap();
292    /// assert!(stream.as_str().starts_with(node.public.node_id.as_str()), "derived from the key");
293    ///
294    /// // a runtime slot that is not 16 hex is not a runtime slot
295    /// let malformed = RuntimeCard { runtime: "zz".into(), ..card };
296    /// assert!(malformed.stream().is_none());
297    /// ```
298    pub fn stream(&self) -> Option<StreamId> {
299        let node = node_id_of_key(&self.pk)?;
300        StreamId::new(node.as_str(), &self.runtime, &self.instance)
301    }
302
303    /// `<node>-<runtime>`, when the card is well-formed: the key the peer table is kept
304    /// under. It deliberately drops the instance the stream carries, so that a runtime
305    /// which restarts is recognised as the same peer rather than accumulating an entry per
306    /// run — the instance is how a restart is noticed, not who the peer is.
307    ///
308    /// ```
309    /// # use majordomus_cli::mesh::identity::NodeIdentity;
310    /// # use majordomus_cli::mesh::link::RuntimeCard;
311    /// # let node = NodeIdentity::ephemeral().unwrap();
312    /// # let card = RuntimeCard {
313    /// #     pk: node.public.public_key.clone(),
314    /// #     runtime: "00000000000000aa".into(),
315    /// #     instance: node.public.instance_id.as_str().into(),
316    /// #     repo: "r".repeat(32),
317    /// #     name: String::new(),
318    /// #     version: String::new(),
319    /// #     features: vec![],
320    /// #     endpoints: vec![],
321    /// # };
322    /// let restarted = RuntimeCard { instance: "0123456789abcdef".into(), ..card.clone() };
323    /// assert_eq!(card.runtime_key(), restarted.runtime_key(), "a restart is the same peer");
324    /// assert_ne!(card.stream(), restarted.stream(), "and a different run of it");
325    /// ```
326    pub fn runtime_key(&self) -> Option<String> {
327        self.stream().map(|s| s.runtime_key())
328    }
329
330    /// Whether the card's runtime carries `feature`. Features are how two executables of
331    /// different ages tell each other what they can do after the protocol version has been
332    /// agreed: an operation that needs something the peer does not carry is refused with
333    /// [`RefusalCode::FeatureUnsupported`], naming the feature, instead of failing
334    /// somewhere further in.
335    ///
336    /// ```
337    /// # use majordomus_cli::mesh::identity::NodeIdentity;
338    /// # use majordomus_cli::mesh::link::{RuntimeCard, FEATURES};
339    /// # let node = NodeIdentity::ephemeral().unwrap();
340    /// # let card = RuntimeCard {
341    /// #     pk: node.public.public_key.clone(),
342    /// #     runtime: "00000000000000aa".into(),
343    /// #     instance: node.public.instance_id.as_str().into(),
344    /// #     repo: "r".repeat(32),
345    /// #     name: String::new(),
346    /// #     version: String::new(),
347    /// #     features: FEATURES.iter().map(|f| f.to_string()).collect(),
348    /// #     endpoints: vec![],
349    /// # };
350    /// assert!(card.supports("reviews"));
351    /// assert!(!card.supports("teleport"), "an unknown word is not a capability");
352    /// ```
353    pub fn supports(&self, feature: &str) -> bool {
354        self.features.iter().any(|f| f == feature)
355    }
356
357    /// Refuse an out-of-bounds card: a key that does not parse, ids of the wrong shape,
358    /// oversized text.
359    ///
360    /// Every field of a card was written by somebody else, so each is bounded before any
361    /// of it is stored or shown. The bounds are small on purpose — a name, a version, a
362    /// handful of features and endpoints — because nothing legitimate needs more, and a
363    /// card is the first thing a stranger gets to put into this process.
364    ///
365    /// ```
366    /// # use majordomus_cli::mesh::identity::NodeIdentity;
367    /// # use majordomus_cli::mesh::link::RuntimeCard;
368    /// # let node = NodeIdentity::ephemeral().unwrap();
369    /// # let card = RuntimeCard {
370    /// #     pk: node.public.public_key.clone(),
371    /// #     runtime: "00000000000000aa".into(),
372    /// #     instance: node.public.instance_id.as_str().into(),
373    /// #     repo: "r".repeat(32),
374    /// #     name: String::new(),
375    /// #     version: String::new(),
376    /// #     features: vec![],
377    /// #     endpoints: vec!["10.0.0.2:8742".into()],
378    /// # };
379    /// assert!(card.check().is_ok());
380    /// assert!(RuntimeCard { repo: String::new(), ..card.clone() }.check().is_err());
381    /// assert!(RuntimeCard { name: "n".repeat(500), ..card.clone() }.check().is_err());
382    /// assert!(
383    ///     RuntimeCard { endpoints: vec!["no-port".into()], ..card }.check().is_err(),
384    ///     "an endpoint without a port is not somewhere to dial",
385    /// );
386    /// ```
387    pub fn check(&self) -> Result<(), String> {
388        if self.stream().is_none() {
389            return Err("the key, runtime or instance is not well-formed".into());
390        }
391        if self.repo.is_empty() || self.repo.len() > 64 {
392            return Err("the repository id is empty or oversized".into());
393        }
394        if self.name.len() > 64
395            || self.version.len() > 32
396            || self.features.len() > 16
397            || self.features.iter().any(|f| f.len() > 32)
398            || self.endpoints.len() > super::protocol::MAX_ENDPOINTS
399            || self
400                .endpoints
401                .iter()
402                .any(|e| e.len() > 64 || !e.contains(':'))
403        {
404            return Err("a field of the card is out of bounds".into());
405        }
406        Ok(())
407    }
408}

410/// A dialer's hello: everything the answerer needs to decide whether to link, in one
411/// signed message. The nonce makes the exchange unrepeatable — the welcome must echo it,
412/// and a nonce is never accepted twice — and the timestamp makes it unstorable, since a
413/// hello captured today cannot be presented tomorrow. The protocol range is stated rather
414/// than assumed, so two executables of different ages find their common version here and
415/// nowhere else.
416///
417/// ```
418/// use majordomus_cli::mesh::identity::NodeIdentity;
419/// use majordomus_cli::mesh::link::{fresh_token, sign, verify_signed, Domain, Hello,
420///     RuntimeCard, LINK_PROTOCOL_MAX, LINK_PROTOCOL_MIN};
421///
422/// let node = NodeIdentity::ephemeral().unwrap();
423/// let hello = Hello {
424///     proto_min: LINK_PROTOCOL_MIN,
425///     proto_max: LINK_PROTOCOL_MAX,
426///     card: RuntimeCard {
427///         pk: node.public.public_key.clone(),
428///         runtime: "00000000000000aa".into(),
429///         instance: node.public.instance_id.as_str().into(),
430///         repo: "r".repeat(32),
431///         name: String::new(),
432///         version: String::new(),
433///         features: vec![],
434///         endpoints: vec![],
435///     },
436///     nonce: fresh_token(),
437///     ts: 1_700_000_000,
438/// };
439/// assert_eq!(hello.nonce.len(), 32, "32 hex of fresh entropy, echoed by the welcome");
440///
441/// let signed = sign(&node, Domain::Hello, serde_json::to_value(&hello).unwrap());
442/// assert!(verify_signed(&node.public.public_key, Domain::Hello, &signed));
443/// ```
444#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
445pub struct Hello {
446    /// The oldest link protocol the dialer speaks.
447    pub proto_min: u32,
448    /// The newest link protocol the dialer speaks.
449    pub proto_max: u32,
450    /// The dialer.
451    pub card: RuntimeCard,
452    /// 32 hex of fresh entropy: the welcome must echo it, and it is never accepted twice.
453    pub nonce: String,
454    /// The dialer's clock, seconds since the Unix epoch.
455    pub ts: u64,
456}

458/// An answerer's welcome: the mirror of the hello, plus the two things that make the link
459/// usable. The link id is what the dialer syncs under afterwards, and the marks let the
460/// very first sync carry exactly the events the answerer lacks instead of a round spent
461/// discovering that. The echoed nonce is what binds this welcome to that hello: without
462/// it, a welcome kept from an earlier exchange would answer a hello it never saw.
463///
464/// ```
465/// use std::collections::BTreeMap;
466/// use majordomus_cli::mesh::identity::NodeIdentity;
467/// use majordomus_cli::mesh::link::{fresh_token, sign, verify_signed, Domain, RuntimeCard,
468///     Welcome, LINK_PROTOCOL_MAX};
469///
470/// let node = NodeIdentity::ephemeral().unwrap();
471/// let nonce = fresh_token();
472/// let welcome = Welcome {
473///     proto: LINK_PROTOCOL_MAX,
474///     link: fresh_token(),
475///     card: RuntimeCard {
476///         pk: node.public.public_key.clone(),
477///         runtime: "00000000000000aa".into(),
478///         instance: node.public.instance_id.as_str().into(),
479///         repo: "r".repeat(32),
480///         name: String::new(),
481///         version: String::new(),
482///         features: vec![],
483///         endpoints: vec![],
484///     },
485///     nonce: nonce.clone(),
486///     ts: 1_700_000_000,
487///     marks: BTreeMap::new(),
488///     heartbeat_seconds: 5,
489///     expiry_seconds: 30,
490/// };
491/// assert_eq!(welcome.nonce, nonce, "this welcome answers that hello and no other");
492/// assert!(welcome.marks.is_empty(), "a runtime that has written nothing has nothing to mark");
493///
494/// let signed = sign(&node, Domain::Welcome, serde_json::to_value(&welcome).unwrap());
495/// assert!(verify_signed(&node.public.public_key, Domain::Welcome, &signed));
496/// ```
497#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
498pub struct Welcome {
499    /// The negotiated protocol version.
500    pub proto: u32,
501    /// The link id the dialer syncs under.
502    pub link: String,
503    /// The answerer.
504    pub card: RuntimeCard,
505    /// The hello's nonce, echoed: this welcome answers that hello and no other.
506    pub nonce: String,
507    /// The answerer's clock.
508    pub ts: u64,
509    /// The answerer's marks, so the first sync carries exactly what it lacks.
510    pub marks: Marks,
511    /// The answerer's heartbeat, informational.
512    pub heartbeat_seconds: u64,
513    /// The answerer's expiry, informational.
514    pub expiry_seconds: u64,
515}

517/// A dialer's sync round: what this runtime holds, and what it believes the answerer
518/// lacks. Sending the events rather than asking for them is what makes one heartbeat a
519/// full replication step in both directions, so replication keeps working when only one
520/// side can open a connection. The counter refuses a replayed round and the instance
521/// refuses a stale one: a restarted dialer carries a new instance and must say hello
522/// again rather than resume a link the answerer no longer has.
523///
524/// ```
525/// use std::collections::BTreeMap;
526/// use majordomus_cli::mesh::identity::NodeIdentity;
527/// use majordomus_cli::mesh::link::{fresh_token, sign, verify_signed, Domain, SyncRequest};
528///
529/// let node = NodeIdentity::ephemeral().unwrap();
530/// let link = fresh_token();
531/// let round = SyncRequest {
532///     link: link.clone(),
533///     counter: 1,
534///     instance: node.public.instance_id.as_str().into(),
535///     marks: BTreeMap::new(),
536///     events: vec![],
537/// };
538/// let next = SyncRequest { counter: round.counter + 1, ..round.clone() };
539/// assert!(next.counter > round.counter, "a round at or below the last accepted is a replay");
540/// assert_eq!(next.link, link, "every round of one link travels under its id");
541///
542/// let signed = sign(&node, Domain::SyncRequest, serde_json::to_value(&next).unwrap());
543/// assert!(verify_signed(&node.public.public_key, Domain::SyncRequest, &signed));
544/// ```
545#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
546pub struct SyncRequest {
547    /// The link id the welcome gave.
548    pub link: String,
549    /// Rising per link: a request at or below the last accepted counter is a replay.
550    pub counter: u64,
551    /// The dialer's instance: a different one is a restarted runtime that must say hello.
552    pub instance: String,
553    /// The dialer's marks.
554    pub marks: Marks,
555    /// The events the answerer lacks, by the answerer's last marks.
556    pub events: Vec<MeshEvent>,
557}
558
559/// An answerer's sync answer: the other half of the round. Its marks are taken *after*
560/// ingesting the request's events, so the dialer learns in one exchange both what arrived
561/// and what it is still missing. The report travels with it because a silently dropped
562/// event is the failure mode this protocol most needs to make visible: a peer that refuses
563/// what it is sent says so, with reasons and numbers, rather than staying quietly behind.
564///
565/// ```
566/// use std::collections::BTreeMap;
567/// use majordomus_cli::mesh::identity::NodeIdentity;
568/// use majordomus_cli::mesh::journal::IngestReport;
569/// use majordomus_cli::mesh::link::{fresh_token, sign, verify_signed, Domain, SyncAnswer};
570///
571/// let node = NodeIdentity::ephemeral().unwrap();
572/// let answer = SyncAnswer {
573///     link: fresh_token(),
574///     counter: 7,
575///     marks: BTreeMap::new(),
576///     events: vec![],
577///     report: IngestReport::default(),
578/// };
579/// assert_eq!(answer.counter, 7, "the request's counter, echoed");
580/// assert_eq!(answer.report.rejected_total(), 0, "nothing was refused this round");
581///
582/// let signed = sign(&node, Domain::SyncAnswer, serde_json::to_value(&answer).unwrap());
583/// assert!(verify_signed(&node.public.public_key, Domain::SyncAnswer, &signed));
584/// ```
585#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
586pub struct SyncAnswer {
587    /// The link id.
588    pub link: String,
589    /// The request's counter, echoed.
590    pub counter: u64,
591    /// The answerer's marks, after ingesting the request's events.
592    pub marks: Marks,
593    /// The events the dialer lacks, by the request's marks.
594    pub events: Vec<MeshEvent>,
595    /// What ingesting the request's events did.
596    pub report: IngestReport,
597}

599/// Why a link message was turned down. Every refusal is typed, counted and shown: a mesh
600/// that refuses a peer says which rule did, instead of dropping a socket.
601///
602/// The codes are the diagnosis a person needs when two runtimes will not link, and they
603/// are deliberately distinct where the remedies differ: `Untrusted` is a key to add,
604/// `RepositoryMismatch` is two different repositories, `SelfLink` is a configuration that
605/// dials its own endpoint, and `UnknownLink` is not a failure at all but an instruction to
606/// say hello again.
607///
608/// ```
609/// use majordomus_cli::mesh::link::RefusalCode;
610///
611/// // the wire word is the code's identity across surfaces, and it is stable
612/// assert_eq!(RefusalCode::RepositoryMismatch.as_str(), "repository_mismatch");
613/// assert_eq!(
614///     serde_json::to_value(RefusalCode::Untrusted).unwrap(),
615///     serde_json::json!("untrusted"),
616/// );
617/// assert_ne!(RefusalCode::Untrusted, RefusalCode::RepositoryMismatch);
618/// ```
619#[derive(
620    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
621)]
622#[serde(rename_all = "snake_case")]
623pub enum RefusalCode {
624    /// Not the shape of the message.
625    Malformed,
626    /// Over [`MAX_LINK_MESSAGE`].
627    Oversized,
628    /// The protocol ranges do not meet.
629    ProtocolUnsupported,
630    /// The signature does not verify.
631    Signature,
632    /// The timestamp is outside the skew window.
633    Stale,
634    /// A nonce or a counter already seen.
635    Replay,
636    /// The peer serves another repository: the isolation rule.
637    RepositoryMismatch,
638    /// The peer's key is not trusted by this runtime's policy: discovery is not trust.
639    Untrusted,
640    /// The peer is this very runtime.
641    SelfLink,
642    /// This runtime's cooperation is not active.
643    NotActive,
644    /// The link id is unknown here: the answerer restarted or expired the link; say hello.
645    UnknownLink,
646    /// A bound of this runtime is full.
647    Capacity,
648    /// The peer lacks a feature the operation needs.
649    FeatureUnsupported,
650}
651
652impl RefusalCode {
653    /// The code's wire word: the snake-case spelling every surface uses, so a refusal
654    /// reads the same in a log line, a JSON answer and a peer's counters. It is written
655    /// out here rather than derived from the variant name, because the word is protocol
656    /// and renaming a variant must not silently change what another runtime reads.
657    ///
658    /// ```
659    /// use majordomus_cli::mesh::link::RefusalCode;
660    ///
661    /// assert_eq!(RefusalCode::ProtocolUnsupported.as_str(), "protocol_unsupported");
662    /// assert_eq!(RefusalCode::SelfLink.as_str(), "self_link");
663    /// assert_eq!(
664    ///     serde_json::to_value(RefusalCode::SelfLink).unwrap(),
665    ///     serde_json::json!(RefusalCode::SelfLink.as_str()),
666    ///     "one spelling, whichever surface prints it",
667    /// );
668    /// ```
669    pub fn as_str(self) -> &'static str {
670        match self {
671            RefusalCode::Malformed => "malformed",
672            RefusalCode::Oversized => "oversized",
673            RefusalCode::ProtocolUnsupported => "protocol_unsupported",
674            RefusalCode::Signature => "signature",
675            RefusalCode::Stale => "stale",
676            RefusalCode::Replay => "replay",
677            RefusalCode::RepositoryMismatch => "repository_mismatch",
678            RefusalCode::Untrusted => "untrusted",
679            RefusalCode::SelfLink => "self_link",
680            RefusalCode::NotActive => "not_active",
681            RefusalCode::UnknownLink => "unknown_link",
682            RefusalCode::Capacity => "capacity",
683            RefusalCode::FeatureUnsupported => "feature_unsupported",
684        }
685    }
686}

688/// A refusal: the code, the reason in words, and the protocol range of the refuser so a
689/// version mismatch is diagnosable from either end.
690///
691/// Carrying the refuser's protocol range in every refusal, not only in the one about
692/// protocols, is what makes a version problem visible from the side that has the log: the
693/// dialer sees what the answerer speaks without anyone having to read the other machine.
694///
695/// ```
696/// use majordomus_cli::mesh::link::{LinkRefusal, RefusalCode, LINK_PROTOCOL_MAX};
697///
698/// let refusal = LinkRefusal::new(RefusalCode::Untrusted, "the key is not in the allowlist");
699/// assert_eq!(refusal.code, RefusalCode::Untrusted);
700/// assert_eq!(refusal.proto_max, LINK_PROTOCOL_MAX);
701/// assert_eq!(refusal.to_string(), "untrusted: the key is not in the allowlist");
702/// ```
703#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
704pub struct LinkRefusal {
705    /// The rule that refused.
706    pub code: RefusalCode,
707    /// Why, in words.
708    pub detail: String,
709    /// The refuser's oldest protocol.
710    pub proto_min: u32,
711    /// The refuser's newest protocol.
712    pub proto_max: u32,
713}
714
715impl LinkRefusal {
716    /// A refusal with this executable's protocol range. Every refusal is built here rather
717    /// than by hand, so that no call site can forget the range and leave the other end
718    /// guessing whether it was turned down over a version or over a rule.
719    ///
720    /// ```
721    /// use majordomus_cli::mesh::link::{LinkRefusal, RefusalCode, LINK_PROTOCOL_MAX,
722    ///     LINK_PROTOCOL_MIN};
723    ///
724    /// let refusal = LinkRefusal::new(RefusalCode::Capacity, "256 peers already linked");
725    /// assert_eq!((refusal.proto_min, refusal.proto_max), (LINK_PROTOCOL_MIN, LINK_PROTOCOL_MAX));
726    /// assert_eq!(refusal.detail, "256 peers already linked");
727    /// ```
728    pub fn new(code: RefusalCode, detail: impl Into<String>) -> Self {
729        LinkRefusal {
730            code,
731            detail: detail.into(),
732            proto_min: LINK_PROTOCOL_MIN,
733            proto_max: LINK_PROTOCOL_MAX,
734        }
735    }
736}
737
738impl std::fmt::Display for LinkRefusal {
739    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
740        write!(f, "{}: {}", self.code.as_str(), self.detail)
741    }
742}

744/// What a hello or a sync answers over the wire: a signed message or a refusal. Both are
745/// successful replies — the transport carries them the same way and neither is an error —
746/// because a refusal is a decision this protocol wants recorded and shown, and an answer
747/// that arrived as an HTTP failure would be indistinguishable from a peer that is down.
748///
749/// ```
750/// use majordomus_cli::mesh::identity::NodeIdentity;
751/// use majordomus_cli::mesh::link::{sign, Domain, LinkReply, RefusalCode};
752///
753/// let node = NodeIdentity::ephemeral().unwrap();
754/// let accepted = LinkReply::accepted(sign(&node, Domain::Welcome, serde_json::json!({})));
755/// assert!(accepted.signed.is_some() && accepted.refusal.is_none());
756///
757/// let refused = LinkReply::refused(RefusalCode::SelfLink, "this runtime dialled itself");
758/// assert!(refused.signed.is_none());
759/// assert_eq!(refused.refusal.unwrap().code, RefusalCode::SelfLink);
760/// ```
761#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
762pub struct LinkReply {
763    /// The signed welcome or sync answer, when accepted.
764    #[serde(default, skip_serializing_if = "Option::is_none")]
765    pub signed: Option<Signed>,
766    /// The refusal, when not.
767    #[serde(default, skip_serializing_if = "Option::is_none")]
768    pub refusal: Option<LinkRefusal>,
769}
770
771impl LinkReply {
772    /// An accepted reply: the signed welcome or sync answer, and no refusal. The two
773    /// halves are built through these constructors rather than by filling the struct in,
774    /// so a reply can never be both at once or neither, which is the one shape a reader
775    /// would have no sensible way to act on.
776    ///
777    /// ```
778    /// use majordomus_cli::mesh::identity::NodeIdentity;
779    /// use majordomus_cli::mesh::link::{sign, Domain, LinkReply};
780    ///
781    /// let node = NodeIdentity::ephemeral().unwrap();
782    /// let reply = LinkReply::accepted(sign(&node, Domain::SyncAnswer, serde_json::json!({})));
783    /// assert!(reply.refusal.is_none(), "an acceptance refuses nothing");
784    /// assert!(reply.signed.unwrap().sig.len() == 128);
785    /// ```
786    pub fn accepted(signed: Signed) -> Self {
787        LinkReply {
788            signed: Some(signed),
789            refusal: None,
790        }
791    }
792
793    /// A refused reply: the typed code that turned the message down, the reason in words,
794    /// and nothing signed. The detail is for the person reading a log and the code is for
795    /// the peer deciding what to do next — retry, say hello again, or stop dialling.
796    ///
797    /// ```
798    /// use majordomus_cli::mesh::link::{LinkReply, RefusalCode};
799    ///
800    /// let reply = LinkReply::refused(RefusalCode::UnknownLink, "say hello again");
801    /// assert!(reply.signed.is_none(), "a refusal signs nothing");
802    /// let refusal = reply.refusal.unwrap();
803    /// assert_eq!(refusal.code, RefusalCode::UnknownLink);
804    /// assert_eq!(refusal.detail, "say hello again");
805    /// ```
806    pub fn refused(code: RefusalCode, detail: impl Into<String>) -> Self {
807        LinkReply {
808            signed: None,
809            refusal: Some(LinkRefusal::new(code, detail)),
810        }
811    }
812}

814/// How a link message reaches a peer. The domain above never names a socket: the HTTP
815/// adapter is the one the shared server answers, and a test can put two runtimes in one
816/// process behind another adapter without changing a line of protocol.
817///
818/// One method is the whole contract, which is what keeps the protocol testable: the
819/// authenticity of a link comes from the signatures inside the messages, never from the
820/// channel, so an implementation owes nothing but delivery.
821///
822/// ```
823/// use majordomus_cli::mesh::link::{LinkReply, LinkTransport, RefusalCode, Signed, HELLO_PATH};
824///
825/// // an adapter that reaches nobody is still a transport; it refuses nothing, it fails
826/// struct Unplugged;
827/// impl LinkTransport for Unplugged {
828///     fn post(&self, endpoint: &str, _path: &str, _message: &Signed) -> Result<LinkReply, String> {
829///         Err(format!("{endpoint}: no network here"))
830///     }
831/// }
832///
833/// let signed = Signed { body: serde_json::json!({}), sig: String::new() };
834/// let error = Unplugged.post("10.0.0.2:8742", HELLO_PATH, &signed).unwrap_err();
835/// assert!(error.contains("10.0.0.2:8742"), "a failure names where it was going");
836///
837/// // a refusal, by contrast, is a reply that arrived
838/// let refusal = LinkReply::refused(RefusalCode::Untrusted, "not in the allowlist");
839/// assert!(refusal.refusal.is_some());
840/// ```
841pub trait LinkTransport: Send + Sync {
842    /// Post `message` to `path` at `endpoint` (`host:port`) and return the peer's reply.
843    /// `Err` is a transport failure — unreachable, timed out, not a reply — never a
844    /// refusal; a refusal is an `Ok` reply that says so.
845    ///
846    /// The distinction is what the caller acts on: a transport failure is a peer to back
847    /// off from and try again later, while a refusal is a decision that will be made the
848    /// same way next time and is recorded instead of retried.
849    ///
850    /// ```
851    /// use majordomus_cli::mesh::link::{HttpTransport, LinkTransport, Signed, HELLO_PATH};
852    ///
853    /// let signed = Signed { body: serde_json::json!({}), sig: String::new() };
854    /// // port 9 discards: unreachable is a transport failure, not a refusal
855    /// let error = HttpTransport.post("127.0.0.1:9", HELLO_PATH, &signed).unwrap_err();
856    /// assert!(error.contains("127.0.0.1:9"), "{error}");
857    /// ```
858    fn post(&self, endpoint: &str, path: &str, message: &Signed) -> Result<LinkReply, String>;
859}

861/// The transport the shared server answers: plain HTTP/1.1 through the crate's one
862/// minimal client (ADR 0032: no HTTP client dependency, no TLS). A private network or an
863/// overlay provides confidentiality; the protocol provides authenticity.
864///
865/// It refuses an oversized message before opening a socket, so the bound the protocol
866/// states is enforced by the sender too and a peer is never asked to read something it
867/// would have to refuse. An endpoint may be written with or without a scheme, because a
868/// person writing a seed should not have to know which.
869///
870/// ```
871/// use majordomus_cli::mesh::link::{HttpTransport, LinkTransport, Signed, MAX_LINK_MESSAGE,
872///     SYNC_PATH};
873///
874/// let oversized = Signed { body: serde_json::json!("x".repeat(MAX_LINK_MESSAGE)), sig: String::new() };
875/// let error = HttpTransport.post("127.0.0.1:9", SYNC_PATH, &oversized).unwrap_err();
876/// assert!(error.contains("exceeds"), "refused before the socket: {error}");
877/// ```
878pub struct HttpTransport;
879
880impl LinkTransport for HttpTransport {
881    fn post(&self, endpoint: &str, path: &str, message: &Signed) -> Result<LinkReply, String> {
882        let body = serde_json::to_string(message).map_err(|e| e.to_string())?;
883        if body.len() > MAX_LINK_MESSAGE {
884            return Err(format!(
885                "a link message of {} bytes exceeds {MAX_LINK_MESSAGE}",
886                body.len()
887            ));
888        }
889        let base = if endpoint.starts_with("http://") {
890            endpoint.to_string()
891        } else {
892            format!("http://{endpoint}")
893        };
894        let reply = crate::mcp::bridge::request(
895            &base,
896            "POST",
897            path,
898            &[("Content-Type", "application/json")],
899            Some(&body),
900            REQUEST_TIMEOUT,
901        )
902        .map_err(|e| format!("{base}{path}: {e}"))?;
903        if reply.status != 200 {
904            return Err(format!("{base}{path}: status {}", reply.status));
905        }
906        serde_json::from_str(&reply.body)
907            .map_err(|e| format!("{base}{path}: not a link reply: {e}"))
908    }
909}

911/// 32 hex characters of OS entropy: nonces and link ids. Both uses need unpredictability
912/// rather than uniqueness alone — a guessable nonce would let a captured hello be answered
913/// in advance, and a guessable link id would let a stranger sync under somebody else's
914/// link. When the OS has no entropy to give, a clock-derived value keeps the link working
915/// and the replay caches still refuse exact repeats.
916///
917/// ```
918/// use majordomus_cli::mesh::link::fresh_token;
919///
920/// let (a, b) = (fresh_token(), fresh_token());
921/// assert_eq!(a.len(), 32);
922/// assert!(a.bytes().all(|c| c.is_ascii_hexdigit()));
923/// assert_ne!(a, b, "a token is fresh, not a counter");
924/// ```
925pub fn fresh_token() -> String {
926    let mut bytes = [0u8; 16];
927    if getrandom::getrandom(&mut bytes).is_err() {
928        // Without entropy a nonce is guessable; a clock-derived fallback keeps the link
929        // working and the replay cache still refuses exact repeats.
930        let nanos = std::time::SystemTime::now()
931            .duration_since(std::time::UNIX_EPOCH)
932            .map(|d| d.as_nanos())
933            .unwrap_or(0);
934        bytes = nanos.to_le_bytes();
935    }
936    bytes.iter().map(|b| format!("{b:02x}")).collect()
937}
