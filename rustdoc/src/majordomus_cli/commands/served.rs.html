1//! `majordomus served`: what each deployment serves, as observed and as it stands now.
2//!
3//! Both subcommands run a capability of the `served` module through the one executor. The
4//! text rendering prints the verdict first and the reason beside it, and `observe` exits
5//! with the verdict's code, so a script reads the same three-way answer a person does:
6//! 0 served, 10 measurably not, 12 not answered.
7

17/// Run `majordomus served`.
18pub fn run(args: ServedArgs) -> Result<u8> {
19    let app = App::load(&args.repo)?;
20    let ctx = &app.context;
21    let stdout = std::io::stdout();
22    let mut out = stdout.lock();
23    match args.command {
24        ServedCommand::Observe {
25            commit,
26            url,
27            identity,
28            deployment,
29            timeout,
30            dry_run,
31        } => {
32            let input = json!({
33                "commit": commit, "url": url, "identity": identity,
34                "deployment": deployment, "timeout_seconds": timeout, "dry_run": dry_run,
35            });
36            let v = execute(ctx, &["served", "observe"], strip_nulls(input))?;
37            match args.format {
38                OutputFormat::Json => writeln!(out, "{}", pretty(&v)),
39                OutputFormat::Text => {
40                    let o = &v["observation"];
41                    write_line(&mut out, o, o["deployment"].as_str().unwrap_or("?"))?;
42                    let recorded = v["recorded"].as_str().unwrap_or("not recorded (dry run)");
43                    writeln!(out, "recorded     {recorded}")
44                }
45            }
46            .map_err(Error::Transport)?;
47            Ok(u8::try_from(v["exit_code"].as_i64().unwrap_or(12)).unwrap_or(12))
48        }
49        ServedCommand::Forge => {
50            let v = execute(ctx, &["served", "forge"], json!({}))?;
51            match args.format {
52                OutputFormat::Json => writeln!(out, "{}", pretty(&v)),
53                OutputFormat::Text => writeln!(out, "{}", forge_line(&v)),
54            }
55            .map_err(Error::Transport)?;
56            Ok(u8::try_from(v["exit_code"].as_i64().unwrap_or(12)).unwrap_or(12))
57        }
58        ServedCommand::Show { commit, deployment } => {
59            let input = strip_nulls(json!({ "commit": commit, "deployment": deployment }));
60            let v = execute(ctx, &["served", "show"], input)?;
61            match args.format {
62                OutputFormat::Json => writeln!(out, "{}", pretty(&v)).map_err(Error::Transport)?,
63                OutputFormat::Text => {
64                    let expected = v["expected"].as_str().unwrap_or("?");
65                    writeln!(out, "against      {}", &expected[..expected.len().min(12)])
66                        .map_err(Error::Transport)?;
67                    let rows = v["deployments"].as_array().cloned().unwrap_or_default();
68                    if rows.is_empty() {
69                        writeln!(out, "nothing observed yet: majordomus served observe")
70                            .map_err(Error::Transport)?;
71                    }
72                    for d in &rows {
73                        let now =
74                            json!({ "verdict": d["now"]["verdict"], "reason": d["now"]["reason"] });
75                        write_line(&mut out, &now, d["deployment"].as_str().unwrap_or("?"))?;
76                        writeln!(
77                            out,
78                            "  seen       {}",
79                            d["observation"]["at"].as_str().unwrap_or("?")
80                        )
81                        .map_err(Error::Transport)?;
82                    }
83                    if let Some(n) = v["unreadable"].as_u64().filter(|n| *n > 0) {
84                        writeln!(out, "unreadable   {n} line(s) of {}", v["observations"])
85                            .map_err(Error::Transport)?;
86                    }
87                }
88            }
89            Ok(0)
90        }
91    }
92}

94/// The receipt of asking the forge, as one line: what it held, or why it was not asked.
95fn forge_line(v: &Value) -> String {
96    if v["observed"].as_bool() == Some(true) {
97        format!(
98            "observed {} deployment(s) to {} environment(s) and {} release(s) of {} at {}",
99            v["deployments"],
100            v["environments"],
101            v["releases"],
102            v["repository"].as_str().unwrap_or("?"),
103            v["observed_at"].as_str().unwrap_or("?"),
104        )
105    } else {
106        format!(
107            "the forge's deployments were not observed: {}",
108            v["reason"].as_str().unwrap_or("no reason was given")
109        )
110    }
111}

113fn write_line(out: &mut std::io::StdoutLock<'_>, judged: &Value, name: &str) -> Result<()> {
114    writeln!(
115        out,
116        "{name:<12} {:<11} {}",
117        judged["verdict"].as_str().unwrap_or("?"),
118        judged["reason"].as_str().unwrap_or("")
119    )
120    .map_err(Error::Transport)
121}

123fn strip_nulls(mut v: Value) -> Value {
124    if let Some(m) = v.as_object_mut() {
125        m.retain(|_, x| !x.is_null());
126    }
127    v
128}
129
130fn pretty(v: &Value) -> String {
131    serde_json::to_string_pretty(v).unwrap_or_default()
132}

134fn execute(ctx: &crate::capability::Context, path: &[&str], input: Value) -> Result<Value> {
135    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
136    let id = ctx
137        .registry
138        .by_cli(&words)
139        .map(|c| c.id.as_str())
140        .ok_or_else(|| Error::Protocol {
141            reason: format!(
142                "no capability is exposed as `majordomus {}`",
143                path.join(" ")
144            ),
145        })?;
146    ctx.execute(id, input).map_err(|e| match e {
147        CapabilityError::InvalidInput(reason) | CapabilityError::Refused(reason) => {
148            Error::Protocol { reason }
149        }
150        other => Error::Protocol {
151            reason: other.to_string(),
152        },
153    })
154}

156#[cfg(test)]
157mod tests {
158    use super::*;
159
160    #[test]
161    fn the_receipt_of_asking_the_forge_is_one_line_either_way() {
162        let held = json!({"observed": true, "deployments": 50, "environments": 2, "releases": 21,
163            "repository": "o/r", "observed_at": "2026-10-12T21:00:00Z", "exit_code": 0});
164        assert_eq!(
165            forge_line(&held),
166            "observed 50 deployment(s) to 2 environment(s) and 21 release(s) of o/r at 2026-10-12T21:00:00Z"
167        );
168        let refused = json!({"observed": false, "reason": "gh could not run", "exit_code": 12});
169        assert_eq!(
170            forge_line(&refused),
171            "the forge's deployments were not observed: gh could not run"
172        );
173        assert!(forge_line(&json!({})).ends_with("no reason was given"));
174    }
175
176    /// `served forge`, run whole, in a repository the forge does not know: exit 12, said.
177    #[test]
178    fn asking_a_forge_that_knows_nothing_ends_on_not_answered() {
179        let repo = crate::synthetic::SyntheticRepository::new(crate::synthetic::Shape::default())
180            .expect("a synthetic repository");
181        for format in [OutputFormat::Text, OutputFormat::Json] {
182            let code = run(ServedArgs {
183                repo: crate::cli::RepoArgs {
184                    repo: Some(repo.root().to_path_buf()),
185                    discovery: crate::cli::DiscoveryMode::Filesystem,
186                    strict: false,
187                    share: Some(
188                        std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../../share"),
189                    ),
190                },
191                command: ServedCommand::Forge,
192                format,
193            })
194            .expect("a receipt, not an error");
195            assert_eq!(code, 12);
196        }
197    }
