1//! The discovery manager: provider lifecycle, the one verification path, and the one
2//! registry. Providers hand raw bytes here; this thread parses, checks bounds and
3//! staleness, verifies the signature, evaluates trust, and only then touches the
4//! registry — in that order, for every transport alike. A provider cannot skip a step
5//! because no provider ever parses.
6//!
7//! The runtime lives on the capability [`Context`](crate::capability::Context) from
8//! process start, inactive, and a shared server activates it when the repository's mesh
9//! declaration says so. Activation is idempotent, bounded, and never blocks startup:
10//! sockets open on their own threads, and a provider that cannot start is a `Failed`
11//! status, not a failed server.
12

39/// Refused datagrams, by reason. The diagnosis of a hostile or misconfigured network:
40/// a mesh that hears garbage says so, with numbers, instead of being quietly empty.
41#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, JsonSchema)]
42pub struct Refusals {
43    /// Longer than the datagram bound.
44    pub oversized: u64,
45    /// Not an envelope.
46    pub malformed: u64,
47    /// A protocol version this executable does not read.
48    pub version: u64,
49    /// Out-of-bounds fields.
50    pub bounds: u64,
51    /// Outside the clock-skew window.
52    pub stale: u64,
53    /// A signature that does not verify.
54    pub signature: u64,
55    /// This node's own datagrams, heard back and skipped.
56    pub self_heard: u64,
57}

59#[derive(Default)]
60struct RefusalCounters {
61    oversized: AtomicU64,
62    malformed: AtomicU64,
63    version: AtomicU64,
64    bounds: AtomicU64,
65    stale: AtomicU64,
66    signature: AtomicU64,
67    self_heard: AtomicU64,
68}
69
70impl RefusalCounters {
71    fn count(&self, refusal: &Refusal) {
72        let counter = match refusal {
73            Refusal::Oversized => &self.oversized,
74            Refusal::Malformed => &self.malformed,
75            Refusal::Version => &self.version,
76            Refusal::Bounds => &self.bounds,
77            Refusal::Stale => &self.stale,
78            Refusal::Signature => &self.signature,
79        };
80        counter.fetch_add(1, Ordering::Relaxed);
81    }
82
83    fn snapshot(&self) -> Refusals {
84        Refusals {
85            oversized: self.oversized.load(Ordering::Relaxed),
86            malformed: self.malformed.load(Ordering::Relaxed),
87            version: self.version.load(Ordering::Relaxed),
88            bounds: self.bounds.load(Ordering::Relaxed),
89            stale: self.stale.load(Ordering::Relaxed),
90            signature: self.signature.load(Ordering::Relaxed),
91            self_heard: self.self_heard.load(Ordering::Relaxed),
92        }
93    }
94}

130/// The runtime: the registry (always there, so a surface can answer before activation)
131/// and the active machinery once a declaration switched it on.
132pub struct MeshRuntime {
133    registry: Arc<MeshRegistry>,
134    refusals: Arc<RefusalCounters>,
135    state: Mutex<Option<Active>>,
136    /// Why the mesh is not active, when it is not.
137    reason: Mutex<String>,
138    /// The cooperation runtime, once a server attached one.
139    cooperation: Mutex<Option<Arc<Cooperation>>>,
140    /// Why cooperation is not running, when the mesh is but cooperation is not.
141    cooperation_reason: Mutex<Option<String>>,
142    /// Whether a shared server decided anything about this runtime: activated it, or
143    /// declined with a reason. A runtime nobody decided on is the command line's, and its

146}

155    /// An inactive runtime: an empty registry and the reason "nothing activated it".
156    pub fn new() -> Self {
157        MeshRuntime {
158            registry: Arc::new(MeshRegistry::new()),
159            refusals: Arc::new(RefusalCounters::default()),
160            state: Mutex::new(None),

164            cooperation: Mutex::new(None),
165            cooperation_reason: Mutex::new(None),
166            decided: AtomicBool::new(false),

170    /// Whether a shared server activated or declined this runtime. `false` in every process
171    /// that is not a server — the command line, a test — where nothing activates the mesh
172    /// and [`MeshRuntime::status`] reports an absence, not a decision.
173    ///
174    /// The distinction is what lets `mesh doctor` judge the server: an enabled declaration
175    /// and an inactive mesh is a failure when a server decided it and nothing at all when
176    /// nobody did.
177    ///
178    /// ```
179    /// use majordomus_cli::mesh::manager::MeshRuntime;
180    ///
181    /// let mesh = MeshRuntime::new();
182    /// assert!(!mesh.decided(), "nothing has decided the command line's runtime");
183    /// assert!(!mesh.status().active);
184    ///
185    /// // a server that could not activate the mesh says why, and that is a decision
186    /// mesh.decline("the node identity did not load");
187    /// assert!(mesh.decided());
188    /// assert!(mesh.status().reason.unwrap().contains("did not load"));
189    /// ```
190    pub fn decided(&self) -> bool {
191        self.decided.load(Ordering::SeqCst)
192    }

194    /// Attach the cooperation runtime a server built for this mesh. The runtime is
195    /// already started; a second attachment replaces nothing.
196    ///
197    /// Discovery and cooperation are built separately — one finds runtimes, the other
198    /// links to them — and this is where the surfaces find both behind one handle. The
199    /// first attachment wins, so a second server in one process cannot quietly take over
200    /// the journal the first one is already writing to.
201    ///
202    /// ```
203    /// # use std::sync::Arc;
204    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
205    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
206    /// # use majordomus_cli::mesh::identity::NodeIdentity;
207    /// # use majordomus_cli::mesh::link::HttpTransport;
208    /// # use majordomus_cli::mesh::registry::MeshRegistry;
209    /// # use majordomus_cli::mesh::repository::of_root_commits;
210    /// # let built = |slot: &str| Cooperation::new(CooperationSetup {
211    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: slot.into(),
212    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
213    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
214    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
215    /// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
216    /// # }).unwrap();
217    /// use majordomus_cli::mesh::manager::MeshRuntime;
218    ///
219    /// let mesh = MeshRuntime::new();
220    /// let first = built("0000000000000001");
221    /// mesh.attach_cooperation(Arc::clone(&first));
222    /// mesh.attach_cooperation(built("0000000000000002"));
223    ///
224    /// let attached = mesh.cooperation().expect("a cooperation runtime");
225    /// assert_eq!(attached.runtime_key(), first.runtime_key(), "the first one keeps the slot");
226    /// ```
227    pub fn attach_cooperation(&self, cooperation: Arc<Cooperation>) {
228        let mut slot = self.cooperation.lock().expect("mesh cooperation");
229        if slot.is_none() {
230            *slot = Some(cooperation);
231        }
232    }

234    /// Record why cooperation is not running while the mesh is.
235    ///
236    /// A mesh that discovers runtimes and links to none is the confusing case: everything
237    /// looks switched on and nothing is shared. The server that decided not to start
238    /// cooperation says so here, so the answer reaches whoever asks rather than living in
239    /// the log of a process they cannot see.
240    ///
241    /// ```
242    /// use majordomus_cli::mesh::manager::MeshRuntime;
243    ///
244    /// let mesh = MeshRuntime::new();
245    /// mesh.decline_cooperation("no endpoint to be dialed at");
246    ///
247    /// let status = mesh.cooperation_status();
248    /// assert!(!status.active);
249    /// assert_eq!(status.reason.as_deref(), Some("no endpoint to be dialed at"));
250    /// ```
251    pub fn decline_cooperation(&self, reason: &str) {
252        *self
253            .cooperation_reason
254            .lock()
255            .expect("mesh cooperation reason") = Some(reason.into());
256    }

258    /// The cooperation runtime, while one is still running. A stopped runtime is not
259    /// handed out: a caller that got it would write to a journal nobody is replicating and
260    /// believe it had told the mesh something. `None` here means the same to every caller
261    /// — do this locally, or say it cannot be done — however cooperation came to be absent.
262    ///
263    /// ```
264    /// # use std::sync::Arc;
265    /// # use majordomus_cli::mesh::config::{CooperationConfig, TrustConfig};
266    /// # use majordomus_cli::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
267    /// # use majordomus_cli::mesh::identity::NodeIdentity;
268    /// # use majordomus_cli::mesh::link::HttpTransport;
269    /// # use majordomus_cli::mesh::registry::MeshRegistry;
270    /// # use majordomus_cli::mesh::repository::of_root_commits;
271    /// # let cooperation = Cooperation::new(CooperationSetup {
272    /// #     identity: Arc::new(NodeIdentity::ephemeral().unwrap()), runtime: "0000000000000001".into(),
273    /// #     repository: of_root_commits(&["root".into()]), endpoints: vec!["127.0.0.1:9".into()],
274    /// #     version: "doc".into(), config: CooperationConfig::default(), trust: TrustConfig::default(),
275    /// #     journal_path: None, registry: Arc::new(MeshRegistry::new()),
276    /// #     transport: Arc::new(HttpTransport), board: None, checkout: CheckoutFacts::default(),
277    /// # }).unwrap();
278    /// use majordomus_cli::mesh::manager::MeshRuntime;
279    ///
280    /// let mesh = MeshRuntime::new();
281    /// assert!(mesh.cooperation().is_none(), "nothing has attached one");
282    ///
283    /// mesh.attach_cooperation(Arc::clone(&cooperation));
284    /// assert!(mesh.cooperation().is_some());
285    ///
286    /// cooperation.stop();
287    /// assert!(mesh.cooperation().is_none(), "a stopped runtime is not handed out");
288    /// ```
289    pub fn cooperation(&self) -> Option<Arc<Cooperation>> {
290        self.cooperation
291            .lock()
292            .expect("mesh cooperation")
293            .as_ref()
294            .filter(|c| !c.stopped())
295            .cloned()
296    }

298    /// Cooperation's status, or why there is none: the mesh's own reason when the mesh is
299    /// off, cooperation's reason when only cooperation is.
300    ///
301    /// The two reasons are kept apart because they call for different acts. A mesh that was
302    /// never switched on is a declaration to write; cooperation declined under a running
303    /// mesh is something about this server — no endpoint, no trusted key — and the more
304    /// specific reason is the one worth telling a person, so it wins when both exist.
305    ///
306    /// ```
307    /// use majordomus_cli::mesh::manager::MeshRuntime;
308    ///
309    /// let mesh = MeshRuntime::new();
310    /// let untouched = mesh.cooperation_status();
311    /// assert!(untouched.reason.unwrap().contains("no shared server"), "the mesh's reason");
312    ///
313    /// mesh.decline_cooperation("the declaration enables the mesh but not cooperation");
314    /// let declined = mesh.cooperation_status();
315    /// assert!(declined.reason.unwrap().contains("not cooperation"), "the nearer reason wins");
316    /// assert!(declined.peers.is_empty(), "and the shape is the one a live runtime answers");
317    /// ```
318    pub fn cooperation_status(&self) -> CooperationStatus {
319        if let Some(c) = self.cooperation() {
320            return c.status();
321        }
322        let reason = self
323            .cooperation_reason
324            .lock()
325            .expect("mesh cooperation reason")
326            .clone()
327            .unwrap_or_else(|| self.reason.lock().expect("mesh reason").clone());
328        CooperationStatus::inactive(&reason)
329    }

336    /// Record why the mesh is not running, without activating anything: a disabled or
337    /// malformed declaration is an answer `mesh.status` must be able to give.
338    pub fn decline(&self, reason: &str) {
339        let state = self.state.lock().expect("mesh state");
340        if state.is_none() {

346    /// Activate the mesh from a declaration. Idempotent: a second activation of an

360    /// [`MeshRuntime::activate`] for one runtime slot of the node: what a shared server
361    /// does, naming its checkout's runtime so that two servers of one machine announce,
362    /// hear and link to each other as the two runtimes they are.
363    ///
364    /// Without the slot, two worktrees of one repository on one machine would announce
365    /// under one identity and each would take the other's announcements for its own echo.
366    /// The slot is what makes them two peers that cooperate, which is the ordinary case on
367    /// a developer's machine and not an exotic one.
368    ///
369    /// ```
370    /// use std::sync::Arc;
371    ///
372    /// use majordomus_cli::mesh::config::MeshConfig;
373    /// use majordomus_cli::mesh::identity::NodeIdentity;
374    /// use majordomus_cli::mesh::manager::MeshRuntime;
375    ///
376    /// // a declaration with every discovery mechanism off: nothing opens a socket
377    /// let config: MeshConfig = serde_json::from_value(serde_json::json!({
378    ///     "schema": "mesh/v1", "kind": "mesh", "id": "majordomus", "enabled": true,
379    ///     "multicast": { "enabled": false },
380    /// })).unwrap();
381    ///
382    /// let mesh = MeshRuntime::new();
383    /// let identity = Arc::new(NodeIdentity::ephemeral().unwrap());
384    /// mesh.activate_as(&config, Arc::clone(&identity), "0000000000000001",
385    ///     vec!["127.0.0.1:8742".into()], vec!["root".into()], "doc").unwrap();
386    ///
387    /// let status = mesh.status();
388    /// assert!(status.active);
389    /// assert_eq!(status.identity.unwrap().node_id, identity.public.node_id);
390    ///
391    /// // idempotent: the second server of this process does not restart the first's mesh
392    /// mesh.activate_as(&config, Arc::clone(&identity), "0000000000000002",
393    ///     vec!["127.0.0.1:8743".into()], vec!["root".into()], "doc").unwrap();
394    /// assert_eq!(mesh.status().started_at, status.started_at);
395    /// ```
396    pub fn activate_as(
397        &self,
398        config: &MeshConfig,
399        identity: Arc<NodeIdentity>,
400        runtime: &str,
401        endpoints: Vec<String>,
402        repos: Vec<String>,
403        version: &str,
404    ) -> Result<(), MeshError> {
405        self.decided.store(true, Ordering::SeqCst);
406        let mut state = self.state.lock().expect("mesh state");
407        if state.is_some() {
408            return Ok(());
409        }
410        if !config.enabled {
411            *self.reason.lock().expect("mesh reason") =
412                "not active: the mesh declaration is disabled".into();
413            return Ok(());
414        }
415        let beacon = Arc::new(
416            Beacon::new(
417                identity,
418                endpoints,
419                vec!["http".into(), "mcp".into(), "ws".into()],
420                repos,
421                version,
422            )
423            .with_runtime(runtime),
424        );
425        let (tx, rx) = channel::<Observation>();
426        let stop = Arc::new(AtomicBool::new(false));
427        let ctx = ProviderContext {
428            tx: tx.clone(),
429            stop: Arc::clone(&stop),
430            beacon: Arc::clone(&beacon),
431        };
432
433        // The providers the declaration asks for. Each failure is that provider's
434        // status; the loop never stops on one.
435        let mut providers: Vec<Box<dyn MeshProvider>> = Vec::new();
436        if config.multicast.enabled {
437            providers.push(Box::new(MulticastProvider::new(config.multicast.clone())));
438        }
439        if config.broadcast.mode != BroadcastMode::Disabled {
440            providers.push(Box::new(BroadcastProvider::new(
441                config.broadcast.clone(),
442                !config.multicast.enabled,
443            )));
444        }
445        if !config.rendezvous.endpoints.is_empty() {
446            providers.push(Box::new(RendezvousProvider::new(
447                config.rendezvous.endpoints.clone(),
448                config.rendezvous.interval_seconds,
449            )));
450        }
451        for provider in providers.iter_mut() {
452            if let Err(e) = provider.start(&ctx) {
453                tracing::warn!(provider = provider.id(), error = %e, "mesh provider did not start; the others run on");
454            }
455        }
456
457        // The manager thread: the one verification path.
458        {
459            let registry = Arc::clone(&self.registry);
460            let refusals = Arc::clone(&self.refusals);
461            let stop = Arc::clone(&stop);
462            let own = (
463                beacon.identity().public.node_id.clone(),
464                beacon.runtime().to_string(),
465            );
466            let trust = config.trust.clone();
467            let _ = std::thread::Builder::new()
468                .name("majordomus-mesh".into())
469                .spawn(move || {
470                    while !stop.load(Ordering::SeqCst) {
471                        let observation = match rx.recv_timeout(RECV_TIMEOUT) {
472                            Ok(o) => o,
473                            Err(std::sync::mpsc::RecvTimeoutError::Timeout) => continue,
474                            Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => break,
475                        };
476                        ingest(&registry, &refusals, &trust, &own, &observation);
477                    }
478                });
479        }
480
481        *state = Some(Active {
482            beacon,
483            tx,
484            stop,
485            providers,
486            trust: config.trust.clone(),
487            started_at: crate::peers::rfc3339(std::time::SystemTime::now()),
488        });
489        Ok(())
490    }

492    /// Attach one more provider to an active mesh. This is the entire registration a
493    /// new discovery mechanism needs: no surface, consumer or registry changes with it.
494    pub fn attach(&self, mut provider: Box<dyn MeshProvider>) -> Result<(), MeshError> {
495        let mut state = self.state.lock().expect("mesh state");
496        let Some(active) = state.as_mut() else {
497            return Err(MeshError::Provider("the mesh is not active".into()));
498        };
499        let ctx = ProviderContext {
500            tx: active.tx.clone(),
501            stop: Arc::clone(&active.stop),
502            beacon: Arc::clone(&active.beacon),
503        };
504        let outcome = provider.start(&ctx);
505        active.providers.push(provider);
506        outcome
507    }

509    /// Tell the mesh to stop, without waiting for anything: every loop and handler reads the
510    /// flag at once, and the draining that needs locks is left to [`stop`](Self::stop). A
511    /// process that is shutting down calls this first, does the work that must not be
512    /// delayed — closing its listeners, releasing its lease — and only then drains.
513    ///
514    /// ```
515    /// use majordomus_cli::mesh::manager::MeshRuntime;
516    ///
517    /// // a mesh that never activated has nothing to tell, and says so rather than failing:
518    /// // a shutdown path must not care whether the thing it is stopping ever started
519    /// let mesh = MeshRuntime::new();
520    /// mesh.begin_stop();
521    /// assert!(!mesh.status().active);
522    /// ```
523    pub fn begin_stop(&self) {
524        if let Some(cooperation) = self.cooperation.lock().expect("mesh cooperation").as_ref() {
525            cooperation.begin_stop();
526        }
527        if let Some(active) = self.state.lock().expect("mesh state").as_ref() {
528            active.stop.store(true, Ordering::SeqCst);
529        }
530    }

532    /// Stop the mesh. Every provider thread and the manager end at their next bounded
533    /// wait; the registry keeps what it saw for whoever still asks.
534    pub fn stop(&self) {
535        if let Some(cooperation) = self.cooperation.lock().expect("mesh cooperation").as_ref() {
536            cooperation.stop();
537        }
538        let mut state = self.state.lock().expect("mesh state");
539        if let Some(active) = state.take() {
540            active.stop.store(true, Ordering::SeqCst);
541            *self.reason.lock().expect("mesh reason") = "not active: stopped".into();
542        }
543    }

545    /// The status, as every surface answers it.
546    pub fn status(&self) -> MeshStatus {
547        let state = self.state.lock().expect("mesh state");
548        let mut status = match state.as_ref() {
549            Some(active) => {
550                let mut providers: Vec<ProviderStatus> =
551                    active.providers.iter().map(|p| p.status()).collect();
552                crate::order::canonical(&mut providers);
553                MeshStatus {
554                    active: true,
555                    reason: None,
556                    identity: Some(active.beacon.identity().public.clone()),
557                    trust_policy: Some(active.trust.policy.as_str().into()),
558                    providers,
559                    tallies: Tallies::default(),
560                    refusals: Refusals::default(),
561                    started_at: Some(active.started_at.clone()),
562                }
563            }
564            None => MeshStatus {
565                active: false,
566                reason: Some(self.reason.lock().expect("mesh reason").clone()),
567                identity: None,
568                trust_policy: None,
569                providers: Vec::new(),
570                tallies: Tallies::default(),
571                refusals: Refusals::default(),
572                started_at: None,
573            },
574        };
575        status.tallies = self.registry.tallies();
576        status.refusals = self.refusals.snapshot();
577        status
578    }

585    /// Handle one rendezvous registration: verify the presented envelope exactly as a
586    /// datagram, record it, and answer with this node's envelope and the candidates the
587    /// registry holds. Registration grants nothing — the caller becomes a record, not a
588    /// peer with rights.
589    pub fn register(&self, envelope: &serde_json::Value, path: &str) -> RegisterAnswer {
590        let state = self.state.lock().expect("mesh state");
591        let Some(active) = state.as_ref() else {
592            return RegisterAnswer {
593                accepted: false,
594                refusal: Some("the mesh is not active on this node".into()),
595                candidates: Vec::new(),
596            };
597        };
598        let own = (
599            active.beacon.identity().public.node_id.clone(),
600            active.beacon.runtime().to_string(),
601        );
602        let trust = active.trust.clone();
603        let own_envelope = active.beacon.next_envelope();
604        drop(state);
605
606        let bytes = serde_json::to_vec(envelope).unwrap_or_default();
607        let refusal = match ingest(
608            &self.registry,
609            &self.refusals,
610            &trust,
611            &own,
612            &Observation {
613                source: MeshSource::Rendezvous,
614                path: path.into(),
615                bytes,
616            },
617        ) {
618            Ingest::Accepted => None,
619            Ingest::Own => Some("that is this node's own envelope".into()),
620            Ingest::Replay => Some("replayed".into()),
621            Ingest::Refused(r) => Some(r.to_string()),
622        };
623        let mut candidates = Vec::with_capacity(MAX_CANDIDATES + 1);
624        if let Ok(own_value) = serde_json::to_value(&own_envelope) {
625            candidates.push(own_value);
626        }
627        candidates.extend(self.registry.candidates(MAX_CANDIDATES));
628        let candidates = candidates
629            .into_iter()
630            .filter_map(|v| serde_json::from_value(v).ok())
631            .collect();
632        RegisterAnswer {
633            accepted: refusal.is_none(),
634            refusal,
635            candidates,
636        }
637    }
638}
639
640enum Ingest {
641    Accepted,
642    Own,
643    Replay,
644    Refused(Refusal),
645}
646
647/// The one verification path: bounds, shape, version, staleness, signature, identity,
648/// trust, registry — in that order, for every provider alike.
649fn ingest(
650    registry: &MeshRegistry,
651    refusals: &RefusalCounters,
652    trust: &TrustConfig,
653    own: &(super::identity::NodeId, String),
654    observation: &Observation,
655) -> Ingest {
656    let envelope = match protocol::parse(&observation.bytes) {
657        Ok(e) => e,
658        Err(refusal) => {
659            refusals.count(&refusal);
660            return Ingest::Refused(refusal);
661        }
662    };
663    let Some(node_id) = envelope.adv.node_id() else {
664        // Unreachable after a verified parse, but hostile input earns no unwrap.
665        refusals.count(&Refusal::Signature);
666        return Ingest::Refused(Refusal::Signature);
667    };
668    // Only this very runtime is "own": another server of the same machine carries the
669    // same key under another runtime slot, and is a runtime to hear, not an echo.
670    if node_id == own.0 && envelope.adv.rt == own.1 {
671        refusals.self_heard.fetch_add(1, Ordering::Relaxed);
672        return Ingest::Own;
673    }
674    let known_key = registry.known_key(&node_id);
675    let previous = registry.trust_of(&node_id);
676    let verdict = trust::evaluate(
677        &trust.policy,
678        &trust.allow,
679        &envelope.adv.pk,
680        known_key.as_deref(),
681        previous.as_ref(),
682    );
683    let raw = serde_json::to_value(&envelope).unwrap_or(serde_json::Value::Null);
684    if registry.observe(
685        &envelope.adv,
686        node_id,
687        verdict,
688        observation.source,
689        &observation.path,
690        raw,
691    ) {
692        Ingest::Accepted
693    } else {
694        Ingest::Replay
695    }
696}
