1//! Discovery: how the repository's web surfaces are found without anybody listing them.
2//!
3//! Four inference sources, none of them a register a person maintains:
4//!
5//! 1. the **executable's own routes**, from the constants that already declare them once —
6//!    the capability prefix, the OpenAPI document, the Swagger UI, MCP and the Cockpit;
7//! 2. the **generated web root**, `target/web/<id>/`, where a producer writes its output
8//!    and a `surface.json` beside it declaring the intent no directory walk can infer;
9//! 3. the **site configuration**, which says the Zola application exists and where it is
10//!    built, so the application is discovered rather than assumed;
11//! 4. the **crate**, whose presence says its rustdoc exists as a surface whether or not
12//!    anybody has run `cargo doc` in this checkout, so the crate's reference is discovered
13//!    from the source it documents rather than from the output of one machine.
14//!
15//! Everything a consumer can be surprised by carries its [`Provenance`], so `web explain`
16//! answers "why is this here?" from data rather than from this module's source.
17

43/// What the running executable offers, which changes what it can answer for.
44///
45/// The static surfaces do not depend on it; the native ones do, because a process that
46/// serves no MCP endpoint must not advertise one.
47#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
48pub struct Runtime {
49    /// This process answers MCP over HTTP.
50    pub mcp: bool,
51    /// This process serves the Cockpit.
52    pub cockpit: bool,
53}
54
55impl Runtime {
56    /// Everything the executable can offer: what `web list` describes when it is not the
57    /// server itself, so a listing shows the topology rather than one process's slice.
58    pub fn full() -> Self {
59        Runtime {
60            mcp: true,
61            cockpit: true,
62        }
63    }
64
65    /// Does this process have the capability a surface needs?
66    ///
67    /// ```
68    /// use majordomus_cli::web::discover::Runtime;
69    /// use majordomus_cli::web::model::Feature;
70    /// assert!(Runtime::full().has(Feature::Cockpit));
71    /// assert!(!Runtime::default().has(Feature::Cockpit));
72    /// ```
73    pub fn has(self, feature: Feature) -> bool {
74        match feature {
75            Feature::Mcp => self.mcp,
76            Feature::Cockpit => self.cockpit,
77        }
78    }
79}

81/// A producer's declaration: the intent that no walk of its output can infer.
82///
83/// `id` and `mount` are the intent. Everything else has a documented default, because a
84/// field a producer must fill in to say what the convention already says is a field that
85/// will disagree with reality one day.
86///
87/// ```
88/// use majordomus_cli::web::discover::Declaration;
89/// let d: Declaration = serde_json::from_str(r#"{
90///     "schema": "web-surface/v1", "id": "tests", "mount": "/tests"
91/// }"#).unwrap();
92/// assert_eq!(d.id, "tests");
93/// assert_eq!(d.index.as_deref(), None);
94/// ```
95#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
96#[serde(deny_unknown_fields)]
97pub struct Declaration {
98    /// The contract this declaration follows.
99    pub schema: String,
100    /// The surface's identity, which is also its selector.
101    pub id: String,
102    /// Where the producer intends its output to be mounted.
103    pub mount: String,
104    /// One line for a listing; the id when absent.
105    #[serde(default, skip_serializing_if = "Option::is_none")]
106    pub title: Option<String>,
107    /// The file the mount itself answers with; `index.html` when absent.
108    #[serde(default, skip_serializing_if = "Option::is_none")]
109    pub index: Option<String>,
110    /// What produced the directory, for a reader who has to rebuild it.
111    #[serde(default, skip_serializing_if = "Option::is_none")]
112    pub producer: Option<String>,
113    /// Whether the surface is published, served, or both; both when absent.
114    #[serde(default, skip_serializing_if = "Option::is_none")]
115    pub availability: Option<Availability>,
116    /// What the surface is for; a generated directory is a report when it says nothing.
117    #[serde(default, skip_serializing_if = "Option::is_none")]
118    pub category: Option<Category>,
119    /// Whether a person is shown it; public when absent, because a producer that went to
120    /// the trouble of declaring a mount meant it to be found.
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    pub visibility: Option<Visibility>,
123    /// The revision the output was built from, when the producer knows it: what makes a
124    /// stale artifact a finding instead of a surprise.
125    #[serde(default, skip_serializing_if = "Option::is_none")]
126    pub built_from: Option<String>,
127}

129/// The whole topology of a repository: the executable's routes, the application, the
130/// crate's rustdoc and the generated reports.
131///
132/// Discovery never fails on an absent producer — a report that has not been generated is
133/// simply not a surface yet, and [`super::validate`] is where a *required* artifact being
134/// missing becomes a finding.
135pub fn discover(root: &Path, runtime: Runtime) -> Result<Topology> {
136    let mut surfaces = native(runtime);
137    surfaces.extend(application(root));
138    surfaces.extend(rustdoc(root));
139    surfaces.extend(generated(root)?);
140    Ok(Topology::new(surfaces))
141}

309/// A mount whose meaning is fixed: the role it serves, the path it is at, and the id of the
310/// surface that must hold it.

312pub struct Reserved {
313    /// What the mount is for, in one word; the key the generated topology publishes it under.
314    pub role: &'static str,
315    /// The path, exactly as the constant that declares it carries it.
316    pub path: &'static str,
317    /// The surface that must own it.
318    pub owner: &'static str,
319}

321/// The mounts this repository reserves.
322///
323/// A path is a name, and this repository has already paid once for a name meaning two
324/// things: the Swagger UI held `/docs` because the word was free, and the day documentation
325/// arrived the obvious path was occupied. These pairs are that decision written down where
326/// something reads it — the validator refuses a topology that breaks one, the generated
327/// topology publishes them, and the documentation renders what is published rather than
328/// restating it.
329///
330/// Each path is the same constant its surface is declared with, so a mount that moves moves
331/// here and the reservation follows it.
332///
333/// ```
334/// use majordomus_cli::web::discover::reserved;
335/// let docs = reserved().into_iter().find(|r| r.role == "documentation").unwrap();
336/// assert_eq!(docs.path, "/docs");
337/// assert_eq!(docs.owner, "docs");
338/// let swagger = reserved().into_iter().find(|r| r.role == "swagger").unwrap();
339/// assert_eq!(swagger.path, "/swagger");
340/// let rustdoc = reserved().into_iter().find(|r| r.owner == "rustdoc").unwrap();
341/// assert_eq!(rustdoc.path, "/rustdoc");
342/// ```
343pub fn reserved() -> Vec<Reserved> {
344    vec![
345        Reserved {
346            role: "home",
347            path: "/",
348            owner: HOME,
349        },
350        Reserved {
351            role: "documentation",
352            path: DOCS_MOUNT,
353            owner: DOCS,
354        },
355        // the crate's reference is documentation too, and a second surface answering its
356        // mount — a producer that decided the rustdoc should be copied somewhere "useful" —
357        // is the duplicate canonical route this refuses while the owner is gone as well
358        Reserved {
359            role: "rustdoc",
360            path: RUSTDOC_MOUNT,
361            owner: RUSTDOC,
362        },
363        Reserved {
364            role: "swagger",
365            path: swagger::SWAGGER_PATH,
366            owner: "swagger",
367        },
368        Reserved {
369            role: "openapi",
370            path: swagger::SPEC_PATH,
371            owner: "openapi",

373        Reserved {
374            role: "capabilities",
375            path: HttpExposure::PREFIX,
376            owner: "api",

378        Reserved {

457/// The crate's rustdoc, when this repository has the crate.
458///
459/// Declared from the source it documents, exactly as the documentation build is declared
460/// from the site's configuration: the crate existing is what says the reference exists,
461/// and whether `cargo doc` has run in this checkout decides only whether it is *built*.
462/// That is what keeps the topology — and the committed projection of it — the same on
463/// every machine; an unbuilt reference is served as a 503 naming [`RUSTDOC_PRODUCER`], and
464/// `web validate --artifacts` is where its absence becomes a finding.
465///
466/// Both worlds hold it. A running process serves it at a top-level mount of its own, and a
467/// publication carries it beside the site under the same path, nested under the
468/// application's root, which is the one nesting the validator allows. The revision is
469/// read from the declaration the producer writes beside its output; nothing else is,
470/// because every other value is decided here and a second statement of it could only
471/// disagree.
472///
473/// ```
474/// use majordomus_cli::web::discover::{rustdoc, RUSTDOC, RUSTDOC_MOUNT};
475/// let dir = tempfile::tempdir().unwrap();
476/// assert!(rustdoc(dir.path()).is_none(), "a repository without the crate documents nothing");
477/// std::fs::create_dir_all(dir.path().join("apps/majordomus-cli/src")).unwrap();
478/// std::fs::write(dir.path().join("apps/majordomus-cli/Cargo.toml"), "").unwrap();
479/// std::fs::write(dir.path().join("apps/majordomus-cli/src/lib.rs"), "//! x\n").unwrap();
480/// let surface = rustdoc(dir.path()).expect("the crate is documented");
481/// assert_eq!(surface.id, RUSTDOC);
482/// assert_eq!(surface.mount.as_str(), RUSTDOC_MOUNT);
483/// // declared before it is built: the producer has not run in this directory
484/// assert!(surface.built_from.is_none());
485/// assert!(surface.availability.is_served() && surface.availability.is_published());
486/// ```

488    crate::capability::builtin::quality::crate_dir(root)?;
489    Some(Surface {
490        id: RUSTDOC.into(),

492        category: Category::Documentation,
493        visibility: Visibility::Public,
494        kind: SurfaceKind::StaticDirectory,
495        mount: Mount::parse(RUSTDOC_MOUNT).expect("the rustdoc mount is a mount"),
496        producer: RUSTDOC_PRODUCER.into(),
497        feature: None,
498        artifact: Some(PathBuf::from(RUSTDOC_ARTIFACT)),
499        index: Some("index.html".into()),
500        availability: Availability::Both,
501        built_from: built_from(&root.join(RUSTDOC_ARTIFACT)),
502        provenance: provenance([
503            (
504                "kind",
505                Provenance::Filesystem {
506                    path: CRATE_DIR.into(),
507                },
508            ),
509            ("mount", Provenance::Default),

511                "artifact",
512                Provenance::Filesystem {
513                    path: RUSTDOC_ARTIFACT.into(),
514                },
515            ),
516        ]),
517    })

520/// The generated directories whose surface is declared by discovery rather than by the
521/// walk of the generated root: the documentation build, known from the site's
522/// configuration, and the rustdoc, known from the crate.
523///
524/// Each producer still writes a declaration beside its output, and that declaration
525/// carries the revision and nothing else the walk would add. Read as a surface of its own
526/// it would be the same surface twice — a duplicate id — and it would exist only in a
527/// checkout where the producer happened to run.
528const DECLARED_BY_DISCOVERY: [&str; 2] = [DOCS_ARTIFACT, RUSTDOC_ARTIFACT];
529
530/// The revision a producer recorded beside its output, when it recorded one.
531fn built_from(dir: &Path) -> Option<String> {
532    let text = std::fs::read_to_string(dir.join(DECLARATION_FILE)).ok()?;
533    let declaration: Declaration = serde_json::from_str(&text).ok()?;
534    declaration.built_from
535}

537/// Every generated static surface: one directory under the generated root, each declaring
538/// itself.
539///
540/// A directory without a declaration is not a surface — it is somebody's scratch space, and
541/// serving it because it happened to be there is how a repository publishes what it did not
542/// mean to.
543pub fn generated(root: &Path) -> Result<Vec<Surface>> {
544    let dir = root.join(GENERATED_ROOT);
545    let mut entries: Vec<PathBuf> = match std::fs::read_dir(&dir) {
546        Ok(read) => read
547            .filter_map(|e| e.ok())
548            .map(|e| e.path())
549            .filter(|p| p.is_dir())
550            .collect(),
551        Err(_) => return Ok(Vec::new()),
552    };
553    entries.sort();
554    let mut out = Vec::new();
555    for path in entries {
556        let declaration = path.join(DECLARATION_FILE);
557        if !declaration.is_file() {
558            continue;
559        }
560        // the documentation build and the rustdoc are discovered from what says they should
561        // exist even when they have not been built; their declarations carry the revision
562        // and nothing else this walk would add
563        let relative = path.file_name().map(|n| Path::new(GENERATED_ROOT).join(n));
564        if relative.is_some_and(|r| DECLARED_BY_DISCOVERY.iter().any(|d| r == Path::new(d))) {
565            continue;
566        }
567        let rel = format!(
568            "{GENERATED_ROOT}/{}/{DECLARATION_FILE}",
569            path.file_name().unwrap_or_default().to_string_lossy()
570        );
571        let text = std::fs::read_to_string(&declaration).map_err(|e| Error::InvalidSurface {
572            surface: rel.clone(),
573            reason: format!("cannot be read: {e}"),
574        })?;
575        let decl: Declaration = serde_json::from_str(&text).map_err(|e| Error::InvalidSurface {
576            surface: rel.clone(),
577            reason: format!("does not parse: {e}; the contract is {DECLARATION_SCHEMA}"),
578        })?;
579        if decl.schema != DECLARATION_SCHEMA {
580            return Err(Error::InvalidSurface {
581                surface: rel.clone(),
582                reason: format!(
583                    "declares schema '{}', and this executable reads {DECLARATION_SCHEMA}",
584                    decl.schema
585                ),
586            });
587        }
588        let mount = Mount::parse(&decl.mount).map_err(|reason| Error::InvalidSurface {
589            surface: decl.id.clone(),
590            reason: format!("{reason} (declared in {rel})"),
591        })?;
592        let artifact = PathBuf::from(GENERATED_ROOT).join(path.file_name().ok_or_else(|| {
593            Error::InvalidSurface {
594                surface: decl.id.clone(),
595                reason: format!("the directory of {rel} has no name"),
596            }
597        })?);
598        out.push(Surface {
599            id: decl.id.clone(),
600            title: decl.title.unwrap_or_else(|| decl.id.clone()),
601            category: decl.category.unwrap_or(Category::Report),
602            visibility: decl.visibility.unwrap_or(Visibility::Public),
603            kind: SurfaceKind::StaticDirectory,
604            mount,
605            producer: decl.producer.unwrap_or_else(|| "unknown".into()),
606            feature: None,
607            artifact: Some(artifact),
608            index: Some(decl.index.unwrap_or_else(|| "index.html".into())),
609            availability: decl.availability.unwrap_or(Availability::Both),
610            built_from: decl.built_from,
611            provenance: provenance([
612                (
613                    "mount",
614                    Provenance::ProducerDeclaration { path: rel.clone() },
615                ),
616                ("kind", Provenance::Filesystem { path: rel.clone() }),
617                ("artifact", Provenance::Filesystem { path: rel }),
618            ]),
619        });
620    }
621    Ok(out)
622}
623
624fn provenance<const N: usize>(pairs: [(&str, Provenance); N]) -> BTreeMap<String, Provenance> {
625    pairs.into_iter().map(|(k, v)| (k.to_string(), v)).collect()
626}
