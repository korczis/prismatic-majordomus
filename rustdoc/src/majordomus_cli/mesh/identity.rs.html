1//! Node identity: who this machine's Majordomus is, independently of any repository,
2//! process, address or port. The identity is an Ed25519 keypair created once per user
3//! and machine and kept under the user's state directory; the node id is a digest of
4//! the public key, so claiming an id means holding the key, and a hostname, an IP
5//! address or a PID is never identity — those are runtime attributes that change.
6//!
7//! The signing key never leaves this module: everything above it sees [`NodeIdentity`],
8//! which signs and exposes public material only. The instance id distinguishes runs of
9//! the same node — a machine that restarts keeps its node id and changes its instance.
10//!
11//! ```
12//! use majordomus_cli::mesh::identity::{verify, NodeIdentity};
13//!
14//! let node = NodeIdentity::ephemeral().unwrap();
15//! let signature = node.sign(b"a majordomus exists here");
16//! assert!(verify(&node.public.public_key, b"a majordomus exists here", &signature));
17//! assert!(!verify(&node.public.public_key, b"something else", &signature));
18//! ```
19

30/// The stable identity of a node: 32 hex characters of the SHA-256 of its Ed25519 public
31/// key. Derived, never chosen: two nodes with one id hold one key.
32#[derive(
33    Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
34)]
35#[serde(transparent)]
36pub struct NodeId(String);
37
38impl NodeId {
39    /// The id of a public key.
40    pub fn of(key: &VerifyingKey) -> Self {
41        let mut hasher = Sha256::new();
42        hasher.update(key.as_bytes());
43        let digest = hasher.finalize();
44        let mut hex = String::with_capacity(32);
45        for byte in digest.iter().take(16) {
46            hex.push_str(&format!("{byte:02x}"));
47        }
48        NodeId(hex)
49    }
50
51    /// The id as text: 32 lowercase hex characters, the form listings and allowlists carry.
52    pub fn as_str(&self) -> &str {
53        &self.0
54    }
55}
56
57impl fmt::Display for NodeId {
58    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
59        f.write_str(&self.0)
60    }
61}

63/// One run of a node: 16 hex characters of OS entropy, new at every process start. A
64/// restart keeps the node id and changes this, which is how a reader tells "the same
65/// node came back" from "the same process is still there".
66#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
67#[serde(transparent)]
68pub struct InstanceId(String);
69
70impl InstanceId {
71    /// A fresh instance id from OS entropy.
72    pub fn fresh() -> Result<Self, MeshError> {
73        let mut bytes = [0u8; 8];
74        getrandom::getrandom(&mut bytes)
75            .map_err(|e| MeshError::Identity(format!("no OS entropy for an instance id: {e}")))?;
76        Ok(InstanceId(hex(&bytes)))
77    }
78
79    /// The id as text: 16 hex characters, stable for the life of this process alone.
80    pub fn as_str(&self) -> &str {
81        &self.0
82    }
83}
84
85impl fmt::Display for InstanceId {
86    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
87        f.write_str(&self.0)
88    }
89}

91/// What everyone may know about this node: the id, the public key, a display name and
92/// this run's instance. Everything an advertisement carries; nothing that signs.
93#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
94pub struct PublicIdentity {
95    /// The node id: a digest of the public key.
96    pub node_id: NodeId,
97    /// The Ed25519 public key, 64 hex characters. Whoever holds the matching secret is
98    /// this node; there is no other proof.
99    pub public_key: String,
100    /// A human name for the node, from the identity file; defaults to the first eight
101    /// characters of the node id. Presentation only — never used for trust.
102    pub display_name: String,
103    /// This run of the node.
104    pub instance_id: InstanceId,
105}
106
107/// The identity a running mesh signs with: the public half plus the signing key.
108pub struct NodeIdentity {
109    /// The public half, as an advertisement carries it.
110    pub public: PublicIdentity,
111    signing: SigningKey,
112}
113
114impl fmt::Debug for NodeIdentity {
115    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
116        // The signing key stays out of every rendering, Debug included.
117        f.debug_struct("NodeIdentity")
118            .field("public", &self.public)
119            .finish_non_exhaustive()
120    }
121}

123/// The identity file: schema, the secret seed, an optional display name. Mode 0600,
124/// under the user's state directory, never inside a repository.
125#[derive(Debug, Serialize, Deserialize)]
126struct IdentityFile {
127    schema: String,
128    secret_key: String,
129    #[serde(default, skip_serializing_if = "Option::is_none")]
130    display_name: Option<String>,
131    created_at: String,
132}
133
134/// The identity file's format version.
135const SCHEMA: &str = "majordomus-mesh-identity/v1";
136
137/// Where the identity lives: `$XDG_STATE_HOME/majordomus/node.json`, else
138/// `$HOME/.local/state/majordomus/node.json`. `None` when neither variable is set —
139/// a process with no home has no node identity, and the mesh stays off.
140pub fn default_identity_path() -> Option<PathBuf> {
141    if let Some(state) = std::env::var_os("XDG_STATE_HOME").filter(|v| !v.is_empty()) {
142        return Some(PathBuf::from(state).join("majordomus").join("node.json"));
143    }
144    std::env::var_os("HOME")
145        .filter(|v| !v.is_empty())
146        .map(|home| {
147            PathBuf::from(home)
148                .join(".local")
149                .join("state")
150                .join("majordomus")
151                .join("node.json")
152        })
153}
154
155impl NodeIdentity {
156    /// Load the identity at `path`, creating it (directories included) when absent.
157    /// A malformed file is an error, never silently replaced: an identity is the one
158    /// thing a node must not lose by accident.
159    pub fn load_or_create(path: &Path) -> Result<Self, MeshError> {
160        if path.is_file() {
161            return Self::load(path);
162        }
163        let identity = Self::create()?;
164        identity.persist(path)?;
165        Ok(identity)
166    }
167
168    fn load(path: &Path) -> Result<Self, MeshError> {
169        let text = std::fs::read_to_string(path)
170            .map_err(|e| MeshError::Identity(format!("{}: {e}", path.display())))?;
171        let file: IdentityFile = serde_json::from_str(&text).map_err(|e| {
172            MeshError::Identity(format!("{}: not an identity file: {e}", path.display()))
173        })?;
174        if file.schema != SCHEMA {
175            return Err(MeshError::Identity(format!(
176                "{}: schema {} is not {SCHEMA}",
177                path.display(),
178                file.schema
179            )));
180        }
181        let seed = unhex(&file.secret_key).ok_or_else(|| {
182            MeshError::Identity(format!("{}: the secret key is not hex", path.display()))
183        })?;
184        let seed: [u8; 32] = seed.try_into().map_err(|_| {
185            MeshError::Identity(format!(
186                "{}: the secret key is not 32 bytes",
187                path.display()
188            ))
189        })?;
190        let signing = SigningKey::from_bytes(&seed);
191        Self::assemble(signing, file.display_name)
192    }
193
194    /// A throwaway identity that touches no disk: what the self-check and tests sign
195    /// with. Never a node's real identity — nothing persists it.
196    pub fn ephemeral() -> Result<Self, MeshError> {
197        Self::create()
198    }
199
200    fn create() -> Result<Self, MeshError> {
201        let mut seed = [0u8; 32];
202        getrandom::getrandom(&mut seed)
203            .map_err(|e| MeshError::Identity(format!("no OS entropy for a node key: {e}")))?;
204        let signing = SigningKey::from_bytes(&seed);
205        Self::assemble(signing, None)
206    }
207
208    fn assemble(signing: SigningKey, display_name: Option<String>) -> Result<Self, MeshError> {
209        let verifying = signing.verifying_key();
210        let node_id = NodeId::of(&verifying);
211        let display_name =
212            display_name.unwrap_or_else(|| node_id.as_str().chars().take(8).collect());
213        Ok(NodeIdentity {
214            public: PublicIdentity {
215                node_id,
216                public_key: hex(verifying.as_bytes()),
217                display_name,
218                instance_id: InstanceId::fresh()?,
219            },
220            signing,
221        })
222    }
223
224    fn persist(&self, path: &Path) -> Result<(), MeshError> {
225        if let Some(dir) = path.parent() {
226            std::fs::create_dir_all(dir)
227                .map_err(|e| MeshError::Identity(format!("{}: {e}", dir.display())))?;
228        }
229        let file = IdentityFile {
230            schema: SCHEMA.into(),
231            secret_key: hex(&self.signing.to_bytes()),
232            display_name: Some(self.public.display_name.clone()),
233            created_at: crate::peers::rfc3339(std::time::SystemTime::now()),
234        };
235        let text =
236            serde_json::to_string_pretty(&file).map_err(|e| MeshError::Identity(e.to_string()))?;
237        write_private(path, &text)
238    }
239
240    /// Sign a message with this node's key.
241    pub fn sign(&self, message: &[u8]) -> String {
242        hex(&self.signing.sign(message).to_bytes())
243    }
244}

246/// Verify `signature` (hex) over `message` against `public_key` (hex). False on any
247/// malformed material: a verifier never panics on hostile input.
248///
249/// ```
250/// use majordomus_cli::mesh::identity::verify;
251/// assert!(!verify("zz", b"m", "zz"), "garbage material is false, not a panic");
252/// ```
253pub fn verify(public_key: &str, message: &[u8], signature: &str) -> bool {
254    let Some(key) = unhex(public_key).and_then(|b| <[u8; 32]>::try_from(b).ok()) else {
255        return false;
256    };
257    let Ok(key) = VerifyingKey::from_bytes(&key) else {
258        return false;
259    };
260    let Some(sig) = unhex(signature).and_then(|b| <[u8; 64]>::try_from(b).ok()) else {
261        return false;
262    };
263    key.verify(message, &Signature::from_bytes(&sig)).is_ok()
264}
265
266/// The node id a public key (hex) yields, when the key parses.
267pub fn node_id_of_key(public_key: &str) -> Option<NodeId> {
268    let key = unhex(public_key).and_then(|b| <[u8; 32]>::try_from(b).ok())?;
269    let key = VerifyingKey::from_bytes(&key).ok()?;
270    Some(NodeId::of(&key))
271}
272
273fn hex(bytes: &[u8]) -> String {
274    let mut out = String::with_capacity(bytes.len() * 2);
275    for byte in bytes {
276        out.push_str(&format!("{byte:02x}"));
277    }
278    out
279}
280
281fn unhex(text: &str) -> Option<Vec<u8>> {
282    if text.len() % 2 != 0 {
283        return None;
284    }
285    (0..text.len())
286        .step_by(2)
287        .map(|i| u8::from_str_radix(text.get(i..i + 2)?, 16).ok())
288        .collect()
289}

291/// Write `text` to `path` with owner-only permissions, atomically enough for one user's
292/// state directory: a temp file beside the target, then a rename.
293fn write_private(path: &Path, text: &str) -> Result<(), MeshError> {
294    let tmp = path.with_extension("tmp");
295    std::fs::write(&tmp, text)
296        .map_err(|e| MeshError::Identity(format!("{}: {e}", tmp.display())))?;
297    #[cfg(unix)]
298    {
299        use std::os::unix::fs::PermissionsExt;
300        let _ = std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o600));
301    }
302    std::fs::rename(&tmp, path).map_err(|e| MeshError::Identity(format!("{}: {e}", path.display())))
303}

309    #[test]
310    fn an_identity_is_created_once_and_loaded_stable() {
311        let dir = tempfile::tempdir().unwrap();
312        let path = dir.path().join("node.json");
313        let first = NodeIdentity::load_or_create(&path).unwrap();
314        let second = NodeIdentity::load_or_create(&path).unwrap();
315        assert_eq!(
316            first.public.node_id, second.public.node_id,
317            "the node id survives a reload"
318        );
319        assert_ne!(
320            first.public.instance_id, second.public.instance_id,
321            "every run is a new instance"
322        );
323    }
324
325    #[test]
326    fn a_signature_verifies_and_a_tampered_message_does_not() {
327        let dir = tempfile::tempdir().unwrap();
328        let identity = NodeIdentity::load_or_create(&dir.path().join("node.json")).unwrap();
329        let sig = identity.sign(b"hello mesh");
330        assert!(verify(&identity.public.public_key, b"hello mesh", &sig));
331        assert!(!verify(&identity.public.public_key, b"hello mess", &sig));
332        assert!(
333            !verify("zz", b"hello mesh", &sig),
334            "malformed key material is false, not a panic"
335        );
336        assert!(!verify(&identity.public.public_key, b"hello mesh", "zz"));
337    }
338
339    #[test]
340    fn the_node_id_is_the_digest_of_the_key() {
341        let dir = tempfile::tempdir().unwrap();
342        let identity = NodeIdentity::load_or_create(&dir.path().join("node.json")).unwrap();
343        assert_eq!(
344            node_id_of_key(&identity.public.public_key),
345            Some(identity.public.node_id.clone())
346        );
347        assert_eq!(identity.public.node_id.as_str().len(), 32);
348    }

350    #[test]
351    fn a_malformed_identity_file_is_an_error_not_a_replacement() {
352        let dir = tempfile::tempdir().unwrap();
353        let path = dir.path().join("node.json");
354        std::fs::write(&path, "not json").unwrap();
355        assert!(NodeIdentity::load_or_create(&path).is_err());
356        assert_eq!(
357            std::fs::read_to_string(&path).unwrap(),
358            "not json",
359            "the file is untouched"
360        );
361    }
