1//! Whether the trunk is released: how much landed work it holds past the last release, since
2//! when, and at which version (I3070).
3//!
4//! The cooperation model's fourth rule says the release seat releases whenever the trunk holds
5//! landed work past the last tag (`project.the-cooperation-model-is-the-owners-word`), and the
6//! owner's order of 2026-10-19 asks for a measured version after every session and milestone.
7//! Neither was measured. The trunk could hold a day of landed work past the last release while
8//! every check, the brief, doctor and the delivery screen stayed green. This module is the one
9//! measurement: `release.currency` answers it, and every surface that shows it reads that answer.
10//!
11//! Three readings make it, each of git as it is when asked:
12//!
13//! - **The release.** The newer of two: the release the layer records
14//!   ([`super::compat::last_release`]), and the newest version tag the trunk contains. A tag is
15//!   pushed before its record lands, so a tag newer than every record is a release that has
16//!   happened and is not recorded yet.
17//! - **The landed work.** Every commit the trunk holds past that release, except merges and
18//!   release commits (`chore(release)`): what the next release would ship.
19//! - **How long it has waited.** A landing is a commit of the trunk's first-parent history: a
20//!   merge of a branch or a top, or a commit made on the trunk itself. The work has waited since
21//!   the oldest landing that brought anything but release commits arrived on the trunk: when
22//!   this clone's remote-tracking trunk first held it, from that ref's reflog. A top is
23//!   composed for hours before it is pushed, so its commit time says when it was made, not when
24//!   it landed. Where the reflog does not show the landing arrive, the commit time is used as
25//!   what it is, an upper bound on the wait, and alone it never makes the work overdue
26//!   ([`LandedAt`]).
27//!
28//! The answer reads refs and the clock, so it is a runtime read and is never committed. A
29//! committed derivation that read refs would go stale with every tag.
30//!
31//! ```
32//! use majordomus_cli::release::currency::{measure, Currency, Stance, OVERDUE_AFTER};
33//! use std::time::{Duration, SystemTime};
34//!
35//! // a directory with no trunk is not measured, and the answer says why
36//! let nowhere = tempfile::tempdir().unwrap();
37//! let answer: Currency = measure(nowhere.path(), &[], SystemTime::now());
38//! assert_eq!(answer.stance, Stance::Unknown);
39//!
40//! // on a trunk, the stance follows the landed work and how long it has waited
41//! assert_eq!(Stance::of(0, OVERDUE_AFTER), Stance::Current);
42//! assert_eq!(Stance::of(3, Duration::from_secs(600)), Stance::Owed);
43//! assert_eq!(Stance::of(3, OVERDUE_AFTER), Stance::Overdue);
44//! ```

58/// How long landed work may wait past the last release before it is overdue: six hours, the
59/// release seat's word (I3070). That is longer than a session, and shorter than the day of
60/// landed work the owner's order closes.
61pub const OVERDUE_AFTER: Duration = Duration::from_secs(6 * 60 * 60);
62
63/// The most landed changes an answer lists. The count is exact however long the list.
64pub const LISTED: usize = 50;

66/// Whether the trunk's landed work is released.
67///
68/// ```
69/// use majordomus_cli::release::currency::{Stance, OVERDUE_AFTER};
70/// use std::time::Duration;
71/// assert_eq!(Stance::of(0, OVERDUE_AFTER), Stance::Current, "nothing landed, nothing owed");
72/// assert_eq!(Stance::of(2, Duration::from_secs(60)), Stance::Owed);
73/// assert_eq!(Stance::of(2, OVERDUE_AFTER), Stance::Overdue);
74/// assert_eq!(Stance::Overdue.as_str(), "overdue");
75/// ```
76#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
77#[serde(rename_all = "snake_case")]
78#[schemars(rename = "ReleaseCurrencyStance")]
79pub enum Stance {
80    /// Nothing has landed past the last release.
81    Current,
82    /// Work has landed past it, and has waited less than [`OVERDUE_AFTER`].
83    Owed,
84    /// Work has landed past it, and has waited at least [`OVERDUE_AFTER`].
85    Overdue,
86    /// It could not be measured: no trunk, no release, or a release this clone cannot read.
87    Unknown,
88}
89
90impl Stance {
91    /// The stance of `landed` changes, the oldest of which has waited `waited`.
92    ///
93    /// ```
94    /// use majordomus_cli::release::currency::{Stance, OVERDUE_AFTER};
95    /// assert_eq!(Stance::of(1, OVERDUE_AFTER / 2), Stance::Owed);
96    /// ```
97    pub fn of(landed: usize, waited: Duration) -> Stance {
98        match (landed, waited >= OVERDUE_AFTER) {
99            (0, _) => Stance::Current,
100            (_, false) => Stance::Owed,
101            (_, true) => Stance::Overdue,
102        }
103    }
104
105    /// The stance as one word, as the brief, doctor and the delivery screen say it.
106    ///
107    /// ```
108    /// use majordomus_cli::release::currency::Stance;
109    /// assert_eq!(Stance::Current.as_str(), "current");
110    /// ```
111    pub fn as_str(self) -> &'static str {
112        match self {
113            Stance::Current => "current",
114            Stance::Owed => "owed",
115            Stance::Overdue => "overdue",
116            Stance::Unknown => "unknown",
117        }
118    }
119}

121/// How the oldest landing was dated, and so which way its wait may be wrong (the plan
122/// critique of 2026-10-19: a top lands by a fast-forward push hours after its commits).
123///
124/// ```
125/// use majordomus_cli::release::currency::LandedAt;
126/// assert_eq!(serde_json::to_value(LandedAt::Reflog).unwrap(), "reflog");
127/// assert_eq!(serde_json::to_value(LandedAt::Commit).unwrap(), "commit");
128/// ```
129#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
130#[serde(rename_all = "snake_case")]
131#[schemars(rename = "ReleaseCurrencyLandedAt")]
132pub enum LandedAt {
133    /// When this clone's remote-tracking trunk first held it, from that ref's reflog. A landing
134    /// is seen no earlier than it lands, so the wait counted from here is a lower bound, and it
135    /// is the one date that can make the stance overdue.
136    Reflog,
137    /// Its commit time, where the reflog does not show it arrive: a commit lands no earlier than
138    /// it is made, so the wait counted from here is an upper bound, and it never makes the
139    /// stance overdue by itself.
140    Commit,
141}

143/// The release the trunk is measured from.
144///
145/// ```
146/// use majordomus_cli::release::currency::Released;
147/// let tagged = Released {
148///     version: "0.30.1".into(),
149///     reference: "v0.30.1".into(),
150///     at: "v0.30.1".into(),
151///     recorded: false,
152/// };
153/// // a tag pushed before its record landed is read at the tag itself
154/// assert_eq!(tagged.at, tagged.reference);
155/// ```
156#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
157#[schemars(rename = "ReleaseCurrencyRelease")]
158pub struct Released {
159    /// `0.30.1`.
160    pub version: String,
161    /// `v0.30.1`: the release as a reader recognises it.
162    pub reference: String,
163    /// What the trunk is measured from: the commit its record names, else its tag.
164    pub at: String,
165    /// Whether the layer holds a record for it yet.
166    pub recorded: bool,
167}

169/// The answer of `release.currency`: whether the trunk's landed work is released.
170///
171/// ```
172/// use majordomus_cli::release::currency::{measure, Currency, Stance};
173/// let nowhere = tempfile::tempdir().unwrap();
174/// let answer: Currency = measure(nowhere.path(), &[], std::time::SystemTime::now());
175/// // a directory with no trunk is not measured, and says why
176/// assert_eq!(answer.stance, Stance::Unknown);
177/// assert!(answer.summary.contains("trunk"), "{}", answer.summary);
178/// ```
179#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
180#[schemars(rename = "ReleaseCurrency")]
181pub struct Currency {
182    /// Whether the landed work is released.
183    pub stance: Stance,
184    /// The answer in one sentence: the brief's and doctor's line.
185    pub summary: String,
186    /// The remote-tracking ref of the trunk it was measured on (`origin/master`).
187    #[serde(default, skip_serializing_if = "Option::is_none")]
188    pub trunk: Option<String>,
189    /// The last release.
190    #[serde(default, skip_serializing_if = "Option::is_none")]
191    pub release: Option<Released>,
192    /// Landed commits past the last release, merges and release commits left out.
193    pub landed: usize,
194    /// The landings of the trunk's first-parent history that brought them.
195    pub landings: usize,
196    /// When the oldest of those landings landed, RFC 3339, dated as [`landed_at`](Self::landed_at)
197    /// says.
198    #[serde(default, skip_serializing_if = "Option::is_none")]
199    pub oldest_landing: Option<String>,
200    /// How [`oldest_landing`](Self::oldest_landing) was dated: when this clone saw it arrive
201    /// on the trunk, or its commit time as an upper bound on the wait.
202    #[serde(default, skip_serializing_if = "Option::is_none")]
203    pub landed_at: Option<LandedAt>,
204    /// How long the work has waited since then, in seconds.
205    #[serde(default, skip_serializing_if = "Option::is_none")]
206    pub waited_seconds: Option<u64>,
207    /// How long it may wait before it is overdue ([`OVERDUE_AFTER`]).
208    pub overdue_after_seconds: u64,
209    /// The version the public contract requires for the landed work (ADR 0051), when it was
210    /// decided.
211    #[serde(default, skip_serializing_if = "Option::is_none")]
212    pub next: Option<String>,
213    /// Why that version was not decided, when it was asked for and could not be.
214    #[serde(default, skip_serializing_if = "Option::is_none")]
215    pub next_unmeasured: Option<String>,
216    /// The landed changes, newest first, at most [`LISTED`].
217    #[serde(default, skip_serializing_if = "Vec::is_empty")]
218    pub changes: Vec<Change>,
219}

231            landed_at: None,
232            waited_seconds: None,
233            overdue_after_seconds: OVERDUE_AFTER.as_secs(),
234            next: None,
235            next_unmeasured: None,
236            changes: Vec::new(),
237        }
238    }

241/// Measure the trunk of the repository at `root` against its last release, as of `now`. Git
242/// is read as it is, the remote-tracking trunk as of the last fetch. The version the contract
243/// requires is not decided here ([`decide_next`]), because that reads the last release's
244/// registry, and the brief and doctor do not need it.
245///
246/// ```
247/// use majordomus_cli::release::currency::{measure, Stance};
248/// let nowhere = tempfile::tempdir().unwrap();
249/// let answer = measure(nowhere.path(), &[], std::time::SystemTime::now());
250/// assert_eq!((answer.stance, answer.landed), (Stance::Unknown, 0));
251/// ```
252pub fn measure(root: &Path, objects: &[Object], now: SystemTime) -> Currency {
253    let trunk = match crate::delivery::revision::trunk_ref(root) {
254        Ok(trunk) => trunk,
255        Err(why) => return Currency::unknown(None, format!("the trunk cannot be told: {why}")),
256    };
257    let release = newest(
258        super::compat::last_release(root, objects),
259        newest_tag_on(root, &trunk),
260    );
261    let Some(release) = release.filter(|r| resolves(root, &r.at)) else {
262        return Currency::unknown(
263            Some(trunk.clone()),
264            format!(
265                "no release to measure {trunk} from: no release record or version tag whose commit this clone holds"
266            ),
267        );
268    };
269    let range = format!("{}..{trunk}", release.at);
270    let changes: Vec<Change> = super::commits::in_range(root, &range, objects)
271        .into_iter()
272        .filter(|c| !is_release(c))
273        .collect();
274    let landed = changes.len();
275    let landings = landings(root, &range);
276    let dated = landings
277        .first()
278        .filter(|_| landed > 0)
279        .map(|(sha, committed)| match first_seen(root, &trunk, sha) {
280            Some(seen) => (LandedAt::Reflog, seen),
281            None => (LandedAt::Commit, *committed),
282        });
283    let now = now.duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
284    let waited = dated.map(|(_, at)| Duration::from_secs(now.saturating_sub(at)));
285    let stance = match (Stance::of(landed, waited.unwrap_or_default()), dated) {
286        // a commit time bounds the wait from above: it says the work may be overdue, never that
287        // it is, so a top composed hours before it landed does not read overdue as it lands
288        (Stance::Overdue, Some((LandedAt::Commit, _))) => Stance::Owed,
289        (stance, _) => stance,
290    };
291    let how = dated.map(|(how, _)| how);
292    Currency {
293        summary: summary(
294            stance,
295            landed,
296            landings.len(),
297            &release,
298            &trunk,
299            waited,
300            how,
301        ),
302        stance,
303        trunk: Some(trunk),
304        release: Some(release),
305        landed,
306        landings: landings.len(),
307        oldest_landing: dated
308            .map(|(_, at)| crate::peers::rfc3339(UNIX_EPOCH + Duration::from_secs(at))),
309        landed_at: how,
310        waited_seconds: waited.map(|w| w.as_secs()),
311        overdue_after_seconds: OVERDUE_AFTER.as_secs(),
312        next: None,
313        next_unmeasured: None,
314        changes: changes.into_iter().take(LISTED).collect(),
315    }
316}

318/// Decide the version the public contract requires for the landed work, where it can be
319/// measured: the release the trunk is measured from must be recorded, because the contract is
320/// measured from a recorded release (ADR 0051), and this checkout must be at the trunk, because
321/// the contract is this tree's. Otherwise `next_unmeasured` says which of the two it is not.
322/// Nothing is decided when nothing has landed.
323///
324/// ```
325/// use majordomus_cli::capability::registry::CapabilityRegistry;
326/// use majordomus_cli::release::currency::{decide_next, measure};
327/// let nowhere = tempfile::tempdir().unwrap();
328/// let mut answer = measure(nowhere.path(), &[], std::time::SystemTime::now());
329/// let registry = CapabilityRegistry::builder().build().unwrap();
330/// decide_next(&mut answer, nowhere.path(), &registry, &[]);
331/// assert_eq!((answer.next, answer.next_unmeasured), (None, None), "nothing landed, nothing decided");
332/// ```
333pub fn decide_next(
334    answer: &mut Currency,
335    root: &Path,
336    registry: &CapabilityRegistry,
337    objects: &[Object],
338) {
339    let Some(release) = answer.release.as_ref().filter(|_| answer.landed > 0) else {
340        return;
341    };
342    let why = if release.recorded {
343        away_from_trunk(root, answer.trunk.as_deref().unwrap_or_default())
344    } else {
345        Some(format!(
346            "{} has no release record yet, and the contract is measured from a recorded release",
347            release.reference
348        ))
349    };
350    match why {
351        Some(why) => answer.next_unmeasured = Some(why),
352        None => {
353            let report = super::version::select(root, registry, objects).report;
354            answer.next = report.next;
355            answer.next_unmeasured = report.contract_unreadable;
356        }
357    }
358}

360/// The newer of the recorded release and the newest tag the trunk contains.
361fn newest(recorded: Option<super::compat::LastRelease>, tag: Option<String>) -> Option<Released> {
362    let from_record = recorded.map(|r| Released {
363        version: r.version,
364        reference: r.reference,
365        at: r.read_at,
366        recorded: r.recorded,
367    });
368    let from_tag = tag.map(|t| Released {
369        version: t.trim_start_matches('v').to_string(),
370        reference: t.clone(),
371        at: t,
372        recorded: false,
373    });
374    match (from_record, from_tag) {
375        (Some(r), Some(t)) if Version::parse(&t.version) > Version::parse(&r.version) => Some(t),
376        (Some(r), _) => Some(r),
377        (None, t) => t,
378    }
379}

381/// The newest version tag whose commit the trunk contains.
382fn newest_tag_on(root: &Path, trunk: &str) -> Option<String> {
383    super::compat::tags(root)
384        .into_iter()
385        .rev()
386        .find(|tag| succeeds(root, &["merge-base", "--is-ancestor", tag, trunk]))
387}

389/// Whether `rev` names a commit this clone holds.
390fn resolves(root: &Path, rev: &str) -> bool {
391    succeeds(
392        root,
393        &[
394            "rev-parse",
395            "--verify",
396            "--quiet",
397            &format!("{rev}^{{commit}}"),
398        ],
399    )

402/// Why this checkout is not where `trunk` is, when it is not.
403fn away_from_trunk(root: &Path, trunk: &str) -> Option<String> {
404    let head = git(root, &["rev-parse", "HEAD"]);
405    let there = git(root, &["rev-parse", trunk]);
406    (head.trim() != there.trim()).then(|| {
407        format!(
408            "this checkout is at {}, not at {trunk} ({}): the contract is measured where the trunk is checked out",
409            short(&head),
410            short(&there)
411        )
412    })
413}

420/// A release commit: what releasing writes (`chore(release): ...`), which no release ships.
421fn is_release(change: &Change) -> bool {
422    change.kind == ChangeKind::Chore && change.scope.as_deref() == Some("release")
423}

425/// One landing of the trunk's first-parent history.
426#[derive(Debug, PartialEq, Eq)]
427struct Landing<'a> {
428    sha: &'a str,
429    /// Its committer time, in seconds since the epoch: when it was made, which is no later than
430    /// when it landed.
431    at: u64,
432    merge: bool,
433}
434
435/// The landings of `log` (`--format=%H %ct %P`), in its order. A line that is not one is left
436/// out rather than guessed.
437fn first_parents(log: &str) -> Vec<Landing<'_>> {
438    log.lines()
439        .filter_map(|line| {
440            let mut words = line.split(' ');
441            let sha = words.next().filter(|w| !w.is_empty())?;
442            let at = words.next()?.parse().ok()?;
443            Some(Landing {
444                sha,
445                at,
446                merge: words.count() > 1,
447            })
448        })
449        .collect()
450}

452/// Every landing in `range` that brought anything but release commits, oldest first: its
453/// commit and its commit time.
454fn landings(root: &Path, range: &str) -> Vec<(String, u64)> {
455    let log = git(
456        root,
457        &[
458            "log",
459            "--first-parent",
460            "--reverse",
461            "--format=%H %ct %P",
462            range,
463        ],
464    );
465    first_parents(&log)
466        .into_iter()
467        .filter(|l| brings_work(root, l))
468        .map(|l| (l.sha.to_string(), l.at))
469        .collect()
470}

472/// When this clone's `trunk` first held `sha`, from the trunk's reflog: the time of the oldest
473/// entry of the run of entries, newest first, that contain it. `None` when no entry contains
474/// it, or when every entry does, because then the reflog does not show it arrive (the clone
475/// was made after it landed, or its reflog was expired).
476fn first_seen(root: &Path, trunk: &str, sha: &str) -> Option<u64> {
477    let log = git(
478        root,
479        &["reflog", "show", "--date=unix", "--format=%H %gd", trunk],
480    );
481    let mut seen = None;
482    for (held, at) in reflog_entries(&log) {
483        if succeeds(root, &["merge-base", "--is-ancestor", sha, held]) {
484            seen = Some(at);
485        } else {
486            return seen;
487        }
488    }
489    None
490}
491
492/// The entries of `git reflog show --date=unix --format='%H %gd'`, newest first, as git lists
493/// them: the commit each entry set the ref to, and when, in seconds since the epoch. A line that
494/// is not one is left out rather than guessed.
495fn reflog_entries(log: &str) -> Vec<(&str, u64)> {
496    log.lines()
497        .filter_map(|line| {
498            let (sha, selector) = line.split_once(' ')?;
499            let at = selector
500                .rsplit_once("@{")?
501                .1
502                .strip_suffix('}')?
503                .parse()
504                .ok()?;
505            Some((sha, at))
506        })
507        .collect()
508}

510/// Whether a landing brought a commit that is not a release commit: a merge, every commit its
511/// second parent added; a commit made on the trunk, itself.
512fn brings_work(root: &Path, landing: &Landing<'_>) -> bool {
513    let range = if landing.merge {
514        format!("{0}^1..{0}", landing.sha)
515    } else {
516        format!("{}^!", landing.sha)
517    };
518    git(root, &["log", "--no-merges", "--format=%s", &range])
519        .lines()
520        .any(|subject| !is_release(&super::commits::parse("", subject, "")))
521}

523fn summary(
524    stance: Stance,
525    landed: usize,
526    landings: usize,
527    release: &Released,
528    trunk: &str,
529    waited: Option<Duration>,
530    how: Option<LandedAt>,
531) -> String {
532    let unrecorded = if release.recorded {
533        ""
534    } else {
535        " (tagged, its record not landed yet)"
536    };
537    match stance {
538        Stance::Current => format!(
539            "{trunk} holds no landed work past {}{unrecorded}",
540            release.reference
541        ),
542        _ => {
543            let waited = waited.unwrap_or_default();
544            let mut s = format!(
545                "{landed} landed commit(s) in {landings} landing(s) wait past {}{unrecorded} on {trunk}",
546                release.reference
547            );
548            match how {
549                Some(LandedAt::Commit) => s.push_str(&format!(
550                    ", the oldest for at most {}: this clone holds no record of when it landed, so its commit time bounds the wait",
551                    age_words(waited)
552                )),
553                _ => s.push_str(&format!(
554                    ", the oldest for {} since this clone saw it land",
555                    age_words(waited)
556                )),
557            }
558            if stance == Stance::Overdue {
559                s.push_str(&format!(
560                    ", longer than the {} a release may wait",
561                    age_words(OVERDUE_AFTER)
562                ));
563            } else if how == Some(LandedAt::Commit) && waited >= OVERDUE_AFTER {
564                s.push_str(&format!(
565                    "; that bound is past the {} a release may wait, which alone makes nothing overdue",
566                    age_words(OVERDUE_AFTER)
567                ));
568            }
569            s
570        }
571    }
572}

574/// A wait in the words a person reads it in: minutes, then hours, then days.
575fn age_words(waited: Duration) -> String {
576    let minutes = waited.as_secs() / 60;
577    match (minutes / (24 * 60), (minutes / 60) % 24, minutes % 60) {
578        (0, 0, m) => format!("{m} min"),
579        (0, h, m) => format!("{h} h {m} min"),
580        (d, h, _) => format!("{d} d {h} h"),
581    }
582}

587        .args(args)

593fn succeeds(root: &Path, args: &[&str]) -> bool {
594    Command::new("git")
595        .current_dir(root)
596        .args(args)
597        .output()
598        .is_ok_and(|out| out.status.success())
599}

608    /// A clone of a bare origin, its trunk `master`, with every commit dated `at`.
609    struct Fixture {
610        dir: tempfile::TempDir,
611    }
612
613    impl Fixture {
614        fn new() -> Fixture {
615            let dir = tempfile::tempdir().expect("a temporary directory");
616            let clone = Fixture { dir };
617            clone.run(
618                &[
619                    "init",
620                    "-q",
621                    "--bare",
622                    "--initial-branch=master",
623                    "origin.git",
624                ],
625                0,
626            );
627            clone.run(&["clone", "-q", "origin.git", "work"], 0);
628            clone.git(&["config", "user.email", "t@example.com"], 0);
629            clone.git(&["config", "user.name", "t"], 0);
630            clone
631        }
632
633        fn work(&self) -> std::path::PathBuf {
634            self.dir.path().join("work")
635        }
636
637        fn command(&self, cwd: &Path, args: &[&str], at: u64) -> String {
638            let out = Command::new("git")
639                .current_dir(cwd)
640                .args(args)
641                .env("GIT_AUTHOR_DATE", format!("@{at} +0000"))
642                .env("GIT_COMMITTER_DATE", format!("@{at} +0000"))
643                .output()
644                .expect("git runs");
645            assert!(out.status.success(), "git {args:?}: {out:?}");
646            String::from_utf8_lossy(&out.stdout).trim().to_string()
647        }
648
649        fn run(&self, args: &[&str], at: u64) -> String {
650            self.command(self.dir.path(), args, at)
651        }
652
653        fn git(&self, args: &[&str], at: u64) -> String {
654            self.command(&self.work(), args, at)
655        }
656
657        /// A commit of `subject` on the checked-out branch, dated `at`, and its sha.
658        fn commit(&self, subject: &str, at: u64) -> String {
659            self.git(&["commit", "-q", "--allow-empty", "-m", subject], at);
660            self.git(&["rev-parse", "HEAD"], at)
661        }
662
663        /// Push the checked-out branch to the trunk and fetch it back, at `at`: the time the
664        /// remote-tracking trunk's reflog records the landing at.
665        fn push(&self, at: u64) {
666            self.git(&["push", "-q", "origin", "HEAD:master", "--tags"], at);
667            self.git(&["fetch", "-q", "origin"], at);
668        }
669    }

671    /// A release record object of the layer's own kind, naming `commit`.
672    fn record(version: &str, commit: &str) -> Object {
673        let mut metadata = serde_json::Map::new();
674        metadata.insert("version".into(), version.into());
675        metadata.insert("tag".into(), format!("v{version}").into());
676        metadata.insert("commit".into(), commit.into());
677        Object {
678            kind: super::super::changelog::RELEASE_KIND.into(),
679            identity: format!("v{version}"),
680            uri: format!("majordomus://release-record/v{version}"),
681            title: None,
682            description: None,
683            metadata: Value::Object(metadata),
684            body: String::new(),
685            content: String::new(),
686            media_type: "application/yaml",
687            provenance: crate::model::Provenance {
688                path: format!(".ai/repo/releases/v{version}.yaml"),
689                directory: ".ai/repo/releases".into(),
690                source_class: "release".into(),
691                section: None,
692                bytes: 0,
693                member: None,
694            },
695        }
696    }

698    fn at(seconds: u64) -> SystemTime {
699        UNIX_EPOCH + Duration::from_secs(seconds)
700    }

702    /// A release with nothing landed past it is current; a fix landed past it is owed, then
703    /// overdue once it has waited past the threshold; a release commit is no landed work; and a
704    /// tag pushed past the record is the release the trunk is measured from.
705    #[test]
706    fn landed_work_past_the_release_is_owed_then_overdue() {
707        let c = Fixture::new();
708        let t0 = 1_700_000_000;
709        let released = c.commit("feat: the first", t0);
710        c.git(&["tag", "v0.1.0"], t0);
711        c.push(t0);
712        let objects = vec![record("0.1.0", &released)];
713        let work = c.work();
714
715        let current = measure(&work, &objects, at(t0 + HOUR));
716        assert_eq!(current.stance, Stance::Current, "{}", current.summary);
717        assert_eq!(current.trunk.as_deref(), Some("origin/master"));
718        assert!(current.release.as_ref().is_some_and(|r| r.recorded));
719        assert!(current.summary.contains("no landed work past v0.1.0"));
720
721        c.commit("fix: the second", t0 + 2 * HOUR);
722        c.commit("chore(release): record v0.1.0", t0 + 2 * HOUR);
723        c.push(t0 + 2 * HOUR);
724        let owed = measure(&work, &objects, at(t0 + 3 * HOUR));
725        assert_eq!(owed.stance, Stance::Owed, "{}", owed.summary);
726        assert_eq!(owed.landed, 1, "the release commit is no landed work");
727        assert_eq!(
728            owed.landings, 1,
729            "the release commit's landing brought no work"
730        );
731        assert_eq!(owed.waited_seconds, Some(HOUR));
732        assert_eq!(owed.landed_at, Some(LandedAt::Reflog));
733        assert_eq!(owed.changes[0].subject, "the second");
734        assert!(owed.oldest_landing.is_some());
735        assert!(owed
736            .summary
737            .contains("1 landed commit(s) in 1 landing(s) wait past v0.1.0"));
738
739        let overdue = measure(&work, &objects, at(t0 + 2 * HOUR + OVERDUE_AFTER.as_secs()));
740        assert_eq!(overdue.stance, Stance::Overdue);
741        assert!(
742            overdue.summary.contains("longer than the 6 h 0 min"),
743            "{}",
744            overdue.summary
745        );
746
747        c.git(&["tag", "v0.1.1"], 0);
748        c.push(t0 + 3 * HOUR);
749        let tagged = measure(&work, &objects, at(t0 + 30 * HOUR));
750        assert_eq!(tagged.stance, Stance::Current, "{}", tagged.summary);
751        let release = tagged.release.expect("the tag");
752        assert_eq!(
753            (release.reference.as_str(), release.recorded),
754            ("v0.1.1", false)
755        );
756        assert!(tagged.summary.contains("record not landed yet"));
757    }

759    /// A top composed hours before it lands waits from when it reached the trunk, as this
760    /// clone's reflog saw it. Where the reflog does not show it arrive, its commit time bounds
761    /// the wait, and that bound alone never makes the work overdue (the plan critique of
762    /// 2026-10-19).
763    #[test]
764    fn a_landing_waits_from_when_it_reached_the_trunk() {
765        let c = Fixture::new();
766        let t0 = 1_700_000_000;
767        let released = c.commit("feat: the first", t0);
768        c.push(t0);
769        let objects = vec![record("0.1.0", &released)];
770        let work = c.work();
771
772        // composed at t0 + 1 h, pushed to the trunk at t0 + 7 h
773        c.commit("fix: a top composed long before it landed", t0 + HOUR);
774        c.push(t0 + 7 * HOUR);
775        let landed = measure(&work, &objects, at(t0 + 8 * HOUR));
776        assert_eq!(
777            landed.landed_at,
778            Some(LandedAt::Reflog),
779            "{}",
780            landed.summary
781        );
782        assert_eq!(
783            landed.waited_seconds,
784            Some(HOUR),
785            "from the push, not the commit"
786        );
787        assert_eq!(landed.stance, Stance::Owed, "{}", landed.summary);
788        assert!(
789            landed.summary.contains("since this clone saw it land"),
790            "{}",
791            landed.summary
792        );
793
794        // the same landing in a clone whose reflog does not show it arrive
795        std::fs::remove_file(work.join(".git/logs/refs/remotes/origin/master"))
796            .expect("the trunk's reflog");
797        let bounded = measure(&work, &objects, at(t0 + 8 * HOUR));
798        assert_eq!(
799            bounded.landed_at,
800            Some(LandedAt::Commit),
801            "{}",
802            bounded.summary
803        );
804        assert_eq!(
805            bounded.waited_seconds,
806            Some(7 * HOUR),
807            "an upper bound, from the commit"
808        );
809        assert_eq!(
810            bounded.stance,
811            Stance::Owed,
812            "a bound alone is never overdue: {}",
813            bounded.summary
814        );
815        assert!(
816            bounded.summary.contains("at most 7 h 0 min"),
817            "{}",
818            bounded.summary
819        );
820        assert!(
821            bounded
822                .summary
823                .contains("which alone makes nothing overdue"),
824            "{}",
825            bounded.summary
826        );
827    }

909    /// The version is decided only for landed work, from a recorded release, at the trunk; each
910    /// other case says which it is not.
911    #[test]
912    fn the_version_is_decided_only_where_the_contract_can_be_measured() {
913        let c = Fixture::new();
914        let t0 = 1_700_000_000;
915        let released = c.commit("feat: the first", t0);
916        c.push(t0);
917        let registry = CapabilityRegistry::builder().build().unwrap();
918        let objects = vec![record("0.1.0", &released)];
919        let work = c.work();
920
921        let mut nothing = measure(&work, &objects, at(t0));
922        decide_next(&mut nothing, &work, &registry, &objects);
923        assert_eq!(
924            (nothing.next.as_ref(), nothing.next_unmeasured.as_ref()),
925            (None, None)
926        );
927
928        c.commit("fix: the second", t0 + HOUR);
929        c.push(t0 + HOUR);
930        let mut here = measure(&work, &objects, at(t0 + HOUR));
931        decide_next(&mut here, &work, &registry, &objects);
932        // this fixture published no registry, so the contract's own refusal is carried
933        assert!(here.next.is_none());
934        assert!(here.next_unmeasured.is_some(), "{here:?}");
935
936        c.git(&["checkout", "-q", "-b", "side"], 0);
937        c.commit("fix: elsewhere", t0 + HOUR);
938        let mut away = measure(&work, &objects, at(t0 + HOUR));
939        decide_next(&mut away, &work, &registry, &objects);
940        assert!(away
941            .next_unmeasured
942            .as_deref()
943            .is_some_and(|w| w.contains("not at origin/master")));
944
945        c.git(&["checkout", "-q", "master"], 0);
946        c.git(&["tag", "v0.1.1"], 0);
947        c.commit("fix: after the tag", t0 + 2 * HOUR);
948        c.push(t0 + 2 * HOUR);
949        let mut unrecorded = measure(&work, &objects, at(t0 + 2 * HOUR));
950        decide_next(&mut unrecorded, &work, &registry, &objects);
951        assert!(unrecorded
952            .next_unmeasured
953            .as_deref()
954            .is_some_and(|w| w.contains("v0.1.1 has no release record yet")));
955    }

957    /// The newer of a record and a tag is the release; a lone record or tag is it; none is none.
958    #[test]
959    fn the_newer_of_the_record_and_the_tag_is_the_release() {
960        let last = |version: &str| super::super::compat::LastRelease {
961            version: version.into(),
962            reference: format!("v{version}"),
963            read_at: "c0ffee".into(),
964            recorded: true,
965        };
966        let newer = newest(Some(last("0.1.0")), Some("v0.2.0".into())).unwrap();
967        assert_eq!(
968            (newer.reference.as_str(), newer.recorded),
969            ("v0.2.0", false)
970        );
971        let older = newest(Some(last("0.3.0")), Some("v0.2.0".into())).unwrap();
972        assert_eq!((older.at.as_str(), older.recorded), ("c0ffee", true));
973        assert_eq!(newest(Some(last("0.1.0")), None).unwrap().version, "0.1.0");
974        assert_eq!(newest(None, Some("v0.1.0".into())).unwrap().at, "v0.1.0");
975        assert_eq!(newest(None, None), None);
976    }
