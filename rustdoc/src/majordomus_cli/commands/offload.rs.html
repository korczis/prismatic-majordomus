1//! `majordomus offload`: the command line's rendering of [`crate::offload`].
2//!
3//! Nothing is decided here. The declaration is read (from `--declaration`, or from the
4//! repository's index), the module answers, and this file prints the answer as lines for a
5//! person or as one JSON document.
6

17/// Run `majordomus offload`.
18pub fn run(args: OffloadArgs) -> Result<u8> {
19    let declaration = load(&args)?;
20    let stdout = std::io::stdout();
21    let mut out = stdout.lock();
22    let w = |e: std::io::Error| Error::io("<stdout>", e);
23    match args.command {
24        OffloadCommand::Hosts => {
25            hosts(declaration.as_ref(), args.format, &mut out).map_err(w)?;
26            Ok(0)
27        }
28        OffloadCommand::Probe => {
29            let d = required(declaration)?;
30            let reports = probe(&d);
31            let chosen = probe_report(&reports, args.format, &mut out).map_err(w)?;
32            Ok(if chosen { 0 } else { EXIT_UNAVAILABLE })
33        }
34        OffloadCommand::Derive {
35            worktree,
36            host,
37            keep,
38        } => {
39            let d = required(declaration)?;
40            let request = OffloadRequest {
41                worktree,
42                host,
43                keep,
44            };
45            let stderr = std::io::stderr();
46            let report = offload(&d, &request, &mut stderr.lock()).map_err(|r| Error::Refused {
47                code: r.code,
48                reason: format!("offload derive: {}", r.reason),
49            })?;
50            derive_report(&report, args.format, &mut out).map_err(w)?;
51            Ok(0)
52        }
53    }
54}

56/// Print every host's measurement; answers whether one was chosen.
57fn probe_report(
58    reports: &[HostReport],
59    format: OutputFormat,
60    out: &mut dyn Write,
61) -> std::io::Result<bool> {
62    let chosen = reports.iter().find(|r| r.chosen).map(|r| r.id.clone());
63    match format {
64        OutputFormat::Json => {
65            let doc = json!({ "chosen": chosen, "hosts": reports });
66            writeln!(
67                out,
68                "{}",
69                serde_json::to_string_pretty(&doc).unwrap_or_default()
70            )?;
71        }
72        OutputFormat::Text => {
73            for r in reports {
74                let free = r.free_gib.map_or("?".into(), |f| f.to_string());
75                let load = r.load.map_or("?".into(), |l| l.to_string());
76                let mark = if r.chosen { "*" } else { " " };
77                let why = r
78                    .reason
79                    .as_deref()
80                    .map(|s| format!("  ({s})"))
81                    .unwrap_or_default();
82                writeln!(
83                    out,
84                    "{mark} {:<12} {:<28} free {free:>4} GiB (min {:>3})  load {load:>2}  {}{why}",
85                    r.id,
86                    r.address,
87                    r.min_free_gib,
88                    standing(r.standing)
89                )?;
90            }
91            match &chosen {
92                Some(id) => writeln!(out, "a derive would go to {id}")?,
93                None => writeln!(out, "no declared host can take a derive")?,
94            }
95        }
96    }
97    Ok(chosen.is_some())
98}

100/// Print what an offloaded derive did.
101fn derive_report(
102    report: &OffloadReport,
103    format: OutputFormat,
104    out: &mut dyn Write,
105) -> std::io::Result<()> {
106    match format {
107        OutputFormat::Json => writeln!(
108            out,
109            "{}",
110            serde_json::to_string_pretty(report).unwrap_or_default()
111        ),
112        OutputFormat::Text => {
113            writeln!(
114                out,
115                "derived {} on {} in {} s ({} s in the queue): {} file(s) staged, nothing committed",
116                report.tag,
117                report.host,
118                report.seconds,
119                report.waited_seconds,
120                report.staged.len()
121            )?;
122            for f in &report.staged {
123                writeln!(out, "  {f}")?;
124            }
125            Ok(())
126        }
127    }
128}

130fn standing(s: Standing) -> &'static str {
131    match s {
132        Standing::Eligible => "eligible",
133        Standing::Unreachable => "unreachable",
134        Standing::UnderMinimum => "under its free-space minimum",
135        Standing::UnknownSpace => "free space unknown",
136    }
137}

139fn load(args: &OffloadArgs) -> Result<Option<OffloadDeclaration>> {
140    let refused = |reason: String| Error::Refused {
141        code: EXIT_UNAVAILABLE,
142        reason: format!("offload: {reason}"),
143    };
144    if let Some(path) = &args.declaration {
145        return OffloadDeclaration::read(path).map(Some).map_err(refused);
146    }
147    let app = crate::app::App::load(&args.repo)?;
148    crate::offload::from_objects(&app.index().objects).map_err(refused)
149}

151fn required(d: Option<OffloadDeclaration>) -> Result<OffloadDeclaration> {
152    d.ok_or_else(|| Error::Refused {
153        code: EXIT_UNAVAILABLE,
154        reason: "offload: this repository declares no derive-offload host (.ai/repo/offload/*.yaml, kind derive-offload); see docs/OFFLOAD.md".into(),
155    })
156}

158fn hosts(
159    d: Option<&OffloadDeclaration>,
160    format: OutputFormat,
161    out: &mut dyn Write,
162) -> std::io::Result<()> {
163    if format == OutputFormat::Json {
164        let doc = json!({ "declared": d.is_some(), "declaration": d });
165        return writeln!(
166            out,
167            "{}",
168            serde_json::to_string_pretty(&doc).unwrap_or_default()
169        );
170    }
171    let Some(d) = d else {
172        return writeln!(out, "no derive-offload declaration in this repository: nothing to offload to (.ai/repo/offload/*.yaml)");
173    };
174    writeln!(
175        out,
176        "declaration {} — base {}, minimum {} GiB free",
177        d.id, d.base, d.min_free_gib
178    )?;
179    writeln!(
180        out,
181        "queue {} (lock {}), a look every {} s, at most {} s",
182        d.queue.directory, d.queue.lock, d.queue.poll_seconds, d.queue.wait_max_seconds
183    )?;
184    writeln!(
185        out,
186        "on a host: checkout ~/{}, worktrees ~/{}, shared build ~/{}, runs {}",
187        d.remote.checkout, d.remote.worktrees, d.remote.target, d.remote.derive
188    )?;
189    for h in &d.hosts {
190        let id = h
191            .identity_file
192            .as_deref()
193            .map(|f| format!("  -i {f}"))
194            .unwrap_or_default();
195        writeln!(
196            out,
197            "  {:<12} {:<28} min {:>3} GiB{id}",
198            h.id,
199            h.address,
200            h.min_free_gib(d)
201        )?;
202    }
203    Ok(())
204}

210    /// A writer that refuses the first write carrying `marker`, as a closed pipe would at
211    /// that line, and keeps everything written before it.
212    struct FailOn {
213        marker: &'static str,
214        written: Vec<u8>,
215    }
216
217    impl FailOn {
218        fn new(marker: &'static str) -> FailOn {
219            FailOn {
220                marker,
221                written: Vec::new(),
222            }
223        }
224
225        fn text(&self) -> String {
226            String::from_utf8(self.written.clone()).unwrap()
227        }
228    }
229
230    impl Write for FailOn {
231        fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
232            if String::from_utf8_lossy(buf).contains(self.marker) {
233                return Err(std::io::Error::from(std::io::ErrorKind::BrokenPipe));
234            }
235            self.written.extend_from_slice(buf);
236            Ok(buf.len())
237        }
238
239        fn flush(&mut self) -> std::io::Result<()> {
240            Ok(())
241        }
242    }

244    fn declaration() -> OffloadDeclaration {
245        serde_json::from_value(json!({
246            "schema": "derive-offload/v1", "kind": "derive-offload", "id": "derive",
247            "hosts": [
248                { "id": "lundra", "address": "u@lundra" },
249                { "id": "macbook", "address": "u@macbook", "identity_file": "~/.ssh/k",
250                  "min_free_gib": 20 }
251            ]
252        }))
253        .unwrap()
254    }

256    fn report(id: &str, standing: Standing, chosen: bool) -> HostReport {
257        HostReport {
258            id: id.into(),
259            address: format!("u@{id}"),
260            min_free_gib: 10,
261            free_gib: (standing != Standing::Unreachable).then_some(50),
262            load: (standing != Standing::Unreachable).then_some(1),
263            standing,
264            reason: (standing != Standing::Eligible).then(|| format!("{id} is out")),
265            chosen,
266        }
267    }

338    #[test]
339    fn a_probe_names_the_host_a_derive_would_take() {
340        let reports = [
341            report("down", Standing::Unreachable, false),
342            report("blind", Standing::UnknownSpace, false),
343            report("roomy", Standing::Eligible, true),
344        ];
345        let mut out = FailOn::new("\0");
346        assert!(probe_report(&reports, OutputFormat::Text, &mut out).unwrap());
347        let text = out.text();
348        let lines: Vec<&str> = text.lines().collect();
349        assert_eq!(lines.len(), 4, "{text}");
350        assert!(
351            lines[0].starts_with("  down ")
352                && lines[0].contains("free    ? GiB")
353                && lines[0].ends_with("unreachable  (down is out)"),
354            "{}",
355            lines[0]
356        );
357        assert!(lines[1].ends_with("free space unknown  (blind is out)"));
358        assert!(lines[2].starts_with("* roomy ") && lines[2].ends_with("load  1  eligible"));
359        assert_eq!(lines[3], "a derive would go to roomy");
360
361        let mut out = FailOn::new("\0");
362        assert!(probe_report(&reports, OutputFormat::Json, &mut out).unwrap());
363        let v: serde_json::Value = serde_json::from_str(&out.text()).unwrap();
364        assert_eq!(v["chosen"], "roomy");
365        assert_eq!(v["hosts"][1]["standing"], "unknown_space");
366
367        let none = [report("down", Standing::Unreachable, false)];
368        let mut out = FailOn::new("\0");
369        assert!(!probe_report(&none, OutputFormat::Text, &mut out).unwrap());
370        assert!(out.text().ends_with("no declared host can take a derive\n"));
371    }
