1//! The design system: one declaration, typed, and every surface a projection of it.
2//!
3//! `share/design/tokens.yaml` is the only place a visual decision is made for any surface
4//! this repository renders — the site published to GitHub Pages, the Cockpit, the pages the
5//! executable renders itself, and the Swagger UI shell. This module reads that file into
6//! [`DesignSystem`], refuses it when it is inconsistent, fingerprints it, and answers the
7//! questions every projection asks of it: what a role resolves to in each theme, which
8//! status a state word carries, which Flowbite name is a synonym of which role.
9//! [`render`] turns the answers into the stylesheets, datasets and documents
10//! `majordomus generate design` writes; the `design` capability module serves them.
11//!
12//! The executable carries a copy of the declaration (`tokens.yaml` beside this file, a
13//! generated artifact kept current by `generate --check`) so that the capabilities, the
14//! Cockpit's inspector and `explain` answer from the declaration the stylesheets were
15//! projected from, and so that the crate still builds when packaged alone.
16//!
17//! The model is the crate's own: nothing outside constructs a [`Role`] or a [`Token`], and
18//! what a client reads is the JSON the `design` capabilities answer with. Its examples are
19//! therefore unit tests rather than `///` examples — see `the_compiled_declaration_answers_
20//! for_every_surface` below, which is the example this header would otherwise carry.
21

62/// A map that keeps the declaration's order. A role, a state, a type step is presented in
63/// the order a person wrote it, on every surface, and the fingerprint is the fingerprint
64/// of that order too; a `BTreeMap` would alphabetise `raised` before `sunken` and put the
65/// page background last.
66#[derive(Debug, Clone, PartialEq, Eq)]
67pub struct Ordered<T>(pub Vec<(String, T)>);
68
69impl<T> Default for Ordered<T> {
70    fn default() -> Self {
71        Ordered(Vec::new())
72    }
73}
74
75impl<T> Ordered<T> {
76    /// The entry under `key`, when there is one.
77    pub fn get(&self, key: &str) -> Option<&T> {
78        self.0.iter().find(|(k, _)| k == key).map(|(_, v)| v)
79    }
80    /// Every entry, in declaration order.
81    pub fn iter(&self) -> impl Iterator<Item = (&str, &T)> {
82        self.0.iter().map(|(k, v)| (k.as_str(), v))
83    }
84    /// Every key, in declaration order.
85    pub fn keys(&self) -> impl Iterator<Item = &str> {
86        self.0.iter().map(|(k, _)| k.as_str())
87    }
88    /// How many entries.
89    pub fn len(&self) -> usize {
90        self.0.len()
91    }
92    /// Whether there are none.
93    pub fn is_empty(&self) -> bool {
94        self.0.is_empty()
95    }
96}
97
98impl<T: Serialize> Serialize for Ordered<T> {
99    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
100        use serde::ser::SerializeMap;
101        let mut map = serializer.serialize_map(Some(self.0.len()))?;
102        for (k, v) in &self.0 {
103            map.serialize_entry(k, v)?;
104        }
105        map.end()
106    }
107}
108
109impl<'de, T: Deserialize<'de>> Deserialize<'de> for Ordered<T> {
110    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
111        struct OrderedVisitor<T>(PhantomData<T>);
112        impl<'de, T: Deserialize<'de>> Visitor<'de> for OrderedVisitor<T> {
113            type Value = Ordered<T>;
114            fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
115                f.write_str("a map whose keys are token names")
116            }
117            fn visit_map<A: MapAccess<'de>>(self, mut access: A) -> Result<Self::Value, A::Error> {
118                let mut out: Vec<(String, T)> = Vec::with_capacity(access.size_hint().unwrap_or(0));
119                while let Some((k, v)) = access.next_entry::<String, T>()? {
120                    if out.iter().any(|(seen, _)| *seen == k) {
121                        return Err(serde::de::Error::custom(format!(
122                            "duplicate token name '{k}'"
123                        )));
124                    }
125                    out.push((k, v));
126                }
127                Ok(Ordered(out))
128            }
129        }
130        deserializer.deserialize_map(OrderedVisitor(PhantomData))
131    }
132}

134impl<T: JsonSchema> JsonSchema for Ordered<T> {
135    fn inline_schema() -> bool {
136        true
137    }
138    fn schema_name() -> Cow<'static, str> {
139        Cow::Owned(format!("Ordered_{}", T::schema_name()))
140    }
141    fn schema_id() -> Cow<'static, str> {
142        Cow::Owned(format!("Ordered<{}>", T::schema_id()))
143    }
144    fn json_schema(generator: &mut SchemaGenerator) -> Schema {
145        let value = generator.subschema_for::<T>();
146        schemars::json_schema!({

149            "propertyNames": { "pattern": NAME_PATTERN },
150            "additionalProperties": value
151        })
152    }

155/// What a token name looks like: lowercase, digits and hyphens, starting with a letter.
156/// The same shape the layer's identifiers have, and the shape a CSS custom property and a
157/// class modifier can carry without escaping.
158pub const NAME_PATTERN: &str = "^[a-z][a-z0-9-]*$";
159
160fn is_name(s: &str) -> bool {
161    let mut chars = s.chars();
162    matches!(chars.next(), Some(c) if c.is_ascii_lowercase())
163        && chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-')
164}

454static COMPILED_MODEL: LazyLock<Result<DesignSystem, String>> =
455    LazyLock::new(|| DesignSystem::parse(COMPILED));

457impl DesignSystem {
458    /// The declaration compiled into this executable.
459    ///
460    /// An error here is a build defect — the compiled copy is a generated artifact of a
461    /// validated source — and is reported, never unwrapped, so a served process says what is
462    /// wrong instead of dying.
463    pub fn compiled() -> Result<&'static DesignSystem, &'static str> {
464        COMPILED_MODEL.as_ref().map_err(String::as_str)
465    }
466
467    /// Read and validate a declaration.
468    ///
469    /// A declaration that does not carry every section is refused by the field that is
470    /// missing; one that carries them all is refused by [`DesignSystem::validate`], which
471    /// names the key path and the reason. Either way nothing is projected.
472    pub fn parse(text: &str) -> Result<DesignSystem, String> {
473        let design: DesignSystem =
474            yaml::parse_into(text).map_err(|e| format!("the declaration does not parse: {e}"))?;
475        let findings = design.validate();
476        if findings.is_empty() {
477            Ok(design)
478        } else if findings.len() == 1 {
479            Err(findings[0].clone())
480        } else {
481            Err(format!("{} (and {} more)", findings[0], findings.len() - 1))
482        }
483    }
484
485    /// Every inconsistency in the declaration, each in a line. Empty means the declaration
486    /// can be projected.
487    ///
488    /// What is refused is what would fail silently downstream: a role naming a palette
489    /// entry that does not exist would become an empty custom property, which CSS treats
490    /// as unset and every reader falls back from; a state word filed under two meanings
491    /// would be coloured by whichever selector the browser saw last; an unbalanced quote
492    /// in a value would be dropped by the parser with every gate still green.
493    pub fn validate(&self) -> Vec<String> {
494        let mut out = Vec::new();
495        let mut refuse = |line: String| out.push(line);
496
497        if self.schema != SCHEMA {
498            refuse(format!(
499                "schema: the declaration is version {}; this executable reads {SCHEMA}",
500                self.schema
501            ));
502        }
503        for (name, value) in self.palette.iter() {
504            if !is_name(name) {
505                refuse(format!("palette.{name}: not a token name ({NAME_PATTERN})"));
506            }
507            if let Some(reason) = malformed(value) {
508                refuse(format!("palette.{name}: {reason}"));
509            }
510        }
511        for stack in [("sans", &self.font.sans), ("mono", &self.font.mono)] {
512            if let Some(reason) = malformed(stack.1) {
513                refuse(format!("font.{}: {reason}", stack.0));
514            }
515        }
516        for (name, role) in self.roles.iter() {
517            if !is_name(name) {
518                refuse(format!("roles.{name}: not a token name ({NAME_PATTERN})"));
519            }
520            for (theme, reference) in [("light", &role.light), ("dark", &role.dark)] {
521                if self.palette.get(reference).is_none() {
522                    refuse(format!(
523                        "roles.{name}.{theme}: '{reference}' is not a palette entry"
524                    ));
525                }
526            }
527        }
528        for (name, status) in self.status.roles.iter() {
529            if !is_name(name) {
530                refuse(format!(
531                    "status.roles.{name}: not a token name ({NAME_PATTERN})"
532                ));
533            }
534            if self.roles.get(name).is_some() {
535                refuse(format!(
536                    "status.roles.{name}: a status may not carry the name of a role"
537                ));
538            }
539            for (part, pair) in [
540                ("fg", &status.fg),
541                ("bg", &status.bg),
542                ("line", &status.line),
543            ] {
544                for (theme, reference) in [("light", &pair.light), ("dark", &pair.dark)] {
545                    if self.palette.get(reference).is_none() && self.roles.get(reference).is_none()
546                    {
547                        refuse(format!(
548                            "status.roles.{name}.{part}.{theme}: '{reference}' is neither a palette entry nor a role"
549                        ));
550                    }
551                }
552            }
553        }
554        let mut seen: Vec<(&str, &str)> = Vec::new();
555        for (role, words) in self.status.states.iter() {
556            if self.status.roles.get(role).is_none() {
557                refuse(format!("status.states.{role}: not a status role"));
558            }
559            for word in words {
560                if !is_name(word) {
561                    refuse(format!(
562                        "status.states.{role}: '{word}' is not a state word ({NAME_PATTERN})"
563                    ));
564                }
565                if let Some((_, other)) = seen.iter().find(|(w, _)| *w == word.as_str()) {
566                    refuse(format!(
567                        "status.states.{role}: '{word}' is already filed under '{other}'; a word carries one meaning"
568                    ));
569                }
570                if word != role && self.status.roles.get(word).is_some() {
571                    refuse(format!(
572                        "status.states.{role}: '{word}' is the name of another status and cannot be filed under this one"
573                    ));
574                }
575                seen.push((word.as_str(), role));
576            }
577        }
578        for (name, step) in self.type_.scale.iter() {
579            if !is_name(name) {
580                refuse(format!(
581                    "type.scale.{name}: not a token name ({NAME_PATTERN})"
582                ));
583            }
584            if !has_length_unit(&step.size) {
585                refuse(format!(
586                    "type.scale.{name}.size: '{}' carries no length unit",
587                    step.size
588                ));
589            }
590            if !(step.leading.is_finite() && step.leading > 0.0) {
591                refuse(format!(
592                    "type.scale.{name}.leading: '{}' is not a positive number",
593                    step.leading
594                ));
595            }
596        }
597        for (section, entries) in [
598            ("type.tracking", &self.type_.tracking),
599            ("layout", &self.layout),
600            ("radius", &self.radius),
601            ("motion", &self.motion),
602        ] {
603            for (name, named) in entries.iter() {
604                if !is_name(name) {
605                    refuse(format!(
606                        "{section}.{name}: not a token name ({NAME_PATTERN})"
607                    ));
608                }
609                if let Some(reason) = malformed(&named.value) {
610                    refuse(format!("{section}.{name}.value: {reason}"));
611                }
612            }
613        }
614        for name in self.layout.keys() {
615            if self.roles.get(name).is_some() || self.status.roles.get(name).is_some() {
616                refuse(format!(
617                    "layout.{name}: collides with a colour token of the same name; both would be --mj-{name}"
618                ));
619            }
620        }
621        if !is_name(&self.theme.class) {
622            refuse(format!(
623                "theme.class: '{}' is not a class name ({NAME_PATTERN})",
624                self.theme.class
625            ));
626        }
627        if self.theme.storage_key.is_empty() || self.theme.storage_key.contains(['\'', '"', '\\']) {
628            refuse("theme.storage-key: must be a plain key with no quotes".to_string());
629        }
630        if self.viewports.is_empty() {
631            refuse("viewports: at least one width is needed".to_string());
632        }
633        if self.viewports.windows(2).any(|w| w[0] >= w[1]) {
634            refuse("viewports: widths are listed ascending and once".to_string());
635        }
636        for (name, target) in self.alias.flowbite.iter() {
637            if !is_name(name) {
638                refuse(format!(
639                    "alias.flowbite.{name}: not a token name ({NAME_PATTERN})"
640                ));
641            }
642            if self.colour_css(target).is_none() {
643                refuse(format!(
644                    "alias.flowbite.{name}: '{target}' is neither a role nor a status colour (ok, ok-bg, ok-line)"
645                ));
646            }
647        }
648        out
649    }

657    pub fn role_of_state(&self, word: &str) -> Option<&str> {

668    /// Parse a status colour reference: `ok` is the text, `ok-bg` the ground, `ok-line`
669    /// the border.
670    pub fn status_ref<'a>(&'a self, token: &'a str) -> Option<StatusRef<'a>> {
671        if let Some((role, _)) = self.status.roles.iter().find(|(k, _)| *k == token) {
672            return Some(StatusRef { role, part: "fg" });
673        }
674        for (suffix, part) in [("-bg", "bg"), ("-line", "line")] {
675            if let Some(role) = token.strip_suffix(suffix) {
676                if let Some((role, _)) = self.status.roles.iter().find(|(k, _)| *k == role) {
677                    return Some(StatusRef { role, part });
678                }
679            }
680        }
681        None
682    }

736    /// Every token, explained, in the inventory's order.
737    pub fn tokens(&self) -> Vec<Token> {
738        let mut out = Vec::new();
739        let aliases_of = |token: &str| -> Vec<String> {
740            self.alias
741                .flowbite
742                .iter()
743                .filter(|(_, target)| *target == token)
744                .map(|(name, _)| format!("--color-{name}"))
745                .collect()
746        };
747        for (name, stack, about) in [
748            (
749                "sans",
750                &self.font.sans,
751                "the system stack every surface reads in",
752            ),
753            (
754                "mono",
755                &self.font.mono,
756                "the monospace stack for code, identifiers and logs",
757            ),
758        ] {
759            out.push(Token {
760                name: name.into(),
761                kind: TokenKind::Font,
762                about: about.into(),
763                css: vec![format!("--font-{name}"), format!("font-{name}")],
764                parts: Vec::new(),
765                value: Some(stack.clone()),
766                role: None,
767                states: Vec::new(),
768                aliases: Vec::new(),
769                projections: render::projections_of(TokenKind::Font),
770            });
771        }
772        for (name, role) in self.roles.iter() {
773            let css = format!("{PREFIX}{name}");
774            out.push(Token {
775                name: name.into(),
776                kind: TokenKind::Role,
777                about: role.about.clone(),
778                css: vec![css.clone()],
779                parts: vec![ColourPart {
780                    part: "fg".into(),
781                    css,
782                    light: self
783                        .resolve(&role.light, false)
784                        .unwrap_or_else(|| unresolved(&role.light)),
785                    dark: self
786                        .resolve(&role.dark, true)
787                        .unwrap_or_else(|| unresolved(&role.dark)),
788                }],
789                value: None,
790                role: None,
791                states: Vec::new(),
792                aliases: aliases_of(name),
793                projections: render::projections_of(TokenKind::Role),
794            });
795        }
796        for (name, status) in self.status.roles.iter() {
797            let parts: Vec<ColourPart> = [
798                ("fg", &status.fg),
799                ("bg", &status.bg),
800                ("line", &status.line),
801            ]
802            .into_iter()
803            .map(|(part, pair)| {
804                let css = if part == "fg" {
805                    format!("{PREFIX}{name}")
806                } else {
807                    format!("{PREFIX}{name}-{part}")
808                };
809                ColourPart {
810                    part: part.into(),
811                    css,
812                    light: self
813                        .resolve(&pair.light, false)
814                        .unwrap_or_else(|| unresolved(&pair.light)),
815                    dark: self
816                        .resolve(&pair.dark, true)
817                        .unwrap_or_else(|| unresolved(&pair.dark)),
818                }
819            })
820            .collect();
821            let mut aliases = aliases_of(name);
822            aliases.extend(aliases_of(&format!("{name}-bg")));
823            aliases.extend(aliases_of(&format!("{name}-line")));
824            out.push(Token {
825                name: name.into(),
826                kind: TokenKind::Status,
827                about: status.about.clone(),
828                css: parts
829                    .iter()
830                    .map(|p| p.css.clone())
831                    .chain([format!(".mj-badge--{name}"), format!(".mj-status--{name}")])
832                    .collect(),
833                parts,
834                value: None,
835                role: None,
836                states: self.status.states.get(name).cloned().unwrap_or_default(),
837                aliases,
838                projections: render::projections_of(TokenKind::Status),
839            });
840        }
841        for (role, words) in self.status.states.iter() {
842            for word in words {
843                if word == role {
844                    continue;
845                }
846                out.push(Token {
847                    name: word.clone(),
848                    kind: TokenKind::State,
849                    about: format!("a state word filed under `{role}`"),
850                    css: vec![format!(".mj-badge--{word}"), format!(".mj-status--{word}")],
851                    parts: Vec::new(),
852                    value: None,
853                    role: Some(role.to_string()),
854                    states: Vec::new(),
855                    aliases: Vec::new(),
856                    projections: render::projections_of(TokenKind::State),
857                });
858            }
859        }
860        for (name, step) in self.type_.scale.iter() {
861            out.push(Token {
862                name: name.into(),
863                kind: TokenKind::Type,
864                about: step.about.clone(),
865                css: vec![
866                    format!("--text-{name}"),
867                    format!("--text-{name}--line-height"),
868                    format!("text-{name}"),
869                ],
870                parts: Vec::new(),
871                value: Some(format!("{}/{}", step.size, step.leading)),
872                role: None,
873                states: Vec::new(),
874                aliases: Vec::new(),
875                projections: render::projections_of(TokenKind::Type),
876            });
877        }
878        for (name, named) in self.type_.tracking.iter() {
879            out.push(scalar(
880                name,
881                TokenKind::Tracking,
882                named,
883                vec![format!("--tracking-{name}"), format!("tracking-{name}")],
884            ));
885        }
886        for (name, named) in self.layout.iter() {
887            out.push(scalar(
888                name,
889                TokenKind::Layout,
890                named,
891                vec![format!("{PREFIX}{name}")],
892            ));
893        }
894        for (name, named) in self.radius.iter() {
895            out.push(scalar(
896                name,
897                TokenKind::Radius,
898                named,
899                vec![format!("--radius-{name}"), format!("rounded-{name}")],
900            ));
901        }
902        for (name, named) in self.motion.iter() {
903            out.push(scalar(
904                name,
905                TokenKind::Motion,
906                named,
907                vec![format!("{PREFIX}motion-{name}")],
908            ));
909        }
910        out.push(Token {
911            name: "theme".into(),
912            kind: TokenKind::Theme,
913            about: format!(
914                "the class `{}` on the root element means dark; the choice is kept under `{}`",
915                self.theme.class, self.theme.storage_key
916            ),
917            css: vec![format!(".{}", self.theme.class)],
918            parts: Vec::new(),
919            value: Some(self.theme.storage_key.clone()),
920            role: None,
921            states: Vec::new(),
922            aliases: Vec::new(),
923            projections: render::projections_of(TokenKind::Theme),
924        });
925        for (name, literal) in self.palette.iter() {
926            out.push(Token {
927                name: name.into(),
928                kind: TokenKind::Palette,
929                about: "a raw palette entry; roles and statuses name it, consumers never do".into(),
930                css: Vec::new(),
931                parts: Vec::new(),
932                value: Some(literal.clone()),
933                role: None,
934                states: Vec::new(),
935                aliases: Vec::new(),
936                projections: render::projections_of(TokenKind::Palette),
937            });
938        }
939        out
940    }

942    /// One token by name — a role, a status, a state word, a type step, a palette entry —
943    /// or by the custom property it becomes: `ok`, `succeeded` and `--mj-ok` all reach the
944    /// `ok` status, and a name nothing declares answers `None` rather than a guess.
945    /// `explain_answers_by_name_and_by_custom_property` below is the example.
946    pub fn explain(&self, name: &str) -> Option<Token> {
947        let wanted = name.trim();
948        let bare = wanted
949            .strip_prefix(PREFIX)
950            .or_else(|| wanted.strip_prefix("--"))
951            .unwrap_or(wanted);
952        let tokens = self.tokens();
953        tokens
954            .iter()
955            .find(|t| t.name == bare)
956            .or_else(|| {
957                tokens.iter().find(|t| {
958                    t.css.iter().any(|c| c == wanted || c == bare)
959                        || t.parts.iter().any(|p| p.css == wanted)
960                })
961            })
962            .cloned()
963    }
964}

989/// Why a CSS value cannot be emitted, when it cannot.
990///
991/// The generator this replaces once wrote `--font-sans: ...Emoji;` with an unbalanced
992/// quote and once wrote `--bg:` with no value at all; the CSS parser dropped both in
993/// silence and every gate stayed green. Neither can be written now.
994fn malformed(value: &str) -> Option<String> {
995    let value = value.trim();
996    if value.is_empty() {
997        return Some("empty value".into());
998    }
999    if value.matches('"').count() % 2 == 1 {
1000        return Some("unbalanced double quote".into());
1001    }
1002    if value.contains([';', '{', '}', '\n']) {
1003        return Some("a value may not carry ';', braces or a line break".into());
1004    }
1005    if let Some(hex) = value.strip_prefix('#') {
1006        if !matches!(hex.len(), 3 | 4 | 6 | 8) || !hex.chars().all(|c| c.is_ascii_hexdigit()) {
1007            return Some(format!("'{value}' is not a hexadecimal colour"));
1008        }
1009    }
1010    None
1011}

1013fn has_length_unit(value: &str) -> bool {
1014    ["px", "rem", "em"].iter().any(|u| {
1015        value
1016            .strip_suffix(u)
1017            .is_some_and(|n| !n.is_empty() && n.parse::<f64>().is_ok())
1018    })
1019}
