1//! The one rule set that makes text a run printed fit to publish.
2//!
3//! Evidence carries text nobody wrote for a reader outside the machine it ran on: the reason
4//! a case skipped, the tail of a failure, a command's captured output. Any of it can hold a
5//! credential the command echoed or a path that names the account it ran under, and once it
6//! is on a public page it cannot be taken back. [`public_text`] is the rule for that text;
7//! wiring the evidence publication to it is I1955, and until then nothing calls it.
8//!
9//! [`redact_secrets`] alone is applied today wherever this process hands back or sends text
10//! it did not write (I2161): every HTTP error body (`http::router::Response::error`), every
11//! JSON-RPC error and MCP tool refusal, the request fields the server logs, and a handover
12//! body before it is signed into the mesh journal and replicated in clear.
13//!
14//! The credential shapes are not a second table. They are a port of `MJ_CAPTURE_SECRETS` and
15//! `MJ_CAPTURE_SECRET_ASSIGN` in `lib/capture.sh`, the table the prompt archive has redacted
16//! with from the start, applied the way `mj_capture_redact` applies it: each shape in table
17//! order over the whole text, every leftmost-longest match replaced by a marker naming the
18//! shape, and then the assignment rule, which keeps the name a value was assigned to. Two
19//! tests hold the port to the table: one reads `lib/capture.sh` and compares the shapes, and
20//! one fixture, `test/fixtures/redaction/shapes.tsv`, is run through both implementations.
21//! The crate has no regular-expression engine, so the shapes are matched by the small
22//! hand-written matcher below, which knows exactly the constructs the table uses.
23//!
24//! Two neighbours are different concerns and stay apart: `execution::redact` removes the
25//! fields a capability's input schema marks sensitive, and `generate::forbidden_in` is the
26//! last word on a published artifact, which [`public_text`] asks after it has done its work.
27//! Before asking, [`public_text`] refuses a text that still carries a secret no shape covers:
28//! one whose announcing shape fired, in this call or in an earlier redaction whose marker
29//! the text carries, and one whose armour or name is left once the shapes have run. Redacting
30//! the announcement is what would let the rest past that check.
31//!
32//! ```
33//! use majordomus_cli::redaction::redact_secrets;
34//!
35//! // assembled at run time, so that no committed file carries a credential's shape
36//! let token = format!("{}{}", "ghp_", "a".repeat(24));
37//! let out = redact_secrets(&format!("pushing with {token} failed"));
38//! assert_eq!(out.text, "pushing with [redacted:github-token] failed");
39//! assert_eq!(out.kinds, ["github-token"]);
40//! ```
41

44/// Text after [`redact_secrets`] or [`public_text`], with the shapes that fired in it.
45///
46/// `kinds` is what `mj_capture_redacted_kinds` reports for the same text: the names its
47/// `[redacted:<name>]` markers carry, each once, in the crate's canonical order, which for
48/// these lower-case hyphenated names is the byte order the shell's `LC_ALL=C sort -u` gives.
49/// They are read off the text, as the shell reads them, so a marker the input already
50/// carried counts as much as one this call wrote: text redacted once and redacted again
51/// still says what was taken out of it. A marker that names no shape of the table is not a
52/// kind here, since a kind is one of [`shape_names`]; the shell would list that name too.
53///
54/// ```
55/// use majordomus_cli::redaction::{redact_secrets, Redacted};
56///
57/// let key = format!("{}{}", "sk-ant-", "b".repeat(20));
58/// let pat = format!("{}{}", "github_pat_", "c".repeat(20));
59/// let out: Redacted = redact_secrets(&format!("{pat} then {key} then {pat}"));
60/// assert_eq!(out.kinds, ["anthropic-key", "github-pat"]);
61/// assert!(!out.text.contains("sk-ant-"));
62///
63/// // redacting again finds nothing new, and still names what the text had taken out
64/// assert_eq!(redact_secrets(&out.text).kinds, out.kinds);
65/// ```
66#[derive(Debug, Clone, PartialEq, Eq)]
67pub struct Redacted {
68    /// The text with every credential replaced by `[redacted:<name>]`.
69    pub text: String,
70    /// The shapes the text's markers name, sorted and unique.
71    pub kinds: Vec<&'static str>,
72}

74/// The names of the credential shapes, in the order they are applied, then `assignment`.
75///
76/// The order is `MJ_CAPTURE_SECRETS`' own, and it matters: the shapes are applied one after
77/// another, so an Anthropic key is named as one before the broader OpenAI shape could claim
78/// it.
79///
80/// ```
81/// use majordomus_cli::redaction::shape_names;
82///
83/// let names = shape_names();
84/// assert_eq!(names.first(), Some(&"anthropic-key"));
85/// assert_eq!(names.last(), Some(&"assignment"));
86/// let anthropic = names.iter().position(|n| *n == "anthropic-key");
87/// let openai = names.iter().position(|n| *n == "openai-key");
88/// assert!(anthropic < openai);
89/// ```
90pub fn shape_names() -> Vec<&'static str> {
91    SHAPES
92        .iter()
93        .chain(std::iter::once(&ASSIGNMENT))
94        .map(|shape| shape.name)
95        .collect()
96}
97
98/// Every credential in `text` replaced by a marker that names its shape.
99///
100/// Byte for byte what `mj_capture_redact` makes of the same text: each shape in table order
101/// over the whole text, every non-overlapping leftmost-longest match replaced as `sed -E`
102/// with `g` replaces it, then the assignment rule, which keeps the name and the separator
103/// and replaces only the value. Text with nothing credential-shaped in it comes back
104/// unchanged, prose that merely talks about a password included.
105///
106/// ```
107/// use majordomus_cli::redaction::redact_secrets;
108///
109/// let value = "d".repeat(16);
110/// let out = redact_secrets(&format!("export API_KEY={value}"));
111/// assert_eq!(out.text, "export API_KEY=[redacted:assignment]");
112/// assert_eq!(out.kinds, ["assignment"]);
113///
114/// let prose = "the password is in the manual";
115/// assert_eq!(redact_secrets(prose).text, prose);
116/// assert!(redact_secrets(prose).kinds.is_empty());
117/// ```
118pub fn redact_secrets(text: &str) -> Redacted {
119    let mut text = text.to_string();
120    for shape in SHAPES.iter().chain(std::iter::once(&ASSIGNMENT)) {
121        if let Some(replaced) = shape.replace_all(&text) {
122            text = replaced;
123        }
124    }
125    let kinds = kinds_in(&text);
126    Redacted { text, kinds }
127}
128
129/// How every marker opens; the shape's name and `]` follow.
130const MARKER: &str = "[redacted:";
131
132/// The shapes named by the markers in `text`, each once and in canonical order: what
133/// `mj_capture_redacted_kinds` reads with `grep -oE '\[redacted:[a-z-]+\]'`, for the names of
134/// the table. No marker can hold another's opening, so every opening is a candidate.
135fn kinds_in(text: &str) -> Vec<&'static str> {
136    let names = shape_names();
137    let mut kinds: Vec<&'static str> = text
138        .match_indices(MARKER)
139        .filter_map(|(at, _)| {
140            let rest = &text[at + MARKER.len()..];
141            let len = rest
142                .find(|c: char| !(c.is_ascii_lowercase() || c == '-'))
143                .unwrap_or(rest.len());
144            rest[len..].starts_with(']').then_some(&rest[..len])
145        })
146        .filter_map(|name| names.iter().copied().find(|shape| *shape == name))
147        .collect();
148    crate::order::canonical_strings(&mut kinds);
149    kinds.dedup();
150    kinds
151}

153/// The repository root written as `<repo>` and the home directory as `<home>`.
154///
155/// These are `mj_uc_normalise`'s first rules, in its order: the canonical (symlinks
156/// resolved) spelling of the root, then the spelling the caller gave, then the home
157/// directory. The canonical spelling goes first because the given one can be a suffix of it
158/// (a temporary directory reached through a symlink), and the root before the home because
159/// a checkout usually lives inside one. A trailing separator is not part of a spelling, and
160/// a spelling that is empty without it, the filesystem root, replaces nothing.
161///
162/// ```
163/// use majordomus_cli::redaction::normalise_machine_paths;
164///
165/// let dir = tempfile::tempdir().unwrap();
166/// let home = dir.path().join("home");
167/// let root = home.join("checkout");
168/// std::fs::create_dir_all(&root).unwrap();
169///
170/// let text = format!("{}/src/lib.rs and {}/.cache", root.display(), home.display());
171/// let out = normalise_machine_paths(&text, &root, Some(&home));
172/// assert_eq!(out, "<repo>/src/lib.rs and <home>/.cache");
173/// ```
174pub fn normalise_machine_paths(text: &str, root: &Path, home: Option<&Path>) -> String {
175    let canonical = std::fs::canonicalize(root).ok();
176    let mut out = text.to_string();
177    for spelling in [canonical.as_deref(), Some(root)]
178        .into_iter()
179        .flatten()
180        .filter_map(spelling_of)
181    {
182        out = out.replace(&spelling, "<repo>");
183    }
184    if let Some(home) = home.and_then(spelling_of) {
185        out = out.replace(&home, "<home>");
186    }
187    out
188}

190/// Text fit to publish: machine paths normalised, then credentials redacted, then checked.
191///
192/// A redaction covers what its shape matched, and two shapes match less than the secret
193/// they announce. So the text is refused, with what it carries and never the text, when:
194///
195/// - its markers name a private key's header, whose body follows on the lines after it, or
196///   an access key id, whose secret access key no shape matches. The markers are read off
197///   the text, so a text redacted once already is refused as the original would be.
198///   Redacting the announcement removes the very marker `generate::forbidden_in` refuses
199///   on, and would publish the rest;
200/// - it still holds a private key's armour line, such as the footer a failure's tail keeps
201///   after cutting the header off, or a secret access key assigned a value: a secret whose
202///   announcement no longer is in the text, or never was;
203/// - `generate::forbidden_in`, the check every published artifact already passes, still
204///   finds a marker: something neither rule can make safe, such as another account's home
205///   directory or a bearer header too short to be a credential's shape.
206///
207/// ```
208/// use majordomus_cli::redaction::public_text;
209///
210/// let dir = tempfile::tempdir().unwrap();
211/// let key = format!("{}{}", "xoxb-", "e".repeat(12));
212/// let text = format!("{}/target: {key}", dir.path().display());
213/// let out = public_text(&text, dir.path(), None).unwrap();
214/// assert_eq!(out.text, "<repo>/target: [redacted:slack-token]");
215///
216/// let elsewhere = format!("{}{}", "/ho", "me/someone/.cache");
217/// let refused = public_text(&elsewhere, dir.path(), None).unwrap_err();
218/// assert!(refused.contains("an absolute path on the machine"));
219///
220/// let pem = format!("{}{}\n{}", "-----BEGIN ", "PRIVATE KEY-----", "f".repeat(64));
221/// let refused = public_text(&pem, dir.path(), None).unwrap_err();
222/// assert!(refused.contains("`private-key-header`"));
223///
224/// let tail = format!("{}\n{}{}", "g".repeat(64), "-----END ", "PRIVATE KEY-----");
225/// let refused = public_text(&tail, dir.path(), None).unwrap_err();
226/// assert!(refused.contains("`PRIVATE KEY-----`"));
227/// ```
228pub fn public_text(text: &str, root: &Path, home: Option<&Path>) -> Result<Redacted, String> {
229    let redacted = redact_secrets(&normalise_machine_paths(text, root, home));
230    if let Some((marker, what)) = uncovered_secret(&redacted) {
231        return Err(format!(
232            "the text carries {what} (`{marker}`), which redacting the shapes of the table \
233             does not cover, so it is not published"
234        ));
235    }
236    match crate::generate::forbidden_in(&redacted.text) {
237        None => Ok(redacted),
238        Some((marker, what)) => Err(format!(
239            "the text still carries {what} (`{marker}`) once its machine paths are normalised \
240             and its credentials redacted, so it is not published"
241        )),
242    }
243}

245/// The first secret `redacted` still carries that no shape covers, as the name it is
246/// refused under and what it is.
247fn uncovered_secret(redacted: &Redacted) -> Option<(&'static str, &'static str)> {
248    let announced = ANNOUNCES_MORE
249        .iter()
250        .find(|(kind, _)| redacted.kinds.contains(kind));
251    let armour = ARMOUR
252        .iter()
253        .find(|(line, _)| contains_ignoring_case(&redacted.text, line));
254    announced.or(armour).copied().or_else(|| {
255        SECRET_ACCESS_KEY.found_in(&redacted.text).then_some((
256            SECRET_ACCESS_KEY.name,
257            "a secret access key assigned a value, which no shape of the table matches",
258        ))
259    })
260}

262/// The shapes whose match announces a secret it does not cover, with what that secret is.
263const ANNOUNCES_MORE: &[(&str, &str)] = &[
264    (
265        "aws-access-key-id",
266        "an access key id, whose secret access key no shape matches",
267    ),
268    (
269        "private-key-header",
270        "a private key's header, whose body no shape matches",

274/// The ends of a private key's armour lines, in any case: every header the shape misses

276const ARMOUR: &[(&str, &str)] = &[
277    (

287/// Does `text` contain `needle`, ASCII letters compared without their case?
288fn contains_ignoring_case(text: &str, needle: &str) -> bool {
289    text.as_bytes()
290        .windows(needle.len())
291        .any(|window| window.eq_ignore_ascii_case(needle.as_bytes()))
292}

294/// A secret access key assigned a long opaque value, in the assignment rule's own form:
295/// `aws_secret_access_key = ...`, `AWS_SECRET_ACCESS_KEY=...`, `"SecretAccessKey": "..."`.
296/// It is not a shape of the table and redacts nothing: [`public_text`] refuses a text it is
297/// found in, since the id it pairs with may be on a line the text no longer has. A name with
298/// no value after it, as a skip reason says it, is not a match.
299const SECRET_ACCESS_KEY: Shape = Shape {
300    name: "secret-access-key",
301    kept: &[
302        Atom::Caseless("secret"),
303        run("-_", 0, Some(1)),
304        Atom::Caseless("access"),
305        run("-_", 0, Some(1)),
306        Atom::Caseless("key"),
307        ASSIGNED,
308    ],
309    secret: &[OPAQUE_VALUE],

312/// A path as the text it appears in, without a trailing separator; `None` when nothing is
313/// left, so that the filesystem root never replaces every separator in the text.
314fn spelling_of(path: &Path) -> Option<String> {
315    let lossy = path.to_string_lossy();
316    let trimmed = lossy.trim_end_matches('/');
317    (!trimmed.is_empty()).then(|| trimmed.to_string())
318}

322/// The inside of a POSIX bracket expression, as `lib/capture.sh` spells it: single bytes and
323/// `a-z` ranges. A backslash is an ordinary member, as POSIX has it and as both BSD and GNU
324/// `sed` read the assignment rule's `[\"']`.
325#[derive(Debug, Clone, Copy)]
326struct Class(&'static str);
327
328impl Class {
329    fn has(self, byte: u8) -> bool {
330        let spelling = self.0.as_bytes();
331        let mut i = 0;
332        while i < spelling.len() {
333            if i + 2 < spelling.len() && spelling[i + 1] == b'-' {
334                if (spelling[i]..=spelling[i + 2]).contains(&byte) {
335                    return true;
336                }
337                i += 3;
338            } else {
339                if spelling[i] == byte {
340                    return true;
341                }
342                i += 1;
343            }
344        }
345        false
346    }
347}

349/// One construct of the table's extended regular expressions.
350#[derive(Debug, Clone, Copy)]
351enum Atom {
352    /// These bytes, exactly.
353    Lit(&'static str),
354    /// These letters in either case, which the table spells `[Aa][Pp][Ii]`.
355    Caseless(&'static str),
356    /// One byte of the class.
357    One(Class),
358    /// At least `min` and at most `max` bytes of the class: `?`, `*`, `+`, `{n}`, `{n,}`.
359    Run {
360        class: Class,
361        min: usize,
362        max: Option<usize>,
363    },
364    /// One of the alternatives, or nothing as well when `optional`: `(a|b)` and `(a)?`.
365    Group {
366        alternatives: &'static [&'static [Atom]],
367        optional: bool,
368    },
369}

371/// A credential shape: `kept` is what the replacement keeps (the `\1\2` of the assignment

374    name: &'static str,
375    kept: &'static [Atom],

379const fn run(class: &'static str, min: usize, max: Option<usize>) -> Atom {
380    Atom::Run {
381        class: Class(class),
382        min,
383        max,
384    }
385}

526/// Every end at which `atoms` can match `text` from `at`, handed to `found`. A POSIX match is
527/// the longest one, so the callers keep the largest end rather than the first.
528fn ends(atoms: &[Atom], text: &[u8], at: usize, found: &mut dyn FnMut(usize)) {
529    let Some((first, rest)) = atoms.split_first() else {
530        found(at);
531        return;
532    };
533    match *first {
534        Atom::Lit(lit) => {
535            if text[at..].starts_with(lit.as_bytes()) {
536                ends(rest, text, at + lit.len(), found);
537            }
538        }
539        Atom::Caseless(word) => {
540            let end = at + word.len();
541            if end <= text.len() && text[at..end].eq_ignore_ascii_case(word.as_bytes()) {
542                ends(rest, text, end, found);
543            }
544        }
545        Atom::One(class) => {
546            if text.get(at).is_some_and(|b| class.has(*b)) {
547                ends(rest, text, at + 1, found);
548            }
549        }
550        Atom::Run { class, min, max } => {
551            let cap = max.unwrap_or(usize::MAX);
552            let mut n = 0;
553            while n < cap && text.get(at + n).is_some_and(|b| class.has(*b)) {
554                n += 1;
555            }
556            for len in (min..=n).rev() {
557                ends(rest, text, at + len, found);
558            }
559        }
560        Atom::Group {
561            alternatives,
562            optional,
563        } => {
564            if optional {
565                ends(rest, text, at, found);
566            }
567            for alternative in alternatives {
568                ends(alternative, text, at, &mut |end| {
569                    ends(rest, text, end, found)
570                });
571            }
572        }
573    }
574}

576impl Shape {
577    /// The longest match starting at `at`, as the end of the kept part and the end of the
578    /// whole; among equally long matches, the one whose kept part is longest, as POSIX
579    /// assigns subexpressions.
580    fn match_at(&self, text: &[u8], at: usize) -> Option<(usize, usize)> {
581        let mut best: Option<(usize, usize)> = None;
582        ends(self.kept, text, at, &mut |kept_end| {
583            ends(self.secret, text, kept_end, &mut |end| {
584                if best.is_none_or(|(k, e)| (end, kept_end) > (e, k)) {
585                    best = Some((kept_end, end));
586                }
587            });
588        });
589        best
590    }
591
592    /// Does the shape match anywhere in `text`?
593    fn found_in(&self, text: &str) -> bool {
594        let bytes = text.as_bytes();
595        (0..bytes.len()).any(|at| self.match_at(bytes, at).is_some_and(|(_, end)| end > at))
596    }
597
598    /// `s%<shape>%<kept>[redacted:<name>]%g` over the whole text, or `None` when the shape
599    /// matched nothing. Every construct of the table starts with an ASCII byte and matches
600    /// only ASCII bytes, so every cut below falls on a character boundary.
601    fn replace_all(&self, text: &str) -> Option<String> {
602        let bytes = text.as_bytes();
603        let mut out = String::with_capacity(text.len());
604        let (mut copied, mut at) = (0, 0);
605        while at < bytes.len() {
606            match self.match_at(bytes, at) {
607                Some((kept_end, end)) if end > at => {
608                    out.push_str(&text[copied..kept_end]);
609                    out.push_str(MARKER);
610                    out.push_str(self.name);
611                    out.push(']');
612                    copied = end;
613                    at = end;
614                }
615                _ => at += 1,
616            }
617        }
618        if copied == 0 {
619            return None;
620        }
621        out.push_str(&text[copied..]);
622        Some(out)
623    }
624}

635    /// An atom as the extended regular expression `lib/capture.sh` spells it.
636    fn ere(atoms: &[Atom]) -> String {
637        atoms.iter().map(atom_ere).collect()
638    }
639
640    fn atom_ere(atom: &Atom) -> String {
641        match *atom {
642            Atom::Lit(lit) => lit.to_string(),
643            Atom::Caseless(word) => word
644                .chars()
645                .map(|c| format!("[{}{}]", c.to_ascii_uppercase(), c.to_ascii_lowercase()))
646                .collect(),
647            Atom::One(class) => format!("[{}]", class.0),
648            Atom::Run { class, min, max } => {
649                let times = match (min, max) {
650                    (0, Some(1)) => "?".to_string(),
651                    (0, None) => "*".to_string(),
652                    (1, None) => "+".to_string(),
653                    (n, None) => format!("{{{n},}}"),
654                    (n, Some(m)) if n == m => format!("{{{n}}}"),
655                    (n, Some(m)) => format!("{{{n},{m}}}"),
656                };
657                format!("[{}]{times}", class.0)
658            }
659            Atom::Group {
660                alternatives,
661                optional,
662            } => {
663                let inner: Vec<String> = alternatives.iter().map(|a| ere(a)).collect();
664                format!("({}){}", inner.join("|"), if optional { "?" } else { "" })
665            }
666        }
667    }

669    /// The value of a single-quoted shell assignment `NAME='...'` in `lib/capture.sh`, with
670    /// the `'\''` idiom decoded, as the shell itself reads it.
671    fn shell_value(source: &str, name: &str) -> String {
672        let opening = format!("\n{name}='");
673        let start = source
674            .find(&opening)
675            .expect("the assignment is in lib/capture.sh");
676        let mut rest = &source[start + opening.len()..];
677        let mut value = String::new();
678        loop {
679            let close = rest.find('\'').expect("the quoted value is closed");
680            value.push_str(&rest[..close]);
681            rest = &rest[close + 1..];
682            match rest.strip_prefix("\\''") {
683                Some(more) => {
684                    value.push('\'');
685                    rest = more;
686                }
687                None => return value,
688            }
689        }
690    }
