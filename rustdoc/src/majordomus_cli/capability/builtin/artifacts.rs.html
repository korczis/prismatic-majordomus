1//! The `artifacts` module: what `majordomus generate` writes, read back.
2//!
3//! The generator declares every artifact it produces — the document it projects, the
4//! encoding it is written in, the JSON Schema its content satisfies, and where it came
5//! from — and commits that declaration as `docs/generated/artifacts.json`. This module is
6//! the reading half: it takes the committed manifest and reconciles it with the working
7//! tree, so that a person, an MCP client, an HTTP caller and the Cockpit all learn from
8//! one answer which generated files exist, which encodings a document is committed in,
9//! and which are stale.
10//!
11//! It does not regenerate anything and it holds no list of its own: a document added to
12//! the generator appears here on the next `majordomus generate`, and a file this module
13//! reports as stale is a file `majordomus generate --check` refuses.
14

32/// Where one generated file stands against the tree it is committed in.
33#[derive(
34    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
35)]
36#[serde(rename_all = "snake_case")]
37pub enum ArtifactState {
38    /// The file is there and its bytes hash to what the manifest recorded.
39    Current,
40    /// The file is there and its bytes differ: it was edited, or the generator moved on.
41    Stale,
42    /// The manifest names it and the tree does not have it.
43    Missing,
44    /// The file is there and the manifest records no hash for it: the manifest's own
45    /// encodings, which cannot hash themselves. `generate --check` compares them.
46    Present,
47}
48
49impl ArtifactState {
50    /// The word as serialised.
51    ///
52    /// ```
53    /// use majordomus_cli::capability::builtin::artifacts::ArtifactState;
54    /// assert_eq!(ArtifactState::Stale.as_str(), "stale");
55    /// ```
56    pub fn as_str(self) -> &'static str {
57        match self {
58            ArtifactState::Current => "current",
59            ArtifactState::Stale => "stale",
60            ArtifactState::Missing => "missing",
61            ArtifactState::Present => "present",
62        }
63    }
64}

100/// One document, and the encodings it is committed in.
101pub struct DocumentView {

113#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
114/// One generated file, as the manifest declares it and as the tree has it.
115pub struct ArtifactView {
116    /// Repository-relative path.
117    pub path: String,
118    /// The document it projects.
119    pub document: String,
120    /// The encoding.
121    pub format: ArtifactFormat,
122    #[serde(default, skip_serializing_if = "Option::is_none")]
123    /// The JSON Schema its content satisfies, when it declares one.
124    pub schema: Option<String>,
125    /// One line: what it was derived from.
126    pub source: String,
127    #[serde(default, skip_serializing_if = "Option::is_none")]
128    /// The size the manifest recorded.
129    pub bytes: Option<u64>,
130    #[serde(default, skip_serializing_if = "Option::is_none")]
131    /// The hash the manifest recorded.
132    pub sha256: Option<String>,
133    /// Where the file stands against it.
134    pub state: ArtifactState,
135}

137#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
138/// How many of each, so a caller needs no arithmetic of its own.
139pub struct ArtifactTallies {
140    /// Documents in the manifest.
141    pub documents: usize,
142    /// Files in the manifest, after any filter.
143    pub artifacts: usize,
144    /// Hashes that match.
145    pub current: usize,
146    /// Hashes that do not.
147    pub stale: usize,
148    /// Files the manifest names and the tree lacks.
149    pub missing: usize,
150    /// Files the manifest records no hash for.
151    pub present: usize,
152}

154#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
155/// The answer of `artifacts.list`.
156pub struct ArtifactReport {
157    /// The manifest this was read from, repository-relative.
158    pub manifest: String,
159    /// Whether the manifest is committed in this repository at all. A repository that has
160    /// never run `majordomus generate` has no generated tree, which is a fact about it and
161    /// not a failure of this call: every list below is then empty and every tally zero.
162    pub present: bool,
163    /// The schema the manifest carries, empty when there is none to read.
164    pub schema: String,
165    /// The documents, in the manifest's order.
166    pub documents: Vec<DocumentView>,
167    /// The files, in the manifest's order, after any filter.
168    pub artifacts: Vec<ArtifactView>,
169    /// The counts.
170    pub tallies: ArtifactTallies,
171    /// The command that rewrites every one of them.
172    pub regenerate: String,
173    /// The command that decides staleness byte for byte, which is stronger than the hash.
174    pub verify: String,
175}

177/// The manifest as committed, deserialised into the shape the generator wrote.
178#[derive(Debug, Deserialize)]
179struct Manifest {
180    schema: String,
181    documents: Vec<DocumentView>,
182    artifacts: Vec<ManifestEntry>,
183}
184
185#[derive(Debug, Deserialize)]
186struct ManifestEntry {
187    path: String,
188    document: String,
189    format: ArtifactFormat,
190    #[serde(default)]
191    schema: Option<String>,
192    source: String,
193    #[serde(default)]
194    bytes: Option<u64>,
195    #[serde(default)]
196    sha256: Option<String>,
197}

199fn artifacts_list(ctx: &Context, input: ArtifactsInput) -> Result<ArtifactReport, CapabilityError> {
200    let rel = format!("{OUT_DIR}/{MANIFEST_ID}.json");
201    let root = std::path::Path::new(&ctx.index.repository.root);
202    let Ok(text) = std::fs::read_to_string(root.join(&rel)) else {
203        return Ok(ArtifactReport {
204            manifest: rel,
205            present: false,
206            schema: String::new(),
207            documents: Vec::new(),
208            artifacts: Vec::new(),
209            tallies: ArtifactTallies {
210                documents: 0,
211                artifacts: 0,
212                current: 0,
213                stale: 0,
214                missing: 0,
215                present: 0,
216            },
217            regenerate: "majordomus generate".into(),
218            verify: "majordomus generate --check".into(),
219        });
220    };
221    let manifest: Manifest = serde_json::from_str(&text).map_err(|e| {
222        CapabilityError::Internal(format!(
223            "{rel} is not the manifest this executable writes: {e}"
224        ))
225    })?;
226    if manifest.schema != MANIFEST_SCHEMA {
227        return Err(CapabilityError::Internal(format!(
228            "{rel} carries schema '{}', and this executable reads {MANIFEST_SCHEMA}",
229            manifest.schema
230        )));
231    }
232
233    let mut artifacts = Vec::new();
234    let mut tallies = ArtifactTallies {
235        documents: manifest.documents.len(),
236        artifacts: 0,
237        current: 0,
238        stale: 0,
239        missing: 0,
240        present: 0,
241    };
242    for e in manifest.artifacts {
243        if input.document.as_deref().is_some_and(|d| d != e.document) {
244            continue;
245        }
246        if input.format.is_some_and(|f| f != e.format) {
247            continue;
248        }
249        let state = match std::fs::read_to_string(root.join(&e.path)) {
250            Err(_) => ArtifactState::Missing,
251            Ok(on_disk) => match &e.sha256 {
252                None => ArtifactState::Present,
253                Some(hash) if *hash == sha256_hex(&on_disk) => ArtifactState::Current,
254                Some(_) => ArtifactState::Stale,
255            },
256        };
257        match state {
258            ArtifactState::Current => tallies.current += 1,
259            ArtifactState::Stale => tallies.stale += 1,
260            ArtifactState::Missing => tallies.missing += 1,
261            ArtifactState::Present => tallies.present += 1,
262        }
263        tallies.artifacts += 1;
264        artifacts.push(ArtifactView {
265            path: e.path,
266            document: e.document,
267            format: e.format,
268            schema: e.schema,
269            source: e.source,
270            bytes: e.bytes,
271            sha256: e.sha256,
272            state,
273        });
274    }
275
276    let documents = match &input.document {
277        Some(id) => manifest
278            .documents
279            .into_iter()
280            .filter(|d| &d.id == id)
281            .collect(),
282        None => manifest.documents,
283    };
284    tallies.documents = documents.len();
285
286    Ok(ArtifactReport {
287        manifest: rel,
288        present: true,
289        schema: manifest.schema,
290        documents,
291        artifacts,
292        tallies,
293        regenerate: "majordomus generate".into(),
294        verify: "majordomus generate --check".into(),
295    })
296}

298/// The module.
299pub fn module() -> ModuleDescriptor {
300    module! {
301        id: "artifacts",
302        title: "Generated artifacts",
303        description: "What `majordomus generate` writes: every document with the encodings it is committed in — JSON for a program, YAML beside it, Markdown for a reader — each with its schema, its source and its hash, reconciled with the working tree. The declaration is the generator's own manifest; nothing here keeps a list.",
304        stability: Stability::BehaviorallyVerified,
305        capabilities: [
306            capability! {
307                id: "artifacts.list",
308                title: "List the generated artifacts",
309                description: "The manifest `majordomus generate` commits as docs/generated/artifacts.json, reconciled with the working tree: every document with the encodings it is written in, and every file with its format, schema, source, size, hash and whether the file on disk still matches. Optionally narrowed to one document or one encoding. Reads only; `majordomus generate` writes and `majordomus generate --check` is the byte-for-byte verdict.",
310                input: ArtifactsInput,
311                output: ArtifactReport,
312                stability: Stability::BehaviorallyVerified,
313                exposure: Exposure {
314                    mcp: Some(McpExposure {
315                        tool: Some("majordomus_artifacts".into()),
316                        resource: Some(McpResource { uri: ARTIFACTS_URI.into(), name: "artifacts".into() }),
317                    }),
318                    http: get("/api/v1/artifacts"),
319                    cli: None,
320                },
321                tags: ["artifacts", "generation", "introspection"],
322                cache: CachePolicy::Process { max_entries: 8, ttl_seconds: Some(5) },
323                handler: artifacts_list,
324            },
325        ],
326    }
327}

329#[cfg(test)]
330mod tests {
331    use super::*;
332
333    /// The declaration is the only place the id, the tool name, the resource URI and the
334    /// route exist. A refactor that dropped one of them would still compile, and every
335    /// suite that tests the report itself would still pass; this is the assertion that
336    /// would not — which is what `project.rust-command-tested-in-file` asks of a command.
337    #[test]
338    fn the_declaration_yields_the_identity_and_the_projections_it_claims() {
339        let m = module();
340        assert_eq!(m.id.as_str(), "artifacts");
341        let ids: Vec<&str> = m
342            .capabilities
343            .iter()
344            .map(|e| e.capability.id.as_str())
345            .collect();
346        assert_eq!(ids, ["artifacts.list"]);
347
348        let c = &m.capabilities[0].capability;
349        let mcp = c.exposure.mcp.as_ref().expect("an MCP projection");
350        assert_eq!(mcp.tool.as_deref(), Some("majordomus_artifacts"));
351        assert_eq!(
352            mcp.resource.as_ref().map(|r| r.uri.as_str()),
353            Some(ARTIFACTS_URI)
354        );
355        assert_eq!(
356            c.exposure.http.as_ref().map(|h| h.path.as_str()),
357            Some("/api/v1/artifacts")
358        );
359        // read-only, and the registry refuses an executable capability that is not
360        assert!(c.kind.is_read_only() && c.kind.is_executable());
361        // and the module stamps its own namespace on what it composes
362        assert!(ids.iter().all(|id| id.starts_with("artifacts.")));
363    }
