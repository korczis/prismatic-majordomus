1//! One rule in full, for the Governance screen's inspector (`/ui/governance?rule=<id>`): what
2//! it states, how it is enforced, every test that proves it with the state of its proof, the
3//! gates that run them, what is still missing to prove it, and the rules it depends on and
4//! that depend on it, each a link to its own inspector.
5//!
6//! Read from `rules.show`, the capability that joins a rule to its proof; a rule the index
7//! does not hold is its refusal, said as such.
8

17/// A rule id as the index writes them (`project.worktree-topology`, `majordomus.adr-integrity@1`):
18/// letters, digits, `.`, `-`, `_` and `@`, and nothing a link could carry anything else in.
19pub fn is_rule_id(id: &str) -> bool {
20    !id.is_empty()
21        && id.len() <= 128
22        && id
23            .bytes()
24            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_' | b'@'))
25}

27/// The rule a query names, when it names one.
28pub fn picked(query: &[(String, String)]) -> Option<String> {
29    query
30        .iter()
31        .find(|(k, _)| k == "rule")
32        .map(|(_, v)| v.trim().to_string())
33        .filter(|v| is_rule_id(v))
34}

36/// Where a rule's inspector is.
37pub fn href(id: &str) -> String {
38    format!("{}?rule={id}#rule", super::governance::ROUTE)
39}
40
41/// The id of a rule as the report or the index gives it: the rule object's `id`, or the
42/// string itself when the report gives one.
43pub fn id_of(rule: &Value) -> String {
44    match rule {
45        Value::String(s) => s.clone(),
46        other => text(other, "id"),
47    }
48}

50/// Ask `rules.show` for `id`.
51pub fn ask(ctx: &Context, id: &str) -> Answer {
52    ask_input(ctx, "rules.show", serde_json::json!({ "rule": id }))
53}

55fn chips(ids: Vec<String>) -> El {
56    if ids.is_empty() {
57        return el("span").class("mj-ui-dim").text("none");
58    }
59    el("span")
60        .class("mj-ui-badges")
61        .children(ids.into_iter().map(|id| {
62            // a dependency is written with its version (`x@1`); the inspector takes either
63            el("a")
64                .class("mj-ui-chip mj-ui-chip--link")
65                .attr("href", href(&id))
66                .child(el("code").class("mj-ui-chip-value").text(id))
67        }))
68}
69
70fn strings(v: &Value, k: &str) -> Vec<String> {
71    v.get(k)
72        .and_then(Value::as_array)
73        .map(|a| {
74            a.iter()
75                .filter_map(|x| {
76                    x.as_str()
77                        .map(str::to_string)
78                        .or_else(|| x.get("id").and_then(Value::as_str).map(str::to_string))
79                })
80                .collect()
81        })
82        .unwrap_or_default()
83}

85/// The inspector of the rule `answer` is about.
86pub fn inspector(answer: &Answer, id: &str) -> El {
87    let body = unanswered(answer).unwrap_or_else(|| {
88        // `unanswered` has spoken for an answer that failed; this one answered
89        let a = answer.ok().cloned().unwrap_or(Value::Null);
90        let proof = a.get("proof").cloned().unwrap_or(Value::Null);
91        let rule = proof.get("rule").cloned().unwrap_or(Value::Null);
92        let enforcement = rule.get("enforcement").cloned().unwrap_or(Value::Null);
93        let fact = |label: &str, value: El| {
94            el("div")
95                .class("mj-ui-fact")
96                .child(el("dt").text(label))
97                .child(el("dd").child(value))
98        };
99        let tests = proof
100            .get("tests")
101            .and_then(Value::as_array)
102            .cloned()
103            .unwrap_or_default()
104            .iter()
105            .map(|t| {
106                el("tr")
107                    .child(el("td").class("mj-ui-mono").text(text(t, "path")))
108                    .child(el("td").child(state(t, "state")))
109                    .child(el("td").class("mj-ui-dim").text(text(t, "meaning")))
110            })
111            .collect::<Vec<_>>();
112        let missing = text(&a, "missing");
113        el("section")
114            .class("mj-ui-detail")
115            .attr("id", "rule")
116            .attr("aria-label", format!("Rule {id}"))
117            .child(
118                el("div")
119                    .class("mj-ui-detail-head")
120                    .child(
121                        el("p")
122                            .class("mj-ui-mono mj-ui-dim")
123                            .text(text(&rule, "identity")),
124                    )
125                    .child(
126                        el("h3")
127                            .class("mj-ui-detail-title")
128                            .text(text(&rule, "title")),
129                    )
130                    .child(
131                        el("p")
132                            .class("mj-ui-badges")
133                            .child(state(&rule, "class"))
134                            .child(state(&rule, "status"))
135                            .child(state(&proof, "state"))
136                            .child(
137                                el("a")
138                                    .class("mj-ui-more")
139                                    .attr("href", super::governance::ROUTE)
140                                    .text("Close"),
141                            ),
142                    ),
143            )
144            .child(el("p").class("mj-ui-lead").text(text(&rule, "statement")))
145            .child(el("p").class("mj-ui-line").text(text(&proof, "meaning")))
146            .child(
147                el("dl")
148                    .class("mj-ui-facts")
149                    .child(fact(
150                        "Enforced",
151                        el("span").class("mj-ui-mono").text(format!(
152                            "{} · {}",
153                            Some(text(&enforcement, "mode"))
154                                .filter(|s| !s.is_empty())
155                                .unwrap_or_else(|| "not said".into()),
156                            Some(text(&enforcement, "validator"))
157                                .filter(|s| !s.is_empty())
158                                .unwrap_or_else(|| "no validator".into())
159                        )),
160                    ))
161                    .child(fact(
162                        "Gates",
163                        el("span").class("mj-ui-mono").text({
164                            let g = strings(&proof, "gates");
165                            if g.is_empty() {
166                                "none".into()
167                            } else {
168                                g.join(", ")
169                            }
170                        }),
171                    ))
172                    .child(fact("Depends on", chips(strings(&a, "depends_on"))))
173                    .child(fact("Required by", chips(strings(&a, "required_by"))))
174                    .child(fact(
175                        "Declared in",
176                        el("span").class("mj-ui-mono").text(text(&rule, "path")),
177                    )),
178            )
179            .child(el("h4").class("mj-ui-panel-title").text("What proves it"))
180            .child(table(
181                &["Test", "State", "What that means"],
182                tests,
183                "It names no test.",
184            ))
185            .when(!missing.is_empty(), |s| {
186                s.child(el("h4").class("mj-ui-panel-title").text("To prove it"))
187                    .child(el("code").class("mj-ui-mono").text(missing))
188            })
189    });
190    panel(
191        "Rule",
192        "One rule of the repository, with what proves it and what is still missing.",
193        body,
194        &[answer],
195    )
196}

214    #[test]
215    fn a_rule_shows_its_proof_and_leads_to_the_rules_around_it() {
216        let a = answer(Ok(serde_json::json!({
217            "proof": {
218                "rule": { "id": "project.a", "identity": "project.a@1", "title": "A holds", "statement": "A is true.",
219                          "class": "blocking", "status": "active", "path": ".ai/repo/rules/project/a.v1.md",
220                          "enforcement": { "mode": "dispatched", "validator": "adr" } },
221                "tests": [{ "path": "test/cases/96_a.sh", "state": "not_run", "meaning": "never run" }],
222                "gates": ["shell-suite"], "state": "not_run", "meaning": "No run is recorded."
223            },
224            "depends_on": ["project.b@1"], "required_by": [{ "id": "project.c" }],
225            "missing": "Run and record: bash test/run.sh 96_a"
226        })));
227        let html = inspector(&a, "project.a").render();
228        for want in [
229            "id=\"rule\"",
230            "A is true.",
231            "test/cases/96_a.sh",
232            "shell-suite",
233            "/ui/governance?rule=project.b@1#rule",
234            "/ui/governance?rule=project.c#rule",
235            "bash test/run.sh 96_a",
236        ] {
237            assert!(html.contains(want), "no {want}: {html}");
238        }
239    }

241    /// A rule its index says little about still opens: no dependency either way, no
242    /// enforcement named, no gate, each said rather than left blank.
243    #[test]
244    fn a_rule_that_says_little_says_what_it_lacks() {
245        let html = inspector(
246            &answer(Ok(serde_json::json!({
247                "proof": { "rule": { "id": "project.z", "title": "Z" }, "tests": [], "state": "not_run" }
248            }))),
249            "project.z",
250        )
251        .render();
252        for want in ["not said", "no validator", "It names no test.", ">none<"] {
253            assert!(html.contains(want), "no {want}: {html}");
254        }
255    }
256
257    #[test]
258    fn an_unknown_rule_is_its_refusal_and_a_hostile_id_is_never_asked() {
259        let refused = inspector(&answer(Err("not found: no rule nope".into())), "nope").render();
260        assert!(refused.contains("no rule nope") && !refused.contains("id=\"rule\""));
261        let q = |v: &str| vec![("rule".to_string(), v.to_string())];
262        assert_eq!(
263            picked(&q("majordomus.adr-integrity@1")),
264            Some("majordomus.adr-integrity@1".into())
265        );
266        assert_eq!(picked(&q("x\" onclick=\"y")), None);
267        assert_eq!(id_of(&serde_json::json!({ "id": "p.q" })), "p.q");
268        assert_eq!(id_of(&serde_json::json!("p.r")), "p.r");
269    }
