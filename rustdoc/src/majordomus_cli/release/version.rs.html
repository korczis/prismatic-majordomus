1//! The version: the one place it is authored, what is derived from that place, what the
2//! contract requires next and what the commits imply as evidence, and the one writer that
3//! raises it.
4//!
5//! The product version is authored in exactly one place, the `[package] version` line of
6//! the crate manifest ([`MANIFEST`]). Every other statement of it is derived from that line
7//! and checked, or is a record of a version that was released:
8//!
9//! ```text
10//!   apps/majordomus-cli/Cargo.toml   authored    the one place; `release bump` writes it
11//!   crate::VERSION                   compiled    cargo's CARGO_PKG_VERSION
12//!   share/version.txt                projected   `majordomus generate`; bin/majordomus reads it
13//!   apps/majordomus-cli/Cargo.lock   derived     cargo's record; the writer keeps it in step
14//!   generator stamps, the changelog  generated   `majordomus generate`
15//!   .ai/repo/releases/*.yaml         records     the release pipeline, after publication
16//! ```
17//!
18//! The shell tool used to state the version a second time, by hand, because an installed
19//! tree has no `Cargo.toml` and the tool cannot read one at run time. It reads
20//! [`PROJECTION`] instead: a typed generated artifact that ships in every archive beside the
21//! tool, that `generate --check` refuses when it falls behind the manifest, and that nobody
22//! edits. [`diagnose`] is what refuses a version anybody writes down by hand where the
23//! tool's own files live.
24//!
25//! One reader of the authority per language: [`declared`] here, `scripts/release-version`
26//! in shell. Nothing else parses the manifest for its version.
27//!
28//! ```
29//! use majordomus_cli::release::version::{declared_in, parse_projection, render_projection};
30//! // the authority, and the one line the shell tool reads back from its projection of it
31//! let authority = declared_in("[package]\nversion = \"0.9.0\"\n").unwrap();
32//! assert_eq!(parse_projection(&render_projection(&authority)).as_deref(), Some("0.9.0"));
33//! ```

114/// How much a set of changes raises a version.
115#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
116pub enum Bump {
117    /// Nothing since the last release.
118    None,
119    /// A fix, or anything that is not a feature.
120    Patch,
121    /// At least one feature.
122    Minor,
123    /// At least one breaking change.
124    Major,
125}

127impl Bump {
128    /// The word the reports use.
129    pub fn as_str(self) -> &'static str {
130        match self {
131            Bump::None => "none",
132            Bump::Patch => "patch",
133            Bump::Minor => "minor",
134            Bump::Major => "major",
135        }
136    }
137
138    /// The bump named by a word, for `release bump --minor` and friends.
139    pub fn parse(word: &str) -> Option<Bump> {
140        match word {
141            "major" => Some(Bump::Major),
142            "minor" => Some(Bump::Minor),
143            "patch" => Some(Bump::Patch),
144            "none" => Some(Bump::None),
145            _ => None,
146        }
147    }

150/// What a set of conventional commits *says about itself*.
151///
152/// The rule is the conventional-commit rule and nothing more: a breaking change is major, a
153/// feature is minor, anything else is patch, and no commits at all is none.
154///
155/// # This is evidence, not authority
156///
157/// It used to decide what `release bump` wrote, and that was the defect this subsystem
158/// exists to remove. A commit message is a label a person chose, and a capability deleted
159/// under a `refactor:` heading was a patch to this function and a broken caller to everyone
160/// else. The compatibility level is now measured from the public contract by
161/// [`crate::release::compat::analyze`]; this answer is carried beside it as
162/// [`crate::release::compat::CommitEvidence`], and when the two disagree the plan says which
163/// one understated the change. Nothing decides a version from this alone.
164pub fn bump_of(changes: &[Change]) -> Bump {
165    if changes.is_empty() {
166        return Bump::None;
167    }
168    if changes.iter().any(|c| c.breaking) {
169        return Bump::Major;
170    }
171    if changes.iter().any(|c| c.kind == ChangeKind::Feat) {
172        return Bump::Minor;
173    }
174    Bump::Patch
175}

177/// One dot-separated identifier of a pre-release part (`alpha`, `1`).
178///
179/// Numeric identifiers compare as numbers and rank below alphanumeric ones, which compare as
180/// ASCII text (semantic versioning 2.0.0, rule 11), so the derived order — `Numeric` before
181/// `Alpha` — is the specified one, and a list of them compares one by one with a shorter
182/// prefix first, which is `Vec`'s own order.
183///
184/// ```
185/// use majordomus_cli::release::version::Identifier;
186/// assert!(Identifier::Numeric(10) > Identifier::Numeric(2), "numbers, not text");
187/// assert!(Identifier::Numeric(99) < Identifier::Alpha("a".into()), "a number sorts first");
188/// assert!(Identifier::Alpha("alpha".into()) < Identifier::Alpha("beta".into()));
189/// ```
190#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
191pub enum Identifier {
192    /// Digits only, with no leading zero: compared as a number.
193    Numeric(u64),
194    /// Anything else of `[0-9A-Za-z-]`: compared as ASCII text.
195    Alpha(String),
196}
197
198impl std::fmt::Display for Identifier {
199    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
200        match self {
201            Identifier::Numeric(n) => write!(f, "{n}"),
202            Identifier::Alpha(s) => f.write_str(s),
203        }
204    }
205}

207/// A semantic version (2.0.0): three numbers, an optional pre-release part and optional build
208/// metadata — `1.0.0-alpha.1+build.5`.
209///
210/// # Precedence is the order
211///
212/// `Ord` is semantic-version precedence and nothing else, the one rule every reader of a
213/// version in this crate uses — the release writer, the compatibility gate, the landing
214/// allocator and the stable pointer of the `distribution` module — so no two of them can rank
215/// the same pair of versions differently (ADR 0004):
216///
217/// - the three numbers, as numbers;
218/// - then a pre-release below its own final: `1.0.0-alpha.1 < 1.0.0`, and `0.29.0 <
219///   1.0.0-alpha.1`;
220/// - then the pre-release identifiers one by one ([`Identifier`]), a shorter list below a
221///   longer one it prefixes.
222///
223/// Build metadata is ignored, by `Ord` and by `==` alike, because two orders that disagree
224/// about equality are not one order: `1.0.0+a == 1.0.0+b`, and [`Display`](std::fmt::Display)
225/// still writes each back as it was read.
226///
227/// ```
228/// use majordomus_cli::release::version::Version;
229/// let v = |s| Version::parse(s).unwrap();
230/// assert!(v("1.0.0-alpha.1") < v("1.0.0-alpha.2"));
231/// assert!(v("1.0.0-alpha.2") < v("1.0.0-alpha.10"), "numeric identifiers are numbers");
232/// assert!(v("1.0.0-alpha.10") < v("1.0.0-beta"));
233/// assert!(v("1.0.0-beta") < v("1.0.0"), "a pre-release sorts below its final");
234/// assert!(v("0.29.0") < v("1.0.0-alpha.1"));
235/// assert_eq!(v("1.0.0+a"), v("1.0.0+b"), "build metadata has no precedence");
236/// assert_eq!(v("1.0.0-alpha.1+b.7").to_string(), "1.0.0-alpha.1+b.7");
237/// ```
238#[derive(Debug, Clone)]
239pub struct Version {
240    /// Major.
241    pub major: u64,
242    /// Minor.
243    pub minor: u64,
244    /// Patch.
245    pub patch: u64,
246    /// The pre-release identifiers; empty for a final release.
247    pub pre: Vec<Identifier>,
248    /// The build metadata identifiers, kept for display and ignored by precedence.
249    pub build: Vec<String>,
250}
251
252impl PartialEq for Version {
253    fn eq(&self, other: &Self) -> bool {
254        self.cmp(other) == std::cmp::Ordering::Equal
255    }
256}

258impl Eq for Version {}

260impl PartialOrd for Version {
261    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {

263    }
264}

266impl Ord for Version {
267    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
268        (self.major, self.minor, self.patch)
269            .cmp(&(other.major, other.minor, other.patch))
270            // a final release (no identifiers) ranks above every pre-release of it
271            .then_with(|| self.pre.is_empty().cmp(&other.pre.is_empty()))
272            .then_with(|| self.pre.cmp(&other.pre))
273    }
274}

276/// Is `text` a non-empty identifier of `[0-9A-Za-z-]`, as semantic versioning requires of
277/// every pre-release and build identifier?
278fn is_identifier(text: &str) -> bool {
279    !text.is_empty() && text.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'-')
280}

282/// A numeric part: digits only, and no leading zero unless it is `0` itself.
283fn numeric(text: &str) -> Option<u64> {
284    if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
285        return None;
286    }
287    if text.len() > 1 && text.starts_with('0') {
288        return None;
289    }
290    text.parse().ok()
291}

293impl Version {
294    /// A final release of three numbers, with no pre-release part and no build metadata.
295    ///
296    /// ```
297    /// use majordomus_cli::release::version::Version;
298    /// assert_eq!(Version::new(0, 29, 0).to_string(), "0.29.0");
299    /// ```
300    pub fn new(major: u64, minor: u64, patch: u64) -> Version {
301        Version {
302            major,
303            minor,
304            patch,
305            pre: Vec::new(),
306            build: Vec::new(),
307        }
308    }
309
310    /// Read a semantic version, `1.2.3`, `1.0.0-alpha.1` or `1.0.0+build`; anything else is
311    /// `None`.
312    ///
313    /// Strict, because a version this accepts is one every reader orders: exactly three
314    /// numbers, none with a leading zero; a pre-release and build identifiers that are
315    /// non-empty `[0-9A-Za-z-]`; and no numeric pre-release identifier with a leading zero.
316    ///
317    /// ```
318    /// use majordomus_cli::release::version::Version;
319    /// assert!(Version::parse("1.0.0-alpha.1").is_some());
320    /// assert!(Version::parse("1.0.0-alpha..1").is_none(), "an empty identifier");
321    /// assert!(Version::parse("1.0.0-alpha.01").is_none(), "a leading zero");
322    /// assert!(Version::parse("1.2.3.4").is_none(), "four numbers");
323    /// ```
324    pub fn parse(text: &str) -> Option<Version> {
325        let text = text.trim();
326        let (text, build) = match text.split_once('+') {
327            Some((v, b)) => (v, Some(b)),
328            None => (text, None),
329        };
330        let (core, pre) = match text.split_once('-') {
331            Some((c, p)) => (c, Some(p)),
332            None => (text, None),
333        };
334        // exactly three parts: a split always yields a first one, so asking for each part in
335        // turn left a branch no input reaches; the count is decided once instead
336        let mut parts = core.split('.');
337        let (Some(major), Some(minor), Some(patch), None) =
338            (parts.next(), parts.next(), parts.next(), parts.next())
339        else {
340            return None;
341        };
342        let (major, minor, patch) = (numeric(major)?, numeric(minor)?, numeric(patch)?);
343        let pre = match pre {
344            None => Vec::new(),
345            Some(p) => p
346                .split('.')
347                .map(|i| {
348                    if !is_identifier(i) {
349                        None
350                    } else if i.bytes().all(|b| b.is_ascii_digit()) {
351                        numeric(i).map(Identifier::Numeric)
352                    } else {
353                        Some(Identifier::Alpha(i.to_string()))
354                    }
355                })
356                .collect::<Option<Vec<_>>>()?,
357        };
358        let build = match build {
359            None => Vec::new(),
360            Some(b) => b
361                .split('.')
362                .map(|i| is_identifier(i).then(|| i.to_string()))
363                .collect::<Option<Vec<_>>>()?,
364        };
365        Some(Version {
366            major,
367            minor,
368            patch,
369            pre,
370            build,
371        })
372    }
373
374    /// Does this version carry a pre-release part (`1.0.0-alpha.1`)? Build metadata does not
375    /// make one: `1.0.0+x` is a final release.
376    ///
377    /// ```
378    /// use majordomus_cli::release::version::Version;
379    /// assert!(Version::parse("1.0.0-alpha.1").unwrap().is_prerelease());
380    /// assert!(!Version::parse("1.0.0+x").unwrap().is_prerelease());
381    /// ```
382    pub fn is_prerelease(&self) -> bool {
383        !self.pre.is_empty()
384    }
385
386    /// The final release this version is a pre-release of — itself, for a final release —
387    /// without build metadata.
388    ///
389    /// ```
390    /// use majordomus_cli::release::version::Version;
391    /// assert_eq!(Version::parse("1.0.0-rc.1+b").unwrap().final_release().to_string(), "1.0.0");
392    /// ```
393    pub fn final_release(&self) -> Version {
394        Version::new(self.major, self.minor, self.patch)
395    }
396
397    /// Does this version meet a floor — the smallest version a tree may declare?
398    ///
399    /// At or above it does, and so does a pre-release *of* a final floor: a floor of `1.0.0`
400    /// is met by `1.0.0-alpha.2`, because a pre-release of the version owed is that version
401    /// on its way, and it promises no compatibility of its own (semantic versioning 2.0.0,
402    /// rule 9). Without this, a project could not publish a second alpha: every raise of
403    /// `1.0.0-alpha.1` is `1.0.0`. A floor that is itself a pre-release is met only by
404    /// precedence, so nothing goes below it.
405    ///
406    /// ```
407    /// use majordomus_cli::release::version::Version;
408    /// let v = |s| Version::parse(s).unwrap();
409    /// assert!(v("1.0.0-alpha.2").reaches(&v("1.0.0")), "a pre-release of what is owed");
410    /// assert!(v("1.0.0-alpha.1").reaches(&v("0.30.0")));
411    /// assert!(!v("0.30.0-rc.1").reaches(&v("1.0.0")));
412    /// assert!(!v("1.0.0-alpha.1").reaches(&v("1.0.0-alpha.2")), "never below a pre-release");
413    /// ```
414    pub fn reaches(&self, floor: &Version) -> bool {
415        self >= floor || (!floor.is_prerelease() && self.final_release() >= *floor)
416    }
417
418    /// This version raised by `bump`.
419    ///
420    /// Arithmetic and nothing else: zero-major is not special-cased here, because whether a
421    /// breaking change below 1.0 costs a major or a minor is a *policy* question and this
422    /// repository answers it in exactly one place, [`crate::release::compat::Policy`]. A
423    /// second answer here is the shape this subsystem was built to remove.
424    ///
425    /// From a pre-release the rule is semantic versioning's: the smallest final release of
426    /// that level above it, which for the level the pre-release already leads up to is its
427    /// own final — `1.0.0-alpha.1` raised by a major, a minor or a patch is `1.0.0`, and
428    /// `1.2.3-rc.1` raised by a patch is `1.2.3`. A raise drops build metadata; no raise at
429    /// all is the version as it is.
430    ///
431    /// ```
432    /// use majordomus_cli::release::version::{Bump, Version};
433    /// let v = |s| Version::parse(s).unwrap();
434    /// assert_eq!(v("1.0.0-alpha.1").raised(Bump::Major).to_string(), "1.0.0");
435    /// assert_eq!(v("1.0.0-alpha.1").raised(Bump::Patch).to_string(), "1.0.0");
436    /// assert_eq!(v("1.2.3-rc.1").raised(Bump::Minor).to_string(), "1.3.0");
437    /// assert_eq!(v("0.29.0").raised(Bump::Major).to_string(), "1.0.0");
438    /// ```
439    pub fn raised(self, bump: Bump) -> Version {
440        let pre = self.is_prerelease();
441        let (major, minor, patch) = (self.major, self.minor, self.patch);
442        match bump {
443            Bump::None => self,
444            Bump::Patch if pre => Version::new(major, minor, patch),
445            Bump::Patch => Version::new(major, minor, patch + 1),
446            Bump::Minor if pre && patch == 0 => Version::new(major, minor, 0),
447            Bump::Minor => Version::new(major, minor + 1, 0),
448            Bump::Major if pre && minor == 0 && patch == 0 => Version::new(major, 0, 0),
449            Bump::Major => Version::new(major + 1, 0, 0),
450        }
451    }
452}

456    ///
457    /// The bridge between the engine's verdict and the arithmetic: the policy has already
458    /// decided what a `0.x` breaking change costs, so this only has to apply it.
459    ///
460    /// ```

510/// The `[package] version` of a manifest's text.
511///
512/// The first `version = "..."` inside `[package]`, which is where the manifest puts it: a
513/// dependency pinned at a version further down is a different package, a key that merely
514/// begins with `version` is a different key, and a trailing comment is not part of the value.
515/// Read without a TOML parser, because the value wanted is one quoted scalar in a table whose
516/// header is unambiguous, and a dependency to read it would be one this crate argues against.
517///
518/// ```
519/// use majordomus_cli::release::version::declared_in;
520/// let manifest = "[package]\nname = \"x\"\nversion = \"1.2.3\"   # bumped\n\n\
521///                 [dependencies]\nserde = { version = \"1.2.4\" }\n";
522/// assert_eq!(declared_in(manifest).as_deref(), Some("1.2.3"));
523/// // outside [package] there is no package version at all
524/// assert_eq!(declared_in("[dependencies]\nversion = \"1.0.0\"\n"), None);
525/// // a key that only starts with the word is another key
526/// assert_eq!(declared_in("[package]\nversioned = \"1.0.0\"\n"), None);
527/// ```
528pub fn declared_in(manifest: &str) -> Option<String> {
529    let mut in_package = false;
530    for line in manifest.lines() {
531        let line = line.trim();
532        if line.starts_with('[') {
533            in_package = line == "[package]";
534            continue;
535        }
536        if !in_package {
537            continue;
538        }
539        let Some(rest) = line.strip_prefix("version") else {
540            continue;
541        };
542        let Some(rest) = rest.trim_start().strip_prefix('=') else {
543            continue;
544        };
545        let rest = rest.trim();
546        let value = match rest.strip_prefix('"') {
547            Some(quoted) => quoted.split('"').next().unwrap_or_default(),
548            None => rest.split('#').next().unwrap_or_default().trim(),
549        };
550        return (!value.is_empty()).then(|| value.to_string());
551    }
552    None
553}

558/// version is a different package and is never read or written here.

601///

635/// How [`PROJECTION`] stands against the authority.
636///
637/// Three answers that lead to three different actions: nothing to do, run the derivation,
638/// or find out who wrote a version nobody authored.
639///
640/// ```
641/// use majordomus_cli::release::version::ProjectionState;
642/// let stale = ProjectionState::Stale { projected: Some("0.8.0".into()), declared: "0.9.0".into() };
643/// // the state every bump leaves behind is not the state nothing could have produced
644/// assert_ne!(stale, ProjectionState::Disagrees { projected: "0.8.0".into(), declared: Some("0.9.0".into()) });
645/// ```
646#[derive(Debug, Clone, PartialEq, Eq)]
647pub enum ProjectionState {
648    /// Neither the manifest nor the projection is here: this repository does not carry the
649    /// product, so there is nothing to project and nothing to compare.
650    NotHere,
651    /// The projection states the version the manifest declares.
652    Current,
653    /// The projection is missing, or states an older version than the manifest: the state
654    /// between `release bump` and the derivation that follows it. `generate` fixes it.
655    Stale {
656        /// What the projection states, when it states anything.
657        projected: Option<String>,
658        /// What the manifest declares.
659        declared: String,
660    },
661    /// The projection states a version newer than the manifest's, or one that cannot be
662    /// compared with it, or there is a projection and no manifest. No derivation writes
663    /// that, so somebody did.
664    Disagrees {
665        /// What the projection states.
666        projected: String,
667        /// What the manifest declares, when there is one.
668        declared: Option<String>,
669    },
670}
671
672/// How [`PROJECTION`] stands against [`MANIFEST`]: compared as versions, so that a projection
673/// behind the manifest — what a bump leaves until the derivation runs — is told apart from
674/// one ahead of it or unrelated to it, which no derivation ever writes.
675///
676/// ```
677/// use majordomus_cli::release::version::{projection_state, ProjectionState, MANIFEST, PROJECTION};
678/// let dir = tempfile::tempdir().unwrap();
679/// let root = dir.path();
680/// assert_eq!(projection_state(root), ProjectionState::NotHere);
681/// std::fs::create_dir_all(root.join("apps/majordomus-cli")).unwrap();
682/// std::fs::create_dir_all(root.join("share")).unwrap();
683/// std::fs::write(root.join(MANIFEST), "[package]\nversion = \"0.9.0\"\n").unwrap();
684/// // raised and not yet derived: stale, which a derivation fixes
685/// std::fs::write(root.join(PROJECTION), "version=0.8.0\n").unwrap();
686/// assert!(matches!(projection_state(root), ProjectionState::Stale { .. }));
687/// // a version the manifest never declared: nothing derives that
688/// std::fs::write(root.join(PROJECTION), "version=1.0.0\n").unwrap();
689/// assert!(matches!(projection_state(root), ProjectionState::Disagrees { .. }));
690/// std::fs::write(root.join(PROJECTION), "version=0.9.0\n").unwrap();
691/// assert_eq!(projection_state(root), ProjectionState::Current);
692/// ```
693pub fn projection_state(root: &Path) -> ProjectionState {
694    match (declared(root), tool(root)) {
695        (None, None) => ProjectionState::NotHere,
696        (None, Some(projected)) => ProjectionState::Disagrees {
697            projected,
698            declared: None,
699        },
700        (Some(declared), None) => ProjectionState::Stale {
701            projected: None,
702            declared,
703        },
704        (Some(declared), Some(projected)) if declared == projected => ProjectionState::Current,
705        (Some(declared), Some(projected)) => {
706            match (Version::parse(&projected), Version::parse(&declared)) {
707                (Some(p), Some(d)) if p < d => ProjectionState::Stale {
708                    projected: Some(projected),
709                    declared,
710                },
711                _ => ProjectionState::Disagrees {
712                    projected,
713                    declared: Some(declared),
714                },
715            }
716        }
717    }
718}

723/// ```

746/// ```

749    /// Repository-relative path.
750    pub path: String,
751    /// One-based line number.

759/// The shape a line states a version in, when it states one.
760///
761/// Three shapes, each a place a version is *declared* rather than mentioned:
762///
763/// - an assignment to a name ending in `version`, in any case and in the languages the
764///   tool's files are written in: shell `MJ_VERSION="0.8.0"` and `export FOO_VERSION=1.2.3`,
765///   TOML `version = "1.2.3"`, JavaScript `const VERSION = '1.2.3'`, Python
766///   `__version__ = "1.2.3"`;
767/// - a `version` member of YAML or JSON at the start of a line: `version: 1.2.3`,
768///   `"version": "1.2.3"`;
769/// - the product's own answer to `version`: `majordomus 0.8.0`.
770///
771/// A version is three numbers, optionally after a `v`; a schema's `version: 1` is not one.
772///
773/// ```
774/// use majordomus_cli::release::version::carrier_shape;
775/// assert!(carrier_shape("MJ_VERSION=\"0.8.0\"").is_some());
776/// assert!(carrier_shape("  export FOO_VERSION=v1.2.3").is_some());
777/// assert!(carrier_shape("declare -r MJ_VERSION=\"1.2.3\"").is_some());
778/// assert!(carrier_shape("version = \"1.2.3\"").is_some());
779/// assert!(carrier_shape("export const VERSION = '1.2.3';").is_some());
780/// assert!(carrier_shape("let appVersion = `1.2.3`;").is_some());
781/// assert!(carrier_shape("__version__ = \"1.2.3\"").is_some());
782/// assert!(carrier_shape("version: 1.2.3").is_some());
783/// assert!(carrier_shape("  \"version\": \"1.2.3\",").is_some());
784/// assert!(carrier_shape("echo \"majordomus 0.8.0\"").is_some());
785/// // read, not stated: a variable, a schema version, a generator stamp, a sentinel, a
786/// // comparison, a property of something else
787/// assert!(carrier_shape("MJ_VERSION=\"${line#version=}\"").is_none());
788/// assert!(carrier_shape("version: 1").is_none());
789/// assert!(carrier_shape("# Generator: majordomus-cli 0.8.0").is_none());
790/// assert!(carrier_shape("MJ_VERSION=\"site\"").is_none());
791/// assert!(carrier_shape("version == \"1.2.3\"").is_none());
792/// assert!(carrier_shape("pkg.version = \"1.2.3\"").is_none());
793/// assert!(carrier_shape("versioned = \"1.2.3\"").is_none());
794/// ```
795pub fn carrier_shape(line: &str) -> Option<&'static str> {
796    let t = line.trim_start();
797
798    // an assignment to a name ending in `version`, once the words that declare it are set aside
799    if let Some((name, value)) = declared_name(t).split_once('=') {
800        let name = name.trim_end();
801        let is_version_name = !name.is_empty()
802            && name
803                .chars()
804                .all(|c| c.is_ascii_alphanumeric() || c == '_' || c == '$')
805            && name
806                .trim_matches('_')
807                .to_ascii_lowercase()
808                .ends_with("version");
809        let value = value.trim_start().trim_start_matches(['"', '\'', '`']);
810        if is_version_name && leads_with_version(value) {
811            return Some("an assignment to a version name");
812        }
813    }
814
815    // a `version` member of YAML or JSON, at the start of a line or of a list item
816    let member = t.strip_prefix("- ").unwrap_or(t);
817    let after_key = member
818        .strip_prefix("\"version\"")
819        .or_else(|| member.strip_prefix("'version'"))
820        .or_else(|| member.strip_prefix("version"));
821    if let Some(rest) = after_key {
822        if let Some(value) = rest.trim_start().strip_prefix(':') {
823            if leads_with_version(value.trim_start().trim_start_matches(['"', '\''])) {
824                return Some("a version member");
825            }
826        }
827    }
828
829    // the product's own answer to `majordomus version`
830    const PRODUCT: &str = "majordomus ";
831    for (at, _) in t.match_indices(PRODUCT) {
832        let before = t[..at].chars().next_back();
833        let joined = before.is_some_and(|c| c.is_ascii_alphanumeric() || "-_./".contains(c));
834        if !joined && leads_with_version(&t[at + PRODUCT.len()..]) {
835            return Some("the product's version line");
836        }
837    }
838    None
839}

841/// A line with the words that declare a name set aside — `export`, `readonly`, `local`,
842/// `declare` and `typeset` with their flags, `const`, `let`, `var` — so that what is left
843/// begins with the name being assigned.
844fn declared_name(mut t: &str) -> &str {
845    const DECLARING: &[&str] = &[
846        "export", "readonly", "local", "declare", "typeset", "const", "let", "var",
847    ];
848    const TAKES_FLAGS: &[&str] = &["export", "readonly", "local", "declare", "typeset"];
849    while let Some((word, rest)) = t.split_once(char::is_whitespace) {
850        if !DECLARING.contains(&word) {
851            break;
852        }
853        t = rest.trim_start();
854        if TAKES_FLAGS.contains(&word) {
855            while let Some((flag, rest)) = t.split_once(char::is_whitespace) {
856                if flag.len() > 1 && flag.starts_with('-') && !flag.contains('=') {
857                    t = rest.trim_start();
858                } else {
859                    break;
860                }
861            }
862        }
863    }
864    t
865}

867/// Whether `text` begins with a version: three dot-separated numbers, optionally after a
868/// `v`, and not followed by a fourth.
869fn leads_with_version(text: &str) -> bool {
870    let bytes = text.strip_prefix('v').unwrap_or(text).as_bytes();
871    let mut at = 0;
872    for group in 0..3 {
873        let start = at;
874        while at < bytes.len() && bytes[at].is_ascii_digit() {
875            at += 1;
876        }
877        if at == start {
878            return false;
879        }
880        if group < 2 {
881            if bytes.get(at) != Some(&b'.') {
882                return false;
883            }
884            at += 1;
885        }
886    }
887    !matches!(bytes.get(at), Some(b) if *b == b'.' || b.is_ascii_digit())
888}

890/// Every version written down by hand under [`CARRIER_ROOTS`].
891///
892/// The files git would commit there — tracked, or new and not ignored, so a file is refused
893/// before it is staged rather than after — or, where this is not a git checkout, every file
894/// under those directories, so that a tree nobody committed is still read rather than
895/// passed. [`PROJECTION`] is not a statement by hand, and neither is any path the generated
896/// artifact manifest names: `majordomus generate` wrote those, and `generate --check` is
897/// what holds them.
898///
899/// ```
900/// use majordomus_cli::release::version::carriers;
901/// let dir = tempfile::tempdir().unwrap();
902/// std::fs::create_dir_all(dir.path().join("lib")).unwrap();
903/// std::fs::write(dir.path().join("lib/x.sh"), "FOO_VERSION=\"1.2.3\"\n").unwrap();
904/// let found = carriers(dir.path());
905/// assert_eq!(found.len(), 1);
906/// assert_eq!((found[0].path.as_str(), found[0].line), ("lib/x.sh", 1));
907/// ```
908pub fn carriers(root: &Path) -> Vec<Carrier> {
909    let files = committable(root).unwrap_or_else(|| walk(root, CARRIER_ROOTS));
910    let generated = generated_paths(root);
911    let mut out = Vec::new();
912    for path in files {
913        if path == PROJECTION || generated.contains(&path) {
914            continue;
915        }
916        let full = root.join(&path);
917        let small =
918            std::fs::symlink_metadata(&full).is_ok_and(|m| m.is_file() && m.len() <= LARGEST_READ);
919        if !small {
920            continue;
921        }
922        let Ok(text) = std::fs::read_to_string(&full) else {
923            continue;
924        };
925        for (i, line) in text.lines().enumerate() {
926            if let Some(shape) = carrier_shape(line) {
927                out.push(Carrier {
928                    path: path.clone(),
929                    line: i + 1,
930                    shape,
931                    text: line.trim().to_string(),
932                });
933            }
934        }
935    }
936    out
937}

939/// The files under [`CARRIER_ROOTS`] git would commit: tracked, or untracked and not
940/// ignored. `None` when git cannot answer, which is not the same as "no files".
941fn committable(root: &Path) -> Option<Vec<String>> {
942    let out = crate::git::read_only(root)
943        .args([
944            "ls-files",
945            "-z",
946            "--cached",
947            "--others",
948            "--exclude-standard",
949            "--",
950        ])
951        .args(CARRIER_ROOTS)
952        .output()
953        .ok()?;
954    if !out.status.success() {
955        return None;
956    }
957    let mut files: Vec<String> = out
958        .stdout
959        .split(|b| *b == 0)
960        .filter(|s| !s.is_empty())
961        .map(|s| String::from_utf8_lossy(s).into_owned())
962        .collect();
963    crate::order::canonical_strings(&mut files);
964    files.dedup();
965    Some(files)
966}

968/// Every file under `roots` — directories walked, files taken as they are —
969/// repository-relative, in a stable order: the answer when git cannot list them.
970fn walk(root: &Path, roots: &[&str]) -> Vec<String> {
971    fn visit(dir: &Path, root: &Path, out: &mut Vec<String>) {
972        let Ok(entries) = std::fs::read_dir(dir) else {
973            return;
974        };
975        for entry in entries.flatten() {
976            let path = entry.path();
977            match entry.file_type() {
978                Ok(t) if t.is_dir() => visit(&path, root, out),
979                Ok(t) if t.is_file() => {
980                    if let Ok(rel) = path.strip_prefix(root) {
981                        out.push(rel.to_string_lossy().replace('\\', "/"));
982                    }
983                }
984                _ => {}
985            }
986        }
987    }
988    let mut out = Vec::new();
989    for dir in roots {
990        let full = root.join(dir);
991        if full.is_file() {
992            out.push((*dir).to_string());
993        } else {
994            visit(&full, root, &mut out);
995        }
996    }
997    crate::order::canonical_strings(&mut out);
998    out
999}

1013/// Every version label on a line: `v`, a major and a minor, and no patch — `v0.1`, `v1.2`.
1014///
1015/// A release is always three numbers (`v0.3.1`), so a label of two names no release: it
1016/// names the version a text was written about, and it stays where it was typed while the
1017/// version moves on. A label joined to a word, a path or a URL is not one, and neither is a
1018/// three-part version or a schema's `version: 1`.
1019///
1020/// ```
1021/// use majordomus_cli::release::version::version_labels;
1022/// assert_eq!(version_labels("What v0.1 does not do"), ["v0.1"]);
1023/// assert_eq!(version_labels("(v1.12), and v2.0."), ["v1.12", "v2.0"]);
1024/// assert!(version_labels("Release `v0.3.1` was published").is_empty());
1025/// assert!(version_labels("see /releases/v0.1.json and api-v0.1").is_empty());
1026/// assert!(version_labels("rule v2, schema version: 1, dev0.1").is_empty());
1027/// assert!(version_labels("v0.1.0 and v0.1x").is_empty());
1028/// ```
1029pub fn version_labels(line: &str) -> Vec<String> {
1030    let bytes = line.as_bytes();
1031    let joined = |b: u8| b.is_ascii_alphanumeric() || b"./_-".contains(&b);
1032    let digits = |mut at: usize| {
1033        while at < bytes.len() && bytes[at].is_ascii_digit() {
1034            at += 1;
1035        }
1036        at
1037    };
1038    let mut out = Vec::new();
1039    for (at, _) in line.match_indices('v') {
1040        if at > 0 && joined(bytes[at - 1]) {
1041            continue;
1042        }
1043        let major = digits(at + 1);
1044        if major == at + 1 || bytes.get(major) != Some(&b'.') {
1045            continue;
1046        }
1047        let minor = digits(major + 1);
1048        if minor == major + 1 {
1049            continue;
1050        }
1051        let patch =
1052            bytes.get(minor) == Some(&b'.') && bytes.get(minor + 1).is_some_and(u8::is_ascii_digit);
1053        let glued = bytes
1054            .get(minor)
1055            .is_some_and(|b| b.is_ascii_alphanumeric() || b"_-".contains(b));
1056        if !patch && !glued {
1057            out.push(line[at..minor].to_string());
1058        }
1059    }
1060    out
1061}

1063/// The tracked hand-written prose: documents, the site's authored pages and data, its
1064/// templates, the layer's own documents and the provider templates.
1065///
1066/// Generated files are not prose anybody wrote — a label in one comes from its source, and
1067/// the source is where it is found — so a path the generated-artifact manifest names, a
1068/// file stamped by its generator, and a page under `site/content/` that the site generator
1069/// projects from `site/content-src/` or writes into a subdirectory are all left out. The
1070/// layer's dated records ([`PROSE_RECORDS`]) are left out too. Where git cannot list the
1071/// tree — not a checkout — every file under the roots is read instead, as [`carriers`]
1072/// does, so a tree nobody committed is still read rather than passed.
1073fn prose_files(root: &Path) -> Vec<String> {
1074    let listed =
1075        crate::git::ls_files_any(root, PROSE_ROOTS).unwrap_or_else(|_| walk(root, PROSE_ROOTS));
1076    let generated = generated_paths(root);
1077    let prose = |p: &str| match p {
1078        "README.md" | "AGENTS.md" | "CLAUDE.md" => true,
1079        _ if PROSE_RECORDS.iter().any(|r| p.starts_with(r)) => false,
1080        _ if p.starts_with("site/content/") => {
1081            let name = &p["site/content/".len()..];
1082            p.ends_with(".md")
1083                && !name.contains('/')
1084                && !root.join("site/content-src").join(name).exists()
1085        }
1086        _ if p.starts_with("site/data/") => {
1087            p.ends_with(".toml") && !p["site/data/".len()..].contains('/')
1088        }
1089        _ if p.starts_with("share/providers/") || p.starts_with(".ai/repo/providers/") => {
1090            p.ends_with(".tmpl")
1091        }
1092        _ if p.starts_with("site/templates/") => true,
1093        _ => p.ends_with(".md"),
1094    };
1095    listed
1096        .into_iter()
1097        .filter(|p| prose(p) && !generated.contains(p))
1098        .collect()
1099}

1101/// A file's text when it is hand-written and small enough to be prose; `None` for one that
1102/// is missing, binary, oversized or stamped by its generator on its first line.
1103fn authored_text(root: &Path, path: &str) -> Option<String> {
1104    const STAMPS: &[&str] = &[
1105        "<!-- generated by",
1106        "<!-- GENERATED FILE",
1107        "# GENERATED FILE",
1108        "// GENERATED FILE",
1109    ];
1110    let full = root.join(path);
1111    let small =
1112        std::fs::symlink_metadata(&full).is_ok_and(|m| m.is_file() && m.len() <= LARGEST_READ);
1113    let text = std::fs::read_to_string(&full).ok().filter(|_| small)?;
1114    let first = text
1115        .lines()
1116        .next()
1117        .unwrap_or("")
1118        .trim_start_matches('\u{feff}');
1119    (!STAMPS.iter().any(|s| first.trim_start().starts_with(s))).then_some(text)
1120}

1122/// The entries of [`LABEL_HISTORY`] as `(line, path, reason)`; an absent file declares none.
1123fn label_history(root: &Path) -> Vec<(usize, String, String)> {
1124    let text = std::fs::read_to_string(root.join(LABEL_HISTORY)).unwrap_or_default();

1127        .filter(|(_, l)| !l.trim().is_empty() && !l.trim_start().starts_with('#'))

1136/// Every version label typed into hand-written prose outside a declared history, and every
1137/// history entry that cannot be honoured.
1138///
1139/// The label is the finding, not the version: `v0.1` on a page that describes the tool as
1140/// it stands was true the day it was typed and is wrong the day after the next release,
1141/// while nothing that derives the version can reach it. A document that is *about* a past
1142/// version names it in [`LABEL_HISTORY`] with a reason. That list is held from both sides:
1143/// an entry with no reason, one naming a path that is not tracked prose, and one whose
1144/// document no longer carries a label are each refused, so the list cannot outlive what it
1145/// exempts.
1146///
1147/// ```
1148/// use majordomus_cli::release::version::{label_diagnostics, LABEL_HISTORY};
1149/// use std::process::Command;
1150/// let dir = tempfile::tempdir().unwrap();
1151/// let root = dir.path();
1152/// std::fs::create_dir_all(root.join("docs")).unwrap();
1153/// std::fs::create_dir_all(root.join(".ai/repo")).unwrap();
1154/// std::fs::write(root.join("README.md"), "What v0.1 does not do.\nRelease v0.3.1 shipped.\n").unwrap();
1155/// std::fs::write(root.join("docs/DESIGN.md"), "The v0.1 specification.\n").unwrap();
1156/// std::fs::write(root.join(LABEL_HISTORY), "docs/DESIGN.md the design phase, dated\n").unwrap();
1157/// let git = |a: &[&str]| {
1158///     Command::new("git")
1159///         .env_remove("GIT_DIR").env_remove("GIT_WORK_TREE").env_remove("GIT_INDEX_FILE")
1160///         .arg("-C").arg(root).args(a).output().unwrap()
1161/// };
1162/// git(&["init", "-q"]);
1163/// git(&["add", "-A"]);
1164/// let found = label_diagnostics(root);
1165/// assert_eq!(found.len(), 1, "{found:?}");
1166/// assert!(found[0].message.starts_with("README.md:1 "), "{}", found[0].message);
1167///
1168/// std::fs::write(root.join("docs/DESIGN.md"), "The founding specification.\n").unwrap();
1169/// let found = label_diagnostics(root);
1170/// assert_eq!(found[1].id, "version-label-history-invalid", "an exemption outliving its label");
1171/// ```
1172pub fn label_diagnostics(root: &Path) -> Vec<Diagnostic> {
1173    let error = |id: &str, message: String| Diagnostic {
1174        id: id.into(),
1175        severity: Severity::Error,
1176        message,
1177    };
1178    let files = prose_files(root);
1179    let history = label_history(root);
1180    let exempt: BTreeSet<&str> = history
1181        .iter()
1182        .filter(|(_, _, reason)| !reason.is_empty())
1183        .map(|(_, path, _)| path.as_str())
1184        .collect();
1185    let mut out = Vec::new();
1186    let mut labelled = BTreeSet::new();
1187    for path in &files {
1188        let Some(text) = authored_text(root, path) else {
1189            continue;
1190        };
1191        for (i, line) in text.lines().enumerate() {
1192            let labels = version_labels(line);
1193            if labels.is_empty() {
1194                continue;
1195            }
1196            labelled.insert(path.as_str());
1197            if exempt.contains(path.as_str()) {
1198                continue;
1199            }
1200            for label in labels {
1201                out.push(error(
1202                    LABEL_IN_PROSE,
1203                    format!(
1204                        "{path}:{} names the version it describes as `{label}`: the label \
1205                         stays while the version moves. Say what is true of the tool as it \
1206                         stands, without a version; a document about a past version is \
1207                         declared in {LABEL_HISTORY} with its reason",
1208                        i + 1
1209                    ),
1210                ));
1211            }
1212        }
1213    }
1214    for (line, path, reason) in &history {
1215        let at = format!("{LABEL_HISTORY}:{line}");
1216        if reason.is_empty() {
1217            out.push(error(
1218                LABEL_HISTORY_INVALID,
1219                format!("{at} exempts {path} and says nothing about why: an exemption states its reason"),
1220            ));
1221        } else if !files.iter().any(|f| f == path) {
1222            out.push(error(
1223                LABEL_HISTORY_INVALID,
1224                format!("{at} exempts {path}, which is not tracked hand-written prose: remove the entry"),
1225            ));
1226        } else if !labelled.contains(path.as_str()) {
1227            out.push(error(
1228                LABEL_HISTORY_INVALID,
1229                format!(
1230                    "{at} exempts {path}, which names no version label any more: remove the entry"
1231                ),
1232            ));
1233        }
1234    }
1235    out
1236}

1238/// Everything wrong with where the version is stated: the projection against the
1239/// authority, and every version written down by hand where the tool's own files live.
1240///
1241/// A stale projection is a [`Severity::Warning`] — it is the state every bump leaves until
1242/// `scripts/derive` runs, `generate --check` is what refuses it, and a writer that refused
1243/// its own un-derived state could not correct a bump it had just made. A projection nothing
1244/// could have derived, and a version written by hand, are [`Severity::Error`]s. Asked only
1245/// where the product lives: a repository without the crate has its own `scripts/` and its
1246/// own versions, and they are none of this module's business.
1247///
1248/// ```
1249/// use majordomus_cli::release::compat::Severity;
1250/// use majordomus_cli::release::version::{diagnose, MANIFEST, PROJECTION};
1251/// let dir = tempfile::tempdir().unwrap();
1252/// let root = dir.path();
1253/// std::fs::create_dir_all(root.join("apps/majordomus-cli")).unwrap();
1254/// std::fs::create_dir_all(root.join("share")).unwrap();
1255/// std::fs::create_dir_all(root.join("lib")).unwrap();
1256/// std::fs::write(root.join(MANIFEST), "[package]\nversion = \"0.9.0\"\n").unwrap();
1257/// std::fs::write(root.join(PROJECTION), "version=0.9.0\n").unwrap();
1258/// assert!(diagnose(root).is_empty(), "authored once, projected once");
1259///
1260/// std::fs::write(root.join("lib/x.sh"), "FOO_VERSION=\"1.2.3\"\n").unwrap();
1261/// let found = diagnose(root);
1262/// assert_eq!(found.len(), 1);
1263/// assert_eq!(found[0].id, "version-stated-by-hand");
1264/// assert_eq!(found[0].severity, Severity::Error);
1265/// ```
1266pub fn diagnose(root: &Path) -> Vec<Diagnostic> {
1267    let mut out = Vec::new();
1268    match projection_state(root) {
1269        ProjectionState::NotHere | ProjectionState::Current => {}
1270        ProjectionState::Stale {
1271            projected,
1272            declared,
1273        } => out.push(Diagnostic {
1274            id: PROJECTION_STALE.into(),
1275            severity: Severity::Warning,
1276            message: format!(
1277                "{PROJECTION} states '{}' and {MANIFEST} declares '{declared}': the projection \
1278                 is stale — run scripts/derive, which builds and then generates it; never edit \
1279                 it or bin/majordomus",
1280                projected.as_deref().unwrap_or("nothing")
1281            ),
1282        }),
1283        ProjectionState::Disagrees {
1284            projected,
1285            declared,
1286        } => out.push(Diagnostic {
1287            id: WRITERS_DISAGREE.into(),
1288            severity: Severity::Error,
1289            message: format!(
1290                "{PROJECTION} states '{projected}' and {MANIFEST} declares '{}': no derivation \
1291                 writes a version the manifest does not declare, so it was written by hand — \
1292                 run scripts/derive, and raise the version only with `majordomus release bump`",
1293                declared.as_deref().unwrap_or("nothing")
1294            ),
1295        }),
1296    }
1297    if root.join(MANIFEST).is_file() {
1298        for c in carriers(root) {
1299            out.push(Diagnostic {
1300                id: STATED_BY_HAND.into(),
1301                severity: Severity::Error,
1302                message: format!(
1303                    "{}:{} states a version by hand, as {}: `{}`. The version is authored \
1304                     only in {MANIFEST}; read it with `scripts/release-version` in shell, \
1305                     `release::version::declared` in Rust, or `bin/majordomus version`",
1306                    c.path, c.line, c.shape, c.text
1307                ),
1308            });
1309        }
1310        out.extend(label_diagnostics(root));
1311    }
1312    out
1313}

1315/// Write `to` into the one place the version is authored, keep the lock's record of it in
1316/// step, and say which files changed.
1317///
1318/// Byte-exact and narrow: the manifest's `version` line inside `[package]`, and the `version`
1319/// line of the lock's own `majordomus-cli` entry — the one line cargo would rewrite. Nothing
1320/// else in either file is read or rewritten, so a dependency at the same version is
1321/// untouched. [`PROJECTION`], the generator stamps and the changelog are not written here:
1322/// they are derived, and `scripts/derive` derives them.
1323///
1324/// ```
1325/// use majordomus_cli::release::version::{declared, write, MANIFEST};
1326/// let dir = tempfile::tempdir().unwrap();
1327/// std::fs::create_dir_all(dir.path().join("apps/majordomus-cli")).unwrap();
1328/// std::fs::write(dir.path().join(MANIFEST), "[package]\nversion = \"0.8.0\"\n").unwrap();
1329/// assert_eq!(write(dir.path(), "0.9.0").unwrap(), vec![MANIFEST.to_string()]);
1330/// assert_eq!(declared(dir.path()).as_deref(), Some("0.9.0"));
1331/// ```
1332pub fn write(root: &Path, to: &str) -> std::io::Result<Vec<String>> {
1333    let mut written = Vec::new();
1334
1335    let manifest = root.join(MANIFEST);
1336    let text = std::fs::read_to_string(&manifest)?;
1337    let mut out = String::with_capacity(text.len());
1338    let mut in_package = false;
1339    let mut done = false;
1340    for line in text.lines() {
1341        let trimmed = line.trim();
1342        if trimmed.starts_with('[') {
1343            in_package = trimmed == "[package]";
1344        }
1345        if in_package && !done && trimmed.starts_with("version") && trimmed.contains('"') {
1346            out.push_str(&format!("version = \"{to}\"\n"));
1347            done = true;
1348            continue;
1349        }
1350        out.push_str(line);
1351        out.push('\n');
1352    }
1353    if out != text {
1354        std::fs::write(&manifest, out)?;
1355        written.push(MANIFEST.to_string());
1356    }
1357
1358    // The lock's own entry, and only it: the `version` line that follows
1359    // `name = "majordomus-cli"`. Every other `version` in the file belongs to a dependency.
1360    let lock = root.join(LOCK);
1361    if let Ok(text) = std::fs::read_to_string(&lock) {
1362        let mut out = String::with_capacity(text.len());
1363        let mut here = false;
1364        let mut done = false;
1365        for line in text.lines() {
1366            let trimmed = line.trim();
1367            if trimmed == "[[package]]" {
1368                here = false;
1369            } else if trimmed == "name = \"majordomus-cli\"" {
1370                here = true;
1371            } else if here && !done && trimmed.starts_with("version") && trimmed.contains('"') {
1372                out.push_str(&format!("version = \"{to}\"\n"));
1373                here = false;
1374                done = true;
1375                continue;
1376            }
1377            out.push_str(line);
1378            out.push('\n');
1379        }
1380        if out != text {
1381            std::fs::write(&lock, out)?;
1382            written.push(LOCK.to_string());
1383        }
1384    }
1385
1386    Ok(written)
1387}

1389/// The version the commit subjects imply: the last release raised by their bump — from the
1390/// declared version only when nothing was released. Evidence, never the answer on its own.
1391///
1392/// It raises the last release and not the declared version, because a declared version
1393/// already raised for the window must not be raised again by the same commits.
1394///
1395/// ```
1396/// use majordomus_cli::release::version::{commits_imply, Bump};
1397/// // released 0.10.0, already declared 0.11.0, only features since: 0.11.0, not 0.12.0
1398/// assert_eq!(commits_imply(Some("0.10.0"), "0.11.0", Bump::Minor).as_deref(), Some("0.11.0"));
1399/// // nothing released: the declared version is the only base there is
1400/// assert_eq!(commits_imply(None, "0.3.0", Bump::Patch).as_deref(), Some("0.3.1"));
1401/// ```
1402pub fn commits_imply(last: Option<&str>, declared: &str, bump: Bump) -> Option<String> {
1403    last.and_then(Version::parse)
1404        .or_else(|| Version::parse(declared))
1405        .map(|v| v.raised(bump).to_string())
1406}

1408/// Who answers `next`, and with what: the contract when it could be measured, and no one
1409/// when it could not.
1410///
1411/// `contract` is the analysis' measured version ([`crate::release::compat::VersionPlan::
1412/// measured_version`]), or why none could be measured. ADR 0051 refuses an unmeasurable
1413/// baseline rather than guessing one, so an `Err` leaves `next` absent, [`DecidedBy::
1414/// Undecided`] and the reason carried — the commit inference stays evidence beside it and
1415/// never stands in for the measurement. The contract's answer is a minimum: when it is no
1416/// greater than `last` — only behaviour behind the public boundary changed, so the contract
1417/// requires no release — it is the release already made, never the next one. Then the
1418/// commits decide whether anything is released at all, and the contract decides how much:
1419/// the smallest release above the last, a patch, labelled [`DecidedBy::ContractAndCommits`].
1420/// The commits never raise `next` beyond what the contract requires.
1421///
1422/// ```
1423/// use majordomus_cli::release::model::DecidedBy;
1424/// use majordomus_cli::release::version::{decide, Bump};
1425/// // the commits say BREAKING, the contract only grew: below 1.0 that owes a minor
1426/// let (next, by, why) = decide(Some("0.10.0"), Bump::Major, Ok("0.11.0".into()));
1427/// assert_eq!((next.as_deref(), by, why), (Some("0.11.0"), DecidedBy::Contract, None));
1428/// // only fixes, the surface unchanged: the contract's minimum is the release already made
1429/// let (next, by, _) = decide(Some("0.10.0"), Bump::Patch, Ok("0.10.0".into()));
1430/// assert_eq!((next.as_deref(), by), (Some("0.10.1"), DecidedBy::ContractAndCommits));
1431/// // no published baseline: no one decides, and the reason is carried
1432/// let (next, by, why) = decide(None, Bump::Minor, Err("no release".into()));
1433/// assert_eq!((next, by), (None, DecidedBy::Undecided));
1434/// assert_eq!(why.as_deref(), Some("no release"));
1435/// ```
1436pub fn decide(
1437    last: Option<&str>,
1438    bump: Bump,
1439    contract: Result<String, String>,
1440) -> (Option<String>, DecidedBy, Option<String>) {
1441    let required = match contract {
1442        Ok(required) => required,
1443        Err(why) => return (None, DecidedBy::Undecided, Some(why)),
1444    };
1445    match (Version::parse(&required), last.and_then(Version::parse)) {
1446        (Some(r), Some(l)) if r <= l => match bump {
1447            Bump::None => (None, DecidedBy::Contract, None),
1448            _ => (
1449                Some(l.raised(Bump::Patch).to_string()),
1450                DecidedBy::ContractAndCommits,
1451                None,
1452            ),
1453        },
1454        _ => (Some(required), DecidedBy::Contract, None),
1455    }
1456}

1458/// The whole version answer: what is declared, whether the projection the shell tool reads
1459/// states it, what the public contract requires next, and what the commits since the last
1460/// release imply — as evidence.
1461///
1462/// `contract` is the analysis' answer ([`decide`] says how it is used); [`select`] is the
1463/// caller that asks it, and the one every surface goes through. The last release is
1464/// [`crate::release::compat::last_release`] — the same baseline `release analyze` measures
1465/// from, the highest version the layer records — so the contract and the commit evidence
1466/// always measure one window. The commit inference raises that release ([`commits_imply`]).
1467pub fn report(root: &Path, objects: &[Object], contract: Result<String, String>) -> VersionReport {
1468    let declared = declared(root).unwrap_or_else(|| "unknown".into());
1469    let tool_version = tool(root).unwrap_or_else(|| "unknown".into());
1470
1471    let last = super::compat::last_release(root, objects);
1472    let changes = match &last {
1473        Some(l) => super::commits::in_range(root, &format!("{}..HEAD", l.read_at), objects),
1474        None => super::commits::in_range(root, "HEAD", objects),
1475    };
1476    let bump = bump_of(&changes);
1477    let last_version = last.map(|l| l.version);
1478    let commits_imply = commits_imply(last_version.as_deref(), &declared, bump);
1479    let (next, decided_by, contract_unreadable) = decide(last_version.as_deref(), bump, contract);
1480
1481    VersionReport {
1482        agree: projection_current(root),
1483        declared,
1484        tool: tool_version,
1485        last_release: last_version,
1486        bump: bump.as_str().to_string(),
1487        commits_imply,
1488        next,
1489        decided_by,
1490        contract_unreadable,
1491        changes,
1492    }
1493}

1495/// The release selection: the contract's analysis against the last release, and the version
1496/// report decided from it — made once, by one function.
1497///
1498/// `release version` (the capability, and every surface that renders it) and `release bump`
1499/// without an explicit target both take their answer from here, so the report and the writer
1500/// cannot disagree about what the next version is: the report's `next` is the version the
1501/// writer raises to ([`default_target`]).
1502///
1503/// ```
1504/// use majordomus_cli::capability::registry::CapabilityRegistry;
1505/// use majordomus_cli::release::model::DecidedBy;
1506/// use majordomus_cli::release::version::{select, Selection};
1507/// let dir = tempfile::tempdir().unwrap();
1508/// let registry = CapabilityRegistry::builder().build().unwrap();
1509/// let selection: Selection = select(dir.path(), &registry, &[]);
1510/// // nothing published: no plan, and so a report that decides nothing — with the same reason
1511/// let why = selection.plan.as_ref().unwrap_err();
1512/// assert_eq!(selection.report.decided_by, DecidedBy::Undecided);
1513/// assert_eq!(selection.report.contract_unreadable.as_ref(), Some(why));
1514/// assert_eq!(selection.report.next, None);
1515/// ```
1516#[derive(Debug)]
1517pub struct Selection {
1518    /// The analysis `release analyze` makes against the last release, or why it could not be
1519    /// made at all (no release to compare with, a baseline whose registry is not committed).
1520    pub plan: Result<super::compat::VersionPlan, String>,
1521    /// The version report decided from that analysis.
1522    pub report: VersionReport,
1523}

1525/// Make the release selection ([`Selection`]): analyse the contract against the last
1526/// release, refuse a guess ([`crate::release::compat::VersionPlan::measured_version`]), and
1527/// decide the report from what is left ([`report`]).
1528///
1529/// ```
1530/// use majordomus_cli::capability::registry::CapabilityRegistry;
1531/// use majordomus_cli::release::version::{default_target, select, Version};
1532/// let dir = tempfile::tempdir().unwrap();
1533/// let registry = CapabilityRegistry::builder().build().unwrap();
1534/// let selection = select(dir.path(), &registry, &[]);
1535/// // the writer's default target is read from this report, so it refuses with its reason
1536/// let current = Version::parse("0.1.0").unwrap();
1537/// assert_eq!(
1538///     default_target(&selection.report, &current).unwrap_err(),
1539///     selection.report.contract_unreadable.clone().unwrap()
1540/// );
1541/// ```
1542pub fn select(
1543    root: &Path,
1544    registry: &crate::capability::registry::CapabilityRegistry,
1545    objects: &[Object],
1546) -> Selection {
1547    let plan = super::compat::analyze(root, registry, objects, None).map_err(|e| e.to_string());
1548    let contract = match &plan {
1549        Ok(p) => p.measured_version(),
1550        Err(why) => Err(why.clone()),
1551    };
1552    Selection {
1553        report: report(root, objects, contract),
1554        plan,
1555    }
1556}

1558/// The version `release bump` raises to when no target is named: the report's `next`, the
1559/// declared version when nothing would be released — and a refusal, with the report's
1560/// reason, when the contract could not be measured and so no one decided.
1561///
1562/// ```
1563/// use majordomus_cli::release::model::{DecidedBy, VersionReport};
1564/// use majordomus_cli::release::version::{default_target, Version};
1565/// let report = |next: Option<&str>, decided_by, why: Option<&str>| VersionReport {
1566///     declared: "0.11.0".into(), tool: "0.11.0".into(), agree: true,
1567///     last_release: Some("0.11.0".into()), bump: "patch".into(),
1568///     commits_imply: Some("0.11.1".into()), next: next.map(Into::into), decided_by,
1569///     contract_unreadable: why.map(Into::into), changes: Vec::new() };
1570/// let current = Version::parse("0.11.0").unwrap();
1571/// let fix = report(Some("0.11.1"), DecidedBy::ContractAndCommits, None);
1572/// assert_eq!(default_target(&fix, &current).unwrap().to_string(), "0.11.1");
1573/// let nothing = report(None, DecidedBy::Contract, None);
1574/// assert_eq!(default_target(&nothing, &current), Ok(current.clone()));
1575/// let unread = report(None, DecidedBy::Undecided, Some("no release"));
1576/// assert_eq!(default_target(&unread, &current), Err("no release".into()));
1577/// ```
1578pub fn default_target(report: &VersionReport, current: &Version) -> Result<Version, String> {
1579    match (&report.next, report.decided_by) {
1580        (_, DecidedBy::Undecided) => Err(report
1581            .contract_unreadable
1582            .clone()
1583            .unwrap_or_else(|| "the public contract could not be measured".into())),
1584        (Some(next), _) => Version::parse(next)
1585            .ok_or_else(|| format!("the selected version '{next}' is not three numbers")),
1586        (None, _) => Ok(current.clone()),
1587    }
1588}
