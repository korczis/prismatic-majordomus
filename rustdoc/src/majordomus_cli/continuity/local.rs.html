1//! The machine's side: what a publication reads from this checkout, what a resume writes
2//! into it, and the one small file of checkout-local state the continuity commands keep.
3//!
4//! Reading: the newest handover record, the task it was written under, the decisions that
5//! task recorded, the open episode, and the source state from git. Every one of those is
6//! written by the shell tool; this module reads them and converts them into the portable
7//! [`Record`](super::record::Record), dropping every value that names this machine.
8//!
9//! Writing, on resume, is the two places where a continued handover has to land for the
10//! rest of the lifecycle to see it without knowing it came from elsewhere: the handovers
11//! directory, through the same writer the mesh uses, and the decision log, appended in its
12//! own format. Then `continuity.json`, which records which record this checkout continues
13//! on each branch — the parent of its next publication — and what the last sync saw, so
14//! that the entry banner can say a resumable handover is waiting without asking a remote.
15

52/// The record this checkout continues on one branch.

54pub struct Position {

100/// The contents of [`LOCAL_FILE`].
101#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
102pub struct LocalState {

104    pub schema: String,

127/// The path of [`LOCAL_FILE`] in `root`.
128pub fn local_path(root: &Path) -> PathBuf {
129    root.join(STATE_DIR).join(LOCAL_FILE)
130}

132/// Read [`LOCAL_FILE`]. Absent is the default; unreadable is an error, never silently
133/// replaced, because it holds the parent of the next publication.
134pub fn load(root: &Path) -> Result<LocalState, String> {
135    let path = local_path(root);
136    match std::fs::read_to_string(&path) {
137        Ok(text) => {
138            let state: LocalState = serde_json::from_str(&text)
139                .map_err(|e| format!("{}: {e}", relative(root, &path)))?;
140            if state.schema != LOCAL_SCHEMA {
141                return Err(format!(
142                    "{}: schema {} is not {LOCAL_SCHEMA}",
143                    relative(root, &path),
144                    state.schema
145                ));
146            }
147            Ok(state)
148        }
149        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(LocalState::default()),
150        Err(e) => Err(format!("{}: {e}", relative(root, &path))),
151    }
152}

154/// Write [`LOCAL_FILE`] atomically.
155pub fn save(root: &Path, state: &LocalState) -> Result<(), String> {
156    let path = local_path(root);
157    let dir = root.join(STATE_DIR);
158    std::fs::create_dir_all(&dir).map_err(|e| format!("{}: {e}", relative(root, &dir)))?;
159    let mut text = serde_json::to_string_pretty(state).expect("a local state is plain data");
160    text.push('\n');
161    let tmp = dir.join(format!(".{LOCAL_FILE}.tmp"));
162    std::fs::write(&tmp, text)
163        .and_then(|()| std::fs::rename(&tmp, &path))
164        .map_err(|e| format!("{}: {e}", relative(root, &path)))
165}

167/// `path` relative to `root`, for a message: a message names the repository's paths, not
168/// the machine's.
169pub fn relative(root: &Path, path: &Path) -> String {
170    path.strip_prefix(root)
171        .map(|p| p.display().to_string())
172        .unwrap_or_else(|_| {
173            path.file_name()
174                .map(|n| n.to_string_lossy().to_string())
175                .unwrap_or_default()
176        })
177}

179fn git_out(root: &Path, args: &[&str]) -> Option<Vec<u8>> {
180    let out = Command::new("git")
181        .env("GIT_OPTIONAL_LOCKS", "0")
182        .env_remove("GIT_DIR")
183        .env_remove("GIT_WORK_TREE")
184        .arg("-C")
185        .arg(root)
186        .args(args)
187        .stdin(Stdio::null())
188        .output();
189    out.ok().filter(|o| o.status.success()).map(|o| o.stdout)
190}

192/// The branch and HEAD of `root`.
193pub fn branch_and_head(root: &Path) -> (Option<String>, Option<String>) {
194    let text = |args: &[&str]| {
195        git_out(root, args)
196            .map(|b| String::from_utf8_lossy(&b).trim().to_string())
197            .filter(|s| !s.is_empty())
198    };
199    (
200        text(&["symbolic-ref", "--quiet", "--short", "HEAD"]),
201        text(&["rev-parse", "--verify", "--quiet", "HEAD"]),
202    )
203}

205/// The source state of `root` now: branch, HEAD, and — when the tree is dirty — the paths
206/// that differ and a fingerprint of the difference. The fingerprint is a digest, never the
207/// content: whether two machines hold the same uncommitted work can be compared, and the
208/// work itself is not published.
209pub fn source_state(root: &Path) -> SourceState {
210    let (branch, head) = branch_and_head(root);
211    let status = git_out(
212        root,
213        &["status", "--porcelain=v1", "-z", "--untracked-files=all"],
214    )
215    .unwrap_or_default();
216    let (mut changed, mut untracked) = changed_paths(&status);
217    crate::order::canonical_strings(&mut changed);
218    changed.dedup();
219    let total = changed.len();
220    let working_tree = if total == 0 {
221        WorkingTree::Clean
222    } else {
223        WorkingTree::Dirty
224    };
225    let fingerprint = (total > 0).then(|| {
226        let mut hasher = Sha256::new();
227        if head.is_some() {
228            hasher.update(git_out(root, &["diff", "--binary", "HEAD"]).unwrap_or_default());
229        }
230        crate::order::canonical_strings(&mut untracked);
231        for path in &untracked {
232            hasher.update(b"\0untracked\0");
233            hasher.update(path.as_bytes());
234            hasher.update(std::fs::read(root.join(path)).unwrap_or_default());
235        }
236        hasher
237            .finalize()
238            .iter()
239            .take(16)
240            .map(|b| format!("{b:02x}"))
241            .collect()
242    });
243    changed.truncate(MAX_CHANGED);
244    SourceState {
245        branch,
246        head,
247        working_tree,
248        changed,
249        changed_total: total,
250        fingerprint,
251    }
252}

254/// The paths `git status --porcelain=v1 -z` names, and those of them that are untracked. An
255/// entry is `XY <path>`; a rename or a copy is followed by the path it came from, which is
256/// not a second change. Anything shorter than an entry can be is passed over.
257fn changed_paths(status: &[u8]) -> (Vec<String>, Vec<String>) {
258    let mut changed = Vec::new();
259    let mut untracked = Vec::new();
260    let mut entries = status.split(|b| *b == 0).filter(|e| !e.is_empty());
261    while let Some(entry) = entries.next() {
262        if entry.len() < 4 {
263            continue;
264        }
265        let code = &entry[..2];
266        let path = String::from_utf8_lossy(&entry[3..]).to_string();
267        if code[0] == b'R' || code[0] == b'C' {
268            entries.next();
269        }
270        if code == b"??" {
271            untracked.push(path.clone());
272        }
273        changed.push(path);
274    }
275    (changed, untracked)
276}

278/// The episode open in this checkout, when its record is this checkout's.
279pub fn open_episode(root: &Path) -> Option<String> {
280    let f = document(&root.join(STATE_DIR).join("session-current.yaml"))?;
281    let worktree = f.get("worktree").cloned().unwrap_or_default();
282    if !worktree.is_empty() && Path::new(&worktree) != root {
283        let same = std::fs::canonicalize(&worktree).ok() == std::fs::canonicalize(root).ok();
284        if !same {
285            return None;
286        }
287    }
288    f.get("session_id").cloned().filter(|s| !s.is_empty())
289}

299/// The task `id` names: the active record when it is that task, else its archived record.
300pub fn task(root: &Path, id: &str) -> Option<TaskRef> {
301    let dir = root.join(STATE_DIR);
302    let found = read_task(&dir.join("current.yaml"))
303        .filter(|t| t.id == id)
304        .or_else(|| {
305            // the archive is named by task id; an id is never a path
306            let safe = id
307                .chars()
308                .all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '_');
309            safe.then(|| read_task(&dir.join("archive").join(format!("{id}.yaml"))))
310                .flatten()
311        })?;
312    Some(TaskRef {
313        id: found.id,
314        title: found.task,
315        profile: Some(found.profile).filter(|p| !p.is_empty() && p != "none"),
316        scope: found.scope,
317        outcome: Some(found.outcome).filter(|o| !o.is_empty()),
318    })
319}

321/// The decisions the log holds for task `id`, oldest first, as written.
322pub fn decisions(root: &Path, id: &str) -> Vec<CarriedDecision> {
323    let Ok(text) = std::fs::read_to_string(root.join(STATE_DIR).join("decisions.md")) else {
324        return Vec::new();
325    };
326    parse_decisions(&text)
327        .into_iter()
328        .filter(|d| {
329            d.text
330                .lines()
331                .any(|l| l.strip_prefix("Task: ").map(str::trim) == Some(id))
332        })
333        .take(MAX_DECISIONS)
334        .collect()
335}

337/// Every entry of a decision log: `## <heading>` and the lines below it. The template's
338/// commented example is not an entry.
339pub fn parse_decisions(text: &str) -> Vec<CarriedDecision> {
340    let mut out: Vec<CarriedDecision> = Vec::new();
341    let mut comment = false;
342    for line in text.lines() {
343        if line.contains("<!--") {
344            comment = true;
345        }
346        if comment {
347            if line.contains("-->") {
348                comment = false;
349            }
350            continue;
351        }
352        if let Some(title) = line.strip_prefix("## ") {
353            out.push(CarriedDecision {
354                title: title.trim().to_string(),
355                text: String::new(),
356            });
357        } else if let Some(d) = out.last_mut() {
358            if d.text.len() + line.len() < MAX_DECISION_BYTES {
359                d.text.push_str(line);
360                d.text.push('\n');
361            }
362        }
363    }
364    for d in &mut out {
365        d.text = d.text.trim_end().to_string();
366        d.text.push('\n');
367    }
368    out
369}

371/// Append carried decisions to this checkout's log, each once: an entry whose heading the
372/// log already holds is not written again. Each appended entry names where it came from.
373/// Returns how many were appended.
374pub fn carry_decisions(
375    root: &Path,
376    carried: &[CarriedDecision],
377    record: &str,
378    device: &str,
379) -> Result<usize, String> {
380    if carried.is_empty() {
381        return Ok(0);
382    }
383    let path = root.join(STATE_DIR).join("decisions.md");
384    let mut text = match std::fs::read_to_string(&path) {
385        Ok(t) => t,
386        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
387            "# Decisions\n\nAppend-only. One dated entry per decision. Newest at the bottom.\n"
388                .to_string()
389        }
390        Err(e) => return Err(format!("{}: {e}", relative(root, &path))),
391    };
392    let held: std::collections::BTreeSet<String> = parse_decisions(&text)
393        .into_iter()
394        .map(|d| d.title)
395        .collect();
396    let mut appended = 0usize;
397    for d in carried {
398        let title: String = d.title.chars().filter(|c| !c.is_control()).collect();
399        if held.contains(&title) {
400            continue;
401        }
402        // a carried entry cannot open a second heading inside itself
403        let body: String = d
404            .text
405            .lines()
406            .filter(|l| !l.starts_with("## ") && !l.contains("<!--"))
407            .map(|l| format!("{l}\n"))
408            .collect();
409        let device: String = device.chars().filter(|c| !c.is_control()).collect();
410        text.push_str(&format!(
411            "\n## {title}\n{body}Carried: continuity {} from {device}\n",
412            &record[..16.min(record.len())]
413        ));
414        appended += 1;
415    }
416    if appended > 0 {
417        let dir = root.join(STATE_DIR);
418        std::fs::create_dir_all(&dir).map_err(|e| format!("{}: {e}", relative(root, &dir)))?;
419        let tmp = path.with_extension("md.tmp");
420        std::fs::write(&tmp, text)
421            .and_then(|()| std::fs::rename(&tmp, &path))
422            .map_err(|e| format!("{}: {e}", relative(root, &path)))?;
423    }
424    Ok(appended)
425}

427/// The values of this process's secret environment variables: every variable whose name
428/// says it holds a credential, with a value long enough to be one. A publication refuses a
429/// record carrying any of them, whatever shape they have.
430pub fn secret_environment() -> Vec<String> {
431    const MARKS: &[&str] = &[
432        "TOKEN",
433        "SECRET",
434        "PASSWORD",
435        "PASSWD",
436        "API_KEY",
437        "APIKEY",
438        "PRIVATE_KEY",
439        "CREDENTIAL",
440        "ACCESS_KEY",
441        "AUTH",
442    ];
443    std::env::vars()
444        .filter(|(k, v)| {
445            let k = k.to_ascii_uppercase();
446            v.len() >= 8 && MARKS.iter().any(|m| k.contains(m))
447        })
448        .map(|(_, v)| v)
449        .collect()
450}
