1//! Which commit an executable was built from, and whether that tree was dirty: the decision
2//! `build.rs` makes once, at build time, and compiles in as `MAJORDOMUS_COMMIT` and
3//! `MAJORDOMUS_DIRTY` (read back as [`crate::COMMIT`] and [`crate::DIRTY`]).
4//!
5//! It lives here rather than inline in the build script so that it can be tested: a build
6//! script has no test harness, and the constants it emits can only be compared with
7//! themselves. `build.rs` compiles this file through `#[path]`, the way it compiles
8//! `generation.rs`, so there is one definition of the decision and the tests below exercise
9//! the one the build runs. For the same reason nothing outside `std` may be used here, and
10//! every item must be reached by the build script or it is dead code in that compilation.
11//!
12//! The dirty flag describes the crate's own directory (the build script asks git with the
13//! pathspec `.` from `CARGO_MANIFEST_DIR`), as it stood when the build script last ran: an
14//! edit to documentation elsewhere in the repository does not mark the executable dirty, and
15//! a commit that touches no crate source does not rerun the build script, so the flag says
16//! what the build saw and not what the tree says now.
17//!
18//! ```
19//! use majordomus_cli::build_identity::{identify, HEAD_ARGS, STATUS_ARGS};
20//!
21//! // a plain `cargo build` in a checkout: git names the commit and reports a clean crate
22//! let sha = "0123456789abcdef0123456789abcdef01234567";
23//! let git = |args: &[&str]| match args {
24//!     a if a == HEAD_ARGS => Some(format!("{sha}\n")),
25//!     a if a == STATUS_ARGS => Some(String::new()),
26//!     _ => None,
27//! };
28//! assert_eq!(identify(None, None, git), (sha.to_string(), "false"));
29//!
30//! // a release build hands the commit and the flag in, and git is never asked
31//! let never = |args: &[&str]| -> Option<String> { panic!("git asked {args:?}") };
32//! assert_eq!(identify(Some(sha), Some("0"), never), (sha.to_string(), "false"));
33//!
34//! // a source tarball has neither git nor a declaration: both halves are `unknown`
35//! assert_eq!(identify(None, None, |_| None), ("unknown".to_string(), "unknown"));
36//! ```
37
38/// The arguments the build script hands git to learn the commit.
39pub const HEAD_ARGS: &[&str] = &["rev-parse", "HEAD"];
40/// The arguments the build script hands git to learn whether the crate's tree is dirty:
41/// tracked changes only, under the build script's working directory (the crate), and without
42/// the optional index refresh — a build is a read and must not rewrite the index of a
43/// repository another session may be staging into.
44pub const STATUS_ARGS: &[&str] = &[
45    "--no-optional-locks",
46    "status",
47    "--porcelain",
48    "--untracked-files=no",
49    "--",
50    ".",
51];
52
53/// The commit the build was handed: `MAJORDOMUS_BUILD_COMMIT` when it names one (neither
54/// empty nor `unknown`), else a hosting provider's `SOURCE_VERSION` when that is a full 40-hex
55/// commit (the Heroku-compatible variable Gigalixir sets to the pushed commit, ADR 0219),
56/// else whatever `MAJORDOMUS_BUILD_COMMIT` held, which [`identify`] then judges exactly as
57/// before. A `SOURCE_VERSION` that is not a commit (a branch name, a short sha) is ignored
58/// rather than compiled in as one (I3820).
59///
60/// ```
61/// use majordomus_cli::build_identity::declared_commit;
62///
63/// let sha = "0123456789abcdef0123456789abcdef01234567";
64/// // a provider build hands only SOURCE_VERSION: it is the commit
65/// assert_eq!(declared_commit(None, Some(sha)), Some(sha.to_string()));
66/// // a branch name is not a commit and is never compiled in as one
67/// assert_eq!(declared_commit(None, Some("master")), None);
68/// ```
69pub fn declared_commit(majordomus: Option<&str>, source_version: Option<&str>) -> Option<String> {
70    let names_one = |v: &str| !v.trim().is_empty() && v.trim() != "unknown";
71    if let Some(v) = majordomus.filter(|v| names_one(v)) {
72        return Some(v.to_string());
73    }
74    let is_commit = |v: &str| v.len() == 40 && v.bytes().all(|b| b.is_ascii_hexdigit());
75    source_version
76        .map(str::trim)
77        .filter(|v| is_commit(v))
78        .map(str::to_string)
79        .or_else(|| majordomus.map(str::to_string))
80}

82/// Decide the commit and the dirty flag from what the build was handed and what git says.
83///
84/// `declared_commit` is `MAJORDOMUS_BUILD_COMMIT` and `declared_dirty` is
85/// `MAJORDOMUS_BUILD_DIRTY`; `git` runs git with the given arguments and returns its standard
86/// output on success. The commit is the declared one when it names something (neither empty
87/// nor `unknown`), else git's `HEAD`, else `unknown`. The flag is `true`/`false` when it was
88/// declared (`true`/`1`, `false`/`0`); otherwise it is read from git only when git also named
89/// the commit — a commit handed in from outside says nothing about the tree it came from —
90/// and is `unknown` when neither said.
91///
92/// ```
93/// use majordomus_cli::build_identity::{identify, STATUS_ARGS};
94///
95/// // git names the commit and finds a tracked change in the crate: the build is dirty
96/// let git = |args: &[&str]| {
97///     Some(if args == STATUS_ARGS { " M src/lib.rs\n" } else { "abc123\n" }.to_string())
98/// };
99/// assert_eq!(identify(None, None, git), ("abc123".to_string(), "true"));
100///
101/// // a declared commit says nothing about the tree, so its dirtiness stays `unknown`
102/// let never = |args: &[&str]| -> Option<String> { panic!("git asked {args:?}") };
103/// assert_eq!(identify(Some(" def456 "), None, never), ("def456".to_string(), "unknown"));
104/// ```
105pub fn identify(
106    declared_commit: Option<&str>,
107    declared_dirty: Option<&str>,
108    mut git: impl FnMut(&[&str]) -> Option<String>,
109) -> (String, &'static str) {
110    let declared = declared_commit
111        .map(str::trim)
112        .filter(|c| !c.is_empty() && *c != "unknown")
113        .map(str::to_string);
114    let from_git = declared.is_none();
115    let commit = declared
116        .or_else(|| {
117            git(HEAD_ARGS)
118                .map(|s| s.trim().to_string())
119                .filter(|s| !s.is_empty())
120        })
121        .unwrap_or_else(|| "unknown".into());
122    let dirty = match declared_dirty.map(str::trim) {
123        Some("true" | "1") => "true",
124        Some("false" | "0") => "false",
125        _ if from_git && commit != "unknown" => match git(STATUS_ARGS) {
126            Some(out) if out.trim().is_empty() => "false",
127            Some(_) => "true",
128            None => "unknown",
129        },
130        _ => "unknown",
131    };
132    (commit, dirty)
133}

135#[cfg(test)]
136mod tests {
137    use super::*;
138
139    const SHA: &str = "0123456789abcdef0123456789abcdef01234567";
140    const OTHER: &str = "fedcba9876543210fedcba9876543210fedcba98";
141
142    /// A git that answers `HEAD` with [`SHA`] and `status` with `status`, and records what
143    /// it was asked.
144    fn git(status: Option<&'static str>) -> impl FnMut(&[&str]) -> Option<String> {
145        move |args: &[&str]| match args {
146            a if a == HEAD_ARGS => Some(format!("{SHA}\n")),
147            a if a == STATUS_ARGS => status.map(str::to_string),
148            other => panic!("git asked something the build does not ask: {other:?}"),
149        }
150    }
151
152    fn no_git(_: &[&str]) -> Option<String> {
153        None
154    }
155
156    #[test]
157    fn git_names_the_commit_and_a_clean_tree() {
158        assert_eq!(identify(None, None, git(Some(""))), (SHA.into(), "false"));
159    }
160
161    #[test]
162    fn git_names_a_dirty_tree() {
163        assert_eq!(
164            identify(None, None, git(Some(" M src/lib.rs\n"))),
165            (SHA.into(), "true")
166        );
167    }
168
169    #[test]
170    fn a_status_git_cannot_answer_is_unknown() {
171        assert_eq!(identify(None, None, git(None)), (SHA.into(), "unknown"));
172    }
173
174    #[test]
175    fn a_declared_commit_without_a_dirty_flag_is_unknown_dirtiness() {
176        // git is not asked about a tree the declared commit may not be
177        let asked = |args: &[&str]| -> Option<String> { panic!("git asked {args:?}") };
178        assert_eq!(
179            identify(Some(OTHER), None, asked),
180            (OTHER.into(), "unknown")
181        );
182    }
183
184    #[test]
185    fn a_declared_dirty_flag_is_taken_as_declared() {
186        for (v, want) in [
187            ("true", "true"),
188            ("1", "true"),
189            ("false", "false"),
190            ("0", "false"),
191        ] {
192            assert_eq!(
193                identify(Some(OTHER), Some(v), no_git),
194                (OTHER.into(), want),
195                "{v}"
196            );
197            assert_eq!(
198                identify(None, Some(v), git(Some(" M x\n"))),
199                (SHA.into(), want),
200                "{v}"
201            );
202        }
203    }
204
205    #[test]
206    fn a_declared_flag_that_is_neither_falls_back() {
207        assert_eq!(
208            identify(None, Some("maybe"), git(Some(""))),
209            (SHA.into(), "false")
210        );
211        assert_eq!(
212            identify(Some(OTHER), Some("unknown"), no_git),
213            (OTHER.into(), "unknown")
214        );
215    }
216
217    #[test]
218    fn a_declared_unknown_or_empty_commit_falls_back_to_git() {
219        for declared in ["unknown", "", "  "] {
220            assert_eq!(
221                identify(Some(declared), None, git(Some(""))),
222                (SHA.into(), "false")
223            );
224        }
225    }
226
227    #[test]
228    fn a_declared_commit_is_trimmed() {
229        let padded = format!(" {OTHER}\n");
230        assert_eq!(
231            identify(Some(&padded), Some("0"), no_git),
232            (OTHER.into(), "false")
233        );
234    }
235
236    #[test]
237    fn a_provider_source_version_is_the_commit_when_nothing_else_names_one() {
238        assert_eq!(declared_commit(None, Some(SHA)), Some(SHA.into()));
239        let padded = format!(" {SHA}\n");
240        assert_eq!(declared_commit(None, Some(&padded)), Some(SHA.into()));
241        // what the build then compiles in: that commit, with its dirtiness unknown
242        let declared = declared_commit(None, Some(SHA));
243        assert_eq!(
244            identify(declared.as_deref(), None, no_git),
245            (SHA.into(), "unknown")
246        );
247    }
248
249    #[test]
250    fn a_named_majordomus_commit_wins_over_the_provider_source_version() {
251        assert_eq!(declared_commit(Some(OTHER), Some(SHA)), Some(OTHER.into()));
252        // `unknown` and empty name nothing, so the provider's commit is taken
253        assert_eq!(
254            declared_commit(Some("unknown"), Some(SHA)),
255            Some(SHA.into())
256        );
257        assert_eq!(declared_commit(Some(" "), Some(SHA)), Some(SHA.into()));
258    }
259
260    #[test]
261    fn a_source_version_that_is_not_a_full_commit_is_ignored() {
262        let not_commits = [
263            "master".to_string(),
264            "0123456".to_string(),
265            SHA[..39].to_string(),
266            format!("{SHA}0"),
267            "z".repeat(40),
268        ];
269        for v in &not_commits {
270            assert_eq!(declared_commit(None, Some(v)), None, "{v:?}");
271            // what MAJORDOMUS_BUILD_COMMIT held is handed on for identify to judge as before
272            assert_eq!(
273                declared_commit(Some("unknown"), Some(v)),
274                Some("unknown".into())
275            );
276        }
277        assert_eq!(declared_commit(None, None), None);
278    }
279
280    #[test]
281    fn no_git_and_nothing_declared_is_unknown() {
282        assert_eq!(identify(None, None, no_git), ("unknown".into(), "unknown"));
283        let empty = |_: &[&str]| Some("\n".to_string());
284        assert_eq!(identify(None, None, empty), ("unknown".into(), "unknown"));
285    }
