1//! The monitor page: everything that happens around this repository, as one live list
2//! (ADR 0133). The first page is rendered here from `monitor.stream`, the same answer every
3//! other surface renders; `monitor.js` then follows the live channel from its cursor
4//! (`/events?follow=blackboard&frames=signals`, ADR 0176) and adds what the server pushes on
5//! top, so nothing is fetched twice and nothing is shown that the capability did not say. A
6//! page whose channel cannot be opened asks for the page after its cursor every few seconds
7//! instead, and says that it polls. Without JavaScript the page is complete, and a refresh
8//! shows the newest signals.
9

21/// How many signals the first page shows.
22const FIRST_PAGE: u64 = 100;
23
24/// The sources the first page reads, by `monitor.stream`'s default, named for the live
25/// channel so that it continues the same cursor.
26const SOURCES: &str = "mesh,executor,github";

28/// `/cockpit/monitor[?for=<names>]`.
29pub fn page(ctx: &Context, query: &[(String, String)]) -> Page {
30    let reader = param(query, "for").filter(|r| !r.trim().is_empty());
31    let mut input = json!({ "limit": FIRST_PAGE, "since": 6 * 3600 });
32    if let Some(reader) = &reader {
33        input["for"] = json!(reader);
34    }
35    let page: MonitorPage = match ask(ctx, "monitor.stream", input) {
36        Ok(p) => p,
37        Err(e) => return failed(Area::Mesh, "Monitor", e),
38    };
39
40    let mine: Vec<&MonitorSignal> = page.signals.iter().rev().filter(|s| s.for_you).collect();
41    let for_you = if reader.is_none() {
42        el("p").class("mj-prose").text(
43            "Say who you are and what names you is listed here first: add ?for=<your names> to this page's address, e.g. ?for=1b,cd.",
44        )
45    } else if mine.is_empty() {
46        el("p")
47            .class("mj-empty")
48            .text("Nothing in this window names you.")
49    } else {
50        rows(&mine)
51    };
52
53    let sources = el("ul")
54        .class("mj-list")
55        .children(page.sources.iter().map(|s| {
56            let state = health_word(s.state);
57            let mut item = el("li")
58                .child(el("strong").text(format!("{:?}", s.source).to_lowercase()))
59                .child(el("span").text(" "))
60                .child(word_badge(state))
61                .child(el("span").text(format!(
62                    " {} signal(s), {} folded away",
63                    s.signals, s.suppressed
64                )));
65            if let Some(reason) = &s.reason {
66                // `mj-identity`: a source's reason may name a key with no break in it
67                item = item.child(
68                    el("span")
69                        .class("mj-muted mj-identity")
70                        .text(format!(" — {reason}")),
71                );
72            }
73            item
74        }));
75
76    let newest_first: Vec<&MonitorSignal> = page.signals.iter().rev().collect();
77    let stream = if newest_first.is_empty() {
78        el("p")
79            .class("mj-empty")
80            .attr("data-mj-monitor-empty", "")
81            .text("Nothing happened in the last six hours.")
82    } else {
83        rows(&newest_first)
84    };
85    let live = el("p")
86        .class("mj-muted")
87        .attr("data-mj-monitor-note", "")
88        .text("While this page is open the server pushes what happens and it is added on top.");
89    let mut source_url = "/api/v1/monitor?limit=200".to_string();
90    let mut live_url = format!("/events?follow=blackboard&frames=signals&sources={SOURCES}");
91    // the page's cursor always, an empty one too: it is the start of the stream, so a page
92    // rendered before anything happened resumes from there and misses nothing since
93    live_url.push_str(&format!(
94        "&after={}&epoch={}",
95        percent_encode(&page.cursor),
96        percent_encode(&crate::http::follow::epoch(ctx))
97    ));
98    if let Some(reader) = &reader {
99        source_url.push_str(&format!("&for={}", percent_encode(reader)));
100        live_url.push_str(&format!("&for={}", percent_encode(reader)));
101    }
102    let frame = el("div")
103        .attr("data-mj-monitor", source_url)
104        .attr("data-mj-monitor-cursor", &page.cursor)
105        .attr("data-mj-monitor-live", live_url)
106        .child(live)
107        .child(stream);
108
109    Page::new(
110        Area::Mesh,
111        "Monitor",
112        el("div")
113            .class("mj-grid")
114            .child(card(format!("For you ({})", mine.len()), for_you))
115            .child(card_with(
116                "Everything, newest first",
117                el("span").class("mj-muted").text(format!(
118                    "{} folded away as noise",
119                    page.suppressed
120                )),
121                frame,
122            ))
123            .child(card("Sources", sources))
124            .child(card(
125                "From a terminal",
126                // the scrolling block every listing uses: a bare <pre> is wider than a phone
127                pre(
128                    "majordomus monitor --for 1b            # this page, for one reader\nmajordomus monitor --follow             # keep printing what is new\nmajordomus monitor --follow --match '#885' --until merged --timeout 1h",
129                ),
130            )),
131    )
132    .subtitle("Everything that happens around this repository, in one stream: sessions, claims, handovers and reviews on every linked machine, the executor's merges and failures, and GitHub's pull requests and runs.")
133    .trail(vec![
134        ("Cockpit", Some("/cockpit")),
135        ("Mesh", Some("/cockpit/mesh")),
136        ("Monitor", None),
137    ])
138    .script("monitor.js")
139}

141/// Signals as table rows: time (UTC), source, machine, severity, summary.
142fn rows(signals: &[&MonitorSignal]) -> El {
143    table(
144        &["Time (UTC)", "Source", "Machine", "", "What happened"],
145        signals.iter().map(|s| row(s)).collect(),
146    )
147}
148
149fn row(s: &MonitorSignal) -> El {
150    let day = s.at % 86_400;
151    let time = format!("{:02}:{:02}:{:02}", day / 3600, day % 3600 / 60, day % 60);
152    let mark = match s.severity {
153        MonitorSeverity::Action => "action",
154        MonitorSeverity::Attention => "attention",
155        MonitorSeverity::Info => "",
156    };
157    let summary = match &s.link {
158        Some(link) => el("a").attr("href", link).text(&s.summary),
159        None => el("span").text(&s.summary),
160    };
161    el("tr")
162        .attr("data-mj-severity", mark)
163        .attr("data-mj-for-you", if s.for_you { "yes" } else { "no" })
164        .child(el("td").class("mj-mono").text(time))
165        .child(el("td").text(format!("{:?}", s.source).to_lowercase()))
166        .child(el("td").text(s.machine.clone().unwrap_or_default()))
167        .child(el("td").child(if mark.is_empty() {
168            el("span")
169        } else {
170            word_badge(mark)
171        }))
172        .child(el("td").child(summary))
173}

175/// The badge word of a source's state.
176fn health_word(state: MonitorSourceHealth) -> &'static str {
177    match state {
178        MonitorSourceHealth::Ok => "ok",
179        MonitorSourceHealth::Absent => "absent",
180        MonitorSourceHealth::Error => "error",
181        MonitorSourceHealth::Skipped => "skipped",
182    }
183}

196    /// A reader named in the address sees what names them first, as rows with the action
197    /// marked, and the live list asks for the same reader; a reader named nowhere is told so.
198    #[test]
199    fn a_named_reader_sees_their_signals_first_and_the_live_list_asks_for_them() {
200        let repo = crate::synthetic::SyntheticRepository::small().unwrap();
201        let ctx = context(&repo);
202        let c = cooperation();
203        session(&c, "s1", "3d", "the runtime train");
204        c.request_review("s1", "1b, read #885".into(), vec![], None, None, None)
205            .unwrap();
206        crate::mesh::testing::attach(&ctx, c);
207
208        let for_1b = page(&ctx, &[("for".into(), "1b".into())]);
209        let html = body(&for_1b);
210        assert!(html.contains("For you (1)"), "{html}");
211        assert!(html.contains("data-mj-severity=\"action\""), "{html}");
212        assert!(html.contains("1b, read #885"), "{html}");
213        assert!(
214            html.contains("/api/v1/monitor?limit=200&amp;for=1b"),
215            "{html}"
216        );
217        // the live channel continues the page's own cursor, for the same reader
218        let live = html
219            .split("data-mj-monitor-live=\"")
220            .nth(1)
221            .and_then(|rest| rest.split('"').next())
222            .expect("the page names its live channel");
223        assert!(
224            live.starts_with("/events?follow=blackboard&amp;frames=signals"),
225            "{live}"
226        );
227        assert!(live.contains("&amp;after="), "{live}");
228        assert!(live.contains("&amp;epoch="), "{live}");
229        assert!(live.ends_with("&amp;for=1b"), "{live}");
230        assert!(html.contains("<strong>mesh</strong>"), "{html}");
231        assert!(
232            !html.contains("Nothing happened in the last six hours"),
233            "{html}"
234        );
235
236        let for_nobody = page(&ctx, &[("for".into(), "zz".into())]);
237        assert!(body(&for_nobody).contains("Nothing in this window names you."));
238        let blank = page(&ctx, &[("for".into(), " ".into())]);
239        assert!(
240            body(&blank).contains("Say who you are"),
241            "a blank name is no name"
242        );
243    }

274    #[test]
275    fn a_row_carries_its_time_severity_and_link() {
276        let mut s = MonitorSignal::new(
277            3_723,
278            MonitorSource::Github,
279            "pr_merged",
280            MonitorSeverity::Attention,

287        assert!(html.contains("data-mj-severity=\"attention\""), "{html}");
288        assert!(html.contains("data-mj-for-you=\"yes\""), "{html}");
289        assert!(
290            html.contains("href=\"https://example.invalid/pull/885\""),
291            "{html}"
292        );
293        let plain = MonitorSignal::new(0, MonitorSource::Mesh, "k", MonitorSeverity::Info, "x");
294        assert!(row(&plain).render().contains("<span>x</span>"));
295    }
