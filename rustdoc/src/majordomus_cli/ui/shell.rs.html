1//! The frame every screen of the UI is drawn in: the site's bar, the sidebar, the tool bar,
2//! and the screen.
3//!
4//! The bar at the top is majordomus.dev's, the reference the product's look is taken from
5//! (`project.the-ui-wears-the-sites-navbar`): the brand, the version, the site's four
6//! questions, its search, its theme toggle and its call to action, as the site renders them
7//! ([`super::navbar`]). The UI adds nothing to it and changes nothing in it but the version it
8//! names and the entry that leads to the UI itself.
9//!
10//! Beneath it the UI's own: the sidebar, [`super::SCREENS`] and nothing else as its
11//! navigation, and at its foot the version card, read from `distribution.releases` like any
12//! other value. The tool bar holds the search and the links to the other surfaces of the
13//! process — the documentation and the API — named by the constants of the modules that
14//! declare them, and the Cockpit as the technical reference. On a narrow screen the sidebar
15//! is a menu that opens without JavaScript (a checkbox and its label).
16

31/// What a screen renders into the frame.
32pub struct View {
33    /// The title, for the document's `<title>`; the screen's heading is in its hero.
34    pub title: String,
35    /// The screen.
36    pub main: El,
37    /// The HTTP status.
38    pub status: u16,
39}
40
41impl View {
42    /// A screen that answered.
43    pub fn new(title: impl Into<String>, main: El) -> Self {
44        View {
45            title: title.into(),
46            main,
47            status: 200,
48        }
49    }
50}

52/// Where the frame is drawn, and with which files.
53pub struct Frame<'a> {
54    /// The route being answered, which marks the current screen.
55    pub current: &'a str,
56    /// The stylesheet URL, with its digest.
57    pub stylesheet: String,
58    /// The favicon URL, with its digest.
59    pub favicon: String,
60    /// The script URL, with its digest.
61    pub script: String,
62    /// The site's own scripts the bar's behaviours are written in — its menus, then its theme
63    /// toggle — served from the UI's files, with their digests.
64    pub bar_scripts: Vec<String>,
65    /// The graph library's URL, with its digest: loaded by the script only on a screen that
66    /// draws a graph.
67    pub graph_lib: String,
68    /// Whether the distribution has the UI's asset directory at all.
69    pub assets_present: bool,
70    /// The executable's version.
71    pub version: &'a str,
72}

74/// One whole page: the frame around a screen.
75pub fn page(ctx: &Context, frame: &Frame<'_>, view: &View) -> String {
76    let head: Vec<El> = vec![
77        el("link")
78            .attr("rel", "stylesheet")
79            .attr("href", frame.stylesheet.clone()),
80        el("link")
81            .attr("rel", "icon")
82            .attr("href", frame.favicon.clone()),
83        // the one inline script, the statement every surface ships for its theme; the policy
84        // allows it by its digest (crate::cockpit::csp)
85        el("script").raw(theme_bootstrap()),
86        el("script")
87            .attr("type", "module")
88            .attr("src", frame.script.clone())
89            .flag("defer"),
90    ]
91    .into_iter()
92    .chain(
93        frame
94            .bar_scripts
95            .iter()
96            .map(|src| el("script").attr("src", src.clone()).flag("defer")),
97    )
98    .collect();
99    let design = DesignSystem::compiled().ok();
100    // the theme contract on the root too, where the site's theme script reads it
101    let root: Vec<(&str, String)> = design
102        .map(|d| {
103            vec![
104                ("data-theme-key", d.theme.storage_key.clone()),
105                ("data-theme-class", d.theme.class.clone()),
106            ]
107        })
108        .unwrap_or_default();
109    let body = el("body")
110        .class("mj-body mj-ui")
111        .attr_if(
112            "data-theme-key",
113            design.map(|d| d.theme.storage_key.clone()),
114        )
115        .attr_if("data-theme-class", design.map(|d| d.theme.class.clone()))
116        .attr_if("data-design", design.map(|d| d.short_fingerprint()))
117        .attr("data-graph-lib", frame.graph_lib.clone())
118        .child(
119            el("a")
120                .class("mj-skip")
121                .attr("href", "#main")
122                .text("Skip to content"),
123        )
124        .raw(navbar::render(frame.version, &[(SURFACE, PREFIX)]))
125        .child(
126            el("input")
127                .class("mj-ui-menu-toggle")
128                .attr("type", "checkbox")
129                .attr("id", "mj-ui-menu")
130                .attr("aria-label", "Show the menu"),
131        )
132        .child(
133            el("div")
134                .class("mj-ui-frame")
135                .child(sidebar(ctx, frame))
136                .child(
137                    el("div")
138                        .class("mj-ui-body")
139                        .child(topbar(&crate::peers::rfc3339(std::time::SystemTime::now())))
140                        .child(
141                            el("main")
142                                .attr("id", "main")
143                                .class("mj-ui-main")
144                                .when(!frame.assets_present, |m| {
145                                    m.child(alert(
146                                        "warn",
147                                        "This distribution has no share/ui/ directory, so the UI is unstyled. Run scripts/cockpit-assets.",
148                                    ))
149                                })
150                                .child(view.main.clone()),
151                        ),
152                ),
153        );
154    document_on(
155        &format!("{} · majordomus ui", view.title),
156        &root,
157        head,
158        body,
159    )
160}

162/// The sidebar: the screens, and the version card.
163fn sidebar(ctx: &Context, frame: &Frame<'_>) -> El {
164    let screens = el("ul")
165        .class("mj-ui-nav-list")
166        .attr("aria-label", "Screens")
167        .children(SCREENS.iter().map(|s| {
168            el("li").child(
169                el("a")
170                    .class("mj-ui-nav-item")
171                    .attr("href", s.route)
172                    .attr("title", s.answers)
173                    .attr_if("aria-current", (frame.current == s.route).then_some("page"))
174                    .child(icon(s.icon))
175                    .child(el("span").class("mj-ui-label").text(s.title)),
176            )
177        }));
178    // no brand of its own: the bar above carries the site's
179    el("aside")
180        .class("mj-ui-sidebar")
181        .attr("aria-label", "majordomus ui")
182        .child(
183            el("nav")
184                .class("mj-ui-nav")
185                .attr("aria-label", "Screens")
186                .child(screens),
187        )
188        .child(version_card(ctx, frame))
189}

191/// The version card: what this executable is, and what was last released, as
192/// `distribution.releases` answers it.
193fn version_card(ctx: &Context, frame: &Frame<'_>) -> El {
194    let releases = ask(ctx, Source::of("distribution.releases"));
195    let latest = releases.read("/latest/version");
196    let published = releases.read("/latest/published_at");
197    let commit = releases.read("/latest/commit");
198    el("div")
199        .class("mj-ui-version-card")
200        .attr("title", format!("read from {}", releases.provenance()))
201        .child(
202            el("p")
203                .class("mj-ui-version-head")
204                .child(
205                    el("span")
206                        .class("mj-ui-version")
207                        .text(format!("v{}", frame.version)),
208                )
209                .child(
210                    el("span")
211                        .class("mj-ui-version-note")
212                        .text("this executable"),
213                ),
214        )
215        .child(
216            el("p")
217                .class("mj-ui-version-line")
218                .text(release_line(&latest, &published, &commit)),
219        )
220}

222/// The latest release in one line: its version, the day it was published and its short
223/// commit, each one said to be missing in the reading's own words when it is.
224fn release_line(latest: &Reading, published: &Reading, commit: &Reading) -> String {
225    let day_of = match published.text() {
226        Some(d) => day(&d),
227        None => shown(published),
228    };
229    let short = match commit.text() {
230        Some(c) => c.chars().take(10).collect::<String>(),
231        None => shown(commit),
232    };
233    format!("Latest release {} · {day_of} · {short}", shown(latest))
234}

236/// The tool bar beneath the site's: the menu button of a narrow screen, the search, the other
237/// surfaces. The theme toggle is the site's, in the bar above.
238fn topbar(read_at: &str) -> El {
239    el("header")
240        .class("mj-ui-topbar")
241        .child(
242            el("label")
243                .class("mj-ui-menu-button")
244                .attr("for", "mj-ui-menu")
245                .child(icon("menu"))
246                .child(el("span").class("mj-sr-only").text("Menu")),
247        )
248        .child(
249            el("form")
250                .class("mj-ui-search")
251                .attr("role", "search")
252                .attr("method", "get")
253                .attr("action", format!("{}/search", crate::cockpit::PREFIX))
254                .child(icon("search"))
255                .child(
256                    el("label")
257                        .class("mj-sr-only")
258                        .attr("for", "mj-ui-q")
259                        .text("Search capabilities, objects, commands"),
260                )
261                .child(
262                    el("input")
263                        .attr("id", "mj-ui-q")
264                        .attr("type", "search")
265                        .attr("name", "q")
266                        .attr("placeholder", "Search capabilities, objects, commands…"),
267                ),
268        )
269        .child(
270            el("p")
271                .class("mj-ui-fresh")
272                .attr("aria-live", "polite")
273                .attr("title", "when this screen read its sources; it reads them again every minute while it is open")
274                .child(el("span").class("mj-ui-fresh-dot").attr("aria-hidden", "true"))
275                .child(
276                    el("span")
277                        .text("Read at ")
278                        .child(
279                            el("time")
280                                .attr("data-read-at", "")
281                                .attr("datetime", read_at)
282                                .text(read_at.replace('T', " ")),
283                        ),
284                ),
285        )
286        .child(
287            el("nav")
288                .class("mj-ui-tools")
289                .attr("aria-label", "Other surfaces")
290                .child(tool(crate::web::discover::DOCS_MOUNT, "book", "Docs"))
291                .child(tool(crate::http::swagger::SWAGGER_PATH, "code", "API"))
292                .child(tool(crate::cockpit::PREFIX, "terminal", "Cockpit")),
293        )
294}

296/// A link to another surface. Below the medium width its label is hidden and only the icon,
297/// which is `aria-hidden`, is drawn, so the link carries its name itself.
298fn tool(href: &str, glyph: &str, label: &str) -> El {
299    el("a")
300        .class("mj-ui-tool")
301        .attr("href", href)
302        .attr("aria-label", label)
303        .child(icon(glyph))
304        .child(el("span").class("mj-ui-label").text(label))
305}

307/// A route under the UI that answers nothing.
308pub fn not_found(path: &str) -> View {
309    View {
310        title: "Not a screen".into(),
311        main: el("div")
312            .class("mj-ui-stack")
313            .child(alert("fail", format!("The UI serves no screen at {path}.")))
314            .child(
315                el("p").child(
316                    el("a")
317                        .class("mj-ui-link")
318                        .attr("href", PREFIX)
319                        .text("Back to the dashboard"),
320                ),
321            ),
322        status: 404,
323    }
324}

347    /// The release line shows the day and the short commit of a release it was given, and
348    /// says which reading is missing when it was not.
349    #[test]
350    fn the_release_line_says_what_it_read_and_what_is_missing() {
351        let releases = crate::ui::source::Answer {
352            source: Source::of("distribution.releases"),
353            route: None,
354            exposure: Default::default(),
355            value: Ok(serde_json::json!({ "latest": {
356                "version": "0.20.0",
357                "published_at": "2026-10-10T16:38:00Z",
358                "commit": "c8f0e4c50123456789",
359            }})
360            .into()),
361            took: std::time::Duration::ZERO,
362            standing: crate::ui::source::Standing::NOW,
363        };
364        let line = release_line(
365            &releases.read("/latest/version"),
366            &releases.read("/latest/published_at"),
367            &releases.read("/latest/commit"),
368        );
369        assert_eq!(line, "Latest release 0.20.0 · 2026-10-10 · c8f0e4c501");
370        let missing = releases.read("/latest/nothing");
371        assert_eq!(
372            release_line(&missing, &missing, &missing),
373            "Latest release none · none · none"
374        );
375    }

436    /// Every page wears the site's bar above the UI's own frame: the site's landmark, its
437    /// theme toggle and its menus, the executable's version in the badge, the entry for the
438    /// product leading to the UI itself, and the theme contract on the root where the site's
439    /// script reads it. The UI draws no theme toggle and no brand of its own beside it.
440    #[test]
441    fn every_page_wears_the_sites_bar() {
442        let html = render(PREFIX);
443        assert_eq!(html.matches("aria-label=\"Main\"").count(), 1, "{html}");
444        assert!(html.find("aria-label=\"Main\"") < html.find("mj-ui-frame"));
445        assert!(html.contains("id=\"theme-toggle\""));
446        assert!(html.contains(">v0.0.0<"), "the badge names the executable");
447        assert!(
448            html.contains("href=\"/ui\" data-nav-primary=\"ui\""),
449            "{html}"
450        );
451        assert!(!html.contains("data-theme-toggle"), "a second theme toggle");
452        assert!(!html.contains("mj-ui-brand"), "a second brand");
453        assert!(
454            html.contains("<html lang=\"en\" data-theme-key=\""),
455            "{html}"
456        );
457        for s in ["/ui/assets/nav.js?v=1", "/ui/assets/theme.js?v=1"] {
458            assert!(html.contains(&format!("<script src=\"{s}\" defer>")), "{s}");
459        }
