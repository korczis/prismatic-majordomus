1//! The tool distribution's data directory: `share/kinds.yaml` and
2//! `share/schemas/<kind>.schema.json`, read at run time. Nothing about kinds or keys is
3//! compiled into the executable; the directory is located per invocation, explicitly or
4//! by convention, and named in every error when it is not.
5//!
6//! # Why nothing is compiled in
7//!
8//! A kind is a fact about a repository's layer, not about the executable that reads it. If
9//! the kinds were compiled in, adding one would mean shipping a binary, and a repository
10//! that added its own would be adding it to somebody else's program. Reading them at run
11//! time is what makes `.ai/repo/knowledge/` able to declare a kind that the same binary
12//! then serves.
13//!
14//! ```
15//! use majordomus_cli::share::Share;
16//!
17//! // located explicitly, and the paths under it are derived rather than joined by callers
18//! let dir = tempfile::tempdir().unwrap();
19//! std::fs::write(dir.path().join("kinds.yaml"), "version: 1\nkinds: []\n").unwrap();
20//! std::fs::create_dir(dir.path().join("schemas")).unwrap();
21//!
22//! let share = Share::locate(Some(dir.path()), dir.path()).unwrap();
23//! assert_eq!(share.origin, "--share", "it says how it was found");
24//! assert!(share.kinds_path().ends_with("kinds.yaml"));
25//! assert!(share.schemas_dir().ends_with("schemas"));
26//! assert_eq!(share.kinds_path().parent(), Some(share.dir()));
27//!
28//! // and when there is none, the error names where it looked
29//! let nowhere = tempfile::tempdir().unwrap();
30//! let err = Share::locate(None, nowhere.path()).unwrap_err();
31//! assert_eq!(err.exit_code(), 12);
32//! ```

68/// A located distribution directory.
69#[derive(Debug, Clone, PartialEq, Eq)]
70pub struct Share {
71    dir: PathBuf,
72    /// How it was found, for diagnostics: `--share`, the environment, the repository, or
73    /// the executable's own installation.
74    pub origin: &'static str,
75}
76
77impl Share {
78    /// Locate the share directory. In order: `explicit` (from `--share`), `MAJORDOMUS_SHARE`,
79    /// `<repository root>/share` when it holds a kinds file (this repository supervising
80    /// itself), and `<directory of the executable>/../share` (an installation laid out as
81    /// `bin/` beside `share/`). A candidate that exists but holds no kinds file is skipped
82    /// only when it came from a convention; an explicit path that lacks one is an error.
83    pub fn locate(explicit: Option<&Path>, repo_root: &Path) -> Result<Self> {
84        let mut tried = Vec::new();
85        if let Some(p) = explicit {
86            return Self::open(p, "--share");
87        }
88        if let Some(p) = std::env::var_os(SHARE_ENV) {
89            return Self::open(Path::new(&p), SHARE_ENV);
90        }
91        let in_repo = repo_root.join("share");
92        if in_repo.join(KINDS_FILE).is_file() {
93            return Self::open(&in_repo, "repository");
94        }
95        tried.push(in_repo);
96        if let Ok(exe) = std::env::current_exe() {
97            if let Some(dir) = exe.parent() {
98                let beside = dir.join("../share");
99                if beside.join(KINDS_FILE).is_file() {
100                    return Self::open(&beside, "installation");
101                }
102                tried.push(beside);
103            }
104        }
105        Err(Error::ShareNotFound { tried })
106    }
107
108    fn open(dir: &Path, origin: &'static str) -> Result<Self> {
109        let kinds = dir.join(KINDS_FILE);
110        if !kinds.is_file() {
111            return Err(Error::ShareNotFound {
112                tried: vec![dir.to_path_buf()],
113            });
114        }
115        let dir = dir.canonicalize().map_err(|e| Error::io(dir, e))?;
116        Ok(Share { dir, origin })
117    }
118

120    pub fn dir(&self) -> &Path {
121        &self.dir
122    }

160    /// The providers the tool has a template for, by id (the file stem), sorted. A
161    /// distribution without the directory has none; that is not an error, it is a
162    /// distribution that ships no adapter.
163    pub fn provider_templates(&self) -> Vec<String> {
164        let Ok(entries) = std::fs::read_dir(self.providers_dir()) else {
165            return Vec::new();
166        };
167        let mut out: Vec<String> = entries
168            .filter_map(|e| e.ok())
169            .filter_map(|e| e.file_name().to_str().map(str::to_string))
170            .filter_map(|n| n.strip_suffix(".tmpl").map(str::to_string))
171            .collect();
172        crate::order::canonical(&mut out);
173        out
174    }

181    /// The provider declarations the distribution ships, joined with its templates: one
182    /// [`ProviderDeclaration`] per id that has either, sorted by id. A distribution without
183    /// the file declares nothing and every template is a provider named by its file; a
184    /// file that does not parse is an error naming it, because a declaration half-read is
185    /// worse than none.
186    pub fn providers(&self) -> Result<ProviderDeclarations> {
187        let path = self.providers_path();
188        let file: ProvidersFile = match std::fs::read_to_string(&path) {
189            Ok(text) => {
190                crate::metadata::yaml::parse_into(&text).map_err(|reason| Error::KindSchema {
191                    reason: format!("{}: {reason}", path.display()),
192                })?
193            }
194            Err(e) if e.kind() == std::io::ErrorKind::NotFound => ProvidersFile::default(),
195            Err(e) => return Err(Error::io(&path, e)),
196        };
197        let templates = self.provider_templates();
198        let mut ids: Vec<String> = templates.clone();
199        ids.extend(file.providers.keys().cloned());
200        crate::order::canonical(&mut ids);
201        ids.dedup();
202        let providers = ids
203            .into_iter()
204            .map(|id| {
205                let decl = file.providers.get(&id);
206                ProviderDeclaration {
207                    title: decl
208                        .and_then(|d| d.title.clone())
209                        .unwrap_or_else(|| id.clone()),
210                    client_config: decl.and_then(|d| d.client_config.clone()),
211                    scratch_roots: decl.map(|d| d.scratch_roots.clone()).unwrap_or_default(),
212                    lifecycle: decl.map(|d| d.lifecycle.clone()).unwrap_or_default(),
213                    prompt_capture: decl.map(|d| d.prompt_capture).unwrap_or(false),
214                    template: templates.contains(&id),
215                    declared: decl.is_some(),
216                    id,
217                }
218            })
219            .collect();
220        Ok(ProviderDeclarations {
221            scratch_roots: file.scratch_roots,
222            providers,
223        })
224    }
225}

264/// One provider: the template's file stem, and what the declaration adds to it.
265#[derive(
266    Debug, Clone, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize, schemars::JsonSchema,
267)]
268pub struct ProviderDeclaration {
269    /// The id: the template's file stem, and the key in the declarations.
270    pub id: String,
271    /// The name a person knows it by; the id when nothing is declared.
272    pub title: String,
273    /// The file it reads a project-scoped MCP client configuration from, when it has one.
274    #[serde(default, skip_serializing_if = "Option::is_none")]
275    pub client_config: Option<String>,
276    /// The scratch roots it creates checkouts under, unexpanded.
277    #[serde(default)]
278    pub scratch_roots: Vec<String>,
279    /// The lifecycle events the tool's adapter for it declares, in the provider's own
280    /// vocabulary (`SessionStart`, `SessionEnd`, `PreCompact`). Empty means the tool ships
281    /// no lifecycle adapter for it, which costs a worker the automation and none of the
282    /// model. Declared, because the two other ways to answer this question are both wrong:
283    /// reading `.claude/hooks/` reports an installation as a capability, and a list written
284    /// into a page is a second source of truth that goes stale the day an adapter changes.
285    #[serde(default)]
286    pub lifecycle: Vec<String>,
287    /// Whether the tool's adapter for it can archive the worker's prompts.
288    #[serde(default)]
289    pub prompt_capture: bool,
290    /// The distribution ships a template for it.
291    pub template: bool,
292    /// The distribution declares it.
293    pub declared: bool,
294}

296impl ProviderDeclarations {
297    /// Every scratch root — the tool's own and every provider's — expanded against
298    /// `primary` (the repository's primary checkout) and the environment, in declaration
299    /// order, with the provider that declared each. A root whose expansion needs a variable
300    /// the environment does not hold is skipped, not an error.
301    pub fn scratch_roots(&self, primary: &Path) -> Vec<(Option<String>, PathBuf)> {
302        let mut out = Vec::new();
303        for r in &self.scratch_roots {
304            if let Some(p) = expand_root(r, primary) {
305                out.push((None, p));
306            }
307        }
308        for p in &self.providers {
309            for r in &p.scratch_roots {
310                if let Some(path) = expand_root(r, primary) {
311                    out.push((Some(p.id.clone()), path));
312                }
313            }
314        }
315        out
316    }
317}

319/// Expand one declared root: `<primary>/`, `~/`, `${NAME:-default}` and `$NAME`, nothing
320/// else. `None` when a variable without a default is unset.
321///
322/// ```
323/// # use std::path::Path;
324/// use majordomus_cli::share::expand_root;
325/// std::env::set_var("MJ_DOCTEST_ROOT", "/srv/x");
326/// std::env::remove_var("MJ_DOCTEST_UNSET");
327/// let primary = Path::new("/repo");
328/// assert_eq!(expand_root("<primary>/.claude/worktrees", primary).unwrap(), Path::new("/repo/.claude/worktrees"));
329/// assert_eq!(expand_root("$MJ_DOCTEST_ROOT/wt", primary).unwrap(), Path::new("/srv/x/wt"));
330/// assert_eq!(expand_root("${MJ_DOCTEST_UNSET:-/d}/wt", primary).unwrap(), Path::new("/d/wt"));
331/// assert!(expand_root("$MJ_DOCTEST_UNSET/wt", primary).is_none());
332/// assert_eq!(expand_root("/tmp", primary).unwrap(), Path::new("/tmp"));
333/// ```
334pub fn expand_root(root: &str, primary: &Path) -> Option<PathBuf> {
335    let mut out = String::new();
336    let mut rest = root;
337    while !rest.is_empty() {
338        if let Some(r) = rest.strip_prefix("<primary>") {
339            out.push_str(&primary.to_string_lossy());
340            rest = r;
341        } else if let Some(r) = rest.strip_prefix('~') {
342            let home = std::env::var_os("HOME")?;
343            out.push_str(&home.to_string_lossy());
344            rest = r;
345        } else if let Some(r) = rest.strip_prefix("${") {
346            let end = r.find('}')?;
347            let (name, default) = match r[..end].split_once(":-") {
348                Some((n, d)) => (n, Some(d)),
349                None => (&r[..end], None),
350            };
351            match std::env::var(name) {
352                Ok(v) if !v.is_empty() => {
353                    out.push_str(&expand_root(&v, primary)?.to_string_lossy())
354                }
355                _ => out.push_str(&expand_root(default?, primary)?.to_string_lossy()),
356            }
357            rest = &r[end + 1..];
358        } else if let Some(r) = rest.strip_prefix('$') {
359            let end = r
360                .find(|c: char| !(c.is_ascii_alphanumeric() || c == '_'))
361                .unwrap_or(r.len());
362            let v = std::env::var(&r[..end]).ok().filter(|v| !v.is_empty())?;
363            out.push_str(&v);
364            rest = &r[end..];
365        } else {
366            let end = rest.find(['<', '~', '$']).unwrap_or(rest.len());
367            let end = if end == 0 { 1 } else { end };
368            out.push_str(&rest[..end]);
369            rest = &rest[end..];
370        }
371    }
372    Some(PathBuf::from(out))
373}

375/// Every `<vendor>/<name>/<name>.v<n>.schema.json` under a schema root, parsed, keyed by
376/// the schema identity its path derives (`majordomus.adr/v1`) and sorted by it. A root that
377/// does not exist yields nothing; that is the repository's case when it adds no schema of
378/// its own.
379///
380/// The identity and the path fix each other, so a schema can be found from a `kinds.yaml`
381/// entry without a directory scan and a file cannot quietly answer to a name it does not
382/// declare. A file whose path does not have the shape is refused by name rather than
383/// skipped, because a schema nobody loads validates nothing and says so nowhere.
384pub fn read_schema_dir(dir: &Path) -> Result<Vec<(String, serde_json::Value)>> {
385    let mut schemas = Vec::new();
386    read_schema_tree(dir, dir, &mut schemas)?;
387    schemas.sort_by(|a, b| a.0.cmp(&b.0));
388    Ok(schemas)
389}
390
391fn read_schema_tree(
392    root: &Path,
393    dir: &Path,
394    out: &mut Vec<(String, serde_json::Value)>,
395) -> Result<()> {
396    if !dir.is_dir() {
397        return Ok(());
398    }
399    for entry in std::fs::read_dir(dir).map_err(|e| Error::io(dir, e))? {
400        let path = entry.map_err(|e| Error::io(dir, e))?.path();
401        if path.is_dir() {
402            // `<schemas>/generated/` holds the contracts of the *generated documents* —
403            // the manifest, the registry, the benchmark matrix. Those are not kinds of the
404            // layer, carry no `<vendor>/<name>/` identity, and are read by
405            // [`read_generated_schema_dir`] instead.
406            if dir == root
407                && path.file_name().and_then(|f| f.to_str()) == Some(GENERATED_SCHEMAS_DIR)
408            {
409                continue;
410            }
411            read_schema_tree(root, &path, out)?;
412            continue;
413        }
414        let Some(file) = path.file_name().and_then(|f| f.to_str()) else {
415            continue;
416        };
417        if !file.ends_with(SCHEMA_SUFFIX) {
418            continue;
419        }
420        let relative = path
421            .strip_prefix(root)
422            .map(|p| p.to_string_lossy().replace('\\', "/"))
423            .unwrap_or_else(|_| file.to_string());
424        let identity = schema_identity(&relative).ok_or_else(|| Error::KindSchema {
425            reason: format!(
426                "{}: a schema's path is <vendor>/<name>/<name>.v<n>{SCHEMA_SUFFIX}, which this is not",
427                path.display()
428            ),
429        })?;
430        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
431        let json: serde_json::Value =
432            serde_json::from_str(&text).map_err(|e| Error::KindSchema {
433                reason: format!("{}: not JSON: {e}", path.display()),
434            })?;
435        out.push((identity, json));
436    }
437    Ok(())
438}

440/// Every `<name>.schema.json` directly under a directory, parsed, keyed by that `<name>`
441/// and sorted by it. A directory that does not exist yields nothing.
442///
443/// This is the reader for `<share>/schemas/generated/`, whose files are the contracts of
444/// the *generated documents* rather than of the layer's kinds: they answer to the document
445/// id they pin in `properties.schema.const`, not to a `<vendor>/<name>` path, so the shape
446/// [`read_schema_dir`] insists on does not apply to them.
447pub fn read_generated_schema_dir(dir: &Path) -> Result<Vec<(String, serde_json::Value)>> {
448    let mut out = Vec::new();
449    if !dir.is_dir() {
450        return Ok(out);
451    }
452    for entry in std::fs::read_dir(dir).map_err(|e| Error::io(dir, e))? {
453        let path = entry.map_err(|e| Error::io(dir, e))?.path();
454        let Some(file) = path.file_name().and_then(|f| f.to_str()) else {
455            continue;
456        };
457        if path.is_dir() || !file.ends_with(SCHEMA_SUFFIX) {
458            continue;
459        }
460        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
461        let json: serde_json::Value =
462            serde_json::from_str(&text).map_err(|e| Error::KindSchema {
463                reason: format!("{}: not JSON: {e}", path.display()),
464            })?;
465        out.push((file[..file.len() - SCHEMA_SUFFIX.len()].to_string(), json));
466    }
467    out.sort_by(|a, b| a.0.cmp(&b.0));
468    Ok(out)
469}

471/// The identity a schema path carries: `majordomus/adr/adr.v1.schema.json` is
472/// `majordomus.adr/v1`. A path of another shape yields nothing.
473///
474/// ```
475/// use majordomus_cli::share::schema_identity;
476/// assert_eq!(
477///     schema_identity("majordomus/adr/adr.v1.schema.json").as_deref(),
478///     Some("majordomus.adr/v1")
479/// );
480/// assert_eq!(schema_identity("adr.schema.json"), None);
481/// assert_eq!(schema_identity("majordomus/adr/other.v1.schema.json"), None);
482/// ```
483pub fn schema_identity(relative: &str) -> Option<String> {
484    let stem = relative.strip_suffix(SCHEMA_SUFFIX)?;
485    let mut parts = stem.split('/');
486    let vendor = parts.next()?;
487    let name = parts.next()?;
488    let file = parts.next()?;
489    if parts.next().is_some() || vendor.is_empty() || name.is_empty() {
490        return None;
491    }
492    let version = file.strip_prefix(&format!("{name}."))?;
493    if !version.starts_with('v') || version.len() < 2 {
494        return None;
495    }
496    Some(format!("{vendor}.{name}/{version}"))
497}
