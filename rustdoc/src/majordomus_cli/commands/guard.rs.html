1//! `majordomus guard`: the edit guard on the command line (I2735).
2//!
3//! `guard hook` is what a provider's pre-tool hook runs. It reads the provider's payload from
4//! stdin, answers `guard.edit` for the path it names without building the index — it runs
5//! before every edit — and says the answer in the one code the provider reads as a refusal:
6//! 2, with the reason and the remedy on stderr, which the provider hands to the model. Every
7//! other outcome is 0, an undecidable one included: a guard that refused because it could
8//! not read its own inputs would stop every edit of a checkout nobody has served.
9//!
10//! A payload of the shell tool is asked the second question first: does its command start
11//! a build on a declared cluster host outside the offload path ([`crate::guard`]'s
12//! `remote_build`, I3956)? That refusal is 2 as well, with the rule and the way.
13
14use std::io::{Read, Write};
15use std::path::{Path, PathBuf};

17use crate::cli::{GuardArgs, GuardCommand, GuardHookArgs};
18use crate::error::Result;
19use crate::guard::remote_build::{declared_hosts, from_payload as from_command, RemoteBuild};
20use crate::guard::{from_payload, judge, server_claims, ClaimSource, EditVerdict};

22/// The code a provider's pre-tool hook reads as "do not run this tool".
23const EXIT_REFUSED: u8 = 2;
24
25/// Run `majordomus guard`.
26pub fn run(args: GuardArgs) -> Result<u8> {
27    match args.command {
28        GuardCommand::Hook(a) => hook(a),
29    }
30}
31
32fn hook(args: GuardHookArgs) -> Result<u8> {
33    // a payload that cannot be read whole names no path, and an edit nobody can name is let
34    // through like every other undecidable one
35    let mut payload = String::new();
36    let _ = std::io::stdin().read_to_string(&mut payload);
37    let base = args.repo.clone().or_else(|| std::env::current_dir().ok());
38    let build = from_command(&payload, base.as_deref(), declared_hosts);
39    if let Some(code) = report_build(build.as_ref(), &mut std::io::stderr()) {
40        return Ok(code);
41    }
42    let verdict = answer(&args.provider, base, &payload, server_claims);
43    Ok(report(verdict.as_ref(), &mut std::io::stderr()))
44}
45
46/// The exit code for a build the payload's command would start on a cluster host, with the
47/// refusal written to `err`; `None` when it starts none and the edit guard decides.
48fn report_build(build: Option<&RemoteBuild>, err: &mut dyn Write) -> Option<u8> {
49    let build = build?;
50    let _ = writeln!(err, "{}", build.refusal());
51    Some(EXIT_REFUSED)
52}
53
54/// The verdict on the edit `payload` names, or `None` when it names no file of a repository.
55///
56/// The checkout is the one the written file belongs to, not the one the hook was installed
57/// in: a session started in the primary checkout edits files of linked worktrees, and each of
58/// those has its own task, its own server and its own claims. A relative path is `base`'s,
59/// the hook's working directory, which is the provider's project.
60fn answer(
61    provider: &str,
62    base: Option<PathBuf>,
63    payload: &str,
64    claims: impl FnOnce(&Path) -> ClaimSource,
65) -> Option<EditVerdict> {
66    let (path, session) = from_payload(payload)?;
67    let target = base.unwrap_or_default().join(path);
68    let start = target.ancestors().find(|a| a.is_dir())?.to_path_buf();
69    let repository = crate::repository::Repository::discover(&start).ok()?;
70    let root = repository.root().to_path_buf();
71    // the root is canonical, so the path is spelled the same way: its existing ancestor
72    // canonicalised (discover has just done so without failing), and whatever of it does not
73    // exist yet appended (`start` is one of its ancestors, so it is a prefix)
74    let rest = target.strip_prefix(&start).unwrap_or(target.as_path());
75    let path = start.canonicalize().unwrap_or_default().join(rest);
76    let session = session.map(|s| format!("{provider}-{s}"));
77    Some(judge(
78        &root,
79        &path.to_string_lossy(),
80        session.as_deref(),
81        || claims(&root),
82    ))
83}
84
85/// The exit code for `verdict`, with the reason and the remedy written to `err` for a
86/// refusal: the provider hands that text to the model.
87fn report(verdict: Option<&EditVerdict>, err: &mut dyn Write) -> u8 {
88    let Some(refused) = verdict.filter(|v| !v.allowed) else {
89        return 0;
90    };
91    let remedy = refused.remedy.as_deref().unwrap_or_default();
92    let _ = writeln!(
93        err,
94        "majordomus guard: {}\n  {remedy}\n  \
95         (session.edits_require_claim in .ai/repo/policy.yaml turns this guard off; I2735)",
96        refused.detail
97    );
98    EXIT_REFUSED
99}

101#[cfg(test)]
102mod tests {
103    use super::*;

105    /// A repository on the mesh whose policy asks every edit for a claim.
106    fn repository() -> tempfile::TempDir {
107        let skeleton = Path::new(env!("CARGO_MANIFEST_DIR")).join("../../share/skeleton");
108        let dir = tempfile::tempdir().unwrap();
109        let ai = dir.path().join(".ai");
110        std::fs::create_dir_all(ai.join("repo/mesh")).unwrap();
111        std::fs::copy(skeleton.join("ai/manifest.yaml"), ai.join("manifest.yaml")).unwrap();
112        let policy = std::fs::read_to_string(skeleton.join("policy.yaml"))
113            .unwrap()
114            .replace("edits_require_claim: false", "edits_require_claim: true");
115        std::fs::write(ai.join("repo/policy.yaml"), policy).unwrap();
116        std::fs::write(
117            ai.join("repo/mesh/mesh.yaml"),
118            "enabled: true\ntrust:\n  policy: tofu\n",
119        )
120        .unwrap();
121        dir
122    }
123
124    fn edit(path: &str) -> String {
125        serde_json::json!({ "session_id": "s1", "tool_input": { "file_path": path } }).to_string()
126    }
127
128    #[test]
129    fn an_unclaimed_edit_is_refused_with_the_reason_and_the_remedy() {
130        let dir = repository();
131        let root = dir.path().canonicalize().unwrap();
132        let verdict = answer(
133            "claude-code",
134            Some(dir.path().to_path_buf()),
135            &edit("docs/new.md"),
136            |asked| {
137                assert_eq!(asked, root.as_path(), "the checkout of the file is asked");
138                ClaimSource::Claims(vec![])
139            },
140        )
141        .expect("a file of the repository");
142        assert!(!verdict.allowed);
143        assert_eq!(
144            verdict.path, "docs/new.md",
145            "a file that does not exist yet is placed"
146        );
147        let mut err = Vec::new();
148        assert_eq!(report(Some(&verdict), &mut err), EXIT_REFUSED);
149        let text = String::from_utf8(err).unwrap();
150        assert!(text.contains("no claim covers docs/new.md"), "{text}");
151        assert!(text.contains("--session claude-code-s1"), "{text}");
152    }
153
154    #[test]
155    fn an_edit_naming_no_file_of_a_repository_is_let_through() {
156        let dir = repository();
157        let none = |_: &Path| -> ClaimSource { unreachable!("nothing to judge") };
158        let base = Some(dir.path().to_path_buf());
159        assert!(answer("claude-code", base.clone(), "not json", none).is_none());
160        let ls = r#"{"tool_input":{"command":"ls"}}"#;
161        assert!(answer("claude-code", base, ls, none).is_none());
162        let bare = tempfile::tempdir().unwrap();
163        let outside = edit(&bare.path().join("a.md").to_string_lossy());
164        assert!(
165            answer("claude-code", None, &outside, none).is_none(),
166            "no manifest is above it"
167        );
168        let nowhere = edit("no-such-directory/y.md");
169        assert!(answer("claude-code", None, &nowhere, none).is_none());
170        let mut err = Vec::new();
171        assert_eq!(report(None, &mut err), 0);
172        assert!(err.is_empty());
173    }

176    fn a_remote_build_is_refused_before_the_edit_guard_is_asked() {
177        let build = crate::guard::remote_build::remote_build(
178            "ssh lundra cargo test",
179            &[serde_json::from_value(

182            .unwrap()],
183        );
184        let mut err = Vec::new();
185        assert_eq!(report_build(build.as_ref(), &mut err), Some(EXIT_REFUSED));
186        let text = String::from_utf8(err).unwrap();
187        assert!(
188            text.contains("majordomus offload derive --host lundra"),
189            "{text}"
190        );
191        let mut quiet = Vec::new();
192        assert_eq!(report_build(None, &mut quiet), None);
193        assert!(quiet.is_empty(), "no build, nothing said");
194    }
