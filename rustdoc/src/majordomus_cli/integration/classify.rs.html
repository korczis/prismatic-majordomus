1//! The one path that decides a disposition. Pure: observations in, an [`PullRequestAssessment`] out.
2//!
3//! The order of the questions is the policy. A pull request that is a draft is a draft
4//! whatever its checks say; one whose patch is already on master is superseded whatever
5//! its conflicts say; and `Ready` is reached only by the last branch, after every other
6//! question was answered in its favour. Nothing upstream of this file decides eligibility
7//! and nothing downstream re-decides it.
8

20/// What the repository requires before a merge, as observed from the forge and the
21/// repository — never a list kept here beside it.
22#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
23pub struct IntegrationPolicy {
24    /// The branch integrated into.
25    pub base: String,
26    /// The status contexts the branch protection requires, as read from the forge.
27    /// `None` when the protection could not be read: every required-check verdict is then
28    /// `unknown`, never `passed`.
29    pub required_checks: Option<Vec<String>>,
30    /// Whether the branch protection requires an approving review; `None` when unread.
31    pub reviews_required: Option<bool>,
32    /// Labels that hold a pull request whatever else is true.
33    pub blocking_labels: Vec<String>,
34    /// The merge method the executor asks the forge for.
35    pub merge_method: String,
36}

75/// The required-check verdict for one head, from the checks reported on it.
76///
77/// ```text
78/// use crate::integration::{required_checks, CheckObservation, CheckRunState, RequiredCheckState};
79/// let ci = |state| vec![CheckObservation { name: "ci".into(), state }];
80/// let req = Some(vec!["ci".to_string()]);
81/// assert_eq!(required_checks(&ci(CheckRunState::Passed), req.as_deref()), RequiredCheckState::Passed);
82/// // a green check that is not the required one proves nothing
83/// let other = vec![CheckObservation { name: "suite".into(), state: CheckRunState::Passed }];
84/// assert_eq!(required_checks(&other, req.as_deref()), RequiredCheckState::Missing);
85/// // unread protection is never a pass
86/// assert_eq!(required_checks(&ci(CheckRunState::Passed), None), RequiredCheckState::Unknown);
87/// ```text
88pub fn required_checks(
89    checks: &[super::model::CheckObservation],
90    required: Option<&[String]>,
91) -> RequiredCheckState {
92    let Some(required) = required else {
93        return RequiredCheckState::Unknown;
94    };
95    let mut verdict = RequiredCheckState::Passed;
96    for name in required {
97        // the newest report of a context wins; the forge lists re-runs as separate entries,
98        // and any failing one among the latest reports is a failure
99        let states: Vec<CheckRunState> = checks
100            .iter()
101            .filter(|c| &c.name == name)
102            .map(|c| c.state)
103            .collect();
104        let this = if states.is_empty() {
105            RequiredCheckState::Missing
106        } else if states.contains(&CheckRunState::Pending) {
107            RequiredCheckState::Pending
108        } else if states.iter().all(|s| *s == CheckRunState::Passed) {
109            RequiredCheckState::Passed
110        } else if states.contains(&CheckRunState::Failed) {
111            RequiredCheckState::Failed
112        } else {
113            // skipped or neutral is not a pass of a required check
114            RequiredCheckState::Missing
115        };
116        verdict = worse(verdict, this);
117    }
118    verdict
119}
120
121fn worse(a: RequiredCheckState, b: RequiredCheckState) -> RequiredCheckState {
122    let rank = |r| match r {
123        RequiredCheckState::Passed => 0,
124        RequiredCheckState::Pending => 1,
125        RequiredCheckState::Missing => 2,
126        RequiredCheckState::Unknown => 3,
127        RequiredCheckState::Failed => 4,
128    };
129    if rank(b) > rank(a) {
130        b
131    } else {
132        a
133    }
134}

136/// The review state under the protection's requirement.
137pub fn review_state(decision: &str, required: Option<bool>) -> PullRequestReview {
138    match (required, decision) {
139        (_, "CHANGES_REQUESTED") => PullRequestReview::ChangesRequested,
140        (_, "APPROVED") => PullRequestReview::Approved,
141        (Some(false), _) => PullRequestReview::NotRequired,
142        (Some(true), _) => PullRequestReview::Pending,
143        (None, _) => PullRequestReview::Unknown,
144    }
145}

147/// Dependencies declared in a body: a line saying `Depends on #N`, `Stacked on #N`,
148/// `Requires #N` or `After #N`, case-insensitively. Prose that merely mentions a number is
149/// not a declaration.
150///
151/// ```text
152/// use crate::integration::declared_dependencies;
153/// assert_eq!(declared_dependencies("Stacked on #601.\nSee #12 for context."), vec![601]);
154/// ```text
155pub fn declared_dependencies(body: &str) -> Vec<u64> {
156    let mut out = BTreeSet::new();
157    for line in body.lines() {
158        let lower = line.to_ascii_lowercase();
159        for marker in ["depends on", "stacked on", "requires", "after"] {
160            let mut rest = lower.as_str();
161            while let Some(at) = rest.find(marker) {
162                rest = &rest[at + marker.len()..];
163                let tail = rest.trim_start().trim_start_matches(['*', '`', '(']);
164                // a list of numbers after one marker: "stacked on #644 and #645"
165                let mut t = tail;
166                while let Some(stripped) = t.strip_prefix('#') {
167                    let digits: String = stripped
168                        .chars()
169                        .take_while(|c| c.is_ascii_digit())
170                        .collect();
171                    let Ok(n) = digits.parse::<u64>() else { break };
172                    out.insert(n);
173                    t = stripped[digits.len()..]
174                        .trim_start_matches([',', ')', '`', '*', '.'])
175                        .trim_start();
176                    t = t
177                        .strip_prefix("and ")
178                        .or_else(|| t.strip_prefix(", "))
179                        .unwrap_or(t)
180                        .trim_start();
181                }
182            }
183        }
184    }
185    out.into_iter().collect()
186}

188/// The planning risk of a change, with its factors.
189pub fn risk_of(authored: &[String]) -> (IntegrationRisk, Vec<String>) {
190    let mut factors = Vec::new();
191    let mut risk = IntegrationRisk::Low;
192    let high: BTreeSet<&str> = authored
193        .iter()
194        .filter_map(|p| HIGH_RISK.iter().find(|h| p.starts_with(**h)).copied())
195        .collect();
196    if !high.is_empty() {
197        risk = IntegrationRisk::High;
198        factors.extend(high.iter().map(|h| format!("touches {h}")));
199    }
200    if authored.len() > LARGE {
201        risk = IntegrationRisk::High;
202        factors.push(format!(
203            "{} authored paths (more than {LARGE})",
204            authored.len()
205        ));
206    }
207    if risk == IntegrationRisk::Low
208        && authored
209            .iter()
210            .any(|p| CODE.iter().any(|c| p.starts_with(c)))
211    {
212        risk = IntegrationRisk::Medium;
213        factors.push("changes executable code".into());
214    }
215    if factors.is_empty() {
216        factors.push("documentation, tests or content only".into());
217    }
218    (risk, factors)
219}

221/// Authored paths shared with other open pull requests.
222pub fn overlaps(number: u64, authored: &BTreeMap<u64, Vec<String>>) -> Vec<PathOverlap> {
223    let Some(mine) = authored.get(&number) else {
224        return Vec::new();
225    };
226    let mine: BTreeSet<&String> = mine.iter().collect();
227    authored
228        .iter()
229        .filter(|(n, _)| **n != number)
230        .filter_map(|(n, theirs)| {
231            let shared: Vec<String> = theirs
232                .iter()
233                .filter(|p| mine.contains(p))
234                .cloned()
235                .collect();
236            (!shared.is_empty()).then_some(PathOverlap {
237                number: *n,
238                paths: shared,
239            })
240        })
241        .collect()
242}

263/// The disposition of one pull request against one master commit.
264pub fn classify(
265    pr: &PullRequestObservation,
266    relation: &RelationToMaster,
267    master_sha: &str,
268    policy: &IntegrationPolicy,
269    queue: &QueueContext,
270) -> PullRequestAssessment {
271    let checks = required_checks(&pr.checks, policy.required_checks.as_deref());
272    let review = review_state(&pr.review_decision, policy.reviews_required);
273    let authored: Vec<String> = match relation {
274        RelationToMaster::UpToDate { authored } | RelationToMaster::Behind { authored, .. } => {
275            authored.clone()
276        }
277        RelationToMaster::Conflicting { paths } => paths.clone(),
278        _ => queue.authored.get(&pr.number).cloned().unwrap_or_default(),
279    };
280    let (risk, risk_factors) = risk_of(&authored);
281
282    let mut dependencies: Vec<PullRequestDependency> = declared_dependencies(&pr.body)
283        .into_iter()
284        .filter(|n| *n != pr.number)
285        .map(|n| PullRequestDependency {
286            number: n,
287            certainty: DependencyCertainty::Confirmed,
288            satisfied: !queue.open.contains(&n),
289        })
290        .collect();
291    let stacked_on = queue.heads.get(&pr.base_ref).copied();
292    if let Some(n) = stacked_on {
293        if !dependencies.iter().any(|d| d.number == n) {
294            dependencies.push(PullRequestDependency {
295                number: n,
296                certainty: DependencyCertainty::Confirmed,
297                satisfied: false,
298            });
299        }
300    }
301
302    let mut evidence = vec![
303        ev(
304            "required_checks",
305            word(&checks),
306            match &policy.required_checks {
307                Some(r) => format!("required: {}", r.join(", ")),
308                None => "the branch protection could not be read".into(),
309            },
310        ),
311        ev("review", word(&review), pr.review_decision.clone()),
312        ev(
313            "relation_to_master",
314            relation_word(relation),
315            relation_detail(relation),
316        ),
317    ];
318    for d in &dependencies {
319        evidence.push(ev(
320            "dependency",
321            if d.satisfied { "satisfied" } else { "open" },
322            format!("#{} ({})", d.number, word(&d.certainty)),
323        ));
324    }
325    let blocking: Vec<&String> = pr
326        .labels
327        .iter()
328        .filter(|l| {
329            policy
330                .blocking_labels
331                .iter()
332                .any(|b| b.eq_ignore_ascii_case(l))
333        })
334        .collect();
335
336    let (disposition, reasons, next): (PullRequestDisposition, Vec<String>, Option<String>) = if pr
337        .base_ref
338        != policy.base
339    {
340        match stacked_on {
341            Some(n) => (
342                PullRequestDisposition::WaitingForDependency,
343                vec![format!("stacked_on:#{n}")],
344                Some(format!(
345                    "land #{n}; its merge retargets this onto {}",
346                    policy.base
347                )),
348            ),
349            None => (
350                PullRequestDisposition::OtherBase,
351                vec![format!("base_is:{}", pr.base_ref)],
352                None,
353            ),
354        }
355    } else if pr.draft {
356        (
357            PullRequestDisposition::Draft,
358            vec!["draft".into()],
359            Some("mark it ready for review".into()),
360        )
361    } else if !blocking.is_empty() {
362        (
363            PullRequestDisposition::Blocked,
364            blocking.iter().map(|l| format!("label:{l}")).collect(),
365            Some("remove the label when it may land".into()),
366        )
367    } else {
368        match relation {
369            RelationToMaster::Contained => (
370                PullRequestDisposition::Superseded,
371                vec!["head_reachable_from_master".into()],
372                Some("close it: every commit is on master".into()),
373            ),
374            RelationToMaster::Superseded => (
375                PullRequestDisposition::Superseded,
376                vec!["merge_changes_nothing".into()],
377                Some("close it: merging it into master changes no file".into()),
378            ),
379            RelationToMaster::DerivedOnly { .. } => (
380                PullRequestDisposition::PossiblyRedundant,
381                vec!["only_derived_artifacts_differ".into()],
382                Some("a person confirms the authored change is on master, then closes it".into()),
383            ),
384            RelationToMaster::Unknown { reason } => (
385                PullRequestDisposition::Unknown,
386                vec![format!("relation_unknown:{reason}")],
387                Some("majordomus prs refresh".into()),
388            ),
389            RelationToMaster::Conflicting { paths } => (
390                PullRequestDisposition::Conflicting,
391                vec![format!("conflicts_on:{}", paths.len())],
392                Some(format!(
393                    "the author merges {} and resolves {}",
394                    policy.base,
395                    paths.join(", ")
396                )),
397            ),
398            RelationToMaster::UpToDate { .. } | RelationToMaster::Behind { .. } => {
399                if let Some(d) = dependencies
400                    .iter()
401                    .find(|d| d.certainty == DependencyCertainty::Confirmed && !d.satisfied)
402                {
403                    (
404                        PullRequestDisposition::WaitingForDependency,
405                        vec![format!("depends_on:#{}", d.number)],
406                        Some(format!("land #{} first", d.number)),
407                    )
408                } else if matches!(
409                    review,
410                    PullRequestReview::ChangesRequested | PullRequestReview::Pending
411                ) {
412                    (
413                        PullRequestDisposition::WaitingForReview,
414                        vec![format!("review:{}", word(&review))],
415                        Some("a reviewer approves it".into()),
416                    )
417                } else if review == PullRequestReview::Unknown {
418                    (
419                        PullRequestDisposition::Unknown,
420                        vec!["review_policy_unread".into()],
421                        Some("majordomus prs refresh".into()),
422                    )
423                } else if checks == RequiredCheckState::Failed {
424                    let mut reasons = vec!["required_check_failed".to_string()];
425                    if let RelationToMaster::Behind { behind, .. } = relation {
426                        reasons.push(format!("behind_master:{behind}"));
427                    }
428                    (
429                        PullRequestDisposition::NeedsRepair,
430                        reasons,
431                        Some("the author fixes the failing required check".into()),
432                    )
433                } else if let (RelationToMaster::Behind { behind, .. }, true) =
434                    (relation, pr.cross_repository)
435                {
436                    (
437                        PullRequestDisposition::NeedsRepair,
438                        vec![format!("behind_master:{behind}"), "fork_head".into()],
439                        Some(format!(
440                            "the author merges {} into the fork's branch",
441                            policy.base
442                        )),
443                    )
444                } else if let RelationToMaster::Behind { behind, .. } = relation {
445                    (
446                            PullRequestDisposition::NeedsRefresh,
447                            vec![format!("behind_master:{behind}")],
448                            Some(format!(
449                                "bring {} in with the derived merge driver, derive, push; CI runs again",
450                                policy.base
451                            )),
452                        )
453                } else {
454                    match checks {
455                        RequiredCheckState::Passed => (
456                            PullRequestDisposition::Ready,
457                            vec!["contains_master".into(), "required_checks_passed".into()],
458                            Some("merge it".into()),
459                        ),
460                        RequiredCheckState::Pending | RequiredCheckState::Missing => (
461                            PullRequestDisposition::WaitingForChecks,
462                            vec![format!("required_checks:{}", word(&checks))],
463                            Some("wait for the required checks on this head".into()),
464                        ),
465                        RequiredCheckState::Unknown => (
466                            PullRequestDisposition::Unknown,
467                            vec!["required_checks_unread".into()],
468                            Some("majordomus prs refresh".into()),
469                        ),
470                        RequiredCheckState::Failed => unreachable!("handled above"),
471                    }
472                }
473            }
474        }
475    };
476    if disposition == PullRequestDisposition::Draft
477        || disposition == PullRequestDisposition::Blocked
478    {
479        for l in &blocking {
480            evidence.push(ev("label", "blocking", (*l).clone()));
481        }
482    }
483
484    PullRequestAssessment {
485        number: pr.number,
486        title: pr.title.clone(),
487        author: pr.author.clone(),
488        head_ref: pr.head_ref.clone(),
489        base_ref: pr.base_ref.clone(),
490        evaluated_against: EvaluatedAgainst {
491            master_sha: master_sha.to_string(),
492            head_sha: pr.head_sha.clone(),
493        },
494        lane: disposition.lane(),
495        disposition,
496        reasons,
497        next_action: next,
498        required_checks: checks,
499        review,
500        relation: relation.clone(),
501        dependencies,
502        overlaps: overlaps(pr.number, &queue.authored),
503        authored_paths: authored,
504        risk,
505        risk_factors,
506        evidence,
507        created_at: pr.created_at.clone(),
508        // the audit trail's to say, not the observation's: queue_of adds it
509        wait: None,
510    }
511}

513/// The serialised word of any unit enum here.
514pub fn word<T: Serialize>(value: &T) -> String {
515    match serde_json::to_value(value) {
516        Ok(serde_json::Value::String(s)) => s,
517        _ => "unknown".into(),
518    }
519}
520
521fn relation_word(r: &RelationToMaster) -> &'static str {
522    match r {
523        RelationToMaster::Contained => "contained",
524        RelationToMaster::Superseded => "superseded",
525        RelationToMaster::DerivedOnly { .. } => "derived_only",
526        RelationToMaster::UpToDate { .. } => "up_to_date",
527        RelationToMaster::Behind { .. } => "behind",
528        RelationToMaster::Conflicting { .. } => "conflicting",
529        RelationToMaster::Unknown { .. } => "unknown",
530    }
531}
532
533fn relation_detail(r: &RelationToMaster) -> String {
534    match r {
535        RelationToMaster::Contained => "the head is an ancestor of master".into(),
536        RelationToMaster::Superseded => "merging it into master changes no file".into(),
537        RelationToMaster::DerivedOnly { paths } => {
538            format!("only {} derived path(s) would change", paths.len())
539        }
540        RelationToMaster::UpToDate { authored } => {
541            format!("contains master; {} authored path(s)", authored.len())
542        }
543        RelationToMaster::Behind { behind, authored } => format!(
544            "{behind} master commit(s) behind; merges cleanly; {} authored path(s)",
545            authored.len()
546        ),
547        RelationToMaster::Conflicting { paths } => format!("conflicts on {}", paths.join(", ")),
548        RelationToMaster::Unknown { reason } => reason.clone(),
549    }
