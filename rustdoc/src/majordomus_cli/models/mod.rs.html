1//! The model catalogue: which AI models exist for this tool's world, what each can do,
2//! and which one a stated need selects — as declared data, never as a live client. ADR
3//! 0044 is the decision; ADR 0032's boundary stands untouched: this crate calls no
4//! model, holds no SDK, and reads no credential's value. The catalogue is
5//! `share/models.yaml` — one declaration, exactly like `share/providers.yaml` — and
6//! `models.list` / `models.route` project it to the CLI, HTTP, OpenAPI, MCP and the
7//! Cockpit, so no consumer keeps a model name of its own.
8//!
9//! Two words are deliberately absent. "Provider" is spent (ADR 0024, 0032): here the
10//! party that serves a model is a *vendor*. And "routing" here is a pure, explainable
11//! function over the declared data — a decision with the reasons attached, never an
12//! opaque dispatcher: ask `models.route` why, and every excluded model answers.
13//!
14//! ```
15//! use majordomus_cli::models::{route, ModelCatalogue, Requirements};
16//!
17//! let catalogue: ModelCatalogue = majordomus_cli::metadata::yaml::parse_into(
18//!     "version: 1\nvendors:\n  - id: acme\n    title: Acme\nmodels:\n  - id: acme-large\n    vendor: acme\n    native_id: acme-large-1\n    context_window: 200000\n    capabilities: [text, tools, vision]\n  - id: acme-mini\n    vendor: acme\n    native_id: acme-mini-1\n    context_window: 32000\n    capabilities: [text]\n",
19//! ).unwrap();
20//! let decision = route(&catalogue, &Requirements { require: vec!["vision".into()], ..Default::default() });
21//! assert_eq!(decision.selected.as_deref(), Some("acme-large"));
22//! assert_eq!(decision.excluded[0].reason, "missing capability: vision");
23//! ```

31/// A party that serves models: a remote API vendor or a local runtime. Not a
32/// "provider" — that word names the AI client tools this repository already adapts to
33/// (ADR 0024), and one word for two registries is how truths fork.
34#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
35#[serde(deny_unknown_fields)]
36pub struct Vendor {
37    /// The vendor id, unique in the catalogue.
38    pub id: String,
39    /// The human name.
40    pub title: String,
41    /// Where inference runs: `remote` (an API over the network) or `local` (a runtime
42    /// on this machine). Typed here so routing's `local_only` is data, not a name test.
43    #[serde(default)]
44    pub inference: Inference,
45    /// The environment variable whose *presence* means a credential is configured.
46    /// Only the name lives here, and only presence is ever reported — no code in this
47    /// crate reads the value, and the environment cache refuses to hold one.
48    #[serde(default, skip_serializing_if = "Option::is_none")]
49    pub credential_env: Option<String>,
50    /// How a call to this vendor can be paid: the sources the vendor offers, each with
51    /// the evidence it was read from. What the vendor *offers* is the distribution's;
52    /// which of them the owner *holds*, and any amount, is the owner's and never here.
53    #[serde(default, skip_serializing_if = "Vec::is_empty")]
54    pub funding: Vec<FundingSource>,
55}

57/// One way a vendor lets a call be paid. A kind and a credential, never an amount: a
58/// price or a balance this tool cannot verify is a fact it must not state.
59///
60/// ```
61/// use majordomus_cli::models::{FundingKind, FundingSource, ModelCatalogue};
62///
63/// let c = ModelCatalogue::parse(concat!(
64///     "version: 1\nvendors:\n  - id: acme\n    title: Acme\n    funding:\n",
65///     "      - id: grant\n        kind: credit_grant\n        credential_env: ACME_API_KEY\n",
66///     "        renews: monthly\n        claim_url: https://acme.example/claim\n",
67///     "        evidence:\n          url: https://acme.example/docs\n          checked: 2026-10-21\n",
68/// )).unwrap();
69/// let grant: &FundingSource = &c.vendors[0].funding[0];
70/// assert_eq!(grant.kind, FundingKind::CreditGrant);
71/// assert_eq!(grant.credential_env.as_deref(), Some("ACME_API_KEY"));
72/// assert!(c.diagnostics().is_empty());
73/// ```
74#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
75#[serde(deny_unknown_fields)]
76pub struct FundingSource {
77    /// The source id, unique within its vendor.
78    pub id: String,
79    /// What kind of source it is.
80    pub kind: FundingKind,
81    /// The environment variable whose presence reaches this source. Two sources may
82    /// name the same one: a credit grant and metered billing are both reached through
83    /// one API credential, and nothing local tells them apart.
84    #[serde(default, skip_serializing_if = "Option::is_none")]
85    pub credential_env: Option<String>,
86    /// The period an allowance renews over, for a source that has one.
87    #[serde(default, skip_serializing_if = "Option::is_none")]
88    pub renews: Option<Renewal>,
89    /// Where the owner claims or links the source, as the vendor published it. Shown as
90    /// evidence; Majordomus never follows it.
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    pub claim_url: Option<String>,
93    /// What the declaration stands on.
94    pub evidence: FundingEvidence,
95    /// One line a person needs when choosing.
96    #[serde(default, skip_serializing_if = "Option::is_none")]
97    pub note: Option<String>,
98}

100/// The kinds of funding source, in the order a policy most often prefers them.
101///
102/// ```
103/// use majordomus_cli::models::FundingKind;
104///
105/// let kind: FundingKind = serde_json::from_str(r#""credit_grant""#).unwrap();
106/// assert_eq!(kind, FundingKind::CreditGrant);
107/// assert!(serde_json::from_str::<FundingKind>(r#""voucher""#).is_err(), "the set is closed");
108/// ```
109#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
110#[serde(rename_all = "snake_case")]
111pub enum FundingKind {
112    /// Inference on this machine; no money changes hands.
113    Local,
114    /// A recurring allowance of API usage that comes with something else the owner
115    /// pays for, such as the API credits a subscription plan includes.
116    CreditGrant,
117    /// A subscription seat, used through the client signed in with it.
118    Subscription,
119    /// Pay-as-you-go usage, billed by what was used.
120    Metered,
121}

173/// A model's lifecycle standing, as the catalogue declares it. There is no default: a
174/// catalogue entry that declares none has an undeclared standing, which is not the same
175/// fact as "available", and no reader may render it as one.
176#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
177#[serde(rename_all = "snake_case")]
178pub enum ModelStatus {
179    /// Generally available.
180    Available,
181    /// Usable, expected to change.
182    Preview,
183    /// Still served, no longer chosen by routing unless named explicitly.
184    Deprecated,
185    /// Gone; kept so an old record's reference still resolves to a row with a story.
186    Retired,
187}

189/// One model: the canonical reference this repository uses, the vendor's own name for
190/// it, and its typed capabilities. Capability words are data the catalogue declares —
191/// consumers compare against a requirement list, never against a hardcoded name.
192#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
193#[serde(deny_unknown_fields)]
194pub struct ModelEntry {
195    /// The canonical id, unique in the catalogue: what sessions, configuration and
196    /// routing name. Stable even when a vendor renames.
197    pub id: String,
198    /// The vendor that serves it.
199    pub vendor: String,
200    /// The vendor's own identifier, verbatim.
201    pub native_id: String,
202    /// Other names that resolve to this model. Aliases live here and nowhere else.
203    #[serde(default, skip_serializing_if = "Vec::is_empty")]
204    pub aliases: Vec<String>,
205    /// Typed capability words: `text`, `vision`, `tools`, `structured_output`,
206    /// `reasoning`, `streaming`, `embeddings` — the declared vocabulary, open for a
207    /// catalogue to extend without a code change.
208    #[serde(default)]
209    pub capabilities: Vec<String>,
210    /// The context window, tokens.
211    #[serde(default, skip_serializing_if = "Option::is_none")]
212    pub context_window: Option<u64>,
213    /// The lifecycle standing, when the catalogue declares one. Absent is undeclared: the
214    /// catalogue admits only facts with provenance, and routing still considers the model,
215    /// but nothing may say it is available on the catalogue's behalf.
216    #[serde(default, skip_serializing_if = "Option::is_none")]
217    pub status: Option<ModelStatus>,
218    /// One line a person needs when choosing; never pricing — a price the tool cannot
219    /// verify is a fact it must not state (nothing here fabricates cost data).
220    #[serde(default, skip_serializing_if = "Option::is_none")]
221    pub note: Option<String>,
222}

224/// The whole catalogue, as `share/models.yaml` declares it.
225#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
226#[serde(deny_unknown_fields)]
227pub struct ModelCatalogue {
228    /// The format version.
229    pub version: u32,
230    /// The vendors, in declaration order.
231    #[serde(default)]
232    pub vendors: Vec<Vendor>,
233    /// The models, in declaration order — which is also routing's preference order:
234    /// the first satisfying model wins, and "why this one" is "it came first among
235    /// those that qualify", an answer a person can verify by reading the file.
236    #[serde(default)]
237    pub models: Vec<ModelEntry>,
238    /// Funding sources [`ModelCatalogue::parse`] could not read, each naming its vendor
239    /// and source. They are set aside rather than failing the file: the advisor catalogue
240    /// loads the same file, and one bad source must not take every advisor down.
241    #[serde(skip)]
242    pub unread_funding: Vec<String>,
243}

245impl ModelCatalogue {
246    /// Load the catalogue from a share directory. No file is an empty catalogue — a
247    /// distribution that declares no models has none, which is an answer; a file that
248    /// does not parse is an error naming it.
249    pub fn load(share: &std::path::Path) -> Result<ModelCatalogue, String> {
250        let path = share.join(MODELS_FILE);
251        match std::fs::read_to_string(&path) {
252            Ok(text) => ModelCatalogue::parse(&text)
253                .map_err(|reason| format!("{}: {reason}", path.display())),
254            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(ModelCatalogue::default()),
255            Err(e) => Err(format!("{}: {e}", path.display())),
256        }
257    }
258
259    /// Parse a catalogue. A funding source without evidence, of a kind nobody declared,
260    /// or otherwise unreadable is set aside and reported by [`ModelCatalogue::diagnostics`]
261    /// naming its vendor and its id — a typed parse alone would fail the whole file and
262    /// say only that a field is missing somewhere.
263    ///
264    /// ```
265    /// use majordomus_cli::models::ModelCatalogue;
266    ///
267    /// let c = ModelCatalogue::parse(concat!(
268    ///     "version: 1\nvendors:\n  - id: acme\n    title: Acme\n    funding:\n",
269    ///     "      - id: grant\n        kind: credit_grant\n",
270    ///     "models: []\n",
271    /// )).unwrap();
272    /// assert!(c.vendors[0].funding.is_empty(), "the unreadable source is set aside");
273    /// assert!(c.diagnostics()[0].contains("vendor 'acme' funding source 'grant' declares no evidence"));
274    /// ```
275    pub fn parse(text: &str) -> Result<ModelCatalogue, String> {
276        let mut map = crate::metadata::yaml::parse_mapping(text)?;
277        let unread = set_aside_unreadable_funding(&mut map);
278        let mut catalogue: ModelCatalogue =
279            serde_json::from_value(serde_json::Value::Object(map)).map_err(|e| e.to_string())?;
280        catalogue.unread_funding = unread;
281        Ok(catalogue)
282    }
283
284    /// Every credential variable the catalogue names, the vendors' and their funding
285    /// sources', once each in declaration order: what no CI gate may depend on.
286    ///
287    /// ```
288    /// use majordomus_cli::models::ModelCatalogue;
289    ///
290    /// let c = ModelCatalogue::parse(concat!(
291    ///     "version: 1\nvendors:\n  - id: acme\n    title: Acme\n    credential_env: ACME_KEY\n",
292    ///     "    funding:\n      - id: seat\n        kind: subscription\n        credential_env: ACME_SEAT\n",
293    ///     "        evidence:\n          url: https://acme.example/docs\n          checked: 2026-10-21\n",
294    ///     "      - id: metered\n        kind: metered\n        credential_env: ACME_KEY\n",
295    ///     "        evidence:\n          url: https://acme.example/docs\n          checked: 2026-10-21\n",
296    /// )).unwrap();
297    /// assert_eq!(c.credential_variables(), ["ACME_KEY", "ACME_SEAT"]);
298    /// ```
299    pub fn credential_variables(&self) -> Vec<&str> {
300        let mut names: Vec<&str> = Vec::new();
301        for vendor in &self.vendors {
302            let sources = vendor
303                .funding
304                .iter()
305                .filter_map(|s| s.credential_env.as_deref());
306            for name in vendor.credential_env.as_deref().into_iter().chain(sources) {
307                if !names.contains(&name) {
308                    names.push(name);
309                }
310            }
311        }
312        names
313    }
314
315    /// The entry a canonical id or an alias resolves to.
316    pub fn resolve(&self, name: &str) -> Option<&ModelEntry> {
317        self.models
318            .iter()
319            .find(|m| m.id == name || m.aliases.iter().any(|a| a == name))
320    }
321
322    /// The declared vendor of an entry.
323    pub fn vendor_of(&self, entry: &ModelEntry) -> Option<&Vendor> {
324        self.vendors.iter().find(|v| v.id == entry.vendor)
325    }

327    /// The catalogue's own consistency: duplicate ids or aliases, and a model naming a
328    /// vendor the catalogue does not declare. Reported, never repaired.
329    pub fn diagnostics(&self) -> Vec<String> {
330        let mut findings = Vec::new();
331        let mut seen = std::collections::BTreeSet::new();
332        for model in &self.models {
333            for name in std::iter::once(&model.id).chain(model.aliases.iter()) {
334                if !seen.insert(name.clone()) {
335                    findings.push(format!("'{name}' names two models; an alias is one truth"));
336                }
337            }
338            if self.vendor_of(model).is_none() {
339                findings.push(format!(
340                    "model '{}' names vendor '{}', which the catalogue does not declare",
341                    model.id, model.vendor
342                ));
343            }
344        }
345        findings.extend(self.unread_funding.iter().cloned());
346        for vendor in &self.vendors {
347            findings.extend(funding_findings(vendor));
348        }
349        findings
350    }
351}

353/// Take out of the raw mapping every funding source that cannot be read — no evidence,
354/// evidence without its page or day, a kind outside [`FundingKind`], or anything else the
355/// typed parse would refuse — and say why, naming the vendor and the source. Done before
356/// the typed parse, so that the rest of the catalogue still loads and the reason names
357/// what it means.
358fn set_aside_unreadable_funding(
359    map: &mut serde_json::Map<String, serde_json::Value>,
360) -> Vec<String> {
361    let mut unread = Vec::new();
362    let vendors = map.get_mut("vendors").and_then(|v| v.as_array_mut());
363    for vendor in vendors.into_iter().flatten() {
364        let vendor_id = vendor
365            .get("id")
366            .and_then(|v| v.as_str())
367            .unwrap_or("?")
368            .to_string();
369        let Some(sources) = vendor.get_mut("funding").and_then(|f| f.as_array_mut()) else {
370            continue;
371        };
372        let mut i = 0;
373        sources.retain(|source| {
374            i += 1;
375            let named = match source.get("id").and_then(|v| v.as_str()) {
376                Some(id) => format!("vendor '{vendor_id}' funding source '{id}'"),
377                None => format!("vendor '{vendor_id}' funding source #{i}"),
378            };
379            let reasons = funding_unreadable(source);
380            for reason in &reasons {
381                unread.push(format!("{named} {reason}; it is set aside"));
382            }
383            reasons.is_empty()
384        });
385    }
386    unread
387}

389/// Why one raw funding source cannot be read; empty when it can.
390fn funding_unreadable(source: &serde_json::Value) -> Vec<String> {
391    let mut reasons = Vec::new();
392    match source.get("kind") {
393        None => reasons.push("declares no kind".to_string()),
394        Some(kind) if serde_json::from_value::<FundingKind>(kind.clone()).is_err() => {
395            reasons.push(format!("declares kind {kind}, which is not a funding kind"));
396        }
397        Some(_) => {}
398    }
399    match source.get("evidence") {
400        None => reasons.push("declares no evidence".to_string()),
401        Some(evidence) => {
402            for field in ["url", "checked"] {
403                let present = evidence
404                    .get(field)
405                    .and_then(|v| v.as_str())
406                    .is_some_and(|s| !s.trim().is_empty());
407                if !present {
408                    reasons.push(format!("has evidence without its {field}"));
409                }
410            }
411        }
412    }
413    if reasons.is_empty() {
414        if let Err(e) = serde_json::from_value::<FundingSource>(source.clone()) {
415            reasons.push(format!("cannot be read: {e}"));
416        }
417    }
418    reasons
419}

421/// A vendor's funding declarations that parse but do not agree with themselves or with
422/// the vendor: reported, never repaired.
423fn funding_findings(vendor: &Vendor) -> Vec<String> {
424    let mut findings = Vec::new();
425    let mut seen = std::collections::BTreeSet::new();
426    for source in &vendor.funding {
427        let named = format!("vendor '{}' funding source '{}'", vendor.id, source.id);
428        if !seen.insert(source.id.as_str()) {
429            findings.push(format!("{named} is declared twice"));
430        }
431        let local_vendor = vendor.inference == Inference::Local;
432        if local_vendor != (source.kind == FundingKind::Local) {
433            findings.push(format!(
434                "{named} is of kind {:?} on a vendor whose inference is {:?}",
435                source.kind, vendor.inference
436            ));
437        }
438        if source.kind == FundingKind::CreditGrant && source.renews.is_none() {
439            findings.push(format!(
440                "{named} is a credit grant that declares no renewal period"
441            ));
442        }
443        if source.kind != FundingKind::Local && source.credential_env.is_none() {
444            findings.push(format!(
445                "{named} names no credential variable that reaches it"
446            ));
447        }
448        if !is_iso_day(&source.evidence.checked) {
449            findings.push(format!(
450                "{named} has evidence checked '{}', which is not a YYYY-MM-DD day",
451                source.evidence.checked
452            ));
453        }
454    }
455    findings
456}

458fn is_iso_day(text: &str) -> bool {
459    let b = text.as_bytes();
460    b.len() == 10
461        && b[4] == b'-'
462        && b[7] == b'-'
463        && b.iter()
464            .enumerate()
465            .all(|(i, c)| i == 4 || i == 7 || c.is_ascii_digit())
466}

498/// The routing answer: what was selected, what stands behind it, who fell out and why.
499/// The whole decision is derived from the catalogue and the requirements — run it
500/// twice, get it twice; there is no state to consult and no health to guess.
501#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
502pub struct RoutingDecision {
503    /// The selected model's canonical id; `None` when nothing qualifies.
504    #[serde(default, skip_serializing_if = "Option::is_none")]
505    pub selected: Option<String>,
506    /// Why the selected model was selected.
507    #[serde(default, skip_serializing_if = "Option::is_none")]
508    pub reason: Option<String>,
509    /// The models that also qualify, in preference order: the fallback chain.
510    #[serde(default, skip_serializing_if = "Vec::is_empty")]
511    pub fallbacks: Vec<String>,
512    /// Every model that did not qualify, with its reason.
513    #[serde(default, skip_serializing_if = "Vec::is_empty")]
514    pub excluded: Vec<ExcludedModel>,
515}

517/// Decide. Pure over the catalogue and the requirements; the preference order is the
518/// catalogue's declaration order, and every exclusion names its first failing check.
519pub fn route(catalogue: &ModelCatalogue, needs: &Requirements) -> RoutingDecision {
520    let mut qualified: Vec<&ModelEntry> = Vec::new();
521    let mut excluded = Vec::new();
522    for model in &catalogue.models {
523        match disqualify(catalogue, model, needs) {
524            Some(reason) => excluded.push(ExcludedModel {
525                model: model.id.clone(),
526                reason,
527            }),
528            None => qualified.push(model),
529        }
530    }
531    let selected = qualified.first().map(|m| m.id.clone());
532    let reason = selected.as_ref().map(|id| match &needs.model {
533        Some(named) => format!("named outright as '{named}' and it satisfies every requirement"),
534        None => format!("'{id}' is the first declared model satisfying every requirement"),
535    });
536    RoutingDecision {
537        selected,
538        reason,
539        fallbacks: qualified.iter().skip(1).map(|m| m.id.clone()).collect(),
540        excluded,
541    }
542}

544/// The first reason a model does not qualify, in a fixed check order; `None` qualifies.
545fn disqualify(
546    catalogue: &ModelCatalogue,
547    model: &ModelEntry,
548    needs: &Requirements,
549) -> Option<String> {
550    if let Some(named) = &needs.model {
551        let resolves = model.id == *named || model.aliases.iter().any(|a| a == named);
552        if !resolves {
553            return Some(format!("not the model named outright ('{named}')"));
554        }
555    }
556    if let Some(status @ (ModelStatus::Deprecated | ModelStatus::Retired)) = model.status {
557        if needs.model.is_none() {
558            return Some(format!(
559                "status is {}; routing only selects it when named outright",
560                match status {
561                    ModelStatus::Deprecated => "deprecated",
562                    _ => "retired",
563                }
564            ));
565        }
566    }
567    if let Some(vendor) = &needs.vendor {
568        if model.vendor != *vendor {
569            return Some(format!("vendor is '{}', not '{vendor}'", model.vendor));
570        }
571    }
572    if needs.local_only {
573        let local = catalogue
574            .vendor_of(model)
575            .is_some_and(|v| v.inference == Inference::Local);
576        if !local {
577            return Some("inference is remote and the need is local_only".into());
578        }
579    }
580    for capability in &needs.require {
581        if !model.capabilities.iter().any(|c| c == capability) {
582            return Some(format!("missing capability: {capability}"));
583        }
584    }
585    if let Some(least) = needs.min_context {
586        match model.context_window {
587            Some(window) if window >= least => {}
588            Some(window) => {
589                return Some(format!(
590                    "context window {window} is under the required {least}"
591                ))
592            }
593            None => return Some(format!("no declared context window; {least} required")),
594        }
595    }
596    None
597}

976    #[test]
977    fn no_field_of_the_catalogue_can_hold_a_secret_value() {
978        // Proves `models-no-secret-fields`: the catalogue reports whether a vendor's
979        // named credential variable is set and has no field a value could hide in.
980        // The vendor declares the *name* of a credential's environment variable; the
981        // schema has no field for a value, so a leak would need a new field to exist.
982        // Property names are what is checked — prose may say "tokens" all it likes.
983        fn property_names(value: &serde_json::Value, out: &mut Vec<String>) {
984            if let Some(properties) = value.get("properties").and_then(|p| p.as_object()) {
985                for (name, nested) in properties {
986                    out.push(name.to_lowercase());
987                    property_names(nested, out);
988                }
989            }
990            for key in ["items", "$defs", "definitions"] {
991                if let Some(nested) = value.get(key) {
992                    if let Some(map) = nested.as_object() {
993                        for entry in map.values() {
994                            property_names(entry, out);
995                        }
996                    } else {
997                        property_names(nested, out);
998                    }
999                }
1000            }
1001        }
1002        let schema = serde_json::to_value(schemars::schema_for!(ModelCatalogue)).unwrap();
1003        let mut names = Vec::new();
1004        property_names(&schema, &mut names);
1005        assert!(
1006            names.iter().any(|n| n == "credential_env"),
1007            "the walk sees the fields"
1008        );
1009        for name in names {
1010            for word in ["secret", "token", "password", "key_value", "bearer"] {
1011                assert!(
1012                    !name.contains(word),
1013                    "the catalogue schema declares a field that smells like a credential: {name}"
1014                );
1015            }
1016        }
1017    }
