1//! The socket: bind, read requests, hand them to the router, write responses. Loopback by
2//! default; the bound address is reported so that a caller who asked for port 0, or whose
3//! port was taken, learns the one in use. A bound socket is accepted by a few worker
4//! threads, and each request is answered on a thread of its own up to a bound, so a slow
5//! request queues nothing behind it — not even the probe a client sends before it would
6//! take the lease over — and the owner's stdio session never waits on HTTP. Stopping is cooperative: every
7//! worker is unblocked and joined, and an in-flight response is finished first. A stopped
8//! server refuses the next connection: `stop` returns after the listening socket is closed. A connection
9//! that reads no request within the declared deadline is closed ([`super::deadline`]), and a
10//! request that finds every handler busy is refused rather than answered on a worker.
11

24/// The largest request body accepted.
25pub const MAX_BODY_BYTES: usize = 1024 * 1024;
26
27/// How many threads accept requests on one socket. They only accept: each request is answered
28/// on a thread of its own (up to [`MAX_HANDLERS`]), so a worker is free again at once and a
29/// lease probe is answered while tool calls run (I2126).
30pub const WORKERS: usize = 4;
31
32/// How many requests are answered at once. A bound, so that a flood costs a fixed number of
33/// threads; far above what clients of one checkout send, so that a few slow tool calls never
34/// hold the probe up. Past it a request is answered 503, except the probe (`GET /`), which a
35/// worker answers itself: it is cheap, and it is what decides whether this server keeps its
36/// lease (I2156).
37pub const MAX_HANDLERS: usize = 64;
38
39/// Requests being answered right now, on threads of their own.
40static HANDLERS: std::sync::atomic::AtomicUsize = std::sync::atomic::AtomicUsize::new(0);
41
42/// When this process last answered a request, in milliseconds since this process started;
43/// zero before the first. Read by the idle timer, so that a person reading the Cockpit or a
44/// client calling the REST API keeps the server they are using (I2131).
45static LAST_REQUEST_MS: AtomicU64 = AtomicU64::new(0);
46
47/// Live `/events` channels open right now: a person watching an execution is using the server
48/// for as long as the socket stays open, whether or not they send anything.
49static OPEN_CHANNELS: AtomicUsize = AtomicUsize::new(0);
50
51fn epoch() -> std::time::Instant {
52    static START: std::sync::OnceLock<std::time::Instant> = std::sync::OnceLock::new();
53    *START.get_or_init(std::time::Instant::now)
54}
55
56fn note_request() {
57    let ms = epoch().elapsed().as_millis() as u64;
58    LAST_REQUEST_MS.store(ms.max(1), Ordering::Relaxed);
59}

61/// How many live `/events` channels are open right now.
62///
63/// ```
64/// // a process that opened no channel holds none
65/// assert_eq!(majordomus_cli::http::server::open_channels(), 0);
66/// ```
67pub fn open_channels() -> usize {
68    OPEN_CHANNELS.load(Ordering::Relaxed)
69}
70
71/// Whether this process was used over HTTP within the last `window`: a request a client sent
72/// answered in it (machine traffic is not counted: [`crate::idle::counts_as_use`]), or a live
73/// channel open now. The idle timer asks this beside the MCP sessions.
74///
75/// ```
76/// use std::time::Duration;
77/// use majordomus_cli::http::server::used_within;
78/// // a process that has answered nothing over HTTP and holds no channel has not been used
79/// assert!(!used_within(Duration::from_secs(60)));
80/// ```
81pub fn used_within(window: std::time::Duration) -> bool {
82    if OPEN_CHANNELS.load(Ordering::Relaxed) > 0 {
83        return true;
84    }
85    let last = LAST_REQUEST_MS.load(Ordering::Relaxed);
86    last != 0 && epoch().elapsed().as_millis() as u64 <= last + window.as_millis() as u64
87}

95/// Bind `host:port`; `0` picks a free port. The error names the address when it fails.
96pub fn bind(host: &str, port: u16) -> Result<Bound> {
97    let server = Arc::new(Server::http((host, port)).map_err(|e| Error::Http {
98        reason: format!("cannot bind {host}:{port}: {e}"),
99    })?);
100    let address = server
101        .server_addr()
102        .to_ip()
103        .map(|a| a.to_string())
104        .unwrap_or_else(|| format!("{host}:{port}"));
105    if address
106        .parse::<std::net::SocketAddr>()
107        .is_ok_and(|a| !a.ip().is_loopback())
108    {
109        tracing::warn!(
110            address = %address,
111            "bound to {address}, which is not a loopback address: every host that can reach this interface can read this repository's AI layer, its diagnostics and its peers, unencrypted; it can change nothing but through a signed mesh message (ADR 0126); bind 127.0.0.1 unless that is intended"
112        );
113    }
114    Ok(Bound { server, address })
115}

117/// Bind the address a deployment object declared. The accidental-bind warning does not
118/// fire here and is not suppressed either: it asks "did you mean this", and `source` is the
119/// answer — the canonical object that said so, named in the log so the operator can read
120/// which file decided the address. A hosted process that bound loopback would be
121/// unreachable inside its own machine, which is why this path exists at all.
122pub fn bind_declared(host: &str, port: u16, source: &str) -> Result<Bound> {
123    let server = Arc::new(Server::http((host, port)).map_err(|e| Error::Http {
124        reason: format!("cannot bind {host}:{port}: {e}"),
125    })?);
126    let address = server
127        .server_addr()
128        .to_ip()
129        .map(|a| a.to_string())
130        .unwrap_or_else(|| format!("{host}:{port}"));
131    tracing::info!(
132        address = %address,
133        source = %source,
134        "bound {address}, the address {source} declares; this repository's AI layer is readable by every host that can reach this interface, which is what a deployment means"
135    );
136    Ok(Bound { server, address })
137}

139/// Bind `host:port`, and when that port is taken bind a free one instead, saying so on
140/// stderr. For a server whose port is a convenience rather than a contract.
141pub fn bind_or_fallback(host: &str, port: u16) -> Result<Bound> {
142    match bind(host, port) {
143        Ok(b) => Ok(b),
144        Err(e) if port != 0 => {
145            tracing::warn!("{e}; taking a free port instead");
146            bind(host, 0)
147        }
148        Err(e) => Err(e),
149    }
150}

163    /// Start [`WORKERS`] threads answering requests through `router`, and the sweeper that
164    /// holds their connections to the declared read deadline.
165    pub fn start(self, router: Router) -> Running {
166        let stopping = Arc::new(AtomicBool::new(false));
167        let local = self
168            .address
169            .parse()
170            .unwrap_or(std::net::SocketAddr::from(([0, 0, 0, 0], 0)));
171        let watch = Arc::new(Watch::new(local, crate::lease::timings().read_deadline));
172        let mut threads: Vec<JoinHandle<()>> = (0..WORKERS)
173            .map(|n| {
174                let server = Arc::clone(&self.server);
175                let router = router.clone();
176                let stopping = Arc::clone(&stopping);
177                let watch = Arc::clone(&watch);
178                std::thread::Builder::new()
179                    .name(format!("http-{n}"))
180                    .spawn(move || worker(&server, &router, &watch, &stopping))
181                    .expect("spawn an http worker")
182            })
183            .collect();
184        let sweeper = {
185            let stopping = Arc::clone(&stopping);
186            std::thread::Builder::new()
187                .name("http-deadline".into())
188                .spawn(move || watch.run(&stopping))
189                .expect("spawn the http deadline sweeper")
190        };
191        threads.push(sweeper);
192        Running {
193            server: self.server,
194            address: self.address,
195            stopping,
196            threads,
197        }
198    }
199}

201/// A served socket.
202pub struct Running {
203    server: Arc<Server>,
204    address: String,
205    stopping: Arc<AtomicBool>,
206    threads: Vec<JoinHandle<()>>,
207}

209impl Running {
210    /// `host:port` as bound.
211    pub fn address(&self) -> &str {
212        &self.address
213    }
214
215    /// `http://host:port`.
216    pub fn url(&self) -> String {
217        format!("http://{}", self.address)
218    }
219
220    /// Unblock every worker and wait for it; an in-flight response is finished first. When
221    /// this returns the address refuses a connection: the next caller learns that nothing was
222    /// sent, not that an answer was lost.
223    pub fn stop(self) {
224        self.stopping.store(true, Ordering::SeqCst);
225        // one more than there are workers is harmless; the sweeper reads `stopping` itself
226        for _ in &self.threads {
227            self.server.unblock();
228        }
229        for t in self.threads {
230            let _ = t.join();
231        }
232        // the requests already handed off are answered before the process goes on to stop,
233        // within a bound: a handler stuck forever does not hold the stop forever
234        let deadline = std::time::Instant::now() + std::time::Duration::from_secs(30);
235        while HANDLERS.load(Ordering::SeqCst) > 0 && std::time::Instant::now() < deadline {
236            std::thread::sleep(std::time::Duration::from_millis(20));
237        }
238        drop(self.server);
239        closed(&self.address);
240        tracing::info!(address = %self.address, "http stopped");
241    }
242}
243
244/// How long [`Running::stop`] waits for the listening socket to close.
245const CLOSE_GRACE: std::time::Duration = std::time::Duration::from_secs(2);
246
247/// Wait until `address` refuses a connection, within [`CLOSE_GRACE`].
248///
249/// Dropping the server only asks its accept thread to end: the listening socket is closed by
250/// that thread, some time after the drop returns. Until then the kernel still completes a
251/// connection nobody will read, and the caller is reset after it has sent its request, so it
252/// cannot tell whether the request ran. A stopped server is one that refuses, and `stop`
253/// returns only when this one does. Another process bound to the same port keeps answering;
254/// the bound is what ends the wait then.
255fn closed(address: &str) {
256    let Ok(address) = address.parse::<std::net::SocketAddr>() else {
257        return;
258    };
259    let deadline = std::time::Instant::now() + CLOSE_GRACE;
260    let attempt = std::time::Duration::from_millis(100);
261    while std::time::Instant::now() < deadline {
262        if std::net::TcpStream::connect_timeout(&address, attempt).is_err() {
263            return;
264        }
265        std::thread::sleep(std::time::Duration::from_millis(5));
266    }
267    tracing::warn!(%address, "the listening socket still accepts after the server stopped");
268}

270fn worker(server: &Server, router: &Router, watch: &Arc<Watch>, stopping: &AtomicBool) {
271    loop {
272        match server.recv() {
273            Ok(request) => hand_off(router, watch, request),
274            Err(_) if stopping.load(Ordering::SeqCst) => break,
275            Err(e) => tracing::warn!("accepting a connection failed: {e}"),
276        }
277    }
278}

280/// Answer `request` on a thread of its own while fewer than [`MAX_HANDLERS`] are, so that the
281/// worker accepts the next request at once. Past the bound the probe is answered here and
282/// anything else is refused 503: a worker that read a slow body itself would be held by it.
283fn hand_off(router: &Router, watch: &Arc<Watch>, request: tiny_http::Request) {
284    if HANDLERS.fetch_add(1, Ordering::SeqCst) >= MAX_HANDLERS {
285        HANDLERS.fetch_sub(1, Ordering::SeqCst);
286        let probe = matches!(
287            request.method(),
288            tiny_http::Method::Get | tiny_http::Method::Head
289        ) && request.url() == "/";
290        if probe {
291            answer(router, watch, request);
292        } else {
293            super::deadline::refused();
294            watch.stop_reading(request.remote_addr());
295            let busy = super::router::Response::error(
296                503,
297                "busy",
298                &format!("{MAX_HANDLERS} requests are being answered; ask again"),
299            )
300            .with_header("Retry-After", "1");
301            respond(request, busy, false);
302        }
303        return;
304    }
305    let router = router.clone();
306    let watch = Arc::clone(watch);
307    let spawned = std::thread::Builder::new()
308        .name("http-request".into())
309        .spawn(move || {
310            answer(&router, &watch, request);
311            HANDLERS.fetch_sub(1, Ordering::SeqCst);
312        });
313    released_unless_spawned(spawned, &HANDLERS);
314}
315
316/// A request whose thread could not be spawned went with the closure, and the client sees its
317/// connection close: give back the slot it took in `handlers`, and say so.
318fn released_unless_spawned(spawned: std::io::Result<JoinHandle<()>>, handlers: &AtomicUsize) {
319    if let Err(e) = spawned {
320        handlers.fetch_sub(1, Ordering::SeqCst);
321        tracing::warn!("a request could not be given a thread: {e}");
322    }
323}

325fn answer(router: &Router, watch: &Watch, request: tiny_http::Request) {
326    // another Majordomus process keeping the machine's servers in touch — a mesh link, a
327    // probe, a sibling's board gather — is not somebody using this one (I4007)
328    let probe = request
329        .headers()
330        .iter()
331        .any(|h| h.field.equiv(crate::idle::PROBE_HEADER));
332    if crate::idle::counts_as_use(&request.method().to_string(), request.url(), probe) {
333        note_request();
334    }
335    let peer = request.remote_addr().copied();
336    watch.reading(peer.as_ref());
337    if let Some(request) = upgrade(router, watch, request) {
338        answer_http(router, watch, request);
339    }
340    watch.answered(peer.as_ref());
341}

343/// Hand the socket to the live channel when the request asks for it, on a thread of its
344/// own so the worker goes back to answering requests.
345///
346/// Returns the request when it was not an upgrade, and nothing when the socket has been
347/// given away. This is the one thing the router cannot do for itself: `tiny_http` yields
348/// the stream only by consuming the request.
349fn upgrade(
350    router: &Router,
351    watch: &Watch,
352    request: tiny_http::Request,
353) -> Option<tiny_http::Request> {
354    let headers: Vec<(String, String)> = request
355        .headers()
356        .iter()
357        .map(|h| (h.field.as_str().to_string(), h.value.as_str().to_string()))
358        .collect();
359    let probe =
360        super::Request::parse_target(&request.method().to_string(), request.url(), Vec::new())
361            .with_headers(headers)
362            .with_remote(request.remote_addr().map(|a| a.ip()));
363    let accepted = match router.websocket(&probe) {
364        None => return Some(request),
365        Some(Ok(accepted)) => accepted,
366        Some(Err(response)) => {
367            watch.busy(request.remote_addr());
368            respond(request, response, false);
369            return None;
370        }
371    };
372    // a live channel is never closed by the read deadline: it waits by design
373    watch.live(request.remote_addr());
374    // and it ends when its client hangs up, not at its next write: an open channel keeps the
375    // server in use, so a closed one must stop counting within the idle timer's tick (I2131)
376    let hung_up = watch.hung_up(request.remote_addr());
377    let key = accepted.accept.clone();
378    let response = HttpResponse::empty(101).with_header(
379        Header::from_bytes("Sec-WebSocket-Accept", key.as_bytes())
380            .expect("the accept value is header-safe"),
381    );
382    let socket = request.upgrade("websocket", response);
383    let spawned = std::thread::Builder::new()
384        .name("events".into())
385        .spawn(move || {
386            OPEN_CHANNELS.fetch_add(1, Ordering::Relaxed);
387            accepted.serve_until(Box::new(socket), hung_up);
388            OPEN_CHANNELS.fetch_sub(1, Ordering::Relaxed);
389            note_request();
390        });
391    if let Err(e) = spawned {
392        tracing::warn!("a live channel could not be started: {e}");
393    }
394    None
395}

397fn answer_http(router: &Router, watch: &Watch, mut request: tiny_http::Request) {
398    let method = request.method().to_string();
399    let head = method == "HEAD";
400    let method = if head { "GET".to_string() } else { method };
401    let target = request.url().to_string();
402    let remote = request.remote_addr().map(|a| a.ip());
403    let headers: Vec<(String, String)> = request
404        .headers()
405        .iter()
406        .map(|h| (h.field.as_str().to_string(), h.value.as_str().to_string()))
407        .collect();
408    let mut body = Vec::new();
409    if let Err(e) = request
410        .as_reader()
411        .take(MAX_BODY_BYTES as u64 + 1)
412        .read_to_end(&mut body)
413    {
414        tracing::warn!("cannot read a request body: {e}");
415        return;
416    }
417    if !watch.busy(request.remote_addr()) {
418        // the deadline closed the connection under the body: what was read is short
419        return;
420    }
421    let response = if body.len() > MAX_BODY_BYTES {
422        // the rest of the body is never read, and dropping the request would read it
423        watch.stop_reading(request.remote_addr());
424        super::router::Response::error(
425            413,
426            "too_large",
427            &format!("the body is over {MAX_BODY_BYTES} bytes"),
428        )
429    } else {
430        router.handle(
431            &super::Request::parse_target(&method, &target, body)
432                .with_headers(headers)
433                .with_remote(remote),
434        )
435    };
436    // the target carries the query string, where a careless client puts a token
437    tracing::debug!(
438        method = %method,
439        target = %crate::redaction::redact_secrets(&target).text,
440        status = response.status,
441        "response"
442    );
443    respond(request, response, head);
444}

446/// Write one response and let the client go.
447fn respond(request: tiny_http::Request, response: super::router::Response, head: bool) {
448    // always Content-Length, never chunked: one less thing a small client must decode;
449    // a HEAD gets the GET's headers and no body
450    let (body, length) = wire_body(&response, head);
451    let mut out = HttpResponse::new(
452        response.status.into(),
453        Vec::new(),
454        std::io::Cursor::new(body),
455        Some(length),
456        None,
457    )
458    .with_chunked_threshold(usize::MAX)
459    .with_status_code(response.status)
460    .with_header(Header::from_bytes("Content-Type", response.content_type).expect("static header"));
461    // `no-store` is right for an answer derived from a repository that a person is editing;
462    // a response that named its own caching (a Cockpit asset whose URL carries its digest)
463    // keeps what it said
464    if !response
465        .headers
466        .iter()
467        .any(|(name, _)| name.eq_ignore_ascii_case("cache-control"))
468    {
469        out.add_header(Header::from_bytes("Cache-Control", "no-store").expect("static header"));
470    }
471    for (name, value) in &response.headers {
472        // the length is the server's to write, and `wire_body` has already read it
473        if name.eq_ignore_ascii_case("content-length") {
474            continue;
475        }
476        if let Ok(h) = Header::from_bytes(name.as_bytes(), value.as_bytes()) {
477            out.add_header(h);
478        }
479    }
480    if let Err(e) = request.respond(out) {
481        tracing::debug!("client went away before the response was written: {e}");
482    }
483}

485/// The bytes to write and the `Content-Length` to announce. A HEAD writes nothing. A
486/// `304 Not Modified` writes nothing either, and announces the length its producer names in
487/// a `Content-Length` header — the 200's, the only value RFC 9110 section 8.6 allows a 304
488/// to carry — or none of its own, an empty body's 0.
489fn wire_body(response: &super::router::Response, head: bool) -> (Vec<u8>, usize) {
490    if response.status == 304 {
491        let named = response
492            .headers
493            .iter()
494            .find(|(name, _)| name.eq_ignore_ascii_case("content-length"))
495            .and_then(|(_, value)| value.trim().parse().ok());
496        return (Vec::new(), named.unwrap_or(0));
497    }
498    let body: Vec<u8> = if head {
499        Vec::new()
500    } else {
501        response.body.as_bytes().to_vec()
502    };
503    let length = body.len();
504    (body, length)
505}

507/// Is stdin a pipe or a socket, as when a parent process spawned us and holds the other end?
508/// Asked of descriptor 0 itself (`fstat`), never of the path `/dev/stdin`: the path
509/// resolves through `/dev/fd`, which some harnesses (a coverage runner, a sandbox) do not
510/// expose, and a wrong "no" here sends `serve` into the run-forever branch while its
511/// parent waits for it to end.
512#[cfg(unix)]
513pub fn stdin_is_a_pipe() -> bool {
514    use std::os::fd::FromRawFd;
515    use std::os::unix::fs::FileTypeExt;
516    // SAFETY: descriptor 0 is the process's stdin for the process's lifetime; ManuallyDrop
517    // keeps this `File` from closing it.
518    let stdin = std::mem::ManuallyDrop::new(unsafe { std::fs::File::from_raw_fd(0) });
519    stdin
520        .metadata()
521        .map(|m| m.file_type().is_fifo() || m.file_type().is_socket())
522        .unwrap_or(false)
523}
524
525/// Is stdin a pipe or a socket? Not known on this platform.
526#[cfg(not(unix))]
527pub fn stdin_is_a_pipe() -> bool {
528    false
529}
