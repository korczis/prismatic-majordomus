1//! CI/CD & Releases: what was released, where it is deployed, and what the CI model runs.
2//! Read from `distribution.releases`, `release.version`, `deploy.list` and `gates.model`.
3//!
4//! How long CI takes is `integration.timings`, measured from the recorded forge observation
5//! (`super::timings`): the CI model says which gates run and when, the timings how long the
6//! runs on the open pull requests took.
7
8use crate::capability::Context;
9use crate::cockpit::html::el;
10use serde_json::Value;
11
12use super::dashboard::releases_panel;
13use super::shell::View;
14use super::source::{ask_all, Source};
15use super::view::{hero, kpi, list, panel, state, table, text, unanswered, wide};
16
17/// The route of the screen.
18pub const ROUTE: &str = "/ui/delivery";
19
20const RELEASES: Source = Source::of("distribution.releases");
21const VERSION: Source = Source::of("release.version");
22const DEPLOYMENTS: Source = Source::of("deploy.list");
23const GATES: Source = Source::of("gates.model");
24
25/// The CI/CD & Releases screen.
26pub fn screen(ctx: &Context) -> View {
27    let answers = ask_all(
28        ctx,
29        &[
30            RELEASES,
31            VERSION,
32            DEPLOYMENTS,
33            GATES,
34            super::timings::TIMINGS,
35        ],
36    );
37    let (releases, version, deployments, gates) = (
38        &answers[&RELEASES],
39        &answers[&VERSION],
40        &answers[&DEPLOYMENTS],
41        &answers[&GATES],
42    );
43
44    let top = hero(
45        "layers",
46        "orange",
47        "CI/CD & Releases",
48        "From a green check to a published release.",
49        "What was released, what the tree declares and what its commits imply next, where it is deployed, and every gate the CI model can run.",
50        vec![
51            kpi("Released", &releases.read("/latest/version")),
52            kpi("Declared", &version.read("/declared")),
53            kpi("Next", &version.read("/next")),
54            kpi("Gates", &gates.read("/count")),
55        ],
56    );
57
58    let deployed = unanswered(deployments).unwrap_or_else(|| {
59        let rows = list(deployments, "/deployments")
60            .iter()
61            .map(|d| {
62                el("tr")
63                    .child(
64                        el("td")
65                            .child(el("span").class("mj-ui-mono mj-ui-dim").text(text(d, "id")))
66                            .child(el("span").class("mj-ui-cell-title").text(text(d, "title"))),
67                    )
68                    .child(el("td").class("mj-ui-mono").text(text(d, "application")))
69                    .child(el("td").child(state(d, "status")))
70            })
71            .collect::<Vec<_>>();
72        table(
73            &["Deployment", "Application", "State"],
74            rows,
75            "No deployment is declared.",
76        )
77    });
78
79    let model = unanswered(gates).unwrap_or_else(|| {
80        let rows = list(gates, "/gates")
81            .iter()
82            .map(|g| {
83                let when = if g.get("always").and_then(Value::as_bool) == Some(true) {
84                    "every plan"
85                } else if g.get("on-demand").and_then(Value::as_bool) == Some(true) {
86                    "on demand"
87                } else {
88                    "when its paths change"
89                };
90                el("tr")
91                    .child(el("td").class("mj-ui-mono").text(text(g, "id")))
92                    .child(el("td").text(text(g, "job")))
93                    .child(el("td").class("mj-ui-dim").text(when))
94                    .child(el("td").class("mj-ui-dim").text(text(g, "summary")))
95            })
96            .collect::<Vec<_>>();
97        table(
98            &["Gate", "Job", "Runs", "What it holds"],
99            rows,
100            "The CI model declares no gate.",
101        )
102    });
103
104    let grid = el("div")
105        .class("mj-ui-grid")
106        .child(releases_panel(releases, version))
107        .child(panel(
108            "Deployments",
109            "Where the product is declared to run.",
110            deployed,
111            &[deployments],
112        ))
113        .child(wide(panel(
114            "CI model",
115            "Every gate the model declares, the job it runs in and when it is selected.",
116            model,
117            &[gates],
118        )))
119        .child(wide(super::timings::panel_of(
120            &answers[&super::timings::TIMINGS],
121        )));
122
123    View::new(
124        "CI/CD & Releases",
125        el("div").class("mj-ui-stack").child(top).child(grid),
126    )
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132    use crate::ui::deployments::tests::furnished;
133
134    /// Every declared deployment is a row, and every gate of the CI model says when it
135    /// runs: in every plan, on demand, or when its paths change.
136    #[test]
137    fn every_deployment_and_every_gate_is_a_row_that_says_when_it_runs() {
138        let repo = furnished();
139        let ctx = repo.context().expect("a context");
140        let html = screen(&ctx).main.render();
141        assert!(
142            html.contains("The Majordomus layer, served from one machine"),
143            "{html}"
144        );
145        assert!(!html.contains("No deployment is declared."), "{html}");
146        for (gate, when) in [
147            ("every-plan", "every plan"),
148            ("when-asked", "on demand"),
149            ("by-path", "when its paths change"),
150        ] {
151            assert!(html.contains(gate), "{gate} is missing: {html}");
152            assert!(html.contains(when), "{gate} does not say {when}: {html}");
153        }
154        assert!(
155            html.contains("runs in every plan"),
156            "the gate's own summary: {html}"
157        );
158        assert!(!html.contains("The CI model declares no gate."), "{html}");
159    }
