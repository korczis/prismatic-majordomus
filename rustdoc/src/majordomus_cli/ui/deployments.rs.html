1//! Deployments: where the product runs and whether what is live is what was built. Read
2//! from `deploy.list` (what the layer declares) and `deploy.observed` (what the forge
3//! recorded: every deployment to every environment, and every release); the delivery report
4//! (`delivery.report`), which probes the public site and judges every feature's delivery, is
5//! slow enough to be a panel of its own.
6//!
7//! Every commit, ref, environment, run and release here is a link, built from the
8//! repository's address as the forge itself gave it — no host is written in this file.

25/// Where one declared deployment is: its row on this screen.
26pub fn href(id: &str) -> String {
27    format!("{ROUTE}#deployment-{}", anchor(id))
28}

30/// One declared deployment as a row: its id leading to its row here, its declaration's
31/// technical view, the application it runs and its state. The CI/CD screen draws the same row.
32pub fn summary_row(ctx: &Context, d: &Value) -> El {
33    let id = text(d, "id");
34    el("tr")
35        .child(
36            el("td")
37                .child(
38                    el("a")
39                        .class("mj-ui-row-link")
40                        .attr("href", href(&id))
41                        .child(
42                            el("span")
43                                .class("mj-ui-mono mj-ui-dim")
44                                .text(value(d, "id")),
45                        )
46                        .child(el("span").class("mj-ui-cell-title").text(value(d, "title"))),
47                )
48                .child(object_if(
49                    ctx,
50                    &format!("majordomus://deployment/{id}"),
51                    &text(d, "file"),
52                )),
53        )
54        .child(el("td").class("mj-ui-mono").text(value(d, "application")))
55        .child(el("td").child(state(d, "status")))
56}

64/// A link that leaves the UI: the forge, a run, a served environment.
65fn outside(href: &str, label: &str) -> El {
66    el("a")
67        .class("mj-ui-link")
68        .attr("href", href)
69        .attr("rel", "noopener noreferrer")
70        .text(label)
71}

73/// A field of an entry as a link out, or `none` when the entry has no such field.
74fn outside_field(entry: &Value, key: &str, label: &str) -> El {
75    match text(entry, key) {
76        href if href.is_empty() => none(),
77        href => outside(&href, label),
78    }
79}

81/// A commit as its short name, leading to the commit on the forge.
82fn commit(repository_url: &str, sha: &str) -> El {
83    let short = &sha[..sha.len().min(10)];
84    el("span")
85        .class("mj-ui-mono")
86        .child(outside(&format!("{repository_url}/commit/{sha}"), short))
87}

89/// The three tables of what the forge recorded: where each environment stands, every
90/// deployment observed, and every release. A checkout that has never asked the forge shows
91/// the reason in their place — the command that asks is in it.
92fn observed_panels(observed: &Answer) -> Vec<El> {
93    if let Some(unread) = unanswered(observed) {
94        return vec![unread];
95    }
96    let root = observed.ok().cloned().unwrap_or(Value::Null);
97    if root.get("observed").and_then(Value::as_bool) != Some(true) {
98        return vec![nothing(&value(&root, "reason"))];
99    }
100    let forge = text(&root, "repository_url");
101    let environments = list(observed, "/environments")
102        .iter()
103        .map(|e| {
104            el("tr")
105                .child(el("td").child(outside(&text(e, "activity_url"), &text(e, "name"))))
106                .child(el("td").child(state(e, "state")))
107                .child(el("td").child(commit(&forge, &text(e, "sha"))))
108                .child(el("td").class("mj-ui-mono").text(value(e, "deployed_at")))
109                .child(el("td").child(outside_field(e, "url", &text(e, "url"))))
110                .child(
111                    el("td").class("mj-ui-mono").text(
112                        e.get("deployments")
113                            .and_then(Value::as_u64)
114                            .unwrap_or(0)
115                            .to_string(),
116                    ),
117                )
118        })
119        .collect::<Vec<_>>();
120    let deployments = list(observed, "/deployments")
121        .iter()
122        .map(|d| {
123            let git_ref = text(d, "git_ref");
124            el("tr")
125                .child(el("td").class("mj-ui-mono").text(value(d, "created_at")))
126                .child(el("td").class("mj-ui-mono").text(value(d, "environment")))
127                .child(el("td").child(state(d, "state")))
128                .child(el("td").child(commit(&forge, &text(d, "sha"))))
129                .child(
130                    el("td")
131                        .class("mj-ui-mono")
132                        .child(match git_ref.is_empty() {
133                            true => none(),
134                            false => outside(&format!("{forge}/tree/{git_ref}"), &git_ref),
135                        }),
136                )
137                .child(el("td").class("mj-ui-mono").text(value(d, "creator")))
138                .child(el("td").child(outside_field(d, "log_url", "the run")))
139        })
140        .collect::<Vec<_>>();
141    let releases = list(observed, "/releases")
142        .iter()
143        .map(|r| {
144            let kind = match (
145                r.get("draft").and_then(Value::as_bool) == Some(true),
146                r.get("prerelease").and_then(Value::as_bool) == Some(true),
147            ) {
148                (true, _) => "draft",
149                (false, true) => "pre-release",
150                (false, false) => "release",
151            };
152            el("tr")
153                .child(
154                    el("td")
155                        .class("mj-ui-mono")
156                        .child(outside(&text(r, "url"), &text(r, "tag"))),
157                )
158                .child(el("td").text(value(r, "name")))
159                .child(el("td").class("mj-ui-mono").text(value(r, "published_at")))
160                .child(el("td").text(kind))
161        })
162        .collect::<Vec<_>>();
163    vec![
164        el("h3")
165            .class("mj-ui-panel-title")
166            .text("Where each environment stands"),
167        table(
168            &[
169                "Environment",
170                "State",
171                "Commit",
172                "Deployed",
173                "Served at",
174                "Deployments",
175            ],
176            environments,
177            "The forge has recorded no deployment.",
178        ),
179        el("h3")
180            .class("mj-ui-panel-title")
181            .text("Every deployment observed"),
182        table(
183            &[
184                "Deployed",
185                "Environment",
186                "State",
187                "Commit",
188                "Ref",
189                "By",
190                "Run",
191            ],
192            deployments,
193            "The forge has recorded no deployment.",
194        ),
195        el("h3").class("mj-ui-panel-title").text("Releases"),
196        table(
197            &["Tag", "Title", "Published", "Kind"],
198            releases,
199            "The forge has published no release.",
200        ),
201    ]
202}

205/// The Deployments screen.
206pub fn screen(ctx: &Context) -> View {
207    let answers = ask_all(ctx, &[DEPLOYMENTS, OBSERVED]);
208    let deployments = &answers[&DEPLOYMENTS];
209    let observed = &answers[&OBSERVED];
210
211    let top = hero(
212        "server",
213        "green",
214        "Deployments",
215        "What is live, and whether it is what was built.",
216        "Every declared deployment, every deployment and release the forge recorded, and — asked on its own — whether the public site serves the trunk and how far each feature has been delivered.",
217        vec![
218            kpi("Deployments", &deployments.read("/count")),
219            kpi_count("Declared", &deployments.read("/deployments")),
220            kpi_count("Environments", &observed.read("/environments")),
221            kpi_count("Observed", &observed.read("/deployments")),
222            kpi_count("Releases", &observed.read("/releases")),
223        ],
224    );
225
226    let declared = unanswered(deployments).unwrap_or_else(|| {
227        let rows = list(deployments, "/deployments")
228            .iter()
229            .map(|d| {
230                summary_row(ctx, d).attr("id", format!("deployment-{}", anchor(&text(d, "id"))))
231            })
232            .collect::<Vec<_>>();
233        table(
234            &["Deployment", "Application", "State"],
235            rows,
236            "No deployment is declared.",
237        )
238    });
239
240    let placeholder = el("section")
241        .class("mj-ui-panel")
242        .attr("aria-label", "Delivery")
243        .attr("data-panel-src", DELIVERY_PANEL)
244        .attr("aria-busy", "false")
245        .child(
246            el("header").class("mj-ui-panel-head").child(
247                el("div")
248                    .class("mj-ui-panel-heading")
249                    .child(el("h2").class("mj-ui-panel-title").text("Delivery"))
250                    .child(el("p").class("mj-ui-panel-sub").text(
251                        "Asked on its own: delivery.report probes the public site and judges every feature, which would hold this screen back.",
252                    )),
253            ),
254        )
255        .child(
256            el("p").class("mj-ui-panel-body").child(
257                el("a").class("mj-ui-link").attr("href", DELIVERY_PANEL).text("Open the delivery report"),
258            ),
259        );
260
261    let grid = el("div")
262        .class("mj-ui-grid")
263        .child(wide(panel(
264            "Declared deployments",
265            "Where the product is declared to run.",
266            declared,
267            &[deployments],
268        )))
269        .child(wide(panel(
270            "What was deployed",
271            "Every deployment the forge recorded, to every environment, and every release — as it was last observed.",
272            observed_panels(observed)
273                .into_iter()
274                .fold(el("div").class("mj-ui-stack"), El::child),
275            &[observed],
276        )))
277        .child(wide(placeholder));
278
279    View::new(
280        "Deployments",
281        el("div").class("mj-ui-stack").child(top).child(grid),
282    )
283}

285/// The delivery report as a panel: the publication's verification and every feature's stage.
286pub fn delivery_panel(ctx: &Context) -> View {
287    View::new("Delivery", delivery(ctx, &ask(ctx, DELIVERY)))
288}

290fn delivery(ctx: &Context, report: &Answer) -> El {
291    let body = unanswered(report).unwrap_or_else(|| {
292        let tallies = strip(vec![
293            kpi("Features", &report.read("/tallies/features")),
294            kpi("On the trunk", &report.read("/tallies/on_master")),
295            kpi("Deployed", &report.read("/tallies/deployed")),
296            kpi(
297                "Publicly verified",
298                &report.read("/tallies/publicly_verified"),
299            ),
300        ]);
301        let url = report.read("/publication/url").text();
302        let publication = el("p")
303            .class("mj-ui-line")
304            .child(state(
305                report
306                    .ok()
307                    .and_then(|v| v.pointer("/publication/verification"))
308                    .unwrap_or(&Value::Null),
309                "state",
310            ))
311            .text(" ")
312            .child(match &url {
313                // the address the report probed, as it gave it
314                Some(u) => link(u.clone(), u.clone()),
315                None => el("span").text("no public address"),
316            })
317            .text(format!(
318                " — {}",
319                report
320                    .read("/publication/verification/detail")
321                    .text()
322                    .unwrap_or_else(|| "none".into())
323            ));
324        let rows = list(report, "/features")
325            .iter()
326            .map(|f| {
327                let blocking: Vec<String> = f
328                    .pointer("/delivery/blocking")
329                    .and_then(Value::as_array)
330                    .map(|b| {
331                        b.iter()
332                            .filter_map(Value::as_str)
333                            .map(str::to_string)
334                            .collect()
335                    })
336                    .unwrap_or_default();
337                let id = text(f, "id");
338                el("tr")
339                    .child(
340                        el("td")
341                            .child(object_if(ctx, &format!("majordomus://feature/{id}"), &id))
342                            .child(el("span").class("mj-ui-cell-title").text(value(f, "title"))),
343                    )
344                    .child(
345                        el("td").child(state(f.get("delivery").unwrap_or(&Value::Null), "stage")),
346                    )
347                    .child(
348                        el("td").child(state(f.get("delivery").unwrap_or(&Value::Null), "state")),
349                    )
350                    .child(el("td").child(if blocking.is_empty() {
351                        none()
352                    } else {
353                        el("span")
354                            .class("mj-ui-mono mj-ui-dim")
355                            .text(blocking.join(", "))
356                    }))
357            })
358            .collect::<Vec<_>>();
359        el("div")
360            .class("mj-ui-stack")
361            .child(tallies)
362            .child(publication)
363            .child(table(
364                &["Feature", "Stage", "Delivery", "What blocks it"],
365                rows,
366                "No feature is declared.",
367            ))
368    });
369    panel_to(
370        "Delivery",
371        "Whether the public site serves the trunk, and how far each feature has been delivered.",
372        None,
373        body,
374        &[report],
375    )
376}

405    /// A synthetic repository furnished with what the screens draw rows from: this
406    /// repository's own deployment, one feature, two claims (one of them a guarantee no
407    /// test supports, so the report has a finding) and a CI model with a gate of each kind
408    /// of selection.
409    pub(crate) fn furnished() -> SyntheticRepository {
410        let repo = SyntheticRepository::small().expect("a synthetic repository");
411        let write = |rel: &str, body: &str| {
412            let path = repo.root().join(rel);
413            std::fs::create_dir_all(path.parent().expect("a parent")).expect("a directory");
414            std::fs::write(path, body).expect("a file");
415        };
416        let sources = repo.root().join(".ai/repo/knowledge/sources.yaml");
417        let mut text = std::fs::read_to_string(&sources).expect("the sources");
418        text.push_str(FURNISHED_SOURCES);
419        std::fs::write(&sources, text).expect("the sources");
420        let deployment = std::fs::read_to_string(concat!(
421            env!("CARGO_MANIFEST_DIR"),
422            "/../../.ai/repo/deployments/majordomus.yaml"
423        ))
424        .expect("this repository's deployment");
425        write(".ai/repo/deployments/majordomus.yaml", &deployment);
426        write(
427            ".ai/repo/features/synthetic.md",
428            "---\nschema: feature/v1\nid: synthetic\nkind: feature\ntitle: A synthetic feature\n\
429             headline: A feature that exists for a test.\n\
430             summary: What a synthetic feature does.\nstatus: stable\n---\n\n## What it does\n\nIt is a fixture.\n",
431        );
432        write(
433            "docs/CLAIMS.yaml",
434            "version: 1\nclaims:\n  - id: synthetic-guarantee\n    claim: A guarantee no test supports\n\
435             \x20   source: docs/DOC_0.md\n    status: guaranteed\n\
436             \x20 - id: synthetic-advice\n    claim: An advisory claim\n    source: docs/DOC_0.md\n\
437             \x20   test: test/cases/1_synthetic.sh\n    status: advisory\n",

440            ".ai/repo/ci/gates.yaml",
