1//! CI/CD & Releases: what was released, whether the landed work since is released, where it
2//! is deployed, and what the CI model runs. Read from `distribution.releases`,
3//! `release.version`, `release.currency`, `deploy.list` and `gates.model`.
4//!
5//! How long CI takes is `integration.timings`, measured from the recorded forge observation
6//! (`super::timings`): the CI model says which gates run and when, the timings how long the
7//! runs on the open pull requests took.

22/// The route of the screen.
23pub const ROUTE: &str = "/ui/delivery";

25/// Where one gate of the CI model is: its row on this screen. A rule's proof and a pull
26/// request's checks name gates by id, and lead here.
27pub fn gate_href(id: &str) -> String {
28    format!("{ROUTE}#gate-{}", anchor(id))
29}

31const RELEASES: Source = Source::of("distribution.releases");
32const VERSION: Source = Source::of("release.version");
33const DEPLOYMENTS: Source = Source::of("deploy.list");
34const GATES: Source = Source::of("gates.model");
35// asked without the version: the hero's Next is release.version's, and one screen names one
36// next version (the plan critique of 2026-10-19)
37const CURRENCY: Source = Source::with("release.currency", r#"{"next":false}"#);

39/// The CI/CD & Releases screen.
40pub fn screen(ctx: &Context) -> View {
41    let answers = ask_all(
42        ctx,
43        &[
44            RELEASES,
45            VERSION,
46            DEPLOYMENTS,
47            GATES,
48            CURRENCY,
49            super::timings::TIMINGS,
50        ],
51    );
52    let (releases, version, deployments, gates, currency) = (
53        &answers[&RELEASES],
54        &answers[&VERSION],
55        &answers[&DEPLOYMENTS],
56        &answers[&GATES],
57        &answers[&CURRENCY],
58    );
59
60    let top = hero(
61        "layers",
62        "orange",
63        "CI/CD & Releases",
64        "From a green check to a published release.",
65        "What was released, whether the work landed since is released, what the tree declares and what its commits imply next, where it is deployed, and every gate the CI model can run.",
66        vec![
67            kpi("Released", &releases.read("/latest/version")),
68            kpi_state("Since then", &currency.read("/stance")),
69            kpi("Declared", &version.read("/declared")),
70            kpi("Next", &version.read("/next")),
71            kpi("Gates", &gates.read("/count")),
72        ],
73    );
74
75    let deployed = unanswered(deployments).unwrap_or_else(|| {
76        let rows = list(deployments, "/deployments")
77            .iter()
78            .map(|d| super::deployments::summary_row(ctx, d))
79            .collect::<Vec<_>>();
80        table(
81            &["Deployment", "Application", "State"],
82            rows,
83            "No deployment is declared.",
84        )
85    });
86
87    let model = unanswered(gates).unwrap_or_else(|| {
88        let rows = list(gates, "/gates")
89            .iter()
90            .map(|g| {
91                let when = if g.get("always").and_then(Value::as_bool) == Some(true) {
92                    "every plan"
93                } else if g.get("on-demand").and_then(Value::as_bool) == Some(true) {
94                    "on demand"
95                } else {
96                    "when its paths change"
97                };
98                let id = text(g, "id");
99                el("tr")
100                    .attr("id", format!("gate-{}", anchor(&id)))
101                    .child(el("td").child(code_link(gate_href(&id), value(g, "id"))))
102                    .child(el("td").text(value(g, "job")))
103                    .child(el("td").class("mj-ui-dim").text(when))
104                    .child(el("td").class("mj-ui-dim").text(value(g, "summary")))
105                    .child(el("td").child(gates_named(g, "implies")))
106            })
107            .collect::<Vec<_>>();
108        table(
109            &["Gate", "Job", "Runs", "What it holds", "Implies"],
110            rows,
111            "The CI model declares no gate.",
112        )
113    });
114
115    let grid = el("div")
116        .class("mj-ui-grid")
117        .child(releases_panel(releases, version, None))
118        .child(panel(
119            "Deployments",
120            "Where the product is declared to run.",
121            deployed,
122            &[deployments],
123        ))
124        // full width, after the pair above, so that adding it moves no panel of the grid
125        .child(wide(currency_panel(currency)))
126        .child(wide(panel(
127            "CI model",
128            "Every gate the model declares, the job it runs in and when it is selected.",
129            model,
130            &[gates],
131        )))
132        .child(wide(super::timings::panel_of(
133            &answers[&super::timings::TIMINGS],
134        )));
135
136    View::new(
137        "CI/CD & Releases",
138        el("div").class("mj-ui-stack").child(top).child(grid),
139    )
140}

142/// Whether the trunk's landed work is released (`release.currency`, I3071): the stance and
143/// its sentence, how much landed work waits, since when and how that was dated, and the
144/// release it is measured from. The version required next is the hero's, read from
145/// `release.version`. The capability reads the remote-tracking trunk as of its last fetch and
146/// the clock, so the panel says that a published snapshot shows it as read when the snapshot
147/// was built.
148fn currency_panel(currency: &Answer) -> El {
149    let body = unanswered(currency)
150        .unwrap_or_else(|| currency_body(currency.ok().unwrap_or(&Value::Null)));
151    panel(
152        "Is the trunk released?",
153        "The landed work past the last release, read from the remote-tracking trunk as of its last fetch. A published snapshot shows it as read when the snapshot was built.",
154        body,
155        &[currency],
156    )
157}

159/// The rows of one answer of `release.currency`, each in the answer's own words: a figure
160/// the answer did not give is said as missing, never as a zero.
161fn currency_body(answer: &Value) -> El {
162    let number = |key: &str| answer.get(key).and_then(Value::as_u64);
163    let waited = match (
164        number("waited_seconds"),
165        answer.get("oldest_landing").and_then(Value::as_str),
166    ) {
167        (Some(seconds), at) => {
168            let dated = match answer.get("landed_at").and_then(Value::as_str) {
169                // a commit time bounds the wait from above, and the row says so
170                Some("commit") => format!(
171                    "at most {}, from its commit time: this clone holds no record of when it landed",
172                    super::timings::duration(seconds)
173                ),
174                _ => format!(
175                    "{}, since this clone saw it land",
176                    super::timings::duration(seconds)
177                ),
178            };
179            match at {
180                Some(at) => format!("{dated} ({at})"),
181                None => dated,
182            }
183        }
184        (None, _) => "nothing waits".into(),
185    };
186    let landed = match (number("landed"), number("landings")) {
187        (Some(commits), Some(landings)) => {
188            format!("{commits} commit(s) in {landings} landing(s)")
189        }
190        _ => "not measured".into(),
191    };
192    let measured_from = answer.get("release").map_or_else(
193        || "no release".to_string(),
194        |release| {
195            let reference = value(release, "reference");
196            if release.get("recorded").and_then(Value::as_bool) == Some(false) {
197                format!("{reference} (tagged; its record has not landed yet)")
198            } else {
199                reference
200            }
201        },
202    );
203    let row = |label: &str, cell: El| {
204        el("tr")
205            .child(el("th").attr("scope", "row").text(label))
206            .child(el("td").child(cell))
207    };
208    table(
209        &["Reading", "Answer"],
210        vec![
211            row(
212                "Stance",
213                el("span")
214                    .child(word_badge(&value(answer, "stance")))
215                    .text(" ")
216                    .text(text(answer, "summary")),
217            ),
218            row("Landed past it", el("span").text(landed)),
219            row("Waited", el("span").text(waited)),
220            row("Measured from", el("span").text(measured_from)),
221        ],
222        "release.currency gave no answer.",
223    )
224}

226/// The gates an entry names under `key`, each a link to its own row; `none` when it names none.
227fn gates_named(g: &Value, key: &str) -> crate::cockpit::html::El {
228    let ids: Vec<String> = g
229        .get(key)
230        .and_then(Value::as_array)
231        .map(|a| {
232            a.iter()
233                .filter_map(Value::as_str)
234                .map(str::to_string)
235                .collect()
236        })
237        .unwrap_or_default();
238    if ids.is_empty() {
239        return none();
240    }
241    el("span")
242        .class("mj-ui-badges")
243        .children(ids.iter().map(|id| code_link(gate_href(id), id.clone())))
244}

251    /// Every declared deployment is a row, and every gate of the CI model says when it
252    /// runs: in every plan, on demand, or when its paths change.
253    #[test]
254    fn every_deployment_and_every_gate_is_a_row_that_says_when_it_runs() {
255        let repo = furnished();
256        let ctx = repo.context().expect("a context");
257        let html = screen(&ctx).main.render();
258        assert!(
259            html.contains("The Majordomus layer, served from its container image"),
260            "{html}"
261        );
262        assert!(!html.contains("No deployment is declared."), "{html}");
263        for (gate, when) in [
264            ("every-plan", "every plan"),
265            ("when-asked", "on demand"),
266            ("by-path", "when its paths change"),
267        ] {
268            assert!(html.contains(gate), "{gate} is missing: {html}");
269            assert!(html.contains(when), "{gate} does not say {when}: {html}");
270        }
271        assert!(
272            html.contains("runs in every plan"),
273            "the gate's own summary: {html}"
274        );
275        assert!(!html.contains("The CI model declares no gate."), "{html}");
276    }

278    /// The release panel says the stance in its own sentence, how much landed work waits and
279    /// since when, the release it is measured from and the version required next (I3071).
280    #[test]
281    fn the_release_panel_says_what_waits_past_the_last_release() {
282        // the answers below are shaped as measure() gives them, asked with next false
283        let owed = currency_body(&serde_json::json!({
284            "stance": "owed",
285            "summary": "1 landed commit(s) in 1 landing(s) wait past v0.30.2 (tagged, its record not landed yet) on origin/master, the oldest for 1 h 2 min since this clone saw it land",
286            "trunk": "origin/master",
287            "release": {"version": "0.30.2", "reference": "v0.30.2", "at": "v0.30.2", "recorded": false},
288            "landed": 1,
289            "landings": 1,
290            "oldest_landing": "2026-10-19T05:00:00Z",
291            "landed_at": "reflog",
292            "waited_seconds": 3720,
293            "overdue_after_seconds": 21600
294        }))
295        .render();
296        for want in [
297            ">owed<",
298            "wait past v0.30.2",
299            "1 commit(s) in 1 landing(s)",
300            "1 h 02 min, since this clone saw it land (2026-10-19T05:00:00Z)",
301            "v0.30.2 (tagged; its record has not landed yet)",
302        ] {
303            assert!(owed.contains(want), "{want}: {owed}");
304        }
305
306        let bounded = currency_body(&serde_json::json!({
307            "stance": "owed",
308            "summary": "1 landed commit(s) in 1 landing(s) wait past v0.30.2 on origin/master, the oldest for at most 7 h 0 min",
309            "trunk": "origin/master",
310            "release": {"version": "0.30.2", "reference": "v0.30.2", "at": "abc", "recorded": true},
311            "landed": 1,
312            "landings": 1,
313            "oldest_landing": "2026-10-18T23:00:00Z",
314            "landed_at": "commit",
315            "waited_seconds": 25200,
316            "overdue_after_seconds": 21600
317        }))
318        .render();
319        assert!(
320            bounded.contains("at most 7 h 00 min, from its commit time"),
321            "{bounded}"
322        );
323
324        let current = currency_body(&serde_json::json!({
325            "stance": "current",
326            "summary": "origin/master holds no landed work past v0.30.2",
327            "trunk": "origin/master",
328            "release": {"version": "0.30.2", "reference": "v0.30.2", "at": "abc", "recorded": true},
329            "landed": 0,
330            "landings": 0,
331            "overdue_after_seconds": 21600
332        }))
333        .render();
334        assert!(current.contains("nothing waits"), "{current}");
335        assert!(
336            current.contains("holds no landed work past v0.30.2"),
337            "{current}"
338        );
339        assert!(!current.contains("its record has not landed"), "{current}");
340
341        // a figure the answer did not give is said as missing, never as a zero
342        let bare = currency_body(&serde_json::json!({
343            "stance": "unknown",
344            "summary": "the trunk cannot be told: no remote-tracking branch",
345            "landed": 0,
346            "landings": 0,
347            "overdue_after_seconds": 21600
348        }))
349        .render();
350        assert!(bare.contains("no release"), "{bare}");
351        assert!(bare.contains("nothing waits"), "{bare}");
352        assert!(bare.contains("no remote-tracking branch"), "{bare}");
353
354        // a wait the answer gave without its landing's date is the wait alone, and counts it
355        // did not give are not measured
356        let partial = currency_body(&serde_json::json!({
357            "stance": "owed",
358            "summary": "landed work waits past v0.30.2 on origin/master",
359            "trunk": "origin/master",
360            "release": {"version": "0.30.2", "reference": "v0.30.2", "at": "abc", "recorded": true},
361            "landed_at": "reflog",
362            "waited_seconds": 3600,
363            "overdue_after_seconds": 21600
364        }))
365        .render();
366        assert!(
367            partial.contains("1 h 00 min, since this clone saw it land"),
368            "{partial}"
369        );
370        assert!(!partial.contains("saw it land ("), "{partial}");
371        assert!(partial.contains("not measured"), "{partial}");
372    }
