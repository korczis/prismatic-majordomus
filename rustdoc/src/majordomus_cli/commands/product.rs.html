1//! `majordomus product`: the product model, through the registry's own `product.*`
2//! capabilities.
3//!
4//! This file renders; it does not know anything. Every answer below is the output of the
5//! capability the CLI exposure names, executed through the one executor, so the command
6//! line, the HTTP routes and the MCP tools cannot answer the same question differently.
7//! Nothing here names a feature, a module, a command, a kind or a provider.
8

18/// The exit code when `validate` finds an error in the model.
19pub const EXIT_INVALID: u8 = 10;
20
21/// Run `majordomus product`.
22pub fn run(args: ProductArgs) -> Result<u8> {
23    let app = App::load(&args.repo)?;
24    let format = args.format;
25    match args.command.as_ref().unwrap_or(&ProductCommand::List) {
26        ProductCommand::List => {
27            let v = call(&app, &["product", "list"], query_of(&args))?;
28            emit(format, &v, list_text)
29        }
30        ProductCommand::Show { id } => {
31            let v = call(&app, &["product", "show"], json!({ "id": id }))?;
32            emit(format, &v, feature_text)
33        }
34        ProductCommand::Matrix => {
35            let v = call(&app, &["product", "matrix"], json!({}))?;
36            emit(format, &v, matrix_text)
37        }
38        ProductCommand::Providers => {
39            let v = call(&app, &["product", "providers"], json!({}))?;
40            emit(format, &v, providers_text)
41        }
42        ProductCommand::Validate => {
43            let v = call(&app, &["product", "validate"], json!({}))?;
44            let code = if v["valid"].as_bool().unwrap_or(false) {
45                0
46            } else {
47                EXIT_INVALID
48            };
49            emit(format, &v, validation_text)?;
50            Ok(code)
51        }
52    }
53}

55/// Every facet the command line offers, as the capability's own input. A facet added to
56/// the query type is a flag here and nowhere else.
57fn query_of(f: &ProductArgs) -> Value {
58    let mut o = serde_json::Map::new();
59    let mut put = |k: &str, v: &Option<String>| {
60        if let Some(v) = v {
61            o.insert(k.into(), Value::String(v.clone()));
62        }
63    };
64    put("area", &f.area);
65    put("module", &f.module);
66    put("command", &f.names_command);
67    put("surface", &f.surface);
68    put("q", &f.query);
69    if f.featured {
70        o.insert("featured".into(), Value::Bool(true));
71    }
72    if f.all {
73        o.insert("status".into(), Value::String("any".into()));
74    }
75    Value::Object(o)
76}

78fn call(app: &App, path: &[&str], input: Value) -> Result<Value> {
79    let ctx = &app.context;
80    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
81    let id = ctx
82        .registry
83        .by_cli(&words)
84        .map(|c| c.id.to_string())
85        .ok_or_else(|| Error::Protocol {
86            reason: format!(
87                "no capability is exposed as `majordomus {}`",
88                path.join(" ")
89            ),
90        })?;
91    ctx.execute(&id, input).map_err(map)
92}

94fn map(e: CapabilityError) -> Error {
95    match e {
96        CapabilityError::NotFound(reason) => Error::NotFound { reason },
97        CapabilityError::InvalidInput(reason) => Error::Protocol { reason },
98        other => Error::Protocol {
99            reason: other.to_string(),
100        },
101    }
102}

104fn emit(format: OutputFormat, v: &Value, text: impl Fn(&Value) -> String) -> Result<u8> {
105    let stdout = std::io::stdout();
106    let mut out = stdout.lock();
107    let body = match format {
108        OutputFormat::Json => serde_json::to_string_pretty(v).unwrap_or_else(|_| v.to_string()),
109        OutputFormat::Text => text(v),
110    };
111    writeln!(out, "{body}").map_err(Error::Transport)?;
112    Ok(0)
113}

115fn s(v: &Value, k: &str) -> String {
116    v[k].as_str().unwrap_or("").to_string()

119fn arr(v: &Value, k: &str) -> Vec<String> {
120    v[k].as_array()
121        .map(|a| {
122            a.iter()
123                .map(|x| x.as_str().unwrap_or("").to_string())
124                .collect()
125        })
126        .unwrap_or_default()
127}

129/// The surfaces of one feature as five marks, in the vocabulary's order, read from the
130/// answer's own `surfaces` object rather than from a list here.
131fn marks(surfaces: &Value) -> String {
132    crate::product::SURFACES
133        .iter()
134        .map(|(id, _, _)| {
135            if surfaces[id].as_bool().unwrap_or(false) {
136                (*id).to_string()
137            } else {
138                "-".repeat(id.len())
139            }
140        })
141        .collect::<Vec<_>>()
142        .join(" ")
143}

145fn list_text(v: &Value) -> String {
146    let features = v["features"].as_array().cloned().unwrap_or_default();
147    let width = features
148        .iter()
149        .map(|f| s(f, "id").len())
150        .max()
151        .unwrap_or(4)
152        .max(4);
153    let mut out = vec![format!(
154        "{:<width$}  {:<26}  {}",
155        "SLUG",
156        "SURFACES",
157        "TITLE",
158        width = width
159    )];
160    for f in &features {
161        out.push(format!(
162            "{:<width$}  {:<26}  {}{}",
163            s(f, "id"),
164            marks(&f["surfaces"]),
165            s(f, "title"),
166            if f["featured"].as_bool().unwrap_or(false) {
167                "  [featured]"
168            } else {
169                ""
170            },
171            width = width
172        ));
173    }
174    let c = &v["counts"];
175    out.push(String::new());
176    out.push(format!(
177        "{} feature(s) listed; {} stable, {} featured, {} provider(s); modules {}/{}, commands {}/{}, kinds {}/{} named by a feature  [{}]",
178        features.len(),
179        c["features"],
180        c["featured"],
181        c["providers"],
182        c["modules_covered"],
183        c["modules"],
184        c["commands_covered"],
185        c["commands"],
186        c["kinds_covered"],
187        c["kinds"],
188        &v["fingerprint"].as_str().unwrap_or("")[..12.min(s(v, "fingerprint").len())],
189    ));
190    out.join("\n")
191}

193fn feature_text(v: &Value) -> String {
194    let mut out = vec![
195        format!("{}  {}", s(v, "id"), s(v, "title")),
196        String::new(),
197        format!("  {}", s(v, "headline")),
198        format!("  {}", s(v, "summary")),
199        String::new(),
200        format!(
201            "  status {}   weight {}   featured {}   route {}",
202            s(v, "status"),
203            v["weight"],
204            v["featured"],
205            s(v, "route")
206        ),
207        format!("  source {}", s(v, "source")),
208        format!("  surfaces {}   (derived)", marks(&v["surfaces"])),
209    ];
210    let row = |label: &str, values: Vec<String>| -> Option<String> {
211        (!values.is_empty()).then(|| format!("  {label:<18}{}", values.join(", ")))
212    };
213    out.push(String::new());
214    for (label, key) in [
215        ("areas", "areas"),
216        ("audiences", "audiences"),
217        ("modules", "modules"),
218        ("commands", "commands"),
219        ("kinds", "kinds"),
220        ("rules", "rules"),
221        ("docs", "docs"),
222        ("adrs", "adrs"),
223        ("claims", "claims"),
224        ("use cases", "use_cases"),
225        ("cockpit", "cockpit"),
226        ("web", "web"),
227        ("related", "related"),
228        ("tags", "tags"),
229    ] {
230        out.extend(row(label, arr(v, key)));
231    }
232    out.push(String::new());
233    out.push("  derived".into());
234    let c = &v["counts"];
235    out.push(format!(
236        "    capabilities {}  mcp tools {}  mcp resources {}  http routes {}  cli paths {}  commands {}  objects {}  rules {} ({} enforced)  docs {}  adrs {}  claims {}  use cases {}  moments {}",
237        c["capabilities"], c["mcp_tools"], c["mcp_resources"], c["http_routes"], c["cli_paths"], c["commands"], c["objects"], c["rules"], c["enforced_rules"], c["docs"], c["adrs"], c["claims"], c["use_cases"], c["moments"]
238    ));
239    if let Some(mods) = v["module_refs"].as_array() {
240        for m in mods {
241            out.push(format!("    module {} — {}", s(m, "id"), s(m, "title")));
242            for c in m["capabilities"].as_array().into_iter().flatten() {
243                out.push(format!(
244                    "      {:<32} {:<9} mcp={} http={} cli={}",
245                    s(c, "id"),
246                    s(c, "kind"),
247                    c["tool"].as_str().unwrap_or("-"),
248                    c["route"].as_str().unwrap_or("-"),
249                    c["cli"].as_str().unwrap_or("-"),
250                ));
251            }
252        }
253    }
254    for (label, key, name, extra) in [
255        ("commands", "command_refs", "id", "summary"),
256        ("kinds", "kind_refs", "name", "objects"),
257        ("rules", "rule_refs", "id", "class"),
258        ("claims", "claim_refs", "id", "status"),
259        ("moments", "moments", "id", "hook"),
260    ] {
261        if let Some(items) = v[key].as_array() {
262            if !items.is_empty() {
263                out.push(format!("    {label}"));
264                for i in items {
265                    let e = match &i[extra] {
266                        Value::String(x) => x.clone(),
267                        other => other.to_string(),
268                    };
269                    out.push(format!("      {:<40} {e}", s(i, name)));
270                }
271            }
272        }
273    }
274    let ev = &v["evidence"]["claims"];
275    if let Some(o) = ev.as_object() {
276        if !o.is_empty() {
277            out.push(format!(
278                "    evidence: {}",
279                o.iter()
280                    .map(|(k, n)| format!("{n} {k}"))
281                    .collect::<Vec<_>>()
282                    .join(", ")
283            ));
284        }
285    }
286    let back = arr(v, "backlinks");
287    if !back.is_empty() {
288        out.push(format!("    named by: {}", back.join(", ")));
289    }
290    out.join("\n")
291}

293fn matrix_text(v: &Value) -> String {
294    let rows = v["rows"].as_array().cloned().unwrap_or_default();
295    let width = rows
296        .iter()
297        .map(|r| s(r, "id").len())
298        .max()
299        .unwrap_or(7)
300        .max(7);
301    let header: String = crate::product::SURFACES
302        .iter()
303        .map(|(id, _, _)| *id)
304        .collect::<Vec<_>>()
305        .join(" ");
306    let mut out = vec![format!(
307        "{:<width$}  {}  STATUS",
308        "FEATURE",
309        header,
310        width = width
311    )];
312    for r in &rows {
313        out.push(format!(
314            "{:<width$}  {}  {}",
315            s(r, "id"),
316            marks(&r["surfaces"]),
317            s(r, "status"),
318            width = width
319        ));
320    }
321    for (label, key) in [
322        ("MODULE", "modules"),
323        ("COMMAND", "commands"),
324        ("KIND", "kinds"),
325    ] {
326        let items = v[key].as_array().cloned().unwrap_or_default();
327        if items.is_empty() {
328            continue;
329        }
330        let w = items
331            .iter()
332            .map(|c| s(c, "id").len())
333            .max()
334            .unwrap_or(4)
335            .max(label.len());
336        out.push(String::new());
337        out.push(format!("{:<w$}  FEATURES", label, w = w));
338        for c in &items {
339            let fs = arr(c, "features");
340            out.push(format!(
341                "{:<w$}  {}",
342                s(c, "id"),
343                if fs.is_empty() {
344                    "(none — a gap)".to_string()
345                } else {
346                    fs.join(", ")
347                },
348                w = w
349            ));
350        }
351    }
352    out.join("\n")
353}

355fn providers_text(v: &Value) -> String {
356    let items = v["providers"].as_array().cloned().unwrap_or_default();
357    let width = items
358        .iter()
359        .map(|p| s(p, "id").len())
360        .max()
361        .unwrap_or(8)
362        .max(8);
363    // EPISODE and PROMPTS are the declared capability — what the vendor offers, cited in
364    // share/providers.yaml — and not what this repository has wired. HOOKS beside them is
365    // the wiring the policy declares, and the two being adjacent is deliberate: a provider
366    // with `hooks` under EPISODE and nothing under HOOKS is a gap in this distribution, and
367    // it used to be invisible because neither column existed. `majordomus capture status`
368    // is the reading that resolves it into one word per aspect.
369    let mut out = vec![format!(
370        "{:<width$}  {:<36}  EPISODE     PROMPTS  BOOTSTRAPS  CLIENT CONFIG  HOOKS  SCRATCH ROOTS",
371        "PROVIDER",
372        "TITLE",
373        width = width
374    )];
375    for p in &items {
376        let boots: Vec<String> = p["bootstraps"]
377            .as_array()
378            .into_iter()
379            .flatten()
380            .map(|b| s(b, "target"))
381            .collect();
382        let hooks = {
383            let h = arr(p, "hooks");
384            if h.is_empty() {
385                "-".to_string()
386            } else {
387                h.join(",")
388            }
389        };
390        let roots = {
391            let r = arr(p, "scratch_roots");
392            if r.is_empty() {
393                "-".to_string()
394            } else {
395                r.join(",")
396            }
397        };
398        let episode = {
399            let v = s(&p["offers"], "episode");
400            if v.is_empty() {
401                "-".to_string()
402            } else {
403                v
404            }
405        };
406        let prompts = {
407            let v = s(&p["offers"], "prompts");
408            if v.is_empty() {
409                "-".to_string()
410            } else {
411                v
412            }
413        };
414        out.push(format!(
415            "{:<width$}  {:<36}  {:<10}  {:<7}  {:<10}  {:<13}  {:<5}  {}",
416            s(p, "id"),
417            s(p, "title"),
418            episode,
419            prompts,
420            if boots.is_empty() {
421                "-".to_string()
422            } else {
423                boots.join(",")
424            },
425            p["client_config"].as_str().unwrap_or("-"),
426            hooks,
427            roots,
428            width = width
429        ));
430    }
431    out.push(String::new());
432    out.push(format!("{} provider(s)", items.len()));
433    out.join("\n")
434}

436fn validation_text(v: &Value) -> String {
437    let mut out = Vec::new();
438    for f in v["findings"].as_array().into_iter().flatten() {
439        let mut line = format!("{:<8} {}", s(f, "severity").to_uppercase(), s(f, "path"));
440        if let Some(id) = f["id"].as_str() {
441            line.push_str(&format!(" [{id}]"));
442        }
443        if let Some(field) = f["field"].as_str() {
444            line.push_str(&format!(":{field}"));
445        }
446        out.push(line);
447        out.push(format!("         {}", s(f, "message")));
448        if let Some(did) = f["did_you_mean"].as_str() {
449            out.push(format!("         did you mean: {did}"));
450        }
451    }
452    if !out.is_empty() {
453        out.push(String::new());
454    }
455    let c = &v["counts"];
456    out.push(format!(
457        "{} feature(s) ({} of every status), {} featured, {} provider(s); modules {}/{}, commands {}/{}, kinds {}/{} named by a feature",
458        c["features"], c["features_all"], c["featured"], c["providers"], c["modules_covered"], c["modules"], c["commands_covered"], c["commands"], c["kinds_covered"], c["kinds"]
459    ));
460    out.push(format!(
461        "{}: {} error(s), {} warning(s)",
462        if v["valid"].as_bool().unwrap_or(false) {
463            "valid"
464        } else {
465            "INVALID"
466        },
467        v["errors"],
468        v["warnings"]
469    ));
470    out.join("\n")
471}

473#[cfg(test)]
474mod tests {
475    use super::*;
476    use serde_json::json;
477
478    /// EPISODE and PROMPTS print what the provider is declared to offer, and a provider that
479    /// declares nothing prints a dash in both, never an empty cell that shifts the columns.
480    #[test]
481    fn the_offers_columns_print_the_declaration_or_a_dash() {
482        let v = json!({ "providers": [
483            {
484                "id": "offering", "title": "Declares both",
485                "offers": { "episode": "hooks", "prompts": "hook" },
486                "bootstraps": [{ "target": "CLAUDE.md" }], "client_config": ".mcp.json",
487                "hooks": ["start", "end"], "scratch_roots": ["/tmp/x"]
488            },
489            { "id": "silent", "title": "Declares nothing" }
490        ]});
491        let text = providers_text(&v);
492        let lines: Vec<&str> = text.lines().collect();
493        assert_eq!(
494            lines[1],
495            format!(
496                "{:<8}  {:<36}  {:<10}  {:<7}  {:<10}  {:<13}  {:<5}  {}",
497                "offering",
498                "Declares both",
499                "hooks",
500                "hook",
501                "CLAUDE.md",
502                ".mcp.json",
503                "start,end",
504                "/tmp/x"
505            )
506        );
507        assert_eq!(
508            lines[2],
509            format!(
510                "{:<8}  {:<36}  {:<10}  {:<7}  {:<10}  {:<13}  {:<5}  {}",
511                "silent", "Declares nothing", "-", "-", "-", "-", "-", "-"
512            )
513        );
514        assert_eq!(lines.last(), Some(&"2 provider(s)"));
515    }
