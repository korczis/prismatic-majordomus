1//! The plan: milestones, issues, the two graphs between them, and everything derived from
2//! them — execution waves, issue status, milestone status, the roadmap order, the active
3//! milestone, the next executable issue and every validation finding.
4//!
5//! **This is a second reader of one set of semantics, not a second set.** `lib/project.awk`
6//! is the incumbent engine: the shell tool, the site generator and the GitHub adapter all
7//! call it, and it has been the only implementation since the model existed. This module
8//! derives the same values in Rust so that the capability registry can answer them — an
9//! agent asking the shared MCP server what to work on next could not be told otherwise —
10//! and the two are held to byte equality by `test/cases/99_plan_capabilities.sh`, which
11//! runs both over every record in this repository and over a fixture built to hit the
12//! edges. Where the two disagree the awk is right; this file is the one that changes.
13//!
14//! The order of derivation below follows `lib/project.awk` step for step, including the
15//! order findings are produced in, because that order is observable: a reader diffing the
16//! two engines diffs a sequence, not a set. Comments name the awk block each step mirrors.
17//!
18//! Nothing here reads a file. Milestones and issues are ordinary declarative objects of the
19//! layer — kinds `milestone` and `issue`, whole-YAML metadata — already discovered, parsed
20//! and schema-validated by the time the index holds them, so building the plan is a walk of
21//! values already in memory. That is what lets it be built on demand, inside the handler of
22//! a plan capability, rather than eagerly when a [`crate::capability::Context`] is composed:
23//! `App::load` has a stated budget and 200 project records have no business inside it.
24

58/// One YAML scalar as the flattener would have written it: the text an awk comparison sees.
59///
60/// `lib/yaml.sh` flattens every value to a string, so `order: 3` is `"3"`, `cancelled: true`
61/// is `"true"` and a missing key is `""`. The index holds the same records as typed JSON,
62/// so the comparisons this module makes have to be made against the same text the awk
63/// compares — otherwise `parallel_safe: false` would be a bool here and the string `"false"`
64/// there, and only one of them would be right.
65///
66/// ```
67/// use majordomus_cli::plan::scalar;
68/// use serde_json::json;
69/// assert_eq!(scalar(Some(&json!(3))), "3");
70/// assert_eq!(scalar(Some(&json!(true))), "true");
71/// assert_eq!(scalar(Some(&json!("x"))), "x");
72/// assert_eq!(scalar(Some(&json!(null))), "");
73/// assert_eq!(scalar(None), "");
74/// ```
75pub fn scalar(v: Option<&Value>) -> String {
76    match v {
77        None | Some(Value::Null) => String::new(),
78        Some(Value::String(s)) => s.clone(),
79        Some(Value::Bool(b)) => b.to_string(),
80        Some(Value::Number(n)) => n.to_string(),
81        Some(other) => other.to_string(),
82    }
83}
84
85fn field(meta: &Value, key: &str) -> String {
86    scalar(meta.get(key))
87}
88
89/// A list field as the flattener's `key.0`, `key.1` … would have carried it: every item as a
90/// scalar, in the file's order. A scalar where a list was expected is one item, which is
91/// what the flattener produces for `scope: apps/` too.
92fn list(meta: &Value, key: &str) -> Vec<String> {
93    match meta.get(key) {
94        None | Some(Value::Null) => Vec::new(),
95        Some(Value::Array(items)) => items.iter().map(|v| scalar(Some(v))).collect(),
96        Some(other) => vec![scalar(Some(other))],
97    }
98}

100/// The `covers` token of every evidence entry that carries one, in file order. The awk
101/// counts the `evidence.N.covers` keys, so an entry without one is not evidence.
102fn evidence_covers(meta: &Value) -> Vec<String> {
103    match meta.get("evidence") {
104        Some(Value::Array(items)) => items
105            .iter()
106            .filter_map(|e| e.get("covers"))
107            .map(|v| scalar(Some(v)))
108            .collect(),
109        _ => Vec::new(),
110    }
111}

396    pub fn warnings(&self) -> usize {
397        self.findings.iter().filter(|f| f.level == "WARN").count()
398    }

400    /// Whether this repository holds a canonical project model at all.
401    pub fn is_present(&self) -> bool {
402        !self.milestones.is_empty() || !self.issues.is_empty()
403    }
404
405    /// The milestone ids in roadmap order: rank, then order, then id. Derived from the
406    /// graph; no list of versions is maintained anywhere.
407    pub fn roadmap(&self) -> Vec<&PlanMilestone> {
408        let mut out: Vec<&PlanMilestone> = self.milestones.iter().collect();
409        out.sort_by(|a, b| {
410            (a.rank, a.order, &a.id)
411                .partial_cmp(&(b.rank, b.order, &b.id))
412                .unwrap_or(std::cmp::Ordering::Equal)
413        });
414        out
415    }
416
417    /// The one issue a worker should take now: the lowest-wave READY issue of the active
418    /// milestone, highest priority first, then id.
419    ///
420    /// The active milestone can have nothing ready while another one does — a milestone
421    /// waiting on its own acceptance evidence, for instance — so the search widens to the
422    /// whole plan rather than answering "none" and sending a worker away from executable
423    /// work. `mj_pj_next_ready` in `lib/project.sh` does exactly this.
424    pub fn next_ready(&self, milestone: Option<&str>) -> Option<&PlanIssue> {
425        let scope = milestone.unwrap_or(self.project.active_milestone.as_str());
426        self.ready_ranked(Some(scope))
427            .or_else(|| self.ready_ranked(None))
428    }
429
430    fn ready_ranked(&self, milestone: Option<&str>) -> Option<&PlanIssue> {
431        self.issues
432            .iter()
433            .filter(|i| i.status == "READY")
434            .filter(|i| milestone.is_none_or(|m| m.is_empty() || i.milestone == m))
435            .min_by_key(|i| (i.wave, priority_rank(&i.priority), i.id.clone()))
436    }

438    /// Build the plan from an index. A walk of values already in memory: no file is opened,
439    /// no subprocess is started, nothing is cached.
440    pub fn build(index: &Index) -> Plan {
441        let (header, milestones, issues) = raws(index);
442        derive(header, milestones, issues)
443    }
444
445    /// The plan as it will read once `transition` has been written to `id`.
446    ///
447    /// The same records, the same [`derive`], one timestamp changed: the status a caller is
448    /// told after a write is computed by the one derivation every reader uses, never
449    /// announced by the writer. This matters because the resulting status is not a function
450    /// of the transition alone — a `done` whose required evidence is absent stays VERIFY,
451    /// and a record marked `cancelled` stays CANCELLED whatever is written to it.
452    ///
453    /// Nothing is read from disk and nothing is written: this is the projection of a write,
454    /// and [`transition`] is what performs one.
455    ///
456    /// ```no_run
457    /// # use majordomus_cli::index::Index;
458    /// use majordomus_cli::plan::{Plan, Transition};
459    /// # fn demo(index: &Index) {
460    /// let before = Plan::build(index);
461    /// let after = Plan::after(index, "I0001", Transition::Done, "2026-09-11T12:00:00Z");
462    /// // The status is derived both times, so an issue whose evidence is incomplete is
463    /// // still VERIFY afterwards and the caller can see that before writing anything.
464    /// assert_eq!(before.issue("I0001").is_some(), after.issue("I0001").is_some());
465    /// # }
466    /// ```
467    pub fn after(index: &Index, id: &str, transition: Transition, now: &str) -> Plan {
468        let (header, milestones, mut issues) = raws(index);
469        if let Some(r) = issues.iter_mut().find(|r| r.id == id) {
470            match transition {
471                Transition::Start => r.started_at = now.to_string(),
472                Transition::Verify => r.verified_at = now.to_string(),
473                Transition::Done => r.completed_at = now.to_string(),
474            }
475        }
476        derive(header, milestones, issues)
477    }
478}

480/// Every record the plan is derived from, as its file declares it.
481///
482/// Split out of [`Plan::build`] so that [`Plan::after`] derives from the same collection
483/// rather than a second reading of the index.
484fn raws(index: &Index) -> (PlanProject, Vec<PlanMilestoneRaw>, Vec<PlanIssueRaw>) {
485    let mut milestones: Vec<PlanMilestoneRaw> = Vec::new();
486    let mut issues: Vec<PlanIssueRaw> = Vec::new();
487    let mut header = PlanProject {
488        name: String::new(),
489        repository: String::new(),
490        default_branch: String::new(),
491        active_milestone: String::new(),
492    };
493    for o in &index.objects {
494        match o.kind.as_str() {
495            MILESTONE => milestones.push(PlanMilestoneRaw::of(&o.identity, &o.metadata)),
496            ISSUE => issues.push(PlanIssueRaw::of(&o.identity, &o.metadata)),
497            PROJECT if o.provenance.path.ends_with("project/project.yaml") => {
498                header.name = field(&o.metadata, "name");
499                header.repository = field(&o.metadata, "repository");
500                header.default_branch = field(&o.metadata, "default_branch");
501            }
502            _ => {}
503        }
504    }
505    // The awk reads the files in `LC_ALL=C sort` order of their names, and the loader
506    // refuses a record whose id disagrees with its file name, so id order is file order.
507    milestones.sort_by(|a, b| a.id.cmp(&b.id));
508    issues.sort_by(|a, b| a.id.cmp(&b.id));
509    (header, milestones, issues)
510}

513    match p {
514        "p0" => 0,
515        "p1" => 1,
516        "p2" => 2,
517        _ => 3,
518    }

544/// Does path `a` contain path `b`, or the reverse? Trailing slashes and `./` are normalised
545/// so that a hand-written scope entry cannot silently miss an overlap. `overlap()` in
546/// `lib/project.awk`.
547///
548/// ```
549/// use majordomus_cli::plan::overlap;
550/// assert!(overlap("apps/cli", "apps/cli/src/main.rs"));
551/// assert!(overlap("./lib/", "lib"));
552/// assert!(!overlap("lib", "libexec"));
553/// ```
554pub fn overlap(a: &str, b: &str) -> bool {
555    let norm = |s: &str| {
556        let s = s.strip_prefix("./").unwrap_or(s);
557        s.trim_end_matches('/').to_string()
558    };
559    let (a, b) = (norm(a), norm(b));
560    a == b || b.starts_with(&format!("{a}/")) || a.starts_with(&format!("{b}/"))
561}

563#[allow(clippy::too_many_lines)] // one derivation, in the order lib/project.awk derives it
564fn derive(mut header: PlanProject, mraw: Vec<PlanMilestoneRaw>, iraw: Vec<PlanIssueRaw>) -> Plan {
565    let mut v = PlanFindings(Vec::new());
566    let iids: Vec<String> = iraw.iter().map(|i| i.id.clone()).collect();
567    let mids: Vec<String> = mraw.iter().map(|m| m.id.clone()).collect();
568    let iseen: BTreeSet<&str> = iids.iter().map(String::as_str).collect();
569    let mseen: BTreeSet<&str> = mids.iter().map(String::as_str).collect();
570    let ix = |id: &str| -> Option<usize> { iraw.iter().position(|r| r.id == id) };
571
572    // --- local facts: is this issue DONE on its own terms?
573    let mut covered = vec![true; iraw.len()];
574    let mut done = vec![false; iraw.len()];
575    for (n, r) in iraw.iter().enumerate() {
576        covered[n] = r
577            .evidence_required
578            .iter()
579            .all(|need| r.evidence_have.contains(need));
580        done[n] = !r.cancelled && !r.completed_at.is_empty() && covered[n];
581    }
582
583    // --- edges, and the errors visible in them
584    let mut edge: BTreeSet<(String, String)> = BTreeSet::new();
585    let mut dependents: Vec<Vec<String>> = vec![Vec::new(); iraw.len()];
586    let mut edges: Vec<PlanEdge> = Vec::new();
587    for (n, r) in iraw.iter().enumerate() {
588        let mut seen_here: BTreeSet<&str> = BTreeSet::new();
589        for d in &r.depends_on {
590            if d == &r.id {
591                v.fail("self_dependency", &r.id, "depends on itself".into());
592                continue;
593            }
594            if !iseen.contains(d.as_str()) {
595                v.fail(
596                    "unknown_dependency",
597                    &r.id,
598                    format!("depends on {d}, which is not an issue"),
599                );
600                continue;
601            }
602            if !seen_here.insert(d.as_str()) {
603                v.warn(
604                    "duplicate_dependency",
605                    &r.id,
606                    format!("names {d} more than once"),
607                );
608                continue;
609            }
610            edge.insert((r.id.clone(), d.clone()));
611            if let Some(dn) = ix(d) {
612                dependents[dn].push(r.id.clone());
613            }
614            edges.push(PlanEdge {
615                from: d.clone(),
616                to: r.id.clone(),
617            });
618        }
619        if r.milestone.is_empty() {
620            v.fail("no_milestone", &r.id, "names no milestone".into());
621        } else if !mseen.contains(r.milestone.as_str()) {
622            v.fail(
623                "unknown_milestone",
624                &r.id,
625                format!("names milestone {}, which does not exist", r.milestone),
626            );
627        }
628        if r.acceptance_criteria == 0 && !r.cancelled {
629            v.fail(
630                "no_acceptance",
631                &r.id,
632                "has no acceptance criteria; an issue without one is a placeholder".into(),
633            );
634        }
635        if r.validation == 0 && !r.cancelled {
636            v.fail("no_validation", &r.id, "names no validation command".into());
637        }
638        if r.evidence_required.is_empty() && !r.cancelled {
639            v.warn(
640                "no_evidence_required",
641                &r.id,
642                "requires no evidence; nothing gates its completion".into(),
643            );
644        }
645        if !r.completed_at.is_empty() && !covered[n] && !r.cancelled {
646            let miss: Vec<&str> = r
647                .evidence_required
648                .iter()
649                .filter(|need| !r.evidence_have.contains(need))
650                .map(String::as_str)
651                .collect();
652            v.warn(
653                "evidence_missing",
654                &r.id,
655                format!(
656                    "completed_at is set but evidence is missing for {}; status stays VERIFY",
657                    miss.join(",")
658                ),
659            );
660        }
661    }
662
663    // --- waves: Kahn layering. A node enters a wave only once every dependency has left,
664    //     so its wave is one past the longest path to it. Nodes that never leave are in or
665    //     downstream of a cycle, and are reported rather than silently given a wave.
666    let mut left = vec![true; iraw.len()];
667    let mut nleft = iraw.len();
668    let mut wave: Vec<Option<u32>> = vec![None; iraw.len()];
669    let mut wavelines: Vec<PlanWave> = Vec::new();
670    let mut layer: u32 = 0;
671    while nleft > 0 {
672        let mut front: Vec<usize> = Vec::new();
673        for (n, r) in iraw.iter().enumerate() {
674            if !left[n] {
675                continue;
676            }
677            let blocked = r.depends_on.iter().any(|d| {
678                d != &r.id && iseen.contains(d.as_str()) && ix(d).is_some_and(|dn| left[dn])
679            });
680            if !blocked {
681                front.push(n);
682            }
683        }
684        if front.is_empty() {
685            let stuck: Vec<&str> = (0..iraw.len())
686                .filter(|n| left[*n])
687                .map(|n| iids[n].as_str())
688                .collect();
689            v.fail(
690                "cycle",
691                "graph",
692                format!(
693                    "a dependency cycle prevents these issues from ever becoming ready: {}",
694                    stuck.join(" ")
695                ),
696            );
697            break;
698        }
699        let mut line: Vec<String> = Vec::new();
700        for n in front {
701            wave[n] = Some(layer);
702            left[n] = false;
703            nleft -= 1;
704            line.push(iids[n].clone());
705        }
706        wavelines.push(PlanWave {
707            wave: layer,
708            issues: line,
709        });
710        layer += 1;
711    }
712
713    // --- issue status
714    let mut status: Vec<String> = vec![String::new(); iraw.len()];
715    let mut blocked_by: Vec<Vec<String>> = vec![Vec::new(); iraw.len()];
716    for (n, r) in iraw.iter().enumerate() {
717        // Every declared dependency, not every edge: a duplicate is a warning, and the awk
718        // walks the declaration here, so a record naming one twice blocks on it twice.
719        let bb: Vec<String> = r
720            .depends_on
721            .iter()
722            .filter(|d| {
723                *d != &r.id && iseen.contains(d.as_str()) && ix(d).is_some_and(|dn| !done[dn])
724            })
725            .cloned()
726            .collect();
727        let st = if r.cancelled {
728            "CANCELLED"
729        } else if done[n] {
730            "DONE"
731        } else if !r.completed_at.is_empty() || !r.verified_at.is_empty() {
732            "VERIFY"
733        } else if !r.started_at.is_empty() {
734            "ACTIVE"
735        } else if !bb.is_empty() {
736            "BLOCKED"
737        } else {
738            "READY"
739        };
740        if !bb.is_empty() && matches!(st, "DONE" | "ACTIVE" | "VERIFY") {
741            v.fail(
742                "premature_execution",
743                &r.id,
744                format!("is {st} while {} is not DONE", bb.join(",")),
745            );
746        }
747        status[n] = st.to_string();
748        blocked_by[n] = bb;
749        if !left[n] && wave[n].is_none() && st != "CANCELLED" {
750            wave[n] = Some(0);
751        }
752    }
753
754    // --- the milestone graph: validated before anything derives from it, under the same
755    //     rules the issue graph lives under, one level up.
756    let mut medge: BTreeSet<(String, String)> = BTreeSet::new();
757    let mut mdependents: Vec<Vec<String>> = vec![Vec::new(); mraw.len()];
758    let mut medges: Vec<PlanEdge> = Vec::new();
759    let mx = |id: &str| -> Option<usize> { mraw.iter().position(|r| r.id == id) };
760    for r in &mraw {
761        for d in &r.depends_on {
762            if d == &r.id {
763                v.fail("milestone_self_dependency", &r.id, "requires itself".into());
764                continue;
765            }
766            if !mseen.contains(d.as_str()) {
767                v.fail(
768                    "milestone_unknown_dependency",
769                    &r.id,
770                    format!("requires {d}, which is not a milestone here"),
771                );
772                continue;
773            }
774            if medge.contains(&(r.id.clone(), d.clone())) {
775                v.warn(
776                    "milestone_duplicate_dependency",
777                    &r.id,
778                    format!("requires {d} more than once"),
779                );
780                continue;
781            }
782            medge.insert((r.id.clone(), d.clone()));
783            if let Some(dn) = mx(d) {
784                mdependents[dn].push(r.id.clone());
785            }
786            medges.push(PlanEdge {
787                from: d.clone(),
788                to: r.id.clone(),
789            });
790        }
791    }
792
793    // --- roadmap rank: Kahn layering over the milestone graph. Rank is what orders the
794    //     roadmap; `order` only breaks ties inside one rank.
795    let mut mleft = vec![true; mraw.len()];
796    let mut mnleft = mraw.len();
797    let mut rank: Vec<u32> = vec![0; mraw.len()];
798    let mut mlayer: u32 = 0;
799    while mnleft > 0 {
800        let mut front: Vec<usize> = Vec::new();
801        for (n, r) in mraw.iter().enumerate() {
802            if !mleft[n] {
803                continue;
804            }
805            let blocked = r.depends_on.iter().any(|d| {
806                medge.contains(&(r.id.clone(), d.clone())) && mx(d).is_some_and(|dn| mleft[dn])
807            });
808            if !blocked {
809                front.push(n);
810            }
811        }
812        if front.is_empty() {
813            break;
814        }
815        for n in front {
816            rank[n] = mlayer;
817            mleft[n] = false;
818            mnleft -= 1;
819        }
820        mlayer += 1;
821    }
822    if mnleft > 0 {
823        let cyc: Vec<&str> = (0..mraw.len())
824            .filter(|n| mleft[*n])
825            .map(|n| mids[n].as_str())
826            .collect();
827        v.fail(
828            "milestone_cycle",
829            &cyc.join(" "),
830            "milestone dependencies form a cycle; no roadmap order exists".into(),
831        );
832        for n in 0..mraw.len() {
833            if mleft[n] {
834                rank[n] = 0;
835            }
836        }
837    }
838
839    // --- milestone status
840    let mut mstatus: Vec<String> = vec![String::new(); mraw.len()];
841    let mut counts: Vec<PlanCounts> = Vec::with_capacity(mraw.len());
842    let mut mempty = vec![false; mraw.len()];
843    for (n, r) in mraw.iter().enumerate() {
844        let c = tally(&iraw, &status, &r.id);
845        let (nd, nr, na, nv) = (
846            *c.by_status.get("DONE").unwrap_or(&0),
847            *c.by_status.get("READY").unwrap_or(&0),
848            *c.by_status.get("ACTIVE").unwrap_or(&0),
849            *c.by_status.get("VERIFY").unwrap_or(&0),
850        );
851        let mcov = r
852            .evidence_required
853            .iter()
854            .all(|need| r.evidence_have.contains(need));
855        let (tot, req) = (c.total, c.required);
856        let ms = if r.cancelled {
857            "CANCELLED"
858        } else if !r.superseded_by.is_empty() {
859            "SUPERSEDED"
860        } else if tot == 0 && !r.evidence_required.is_empty() && mcov {
861            "DONE"
862        } else if tot == 0 {
863            "PLANNED"
864        } else if req > 0 && nd == req && mcov {
865            "DONE"
866        } else if req > 0 && nd == req {
867            "VERIFY"
868        } else if na > 0 || nv > 0 || nd > 0 {
869            "ACTIVE"
870        } else if nr == 0 {
871            "BLOCKED"
872        } else {
873            "PLANNED"
874        };
875        mstatus[n] = ms.to_string();
876        mempty[n] = tot == 0;
877        if req > 0 && nd == req && !mcov {
878            let miss: Vec<&str> = r
879                .evidence_required
880                .iter()
881                .filter(|need| !r.evidence_have.contains(need))
882                .map(String::as_str)
883                .collect();
884            v.warn(
885                "milestone_evidence_missing",
886                &r.id,
887                format!(
888                    "every issue is DONE but milestone evidence is missing for {}; status stays VERIFY",
889                    miss.join(",")
890                ),
891            );
892        }
893        counts.push(c);
894    }
895
896    // --- the gate. A milestone whose required outcomes are not yet real cannot be
897    //     executable, whatever its own issues say. Run in rank order, so a dependency is
898    //     always decided before the milestone requiring it and the block cascades.
899    let mut mblocked: Vec<Vec<String>> = vec![Vec::new(); mraw.len()];
900    for r in 0..=mlayer {
901        for (n, m) in mraw.iter().enumerate() {
902            if rank[n] != r {
903                continue;
904            }
905            let block: Vec<String> = m
906                .depends_on
907                .iter()
908                .filter(|d| medge.contains(&(m.id.clone(), (*d).clone())))
909                .filter(|d| mx(d).is_some_and(|dn| mstatus[dn] != "DONE"))
910                .cloned()
911                .collect();
912            mblocked[n] = block.clone();
913            if block.is_empty() {
914                continue;
915            }
916            if mstatus[n] == "CANCELLED" || mstatus[n] == "SUPERSEDED" {
917                continue;
918            }
919            if matches!(mstatus[n].as_str(), "DONE" | "VERIFY" | "ACTIVE") {
920                v.fail(
921                    "milestone_premature",
922                    &m.id,
923                    format!("is {} while {} is not DONE", mstatus[n], block.join(",")),
924                );
925            }
926            mstatus[n] = "BLOCKED".into();
927        }
928    }
929
930    for (n, m) in mraw.iter().enumerate() {
931        if !mempty[n] {
932            continue;
933        }
934        if matches!(mstatus[n].as_str(), "DONE" | "CANCELLED" | "SUPERSEDED") {
935            continue;
936        }
937        if !mblocked[n].is_empty() {
938            continue;
939        }
940        v.warn(
941            "empty_milestone",
942            &m.id,
943            "is reachable and has no issues; it is an outcome nobody is executing".into(),
944        );
945    }
946
947    // --- the gate reaches the work. An issue inside a milestone the gate has blocked is not
948    //     executable, whatever its own dependencies say: the outcome it belongs to is not
949    //     reachable yet.
950    for (n, r) in iraw.iter().enumerate() {
951        if r.milestone.is_empty() {
952            continue;
953        }
954        let Some(mn) = mx(&r.milestone) else { continue };
955        if mblocked[mn].is_empty() {
956            continue;
957        }
958        if status[n] == "CANCELLED" || status[n] == "DONE" {
959            continue;
960        }
961        if status[n] == "ACTIVE" || status[n] == "VERIFY" {
962            v.fail(
963                "premature_execution",
964                &r.id,
965                format!(
966                    "is {} while its milestone {} waits on {}",
967                    status[n],
968                    r.milestone,
969                    mblocked[mn].join(",")
970                ),
971            );
972        }
973        status[n] = "BLOCKED".into();
974        blocked_by[n].push(format!("milestone:{}", r.milestone));
975    }
976
977    // counts follow the statuses, so a blocked milestone cannot report ready work
978    for (n, m) in mraw.iter().enumerate() {
979        if mblocked[n].is_empty() {
980            continue;
981        }
982        counts[n] = tally(&iraw, &status, &m.id);
983    }
984
985    // --- scope conflict: two issues that could run together but touch the same paths.
986    //     Reported conservatively — an uncertain overlap serialises rather than races.
987    for a in 0..iraw.len() {
988        for b in (a + 1)..iraw.len() {
989            if !matches!(status[a].as_str(), "READY" | "ACTIVE") {
990                continue;
991            }
992            if !matches!(status[b].as_str(), "READY" | "ACTIVE") {
993                continue;
994            }
995            if wave[a] != wave[b] {
996                continue;
997            }
998            'pair: for x in &iraw[a].scope {
999                for y in &iraw[b].scope {
1000                    if overlap(x, y) {
1001                        v.warn(
1002                            "scope_conflict",
1003                            &iraw[a].id,
1004                            format!(
1005                                "shares {x} with {}; they may not run concurrently",
1006                                iraw[b].id
1007                            ),
1008                        );
1009                        break 'pair;
1010                    }
1011                }
1012            }
1013        }
1014    }
1015
1016    // --- the active milestone: the lowest-ranked unblocked milestone that is ACTIVE, else
1017    //     the lowest-ranked unblocked one not finished.
1018    let mut best: Option<usize> = None;
1019    for pass in 1..=2 {
1020        for n in 0..mraw.len() {
1021            if mstatus[n] == "DONE" || mstatus[n] == "CANCELLED" {
1022                continue;
1023            }
1024            if pass == 1 && mstatus[n] != "ACTIVE" {
1025                continue;
1026            }
1027            if !mblocked[n].is_empty() {
1028                continue;
1029            }
1030            let o = i64::from(rank[n]) * 1_000_000 + order_of(&mraw[n].order);
1031            if best.is_none_or(|b| o < i64::from(rank[b]) * 1_000_000 + order_of(&mraw[b].order)) {
1032                best = Some(n);
1033            }
1034        }
1035        if best.is_some() {
1036            break;
1037        }
1038    }
1039    header.active_milestone = best.map(|n| mids[n].clone()).unwrap_or_default();
1040
1041    // --- reconciliation: a status assigned but not declared would reach a surface as a
1042    //     count nobody renders and a label nobody has a colour for.
1043    for (n, r) in iraw.iter().enumerate() {
1044        if !ISSUE_STATUSES.contains(&status[n].as_str()) {
1045            v.fail(
1046                "undeclared_status",
1047                &r.id,
1048                format!(
1049                    "is {}, which the issue status vocabulary does not declare",
1050                    status[n]
1051                ),
1052            );
1053        }
1054    }
1055    for (n, m) in mraw.iter().enumerate() {
1056        if !MILESTONE_STATUSES.contains(&mstatus[n].as_str()) {
1057            v.fail(
1058                "undeclared_status",
1059                &m.id,
1060                format!(
1061                    "is {}, which the milestone status vocabulary does not declare",
1062                    mstatus[n]
1063                ),
1064            );
1065        }
1066    }
1067
1068    // ---------------------------------------------------------------- assemble
1069    edges.sort();
1070    edges.dedup();
1071    medges.sort();
1072    medges.dedup();
1073
1074    let issues: Vec<PlanIssue> = iraw
1075        .iter()
1076        .enumerate()
1077        .map(|(n, r)| PlanIssue {
1078            id: r.id.clone(),
1079            milestone: r.milestone.clone(),
1080            status: status[n].clone(),
1081            wave: wave[n].unwrap_or(0),
1082            priority: r.priority.clone(),
1083            profile: r.profile.clone(),
1084            parallel_safe: r.parallel_safe != "false",
1085            title: r.title.clone(),
1086            slug: r.slug.clone(),
1087            depends_on: r.depends_on.clone(),
1088            blocked_by: blocked_by[n].clone(),
1089            dependents: dependents[n].clone(),
1090            scope: r.scope.clone(),
1091            objective: r.objective.clone(),
1092            evidence_have: u32::try_from(r.evidence_have.len()).unwrap_or(u32::MAX),
1093            evidence_need: u32::try_from(r.evidence_required.len()).unwrap_or(u32::MAX),
1094            started_at: r.started_at.clone(),
1095            verified_at: r.verified_at.clone(),
1096            completed_at: r.completed_at.clone(),
1097        })
1098        .collect();
1099
1100    let milestones: Vec<PlanMilestone> = mraw
1101        .iter()
1102        .enumerate()
1103        .map(|(n, r)| PlanMilestone {
1104            id: r.id.clone(),
1105            status: mstatus[n].clone(),
1106            order: order_of(&r.order),
1107            priority: r.priority.clone(),
1108            title: r.title.clone(),
1109            slug: r.slug.clone(),
1110            version: r.version.clone(),
1111            rank: rank[n],
1112            depends_on: r
1113                .depends_on
1114                .iter()
1115                .filter(|d| medge.contains(&(r.id.clone(), (*d).clone())))
1116                .cloned()
1117                .collect(),
1118            blocked_by: mblocked[n].clone(),
1119            dependents: mdependents[n].clone(),
1120            claims: r.claims.clone(),
1121            counts: counts[n].clone(),
1122            issues: iraw
1123                .iter()
1124                .filter(|i| i.milestone == r.id)
1125                .map(|i| i.id.clone())
1126                .collect(),
1127            outcome: r.outcome.clone(),
1128        })
1129        .collect();
1130
1131    Plan {
1132        project: header,
1133        statuses: PlanVocabulary {
1134            issue: ISSUE_STATUSES.iter().map(|s| (*s).to_string()).collect(),
1135            milestone: MILESTONE_STATUSES
1136                .iter()
1137                .map(|s| (*s).to_string())
1138                .collect(),
1139        },
1140        milestones,
1141        issues,
1142        waves: wavelines,
1143        edges,
1144        milestone_edges: medges,
1145        findings: v.0,
1146    }
1147}

1149/// `order` as a number: the awk compares it with `+ 0`, which makes anything unparseable
1150/// zero. A milestone with no `order` is not an error, it is order zero.
1151fn order_of(raw: &str) -> i64 {
1152    raw.trim().parse::<i64>().unwrap_or(0)
1153}

1155/// The counts of one milestone: the two denominators, then one entry per declared status.
1156/// `mcounts()` in `lib/project.awk`, and computed the same twice — once from the statuses
1157/// before the gate and once after it, because the gate changes statuses underneath it.
1158fn tally(iraw: &[PlanIssueRaw], status: &[String], milestone: &str) -> PlanCounts {
1159    let mut by: BTreeMap<String, u32> = ISSUE_STATUSES
1160        .iter()
1161        .map(|s| ((*s).to_string(), 0))
1162        .collect();
1163    let mut total = 0u32;
1164    for (n, r) in iraw.iter().enumerate() {
1165        if r.milestone != milestone {
1166            continue;
1167        }
1168        total += 1;
1169        *by.entry(status[n].clone()).or_insert(0) += 1;
1170    }
1171    let cancelled = *by.get("CANCELLED").unwrap_or(&0);
1172    PlanCounts {
1173        total,
1174        required: total - cancelled,
1175        by_status: by,
1176    }
1177}

1388// ---------------------------------------------------------------- transitions
1389//
1390// The write half of the plan, and the first mutating capability of the development
1391// lifecycle. Everything above this line derives; everything below it changes one field of
1392// one record and records that it did.
1393//
1394// `lib/plan.sh`'s `mj_plan_transition` has been the only implementation of these three
1395// moves. This is the second, and the contract it is held to is byte equality of the file it
1396// writes — `test/cases/133_plan_transition.sh` performs the same transition through both
1397// engines over the same fixture and diffs the record. The guards are stated here in the
1398// same order the shell states them, because the *first* refusal a caller meets is part of
1399// the behaviour: an issue that is both blocked and missing its evidence must be told about
1400// the blocker, which is the thing it can act on.
1401
1402/// One lifecycle move.
1403///
1404/// Three, not four: `evidence` appends a record rather than moving an issue, and it belongs
1405/// with the record it proves rather than with the moves it unblocks.
1406///
1407/// ```
1408/// use majordomus_cli::plan::Transition;
1409///
1410/// // The wire spelling is the one a caller types on the command line.
1411/// let t: Transition = serde_json::from_str("\"done\"").unwrap();
1412/// assert_eq!(t, Transition::Done);
1413/// assert_eq!(serde_json::to_string(&Transition::Start).unwrap(), "\"start\"");
1414///
1415/// // Each move owns the field it stamps and the event it records; nothing else decides
1416/// // either, so a new move cannot be added without saying both.
1417/// for (t, field, event) in [
1418///     (Transition::Start, "started_at", "plan_start"),
1419///     (Transition::Verify, "verified_at", "plan_verify"),
1420///     (Transition::Done, "completed_at", "plan_done"),
1421/// ] {
1422///     assert_eq!(t.field(), field);
1423///     assert_eq!(t.event(), event);
1424/// }
1425/// ```
1426#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
1427#[serde(rename_all = "lowercase")]
1428// `session::state::Transition` already publishes the bare name, and one schema component
1429// cannot mean two things: the move an issue makes is named for the plan it belongs to.
1430#[schemars(rename = "PlanTransition")]
1431pub enum Transition {
1432    /// Execution began. Refused unless the issue is READY.
1433    Start,
1434    /// Implementation is complete and the evidence is still outstanding.
1435    Verify,
1436    /// Complete. Refused while anything it waits on is unfinished, or while any evidence
1437    /// its own record requires is absent.
1438    Done,
1439}
1440
1441impl Transition {
1442    /// The record field this move stamps, and the only field it writes besides `updated_at`.
1443    ///
1444    /// The status is never stored: an issue records what happened to it and the status
1445    /// follows. So a move owns a timestamp, not a state, and this names which one.
1446    ///
1447    /// ```
1448    /// use majordomus_cli::plan::Transition;
1449    /// assert_eq!(Transition::Start.field(), "started_at");
1450    /// assert_eq!(Transition::Done.field(), "completed_at");
1451    /// ```
1452    pub fn field(self) -> &'static str {
1453        match self {
1454            Self::Start => "started_at",
1455            Self::Verify => "verified_at",
1456            Self::Done => "completed_at",
1457        }
1458    }
1459
1460    /// The event `share/events.yaml` declares for it.
1461    ///
1462    /// The four plan events are spelled with an underscore rather than the dot the rest of
1463    /// the vocabulary uses. They are registered as written, because the ledger of this
1464    /// repository already holds dozens of them and a rename would orphan every one.
1465    ///
1466    /// ```
1467    /// use majordomus_cli::plan::Transition;
1468    /// assert_eq!(Transition::Verify.event(), "plan_verify");
1469    /// assert!(Transition::Done.event().starts_with("plan_"));
1470    /// ```
1471    pub fn event(self) -> &'static str {
1472        match self {
1473            Self::Start => "plan_start",
1474            Self::Verify => "plan_verify",
1475            Self::Done => "plan_done",
1476        }
1477    }
1478}

1480/// Why a transition did not happen.
1481///
1482/// Each variant carries what the caller needs to act, never only that it failed: which
1483/// status the issue is actually in, which dependency holds it, which evidence token is
1484/// missing. The messages are the shell's messages.
1485///
1486/// ```
1487/// use majordomus_cli::plan::TransitionError;
1488///
1489/// // A refusal says what is in the way, because that is the thing the caller can change.
1490/// let refused = TransitionError::Refused("I0001 is DONE, not READY".into());
1491/// assert_eq!(refused.to_string(), "I0001 is DONE, not READY");
1492///
1493/// // A missing issue is a different answer from an illegal move, and callers map them to
1494/// // different statuses: not_found rather than refused.
1495/// let missing = TransitionError::NoSuchIssue("I9999".into());
1496/// assert_eq!(missing.to_string(), "no issue 'I9999'");
1497/// assert_ne!(missing, refused);
1498/// ```
1499#[derive(Debug, Clone, PartialEq, Eq)]
1500pub enum TransitionError {
1501    /// No issue of that id.
1502    NoSuchIssue(String),
1503    /// The move is illegal from the status the issue is in.
1504    Refused(String),
1505    /// The record could not be rewritten.
1506    Write {
1507        /// The record.
1508        path: std::path::PathBuf,
1509        /// What the filesystem said.
1510        reason: String,
1511    },
1512}
1513
1514impl std::fmt::Display for TransitionError {
1515    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
1516        match self {
1517            Self::NoSuchIssue(id) => write!(f, "no issue '{id}'"),
1518            Self::Refused(m) => f.write_str(m),
1519            Self::Write { path, reason } => write!(f, "{}: {reason}", path.display()),
1520        }
1521    }
1522}

1526/// Whether `transition` is legal for `issue` as the plan reads now, and why not when it is
1527/// not.
1528///
1529/// Separate from the write so that a surface can offer or withhold a control without
1530/// performing it: the Cockpit disables a button by asking this, and never by reimplementing
1531/// the rule.
1532///
1533/// ```no_run
1534/// # use majordomus_cli::index::Index;
1535/// use majordomus_cli::plan::{check, Plan, Transition, TransitionError};
1536/// # fn demo(index: &Index) {
1537/// let plan = Plan::build(index);
1538/// let now = "2026-09-11T12:00:00Z";
1539/// // Offer the control only when the model would accept it, and show the reason when not.
1540/// let offered = check(index, &plan, "I0001", Transition::Start, now).is_ok();
1541///
1542/// // An issue the plan does not hold is not found, never merely refused: the two are
1543/// // different answers and a surface maps them to different statuses.
1544/// let missing = check(index, &plan, "no-such-issue", Transition::Start, now);
1545/// assert_eq!(missing, Err(TransitionError::NoSuchIssue("no-such-issue".into())));
1546///
1547/// // Asking twice never changes anything: this is the question, not the move.
1548/// assert_eq!(offered, check(index, &plan, "I0001", Transition::Start, now).is_ok());
1549/// # }
1550/// ```
1551pub fn check(
1552    index: &Index,
1553    plan: &Plan,
1554    id: &str,
1555    transition: Transition,
1556    now: &str,
1557) -> Result<(), TransitionError> {
1558    let issue = plan
1559        .issue(id)
1560        .ok_or_else(|| TransitionError::NoSuchIssue(id.to_string()))?;
1561    match transition {
1562        Transition::Start => {
1563            if issue.status != "READY" {
1564                let waiting = if issue.blocked_by.is_empty() {
1565                    String::new()
1566                } else {
1567                    format!(" (waiting on {})", issue.blocked_by.join(" "))
1568                };
1569                return Err(TransitionError::Refused(format!(
1570                    "{id} is {}, not READY{waiting}",
1571                    issue.status
1572                )));
1573            }
1574        }
1575        Transition::Verify => {
1576            if !matches!(issue.status.as_str(), "ACTIVE" | "VERIFY") {
1577                return Err(TransitionError::Refused(format!(
1578                    "{id} is {}; only an ACTIVE issue can move to VERIFY",
1579                    issue.status
1580                )));
1581            }
1582        }
1583        Transition::Done => {
1584            // The blocker first: an issue that is both blocked and short of evidence is
1585            // told about the blocker, because that is the one it cannot act on alone.
1586            if !issue.blocked_by.is_empty() {
1587                return Err(TransitionError::Refused(format!(
1588                    "{id} cannot be DONE while {} is not DONE",
1589                    issue.blocked_by.join(" ")
1590                )));
1591            }
1592            // Whether the evidence is complete is not a count — a record may carry two
1593            // entries covering one token and none covering another — and the rule for it
1594            // already exists inside `derive`. So rather than restate it, ask what the plan
1595            // would say: write the stamp in memory and see whether the issue reaches DONE.
1596            // An issue that stays VERIFY with `completed_at` set is exactly the record
1597            // `derive` reports as "completed_at is set but evidence is missing".
1598            let after = Plan::after(index, id, transition, now);
1599            if after.issue(id).is_some_and(|i| i.status != "DONE") {
1600                return Err(TransitionError::Refused(format!(
1601                    "{id} has evidence for {} of {} requirement(s) (run: majordomus plan evidence {id} --covers <token> ...)",
1602                    issue.evidence_have, issue.evidence_need
1603                )));
1604            }
1605        }
1606    }
1607    Ok(())
1608}

1610/// Perform one transition: check it, stamp the record, append the event.
1611///
1612/// The order is the shell's and is not an implementation detail. The guard runs against the
1613/// plan as it reads now; the record is rewritten; only then is the event appended. A ledger
1614/// line therefore never describes a change that did not reach the file, which is the
1615/// property that lets the record be read as history rather than as intent.
1616///
1617/// `path` is the issue's own file, as the index recorded it. It is passed in rather than
1618/// derived from the id here, because the index already knows where every record lives and a
1619/// second convention for locating one is how two readers come to disagree about which file
1620/// is the record.
1621///
1622/// ```no_run
1623/// # use std::path::Path;
1624/// # use majordomus_cli::index::Index;
1625/// # use majordomus_cli::git::GitState;
1626/// use majordomus_cli::ledger::{self, Vocabulary};
1627/// use majordomus_cli::plan::{transition, Plan, Transition};
1628/// # fn demo(index: &Index, root: &Path, record: &Path, git: &GitState) {
1629/// let plan = Plan::build(index);
1630/// let vocabulary = Vocabulary::load(&root.join("share/events.yaml")).unwrap();
1631/// let at = ledger::now();
1632/// // The guard runs first, the record is rewritten, and only then is the event appended:
1633/// // a ledger line never describes a change that did not reach the file.
1634/// transition(root, record, index, &plan, "I0001", Transition::Start, &at, &vocabulary, git)
1635///     .expect("a READY issue may start");
1636/// # }
1637/// ```
1638#[allow(clippy::too_many_arguments)]
1639pub fn transition(
1640    root: &std::path::Path,
1641    path: &std::path::Path,
1642    index: &Index,
1643    plan: &Plan,
1644    id: &str,
1645    transition: Transition,
1646    now: &str,
1647    vocabulary: &crate::ledger::Vocabulary,
1648    git: &crate::git::GitState,
1649) -> Result<(), TransitionError> {
1650    check(index, plan, id, transition, now)?;
1651    let text = std::fs::read_to_string(path).map_err(|e| TransitionError::Write {
1652        path: path.to_path_buf(),
1653        reason: e.to_string(),
1654    })?;
1655    // Both fields, in the order the shell writes them: the move's own stamp, then
1656    // `updated_at`. A record whose `updated_at` preceded its `completed_at` would be a
1657    // record that says it was finished before it was last touched.
1658    let text = set_field(&text, transition.field(), now);
1659    let text = set_field(&text, "updated_at", now);
1660    std::fs::write(path, text).map_err(|e| TransitionError::Write {
1661        path: path.to_path_buf(),
1662        reason: e.to_string(),
1663    })?;
1664    crate::ledger::append(
1665        root,
1666        vocabulary,
1667        git,
1668        now,
1669        transition.event(),
1670        &[("issue", id.to_string())],
1671    )
1672    .map_err(|e| TransitionError::Write {
1673        path: root.join(".ai/local/state/ledger.jsonl"),
1674        reason: e.to_string(),
1675    })?;
1676    Ok(())
1677}

1679/// Set one top-level scalar field of a record, exactly as `mj_pj_set_field` in
1680/// `lib/project.sh` does.
1681///
1682/// The rule is small and every part of it is load-bearing:
1683///
1684/// * a line beginning `key:` is replaced in place, keeping its position in the file, so a
1685///   record's field order survives every write;
1686/// * otherwise the field is inserted immediately *before* `evidence:`, because evidence is
1687///   a block and a scalar appended after it would be read as one of its entries;
1688/// * with no `evidence:` line, the field is appended at the end.
1689///
1690/// Only the first `evidence:` line is an anchor, and matching is on the line's start, which
1691/// is what `grep -qE "^$key:"` and the awk beside it do. A nested key of the same name is
1692/// indented and is therefore not matched — the same blindness, deliberately, because the
1693/// two engines must be blind in the same places.
1694fn set_field(text: &str, key: &str, value: &str) -> String {
1695    let prefix = format!("{key}:");
1696    let line = format!("{key}: {value}");
1697    if text.lines().any(|l| l.starts_with(&prefix)) {
1698        let mut out = String::with_capacity(text.len() + line.len());
1699        for l in text.lines() {
1700            if l.starts_with(&prefix) {
1701                out.push_str(&line);
1702            } else {
1703                out.push_str(l);
1704            }
1705            out.push('\n');
1706        }
1707        return out;
1708    }
1709    let mut out = String::with_capacity(text.len() + line.len() + 1);
1710    let mut inserted = false;
1711    for l in text.lines() {
1712        if !inserted && l.starts_with("evidence:") {
1713            out.push_str(&line);
1714            out.push('\n');
1715            inserted = true;
1716        }
1717        out.push_str(l);
1718        out.push('\n');
1719    }
1720    if !inserted {
1721        out.push_str(&line);
1722        out.push('\n');
1723    }
1724    out
1725}
