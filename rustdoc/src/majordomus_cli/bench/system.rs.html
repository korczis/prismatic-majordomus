1//! The transports' own operations, benchmarked beside the capabilities: the MCP protocol
2//! methods a client calls before any tool, and the HTTP projection's infrastructure
3//! routes. Declared once, here, as data; the projection turns them into targets and the
4//! coverage into requirements.
5
6use schemars::JsonSchema;
7use serde::{Deserialize, Serialize};
8
9/// One system target.
10#[derive(
11    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
12)]
13#[serde(rename_all = "snake_case")]
14pub enum SystemTarget {
15    /// A fresh `majordomus mcp` process: spawn, `initialize`, first `tools/list`.
16    McpProcessCold,
17    /// `initialize` on a running process.
18    McpInitialize,
19    /// `ping`.
20    McpPing,
21    /// `tools/list`.
22    McpToolsList,
23    /// `resources/list`.
24    McpResourcesList,
25    /// `resources/read` of the first declarative resource.
26    McpResourcesRead,
27    /// `GET /`: the topology as JSON, what a client that asks for no page receives.
28    HttpIndex,
29    /// `GET /` with `Accept: text/html`: the home page, rendered from the topology.
30    HttpHome,
31    /// `GET /openapi.json`.
32    HttpOpenApi,
33    /// `GET /swagger`.
34    HttpSwagger,
35    /// `GET /cockpit`: the Cockpit's landing page, rendered.
36    HttpCockpitOverview,
37    /// `GET /cockpit/capabilities`: the whole registry as a table, the widest page there is.
38    HttpCockpitCapabilities,
39    /// `GET /cockpit/graphs/registry`: a page whose content is a derived graph.
40    HttpCockpitGraph,
41}
42
43impl SystemTarget {
44    /// Every system target, in a stable order.
45    pub const ALL: [SystemTarget; 13] = [
46        SystemTarget::McpProcessCold,
47        SystemTarget::McpInitialize,
48        SystemTarget::McpPing,
49        SystemTarget::McpToolsList,
50        SystemTarget::McpResourcesList,
51        SystemTarget::McpResourcesRead,
52        SystemTarget::HttpIndex,
53        SystemTarget::HttpHome,
54        SystemTarget::HttpOpenApi,
55        SystemTarget::HttpSwagger,
56        SystemTarget::HttpCockpitOverview,
57        SystemTarget::HttpCockpitCapabilities,
58        SystemTarget::HttpCockpitGraph,
59    ];
60
61    /// The stable key results and baselines use.
62    pub fn key(self) -> &'static str {
63        match self {
64            SystemTarget::McpProcessCold => "system.mcp.process_cold",
65            SystemTarget::McpInitialize => "system.mcp.initialize",
66            SystemTarget::McpPing => "system.mcp.ping",
67            SystemTarget::McpToolsList => "system.mcp.tools_list",
68            SystemTarget::McpResourcesList => "system.mcp.resources_list",
69            SystemTarget::McpResourcesRead => "system.mcp.resources_read",
70            SystemTarget::HttpIndex => "system.http.index",
71            SystemTarget::HttpHome => "system.http.home",
72            SystemTarget::HttpOpenApi => "system.http.openapi",
73            SystemTarget::HttpSwagger => "system.http.swagger",
74            SystemTarget::HttpCockpitOverview => "system.http.cockpit_overview",
75            SystemTarget::HttpCockpitCapabilities => "system.http.cockpit_capabilities",
76            SystemTarget::HttpCockpitGraph => "system.http.cockpit_graph",
77        }
78    }
79
80    /// The transport the target belongs to.
81    pub fn transport(self) -> super::Transport {
82        match self {
83            SystemTarget::McpProcessCold
84            | SystemTarget::McpInitialize
85            | SystemTarget::McpPing
86            | SystemTarget::McpToolsList
87            | SystemTarget::McpResourcesList
88            | SystemTarget::McpResourcesRead => super::Transport::Mcp,
89            SystemTarget::HttpIndex
90            | SystemTarget::HttpHome
91            | SystemTarget::HttpOpenApi
92            | SystemTarget::HttpSwagger
93            | SystemTarget::HttpCockpitOverview
94            | SystemTarget::HttpCockpitCapabilities
95            | SystemTarget::HttpCockpitGraph => super::Transport::Http,
96        }
97    }
98
99    /// One line of what is measured.
100    pub fn description(self) -> &'static str {
101        match self {
102            SystemTarget::McpProcessCold => {
103                "spawn a majordomus mcp process, initialize, first tools/list"
104            }
105            SystemTarget::McpInitialize => "initialize on a running process",
106            SystemTarget::McpPing => "ping: the protocol round trip with nothing behind it",
107            SystemTarget::McpToolsList => "tools/list",
108            SystemTarget::McpResourcesList => "resources/list",
109            SystemTarget::McpResourcesRead => "resources/read of the first declarative resource",
110            SystemTarget::HttpIndex => "GET / (the topology as JSON)",
111            SystemTarget::HttpHome => {
112                "GET / with Accept: text/html (the home page, rendered from the topology)"
113            }
114            SystemTarget::HttpOpenApi => "GET /openapi.json",
115            SystemTarget::HttpSwagger => "GET /swagger (the Swagger UI shell)",
116            SystemTarget::HttpCockpitOverview => {
117                "GET /cockpit (the Cockpit's landing page, server-rendered)"
118            }
119            SystemTarget::HttpCockpitCapabilities => {
120                "GET /cockpit/capabilities (every capability as a table: the widest page)"
121            }
122            SystemTarget::HttpCockpitGraph => {
123                "GET /cockpit/graphs/registry (a page whose content is a derived graph)"
124            }
125        }
126    }
