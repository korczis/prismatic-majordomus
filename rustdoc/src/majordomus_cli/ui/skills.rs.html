1//! Skills: what an agent can be asked by name, and where each name comes from (ADR 0165).
2//! Read from `skills.status` (the layer's workflow skills) and `editors.status` (every file
3//! an editor or an agent reads, the derived domain and command skills among them); nothing
4//! here names a skill, so a skill the projection derives is a row here as soon as it exists.

18const SKILLS: Source = Source::of("skills.status");
19const EDITORS: Source = Source::of("editors.status");

21/// The Skills screen.
22pub fn screen(ctx: &Context) -> View {
23    let answers = ask_all(ctx, &[SKILLS, EDITORS]);
24    let (skills, editors) = (&answers[&SKILLS], &answers[&EDITORS]);
25
26    let top = hero(
27        "sparkles",
28        "violet",
29        "Skills",
30        "What an agent can be asked by name, and where each name comes from.",
31        "Workflow skills are written once under the layer; command and domain skills are derived from the capability registry. Every editor and agent reads the same projection.",
32        vec![
33            kpi_count("Workflow skills", &skills.read("/skills")),
34            kpi_count("Projected files", &editors.read("/artifacts")),
35            kpi_count("Editors and agents", &editors.read("/adapters")),
36            kpi_count("Findings", &editors.read("/diagnostics")),
37        ],
38    );
39
40    let grid = el("div")
41        .class("mj-ui-grid")
42        .child(wide(panel(
43            "Workflow skills",
44            "The layer's own procedures, one directory each under the skills section.",
45            workflows(skills),
46            &[skills],
47        )))
48        .child(wide(panel(
49            "By name, in every editor and agent",
50            "Each name as it is typed, the adapter that reads it, and what it projects: a workflow skill, a capability, or a domain of the registry.",
51            projected(editors),
52            &[editors],
53        )));
54
55    View::new(
56        "Skills",
57        el("div").class("mj-ui-stack").child(top).child(grid),
58    )
59}

61fn workflows(answer: &Answer) -> El {
62    unanswered(answer).unwrap_or_else(|| {
63        let rows = list(answer, "/skills")
64            .iter()
65            .map(|s| {
66                el("tr")
67                    .attr("id", format!("skill-{}", text(s, "id")))
68                    .child(
69                        el("td")
70                            .child(el("span").class("mj-ui-mono mj-ui-dim").text(text(s, "id")))
71                            .child(el("span").class("mj-ui-cell-title").text(text(s, "title"))),
72                    )
73                    .child(el("td").class("mj-ui-dim").text(text(s, "description")))
74                    .child(el("td").child(state(s, "status")))
75            })
76            .collect::<Vec<_>>();
77        table(
78            &["Skill", "When to use it", "Status"],
79            rows,
80            "The layer declares no workflow skill.",
81        )
82    })
83}

85/// The name a file is typed by: its skill directory for a `SKILL.md`, the path otherwise.
86fn typed(path: &str) -> String {
87    match path.strip_suffix("/SKILL.md") {
88        Some(dir) => format!("/{}", dir.rsplit('/').next().unwrap_or(dir)),
89        None => path.to_string(),
90    }
91}
92
93fn projected(answer: &Answer) -> El {
94    unanswered(answer).unwrap_or_else(|| {
95        let mut rows = Vec::new();
96        for adapter in list(answer, "/adapters") {
97            let title = text(&adapter, "title");
98            let capabilities = adapter
99                .get("capabilities")
100                .and_then(Value::as_array)
101                .cloned()
102                .unwrap_or_default();
103            for capability in capabilities {
104                let what = text(&capability, "id");
105                for path in capability
106                    .get("artifacts")
107                    .and_then(Value::as_array)
108                    .into_iter()
109                    .flatten()
110                    .filter_map(Value::as_str)
111                {
112                    rows.push(
113                        el("tr")
114                            .child(el("td").class("mj-ui-mono").text(typed(path)))
115                            .child(el("td").text(title.clone()))
116                            .child(el("td").class("mj-ui-mono mj-ui-dim").text(what.clone()))
117                            .child(el("td").child(state(&capability, "support"))),
118                    );
119                }
120            }
121        }
122        table(
123            &["Name", "Read by", "Projects", "Support"],
124            rows,
125            "No editor or agent reads anything from this repository.",
126        )
127    })
128}

134    #[test]
135    fn a_skill_is_named_as_it_is_typed() {
136        assert_eq!(typed(".claude/skills/mj-mesh/SKILL.md"), "/mj-mesh");

141    fn answer(value: Result<Value, String>) -> Answer {
142        Answer {
143            source: SKILLS,
144            route: None,
145            exposure: Default::default(),
146            value: value.map(Into::into),
147            took: std::time::Duration::ZERO,
148            standing: super::super::source::Standing::NOW,
149        }
150    }
151
152    /// Each workflow skill is a row anchored by its id, carrying its title, its description
153    /// and its status; a skill with no status is `unknown`, never an empty badge.
154    #[test]
155    fn a_workflow_skill_is_a_row_with_its_title_description_and_status() {
156        let html = workflows(&answer(Ok(serde_json::json!({
157            "skills": [
158                {
159                    "id": "evolve",
160                    "title": "Evolve",
161                    "description": "Turn observations into knowledge.",
162                    "status": "Active"
163                },
164                {"id": "implement", "title": "Implement", "description": "Carry a change."}
165            ]
166        }))))
167        .render();
168        for said in [
169            "id=\"skill-evolve\"",
170            "id=\"skill-implement\"",
171            "Evolve",
172            "Turn observations into knowledge.",
173            "Carry a change.",
174            "active",
175            "unknown",
176        ] {
177            assert!(html.contains(said), "{said:?} is missing from:\n{html}");
178        }
179        assert!(!html.contains("declares no workflow skill"), "{html}");
180    }
181
182    /// No skill is said in words, and a source that could not answer says why.
183    #[test]
184    fn no_workflow_skill_is_said_and_an_unanswered_source_says_why() {
185        let none = workflows(&answer(Ok(serde_json::json!({"skills": []})))).render();
186        assert!(
187            none.contains("The layer declares no workflow skill."),
188            "{none}"
189        );
190        assert!(!none.contains("<tr"), "{none}");
191
192        let failed = workflows(&answer(Err("the index is rebuilding".into()))).render();
193        assert!(
194            failed.contains("skills.status could not answer: the index is rebuilding"),
195            "{failed}"
196        );
197    }
