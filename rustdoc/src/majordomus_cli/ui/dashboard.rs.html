1//! The Dashboard: how the delivery stands, on one screen.
2//!
3//! Its first paint asks only sources that answer in well under a second on this
4//! repository, all at once: `repository.info`, `health.report`, `release.version`,
5//! `distribution.releases`, `plan.status`, `plan.roadmap` and `mesh.peers`. Measured one at a
6//! time on 2026-10-11, the slowest was `release.version` at 0.5-0.8 s. The pull-request queue
7//! (`integration.queue`) took 50 s for 1.2 MB, so it is a panel of its own at
8//! [`PULL_REQUESTS_PANEL`]: the script pulls it in once the screen has painted, and a reader
9//! without JavaScript follows its link.
10

36/// The sources the first paint asks, together.
37pub const FIRST_PAINT: &[Source] = &[REPOSITORY, HEALTH, VERSION, RELEASES, PLAN, ROADMAP, MESH];

39/// How many rows a panel shows before it says how many there are in all.
40pub const ROWS: usize = 8;

42/// The Dashboard screen.
43pub fn screen(ctx: &Context) -> View {
44    let answers = ask_all(ctx, FIRST_PAINT);
45    let a = |s: Source| &answers[&s];
46
47    let git = |p: &'static str| a(REPOSITORY).read(p).text();
48    let lead = match (git("/repository/git/branch"), git("/repository/git/head")) {
49        (Some(branch), Some(head)) => format!(
50            "{branch} at {} · working tree {}",
51            &head[..12.min(head.len())],
52            git("/repository/git/working_tree")
53                .as_deref()
54                .unwrap_or("unknown")
55        ),
56        _ => "The repository's version control could not be read.".into(),
57    };
58
59    let top = hero(
60        "home",
61        "blue",
62        "Dashboard",
63        "How the delivery stands, read from the repository itself.",
64        &lead,
65        vec![
66            // the repository's verdict, not the whole report's: the whole one is `unknown`
67            // in every process that is no server, a published snapshot's reader among them,
68            // and the panel this leads to still lists every dimension with its own word
69            kpi_to(
70                kpi_state("Repository health", &a(HEALTH).read("/repository")),
71                "#health",
72            ),
73            kpi_to(
74                kpi("Released", &a(RELEASES).read("/latest/version")),
75                super::delivery::ROUTE,
76            ),
77            kpi_to(
78                kpi("Next release", &a(VERSION).read("/next")),
79                super::delivery::ROUTE,
80            ),
81            kpi_to(
82                kpi("Issues ready", &a(PLAN).read("/counts/by_status/READY")),
83                super::work::ROUTE,
84            ),
85            kpi_to(
86                kpi("Issues blocked", &a(PLAN).read("/counts/by_status/BLOCKED")),
87                super::roadmap::ROUTE,
88            ),
89            kpi_to(mesh_kpi(a(MESH)), super::mesh::ROUTE),
90        ],
91    );
92
93    let grid = el("div")
94        .class("mj-ui-grid")
95        .child(plan_panel(a(ROADMAP), a(PLAN)))
96        .child(issues_panel(a(PLAN)))
97        .child(wide(queue_placeholder()))
98        .child(releases_panel(a(RELEASES), a(VERSION), Some(ROWS)))
99        .child(mesh_panel(a(MESH)))
100        .child(health_panel(a(HEALTH)));
101
102    View::new(
103        "Dashboard",
104        el("div").class("mj-ui-stack").child(top).child(grid),
105    )
106}

121/// The milestones still open, in the plan's own order, each with the issues it has done
122/// out of those it holds.
123pub fn plan_panel(roadmap: &Answer, plan: &Answer) -> El {
124    let body = unanswered(roadmap).unwrap_or_else(|| {
125        let open: Vec<Value> = list(roadmap, "/milestones")
126            .into_iter()
127            .filter(|m| m.get("status").and_then(Value::as_str) != Some("DONE"))
128            .collect();
129        let rows = open
130            .iter()
131            .take(ROWS)
132            .map(milestone_row)
133            .collect::<Vec<_>>();
134        el("div")
135            .child(el("p").class("mj-ui-line").text(format!(
136                "Now {} · next {}",
137                shown(&roadmap.read("/now")),
138                shown(&roadmap.read("/next"))
139            )))
140            .child(table(
141                &["Milestone", "State", "Issues done"],
142                rows,
143                "No milestone is open.",
144            ))
145            .children(more(
146                ROWS.min(open.len()),
147                open.len(),
148                Some(super::roadmap::ROUTE),
149            ))
150    });
151    panel_to(
152        "Roadmap",
153        "The milestones still open, in the plan's own order.",
154        Some(super::roadmap::ROUTE),
155        body,
156        &[roadmap, plan],
157    )
158}

193/// The label of the tile counting merges: the window is the queue's own (`throughput/
194/// window_days`), so the label says what the queue counted rather than a window of its own.
195pub fn merged_label(queue: &Answer) -> String {
196    match queue.read("/throughput/window_days").count() {
197        Some(days) => format!("Merged, {days} days"),
198        None => "Merged, over a window the queue does not state".into(),
199    }
200}

202/// The first two reasons of a pull request, and how many more there are: a row has room for
203/// two, and its inspector holds them all.
204pub fn first_reasons(reasons: &[String]) -> String {
205    match reasons.len() {
206        0 => "none".into(),
207        1 | 2 => reasons.join(", "),
208        n => format!("{} and {} more", reasons[..2].join(", "), n - 2),
209    }
210}

212/// What was published, newest first, as a timeline: the first `limit` of them, saying how
213/// many more there are, or every one of them when there is no limit.
214pub fn releases_panel(releases: &Answer, version: &Answer, limit: Option<usize>) -> El {
215    let body = unanswered(releases).unwrap_or_else(|| {
216        let all = list(releases, "/releases");
217        let limit = limit.unwrap_or(all.len());
218        let entries = all
219            .iter()
220            .take(limit)
221            .map(|r| {
222                let commit = text(r, "commit");
223                let record = text(r, "metadata_url");
224                (
225                    text(r, "published_at"),
226                    format!("v{}", value(r, "version")),
227                    el("span")
228                        .class("mj-ui-dim")
229                        .text(format!(
230                            "{} · {}",
231                            value(r, "channel"),
232                            commit.get(..10).unwrap_or(if commit.is_empty() {
233                                "no commit"
234                            } else {
235                                &commit
236                            })
237                        ))
238                        // the release's own record, at the address the distribution gives
239                        .when(!record.is_empty(), |s| {
240                            s.text(" · ")
241                                .child(super::view::link(record.clone(), "its record"))
242                        }),
243                )
244            })
245            .collect::<Vec<_>>();
246        el("div")
247            .child(el("p").class("mj-ui-line").text(format!(
248                "Declared {} · the commits since the last release imply a {} release",
249                shown(&version.read("/declared")),
250                shown(&version.read("/bump"))
251            )))
252            .child(if entries.is_empty() {
253                nothing("Nothing has been released yet.")
254            } else {
255                timeline(entries)
256            })
257            .children(more(
258                limit.min(all.len()),
259                all.len(),
260                Some(super::delivery::ROUTE),
261            ))
262    });
263    panel_to(
264        "Releases",
265        "What was published, newest first.",
266        Some(super::delivery::ROUTE),
267        body,
268        &[releases, version],
269    )
270}

272/// The mesh as one figure: how many machines the runtime knows, and where no runtime is
273/// active in this process, that — never a count. A process without a mesh runtime knows no
274/// machine because it asked nobody, and `0` there would read as a repository nobody works on.
275fn mesh_kpi(mesh: &Answer) -> El {
276    match mesh.read("/active").state {
277        Availability::Available(Value::Bool(false)) => kpi("Mesh active", &mesh.read("/active")),
278        _ => kpi_count("Mesh machines", &mesh.read("/machines")),
279    }
280}

312/// One machine as a table row: its name, whether it is this one, and the liveness of each
313/// of its runtimes as the mesh runtime observed it.
314pub fn machine_row(m: &Value) -> El {
315    let local = m.get("local").and_then(Value::as_bool);
316    let words: Vec<String> = m
317        .get("runtimes")
318        .and_then(Value::as_array)
319        .map(|rs| {
320            rs.iter()
321                .filter_map(|r| r.get("liveness").and_then(Value::as_str))
322                .map(str::to_string)
323                .collect()
324        })
325        .unwrap_or_default();
326    el("tr")
327        .child(el("td").class("mj-ui-mono").text(text(m, "name")))
328        .child(el("td").text(match local {
329            Some(true) => "this machine",
330            Some(false) => "remote",
331            None => "unknown",
332        }))
333        .child(
334            el("td")
335                .class("mj-ui-badges")
336                .children(words.iter().map(|w| word_badge(w))),
337        )
338}

397/// The queue: its tallies, its throughput and the pull requests in rank order.
398pub fn queue_panel(queue: &Answer, selected: Option<u64>) -> El {
399    let body = unanswered(queue).unwrap_or_else(|| {
400        if queue.read("/observed").state != Availability::Available(Value::Bool(true)) {
401            return el("p").class("mj-ui-empty").text(format!(
402                "The forge was not observed in this checkout: {}",
403                shown(&queue.read("/reason"))
404            ));
405        }
406        let tally = |p: &'static str| queue.read(p);
407        let tallies = strip(vec![
408            kpi("Open", &tally("/queue/tallies/open")),
409            kpi(
410                "Needs refresh",
411                &tally("/queue/tallies/by_disposition/needs_refresh"),
412            ),
413            kpi(
414                "Needs repair",
415                &tally("/queue/tallies/by_disposition/needs_repair"),
416            ),
417            kpi(
418                "Conflicting",
419                &tally("/queue/tallies/by_disposition/conflicting"),
420            ),
421            kpi(&merged_label(queue), &tally("/throughput/merges")),
422            kpi("Merges per day", &tally("/throughput/merges_per_day")),
423        ]);
424        let assessments = list(queue, "/queue/assessments");
425        let rows = assessments
426            .iter()
427            .take(ROWS * 2)
428            .map(|a| {
429                let number = a
430                    .get("number")
431                    .and_then(Value::as_u64)
432                    .map_or_else(|| "unknown".to_string(), |n| format!("#{n}"));
433                let reasons: Vec<String> = a
434                    .get("reasons")
435                    .and_then(Value::as_array)
436                    .map(|rs| {
437                        rs.iter()
438                            .filter_map(Value::as_str)
439                            .map(str::to_string)
440                            .collect()
441                    })
442                    .unwrap_or_default();
443                let n = a.get("number").and_then(Value::as_u64);
444                el("tr")
445                    .attr_if("id", n.map(|n| format!("pr-{n}")))
446                    .child(el("td").class("mj-ui-mono").text(number))
447                    .child(
448                        el("td").child(
449                            el("a")
450                                .class("mj-ui-row-link")
451                                .attr(
452                                    "href",
453                                    n.map_or_else(
454                                        || super::prs::ROUTE.to_string(),
455                                        |n| format!("{}?pr={n}#pr", super::prs::ROUTE),
456                                    ),
457                                )
458                                .child(el("span").class("mj-ui-cell-title").text(text(a, "title"))),
459                        ),
460                    )
461                    .child(el("td").child(state(a, "disposition")))
462                    .child(el("td").text(value(a, "lane")))
463                    .child(
464                        el("td")
465                            .class("mj-ui-mono mj-ui-dim")
466                            .text(first_reasons(&reasons)),
467                    )
468            })
469            .collect::<Vec<_>>();
470        let observed = queue
471            .read("/queue/observed_at")
472            .text()
473            .map(|t| t.replace('T', " "))
474            .unwrap_or_else(|| "at an unknown time".into());
475        let detail = selected.map(|n| {
476            list(queue, "/queue/assessments")
477                .into_iter()
478                .find(|a| a.get("number").and_then(Value::as_u64) == Some(n))
479                .map_or_else(
480                    || {
481                        nothing(&format!(
482                            "#{n} is not among the open pull requests the queue observed."
483                        ))
484                    },
485                    |a| pull_request_detail(&a),
486                )
487        });
488        el("div")
489            .class("mj-ui-stack")
490            .children(detail)
491            .child(tallies)
492            .child(table(
493                &["PR", "Title", "Disposition", "Lane", "Why"],
494                rows,
495                "No pull request is open.",
496            ))
497            .children(more(
498                (ROWS * 2).min(assessments.len()),
499                assessments.len(),
500                Some(super::prs::ROUTE),
501            ))
502            .child(
503                el("p")
504                    .class("mj-ui-line")
505                    .text(format!("Observed {observed} ({})", day(&observed))),
506            )
507    });
508    panel_to(
509        "Pull-request queue",
510        "What the integration lane would merge, repair or wait for, and why.",
511        Some(super::prs::ROUTE),
512        body,
513        &[queue],
514    )
515}

517/// One pull request as the queue assessed it: what it is, every gate with its verdict, the
518/// evidence each verdict was read from, why it stands where it stands, what would move it,
519/// and the factors of its rank. Read from the queue's own answer, so it costs nothing more.
520fn pull_request_detail(a: &Value) -> El {
521    let s = |k: &str| text(a, k);
522    // an assessment without a number is shown as unknown, never as #0
523    let number = a
524        .get("number")
525        .and_then(Value::as_u64)
526        .map_or_else(|| "unknown".to_string(), |n| n.to_string());
527    let strings = |k: &str| -> Vec<String> {
528        a.get(k)
529            .and_then(Value::as_array)
530            .map(|v| {
531                v.iter()
532                    .filter_map(Value::as_str)
533                    .map(str::to_string)
534                    .collect()
535            })
536            .unwrap_or_default()
537    };
538    let gates = el("ul").class("mj-ui-checks").children(
539        a.get("gates")
540            .and_then(Value::as_array)
541            .cloned()
542            .unwrap_or_default()
543            .iter()
544            .map(|g| {
545                let passed = g.get("passed").and_then(Value::as_bool);
546                el("li")
547                    .class("mj-ui-check")
548                    .child(
549                        el("span")
550                            .class("mj-ui-check-title mj-ui-mono")
551                            .text(text(g, "gate")),
552                    )
553                    .child(word_badge(match passed {
554                        Some(true) => "pass",
555                        Some(false) => "fail",
556                        None => "unknown",
557                    }))
558            }),
559    );
560    let evidence = a
561        .get("evidence")
562        .and_then(Value::as_array)
563        .cloned()
564        .unwrap_or_default()
565        .iter()
566        .map(|e| {
567            el("tr")
568                .child(el("td").class("mj-ui-mono").text(text(e, "kind")))
569                .child(el("td").child(state(e, "status")))
570                .child(el("td").class("mj-ui-dim").text(text(e, "detail")))
571        })
572        .collect::<Vec<_>>();
573    let factors = a
574        .get("rank_factors")
575        .and_then(Value::as_object)
576        .map(|m| {
577            m.iter()
578                .map(|(k, v)| {
579                    el("div")
580                        .class("mj-ui-fact")
581                        .child(el("dt").text(k.replace('_', " ")))
582                        .child(el("dd").class("mj-ui-mono").text(match v {
583                            Value::String(s) => s.clone(),
584                            other => other.to_string(),
585                        }))
586                })
587                .collect::<Vec<_>>()
588        })
589        .unwrap_or_default();
590    el("section")
591        .class("mj-ui-detail")
592        .attr("id", "pr")
593        .attr("aria-label", format!("Pull request #{number}"))
594        .child(
595            el("div")
596                .class("mj-ui-detail-head")
597                .child(el("p").class("mj-ui-mono mj-ui-dim").text(format!(
598                    "#{number} · {} · {} → {}",
599                    s("author"),
600                    s("head_ref"),
601                    s("base_ref")
602                )))
603                .child(el("h3").class("mj-ui-detail-title").text(s("title")))
604                .child(
605                    el("p")
606                        .class("mj-ui-badges")
607                        .child(state(a, "disposition"))
608                        .child(state(a, "lane"))
609                        .child(state(a, "risk")),
610                ),
611        )
612        .child(el("p").class("mj-ui-line").text(format!(
613            "Next: {}",
614            if s("next_action").is_empty() {
615                "nothing to do".into()
616            } else {
617                s("next_action")
618            }
619        )))
620        .child(
621            el("p")
622                .class("mj-ui-mono mj-ui-dim")
623                .text(strings("reasons").join(" · ")),
624        )
625        .child(el("h4").class("mj-ui-panel-title").text("Gates"))
626        .child(gates)
627        .child(el("h4").class("mj-ui-panel-title").text("Evidence"))
628        .child(table(
629            &["Kind", "Verdict", "What it says"],
630            evidence,
631            "No evidence was recorded.",
632        ))
633        .child(el("h4").class("mj-ui-panel-title").text("Paths it authors"))
634        .child(el("p").class("mj-ui-mono mj-ui-dim").text({
635            let paths = strings("authored_paths");
636            if paths.is_empty() {
637                "none".into()
638            } else {

701        let repo = SyntheticRepository::new(Shape::default()).expect("a synthetic repository");

704        let cite = "read from health.report /repository";

710            .find("mj-ui-kpi-source")

712        assert!(tile.contains("Repository health"), "{tile}");
713        // the word shown is the report's own, over the repository's dimensions alone
