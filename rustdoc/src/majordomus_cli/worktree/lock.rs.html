1//! One lock per repository, around every operation that changes the worktree topology.
2//!
3//! Two agents working the same repository is the normal case here, not the exotic one, and
4//! `git worktree add` twice at once on the same destination is how a half-registered
5//! worktree happens. The lock lives under the *common* git directory —
6//! `<git-common-dir>/majordomus/locks/worktrees.lock` — which is the one directory every
7//! linked worktree of a repository shares and which no other repository can see. So the
8//! lock is visible from every worktree, there is exactly one of it, and two different
9//! repositories never wait on each other.
10//!
11//! It is not under `.ai/local/`: that is per-checkout, so a lock there would be one lock per
12//! worktree, which is no lock at all. It is not the MCP server's lease either: that elects
13//! a process to serve, which is a different question with a different lifetime. Reads take
14//! no lock; only a mutation serialises.
15
16use std::fs;
17use std::io::Write;
18use std::path::{Path, PathBuf};
19use std::time::{Duration, Instant, SystemTime, UNIX_EPOCH};
20
21use super::error::{Result, WorktreeError};
22
23/// Where the lock lives under the common git directory.
24pub const LOCK_RELATIVE: &str = "majordomus/locks/worktrees.lock";
25
26/// How long a caller waits for the lock before saying who has it.
27pub const WAIT: Duration = Duration::from_secs(20);
28
29/// A lock file older than this is treated as abandoned even when its owner cannot be judged
30/// (a file with no owner written in it, an owner of another machine's boot that might still
31/// be alive). Generous, because a migration of a large worktree across devices can take a
32/// while.
33pub const STALE_AFTER: Duration = Duration::from_secs(900);
34
35/// The held lock. Dropping it releases it; a panic or an early return therefore cannot leave
36/// the repository locked. A process that dies without dropping it — a signal, `SIGKILL`, a
37/// crash — leaves the file, and the file names its owner: the next caller sees that the owner
38/// is gone (`Owner::is_gone`) and takes the lock at once, instead of waiting out
39/// `STALE_AFTER` (OpenTTD, 2026-11-22: `worktrees.lock` held by dead pid 74771 until a
40/// person removed it).
41#[derive(Debug)]
42pub struct WorktreeLock {
43    path: PathBuf,
44    token: String,
45}
46
47impl WorktreeLock {
48    /// The lock file for a repository, given its common git directory.
49    pub fn path_for(git_common_dir: &Path) -> PathBuf {
50        git_common_dir.join(LOCK_RELATIVE)
51    }
52
53    /// Take the lock, waiting up to `WAIT` for another holder to finish.
54    ///
55    /// The primitive is `O_CREAT|O_EXCL`, which is atomic on every filesystem this runs on:
56    /// exactly one caller creates the file. A file whose owner is gone, or older than
57    /// `STALE_AFTER`, is removed and the attempt repeats, so a killed process is recovered
58    /// from without a person being asked to delete anything.
59    pub fn acquire(git_common_dir: &Path) -> Result<Self> {
60        Self::acquire_with(git_common_dir, WAIT)
61    }
62
63    /// The same, with the waiting time as an argument, so a test does not wait twenty
64    /// seconds to prove that a held lock is held.
65    pub fn acquire_with(git_common_dir: &Path, wait: Duration) -> Result<Self> {
66        let path = Self::path_for(git_common_dir);
67        if let Some(dir) = path.parent() {
68            fs::create_dir_all(dir).map_err(|e| WorktreeError::io(dir, &e))?;
69        }
70        let token = Owner::this_process().token();
71        let deadline = Instant::now() + wait;
72        loop {
73            match fs::OpenOptions::new()
74                .write(true)
75                .create_new(true)
76                .open(&path)
77            {
78                Ok(mut f) => {
79                    // Best effort: the content is for the person who reads the message, and
80                    // the lock is the file's existence, not its content.
81                    let _ = f.write_all(token.as_bytes());
82                    return Ok(WorktreeLock { path, token });
83                }
84                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
85                    if Self::is_stale(&path) {
86                        Self::reclaim(&path);
87                        continue;
88                    }
89                    if Instant::now() >= deadline {
90                        return Err(WorktreeError::LockUnavailable {
91                            holder: fs::read_to_string(&path).ok().map(|s| s.trim().to_string()),
92                            path,
93                        });
94                    }
95                    std::thread::sleep(Duration::from_millis(50));
96                }
97                Err(e) => return Err(WorktreeError::io(&path, &e)),
98            }
99        }
100    }
101
102    /// Remove the lock file at `path` if it is still stale, deciding and removing as one step.
103    ///
104    /// Judging a file stale and then removing it are two steps, and between them another
105    /// recoverer can remove the same stale file and create its own lock: a second recoverer
106    /// that removed whatever was there then deleted a live lock, and both held it. That was
107    /// rare while a lock was reclaimed only after 900 s; once a dead owner's lock is taken at
108    /// once, every caller that waited behind the dead owner judges it in the same moment. So
109    /// the judgement is repeated, and the removal made, under an exclusive `flock` on a file
110    /// beside the lock (`worktrees.lock.reclaim`): the recoverer that comes second finds the
111    /// lock already gone or a live owner's, and leaves it. That file is never removed, and
112    /// the kernel releases its lock when its holder exits however it exits, so it cannot be
113    /// left held. A recoverer that cannot open it removes the stale file as before.
114    fn reclaim(path: &Path) {
115        let mut name = path.file_name().unwrap_or_default().to_os_string();
116        name.push(".reclaim");
117        // A sidecar that cannot be opened leaves the removal unguarded, as it was before.
118        let _guard = fs::OpenOptions::new()
119            .create(true)
120            .truncate(false)
121            .write(true)
122            .open(path.with_file_name(name))
123            .inspect(|file| {
124                #[cfg(unix)]
125                {
126                    use std::os::unix::io::AsRawFd;
127                    // SAFETY: flock(2) on the descriptor of `file`, which stays open until
128                    // `_guard` drops at the end of this function; closing it releases the
129                    // lock. A lock that cannot be taken leaves the removal unguarded.
130                    unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX) };
131                }
132                #[cfg(not(unix))]
133                let _ = file;
134            });
135        if Self::is_stale(path) {
136            let _ = fs::remove_file(path);
137        }
138    }
139
140    fn is_stale(path: &Path) -> bool {
141        let owner = fs::read_to_string(path).ok();
142        if owner
143            .as_deref()
144            .and_then(Owner::parse)
145            .is_some_and(|o| o.is_gone())
146        {
147            return true;
148        }
149        let Ok(meta) = fs::metadata(path) else {
150            return false;
151        };
152        let Ok(modified) = meta.modified() else {
153            return false;
154        };
155        SystemTime::now()
156            .duration_since(modified)
157            .map(|age| age > STALE_AFTER)
158            .unwrap_or(false)
159    }
160
161    /// The lock file, for a message.
162    pub fn path(&self) -> &Path {
163        &self.path
164    }
165}

167/// Who wrote a lock file: `pid <pid> at <seconds> boot <boot>`, the boot as
168/// [`crate::process::boot_id`] names it. A file written before the boot was recorded reads
169/// `pid <pid> at <seconds>` and is judged by its pid alone.
170///
171/// The lock lives under the common git directory, which is on this machine's disk, as git's
172/// own `index.lock` is; its pid is therefore a pid of this machine.
173#[derive(Debug, Clone, PartialEq, Eq)]
174pub struct Owner {
175    pid: u32,
176    at: u64,
177    boot: Option<String>,
178}
179
180impl Owner {
181    /// This process, now, in this boot.
182    pub fn this_process() -> Self {
183        Owner {
184            pid: std::process::id(),
185            at: SystemTime::now()
186                .duration_since(UNIX_EPOCH)
187                .map(|d| d.as_secs())
188                .unwrap_or(0),
189            boot: crate::process::boot_id(),
190        }
191    }
192
193    /// The text written into the lock file.
194    pub fn token(&self) -> String {
195        match &self.boot {
196            Some(boot) => format!("pid {} at {} boot {boot}", self.pid, self.at),
197            None => format!("pid {} at {}", self.pid, self.at),
198        }
199    }
200
201    /// The owner a lock file names, or `None` for a file that names none (empty, written
202    /// only in part, or not this format).
203    pub fn parse(text: &str) -> Option<Self> {
204        let mut words = text.split_whitespace();
205        let (Some("pid"), Some(pid), Some("at"), Some(at)) =
206            (words.next(), words.next(), words.next(), words.next())
207        else {
208            return None;
209        };
210        let boot = match (words.next(), words.next(), words.next()) {
211            (None, _, _) => None,
212            (Some("boot"), Some(boot), None) => Some(boot.to_string()),
213            _ => return None,
214        };
215        Some(Owner {
216            pid: pid.parse().ok()?,
217            at: at.parse().ok()?,
218            boot,
219        })
220    }
221
222    /// Whether the process that wrote the lock cannot be holding it: it wrote it in another
223    /// boot of this machine, or no process with its pid is alive.
224    pub fn is_gone(&self) -> bool {
225        let other_boot = match (&self.boot, crate::process::boot_id()) {
226            (Some(theirs), Some(ours)) => *theirs != ours,
227            _ => false,
228        };
229        other_boot || !crate::lease::alive(self.pid)
230    }
231}
232
233impl Drop for WorktreeLock {
234    fn drop(&mut self) {
235        // Only remove the file if it is still this process's: a lock reclaimed as stale by
236        // someone else must not then be deleted by the process that lost it.
237        if let Ok(content) = fs::read_to_string(&self.path) {
238            if content.trim() != self.token {
239                return;
240            }
241        }
242        let _ = fs::remove_file(&self.path);
243    }
244}

292    fn plant(dir: &Path, content: &str) -> PathBuf {

323    /// Callers waiting behind a dead owner all judge its lock stale at once; exactly one of
324    /// them takes it at a time. Each holder counts itself in while it holds the lock, and no
325    /// holder may ever find another already counted. Before the judgement and the removal
326    /// were one step, a caller that judged the dead owner's file stale removed the lock the
327    /// first recoverer had just created, and both held it.
328    #[test]
329    fn callers_reclaiming_one_dead_owner_take_the_lock_one_at_a_time() {
330        use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
331        use std::sync::{Arc, Barrier};
332        const TAKERS: usize = 8;
333        let dead = dead_pid();
334        let holders = Arc::new(AtomicUsize::new(0));
335        let shared = Arc::new(AtomicBool::new(false));
336        for _round in 0..25 {
337            let dir = tempfile::tempdir().unwrap();
338            plant(dir.path(), &format!("pid {dead} at 1"));
339            let start = Arc::new(Barrier::new(TAKERS));
340            let takers: Vec<_> = (0..TAKERS)
341                .map(|_| {
342                    let (root, start) = (dir.path().to_path_buf(), Arc::clone(&start));
343                    let (holders, shared) = (Arc::clone(&holders), Arc::clone(&shared));
344                    std::thread::spawn(move || {
345                        start.wait();
346                        let held = WorktreeLock::acquire_with(&root, Duration::from_secs(20))
347                            .expect("every taker gets the lock in turn");
348                        if holders.fetch_add(1, Ordering::SeqCst) != 0 {
349                            shared.store(true, Ordering::SeqCst);
350                        }
351                        std::thread::sleep(Duration::from_millis(2));
352                        holders.fetch_sub(1, Ordering::SeqCst);
353                        drop(held);
354                    })
355                })
356                .collect();
357            for taker in takers {
358                taker.join().expect("a taker finished");
359            }
360            assert!(
361                !shared.load(Ordering::SeqCst),
362                "two callers held the lock at once"
363            );
364            assert!(!WorktreeLock::path_for(dir.path()).exists(), "all released");
365        }
366    }

435        let root = dir.path().to_path_buf();
436        let panicked = std::panic::catch_unwind(move || {

448    /// A process killed while it holds the lock (`SIGKILL`: no drop, no handler) leaves the
449    /// file, and the next caller takes the lock at once rather than after `STALE_AFTER`.
450    #[cfg(unix)]
451    #[test]
452    fn a_holder_killed_outright_does_not_keep_the_lock() {
453        use std::os::unix::process::ExitStatusExt;
454        if let Some(dir) = std::env::var_os(CHILD) {
455            let held = WorktreeLock::acquire(Path::new(&dir)).unwrap();
456            std::mem::forget(held);
457            // SAFETY: ending this process with a signal no handler can take
458            unsafe { libc::raise(libc::SIGKILL) };
459            unreachable!("SIGKILL ends the process");
460        }
461        let dir = tempfile::tempdir().unwrap();
462        let out = std::process::Command::new(std::env::current_exe().unwrap())
463            .args([
464                "--exact",
465                "worktree::lock::tests::a_holder_killed_outright_does_not_keep_the_lock",
466                "--nocapture",
467                "--test-threads",
468                "1",
469            ])
470            .env(CHILD, dir.path())
471            .output()
472            .expect("the child process runs");
473        assert_eq!(out.status.signal(), Some(libc::SIGKILL), "{out:?}");
474        let left = WorktreeLock::path_for(dir.path());
475        assert!(left.exists(), "the killed holder left its lock file");
476        let taken = WorktreeLock::acquire_with(dir.path(), Duration::from_millis(500))
477            .expect("the dead holder's lock is taken at once");
478        drop(taken);
479        assert!(!left.exists());
480    }
