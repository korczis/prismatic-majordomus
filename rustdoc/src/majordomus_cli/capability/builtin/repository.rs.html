18/// The URI under which `repository.info` is read as an MCP resource, and the one URI
19/// `objects.get` answers by executing a query rather than by reading the index.
20pub const REPOSITORY_URI: &str = "majordomus://repository";
21
22/// The repository, its layer, its git state, and the state of this process's index.
23#[derive(Debug, Serialize, Deserialize, JsonSchema)]
24pub struct RepositoryReport {
25    /// The repository: root, layer schema, sections, git state, discovery mode, source classes, kind sources.
26    pub repository: RepositoryInfo,
27    /// `ok` when every discovered file became an object, `degraded` otherwise.
28    pub state: State,
29    /// How many objects the index holds.
30    pub objects: usize,
31    /// Objects per kind.
32    pub kinds: std::collections::BTreeMap<String, usize>,
33    /// Every diagnostic the index produced.
34    pub diagnostics: Vec<Diagnostic>,
35    /// The capability registry, counted.
36    pub capabilities: Summary,
37}
38
39fn repository_info(ctx: &Context, _: Empty) -> Result<RepositoryReport, CapabilityError> {
40    let index = &ctx.index;
41    Ok(RepositoryReport {
42        repository: index.repository.clone(),
43        state: index.state,
44        objects: index.objects.len(),
45        kinds: index
46            .kinds()
47            .into_iter()
48            .map(|(k, v)| (k.to_string(), v))
49            .collect(),
50        diagnostics: index.diagnostics.clone(),
51        capabilities: ctx.registry.summary(),
52    })
53}

55/// The capability `majordomus scope <PATHS>` answers with when it is given paths.
56///
57/// Named here, beside the declaration, because the command module must reach it by
58/// identity: it has no command line of its own to be found by.
59///
60/// ```
61/// use majordomus_cli::capability::builtin::repository::SCOPE_CLASSIFY;
62/// assert_eq!(SCOPE_CLASSIFY, "repository.scope_classify");
63/// ```
64pub const SCOPE_CLASSIFY: &str = "repository.scope_classify";
65
66/// The module.
67pub fn module() -> ModuleDescriptor {
68    module! {
69        id: "repository",
70        title: "Repository",
71        description: "The repository this process serves: its layer, its git state, the state of the index built from it, and its scope: what a worker reads of it and what it never reads.",
72        stability: Stability::BehaviorallyVerified,
73        capabilities: [
74            capability! {
75                id: "repository.info",
76                title: "Repository and index state",
77                description: "The repository root, layer sections, git state, discovery mode, kinds present, every diagnostic, and the capability registry counted.",
78                input: Empty,
79                output: RepositoryReport,
80                stability: Stability::BehaviorallyVerified,
81                exposure: Exposure {
82                    mcp: Some(McpExposure {
83                        tool: Some("majordomus_repository".into()),
84                        resource: Some(McpResource { uri: REPOSITORY_URI.into(), name: "repository".into() }),
85                    }),
86                    http: get("/api/v1/repository"),
87                    cli: None,
88                },
89                tags: ["repository", "introspection"],
90                handler: repository_info,
91            },
92            capability! {
93                id: "repository.scope",
94                title: "The repository scope",
95                description: "The scope declaration as read, where it came from (the repository's own or the distribution's default), and every tracked file tallied against it: how many are in, how many are out for each reason, and which.",
96                input: Empty,
97                output: ScopeReport,
98                stability: Stability::BehaviorallyVerified,
99                exposure: Exposure {
100                    mcp: Some(McpExposure {
101                        tool: Some("majordomus_scope".into()),
102                        resource: Some(McpResource { uri: SCOPE_URI.into(), name: "scope".into() }),
103                    }),
104                    http: get("/api/v1/scope"),
105                    cli: Some(CliExposure { path: vec!["scope".into()] }),
106                },
107                tags: ["repository", "scope", "introspection"],
108                handler: scope_info,
109            },
110            capability! {
111                id: "repository.scope_classify",
112                title: "Judge a path against the scope",
113                description: "Whether a repository-relative path is in or out of the scope, the reason when it is out, and the pattern or limit that decided; an existing file is judged by name, then size, then content.",
114                input: ClassifyInput,
115                output: Classification,
116                stability: Stability::BehaviorallyVerified,
117                // No CLI exposure: `majordomus scope <PATHS>` is one command that answers
118                // with this capability when it is given paths and with `repository.scope`
119                // when it is not, and clap has no `scope classify` subcommand for anybody
120                // to type. The path used to be declared here so that the command module
121                // could find the id with `by_cli`, which made a lookup key look like a
122                // public command line — `capabilities describe`, `docs/generated/cli.md`
123                // and the site's registry all repeated a command that does not exist. The
124                // command module names [`SCOPE_CLASSIFY`] instead, and this exposure says
125                // what is true: the capability is reached through the command above.
126                exposure: Exposure {
127                    mcp: mcp("majordomus_scope_classify"),
128                    http: get("/api/v1/scope/classify"),
129                    cli: None,
130                },
131                tags: ["repository", "scope"],
132                handler: scope_classify,
133            },
134        ],
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    /// The declaration is the only place these names exist, and every projection — the MCP
143    /// tool, the HTTP route, the OpenAPI operation, the benchmark target — is derived from
144    /// it. A refactor that dropped an exposure or renamed a route would still compile, and
145    /// the suites that exercise the behaviour behind it would still pass. This is the
146    /// assertion that would not.
147    #[test]
148    fn the_declaration_yields_the_projections_it_claims() {
149        let m = module();
150        assert_eq!(m.id.as_str(), "repository");
151        let expected: &[(&str, &str, &str)] = &[
152            (
153                "repository.info",
154                "majordomus_repository",
155                "/api/v1/repository",
156            ),
157            ("repository.scope", "majordomus_scope", "/api/v1/scope"),
158            (
159                "repository.scope_classify",
160                "majordomus_scope_classify",
161                "/api/v1/scope/classify",
162            ),
163        ];
164        let ids: Vec<&str> = m
165            .capabilities
166            .iter()
167            .map(|e| e.capability.id.as_str())
168            .collect();
169        let want: Vec<&str> = expected.iter().map(|(id, _, _)| *id).collect();
170        assert_eq!(
171            ids, want,
172            "the module declares a different set of capabilities"
173        );
174        for (executable, (id, tool, path)) in m.capabilities.iter().zip(expected) {
175            let exposure = &executable.capability.exposure;
176            assert_eq!(
177                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
178                Some(*tool),
179                "{id} lost or renamed its MCP tool"
180            );
181            assert_eq!(
182                exposure.http.as_ref().map(|h| h.path.as_str()),
183                Some(*path),
184                "{id} lost or renamed its HTTP route"
185            );
186        }
187    }
