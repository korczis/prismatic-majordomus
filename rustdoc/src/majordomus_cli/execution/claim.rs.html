1//! The mesh claim an execution started for an issue holds while it runs (I4004).
2//!
3//! Two workers that start the same issue at once must not both run it. The mesh already
4//! decides that question for sessions — an exclusive claim that meets a live exclusive claim
5//! of another session is refused, on this runtime and on every runtime it has heard — so an
6//! execution started for an issue asks it the same way a person's session does: through
7//! [`Cooperation::claim`], the function `mesh.claim` calls, as the mesh session
8//! `execution-<id>`, for the issue's declared scope.
9//!
10//! ```text
11//!   queued ──► take ──► held ──────► claimed ──► running ──► final ──► released
12//!                  │                                     └─► unknown_outcome ──► kept
13//!                  ├─► conflict ──► refused  (the holder named; nothing runs)
14//!                  └─► no mesh ───► unclaimed ──► claimed ──► running ...   (said, never a failure)
15//! ```
16//!
17//! The claim is given back when the execution reaches a final state, by the worker that ran
18//! it, and — through `HeldClaim`'s `Drop` — when that worker unwinds instead. An outcome
19//! that is not known keeps the claim: whether the work happened is still to be read, and the
20//! scope is not free until it has been. A kept claim, or one whose process died, ends when
21//! its runtime stops beating, as every mesh claim does. Fencing a holder that comes back
22//! after its claim expired is not done here (I3174).
23//!
24//! ```
25//! use majordomus_cli::execution::claim::{disposition, scope_of, session_of};
26//! use majordomus_cli::execution::{ClaimStanding, ExecutionId, ExecutionState};
27//! let repo = majordomus_cli::synthetic::SyntheticRepository::small().unwrap();
28//! let ctx = repo.context().unwrap();
29//! let id = ExecutionId::fresh();
30//! assert_eq!(session_of(&id), format!("execution-{id}"));
31//! assert_eq!(scope_of(&ctx, "I4004"), ["issue/I4004"]);
32//! assert_eq!(disposition(ExecutionState::Cancelled), ClaimStanding::Released);
33//! ```

44/// The client name an execution's mesh session is opened with.
45pub const CLIENT: &str = "majordomus-execution";

47/// The mesh session an execution claims as: `execution-<id>`, so that a refusal that names
48/// the session names the execution that holds the scope.
49///
50/// ```
51/// use majordomus_cli::execution::{claim::session_of, ExecutionId};
52/// let id = ExecutionId::parse("x-20261023T010203Z-0a1b2c3d").unwrap();
53/// assert_eq!(session_of(&id), "execution-x-20261023T010203Z-0a1b2c3d");
54/// ```
55pub fn session_of(id: &ExecutionId) -> String {
56    format!("execution-{id}")
57}

59/// The scope an issue's claim covers: the paths its record declares under `scope`, or the
60/// issue's pseudo-path `issue/<id>` when the record declares none or is not in this checkout.
61///
62/// ```
63/// use majordomus_cli::execution::claim::scope_of;
64/// use majordomus_cli::synthetic::SyntheticRepository;
65/// let repo = SyntheticRepository::small().unwrap();
66/// let ctx = repo.context().unwrap();
67/// assert_eq!(scope_of(&ctx, "I9999"), ["issue/I9999"], "no such record here");
68/// ```
69pub fn scope_of(ctx: &Context, issue: &str) -> Vec<String> {
70    let declared: Vec<String> = crate::plan::Plan::build(&ctx.index)
71        .issue(issue)
72        .map(|i| i.scope.clone())
73        .unwrap_or_default()
74        .into_iter()
75        .filter(|p| !p.trim().is_empty())
76        .collect();
77    if declared.is_empty() {
78        vec![format!("issue/{issue}")]
79    } else {
80        declared
81    }
82}

84/// What the claim does when the execution reaches `state`: kept for an outcome that is not
85/// known, released for anything else — a final state, or a worker that is leaving.
86///
87/// ```
88/// use majordomus_cli::execution::claim::disposition;
89/// use majordomus_cli::execution::{ClaimStanding, ExecutionState};
90/// assert_eq!(disposition(ExecutionState::Succeeded), ClaimStanding::Released);
91/// assert_eq!(disposition(ExecutionState::Refused), ClaimStanding::Released);
92/// assert_eq!(disposition(ExecutionState::UnknownOutcome), ClaimStanding::Kept);
93/// ```
94pub fn disposition(state: ExecutionState) -> ClaimStanding {
95    if state == ExecutionState::UnknownOutcome {
96        ClaimStanding::Kept
97    } else {
98        ClaimStanding::Released
99    }
100}

102/// How the attempt to take an issue's claim came out: held, unclaimed because no mesh runs,
103/// or refused because another session holds the scope.
104#[derive(Debug)]
105pub(crate) enum Taken {
106    /// The exclusive claim is held; it goes back when the guard is settled or dropped.
107    Held(HeldClaim),
108    /// No mesh cooperation runs in this process: the execution runs without a claim, and
109    /// the claim says why.
110    Unclaimed(IssueClaim),
111    /// The claim was refused: nothing may run.
112    Refused {
113        /// The claim, standing `refused`, with the holder or the reason in its detail.
114        claim: IssueClaim,
115        /// `true` when another session holds the scope (`claim_conflict`); `false` when the
116        /// mesh refused the claim for another reason.
117        conflict: bool,
118        /// The sentence for the execution's error.
119        message: String,
120    },
121}
122
123/// Take the exclusive mesh claim of `issue` for the execution `id`, before it runs: held,
124/// refused with the holder named, or unclaimed with the reason when no mesh cooperation runs in
125/// this process. The tests below hold each branch; a doc example cannot, because a running
126/// mesh is only ever constructed inside `src/mesh/` (scripts/ci/mesh-check).
127pub(crate) fn take(ctx: &Context, id: &ExecutionId, intent: &str, issue: &str) -> Taken {
128    let scope = scope_of(ctx, issue);
129    let Some(cooperation) = ctx.mesh.cooperation() else {
130        let reason = unmeshed(ctx);
131        return Taken::Unclaimed(IssueClaim {
132            standing: ClaimStanding::Unclaimed,
133            scope,
134            detail: Some(format!(
135                "ran without a mesh claim: {reason}; nothing on the mesh kept a second start of {issue} out"
136            )),
137            ..IssueClaim::wanted(issue)
138        });
139    };
140    take_with(cooperation, id, intent, issue, scope)
141}
142
143/// Why no mesh can be asked in this process, in the mesh's own words: without a running
144/// cooperation its status is `CooperationStatus::inactive`, which always carries the reason.
145fn unmeshed(ctx: &Context) -> String {
146    ctx.mesh.cooperation_status().reason.unwrap_or_default()
147}
148
149fn take_with(
150    cooperation: Arc<Cooperation>,
151    id: &ExecutionId,
152    intent: &str,
153    issue: &str,
154    scope: Vec<String>,
155) -> Taken {
156    let session = session_of(id);
157    let info = SessionInfo {
158        session: session.clone(),
159        client: CLIENT.into(),
160        issue: Some(issue.to_string()),
161        intent: Some(intent.to_string()),
162        ..SessionInfo::default()
163    };
164    match cooperation.claim(
165        &info,
166        scope.clone(),
167        Some(intent.to_string()),
168        ClaimMode::Exclusive,
169        Some(issue.to_string()),
170    ) {
171        Ok(written) => Taken::Held(HeldClaim {
172            cooperation,
173            claim: IssueClaim {
174                standing: ClaimStanding::Held,
175                scope,
176                key: Some(written.key),
177                session: Some(session),
178                ..IssueClaim::wanted(issue)
179            },
180            armed: true,
181        }),
182        Err(CooperationError::Conflict(holders)) => {
183            let named = name_holders(holders.iter());
184            let message =
185                format!("claim_conflict: {issue} is claimed on the mesh: {named}; nothing ran");
186            Taken::Refused {
187                claim: IssueClaim {
188                    standing: ClaimStanding::Refused,
189                    scope,
190                    detail: Some(named),
191                    ..IssueClaim::wanted(issue)
192                },
193                conflict: true,
194                message,
195            }
196        }
197        Err(other) => {
198            let message =
199                format!("the mesh claim of {issue} could not be taken: {other}; nothing ran");
200            Taken::Refused {
201                claim: IssueClaim {
202                    standing: ClaimStanding::Refused,
203                    scope,
204                    detail: Some(other.to_string()),
205                    ..IssueClaim::wanted(issue)
206                },
207                conflict: false,
208                message,
209            }
210        }
211    }
212}

214/// The claims that keep a scope out, one sentence: which session holds which claim on what,
215/// and for which issue. What a refusal names, whether it came from taking the claim or from
216/// asking whether it could be taken.
217fn name_holders<'a>(holders: impl Iterator<Item = &'a ClaimView>) -> String {
218    holders
219        .map(|h| {
220            format!(
221                "session {} holds {} on {}{}",
222                h.session,
223                h.key,
224                h.scope.join(", "),
225                h.issue
226                    .as_deref()
227                    .map(|i| format!(" for {i}"))
228                    .unwrap_or_default()
229            )
230        })
231        .collect::<Vec<_>>()
232        .join("; ")
233}

235/// Whether an exclusive claim of an issue would be admitted now, asked without taking it
236/// (I4003).
237///
238/// `devtask.implement` creates a worktree before the execution that holds the claim is
239/// started, and an overlap must refuse it before anything exists. It asks the question the
240/// claim itself will ask — [`admission_conflicts`] over the same state, the same scope and
241/// the exclusive mode — so the two cannot disagree except by a claim taken between them, which
242/// the execution's own claim then refuses, with the holder named, before its handler runs.
243#[derive(Debug, PartialEq, Eq)]
244pub(crate) enum Admission {
245    /// Nothing live and exclusive meets the scope: the execution may take it.
246    Free {
247        /// The scope the claim would cover.
248        scope: Vec<String>,
249    },
250    /// No mesh cooperation runs in this process: nothing on the mesh can be asked, and the
251    /// execution runs unclaimed, as I4004 says it does.
252    Unmeshed {
253        /// The scope the claim would have covered.
254        scope: Vec<String>,
255        /// Why there is no mesh to ask.
256        reason: String,
257    },
258    /// Another session holds the scope: the refusal, starting `claim_conflict:`.
259    Conflict {
260        /// The sentence for the refusal, naming every holder.
261        message: String,
262    },
263}
264
265/// Ask whether the exclusive claim of `issue` would be admitted now; nothing is written.
266pub(crate) fn admission(ctx: &Context, issue: &str) -> Admission {
267    let scope = scope_of(ctx, issue);
268    match ctx.mesh.cooperation() {
269        Some(cooperation) => admission_with(&cooperation, issue, scope),
270        None => Admission::Unmeshed {
271            scope,
272            reason: unmeshed(ctx),
273        },
274    }
275}
276
277fn admission_with(cooperation: &Cooperation, issue: &str, scope: Vec<String>) -> Admission {
278    let state = cooperation.state();
279    // no session of ours holds anything yet, so no claim is passed over as our own
280    let conflicts = admission_conflicts(&state, "", &scope, ClaimMode::Exclusive);
281    if conflicts.is_empty() {
282        Admission::Free { scope }
283    } else {
284        Admission::Conflict {
285            message: format!(
286                "claim_conflict: {issue} is claimed on the mesh: {}; nothing was created",
287                name_holders(conflicts.into_iter())
288            ),
289        }
290    }
291}

293/// An exclusive mesh claim an execution holds. Settled when the execution ends; released by
294/// `Drop` when the worker unwinds without settling it, so a panic does not leave the scope
295/// claimed for as long as the runtime beats.
296pub(crate) struct HeldClaim {
297    cooperation: Arc<Cooperation>,
298    claim: IssueClaim,
299    armed: bool,
300}
301
302impl std::fmt::Debug for HeldClaim {
303    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
304        f.debug_struct("HeldClaim")
305            .field("claim", &self.claim)
306            .field("armed", &self.armed)
307            .finish()
308    }
309}
310
311impl HeldClaim {
312    /// The claim as held: its key, its session and its scope, which the execution's
313    /// `execution.issue_claim` event and the `claimed` holder are made from.
314    pub(crate) fn claim(&self) -> &IssueClaim {
315        &self.claim
316    }
317
318    /// Settle the claim for an execution that reached `state` (`None`: the record is gone):
319    /// released, with its session closed, or kept for an outcome that is not known. The
320    /// answer is the claim as it now stands, for the execution's event.
321    pub(crate) fn settle(mut self, state: Option<ExecutionState>) -> IssueClaim {
322        match state.map(disposition) {
323            Some(ClaimStanding::Kept) => {
324                self.armed = false;
325                IssueClaim {
326                    standing: ClaimStanding::Kept,
327                    detail: Some(
328                        "the outcome is not known, so the claim is kept until a reconciliation reads what happened; it ends when this runtime stops beating"
329                            .into(),
330                    ),
331                    ..self.claim.clone()
332                }
333            }
334            _ => self.release_now(),
335        }
336    }
337
338    fn release_now(&mut self) -> IssueClaim {
339        self.armed = false;
340        let mut problems = Vec::new();
341        if let Some(key) = &self.claim.key {
342            if let Err(e) = self.cooperation.release(key) {
343                problems.push(format!("the release was refused: {e}"));
344            }
345        }
346        if let Some(session) = &self.claim.session {
347            if let Err(e) = self.cooperation.close_session(session) {
348                problems.push(format!("the session did not close: {e}"));
349            }
350        }
351        if !problems.is_empty() {
352            tracing::warn!(claim = ?self.claim.key, "an execution's issue claim was not given back cleanly: {}", problems.join("; "));
353        }
354        IssueClaim {
355            standing: ClaimStanding::Released,
356            detail: (!problems.is_empty()).then(|| problems.join("; ")),
357            ..self.claim.clone()
358        }
359    }
360}
361
362impl Drop for HeldClaim {
363    fn drop(&mut self) {
364        if self.armed {
365            let _ = self.release_now();
366        }
367    }
368}

633                    && s.state == crate::mesh::state::SessionState::Closed)
634        );
635
636        // no session: only the key is released
637        let key = c
638            .claim(
639                &SessionInfo::named("s2", "test"),
640                vec!["issue/I1".into()],
641                None,
642                ClaimMode::Exclusive,
643                Some("I1".into()),

645            .unwrap()
646            .key;
647        let settled = armed(&c, Some(key.clone()), None).settle(Some(ExecutionState::Failed));
648        assert_eq!(settled.standing, ClaimStanding::Released);
649        assert_eq!(settled.detail, None, "given back cleanly");
650        assert!(!live(&c, &key), "released");
651    }
