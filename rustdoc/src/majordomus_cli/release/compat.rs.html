1//! The compatibility engine: how much the public contract moved, and therefore what the
2//! smallest honest version is.
3//!
4//! # The authority this takes away
5//!
6//! Before this module there were two answers to one question. `scripts/ci/version-matches-surface`
7//! compared the registry of two refs and said what the *surface* required; [`super::version`]
8//! counted `feat:` and `fix:` in the commit subjects and said what the *commits* implied, and
9//! that second answer was the one `release bump` actually wrote. So a capability deleted
10//! under a `refactor:` heading was a patch to the writer and a breaking change to the gate,
11//! and the tree could carry an under-versioned public contract until CI happened to look.
12//!
13//! There is one authority here now, and it is the contract:
14//!
15//! ```text
16//!   surface(base) ─┐
17//!                  ├─ diff ─ implied impact ─ policy ─ required impact ─ required version
18//!   surface(head) ─┘                                          │
19//!                                                             ▼
20//!                                              one writer, which may not go under it
21//! ```
22//!
23//! Conventional commits did not go away and are not worthless — they are how a change
24//! explains itself, and they are the whole of the changelog's prose. They are carried here
25//! as [`CommitEvidence`], beside the verdict rather than deciding it, and when they
26//! *understate* what the contract did the plan says so explicitly. That line is the point of
27//! the subsystem: a human label and a measured fact, disagreeing, in the open.
28//!
29//! # Why unknown means major
30//!
31//! `compare_schema` models the schema mutations it can name and refuses to guess at the
32//! rest: a key it does not understand, changed, is [`Impact::Major`] with a diagnostic
33//! saying which pointer and that it is unmodelled. A false major is an argument; a false
34//! patch is a broken caller who finds out by breaking. Prose keys are normalised away first
35//! (see [`super::surface::normalise`]) precisely so that this conservatism costs nothing on
36//! the changes that are not contract changes at all.
37//!
38//! ```
39//! use majordomus_cli::release::compat::{Impact, Mode, Policy};
40//! use majordomus_cli::release::version::Version;
41//!
42//! // Below 1.0.0 a removal costs a minor rather than 1.0.0 — and is still named breaking.
43//! let policy = Policy::for_version(Version::parse("0.5.0").unwrap());
44//! assert_eq!(policy.mode, Mode::Pre1_0Strict);
45//! assert_eq!(policy.required_of(Impact::Major), Impact::Minor);
46//!
47//! // At 1.0.0 the shift ends and the implied impact is the requirement.
48//! let policy = Policy::for_version(Version::parse("1.2.0").unwrap());
49//! assert_eq!(policy.required_of(Impact::Major), Impact::Major);
50//! assert_eq!(policy.required_of(Impact::None), Impact::None, "an unchanged surface owes nothing");
51//! ```

68/// How much a change moves the contract.
69///
70/// Ordered, so that "the declared bump covers what is required" is a comparison rather than
71/// a table:
72///
73/// ```
74/// use majordomus_cli::release::compat::Impact;
75/// assert!(Impact::Major > Impact::Minor);
76/// assert!(Impact::Minor > Impact::Patch);
77/// assert!(Impact::Patch > Impact::None);
78/// ```
79#[derive(
80    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
81)]
82#[serde(rename_all = "snake_case")]
83#[schemars(rename = "ReleaseImpact")]
84pub enum Impact {
85    /// The contract did not move.
86    None,
87    /// It moved behind the boundary: a caller cannot tell.
88    Patch,
89    /// Something arrived; every existing caller still works.
90    Minor,
91    /// Something a caller could hold is gone or changed under it.
92    Major,
93}
94
95impl Impact {
96    /// The word every surface renders: the command line's report, the JSON every machine
97    /// surface answers with, the Cockpit's badge and the gate's refusal all print this and
98    /// never a wording of their own.
99    ///
100    /// ```
101    /// use majordomus_cli::release::compat::Impact;
102    /// assert_eq!(Impact::Major.as_str(), "major");
103    /// assert_eq!(Impact::None.as_str(), "none");
104    /// ```
105    pub fn as_str(self) -> &'static str {
106        match self {
107            Impact::None => "none",
108            Impact::Patch => "patch",
109            Impact::Minor => "minor",
110            Impact::Major => "major",
111        }
112    }
113
114    /// The impact a word names.
115    ///
116    /// Every impact round-trips through its word, and nothing else is one:
117    ///
118    /// ```
119    /// use majordomus_cli::release::compat::Impact;
120    /// assert_eq!(Impact::parse("minor"), Some(Impact::Minor));
121    /// assert_eq!(Impact::parse("enormous"), None);
122    /// ```
123    pub fn parse(word: &str) -> Option<Impact> {
124        match word {
125            "none" => Some(Impact::None),
126            "patch" => Some(Impact::Patch),
127            "minor" => Some(Impact::Minor),
128            "major" => Some(Impact::Major),
129            _ => None,
130        }
131    }
132}

134/// Whether a change added, removed or altered part of the contract.
135///
136/// ```
137/// use majordomus_cli::release::compat::Direction;
138/// assert_eq!(Direction::Removed.sign(), '-');
139/// ```
140#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
141#[serde(rename_all = "snake_case")]
142#[schemars(rename = "ReleaseChangeDirection")]
143pub enum Direction {
144    /// It was not there before.
145    Added,
146    /// It was there before and is not now.
147    Removed,
148    /// It is there in both, and differs.
149    Changed,
150}
151
152impl Direction {
153    /// The sign the reports print beside a movement, so that a list of changes reads as a
154    /// diff does: what arrived, what left, and what stayed and differs.
155    ///
156    /// ```
157    /// use majordomus_cli::release::compat::Direction;
158    /// assert_eq!(Direction::Added.sign(), '+');
159    /// assert_eq!(Direction::Changed.sign(), '~');
160    /// ```
161    pub fn sign(self) -> char {
162        match self {
163            Direction::Added => '+',
164            Direction::Removed => '-',
165            Direction::Changed => '~',
166        }
167    }
168}

170/// Which projection of the contract a change is in.
171///
172/// Not where it was *found* — everything here is found in the registry — but which promise
173/// it is, so that a reader sees "a route is gone" rather than a JSON pointer.
174///
175/// ```
176/// use majordomus_cli::release::compat::SurfaceKind;
177/// assert_eq!(SurfaceKind::Http.as_str(), "route");
178/// ```

180#[serde(rename_all = "snake_case")]
181#[schemars(rename = "ReleaseSurfaceKind")]
182pub enum SurfaceKind {
183    /// The capability itself.
184    Capability,
185    /// Whether calling it changes anything.
186    Kind,
187    /// An MCP tool name.
188    McpTool,
189    /// An MCP resource URI.
190    McpResource,
191    /// An HTTP method and path.
192    Http,
193    /// A command-line path.
194    Cli,
195    /// What a caller may send.
196    Input,
197    /// What a caller is promised back.
198    Output,
199}

226/// One movement of the public contract, with the reason it counts for what it does.
227///
228/// ```
229/// use majordomus_cli::release::compat::{Direction, Impact, SurfaceChange, SurfaceKind};
230/// let change = SurfaceChange {
231///     impact: Impact::Major,
232///     direction: Direction::Removed,
233///     surface: SurfaceKind::Http,
234///     capability: "alpha.list".into(),
235///     id: "GET /api/v1/alpha".into(),
236///     detail: "the capability survives; this way of reaching it does not".into(),
237/// };
238/// assert_eq!(change.to_string(), "- route GET /api/v1/alpha");
239/// ```
240#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
241#[schemars(rename = "ReleaseSurfaceChange")]
242pub struct SurfaceChange {
243    /// How much it moves the contract on its own.
244    pub impact: Impact,
245    /// Added, removed, or altered.
246    pub direction: Direction,
247    /// Which promise it is.
248    pub surface: SurfaceKind,
249    /// The capability it belongs to.
250    pub capability: String,
251    /// The thing itself: `GET /api/v1/foo`, `majordomus_release_analysis`, `release.analysis`.
252    pub id: String,
253    /// Why this impact and not a smaller one, in one clause.
254    pub detail: String,
255}

266    }

271/// Every way the contract moved between two surfaces.
272///
273/// A capability that was renamed appears as a removal and an addition, and deliberately so:
274/// that *is* what happened to a caller holding the old name, and collapsing the two into one
275/// "rename" would lose the breaking half while looking tidier.
276///
277/// ```
278/// use majordomus_cli::release::compat::{diff, Impact};
279/// use majordomus_cli::release::surface::{public_capabilities, Origin, Surface, SURFACE_SCHEMA};
280/// use serde_json::json;
281///
282/// let surface = |caps| Surface {
283///     schema: SURFACE_SCHEMA.to_string(),
284///     origin: Origin::WorkingTree,
285///     capabilities: public_capabilities(&json!({"capabilities": caps})),
286/// };
287/// let before = surface(json!([{"id": "a.one", "kind": "query", "visibility": "public"}]));
288/// let after = surface(json!([
289///     {"id": "a.one", "kind": "query", "visibility": "public"},
290///     {"id": "a.two", "kind": "query", "visibility": "public"},
291/// ]));
292///
293/// assert!(diff(&before, &before).is_empty(), "a surface diffed with itself is empty");
294/// let found = diff(&before, &after);
295/// assert_eq!(found.len(), 1);
296/// assert_eq!(found[0].impact, Impact::Minor, "an addition breaks nobody");
297/// // and the other way round, the same movement is breaking
298/// assert_eq!(diff(&after, &before)[0].impact, Impact::Major);
299/// ```
300pub fn diff(base: &Surface, head: &Surface) -> Vec<SurfaceChange> {
301    let mut changes = Vec::new();
302    let ids: BTreeSet<&String> = base
303        .capabilities
304        .keys()
305        .chain(head.capabilities.keys())
306        .collect();
307
308    for id in ids {
309        match (base.capabilities.get(id), head.capabilities.get(id)) {
310            (None, Some(_)) => changes.push(SurfaceChange {
311                impact: Impact::Minor,
312                direction: Direction::Added,
313                surface: SurfaceKind::Capability,
314                capability: id.clone(),
315                id: id.clone(),
316                detail: "a capability nothing could call before".into(),
317                // Its bindings are not listed separately: they arrived with it, and a
318                // reader counting "one new capability" should not also count three atoms.
319            }),
320            (Some(_), None) => changes.push(SurfaceChange {
321                impact: Impact::Major,
322                direction: Direction::Removed,
323                surface: SurfaceKind::Capability,
324                capability: id.clone(),
325                id: id.clone(),
326                detail: "a caller that held it has nothing to call".into(),
327            }),
328            (Some(b), Some(h)) => compare_capability(b, h, &mut changes),
329            (None, None) => unreachable!("the id came from one of the two maps"),
330        }
331    }
332    changes.sort_by(|a, b| {
333        // Worst first, then stable by what it is, so two runs print the same report.
334        b.impact
335            .cmp(&a.impact)
336            .then_with(|| a.capability.cmp(&b.capability))
337            .then_with(|| a.id.cmp(&b.id))
338            .then_with(|| a.detail.cmp(&b.detail))
339    });
340    changes
341}

343/// One capability present in both: every binding, then both schemas.
344fn compare_capability(
345    base: &PublicCapability,
346    head: &PublicCapability,
347    out: &mut Vec<SurfaceChange>,
348) {
349    let id = &base.id;
350
351    if base.kind != head.kind {
352        out.push(SurfaceChange {
353            impact: Impact::Major,
354            direction: Direction::Changed,
355            surface: SurfaceKind::Kind,
356            capability: id.clone(),
357            id: format!("{} -> {}", base.kind, head.kind),
358            detail: "whether calling it changes anything is part of the contract".into(),
359        });
360    }
361
362    binding(
363        out,
364        id,
365        SurfaceKind::McpTool,
366        base.mcp_tool.as_deref(),
367        head.mcp_tool.as_deref(),
368    );
369    binding(
370        out,
371        id,
372        SurfaceKind::McpResource,
373        base.mcp_resource.as_deref(),
374        head.mcp_resource.as_deref(),
375    );
376    binding(
377        out,
378        id,
379        SurfaceKind::Cli,
380        base.cli.as_deref(),
381        head.cli.as_deref(),
382    );
383    let b_http = base.http.as_ref().map(ToString::to_string);
384    let h_http = head.http.as_ref().map(ToString::to_string);
385    binding(
386        out,
387        id,
388        SurfaceKind::Http,
389        b_http.as_deref(),
390        h_http.as_deref(),
391    );
392
393    schema(
394        out,
395        id,
396        SurfaceKind::Input,
397        base.input.as_ref(),
398        head.input.as_ref(),
399    );
400    schema(
401        out,
402        id,
403        SurfaceKind::Output,
404        base.output.as_ref(),
405        head.output.as_ref(),
406    );
407}

409/// One optional name a caller can hold: gone is major, arrived is minor, moved is both.
410fn binding(
411    out: &mut Vec<SurfaceChange>,
412    capability: &str,
413    kind: SurfaceKind,
414    base: Option<&str>,
415    head: Option<&str>,
416) {
417    match (base, head) {
418        (Some(b), Some(h)) if b != h => {
419            // Two entries, not one "changed": a caller holding `b` is broken, and `h` is a
420            // thing that did not exist. Reporting a single move would understate the first.
421            out.push(SurfaceChange {
422                impact: Impact::Major,
423                direction: Direction::Removed,
424                surface: kind,
425                capability: capability.to_string(),
426                id: b.to_string(),
427                detail: format!("moved to `{h}`; a caller holding this one breaks"),
428            });
429            out.push(SurfaceChange {
430                impact: Impact::Minor,
431                direction: Direction::Added,
432                surface: kind,
433                capability: capability.to_string(),
434                id: h.to_string(),
435                detail: format!("where `{b}` used to answer"),
436            });
437        }
438        (Some(b), None) => out.push(SurfaceChange {
439            impact: Impact::Major,
440            direction: Direction::Removed,
441            surface: kind,
442            capability: capability.to_string(),
443            id: b.to_string(),
444            detail: "the capability survives; this way of reaching it does not".into(),
445        }),
446        (None, Some(h)) => out.push(SurfaceChange {
447            impact: Impact::Minor,
448            direction: Direction::Added,
449            surface: kind,
450            capability: capability.to_string(),
451            id: h.to_string(),
452            detail: "a new way to reach a capability that already existed".into(),
453        }),
454        _ => {}
455    }
456}

458/// One of the two contracts, compared structurally.
459fn schema(
460    out: &mut Vec<SurfaceChange>,
461    capability: &str,
462    role: SurfaceKind,
463    base: Option<&Value>,
464    head: Option<&Value>,
465) {
466    match (base, head) {
467        (Some(b), Some(h)) => {
468            let mut found = Vec::new();
469            compare_schema(role, b, h, "", &mut found);
470            for (impact, pointer, detail) in found {
471                out.push(SurfaceChange {
472                    impact,
473                    direction: Direction::Changed,
474                    surface: role,
475                    capability: capability.to_string(),
476                    id: if pointer.is_empty() {
477                        "the schema".to_string()
478                    } else {
479                        pointer
480                    },
481                    detail,
482                });
483            }
484        }
485        // A contract that appears constrains a caller that had none; one that disappears
486        // stops promising. Neither is a shape this repository produces — every capability
487        // declares both — so both are reported rather than ignored.
488        (None, Some(_)) => out.push(SurfaceChange {
489            impact: if role == SurfaceKind::Input {
490                Impact::Major
491            } else {
492                Impact::Minor
493            },
494            direction: Direction::Added,
495            surface: role,
496            capability: capability.to_string(),
497            id: "the schema".into(),
498            detail: "a contract where there was none".into(),
499        }),
500        (Some(_), None) => out.push(SurfaceChange {
501            impact: Impact::Major,
502            direction: Direction::Removed,
503            surface: role,
504            capability: capability.to_string(),
505            id: "the schema".into(),
506            detail: "the contract it stated is gone".into(),
507        }),
508        (None, None) => {}
509    }
510}

512/// The schema comparison, by JSON pointer.
513///
514/// `role` decides the direction compatibility runs in, and it is the whole reason the two
515/// are not one function: an input is what a caller *sends*, so accepting more is compatible
516/// and demanding more is breaking; an output is what a caller *receives*, so promising more
517/// is compatible and promising less is breaking. The same edit is minor in one and major in
518/// the other, which is exactly the mistake a textual diff of two schemas makes.
519fn compare_schema(
520    role: SurfaceKind,
521    base: &Value,
522    head: &Value,
523    at: &str,
524    out: &mut Vec<(Impact, String, String)>,
525) {
526    if base == head {
527        return;
528    }
529    let input = role == SurfaceKind::Input;
530    let here = |suffix: &str| {
531        if suffix.is_empty() {
532            at.to_string()
533        } else if at.is_empty() {
534            suffix.to_string()
535        } else {
536            format!("{at}.{suffix}")
537        }
538    };
539
540    let (Some(b), Some(h)) = (base.as_object(), head.as_object()) else {
541        // `true` / `false` schemas, or one side replaced wholesale.
542        out.push((
543            Impact::Major,
544            here(""),
545            "the schema was replaced by one of a different shape".into(),
546        ));
547        return;
548    };
549
550    // ------------------------------------------------------------------------ type
551    if b.get("type") != h.get("type") {
552        out.push((
553            Impact::Major,
554            here("type"),
555            format!(
556                "the type changed from {} to {}",
557                render(b.get("type")),
558                render(h.get("type"))
559            ),
560        ));
561    }
562
563    // ------------------------------------------------------------------------ enum
564    // A value that was accepted and is not any more breaks the caller that sends it; one
565    // that is accepted now and was not breaks nobody. The same holds for an output read by
566    // a caller that only knows the old values, which is why widening is minor and not none.
567    values(b.get("enum"), h.get("enum"), &mut |gone, new| {
568        for v in gone {
569            out.push((
570                Impact::Major,
571                here("enum"),
572                format!("the value {v} is no longer accepted"),
573            ));
574        }
575        for v in new {
576            out.push((
577                Impact::Minor,
578                here("enum"),
579                format!("the value {v} is accepted as well"),
580            ));
581        }
582    });
583
584    // --------------------------------------------------------- oneOf / anyOf / allOf
585    for key in ["oneOf", "anyOf"] {
586        values(b.get(key), h.get(key), &mut |gone, new| {
587            for v in gone {
588                out.push((Impact::Major, here(key), format!("the variant {v} is gone")));
589            }
590            for v in new {
591                out.push((Impact::Minor, here(key), format!("the variant {v} is new")));
592            }
593        });
594    }
595    if b.get("allOf") != h.get("allOf") {
596        // Every branch of an allOf must hold at once, so adding one narrows and removing
597        // one widens; rather than model that, it is named and called breaking.
598        out.push((
599            Impact::Major,
600            here("allOf"),
601            "the conjunction of constraints changed".into(),
602        ));
603    }
604
605    // -------------------------------------------------------------------- required
606    let b_req = string_set(b.get("required"));
607    let h_req = string_set(h.get("required"));
608    for name in h_req.difference(&b_req) {
609        out.push(if input {
610            (
611                Impact::Major,
612                here(&format!("required.{name}")),
613                format!("`{name}` is now required; a caller that did not send it is refused"),
614            )
615        } else {
616            (
617                Impact::Minor,
618                here(&format!("required.{name}")),
619                format!("`{name}` is now always present; a stronger promise than before"),
620            )
621        });
622    }
623    for name in b_req.difference(&h_req) {
624        out.push(if input {
625            (
626                Impact::Minor,
627                here(&format!("required.{name}")),
628                format!("`{name}` is no longer required; every existing caller still works"),
629            )
630        } else {
631            (
632                Impact::Major,
633                here(&format!("required.{name}")),
634                format!("`{name}` is no longer promised; a caller that reads it may find nothing"),
635            )
636        });
637    }
638
639    // ------------------------------------------------------------------ properties
640    let empty = serde_json::Map::new();
641    let b_props = b
642        .get("properties")
643        .and_then(|p| p.as_object())
644        .unwrap_or(&empty);
645    let h_props = h
646        .get("properties")
647        .and_then(|p| p.as_object())
648        .unwrap_or(&empty);
649    let names: BTreeSet<&String> = b_props.keys().chain(h_props.keys()).collect();
650    for name in names {
651        match (b_props.get(name), h_props.get(name)) {
652            (None, Some(_)) => out.push((
653                Impact::Minor,
654                here(&format!("properties.{name}")),
655                if input {
656                    format!("`{name}` may now be sent; nothing that worked stopped working")
657                } else {
658                    format!("`{name}` is returned as well; a caller ignoring it is unaffected")
659                },
660            )),
661            (Some(_), None) => out.push((
662                Impact::Major,
663                here(&format!("properties.{name}")),
664                if input {
665                    // Every input schema in this repository derives `deny_unknown_fields`,
666                    // so a property that is gone is not ignored — it is rejected.
667                    format!("`{name}` is refused now; a caller still sending it gets an error")
668                } else {
669                    format!("`{name}` is gone; a caller reading it finds nothing")
670                },
671            )),
672            (Some(bp), Some(hp)) => {
673                compare_schema(role, bp, hp, &here(&format!("properties.{name}")), out)
674            }
675            (None, None) => unreachable!("the name came from one of the two maps"),
676        }
677    }
678
679    // ----------------------------------------------------------------------- $defs
680    for key in ["$defs", "definitions"] {
681        let b_defs = b.get(key).and_then(|d| d.as_object()).unwrap_or(&empty);
682        let h_defs = h.get(key).and_then(|d| d.as_object()).unwrap_or(&empty);
683        let names: BTreeSet<&String> = b_defs.keys().chain(h_defs.keys()).collect();
684        for name in names {
685            match (b_defs.get(name), h_defs.get(name)) {
686                (Some(bd), Some(hd)) => {
687                    compare_schema(role, bd, hd, &here(&format!("{key}.{name}")), out)
688                }
689                // A definition is only emitted when something refers to it, so one arriving
690                // or leaving is a referenced shape arriving or leaving.
691                (None, Some(_)) => out.push((
692                    Impact::Minor,
693                    here(&format!("{key}.{name}")),
694                    format!("the shape `{name}` is part of the contract now"),
695                )),
696                (Some(_), None) => out.push((
697                    Impact::Major,
698                    here(&format!("{key}.{name}")),
699                    format!("the shape `{name}` is gone"),
700                )),
701                (None, None) => unreachable!("the name came from one of the two maps"),
702            }
703        }
704    }
705
706    // ----------------------------------------------------------------------- items
707    for key in ["items", "additionalItems", "contains"] {
708        match (b.get(key), h.get(key)) {
709            (Some(bi), Some(hi)) => compare_schema(role, bi, hi, &here(key), out),
710            (None, None) => {}
711            _ => out.push((
712                Impact::Major,
713                here(key),
714                format!("`{key}` is stated on one side only"),
715            )),
716        }
717    }
718    if let (Some(Value::Array(ba)), Some(Value::Array(ha))) =
719        (b.get("prefixItems"), h.get("prefixItems"))
720    {
721        // Positional: index i of one is index i of the other, and a length change is a
722        // change to what a caller must supply or may read.
723        if ba.len() != ha.len() {
724            out.push((
725                Impact::Major,
726                here("prefixItems"),
727                format!("the tuple went from {} to {} positions", ba.len(), ha.len()),
728            ));
729        }
730        for (i, (bi, hi)) in ba.iter().zip(ha).enumerate() {
731            compare_schema(role, bi, hi, &here(&format!("prefixItems.{i}")), out);
732        }
733    }
734
735    // ------------------------------------------------------ additionalProperties, $ref
736    if b.get("additionalProperties") != h.get("additionalProperties") {
737        match (b.get("additionalProperties"), h.get("additionalProperties")) {
738            // Closing an open object refuses input that used to be accepted; opening a
739            // closed one accepts more. For an output the reasoning runs the other way, and
740            // neither direction is common enough here to be worth more than naming it.
741            (Some(Value::Bool(true)) | None, Some(Value::Bool(false))) => out.push((
742                Impact::Major,
743                here("additionalProperties"),
744                "unknown properties are refused now".into(),
745            )),
746            (Some(Value::Bool(false)), Some(Value::Bool(true)) | None) => out.push((
747                if input { Impact::Minor } else { Impact::Major },
748                here("additionalProperties"),
749                if input {
750                    "unknown properties are accepted now".into()
751                } else {
752                    "the shape of what is returned is no longer closed".to_string()
753                },
754            )),
755            _ => out.push((
756                Impact::Major,
757                here("additionalProperties"),
758                "what may appear beyond the declared properties changed".into(),
759            )),
760        }
761    }
762    if b.get("$ref") != h.get("$ref") {
763        out.push((
764            Impact::Major,
765            here("$ref"),
766            format!(
767                "it points at {} instead of {}",
768                render(h.get("$ref")),
769                render(b.get("$ref"))
770            ),
771        ));
772    }
773
774    // ------------------------------------------------------- everything not modelled
775    //
776    // A constraint this comparator has no rule for, that differs, is breaking — because the
777    // alternative is to call an unknown contract change a patch, and a caller finds that out
778    // by breaking. The pointer and the key are named so the answer is arguable rather than
779    // mysterious, and every key above is excluded because it already has a rule.
780    const MODELLED: &[&str] = &[
781        "type",
782        "enum",
783        "oneOf",
784        "anyOf",
785        "allOf",
786        "required",
787        "properties",
788        "$defs",
789        "definitions",
790        "items",
791        "additionalItems",
792        "contains",
793        "prefixItems",
794        "additionalProperties",
795        "$ref",
796    ];
797    let keys: BTreeSet<&String> = b.keys().chain(h.keys()).collect();
798    for key in keys {
799        if MODELLED.contains(&key.as_str()) {
800            continue;
801        }
802        if b.get(key) != h.get(key) {
803            out.push((
804                Impact::Major,
805                here(key),
806                format!(
807                    "`{key}` changed from {} to {}, and this comparator does not model what \
808                     that does to a caller; it is counted as breaking rather than guessed at",
809                    render(b.get(key)),
810                    render(h.get(key))
811                ),
812            ));
813        }
814    }
815}

817/// Compare two arrays as sets of values, calling back with what went and what arrived.
818fn values(
819    base: Option<&Value>,
820    head: Option<&Value>,
821    f: &mut impl FnMut(Vec<String>, Vec<String>),
822) {
823    let set = |v: Option<&Value>| -> BTreeSet<String> {
824        v.and_then(|v| v.as_array())
825            .map(|a| a.iter().map(compact).collect())
826            .unwrap_or_default()
827    };
828    let (b, h) = (set(base), set(head));
829    if b == h {
830        return;
831    }
832    f(
833        b.difference(&h).cloned().collect(),
834        h.difference(&b).cloned().collect(),
835    )
836}

838/// The `required` list as a set.
839fn string_set(v: Option<&Value>) -> BTreeSet<String> {
840    v.and_then(|v| v.as_array())
841        .map(|a| {
842            a.iter()
843                .filter_map(|x| x.as_str())
844                .map(str::to_string)
845                .collect()
846        })
847        .unwrap_or_default()
848}

850/// One value, compactly, for a message a person reads.
851fn compact(v: &Value) -> String {
852    match v {
853        Value::String(s) => format!("`{s}`"),
854        other => serde_json::to_string(other).unwrap_or_else(|_| "?".into()),
855    }
856}

858/// A value that may be absent, for a message a person reads.
859fn render(v: Option<&Value>) -> String {
860    v.map_or_else(|| "absent".to_string(), compact)

865/// How this project turns a measured impact into the version it owes.
866///
867/// Semantic versioning hands `0.y.z` a blanket exemption — "anything MAY change at any
868/// time" — and Elm refuses the exemption by starting every package at `1.0.0`, because a
869/// rule with an escape hatch is a rule about when to use the hatch. This repository is `0.x`
870/// and has not earned `1.0.0`, so neither answer is available: demanding `major` would
871/// demand `1.0.0` for a single removal, and granting the exemption would let a breaking
872/// change ship as a patch.
873///
874/// So below `1.0.0` the floor is a minor. Any movement of the surface, gone or new, is at
875/// least a minor release; only an unchanged surface owes nothing. That is the strongest
876/// signal `0.x` has, and a removal is still *named* as breaking whatever the verdict, so the
877/// evidence does not quietly become a minor along with the number.
878///
879/// At `1.0.0` the shift ends: [`Mode::Semver`] is the ordinary rule, and the transition is
880/// one line of [`Policy::required_of`].
881///
882/// ```
883/// use majordomus_cli::release::compat::{Mode, Policy};
884/// use majordomus_cli::release::version::Version;
885/// assert_eq!(Policy::for_version(Version::parse("0.9.0").unwrap()).mode, Mode::Pre1_0Strict);
886/// assert_eq!(Policy::for_version(Version::parse("1.0.0").unwrap()).mode, Mode::Semver);
887/// ```
888#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
889#[serde(rename_all = "kebab-case")]
890#[schemars(rename = "ReleasePolicyMode")]
891pub enum Mode {
892    /// Below `1.0.0`: breaking changes are allowed, and cost a minor rather than nothing.
893    Pre1_0Strict,
894    /// At or above `1.0.0`: the impact is the requirement.
895    Semver,
896}
897
898/// The compatibility policy, named and versioned.
899///
900/// ```
901/// use majordomus_cli::release::compat::{Impact, Policy, POLICY_SCHEMA};
902/// use majordomus_cli::release::version::Version;
903/// let policy = Policy::for_version(Version::parse("0.5.0").unwrap());
904/// assert_eq!(policy.schema, POLICY_SCHEMA, "a verdict carries the rules it was reached under");
905/// assert_eq!(policy.required_of(Impact::Major), Impact::Minor);
906/// ```
907#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
908#[schemars(rename = "ReleasePolicy")]
909pub struct Policy {
910    /// `majordomus/version-policy/v1`.
911    pub schema: String,
912    /// Which rule is in force, decided by the baseline version alone.
913    pub mode: Mode,
914}
915
916impl Policy {
917    /// The policy that governs a project at this version.
918    ///
919    /// ```
920    /// use majordomus_cli::release::compat::{Mode, Policy};
921    /// use majordomus_cli::release::version::Version;
922    /// // the baseline decides the mode, not the version being written
923    /// assert_eq!(Policy::for_version(Version::parse("0.1.0").unwrap()).mode, Mode::Pre1_0Strict);
924    /// assert_eq!(Policy::for_version(Version::parse("2.3.4").unwrap()).mode, Mode::Semver);
925    /// ```
926    pub fn for_version(base: Version) -> Policy {
927        Policy {
928            schema: POLICY_SCHEMA.to_string(),
929            mode: if base.major == 0 {
930                Mode::Pre1_0Strict
931            } else {
932                Mode::Semver
933            },
934        }
935    }
936
937    /// The smallest release this impact is allowed to ship as.
938    ///
939    /// An unchanged surface owes nothing under either mode. This runs over every tree, not
940    /// only over a release, and most commits are behind the boundary: demanding a bump for
941    /// each of them would make the gate a thing to be worked around within a day.
942    ///
943    /// ```
944    /// use majordomus_cli::release::compat::{Impact, Policy};
945    /// use majordomus_cli::release::version::Version;
946    ///
947    /// let pre = Policy::for_version(Version::parse("0.5.0").unwrap());
948    /// assert_eq!(pre.required_of(Impact::None), Impact::None);
949    /// assert_eq!(pre.required_of(Impact::Minor), Impact::Minor);
950    /// assert_eq!(pre.required_of(Impact::Major), Impact::Minor, "0.x floors a removal at a minor");
951    ///
952    /// let one = Policy::for_version(Version::parse("1.0.0").unwrap());
953    /// assert_eq!(one.required_of(Impact::Major), Impact::Major);
954    /// ```
955    pub fn required_of(&self, implied: Impact) -> Impact {
956        match implied {
957            Impact::None | Impact::Patch => Impact::None,
958            Impact::Minor => Impact::Minor,
959            Impact::Major => match self.mode {
960                Mode::Pre1_0Strict => Impact::Minor,
961                Mode::Semver => Impact::Major,
962            },
963        }
964    }
965
966    /// How this policy explains itself, in the plan and in `--explain`.
967    ///
968    /// ```
969    /// use majordomus_cli::release::compat::Policy;
970    /// use majordomus_cli::release::version::Version;
971    /// let policy = Policy::for_version(Version::parse("0.5.0").unwrap());
972    /// assert!(policy.statement().contains("below 1.0.0"));
973    /// ```
974    pub fn statement(&self) -> &'static str {
975        match self.mode {
976            Mode::Pre1_0Strict => {
977                "below 1.0.0: an unchanged surface owes nothing, and any movement of it — \
978                 arrived or gone — is at least a minor; a removal is named as breaking but \
979                 does not demand 1.0.0"
980            }
981            Mode::Semver => {
982                "at or above 1.0.0: an unchanged surface owes nothing, an addition is a \
983                 minor, and anything a caller could hold that is gone or changed under it \
984                 is a major"
985            }
986        }
987    }
988}

1071/// Every version tag this repository carries, newest last.
1072fn tags(root: &Path) -> Vec<String> {
1073    let out = std::process::Command::new("git")
1074        .current_dir(root)
1075        .args(["tag", "--list", "v[0-9]*"])
1076        .output();
1077    let Ok(out) = out else { return Vec::new() };
1078    let mut list: Vec<(Version, String)> = String::from_utf8_lossy(&out.stdout)
1079        .lines()
1080        .map(str::trim)
1081        .filter(|l| !l.is_empty())
1082        .filter_map(|l| Version::parse(l.trim_start_matches('v')).map(|v| (v, l.to_string())))
1083        .collect();
1084    list.sort_by_key(|(v, _)| (v.major, v.minor, v.patch));
1085    list.into_iter().map(|(_, t)| t).collect()
1086}

1088/// The release records the layer holds, newest last by version.
1089fn records(objects: &[Object]) -> Vec<(Version, String, Option<String>, Option<String>)> {
1090    let mut list: Vec<_> = objects
1091        .iter()
1092        .filter(|o| o.kind == super::changelog::RELEASE_KIND)
1093        .filter_map(|o| {
1094            let raw = o.metadata.get("version")?.as_str()?;
1095            let v = Version::parse(raw)?;
1096            let tag = o
1097                .metadata
1098                .get("tag")
1099                .and_then(|t| t.as_str())
1100                .map(str::to_string);
1101            let commit = o
1102                .metadata
1103                .get("commit")
1104                .and_then(|c| c.as_str())
1105                .map(str::to_string);
1106            Some((v, raw.to_string(), tag, commit))
1107        })
1108        .collect();
1109    list.sort_by_key(|(v, _, _, _)| (v.major, v.minor, v.patch));
1110    list
1111}

1113/// Find the release to measure against, and say what is incoherent about the release state
1114/// on the way.
1115///
1116/// The canonical baseline is the newest release the *layer records*, because a record is
1117/// what this repository writes when something is actually published; a tag is git's copy of
1118/// that fact and can be created by anyone. When the two disagree the record still wins and
1119/// the disagreement is reported, so the answer is never silently drawn from whichever source
1120/// happened to be newer.
1121fn resolve_baseline(
1122    root: &Path,
1123    objects: &[Object],
1124    since: Option<&str>,
1125    diagnostics: &mut Vec<Diagnostic>,
1126) -> Result<Baseline, SurfaceError> {
1127    let tags = tags(root);
1128    let records = records(objects);
1129
1130    // Every tag should have a record and every record a tag; neither direction is checked
1131    // anywhere else, and each is how a release goes half-published.
1132    let tagged: BTreeSet<String> = tags.iter().cloned().collect();
1133    let recorded_tags: BTreeSet<String> = records
1134        .iter()
1135        .map(|(_, raw, tag, _)| tag.clone().unwrap_or_else(|| format!("v{raw}")))
1136        .collect();
1137    for t in tagged.difference(&recorded_tags) {
1138        diagnostics.push(Diagnostic {
1139            id: "tag-without-record".into(),
1140            severity: Severity::Warning,
1141            message: format!(
1142                "{t} is tagged and the layer holds no release record for it; \
1143                 the changelog and every baseline read the records, so this release is invisible to them"
1144            ),
1145        });
1146    }
1147    for t in recorded_tags.difference(&tagged) {
1148        diagnostics.push(Diagnostic {
1149            id: "record-without-tag".into(),
1150            severity: Severity::Warning,
1151            message: format!(
1152                "the layer records {t} and no such tag exists in this clone; \
1153                 either it was never pushed or the tags were not fetched"
1154            ),
1155        });
1156    }
1157
1158    // The ref to read the surface from. For a record, that is the *commit* it names rather
1159    // than its tag: a record always carries the commit it published, and a tag is git's copy
1160    // of that fact — absent from a shallow clone, absent from a fork, and movable. The tag
1161    // stays the name the reports print, because that is what a reader recognises.
1162    let (version, reference, read_at) = match since {
1163        Some(r) => (
1164            Version::parse(r.trim_start_matches('v')).map_or_else(String::new, |v| v.to_string()),
1165            r.to_string(),
1166            r.to_string(),
1167        ),
1168        None => match records.last() {
1169            Some((_, raw, tag, commit)) => (
1170                raw.clone(),
1171                tag.clone().unwrap_or_else(|| format!("v{raw}")),
1172                commit
1173                    .clone()
1174                    .or_else(|| tag.clone())
1175                    .unwrap_or_else(|| format!("v{raw}")),
1176            ),
1177            None => match tags.last() {
1178                Some(t) => {
1179                    diagnostics.push(Diagnostic {
1180                        id: "baseline-from-tag".into(),
1181                        severity: Severity::Warning,
1182                        message: format!(
1183                            "the layer records no release, so {t} was taken as the baseline from git's tags alone"
1184                        ),
1185                    });
1186                    (t.trim_start_matches('v').to_string(), t.clone(), t.clone())
1187                }
1188                None => return Err(SurfaceError::NothingPublished),
1189            },
1190        },
1191    };
1192
1193    let surface = Surface::read_ref(root, &read_at)?;
1194    let commit = match &surface.origin {
1195        super::surface::Origin::Ref { commit, .. } => commit.clone(),
1196        super::surface::Origin::WorkingTree => String::new(),
1197    };
1198
1199    // The record's own commit against the one the tag points at: a tag moved after the
1200    // record was written makes every comparison since then measure the wrong tree.
1201    let recorded = records.iter().find(|(_, raw, _, _)| *raw == version);
1202    if let Some((_, _, _, Some(record_commit))) = recorded {
1203        if !commit.is_empty() && record_commit != &commit {
1204            diagnostics.push(Diagnostic {
1205                id: "tag-commit-mismatch".into(),
1206                severity: Severity::Error,
1207                message: format!(
1208                    "the release record for {version} names commit {} and {reference} points at {}; \
1209                     the surface being compared with is not the one that was published",
1210                    &record_commit[..record_commit.len().min(12)],
1211                    &commit[..commit.len().min(12)]
1212                ),
1213            });
1214        }
1215    }
1216
1217    Ok(Baseline {
1218        version,
1219        reference,
1220        read_at,
1221        commit,
1222        recorded: recorded.is_some(),
1223        atoms: surface.atoms(),
1224        fingerprint: surface.fingerprint(),
1225    })
1226}

1275/// The whole answer: what moved, what that requires, and what the tree declares.
1276///
1277/// One value, and every surface renders it — the command line, `/api/v1/release/analysis`,
1278/// the MCP tool, the Cockpit panel, the generated document and the CI gate. There is no
1279/// second computation anywhere, which is the property that makes the number mean something.
1280///
1281/// ```
1282/// use majordomus_cli::release::compat::{Status, VersionPlan};
1283/// # use majordomus_cli::release::compat::{Baseline, CommitEvidence, Impact, Policy};
1284/// # use majordomus_cli::release::version::Version;
1285/// let plan = VersionPlan {
1286///     policy: Policy::for_version(Version::parse("0.5.0").unwrap()),
1287///     baseline: Baseline {
1288///         version: "0.5.0".into(), reference: "v0.5.0".into(), read_at: "3a032c1e".into(),
1289///         commit: "3a032c1e".into(), recorded: true, atoms: 300, fingerprint: "sha256:a".into(),
1290///     },
1291///     declared_version: "0.5.0".into(),
1292///     tool_version: "0.5.0".into(),
1293///     writers_agree: true,
1294///     atoms: 333,
1295///     fingerprint: "sha256:b".into(),
1296///     implied: Impact::Minor,
1297///     required: Impact::Minor,
1298///     declared: Impact::None,
1299///     required_version: "0.6.0".into(),
1300///     status: Status::Blocked,
1301///     breaking: false,
1302///     changes: Vec::new(),
1303///     commits: CommitEvidence { commits: 24, implied: Impact::Patch, breaking: Vec::new() },
1304///     understated: true,
1305///     diagnostics: Vec::new(),
1306/// };
1307/// // The contract requires a minor and the version declares none: this tree may not land.
1308/// assert_eq!(plan.status, Status::Blocked);
1309/// assert!(plan.declared < plan.required);
1310/// assert!(plan.understated, "the commit subjects claim less than the contract did");
1311/// ```
1312#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
1313#[schemars(rename = "ReleaseVersionPlan")]
1314pub struct VersionPlan {
1315    /// The policy the verdict was reached under.
1316    pub policy: Policy,
1317    /// The release this was measured against.
1318    pub baseline: Baseline,
1319    /// The version the crate manifest declares — the one place it is authored.
1320    pub declared_version: String,
1321    /// The version `bin/majordomus` prints: what its projection, `share/version.txt`, states.
1322    pub tool_version: String,
1323    /// Whether that projection is current — states what the manifest declares. The name is
1324    /// kept from when the version had two writers; the manifest is the only one now.
1325    pub writers_agree: bool,
1326    /// How many public atoms this tree carries.
1327    pub atoms: usize,
1328    /// The fingerprint of this tree's surface.
1329    pub fingerprint: String,
1330    /// What the surface movement amounts to on its own.
1331    pub implied: Impact,
1332    /// What the policy requires of it.
1333    pub required: Impact,
1334    /// What the two version numbers already represent.
1335    pub declared: Impact,
1336    /// The smallest version this tree may declare.
1337    pub required_version: String,
1338    /// Whether it does.
1339    pub status: Status,
1340    /// Whether anything a caller could hold is gone.
1341    pub breaking: bool,
1342    /// Every movement of the contract, worst first.
1343    pub changes: Vec<SurfaceChange>,
1344    /// What the commits said about themselves.
1345    pub commits: CommitEvidence,
1346    /// Whether the commit subjects understate what the contract actually did.
1347    pub understated: bool,
1348    /// Anything incoherent about the release state, whatever the verdict.
1349    #[serde(default, skip_serializing_if = "Vec::is_empty")]
1350    pub diagnostics: Vec<Diagnostic>,

1353impl VersionPlan {
1354    /// How many changes moved the contract in each direction.
1355    ///
1356    /// ```
1357    /// # use majordomus_cli::release::compat::*;
1358    /// # use majordomus_cli::release::version::Version;
1359    /// # fn plan(changes: Vec<SurfaceChange>) -> VersionPlan {
1360    /// #   VersionPlan { policy: Policy::for_version(Version::parse("0.5.0").unwrap()),
1361    /// #     baseline: Baseline { version: "0.5.0".into(), reference: "v0.5.0".into(),
1362    /// #       read_at: "c".into(), commit: "c".into(), recorded: true, atoms: 1,
1363    /// #       fingerprint: "sha256:a".into() },
1364    /// #     declared_version: "0.5.0".into(), tool_version: "0.5.0".into(), writers_agree: true,
1365    /// #     atoms: 1, fingerprint: "sha256:b".into(), implied: Impact::Minor,
1366    /// #     required: Impact::Minor, declared: Impact::None, required_version: "0.6.0".into(),
1367    /// #     status: Status::Blocked, breaking: false, changes,
1368    /// #     commits: CommitEvidence { commits: 0, implied: Impact::None, breaking: Vec::new() },
1369    /// #     understated: false, diagnostics: Vec::new() }
1370    /// # }
1371    /// # fn change(direction: Direction) -> SurfaceChange {
1372    /// #   SurfaceChange { impact: Impact::Minor, direction, surface: SurfaceKind::Capability,
1373    /// #     capability: "a.one".into(), id: "a.one".into(), detail: String::new() }
1374    /// # }
1375    /// let p = plan(vec![change(Direction::Added), change(Direction::Added), change(Direction::Removed)]);
1376    /// assert_eq!(p.counts(), (2, 0, 1), "added, changed, removed");
1377    /// ```
1378    pub fn counts(&self) -> (usize, usize, usize) {
1379        let n = |d: Direction| self.changes.iter().filter(|c| c.direction == d).count();
1380        (
1381            n(Direction::Added),
1382            n(Direction::Changed),
1383            n(Direction::Removed),
1384        )
1385    }

1388    ///
1389    /// ```
1390    /// # use majordomus_cli::release::compat::*;
1391    /// # use majordomus_cli::release::version::Version;

1393    /// #   surface: SurfaceKind::Http, capability: "a.one".into(), id: "GET /x".into(),

1396    /// # let plan = VersionPlan { policy: Policy::for_version(Version::parse("1.0.0").unwrap()),
1397    /// #   baseline: Baseline { version: "1.0.0".into(), reference: "v1.0.0".into(),
1398    /// #     read_at: "c".into(), commit: "c".into(), recorded: true, atoms: 1,
1399    /// #     fingerprint: "sha256:a".into() },
1400    /// #   declared_version: "1.0.0".into(), tool_version: "1.0.0".into(), writers_agree: true,
1401    /// #   atoms: 1, fingerprint: "sha256:b".into(), implied: Impact::Major,
1402    /// #   required: Impact::Major, declared: Impact::None, required_version: "2.0.0".into(),
1403    /// #   status: Status::Blocked, breaking: true, changes: vec![breaking, additive],
1404    /// #   commits: CommitEvidence { commits: 0, implied: Impact::None, breaking: Vec::new() },
1405    /// #   understated: false, diagnostics: Vec::new() };
1406    /// assert_eq!(plan.breaking_changes().count(), 1, "only what is breaking on its own");

1408    pub fn breaking_changes(&self) -> impl Iterator<Item = &SurfaceChange> {
1409        self.changes.iter().filter(|c| c.impact == Impact::Major)

1416    /// # use majordomus_cli::release::version::Version;

1421    /// #   declared_version: "0.5.0".into(), tool_version: "0.5.0".into(), writers_agree: true,
1422    /// #   atoms: 1, fingerprint: "sha256:b".into(), implied: Impact::None,
1423    /// #   required: Impact::None, declared: Impact::None, required_version: "0.5.0".into(),
1424    /// #   status: Status::Ok, breaking: false, changes: Vec::new(),
1425    /// #   commits: CommitEvidence { commits: 0, implied: Impact::None, breaking: Vec::new() },
1426    /// #   understated: false, diagnostics: Vec::new() };
1427    /// assert!(!plan.has_errors());

1431    /// assert!(!plan.has_errors());

1438            .iter()
1439            .any(|d| d.severity == Severity::Error)

1443/// The bump two version numbers already represent.
1444///
1445/// A number going *down* is not a bump at all, and is reported as [`Impact::None`] so that
1446/// the comparison with the requirement refuses it rather than ranking it.
1447///
1448/// ```
1449/// use majordomus_cli::release::compat::{declared_impact, Impact};
1450/// use majordomus_cli::release::version::Version;
1451/// let v = |s| Version::parse(s).unwrap();
1452/// assert_eq!(declared_impact(v("0.5.0"), v("0.6.0")), Impact::Minor);
1453/// assert_eq!(declared_impact(v("0.5.0"), v("1.0.0")), Impact::Major);
1454/// // A version going down is no bump at all, so it can never satisfy a requirement.
1455/// assert_eq!(declared_impact(v("0.5.0"), v("0.4.0")), Impact::None);
1456/// ```
1457pub fn declared_impact(from: Version, to: Version) -> Impact {
1458    if to.major > from.major {
1459        Impact::Major
1460    } else if to.major < from.major {
1461        Impact::None
1462    } else if to.minor > from.minor {
1463        Impact::Minor
1464    } else if to.minor < from.minor {
1465        Impact::None
1466    } else if to.patch > from.patch {
1467        Impact::Patch
1468    } else {
1469        Impact::None
1470    }
1471}

1473/// The one analysis. Everything that answers a version question calls this.
1474///
1475/// ```
1476/// use majordomus_cli::capability::registry::CapabilityRegistry;
1477/// use majordomus_cli::release::compat::analyze;
1478///
1479/// let registry = CapabilityRegistry::builder().build().expect("an empty registry composes");
1480/// let dir = tempfile::tempdir().unwrap();
1481/// // A repository that has published nothing has no baseline, and the analysis refuses
1482/// // rather than reporting an empty diff — which would read as "nothing changed".
1483/// assert!(analyze(dir.path(), &registry, &[], None).is_err());
1484/// ```
1485pub fn analyze(
1486    root: &Path,
1487    registry: &crate::capability::registry::CapabilityRegistry,
1488    objects: &[Object],
1489    since: Option<&str>,
1490) -> Result<VersionPlan, SurfaceError> {
1491    let mut diagnostics = Vec::new();
1492    let baseline = resolve_baseline(root, objects, since, &mut diagnostics)?;
1493    let base = Surface::read_ref(root, &baseline.read_at)?;
1494    let head = Surface::of_registry(registry);
1495    let changes = diff(&base, &head);
1496
1497    let implied = changes
1498        .iter()
1499        .map(|c| c.impact)
1500        .max()
1501        .unwrap_or(Impact::None);
1502    let breaking = implied == Impact::Major;
1503
1504    let declared_version = super::version::declared(root).unwrap_or_default();
1505    let tool_version = super::version::tool(root).unwrap_or_default();
1506
1507    let base_v = Version::parse(&baseline.version);
1508    let policy = Policy::for_version(base_v.unwrap_or(Version {
1509        major: 0,
1510        minor: 0,
1511        patch: 0,
1512    }));
1513    let required = policy.required_of(implied);
1514
1515    if base_v.is_none() {
1516        diagnostics.push(Diagnostic {
1517            id: "baseline-version-malformed".into(),
1518            severity: Severity::Error,
1519            message: format!(
1520                "the baseline version '{}' is not three numbers, so the policy that governs it \
1521                 cannot be decided; pre-1.0 rules were assumed",
1522                baseline.version
1523            ),
1524        });
1525    }
1526    let declared_v = Version::parse(&declared_version);
1527    if declared_v.is_none() {
1528        diagnostics.push(Diagnostic {
1529            id: "declared-version-malformed".into(),
1530            severity: Severity::Error,
1531            message: format!(
1532                "the tree declares '{declared_version}', which is not three numbers; \
1533                 {} states it",
1534                super::version::MANIFEST
1535            ),
1536        });
1537    }
1538    // Where the version is stated: the projection against the authority, and any version
1539    // written down by hand. A stale projection is a warning — every bump leaves one until the
1540    // derivation runs, `generate --check` refuses it, and an error here would make the writer
1541    // refuse to correct a bump it had just made. The rest are errors.
1542    let writers_agree = super::version::projection_current(root);
1543    diagnostics.extend(super::version::diagnose(root));
1544
1545    let (declared, required_version) = match (base_v, declared_v) {
1546        (Some(b), Some(d)) => {
1547            let owed = b.raised_to(required);
1548            // The smallest acceptable version, which is the baseline raised by the
1549            // requirement — unless the tree already declares something at least that big.
1550            let required_version = if declared_impact(b, d) >= required && d >= owed {
1551                d.to_string()
1552            } else {
1553                owed.to_string()
1554            };
1555            (declared_impact(b, d), required_version)
1556        }
1557        _ => (Impact::None, declared_version.clone()),
1558    };
1559
1560    // The human account of the same window, kept beside the measured one.
1561    let commit_changes =
1562        super::commits::in_range(root, &format!("{}..HEAD", baseline.reference), objects);
1563    let commits = CommitEvidence {
1564        commits: commit_changes.len(),
1565        implied: match super::version::bump_of(&commit_changes) {
1566            super::version::Bump::None => Impact::None,
1567            super::version::Bump::Patch => Impact::Patch,
1568            super::version::Bump::Minor => Impact::Minor,
1569            super::version::Bump::Major => Impact::Major,
1570        },
1571        breaking: commit_changes
1572            .iter()
1573            .filter(|c| c.breaking)
1574            .map(|c| c.subject.clone())
1575            .collect(),
1576    };
1577    let understated = commits.implied < implied;
1578
1579    let status = if declared < required || diagnostics.iter().any(|d| d.severity == Severity::Error)
1580    {
1581        Status::Blocked
1582    } else {
1583        Status::Ok
1584    };
1585
1586    Ok(VersionPlan {
1587        policy,
1588        baseline,
1589        declared_version,
1590        tool_version,
1591        writers_agree,
1592        atoms: head.atoms(),
1593        fingerprint: head.fingerprint(),
1594        implied,
1595        required,
1596        declared,
1597        required_version,
1598        status,
1599        breaking,
1600        changes,
1601        commits,
1602        understated,
1603        diagnostics,
1604    })
1605}
