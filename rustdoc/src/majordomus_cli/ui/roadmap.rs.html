1//! Roadmap: the milestones in the plan's own order, the intents they serve, and the issues
2//! they hold. Read from `plan.roadmap`, `plan.status` and `intents.list`, and from
3//! `plan.flight` for where the work is (ADR 0166, decision 4):
4//!
5//! - **Now** is the milestones with work in flight (an open pull request, a live mesh claim,
6//!   an ACTIVE issue), never merely the first open one in the plan's order, and **Next** is
7//!   what the graph allows once that work lands.
8//! - Every issue in flight, and every issue of an inspected milestone, shows its declared
9//!   status beside its inferred stage, linked to the objects that prove it: the pull
10//!   request, the merge commit, the release tag, the published `latest.json`. A source that
11//!   was not read says so, with its reason.
12//! - The latest release is read from the releases when the screen is drawn — the records,
13//!   the tags and the published `latest.json` — so a snapshot never shows a release older
14//!   than the executable that drew it.
15//!
16//! Milestones carry no dates in this repository's plan, so the roadmap is ordered by the
17//! plan's dependency graph and drawn as a sequence, never as a calendar: a timeline with
18//! invented dates would be a claim nobody made.
19

47/// The Roadmap screen; `query` may pick one milestone to inspect (`?milestone=M004`).
48pub fn screen(ctx: &Context, query: &[(String, String)]) -> View {
49    let picked = super::milestone::picked(query);
50    let intent = super::intent::picked(query);
51    // the issues are asked only when a milestone or an intent is inspected, their one reader
52    let mut sources = vec![ROADMAP, PLAN, INTENTS];
53    if picked.is_some() || intent.is_some() {
54        sources.push(ISSUES);
55    }
56    // a milestone that is inspected is asked of `plan.flight` with every one of its issues;
57    // otherwise where the work is is asked beside the other sources, at once
58    let inspecting = picked.is_some();
59    if !inspecting {
60        sources.push(FLIGHT);
61    }
62    let mut answers = ask_all(ctx, &sources);
63    if let Some(id) = &picked {
64        let declared = answers[&ROADMAP]
65            .ok()
66            .and_then(|v| v.pointer("/milestones"))
67            .and_then(Value::as_array)
68            .is_some_and(|ms| ms.iter().any(|m| text(m, "id") == *id));
69        let input = if declared {
70            serde_json::json!({ "milestone": id })
71        } else {
72            serde_json::json!({})
73        };
74        answers.insert(FLIGHT, ask_input(ctx, FLIGHT.capability, input));
75    }
76    let (roadmap, plan, intents) = (&answers[&ROADMAP], &answers[&PLAN], &answers[&INTENTS]);
77    let flight = &answers[&FLIGHT];
78
79    let top = hero(
80        "target",
81        "violet",
82        "Roadmap",
83        "From intent to impact, in the order the plan's dependencies allow.",
84        "Every milestone of the plan with the issues it holds, and the intents the work serves.",
85        vec![
86            kpi_count("Now: milestones in flight", &flight.read("/now")),
87            kpi("Next", &flight.read("/next")),
88            kpi("Latest release", &flight.read("/release/latest")),
89            kpi_count("Milestones", &roadmap.read("/milestones")),
90            kpi_count("Intents", &intents.read("/intents")),
91        ],
92    );
93
94    let sequence = unanswered(roadmap).unwrap_or_else(|| {
95        let rows = list(roadmap, "/milestones")
96            .iter()
97            .map(milestone_row)
98            .collect::<Vec<_>>();
99        table(
100            &["Milestone", "State", "Issues done"],
101            rows,
102            "The plan holds no milestone.",
103        )
104    });
105
106    let dependencies = unanswered(roadmap).unwrap_or_else(|| {
107        let rows = list(roadmap, "/milestones")
108            .iter()
109            .filter(|m| {
110                m.get("depends_on")
111                    .and_then(Value::as_array)
112                    .is_some_and(|d| !d.is_empty())
113            })
114            .map(|m| {
115                let on = |key: &str| {
116                    m.get(key)
117                        .and_then(Value::as_array)
118                        .map(|d| {
119                            d.iter()
120                                .filter_map(Value::as_str)
121                                .collect::<Vec<_>>()
122                                .join(", ")
123                        })
124                        .unwrap_or_default()
125                };
126                el("tr")
127                    .child(el("td").class("mj-ui-mono").text(text(m, "id")))
128                    .child(
129                        el("td")
130                            .class("mj-ui-mono mj-ui-dim")
131                            .text(on("depends_on")),
132                    )
133                    .child(
134                        el("td")
135                            .class("mj-ui-mono mj-ui-dim")
136                            .text(on("blocked_by")),
137                    )
138            })
139            .collect::<Vec<_>>();
140        table(
141            &["Milestone", "Depends on", "Blocked by"],
142            rows,
143            "No milestone depends on another.",
144        )
145    });
146
147    // the same edges as a graph: every milestone a node in the tone of its status, every
148    // dependency the plan records an edge, the table above the fallback without a script
149    let dependencies = match unanswered(roadmap) {
150        Some(failed) => failed,
151        None => {
152            let milestones = list(roadmap, "/milestones");
153            let nodes: Vec<GraphNode> = milestones
154                .iter()
155                .map(|m| {
156                    let id = text(m, "id");
157                    GraphNode {
158                        label: id.clone(),
159                        tone: status_tone(&text(m, "status")),
160                        href: super::milestone::href(&id),
161                        kind: "node",
162                        id,
163                    }
164                })
165                .collect();
166            let edges: Vec<(String, String)> = milestones
167                .iter()
168                .flat_map(|m| {
169                    let to = text(m, "id");
170                    m.get("depends_on")
171                        .and_then(Value::as_array)
172                        .cloned()
173                        .unwrap_or_default()
174                        .into_iter()
175                        .filter_map(move |d| d.as_str().map(|from| (from.to_string(), to.clone())))
176                })
177                .collect();
178            graph(
179                "The milestones of the plan and the dependencies between them",
180                Layout::Directed,
181                &nodes,
182                &edges,
183                dependencies,
184            )
185        }
186    };
187
188    let intents_body = unanswered(intents).unwrap_or_else(|| {
189        let rows = list(intents, "/intents")
190            .iter()
191            .map(|i| {
192                el("tr")
193                    .child(
194                        el("td").child(
195                            el("a")
196                                .class("mj-ui-row-link")
197                                .attr("href", super::intent::href(&text(i, "id")))
198                                .child(el("span").class("mj-ui-mono mj-ui-dim").text(text(i, "id")))
199                                .child(el("span").class("mj-ui-cell-title").text(text(i, "title"))),
200                        ),
201                    )
202                    .child(el("td").child(state(i, "stage")))
203                    .child(el("td").child(super::intent::verdict(i)))
204            })
205            .collect::<Vec<_>>();
206        table(
207            &["Intent", "Stage", "Verdict"],
208            rows,
209            "No intent is recorded.",
210        )
211    });
212
213    let mut grid = el("div").class("mj-ui-grid");
214    if let (Some(id), Some(issues)) = (&intent, answers.get(&ISSUES)) {
215        grid = grid.child(wide(super::intent::inspector(intents, issues, id)));
216    }
217    if let (Some(id), Some(issues)) = (&picked, answers.get(&ISSUES)) {
218        // the milestone's own record, for the prose no plan summary carries
219        let record = super::source::ask_input(
220            ctx,
221            "objects.get",
222            serde_json::json!({ "uri": format!("majordomus://milestone/{id}") }),
223        );
224        grid = grid.child(wide(super::milestone::inspector_with(
225            roadmap, issues, &record, id,
226        )));
227        grid = grid.child(wide(delivery_panel(flight, id)));
228    }
229    let grid = grid
230        .child(wide(now_panel(flight)))
231        .child(wide(panel(
232            "Milestones",
233            "Every milestone, in the plan's order; the plan declares no dates, so none are drawn.",
234            sequence,
235            &[roadmap],
236        )))
237        .child(panel(
238            "Intents",
239            "What the work is for, and whether the evidence says it is met.",
240            intents_body,
241            &[intents],
242        ))
243        .child(issues_panel(plan))
244        .child(release_panel(flight))
245        .child(wide(panel(
246            "Dependencies",
247            "Which milestone waits for which, as the plan records it.",
248            dependencies,
249            &[roadmap],
250        )));
251
252    View::new(
253        "Roadmap",
254        el("div").class("mj-ui-stack").child(top).child(grid),
255    )
256}

258/// The milestones with work in flight, each with the issues that put it there: why, the
259/// declared status beside the inferred stage, and what proves the stage.
260fn now_panel(flight: &Answer) -> El {
261    let body = unanswered(flight).unwrap_or_else(|| {
262        let milestones = list(flight, "/milestones");
263        let mut body = el("div").child(el("p").class("mj-ui-line").text(format!(
264            "Now {} · next {}",
265            joined(&list(flight, "/now"), "nothing in flight"),
266            shown(&flight.read("/next"))
267        )));
268        if milestones.is_empty() {
269            body = body.child(nothing(
270                "No work is in flight: no open pull request, live mesh claim or ACTIVE issue \
271                 names an open issue of an open milestone.",
272            ));
273        }
274        for m in &milestones {
275            let id = text(m, "id");
276            body = body
277                .child(
278                    el("h3").class("mj-ui-detail-title").child(
279                        el("a")
280                            .class("mj-ui-row-link")
281                            .attr("href", super::milestone::href(&id))
282                            .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))
283                            .child(el("span").class("mj-ui-cell-title").text(text(m, "title"))),
284                    ),
285                )
286                .child(issue_table(m, true));
287        }
288        body.children(unread_sources(flight))
289    });
290    panel(
291        "Now: where the work is",
292        "The milestones holding an issue with an open pull request, a live mesh claim or an \
293         ACTIVE status, in the plan's order; next is the first one the graph allows after them.",
294        body,
295        &[flight],
296    )
297}

299/// Every issue of the inspected milestone, its declared status beside its inferred stage.
300fn delivery_panel(flight: &Answer, id: &str) -> El {
301    let body = unanswered(flight).unwrap_or_else(|| {
302        match flight.ok().and_then(|v| v.pointer("/inspected")) {
303            Some(m) => el("div")
304                .child(issue_table(m, false))
305                .children(unread_sources(flight)),
306            None => nothing(&format!("{id} is not a milestone of this plan.")),
307        }
308    });
309    panel(
310        &format!("Delivery of {id}"),
311        "Each issue's declared status beside how far its work has travelled, and what proves \
312         it.",
313        body,
314        &[flight],
315    )
316}

318/// A table of a milestone's issues: declared, inferred, why in flight (when `why`), proof.
319fn issue_table(m: &Value, why: bool) -> El {
320    let issues = m
321        .get("issues")
322        .and_then(Value::as_array)
323        .cloned()
324        .unwrap_or_default();
325    let rows = issues
326        .iter()
327        .map(|i| {
328            let id = text(i, "issue");
329            let unread = strings(i, "unread");
330            let stage =
331                el("td")
332                    .child(word_badge(&value(i, "stage")))
333                    .when(!unread.is_empty(), |td| {
334                        td.child(
335                            el("span")
336                                .class("mj-ui-dim")
337                                .text(format!(" unread: {}", unread.join(", "))),
338                        )
339                    });
340            let row = el("tr")
341                .child(
342                    el("td").child(
343                        el("a")
344                            .class("mj-ui-row-link")
345                            .attr("href", format!("{}?issue={id}#issue", super::prs::ROUTE))
346                            .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))
347                            .child(el("span").class("mj-ui-cell-title").text(text(i, "title"))),
348                    ),
349                )
350                .child(el("td").child(state(i, "status")))
351                .child(stage);
352            let row = if why {
353                row.child(
354                    el("td")
355                        .class("mj-ui-mono mj-ui-dim")
356                        .text(strings(i, "in_flight").join(", ")),
357                )
358            } else {
359                row
360            };
361            row.child(el("td").child(proof_list(i)))
362        })
363        .collect::<Vec<_>>();
364    let headers: &[&str] = if why {
365        &["Issue", "Declared", "Inferred", "In flight by", "Proof"]
366    } else {
367        &["Issue", "Declared", "Inferred", "Proof"]
368    };
369    table(headers, rows, "The milestone holds no issue.")
370}

372/// The objects that prove an issue's stage: a link where the forge has the object, the name
373/// alone where only this clone (or the mesh) does.
374fn proof_list(i: &Value) -> El {
375    let proofs = i
376        .get("proofs")
377        .and_then(Value::as_array)
378        .cloned()
379        .unwrap_or_default();
380    if proofs.is_empty() {
381        return el("span").class("mj-ui-dim").text("nothing names it");
382    }
383    let mut out = el("span");
384    for (n, p) in proofs.iter().enumerate() {
385        if n > 0 {
386            out = out.text(" · ");
387        }
388        let label = format!("{} {}", text(p, "kind").replace('_', " "), text(p, "label"));
389        out = match (text(p, "kind").as_str(), text(p, "href")) {
390            ("claim", _) => out.child(super::view::link(super::mesh::ROUTE, label)),
391            (_, href) if !href.is_empty() => out.child(
392                el("a")
393                    .class("mj-ui-link")
394                    .attr("href", href)
395                    .attr("rel", "noopener noreferrer")
396                    .text(label),
397            ),
398            _ => out.child(el("span").class("mj-ui-mono").text(label)),
399        };
400    }
401    out
402}

404/// The latest release, and what each source of it answered.
405fn release_panel(flight: &Answer) -> El {
406    let body = unanswered(flight).unwrap_or_else(|| {
407        let rows = list(flight, "/release/readings")
408            .iter()
409            .map(|r| {
410                let detail = text(r, "detail");
411                let href = text(r, "href");
412                el("tr")
413                    .child(el("td").class("mj-ui-mono").text(text(r, "source")))
414                    .child(el("td").child(state(r, "state")))
415                    .child(el("td").class("mj-ui-mono").text(value(r, "version")))
416                    .child(el("td").child(if href.is_empty() {
417                        el("span").class("mj-ui-dim").text(detail)
418                    } else {
419                        el("a")
420                            .class("mj-ui-link mj-ui-mono")
421                            .attr("href", href)
422                            .attr("rel", "noopener noreferrer")
423                            .text(detail)
424                    }))
425            })
426            .collect::<Vec<_>>();
427        // an answered flight always says whether this executable is tagged
428        let tagged = match flight.read("/release/executable_tagged").text().as_deref() {
429            Some("true") => "is tagged",
430            _ => "is not tagged",
431        };
432        el("div")
433            .child(el("p").class("mj-ui-line").text(format!(
434                "Latest release {} (from {}) · this executable {} {tagged}",
435                shown(&flight.read("/release/latest")),
436                shown(&flight.read("/release/source")),
437                shown(&flight.read("/release/executable")),
438            )))
439            .child(table(
440                &["Source", "State", "Version", "Read from"],
441                rows,
442                "No source of releases was asked.",
443            ))
444    });
445    panel(
446        "Latest release",
447        "The highest stable version the release records, the tags and the published \
448         latest.json name, read when this screen was drawn.",
449        body,
450        &[flight],
451    )
452}

454/// The sources `plan.flight` could not read, each with why: an unread source is said, so
455/// "nothing in flight" is never mistaken for "nothing was looked at".
456fn unread_sources(flight: &Answer) -> Option<El> {
457    let unread: Vec<Value> = list(flight, "/sources")
458        .into_iter()
459        .filter(|s| text(s, "state") == "unread")
460        .collect();
461    (!unread.is_empty()).then(|| {
462        let mut ul = el("ul").class("mj-ui-prose-list");
463        for s in &unread {
464            ul = ul.child(
465                el("li")
466                    .child(word_badge("unread"))
467                    .text(" ")
468                    .child(el("code").class("mj-mono").text(text(s, "name")))
469                    .text(format!(": {}", text(s, "detail"))),
470            );
471        }
472        ul
473    })
474}

476/// A list of strings joined, or `empty` when there are none.
477fn joined(items: &[Value], empty: &str) -> String {
478    let words: Vec<&str> = items.iter().filter_map(Value::as_str).collect();
479    if words.is_empty() {
480        empty.to_string()
481    } else {
482        words.join(", ")
483    }
484}
485
486fn strings(v: &Value, key: &str) -> Vec<String> {
487    v.get(key)
488        .and_then(Value::as_array)
489        .map(|a| {
490            a.iter()
491                .filter_map(Value::as_str)
492                .map(str::to_string)
493                .collect()
494        })
495        .unwrap_or_default()
496}

516    /// A context over `repo` whose registry holds every built-in module except those named:
517    /// a screen asked there meets a capability that cannot answer, as it would on a server
518    /// whose registry lost one.
519    fn context_without(repo: &SyntheticRepository, absent: &[&str]) -> std::sync::Arc<Context> {
520        let index = repo.index().expect("an index");
521        let modules = builtin::modules()
522            .into_iter()
523            .filter(|m| !absent.contains(&m.id.as_str()))
524            .collect();
525        let registry = CapabilityRegistry::builder()
526            .with_modules(modules)
527            .with_index(&index)
528            .build()
529            .expect("a registry");
530        std::sync::Arc::new(Context::new(
531            std::sync::Arc::new(index),
532            std::sync::Arc::new(registry),
533        ))
534    }
