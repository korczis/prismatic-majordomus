1//! How long a pull request has waited for the executor, and how often another was chosen
2//! instead: starvation, made visible.
3//!
4//! Everything here is a fold over the audit trail ([`super::drain::events`]); nothing is
5//! stored beside it. The executor records two facts the fold needs: when a pull request
6//! *becomes* actionable (ready, or refreshable — the two dispositions the executor acts on)
7//! and when it stops being so, both as transitions between one executor step's observation
8//! and the trail's last word; and, on every selection, the other actionable pull requests it
9//! passed over. A dry run records neither, so a dry run still leaves no trace.
10//!
11//! The rank is not changed by any of it. Its age tie-break already prefers the older of two
12//! otherwise equal candidates; a long wait is a fact for a person ([`STARVING_AFTER`]), and
13//! never a reason to merge something less safe sooner.
14
15use std::collections::{BTreeMap, BTreeSet};
16use std::path::Path;
17
18use super::drain::{events, record, IntegrationEvent};
19use super::{ExecutorWait, IntegrationQueue, PassedOver, PullRequestDisposition};
20
21/// A pull request passed over this many times while actionable is called starving.
22pub const STARVING_AFTER: u32 = 3;
23
24/// The trail's word for a pull request that became actionable.
25pub const BECAME_ACTIONABLE: &str = "became_actionable";
26/// The trail's word for one that stopped being actionable.
27pub const LEFT_ACTIONABLE: &str = "left_actionable";
28
29/// Whether the executor acts on a disposition: it merges a ready one and brings master into
30/// a refreshable one.
31pub fn actionable(d: PullRequestDisposition) -> bool {
32    matches!(
33        d,
34        PullRequestDisposition::Ready | PullRequestDisposition::NeedsRefresh
35    )
36}
37
38/// Every pull request the trail says is actionable now, with how long and how often passed
39/// over. A fold, oldest event first; deterministic for a given trail.
40pub fn waits(trail: &[IntegrationEvent]) -> BTreeMap<u64, ExecutorWait> {
41    let mut out: BTreeMap<u64, ExecutorWait> = BTreeMap::new();
42    for e in trail {
43        match (e.action.as_str(), e.pr) {
44            (BECAME_ACTIONABLE, Some(n)) => {
45                out.insert(
46                    n,
47                    ExecutorWait {
48                        actionable_since: e.at.clone(),
49                        passed_over: 0,
50                        last_passed_over: None,
51                    },
52                );
53            }
54            (LEFT_ACTIONABLE | "merge_succeeded" | "closed_superseded", Some(n)) => {
55                out.remove(&n);
56            }
57            ("selected" | "refresh_selected", Some(chosen)) => {
58                for n in &e.passed_over {
59                    if let Some(w) = out.get_mut(n) {
60                        w.passed_over += 1;
61                        w.last_passed_over = Some(PassedOver {
62                            at: e.at.clone(),
63                            for_pr: chosen,
64                            action: e.action.clone(),
65                        });
66                    }
67                }
68            }
69            _ => {}
70        }
71    }
72    out
73}

75/// Record the transitions between what the trail says is actionable and what `queue` says
76/// is: one `became_actionable` per pull request that newly is, one `left_actionable` per
77/// pull request that no longer is, with the disposition it has now (or that it is no longer
78/// open). Nothing is recorded when nothing changed. Returns how many were recorded.
79pub fn record_transitions(root: &Path, queue: &IntegrationQueue) -> usize {
80    let before: BTreeSet<u64> = waits(&events(root)).into_keys().collect();
81    let now: BTreeSet<u64> = queue
82        .assessments
83        .iter()
84        .filter(|a| actionable(a.disposition))
85        .map(|a| a.number)
86        .collect();
87    let mut n = 0;
88    for pr in now.difference(&before) {
89        let a = queue.get(*pr);
90        record(
91            root,
92            IntegrationEvent {
93                at: String::new(),
94                actor: String::new(),
95                action: BECAME_ACTIONABLE.into(),
96                pr: Some(*pr),
97                master_before: a.map(|a| a.evaluated_against.master_sha.clone()),
98                head_sha: a.map(|a| a.evaluated_against.head_sha.clone()),
99                master_after: None,
100                reasons: a.map(|a| a.reasons.clone()).unwrap_or_default(),
101                detail: a
102                    .map(|a| a.disposition.as_str().to_string())
103                    .unwrap_or_default(),
104                passed_over: Vec::new(),
105            },
106        );
107        n += 1;
108    }
109    for pr in before.difference(&now) {
110        let detail = match queue.get(*pr) {
111            Some(a) => format!("it is {} now", a.disposition.as_str()),
112            None => "it is no longer open".to_string(),
113        };
114        record(
115            root,
116            IntegrationEvent {
117                at: String::new(),
118                actor: String::new(),
119                action: LEFT_ACTIONABLE.into(),
120                pr: Some(*pr),
121                master_before: Some(queue.master_sha.clone()),
122                head_sha: queue.get(*pr).map(|a| a.evaluated_against.head_sha.clone()),
123                master_after: None,
124                reasons: Vec::new(),
125                detail,
126                passed_over: Vec::new(),
127            },
128        );
129        n += 1;
130    }
131    n
132}

134/// Put the trail's waits on the queue: each actionable assessment gets its wait, and the
135/// queue names the starving ones, in rank order. A pull request the trail calls actionable
136/// but this queue does not — master moved since the executor last looked — gets none.
137pub fn annotate(queue: &mut IntegrationQueue, trail: &[IntegrationEvent]) {
138    let waits = waits(trail);
139    for a in &mut queue.assessments {
140        a.wait = if actionable(a.disposition) {
141            waits.get(&a.number).cloned()
142        } else {
143            None
144        };
145    }
146    queue.starving = queue
147        .assessments
148        .iter()
149        .filter(|a| {
150            a.wait
151                .as_ref()
152                .is_some_and(|w| w.passed_over >= STARVING_AFTER)
153        })
154        .map(|a| a.number)
155        .collect();
156}
