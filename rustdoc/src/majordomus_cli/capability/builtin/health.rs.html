1//! The `health` module: whether what this process serves is healthy, answered by the
2//! engines that already decide it rather than by checks written a second time here.
3//!
4//! The module is `health` and not `system` because the benchmark projection reserves
5//! `system` for the transports' own targets (`system.http.index`, `system.mcp.ping`) and
6//! buckets its coverage by that name. A capability module called `system` would have its
7//! lines tallied as transport targets and silently vanish from the per-transport
8//! denominators; `reserved_namespace_is_not_a_module` holds that shut.
9//!
10//! Every check delegates: the index's own diagnostics say whether the layer read cleanly,
11//! the benchmark projection's coverage says whether every executable capability is timed,
12//! `generate --check` — the same comparison the command runs — says whether the committed
13//! projections are stale, and [`crate::capability::builtin::server::standing_at`] — the
14//! reading `server.status` answers from — says where the shared server of this checkout
15//! stands. A check that computed its own verdict would be a second opinion, and two
16//! opinions drift.
17//!
18//! # Four readings of "ready", and why none of them is this one
19//!
20//! This crate answers "ready" four times, about four different subjects. The full table,
21//! with what asks each one, is in `docs/MCP.md`; in short:
22//!
23//! - [`crate::http::Served::ready`] — is a *surface's* producer's output on disk? Per
24//!   surface, and what the home page renders as `not built`.
25//! - [`Readiness`] (`health.ready`, `GET /api/v1/ready`) — can *this process* answer a
26//!   request? Local initialisation only, never a dependency: a readiness probe that
27//!   contacts another service fails a deployment for something that is not this process.
28//! - [`crate::environment::ServiceAvailability`] — does *anything* accept a connection at
29//!   the address the lease published? One bounded TCP connect, no DNS, because it runs on
30//!   a shell prompt.
31//! - [`crate::capability::builtin::server::ServerStanding`] — is what answers at that
32//!   address *current*: this checkout's server, from the file on disk, at this version?
33//!
34//! The `server` check below is the fourth, and it is the only one of them that a health
35//! report should carry: the first is per surface, the second is the report's own caller,
36//! and the third cannot tell a live server from a socket somebody else holds.

60/// Where one dimension of the system stands. Ordered by severity, so the worst check
61/// decides the whole.
62#[derive(
63    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
64)]
65#[serde(rename_all = "snake_case")]
66pub enum HealthStatus {
67    /// The engine that decides this dimension is satisfied.
68    Ok,
69    /// Nothing is broken and something is worth looking at.
70    Warn,
71    /// The engine that decides this dimension is not satisfied.
72    Fail,
73    /// This process cannot decide it, and says so rather than reporting `ok`.
74    Unknown,
75}
76
77impl HealthStatus {
78    /// The word as serialised.
79    ///
80    /// ```
81    /// use majordomus_cli::capability::builtin::health::HealthStatus;
82    /// assert_eq!(HealthStatus::Warn.as_str(), "warn");
83    /// ```
84    pub fn as_str(self) -> &'static str {
85        match self {
86            HealthStatus::Ok => "ok",
87            HealthStatus::Warn => "warn",
88            HealthStatus::Fail => "fail",
89            HealthStatus::Unknown => "unknown",
90        }
91    }
92
93    /// The worse of two: the whole is as good as its worst check, and `unknown` is worse
94    /// than `warn` because an undecided dimension is not a healthy one.
95    ///
96    /// ```
97    /// use majordomus_cli::capability::builtin::health::HealthStatus::*;
98    /// assert_eq!(Ok.worse(Warn), Warn);
99    /// assert_eq!(Fail.worse(Unknown), Unknown);
100    /// ```
101    pub fn worse(self, other: Self) -> Self {
102        if other > self {
103            other
104        } else {
105            self
106        }
107    }
108}

110/// One dimension of the system, decided by one engine, with what a reader needs to check
111/// it themselves.
112#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
113pub struct HealthCheck {
114    /// A stable id, `[a-z][a-z0-9-]*`.
115    pub id: String,
116    /// The short name.
117    pub title: String,
118    /// Where it stands.
119    pub status: HealthStatus,
120    /// One line: what the engine decided and on what.
121    pub detail: String,
122    /// The engine that decided it, named so the reader knows what to fix.
123    pub decided_by: String,
124    /// What reproduces the verdict: a command, or a path.
125    #[serde(default, skip_serializing_if = "Vec::is_empty")]
126    pub evidence: Vec<String>,
127    /// Findings behind the verdict, when the engine produced any: the diagnostic
128    /// messages, the stale files, the missing coverage lines.
129    #[serde(default, skip_serializing_if = "Vec::is_empty")]
130    pub findings: Vec<String>,
131}

133/// The health of what this process serves.
134#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
135pub struct Health {
136    /// The worst check's status.
137    pub status: HealthStatus,
138    /// How many checks stand where, by status word.
139    pub tallies: BTreeMap<String, usize>,
140    /// Every dimension, in a stable order.
141    pub checks: Vec<HealthCheck>,
142    /// When the state a check decided on was read, by check id: the freshness contract
143    /// ([`crate::index::AnswerObservation`]). A check absent here decided live, during this call.
144    /// `layer` and `git` are here because they judge the picture the index took, which is
145    /// as old as the index.
146    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
147    pub observed: BTreeMap<String, crate::index::AnswerObservation>,
148}

150/// The answer to "is this process alive": the cheapest true thing this executable can
151/// say about itself. Deliberately not a summary of anything — a probe that grew an
152/// opinion would stop reporting the one fact the platform polls it for.
153#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
154pub struct Liveness {
155    /// Always `true`. A process that could not answer would not answer at all, which is
156    /// the signal.
157    pub alive: bool,
158    /// This executable's version, so a rolling deployment can tell which build answered.
159    pub version: String,
160    /// The commit this executable was built from, in full, or `unknown` when the build could
161    /// not name one. A version is what a build calls itself and two revisions share it; this
162    /// is what a deployment is checked against. Never omitted: an absent field would read as
163    /// a server too old to say, and `unknown` is a different, truthful answer.
164    pub commit: String,
165    /// Whether that commit's tree carried uncommitted changes when this executable was built;
166    /// `null` when the build did not know.
167    pub dirty: Option<bool>,
168}

170/// The answer to "can this process serve traffic": the local initialisation a request
171/// would need, and nothing beyond this process. A readiness check that grew a dependency
172/// probe fails a deployment because an unrelated service is down.
173#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
174pub struct Readiness {
175    /// Whether every part below is in place.
176    pub ready: bool,
177    /// This executable's version.
178    pub version: String,
179    /// The commit this executable was built from, or `unknown`; the same value `health.live`
180    /// answers.
181    pub commit: String,
182    /// Whether the build's tree was dirty; `null` when the build did not know.
183    pub dirty: Option<bool>,
184    /// How many capabilities the registry holds; zero would mean nothing to serve.
185    pub capabilities: usize,
186    /// How many objects the index holds. Read from the index this process built at
187    /// start-up: it is already in memory, and reading it walks nothing.
188    pub objects: usize,
189    /// Whether the layer read cleanly. A degraded layer is still served — the diagnostics
190    /// are the point — so this reports rather than refuses.
191    pub layer: HealthStatus,
192    /// Why this process is serving code that is no longer on disk: the executable it was
193    /// started from has been replaced or removed since. Present, `ready` is false — a server
194    /// answering with yesterday's code says so in the place it says it is ready, to a client
195    /// holding nothing but this answer (I1502).
196    #[serde(default, skip_serializing_if = "Option::is_none")]
197    pub stale: Option<String>,
198}

200/// Liveness. No filesystem, no index traversal, no network: two fields this process can
201/// answer with while doing anything else.
202fn liveness(_: &Context, _: Empty) -> Result<Liveness, CapabilityError> {
203    Ok(Liveness {
204        alive: true,
205        version: crate::VERSION.into(),
206        commit: crate::COMMIT.into(),
207        dirty: crate::DIRTY,
208    })
209}

211/// Readiness. Only what this process needs in order to answer a request, all of it already
212/// resident: the registry it built, the index it built, and how that index read. Nothing
213/// here contacts a provider, a database or another service, and nothing here walks a tree.
214fn readiness(ctx: &Context, _: Empty) -> Result<Readiness, CapabilityError> {
215    let capabilities = ctx.registry.summary().total;
216    let objects = ctx.index.objects.len();
217    let stale = crate::lease::serving_replaced_code();
218    Ok(Readiness {
219        ready: capabilities > 0 && stale.is_none(),
220        version: crate::VERSION.into(),
221        commit: crate::COMMIT.into(),
222        dirty: crate::DIRTY,
223        capabilities,
224        objects,
225        layer: match ctx.index.state {
226            State::Degraded => HealthStatus::Warn,
227            State::Ok => HealthStatus::Ok,
228        },
229        stale,
230    })
231}

233/// Record one dimension, and say on the execution's stream that it was decided.
234///
235/// The report is the same whoever asked for it: a `GET /api/v1/health`, an MCP tool call
236/// and a browser watching an execution all run this function, and the reporting is silent
237/// for the two of them that are not executions. That is the whole of what it took to make
238/// an existing capability observable — no second implementation, no second route, and no
239/// change to what it answers.
240fn record(checks: &mut Vec<HealthCheck>, p: &crate::execution::Progress, check: HealthCheck) {
241    p.step_done(
242        &check.id,
243        check.status != HealthStatus::Fail,
244        Some(check.detail.clone()),
245    );
246    checks.push(check);
247    p.progress(checks.len() as u64, None, "dimension(s) decided");
248}

250/// The attached-clients dimension, from the board this process holds (`None` when it holds
251/// none). Live peers only are counted as attached; a peer that is gone with its claims still
252/// shown, or one attached that has announced nothing, is worth a look, because the board then
253/// understates or overstates who is working here.
254fn peers_check(board: Option<&[crate::peers::Peer]>) -> HealthCheck {
255    let decided_by = "the peer board of the process holding this checkout's lease, tallied as `environment.preflight` tallies it";
256    let Some(peers) = board else {
257        return HealthCheck {
258            id: "peers".into(),
259            title: "Attached clients".into(),
260            status: HealthStatus::Unknown,
261            detail: "this process holds no peer board: it does not hold this checkout's lease, so who is attached is the server's to say".into(),
262            decided_by: decided_by.into(),
263            evidence: vec!["majordomus_peers".into(), "majordomus serve status".into()],
264            findings: Vec::new(),
265        };
266    };
267    let t = crate::environment::preflight::PeersObservation::of(peers);
268    let silent = t.attached - t.announced;
269    let mut findings = Vec::new();
270    if t.detached > 0 {
271        findings.push(format!(
272            "{} peer(s) are gone and the board still shows their claims",
273            t.detached
274        ));
275    }
276    if silent > 0 {
277        findings.push(format!(
278            "{silent} attached peer(s) have announced nothing, so the board understates who is here"
279        ));
280    }
281    HealthCheck {
282        id: "peers".into(),
283        title: "Attached clients".into(),
284        status: if findings.is_empty() {
285            HealthStatus::Ok
286        } else {
287            HealthStatus::Warn
288        },
289        detail: format!(
290            "{} attached ({} announced) · {} gone with claims still shown",
291            t.attached, t.announced, t.detached
292        ),
293        decided_by: decided_by.into(),
294        evidence: vec!["majordomus_peers".into()],
295        findings,
296    }
297}

299fn health(ctx: &Context, _: Empty) -> Result<Health, CapabilityError> {
300    let index = &ctx.index;
301    let mut checks = Vec::new();
302
303    // --- the layer, as the index read it
304    let errors = index
305        .diagnostics
306        .iter()
307        .filter(|d| d.severity == Severity::Error)
308        .count();
309    let warnings = index
310        .diagnostics
311        .iter()
312        .filter(|d| d.severity == Severity::Warning)
313        .count();
314    record(
315        &mut checks,
316        &ctx.progress,
317        HealthCheck {
318            id: "layer".into(),
319            title: "The layer as it was read".into(),
320            status: match (index.state, warnings) {
321                (State::Degraded, _) => HealthStatus::Fail,
322                (State::Ok, 0) => HealthStatus::Ok,
323                (State::Ok, _) => HealthStatus::Warn,
324            },
325            detail: format!(
326                "{} object(s) indexed; {errors} error(s), {warnings} warning(s)",
327                index.objects.len()
328            ),
329            decided_by: "the index's own diagnostics".into(),
330            evidence: vec!["majordomus capabilities validate".into()],
331            findings: index
332                .diagnostics
333                .iter()
334                .filter(|d| d.severity != Severity::Info)
335                .map(|d| {
336                    format!(
337                        "{}: {} ({})",
338                        d.code,
339                        d.message,
340                        d.path.as_deref().unwrap_or("-")
341                    )
342                })
343                .collect(),
344        },
345    );
346
347    // --- the registry: it exists, therefore it validated; a registry that does not build
348    // stops the process before any transport is bound
349    let summary = ctx.registry.summary();
350    record(
351        &mut checks,
352        &ctx.progress,
353        HealthCheck {
354            id: "registry".into(),
355            title: "The capability registry".into(),
356            status: HealthStatus::Ok,
357            detail: format!(
358                "{} capabilities ({} builtin, {} declarative) across {} modules; fingerprint {}",
359                summary.total,
360                summary.builtin,
361                summary.declarative,
362                summary.modules,
363                &ctx.registry.fingerprint()[..12.min(ctx.registry.fingerprint().len())]
364            ),
365            decided_by:
366                "the registry builder; it refuses to build on a duplicate or a malformed exposure"
367                    .into(),
368            evidence: vec!["majordomus capabilities validate".into()],
369            findings: Vec::new(),
370        },
371    );
372
373    // --- the scope: what a worker reads of this repository and what it never reads
374    let tally = &index.scoped.tally;
375    record(
376        &mut checks,
377        &ctx.progress,
378        HealthCheck {
379            id: "scope".into(),
380            title: "The declared scope".into(),
381            status: HealthStatus::Ok,
382            detail: format!(
383                "{} of {} tracked file(s) in scope, {} out, read from {}",
384                tally.r#in, tally.files, tally.out, index.repository.scope_path
385            ),
386            decided_by: "the scope declaration the index was built under".into(),
387            evidence: vec!["majordomus scope".into()],
388            findings: Vec::new(),
389        },
390    );
391
392    // --- git
393    let (git_status, git_detail) = match &index.repository.git {
394        GitState::Available(info) => {
395            let head = info.head.as_deref().unwrap_or("(unborn)");
396            (
397                if info.working_tree == "clean" {
398                    HealthStatus::Ok
399                } else {
400                    HealthStatus::Warn
401                },
402                format!(
403                    "{} at {}, working tree {}",
404                    info.branch.as_deref().unwrap_or("(detached)"),
405                    &head[..12.min(head.len())],
406                    info.working_tree
407                ),
408            )
409        }
410        GitState::Unavailable { reason } => (HealthStatus::Unknown, reason.clone()),
411    };
412    record(
413        &mut checks,
414        &ctx.progress,
415        HealthCheck {
416            id: "git".into(),
417            title: "Version control".into(),
418            status: git_status,
419            detail: git_detail,
420            decided_by: "git, as the index asked it when it was built".into(),
421            evidence: vec!["git status".into()],
422            findings: Vec::new(),
423        },
424    );
425
426    // --- benchmark coverage, from the projection that defines the targets
427    let projection = BenchmarkProjection::from_context(ctx);
428    let coverage = Coverage::compute(ctx, &projection);
429    let total = coverage.tallies.get("total").cloned().unwrap_or_default();
430    record(
431        &mut checks,
432        &ctx.progress,
433        HealthCheck {
434            id: "benchmark-coverage".into(),
435            title: "Benchmark coverage".into(),
436            status: if total.missing > 0 {
437                HealthStatus::Fail
438            } else if total.waived > 0 {
439                HealthStatus::Warn
440            } else {
441                HealthStatus::Ok
442            },
443            detail: format!(
444                "{} of {} target(s) covered, {} missing, {} waived",
445                total.covered, total.required, total.missing, total.waived
446            ),
447            decided_by:
448                "the benchmark projection's coverage, the same one `bench coverage --check` reads"
449                    .into(),
450            evidence: vec!["majordomus bench coverage --check".into()],
451            findings: coverage
452                .lines
453                .iter()
454                .filter(|l| matches!(l.state, crate::bench::CoverageState::Missing))
455                .map(|l| format!("{} on {}: no case", l.subject, l.transport.name()))
456                .collect(),
457        },
458    );
459
460    // --- the committed registry manifest against the registry this process built.
461    // Only that one artifact: it is derived from the code alone, and rendering it costs
462    // nothing but serialising the descriptors already in memory. Rendering the OpenAPI
463    // document, the reference and the benchmark matrix on every health call would rebuild
464    // canonical state per request, which this executable does not do; `generate --check`
465    // is the complete answer and this check names it.
466    let root = std::path::Path::new(&index.repository.root);
467    let manifest = generate::Document::new(
468        "registry",
469        generate::REGISTRY_SCHEMA,
470        "the canonical capability registry",
471        generate::registry_manifest(&ctx.registry),
472    )
473    .artifacts(crate::VERSION);
474    let manifest = manifest
475        .iter()
476        .find(|a| a.format == generate::ArtifactFormat::Json)
477        .map(|a| a.content.clone())
478        .unwrap_or_default();
479    let path = format!("{}/registry.json", generate::OUT_DIR);
480    let committed = std::fs::read_to_string(root.join(&path));
481    let (status, detail, findings) = match &committed {
482        Ok(text) if *text == manifest => (
483            HealthStatus::Ok,
484            format!("{path} matches the registry this process built"),
485            Vec::new(),
486        ),
487        Ok(_) => (
488            HealthStatus::Fail,
489            format!("{path} differs from the registry this process built"),
490            vec![format!("{path} (differs)")],
491        ),
492        Err(_) => (
493            HealthStatus::Unknown,
494            format!("{path} could not be read"),
495            vec![format!("{path} (missing)")],
496        ),
497    };
498    record(
499        &mut checks,
500        &ctx.progress,
501        HealthCheck {
502            id: "generated-registry".into(),
503            title: "Committed projections".into(),
504            status,
505            detail,
506            decided_by:
507                "the same rendering `majordomus generate` writes, compared with what is committed"
508                    .into(),
509            evidence: vec!["majordomus generate --check".into()],
510            findings,
511        },
512    );
513
514    // --- handovers published from other machines
515    //
516    // `continuity.status` decides it: every record of the local continuity store admitted
517    // or refused, and the lineage read from what was admitted. A refused record (forged,
518    // foreign, leaking) or a broken line fails; a diverged line, a dangling parent or a
519    // record too new to read warns. A repository that never published is not a fault.
520    let continuity = crate::continuity::Machine::open_read(
521        root,
522        super::mesh::declaration(ctx).and_then(Result::ok),
523    )
524    .and_then(|m| crate::continuity::status(&m));
525    let (status, detail, findings) = continuity_health(continuity);
526    record(
527        &mut checks,
528        &ctx.progress,
529        HealthCheck {
530            id: "continuity".into(),
531            title: "Handovers published across machines".into(),
532            status,
533            detail,
534            decided_by: "the admission and lineage checks of `continuity.status`".into(),
535            evidence: vec!["majordomus-cli continuity status".into()],
536            findings,
537        },
538    );
539
540    // --- the shared server of this checkout
541    //
542    // Not a second opinion and not a second reading: `server::standing_at` is what answers
543    // `GET /api/v1/server`, and this reports what it says. It costs one small file and,
544    // when the lease names an address, one probe bounded by `lease::PROBE_TIMEOUT`; the
545    // five-second cache on `health.report` is as fresh as a file other processes write can
546    // usefully be, and `server.status` is the uncached answer for a caller that needs one.
547    let checkout = root.canonicalize().unwrap_or_else(|_| root.to_path_buf());
548    let local_half = server::local_half(&checkout);
549    let lease_path = crate::lease::lease_file(&checkout, &local_half);
550    let (standing, reason, lease) = server::standing_at(&checkout, &local_half);
551    let detail = match lease.document() {
552        Some(doc) => format!(
553            "{}: {}, version {}, pid {}, since {}",
554            standing.as_str(),
555            doc.url.as_deref().unwrap_or("no address published yet"),
556            doc.version.as_deref().unwrap_or("unstated"),
557            doc.pid,
558            doc.started_at
559        ),
560        None if standing == ServerStanding::Absent => format!(
561            "absent: no lease at {}, so nothing serves this checkout",
562            lease_path.display()
563        ),
564        None => format!(
565            "{}: the lease at {} holds no document",
566            standing.as_str(),
567            lease_path.display()
568        ),
569    };
570    // Whose report this is. Everything above describes the checkout's server, read from the
571    // lease — and a process whose lease was taken over describes, accurately, a server that
572    // is not itself. The one caller that has reached the wrong process is the one caller
573    // who cannot afford to be told that the right server is fine, so this outranks the
574    // standing rather than qualifying it.
575    let orphaned = crate::lease::was_lost();
576    let mut findings: Vec<String> = reason.into_iter().collect();
577    if orphaned {
578        findings.push(
579            "this process no longer holds the lease: it serves the sessions it already had \
580             and takes on no new ones; what is described above is another process"
581                .into(),
582        );
583    }
584    record(
585        &mut checks,
586        &ctx.progress,
587        HealthCheck {
588            id: "server".into(),
589            title: "The shared server".into(),
590            status: if orphaned {
591                HealthStatus::Warn
592            } else {
593                match standing {
594                    // A checkout nobody serves is not an unhealthy one: nothing is running
595                    // and nothing claims to be. A lease still binding resolves itself
596                    // within the bind grace, and is reported as stale by this same engine
597                    // when it does not. Neither is a fault of what this process serves.
598                    ServerStanding::Absent | ServerStanding::Starting | ServerStanding::Ready => {
599                        HealthStatus::Ok
600                    }
601                    // A server answering from code this tree no longer has says yesterday's
602                    // truth in today's words, and a lease naming one that answers for
603                    // nobody sends the next reader at a dead address. Both are somebody's
604                    // to clear; neither stops this process from serving, so neither is a
605                    // failure.
606                    ServerStanding::Outdated | ServerStanding::Stale => HealthStatus::Warn,
607                }
608            },
609            detail: if orphaned {
610                format!("{detail}; and this process is not it")
611            } else {
612                detail
613            },
614            decided_by:
615                "the decision `server.status` makes, from this checkout's lease and one probe of the server it names"
616                    .into(),
617            evidence: vec!["majordomus serve status".into()],
618            findings,
619        },
620    );
621
622    // --- the peers attached to this process
623    //
624    // Decided from the board and tallied the way preflight tallies it, so the two cards on
625    // one page cannot count the same board two ways. Only a process that holds this
626    // checkout's lease holds its board: anything else — the command line, a process whose
627    // lease was never published — has an empty board of its own, and an empty board it does
628    // not own is no evidence that nobody is attached.
629    record(
630        &mut checks,
631        &ctx.progress,
632        peers_check(
633            crate::lease::held()
634                .is_some()
635                .then(|| ctx.peers.list())
636                .as_deref(),
637        ),
638    );
639
640    let status = checks
641        .iter()
642        .fold(HealthStatus::Ok, |acc, c| acc.worse(c.status));
643    let mut tallies: BTreeMap<String, usize> = BTreeMap::new();
644    for c in &checks {
645        *tallies.entry(c.status.as_str().into()).or_insert(0) += 1;
646    }
647    let observed = [
648        ("layer", &index.repository.observed.index),
649        ("git", &index.repository.observed.git),
650    ]
651    .into_iter()
652    .filter(|(id, o)| !o.observed_at.is_empty() && checks.iter().any(|c| c.id == *id))
653    .map(|(id, o)| (id.to_string(), o.clone()))
654    .collect();
655    Ok(Health {
656        status,
657        tallies,
658        checks,
659        observed,
660    })
661}

716/// What the continuity store says about the repository's health: the worst severity among
717/// the status's diagnostics, how many records were admitted, refused and are resumable, and
718/// every diagnostic that is more than information. A store that could not be read is
719/// unknown, never ok.
720fn continuity_health(
721    status: Result<crate::continuity::Status, String>,
722) -> (HealthStatus, String, Vec<String>) {
723    match status {
724        Ok(st) => {
725            let worst = st
726                .diagnostics
727                .iter()
728                .map(|d| match d.severity {
729                    Severity::Error => HealthStatus::Fail,
730                    Severity::Warning => HealthStatus::Warn,
731                    _ => HealthStatus::Ok,
732                })
733                .fold(HealthStatus::Ok, HealthStatus::worse);
734            (
735                worst,
736                format!(
737                    "{} record(s) admitted, {} refused, {} resumable from another device",
738                    st.store.records,
739                    st.store.refused,
740                    st.resumable.len()
741                ),
742                st.diagnostics
743                    .iter()
744                    .filter(|d| d.severity != Severity::Info)
745                    .map(|d| {
746                        format!(
747                            "{} {}{}",
748                            d.code,
749                            d.path
750                                .as_deref()
751                                .map(|p| format!("{p}: "))
752                                .unwrap_or_default(),
753                            d.message
754                        )
755                    })
756                    .collect(),
757            )
758        }
759        Err(e) => (
760            HealthStatus::Unknown,
761            format!("the continuity store could not be read: {e}"),
762            Vec::new(),
763        ),
764    }
765}

868    /// note that is only information is not a finding, and a store that could not be read

873        use crate::continuity::{self as domain, PublishRequest};

875        let a_root = w.root("a");

879        assert_eq!(status, HealthStatus::Ok);

884        assert!(findings.is_empty(), "{findings:?}");
