1//! The UI as a static snapshot: `majordomus ui`, read once at a pinned commit and written
2//! as files a publication can carry (ADR 0141).
3//!
4//! The UI (ADR 0135) is a projection a running process answers; a publication has no
5//! process. So the snapshot asks the UI itself — the same [`crate::ui::Ui`] the server
6//! routes to, in-process, over one reading of the repository — for every screen and panel,
7//! and then for the pages those link to, breadth first and up to a bound. Nothing is
8//! rendered here: each page is the UI's own HTML. What changes is only what a file cannot
9//! do that a route can:
10//!
11//! - **a link** to another page of the UI becomes a relative link to that page's file, so
12//!   the directory works at any mount and from disk. An inspector's query (`?rule=…`) is a
13//!   directory of its own; a query of more than one pair, or a page past the bound, is not
14//!   in the snapshot and its link says so (below);
15//! - **a link out of the UI** stays as it is when a publication holds what it names (the
16//!   topology says which surfaces are published), and otherwise leads to the snapshot's one
17//!   page about the live UI, carrying the path it named, because only a running process
18//!   answers it: never a link to nothing;
19//! - **the script** is told it is in a snapshot (`data-snapshot` on the body), so it reads
20//!   nothing again and fetches no panel: a static page that refreshed itself would replace
21//!   what was read with whatever the host answers for a query it ignores;
22//! - **every page says what it is**: read at which commit, when, and by which executable,
23//!   and how to see it live.
24//!
25//! A value the UI could not read when the snapshot was taken reads as the UI says it — none,
26//! unknown or not answered — and never as zero, because the UI's own rule (`ui-derived-state`)
27//! holds for every page this copies.
28

42/// The surface a snapshot is taken of: what tells the UI's snapshot from the Cockpit's.
43///
44/// One exporter reads both. Everything that differs between them is a value of this type,
45/// so that a third surface is a constant here and not a second exporter.
46#[derive(Debug, Clone, Copy, PartialEq, Eq)]
47pub struct Subject {
48    /// The snapshot's identity in the web topology, and its directory under the web root.
49    pub id: &'static str,
50    /// What produces it, as a reader would run it.
51    pub producer: &'static str,
52    /// The identity of the surface a process serves at the same mount.
53    pub served: &'static str,
54    /// The mount: the snapshot is published where the process serves the surface.
55    pub prefix: &'static str,
56    /// The prefix of the surface's assets.
57    pub assets: &'static str,
58    /// The surface in a sentence: "UI", "Cockpit".
59    pub name: &'static str,
60    /// The declaration's title.
61    pub title: &'static str,
62    /// The title of the page about the live surface.
63    pub live_title: &'static str,
64    /// A query key that asks for a part of a page and is no page of its own; empty for none.
65    pub fragment: &'static str,
66    /// Whether a page keeps its scripts. The UI's one script knows a snapshot and reads
67    /// nothing again in one. The Cockpit's scripts open sockets, poll and post to the
68    /// process, so a snapshot of it carries none: every page is complete without them.
69    pub scripts: bool,
70    /// Routes a publication never holds, whatever links to them: a page about the machine
71    /// that renders it and not about the repository. Such a route is no page of the
72    /// snapshot, so it is never read, and a link to it leads to the live page.
73    pub withheld: &'static [&'static str],
74}

112/// The pages a snapshot holds at most when the caller names no bound: every screen and

116/// How long a run reads inspectors for when the caller names no budget, in seconds. The
117/// screens and panels are read whatever they cost; what they link to is read until the page
118/// bound or this is reached, whichever is first, because an inspector's cost is its source's
119/// (a claim's history, a pull request's checks) and a publication must not wait on the sum.

129/// What a snapshot run wrote.
130#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
131pub struct Snapshot {
132    /// The directory, absolute.
133    #[serde(skip)]
134    pub dir: PathBuf,
135    /// Where it came from.
136    pub origin: Origin,
137    /// The mount it is published at: the surface's own.
138    pub mount: String,
139    /// Every page written, as the target it was read from and the file it is in.
140    pub pages: BTreeMap<String, String>,
141    /// The asset files copied beside the pages.
142    pub assets: Vec<String>,
143    /// UI pages a page links to that are not in the snapshot: past the bound, or a query no
144    /// file can stand for. Each link to one leads to the live page.
145    pub left_out: Vec<String>,
146    /// The bound the run was given.
147    pub limit: usize,
148    /// The seconds the run was given for the pages the screens link to.
149    pub budget_seconds: u64,
150    /// What ended the reading: `complete` (every linked page was written), `limit` or `budget`.
151    pub stopped_by: &'static str,
152}

154/// Read a surface and write it under the generated web root, with the declaration that makes
155/// the directory a surface of every publication.
156///
157/// `handle` is the surface itself, asked in-process: the same function the server routes a
158/// request to. `seeds` are the pages read first and whatever they cost, in order; what they
159/// link to is read after, until the bound or the budget.
160pub fn render(
161    root: &Path,
162    subject: &Subject,
163    handle: &dyn Fn(&Request) -> Response,
164    seeds: Vec<String>,
165    topology: &Topology,
166    limit: usize,
167    budget_seconds: u64,
168) -> Result<Snapshot> {
169    let started = std::time::Instant::now();
170    let budget = std::time::Duration::from_secs(budget_seconds);
171    let mut stopped_by = "limit";
172    let dir = root.join(GENERATED_ROOT).join(subject.id);
173    if dir.exists() {
174        // a snapshot is one reading: pages of an earlier one must not survive into it
175        std::fs::remove_dir_all(&dir).map_err(|e| Error::io(dir.display().to_string(), e))?;
176    }
177    std::fs::create_dir_all(&dir).map_err(|e| Error::io(dir.display().to_string(), e))?;
178    let origin = Origin::read(root);
179    let outside = Outside::from(root, subject, topology);
180
181    // The screens and panels first, in the navigation's order. What they link to is then
182    // taken one page per family in turn (a family is a route and the one key it is asked
183    // with: every rule's inspector, every issue's), so that a bound leaves each family the
184    // same share instead of the first long list all of it.
185    let mut seeds: VecDeque<String> = seeds.into();
186    let mut seen: BTreeSet<String> = seeds.iter().cloned().collect();
187    let mut families: Vec<(String, VecDeque<String>)> = Vec::new();
188    let mut turn = 0usize;
189    let mut pages: BTreeMap<String, String> = BTreeMap::new();
190    let mut rendered: Vec<(String, String)> = Vec::new();
191    let mut assets: BTreeSet<String> = BTreeSet::new();
192    while pages.len() < limit.max(1) {
193        let target = match seeds.pop_front() {
194            Some(seed) => seed,
195            None => {
196                // the next family that still has a page, starting after the last one served
197                let waiting = families.len();
198                let next = (0..waiting)
199                    .map(|i| (turn + i) % waiting)
200                    .find(|i| !families[*i].1.is_empty());
201                let Some(i) = next else {
202                    stopped_by = "complete";
203                    break;
204                };
205                if started.elapsed() >= budget {
206                    stopped_by = "budget";
207                    break;
208                }
209                turn = i + 1;
210                match families[i].1.pop_front() {
211                    Some(target) => target,
212                    None => break,
213                }
214            }
215        };
216        let Some(file) = subject.page_file(&target) else {
217            continue;
218        };
219        let asked = std::time::Instant::now();
220        let response = handle(&Request::parse_target("GET", &target, Vec::new()));
221        tracing::debug!(target = %target, ms = asked.elapsed().as_millis() as u64, status = response.status, "snapshot page");
222        if response.status != 200 {
223            continue;
224        }
225        let html = response.body.text().into_owned();
226        for link in links(&html) {
227            let link = unescape(&link);
228            if let Some(name) = subject.asset_name(&link) {
229                assets.insert(name);
230            } else if subject.page_file(&link).is_some() {
231                let key = subject.canonical(&link);
232                if seen.insert(key.clone()) {
233                    let family = family(&key);
234                    match families.iter_mut().find(|(name, _)| *name == family) {
235                        Some((_, queue)) => queue.push_back(key),
236                        None => families.push((family, VecDeque::from([key]))),
237                    }
238                }
239            }
240        }
241        pages.insert(target.clone(), file.clone());
242        rendered.push((file, html));
243    }
244
245    // the assets first: a page names only the ones the snapshot holds
246    let mut copied = Vec::new();
247    for name in &assets {
248        let response = handle(&Request::parse_target(
249            "GET",
250            &format!("{}{name}", subject.assets),
251            Vec::new(),
252        ));
253        if response.status != 200 {
254            continue; // the page's own fallback stands: the tables beside a graph, unstyled markup
255        }
256        let path = dir.join("assets").join(name);
257        if let Some(parent) = path.parent() {
258            std::fs::create_dir_all(parent)
259                .map_err(|e| Error::io(parent.display().to_string(), e))?;
260        }
261        std::fs::write(&path, response.body.as_bytes())
262            .map_err(|e| Error::io(path.display().to_string(), e))?;
263        copied.push(name.clone());
264    }
265    let held: BTreeSet<String> = copied.iter().cloned().collect();
266    // the pages a link may lead to are exactly the ones written: decided before any rewrite
267    let written: BTreeMap<String, String> = pages.clone();
268    let mut left_out: BTreeSet<String> = BTreeSet::new();
269    let mut frame: Option<String> = None;
270    for (file, html) in &rendered {
271        let page = rewrite(
272            subject,
273            html,
274            file,
275            &written,
276            &held,
277            &outside,
278            &origin,
279            &mut left_out,
280        );
281        write(&dir, file, &page)?;
282        if file == "index.html" {
283            frame = Some(page);
284        }
285    }
286    let live = live_page(subject, frame.as_deref(), &origin);
287    write(&dir, LIVE_PAGE, &live)?;
288
289    let snapshot = Snapshot {
290        dir: dir.clone(),
291        origin,
292        mount: subject.prefix.to_string(),
293        pages,
294        assets: copied.iter().map(|n| format!("assets/{n}")).collect(),
295        left_out: left_out.into_iter().collect(),
296        limit,
297        budget_seconds,
298        stopped_by,
299    };
300    write_json(&dir, "snapshot.json", &snapshot)?;
301    declare(subject, &dir, &snapshot.origin)?;
302    // the snapshots written before this one led their links into this surface to their live
303    // page, because this one did not exist yet: now it does, and they lead to its pages
304    for other in SUBJECTS.iter().filter(|other| other.id != subject.id) {
305        relink_into(root, other, subject, &snapshot.pages)?;
306    }
307    Ok(snapshot)
308}

310/// Lead the links of an earlier snapshot (`from`) into a surface that has a snapshot now
311/// (`to`, holding `pages`): every link `from` led to its live page because `to` had none is
312/// led to the page itself when `to` holds it, and left where it was when it does not.
313///
314/// Two snapshots link into each other, and one of them is written first. Without this the
315/// first would be read twice, once to learn what it links to and once to link it; with it,
316/// each surface is read once, in either order, and the files say the same thing. A snapshot
317/// that is not there has nothing to correct. Returns how many links were led.
318fn relink_into(
319    root: &Path,
320    from: &Subject,
321    to: &Subject,
322    pages: &BTreeMap<String, String>,
323) -> Result<usize> {
324    let Some(files) = pages_of(root, from) else {
325        return Ok(0);
326    };
327    let dir = root.join(GENERATED_ROOT).join(from.id);
328    let led = regex::Regex::new(&format!(
329        r##"(?:\.\./)*{}#({}(?:/[^"#]*)?)"##,
330        regex::escape(LIVE_PAGE),
331        regex::escape(to.prefix)
332    ))
333    .expect("the pattern of a link led to the live page compiles");
334    let mut count = 0;
335    let mut names: Vec<String> = files.into_values().collect();
336    crate::order::canonical_strings(&mut names);
337    for name in names {
338        let path = dir.join(&name);
339        let Ok(html) = std::fs::read_to_string(&path) else {
340            continue;
341        };
342        let mut changed = 0;
343        let page = led.replace_all(&html, |c: &regex::Captures<'_>| {
344            match pages.get(&to.canonical(&unescape(&c[1]))) {
345                Some(file) => {
346                    changed += 1;
347                    let directory = file.strip_suffix("index.html").unwrap_or(file);
348                    escape(&format!("{}/{directory}", to.prefix))
349                }
350                None => c[0].to_string(),
351            }
352        });
353        if changed > 0 {
354            std::fs::write(&path, page.as_bytes())
355                .map_err(|e| Error::io(path.display().to_string(), e))?;
356            count += changed;
357        }
358    }
359    Ok(count)
360}

362/// The pages the Cockpit's snapshot reads first: every static route but the one page about
363/// this machine, then every Cockpit page the UI's snapshot under `root` wanted to link to.
364///
365/// The page about this machine is the subject's to withhold ([`Subject::withheld`]), so it
366/// is no seed and no link reaches it either. The UI's links come from the UI's own
367/// snapshot, where a link into a Cockpit that was not yet published was led to the live
368/// page with the path it meant: those paths are what a reader of the UI will follow first,
369/// so they are read before the bound can leave them out. No list of them is kept anywhere.
370pub fn cockpit_seeds(root: &Path) -> Vec<String> {
371    let mut seeds: Vec<String> = crate::cockpit::STATIC_ROUTES
372        .iter()
373        .map(|(path, _)| path.to_string())
374        .filter(|path| COCKPIT.page_file(path).is_some())
375        .collect();
376    let wanted = regex::Regex::new(&format!(
377        r##"{}#({}/[^"#&<> ]*)"##,
378        regex::escape(LIVE_PAGE),
379        regex::escape(COCKPIT.prefix)
380    ))
381    .expect("the pattern of a link the UI led to its live page compiles");
382    let ui = root.join(GENERATED_ROOT).join(UI.id);
383    let mut names: Vec<String> = pages_of(root, &UI)
384        .unwrap_or_default()
385        .into_values()
386        .collect();
387    crate::order::canonical_strings(&mut names);
388    for file in names.into_iter().map(|name| ui.join(name)) {
389        let Ok(html) = std::fs::read_to_string(&file) else {
390            continue;
391        };
392        for found in wanted.captures_iter(&html) {
393            let target = COCKPIT.canonical(&unescape(&found[1]));
394            if COCKPIT.page_file(&target).is_some() && !seeds.contains(&target) {
395                seeds.push(target);
396            }
397        }
398    }
399    seeds
400}

402/// The declaration beside the pages: a surface of every publication at the surface's own
403/// mount, which no process serves, because a process serves the surface itself there.
404fn declare(subject: &Subject, dir: &Path, origin: &Origin) -> Result<()> {
405    let declaration = Declaration {
406        schema: DECLARATION_SCHEMA.into(),
407        id: subject.id.into(),
408        mount: subject.prefix.into(),
409        title: Some(subject.title.into()),
410        index: Some("index.html".into()),
411        producer: Some(subject.producer.into()),
412        availability: Some(Availability::PublishedOnly),
413        category: Some(Category::Interface),
414        visibility: Some(Visibility::Public),
415        built_from: origin.revision.clone(),
416    };
417    let mut body =
418        serde_json::to_string_pretty(&declaration).map_err(|e| Error::InvalidSurface {
419            surface: subject.id.into(),
420            reason: format!("its declaration cannot be serialised: {e}"),
421        })?;
422    body.push('\n');
423    write(dir, DECLARATION_FILE, &body)
424}

426/// Where a link that leaves the surface goes in a publication.
427struct Outside {
428    /// Mounts a publication holds, so a link under one keeps its path.
429    kept: Vec<crate::web::model::Mount>,
430    /// Mounts a process serves from the application's own source (the documentation build),
431    /// whose pages a publication holds at the application's root instead.
432    rebased: Vec<crate::web::model::Mount>,
433    /// The other snapshots under the same web root, each with the pages it holds. A link
434    /// into one leads to the page when the snapshot has it and to the live page when it
435    /// does not: a snapshot is bounded, so its mount being published says nothing about one
436    /// page of it.
437    snapshots: Vec<(&'static Subject, BTreeMap<String, String>)>,
438}

440/// The pages another snapshot under `root` holds, read from its own account of itself;
441/// none when it has not been written.
442fn pages_of(root: &Path, subject: &Subject) -> Option<BTreeMap<String, String>> {
443    let path = root
444        .join(GENERATED_ROOT)
445        .join(subject.id)
446        .join("snapshot.json");
447    let text = std::fs::read_to_string(path).ok()?;
448    let value: serde_json::Value = serde_json::from_str(&text).ok()?;
449    let pages = value.get("pages")?.as_object()?;
450    Some(
451        pages
452            .iter()
453            .filter_map(|(target, file)| Some((target.clone(), file.as_str()?.to_string())))
454            .collect(),
455    )
456}

458impl Outside {
459    fn from(root: &Path, subject: &Subject, topology: &Topology) -> Self {
460        let from_site = |s: &crate::web::model::Surface| {
461            matches!(
462                s.provenance.get("kind"),
463                Some(Provenance::SiteConfig { .. })
464            )
465        };
466        let published_root = topology
467            .surfaces
468            .iter()
469            .any(|s| s.mount.is_root() && s.availability.is_published() && from_site(s));
470        Outside {
471            kept: topology
472                .surfaces
473                .iter()
474                .filter(|s| !s.mount.is_root() && s.availability.is_published())
475                // a surface a snapshot stands for is never kept by its mount alone
476                .filter(|s| !SUBJECTS.iter().any(|x| s.id == x.served || s.id == x.id))
477                .map(|s| s.mount.clone())
478                .collect(),
479            rebased: topology
480                .surfaces
481                .iter()
482                .filter(|s| published_root && !s.availability.is_published())
483                .filter(|s| s.kind == SurfaceKind::StaticDirectory && from_site(s))
484                .map(|s| s.mount.clone())
485                .collect(),
486            snapshots: SUBJECTS
487                .iter()
488                .filter(|other| other.id != subject.id)
489                .filter_map(|other| Some((*other, pages_of(root, other)?)))
490                .collect(),
491        }
492    }
493
494    /// Where a link into another snapshot leads: `Some(Some(href))` for a page it holds,
495    /// `Some(None)` for a page of that surface it does not hold, and `None` for a link that
496    /// is not into another snapshot's surface at all. `link` carries its query and no
497    /// fragment.
498    fn snapshot(&self, link: &str) -> Option<Option<String>> {
499        let path = link.split('?').next().unwrap_or(link);
500        let (other, pages) = self.snapshots.iter().find(|(other, _)| {
501            path == other.prefix || path.starts_with(&format!("{}/", other.prefix))
502        })?;
503        Some(pages.get(&other.canonical(link)).map(|file| {
504            let directory = file.strip_suffix("index.html").unwrap_or(file);
505            format!("{}/{directory}", other.prefix)
506        }))
507    }
508
509    /// The link a publication answers for `path`, or none when only a process does.
510    fn resolve(&self, path: &str) -> Option<String> {
511        if path == "/" {
512            return Some("/".into());
513        }
514        if self.kept.iter().any(|m| m.owns(path)) {
515            return Some(path.into());
516        }
517        self.rebased.iter().find(|m| m.owns(path)).map(|m| {
518            let rest = path[m.as_str().len()..].trim_start_matches('/');
519            format!("/{rest}")
520        })
521    }
522}

524impl Subject {
525    /// The file a target is written to, relative to the snapshot's root; none for a target
526    /// that is not a page of the surface, or that no file can stand for.
527    ///
528    /// ```text
529    /// /ui                       index.html
530    /// /ui/roadmap               roadmap/index.html
531    /// /ui/roadmap?milestone=M1  roadmap/_milestone/M1/index.html
532    /// /ui/issues?a=1&b=2        none: two pairs are a query, not a page
533    /// ```
534    fn page_file(&self, target: &str) -> Option<String> {
535        let (path, query) = target.split_once('?').unwrap_or((target, ""));
536        let path = path.split('#').next().unwrap_or(path);
537        let route = path.trim_end_matches('/');
538        if self
539            .withheld
540            .iter()
541            .any(|w| route == *w || route.starts_with(&format!("{w}/")))
542        {
543            return None;
544        }
545        if path.starts_with(self.assets) {
546            return None;
547        }
548        let rest = if path == self.prefix {
549            ""
550        } else {
551            path.strip_prefix(&format!("{}/", self.prefix))?
552        };
553        let rest = rest.trim_end_matches('/');
554        if rest
555            .split('/')
556            .any(|s| s.is_empty() && !rest.is_empty() || s == "." || s == "..")
557        {
558            return None;
559        }
560        let base = if rest.is_empty() {
561            String::new()
562        } else {
563            format!("{rest}/")
564        };
565        let pairs: Vec<(String, String)> = query
566            .split('&')
567            .filter(|p| !p.is_empty())
568            .map(|p| {
569                let (k, v) = p.split_once('=').unwrap_or((p, ""));
570                (decode(k), decode(v))
571            })
572            .filter(|(k, _)| self.fragment.is_empty() || k != self.fragment)
573            .collect();
574        match pairs.as_slice() {
575            [] => Some(format!("{base}index.html")),
576            [(k, v)] if !k.is_empty() && !v.is_empty() => {
577                Some(format!("{base}_{}/{}/index.html", slug(k), slug(v)))
578            }
579            _ => None,
580        }
581    }
582}

584/// The family of a page: its route and the key it is asked with, without the value.
585fn family(target: &str) -> String {
586    let (path, query) = target.split_once('?').unwrap_or((target, ""));
587    let key = query.split('=').next().unwrap_or("");
588    format!("{path}?{key}")
589}

591impl Subject {
592    /// One spelling per page, so that two links to one page are one page.
593    fn canonical(&self, target: &str) -> String {
594        let target = target.split('#').next().unwrap_or(target);
595        let (path, query) = target.split_once('?').unwrap_or((target, ""));
596        let path = if path.len() > self.prefix.len() {
597            path.trim_end_matches('/')
598        } else {
599            path
600        };
601        let query: Vec<&str> = query
602            .split('&')
603            .filter(|p| !p.is_empty() && (self.fragment.is_empty() || *p != self.fragment))
604            .collect();
605        if query.is_empty() {
606            path.to_string()
607        } else {
608            format!("{path}?{}", query.join("&"))
609        }
610    }
611}

613/// A value as a path segment of its own: the characters a URL and every file system carry
614/// unchanged, and every other byte as `_` and its hex, so two values never share a file.
615fn slug(value: &str) -> String {
616    let mut out = String::with_capacity(value.len());
617    for b in value.bytes() {
618        if b.is_ascii_alphanumeric() || b == b'-' || b == b'.' {
619            out.push(b as char);
620        } else {
621            out.push_str(&format!("_{b:02x}"));
622        }
623    }
624    if out.starts_with('.') {
625        out.replace_range(0..1, "_2e");
626    }
627    out
628}
629
630/// A percent-decoded query component.
631fn decode(raw: &str) -> String {
632    let bytes = raw.as_bytes();
633    let hex = |b: u8| (b as char).to_digit(16).map(|d| d as u8);
634    let mut out = Vec::with_capacity(bytes.len());
635    let mut i = 0;
636    while i < bytes.len() {
637        match (
638            bytes[i],
639            bytes.get(i + 1).copied().and_then(hex),
640            bytes.get(i + 2).copied().and_then(hex),
641        ) {
642            (b'%', Some(hi), Some(lo)) => {
643                out.push(hi * 16 + lo);
644                i += 3;
645            }
646            (b'+', _, _) => {
647                out.push(b' ');
648                i += 1;
649            }
650            (b, _, _) => {
651                out.push(b);
652                i += 1;
653            }
654        }
655    }
656    String::from_utf8_lossy(&out).into_owned()
657}

659impl Subject {
660    /// The name of an asset of the surface a link names, without its digest.
661    fn asset_name(&self, link: &str) -> Option<String> {
662        let rest = link.strip_prefix(self.assets)?;
663        let name = rest.split(['?', '#']).next().unwrap_or(rest);
664        (!name.is_empty() && !name.split('/').any(|s| s.is_empty() || s == ".."))
665            .then(|| name.to_string())
666    }

669/// The attributes whose value is a URL a reader or the script follows.
670const URL_ATTRIBUTES: &str = "href|src|action|data-graph-lib|data-panel-src";
671
672fn attribute_pattern() -> regex::Regex {
673    regex::Regex::new(&format!(r#"\b({URL_ATTRIBUTES})="([^"]*)""#))
674        .expect("the attribute pattern compiles")
675}
676
677/// The link a graph's node carries, inside its JSON in an attribute.
678fn graph_href_pattern() -> regex::Regex {
679    regex::Regex::new(r#"&quot;href&quot;:&quot;(.*?)&quot;"#).expect("the graph pattern compiles")
680}

682/// Every URL a page names, as written (still escaped).
683fn links(html: &str) -> Vec<String> {
684    let mut out: Vec<String> = attribute_pattern()
685        .captures_iter(html)
686        .map(|c| c[2].to_string())
687        .collect();
688    out.extend(
689        graph_href_pattern()
690            .captures_iter(html)
691            .map(|c| c[1].to_string()),
692    );
693    out
694}

696fn unescape(value: &str) -> String {
697    value
698        .replace("&quot;", "\"")
699        .replace("&#39;", "'")
700        .replace("&lt;", "<")
701        .replace("&gt;", ">")
702        .replace("&amp;", "&")
703}
704
705fn escape(value: &str) -> String {
706    value
707        .replace('&', "&amp;")
708        .replace('"', "&quot;")
709        .replace('<', "&lt;")
710        .replace('>', "&gt;")
711}

713/// The path from a file of the snapshot back to its root.
714fn up(file: &str) -> String {
715    "../".repeat(file.matches('/').count())
716}

718/// Where one link of the page in `file` leads in the snapshot.
719fn relink(
720    subject: &Subject,
721    link: &str,
722    file: &str,
723    written: &BTreeMap<String, String>,
724    held: &BTreeSet<String>,
725    outside: &Outside,
726    left_out: &mut BTreeSet<String>,
727) -> String {
728    if link.is_empty() || link.starts_with('#') || !link.starts_with('/') || link.starts_with("//")
729    {
730        return link.to_string(); // the page itself, another origin, or a scheme: unchanged
731    }
732    let fragment = link.find('#').map(|i| &link[i..]).unwrap_or("");
733    if let Some(name) = subject.asset_name(link) {
734        // an asset the distribution did not have is named nowhere: the page's own fallback
735        // stands (the tables beside a graph), and no link leads to nothing
736        return if held.contains(&name) {
737            format!("{}assets/{name}", up(file))
738        } else {
739            String::new()
740        };
741    }
742    if subject.page_file(link).is_some()
743        || link == subject.prefix
744        || link.starts_with(&format!("{}/", subject.prefix))
745    {
746        if let Some(target) = written.get(&subject.canonical(link)) {
747            return format!("{}{target}{fragment}", up(file));
748        }
749        let missing = subject.canonical(link);
750        left_out.insert(missing.clone());
751        return format!("{}{LIVE_PAGE}#{}", up(file), missing);
752    }
753    let bare = link.split('#').next().unwrap_or(link);
754    // a page of another snapshot: the page itself when that snapshot holds it, and what
755    // says only a process answers it when it does not
756    if let Some(held) = outside.snapshot(bare) {
757        return match held {
758            Some(href) => format!("{href}{fragment}"),
759            None => format!("{}{LIVE_PAGE}#{bare}", up(file)),
760        };
761    }
762    let path = link.split(['?', '#']).next().unwrap_or(link);
763    match outside.resolve(path) {
764        Some(resolved) => format!("{resolved}{}", &link[path.len()..]),
765        None => format!(
766            "{}{LIVE_PAGE}#{}",
767            up(file),
768            link.split('#').next().unwrap_or(link)
769        ),
770    }
771}

773/// One page of the surface as a file of the snapshot.
774#[allow(clippy::too_many_arguments)]
775fn rewrite(
776    subject: &Subject,
777    html: &str,
778    file: &str,
779    written: &BTreeMap<String, String>,
780    held: &BTreeSet<String>,
781    outside: &Outside,
782    origin: &Origin,
783    left_out: &mut BTreeSet<String>,
784) -> String {
785    let attributes = attribute_pattern();
786    let page = attributes.replace_all(html, |c: &regex::Captures<'_>| {
787        let name = &c[1];
788        let link = unescape(&c[2]);
789        let target = if name == "action" {
790            // a form asks the process: in a snapshot it leads to what says so
791            let path = link.split(['?', '#']).next().unwrap_or(&link).to_string();
792            format!("{}{LIVE_PAGE}#{path}", up(file))
793        } else {
794            relink(subject, &link, file, written, held, outside, left_out)
795        };
796        format!(r#"{name}="{}""#, escape(&target))
797    });
798    let graphs = graph_href_pattern();
799    let page = graphs.replace_all(&page, |c: &regex::Captures<'_>| {
800        let target = relink(
801            subject,
802            &unescape(&c[1]),
803            file,
804            written,
805            held,
806            outside,
807            left_out,
808        );
809        format!("&quot;href&quot;:&quot;{}&quot;", escape(&target))
810    });
811    let page = page.replacen("<body ", &format!("<body {SNAPSHOT_ATTRIBUTE} "), 1);
812    // a surface whose scripts talk to the process carries none of them in a snapshot: a
813    // script that opened a socket or posted a form would ask a host that answers neither
814    let page = if subject.scripts {
815        page
816    } else {
817        script_pattern().replace_all(&page, "").into_owned()
818    };
819    let banner = banner(origin, &format!("{}{LIVE_PAGE}", up(file))).render();
820    match page.find("<main") {
821        Some(start) => match page[start..].find('>') {
822            Some(end) => {
823                let at = start + end + 1;
824                format!("{}{banner}{}", &page[..at], &page[at..])
825            }
826            None => page,
827        },
828        None => page,
829    }
830}

832/// A script a page loads from a file: `<script … src="…"></script>`, with nothing inside.
833fn script_pattern() -> regex::Regex {
834    regex::Regex::new(r#"<script\b[^>]*\bsrc="[^"]*"[^>]*>\s*</script>"#)
835        .expect("the script pattern compiles")
836}

857/// The one page about the live surface: what a link to a running process meant, and how to
858/// get one. Drawn in the surface's own frame, taken from the first page the snapshot read.
859fn live_page(subject: &Subject, frame: Option<&str>, origin: &Origin) -> String {
860    let body = el("div")
861        .class("mj-ui-stack")
862        .child(
863            el("h1")
864                .class("mj-ui-title")
865                .text(format!("The live {}", subject.name)),
866        )
867        .child(el("p").text(
868            "This link leads to something only a running process answers: a page read again as the repository changes, a query, a form, a page past the snapshot's bound, another surface of the server or its API. Run the server in a checkout and open the same path there:",
869        ))
870        .child(crate::cockpit::view::pre(format!(
871            "majordomus serve\n# then open http://localhost:<port>{}",
872            subject.prefix
873        )))
874        .child(
875            el("p")
876                .text("The path this link named: ")
877                .child(el("code").attr("data-snapshot-path", "").text("(the part of this page's address after #)")),
878        )
879        .child(el("p").text(format!(
880            "The snapshot was read at commit {} on {}.",
881            origin.revision.as_deref().unwrap_or("an unknown commit"),
882            origin.rendered_at
883        )));
884    let bare = || crate::cockpit::html::document(subject.live_title, Vec::new(), body.clone());
885    // the frame of the dashboard as the snapshot wrote it, at the root: its <main> is replaced
886    let Some(html) = frame else { return bare() };
887    let (Some(open), Some(close)) = (html.find("<main"), html.rfind("</main>")) else {
888        return bare();
889    };
890    let Some(open_end) = html[open..].find('>').map(|e| open + e + 1) else {
891        return bare();
892    };
893    // the dashboard is at the root and this page one directory down: every relative link of
894    // the frame gains one step up, and nothing in the frame is the current screen here
895    let relocate = |part: &str| {
896        attribute_pattern()
897            .replace_all(part, |c: &regex::Captures<'_>| {
898                let v = &c[2];
899                if v.is_empty() || v.starts_with('/') || v.starts_with('#') || v.contains("://") {
900                    c[0].to_string()
901                } else {
902                    format!(r#"{}="../{v}""#, &c[1])
903                }
904            })
905            .replace(r#" aria-current="page""#, "")
906    };
907    let head = relocate(&html[..open_end]);
908    let head = match (head.find("<title>"), head.find("</title>")) {
909        (Some(a), Some(b)) if a < b => {
910            format!("{}<title>{}{}", &head[..a], subject.live_title, &head[b..])
911        }
912        _ => head,
913    };
914    format!("{head}{}{}", body.render(), relocate(&html[close..]))
915}

1145    fn holds_what_it_links(subject: &Subject, snapshot: &Snapshot) {
1146        let dir = &snapshot.dir;
1147        for file in snapshot
1148            .pages
1149            .values()
1150            .chain(std::iter::once(&LIVE_PAGE.to_string()))
1151        {
1152            let html = std::fs::read_to_string(dir.join(file)).unwrap();
1153            for link in links(&html) {
1154                let link = unescape(&link);
1155                assert!(
1156                    link != subject.prefix && !link.starts_with(&format!("{}/", subject.prefix)),
1157                    "{file} keeps an absolute link into its own surface: {link}"
1158                );
1159                if link.is_empty()
1160                    || link.starts_with('/')
1161                    || link.starts_with('#')
1162                    || link.contains("://")
1163                {
1164                    continue;
1165                }
1166                let path = link.split(['#', '?']).next().unwrap();
1167                let base = Path::new(file).parent().unwrap_or(Path::new(""));
1168                assert!(
1169                    dir.join(base).join(path).exists(),
1170                    "{file} links {link}, which is not in the snapshot"
1171                );
1172            }
1173        }
1174    }

1253    /// Two snapshots under one root link into each other by the pages each holds: the one
1254    /// written first is corrected by the one written after it, a page the other does not
1255    /// hold still leads to the live page, and the second one's seeds are what the first
1256    /// wanted to link to.
1257    #[test]
1258    fn two_snapshots_link_only_the_pages_the_other_holds() {
1259        let repo = crate::synthetic::SyntheticRepository::new(crate::synthetic::Shape::default())
1260            .expect("a synthetic repository");
1261        let ctx = repo.context().expect("a context");
1262        let share = Path::new(env!("CARGO_MANIFEST_DIR")).join("../../share");
1263        let ui = crate::ui::Ui::new(std::sync::Arc::clone(&ctx), "0.0.0-test", Some(&share));
1264        let cockpit =
1265            crate::cockpit::Cockpit::new(std::sync::Arc::clone(&ctx), "0.0.0-test", Some(&share));
1266        let root = tempfile::tempdir().unwrap();
1267        let topology =
1268            crate::web::discover::discover(root.path(), crate::web::discover::Runtime::full())
1269                .expect("a topology");
1270        let screens: Vec<String> = crate::ui::SCREENS
1271            .iter()
1272            .map(|s| s.route.to_string())
1273            .collect();
1274        let first = render(
1275            root.path(),
1276            &UI,
1277            &|request| ui.handle(request),
1278            screens,
1279            &topology,
1280            40,
1281            0,
1282        )
1283        .expect("the UI's snapshot");
1284        let cockpit_links =
1285            |dir: &Path, pages: &BTreeMap<String, String>| -> (Vec<String>, usize) {
1286                let mut held = Vec::new();
1287                let mut live = 0;
1288                for file in pages.values() {
1289                    let html = std::fs::read_to_string(dir.join(file)).unwrap();
1290                    for link in links(&html) {
1291                        let link = unescape(&link);
1292                        if link.starts_with("/cockpit") {
1293                            held.push(link);
1294                        } else if link.contains(&format!("{LIVE_PAGE}#/cockpit")) {
1295                            live += 1;
1296                        }
1297                    }
1298                }
1299                (held, live)
1300            };
1301        let (held, live) = cockpit_links(&first.dir, &first.pages);
1302        assert!(held.is_empty(), "no Cockpit snapshot exists yet: {held:?}");
1303        assert!(live > 0, "the UI's source lines lead to the Cockpit");
1304
1305        // the Cockpit's seeds are what the UI wanted to link to
1306        let seeds = cockpit_seeds(root.path());
1307        assert!(
1308            seeds
1309                .iter()
1310                .any(|s| s.starts_with("/cockpit/capabilities/")),
1311            "the capabilities the UI cites are read first: {seeds:?}"
1312        );
1313        let second = render(
1314            root.path(),
1315            &COCKPIT,
1316            &|request| cockpit.handle(request),
1317            seeds,
1318            &topology,
1319            400,
1320            600,
1321        )
1322        .expect("the Cockpit's snapshot");
1323
1324        // the UI's files were corrected: a link into the Cockpit names a page it holds
1325        let (held, _) = cockpit_links(&first.dir, &first.pages);
1326        assert!(
1327            !held.is_empty(),
1328            "the UI's links were led into the Cockpit's snapshot"
1329        );
1330        for link in &held {
1331            let path = link.split(['#', '?']).next().unwrap();
1332            let rest = path
1333                .strip_prefix("/cockpit")
1334                .unwrap()
1335                .trim_start_matches('/');
1336            assert!(
1337                second.dir.join(rest).join("index.html").is_file(),
1338                "the UI links {link}, which the Cockpit's snapshot does not hold"
1339            );
1340        }
1341        // and a page of another snapshot it does not hold is the live page, never a guess
1342        let outside = Outside::from(root.path(), &UI, &topology);
1343        assert_eq!(
1344            outside.snapshot("/cockpit/capabilities"),
1345            Some(Some("/cockpit/capabilities/".to_string()))
1346        );
1347        assert_eq!(outside.snapshot("/cockpit/no/such/page"), Some(None));
1348        assert_eq!(
1349            outside.snapshot("/rustdoc/x"),
1350            None,
1351            "not a snapshot's surface"
1352        );
1353    }
