1//! Governance: the rules that bind this repository and how each is proven. Read from
2//! `rules.report`: its verdict, its coverage, the state of each rule's proof and its
3//! findings.

15/// The route of the screen.
16pub const ROUTE: &str = "/ui/governance";
17
18const RULES: Source = Source::of("rules.report");
19
20/// The Governance screen; `query` may pick one rule to inspect (`?rule=<id>`).
21pub fn screen(ctx: &Context, query: &[(String, String)]) -> View {
22    let answers = ask_all(ctx, &[RULES]);
23    let rules = &answers[&RULES];
24    let inspected = super::rule::picked(query)
25        .map(|id| super::rule::inspector(&super::rule::ask(ctx, &id), &id));
26
27    let top = hero(
28        "shield-check",
29        "rose",
30        "Governance",
31        "Rules that are proven, not only declared.",
32        "Every rule of the repository, what proves it — a validator, a test, a gate — and whether that proof ran and passed.",
33        vec![
34            kpi_state("Verdict", &rules.read("/verdict")),
35            kpi("Rules", &rules.read("/coverage/rules")),
36            kpi("Blocking", &rules.read("/coverage/blocking")),
37            kpi("Gated", &rules.read("/coverage/gated")),
38            kpi_count("Findings", &rules.read("/findings")),
39        ],
40    );
41
42    let table_body = unanswered(rules).unwrap_or_else(|| {
43        let rows = list(rules, "/rules")
44            .iter()
45            .map(|r| {
46                // the report gives the rule as an object: its id and title, a link to it
47                let rule = r.get("rule").cloned().unwrap_or(serde_json::Value::Null);
48                let id = super::rule::id_of(&rule);
49                el("tr")
50                    .child(
51                        el("td").child(
52                            el("a")
53                                .class("mj-ui-row-link")
54                                .attr("href", super::rule::href(&id))
55                                .child(el("span").class("mj-ui-mono mj-ui-dim").text(id.clone()))
56                                .child(
57                                    el("span")
58                                        .class("mj-ui-cell-title")
59                                        .text(text(&rule, "title")),
60                                ),
61                        ),
62                    )
63                    .child(el("td").child(state(r, "state")))
64                    .child(el("td").class("mj-ui-dim").text(text(r, "meaning")))
65            })
66            .collect::<Vec<_>>();
67        table(
68            &["Rule", "Proof", "What that means"],
69            rows,
70            "No rule is declared.",
71        )
72    });
73
74    let mut grid = el("div").class("mj-ui-grid");
75    if let Some(inspector) = inspected {
76        grid = grid.child(wide(inspector));
77    }
78    let grid = grid
79        .child(panel(
80            "Proof states",
81            "How far each rule's proof reaches, counted by the report.",
82            unanswered(rules).unwrap_or_else(|| donut("rules", &counts(rules, "/states"))),
83            &[rules],
84        ))
85        .child(panel(
86            "Coverage",
87            "The rules by class and by what proves them; the classes overlap, so they are listed, not summed.",
88            unanswered(rules).unwrap_or_else(|| {
89                el("dl").class("mj-ui-facts").children(
90                    counts(rules, "/coverage")
91                        .into_iter()
92                        .map(|(name, n)| {
93                            el("div")
94                                .class("mj-ui-fact")
95                                .child(el("dt").text(name.replace('_', " ")))
96                                .child(el("dd").text(n.to_string()))
97                        }),
98                )
99            }),
100            &[rules],
101        ))
102        .child(wide(panel(
103            "Rules",
104            "Every rule with the state of its proof.",
105            table_body,
106            &[rules],
107        )));
108
109    View::new(
110        "Governance",
111        el("div").class("mj-ui-stack").child(top).child(grid),
112    )
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118    use crate::synthetic::SyntheticRepository;
119
120    fn q(pairs: &[(&str, &str)]) -> Vec<(String, String)> {
121        pairs
122            .iter()
123            .map(|(k, v)| (k.to_string(), v.to_string()))
124            .collect()
125    }
126
127    /// The screen reads the report it names: its verdict and coverage, the proof states and
128    /// every rule as a row that leads to the rule's own inspector.
129    #[test]
130    fn every_rule_is_a_row_that_leads_to_its_inspector() {
131        let repo = SyntheticRepository::small().expect("a synthetic repository");
132        let ctx = repo.context().expect("a context");
133        let view = screen(&ctx, &[]);
134        assert_eq!(view.title, "Governance");
135        let html = view.main.render();
136        for want in ["Proof states", "Coverage", "Rules", "rules.report"] {
137            assert!(html.contains(want), "no {want}: {html}");
138        }
139        let report = ctx
140            .execute("rules.report", serde_json::json!({}))
141            .expect("the report");
142        let ids: Vec<String> = report["rules"]
143            .as_array()
144            .map(|a| {
145                a.iter()
146                    .map(|r| {
147                        super::super::rule::id_of(r.get("rule").unwrap_or(&serde_json::Value::Null))
148                    })
149                    .filter(|id| !id.is_empty())
150                    .collect()
151            })
152            .unwrap_or_default();
153        if ids.is_empty() {
154            assert!(html.contains("No rule is declared."), "{html}");
155        }
156        for id in &ids {
157            let href = super::super::rule::href(id);
158            assert!(
159                html.contains(&format!("href=\"{href}\"")),
160                "{id} leads nowhere: {html}"
161            );
162        }
163        assert!(
164            !html.contains("id=\"rule\""),
165            "no rule is inspected unless asked"
166        );
167    }
168
169    /// `?rule=` opens the inspector above the table; a value that is no rule id opens none and
170    /// is never asked.
171    #[test]
172    fn a_picked_rule_opens_its_inspector_and_a_bad_one_none() {
173        let repo = SyntheticRepository::small().expect("a synthetic repository");
174        let ctx = repo.context().expect("a context");
175        let unknown = screen(&ctx, &q(&[("rule", "no.such-rule")])).main.render();
176        assert!(
177            unknown.contains("rules.show"),
178            "the inspector asked: {unknown}"
179        );
180        assert!(!unknown.contains("id=\"rule\""), "{unknown}");
181        let hostile = screen(&ctx, &q(&[("rule", "x\" onclick=\"y")]))
182            .main
183            .render();
184        assert!(
185            !hostile.contains("rules.show"),
186            "a hostile id is never asked: {hostile}"
187        );
188        assert!(!hostile.contains("onclick=\"y"), "{hostile}");
189    }
