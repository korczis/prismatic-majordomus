1//! The session domain: one typed account of what an episode is, what identifies it, how it
2//! moves, and how it is closed exactly once.
3//!
4//! # Why this module exists
5//!
6//! This repository runs two programs over one subsystem. The shell tool (`lib/session.sh`,
7//! `lib/capture.sh`, `lib/checkpoint.sh`, `lib/handover.sh`) owns the writes, the lifecycle
8//! and the state machine; this executable owns the typed read model, the server, the HTTP
9//! API, MCP and the Cockpit. [`crate::capability::builtin::continuity`] opens by declaring
10//! the split — *"It is read, never written. The lifecycle is the shell tool's."*
11//!
12//! ADR 0052 found what the split costs. For six days this repository wrote no checkpoint
13//! and no handover while `doctor` and `watch` reported health, because the invariant that
14//! would have caught it — *lifecycle events are arriving, therefore derived state must
15//! advance* — spans both halves and is owned by neither. The reader faithfully reported the
16//! newest record it could find; the writer faithfully refused to write one. ADR 0052 names
17//! the target: **one canonical session/continuity domain service in this crate**, with
18//! `.envrc`, the provider hook shims, the CLI, MCP, HTTP and the Cockpit as adapters over
19//! it. This module is that service's domain.
20//!
21//! # What this change is, and what it is not
22//!
23//! It is **additive**. The shell keeps every write it has. Nothing here changes the
24//! behaviour of the running lifecycle: the projections are read-only queries, and the write
25//! path ([`SessionStore::close`]) exists, is tested, and is reachable only through
26//! [`SessionStore::writable`] — a constructor no surface calls. ADR 0047 records the cutover
27//! and its order. Four workers are editing `lib/*.sh` as this is written, and a cutover in
28//! the same change would be a merge whose conflicts are semantic rather than textual, in the
29//! one subsystem where a wrong merge is invisible until somebody resumes from a record that
30//! is not theirs.
31//!
32//! # The four things it settles
33//!
34//! **[Identity](identity).** Six things the stores spell as three words become six types
35//! with no conversion between them. [`RepositoryId`] and [`CheckoutId`] wrap the digests
36//! this crate already computes for its lease, index and worktree topology, rather than
37//! inventing a seventh notion of "which repository".
38//!
39//! **[The aggregates](episode).** An [`Episode`] carries `Option<TaskId>` and nothing else
40//! of a task. ADR 0052's defect — a task's state deciding whether an episode's artefact is
41//! written — is unrepresentable, because [`EpisodeState::may_move_to`] takes no task.
42//!
43//! **[The machine](state).** Four states, six transitions, one terminal state, declared
44//! once and projected as data by `session.machine` so that no surface draws its own.
45//!
46//! **[Persistence](store).** Closing takes an `O_EXCL` claim before the expensive part, so
47//! four repeated closes and eight simultaneous ones yield exactly one record. The episode
48//! that has four records in this repository is the reason.
49//!
50//! **[Events](ledger).** Typed, over the ledger that already exists, against the vocabulary
51//! `share/events.yaml` already declares. No second store: locked append-only JSONL meets
52//! concurrency, corruption detection and recovery, and ADR 0047 records why a database was
53//! refused rather than merely not chosen.
54//!
55//! # Freshness is consumed, not restated
56//!
57//! `fresh | aging | stale | unknown | invalid`, `Thresholds::judge` and `epoch_seconds`
58//! arrive with ADR 0052 in `continuity.rs`. This domain does not define a second vocabulary
59//! for age: it takes Unix seconds and answers the single predicate the machine needs,
60//! [`Episode::stranded_after`].
61//!
62//! ```
63//! use majordomus_cli::session::{EpisodeState, Machine, open_episodes};
64//!
65//! // the machine is derived from the types and has one terminal state
66//! let machine = Machine::describe();
67//! assert_eq!(machine.states.len(), EpisodeState::ALL.len());
68//! assert_eq!(machine.states.iter().filter(|s| s.terminal).count(), 1);
69//!
70//! // and a checkout that has never run the lifecycle reports absence, not a fault
71//! let dir = tempfile::tempdir().unwrap();
72//! assert!(open_episodes(dir.path()).is_empty());
73//! ```

110/// The open episodes of one checkout, as the store holds them.
111///
112/// Read from `.ai/local/state/sessions-open/`, which is a file per provider session rather
113/// than a singleton: an episode used to be one `session-current.yaml` per checkout, and a
114/// second provider session opening one was folded into the first. Measured in this
115/// repository on 2026-09-09: seven concurrent sessions, one record between them.
116///
117/// `last_sign_of_life` is the later of the episode's `started_at` and the newest ledger
118/// line it stamped, which is what makes "nobody has been here for a day" a measurement
119/// rather than an inference from the open record's own age.
120///
121/// ```
122/// use majordomus_cli::session::open_episodes;
123///
124/// let dir = tempfile::tempdir().unwrap();
125/// let open = dir.path().join(".ai/local/state/sessions-open");
126/// std::fs::create_dir_all(&open).unwrap();
127/// std::fs::write(
128///     open.join("01Bv2gJsf.yaml"),
129///     "session_id: s-20260910205542-e2a6\nstarted_at: 2026-09-10T20:55:42Z\nowner: \"korczis\"\n\
130///      provider: \"claude-code\"\nprovider_session: \"01Bv2gJsf\"\nbranch: master\nstart_head: 9f3a2a0\n",
131/// ).unwrap();
132///
133/// let episodes = open_episodes(dir.path());
134/// assert_eq!(episodes.len(), 1);
135/// assert_eq!(episodes[0].id.as_str(), "s-20260910205542-e2a6");
136/// assert!(episodes[0].task.is_none(), "an episode without a task is ordinary");
137/// assert_eq!(episodes[0].provider_session.as_ref().map(|p| p.as_str()), Some("01Bv2gJsf"));
138/// assert_eq!(episodes[0].last_sign_of_life, Some(1_789_073_742));
139/// ```
140pub fn open_episodes(root: &Path) -> Vec<Episode> {
141    let dir = root.join(store::OPEN_DIR);
142    let Ok(entries) = std::fs::read_dir(&dir) else {
143        return Vec::new();
144    };
145    let stamps = ledger_stamps(root);
146    let mut out: Vec<Episode> = Vec::new();
147    for entry in entries.filter_map(|e| e.ok()) {
148        let path = entry.path();
149        if path.extension().is_none_or(|x| x != "yaml") {
150            continue;
151        }
152        let Ok(text) = std::fs::read_to_string(&path) else {
153            continue;
154        };
155        let Ok(map) = yaml::parse_mapping(&text) else {
156            continue;
157        };
158        let s = |k: &str| map.get(k).and_then(yaml::scalar_string).unwrap_or_default();
159        let Some(id) = EpisodeId::parse(&s("session_id")) else {
160            continue;
161        };
162        let started_at = s("started_at");
163        let provider_session = {
164            let raw = s("provider_session");
165            (!raw.is_empty()).then(|| ProviderSessionId::new(raw))
166        };
167        let opened = crate::peers::epoch_seconds(&started_at);
168        let newest = stamps
169            .iter()
170            .filter(|(episode, _)| episode == id.as_str())
171            .filter_map(|(_, at)| *at)
172            .max();
173        let last = match (opened, newest) {
174            (Some(a), Some(b)) => Some(a.max(b)),
175            (a, b) => a.or(b),
176        };
177        let mut episode = Episode {
178            id,
179            state: EpisodeState::Open,
180            repository: RepositoryId::of(root),
181            checkout: CheckoutId::of(root),
182            provider: s("provider"),
183            provider_session,
184            worker: WorkerId::supplied(&s("worker")),
185            task: None,
186            started_at,
187            branch: s("branch"),
188            start_head: s("start_head"),
189            last_sign_of_life: last,
190        };
191        // The task, when the checkout has one open. A relation the episode carries; its
192        // absence is ordinary and is never a reason to skip anything (ADR 0052).
193        if let Some(task) = active_task(root) {
194            episode.task = Some(task);
195        }
196        out.push(episode);
197    }
198    crate::order::canonical(&mut out);
199    out
200}

202/// The task this checkout has open, when it has one. Read from `state/current.yaml`, the
203/// active task record — which `lib/start.sh` writes and check, checkpoint, handover and
204/// finish update.
205///
206/// It reads the id and nothing else. An episode relates to a task; it does not hold one,
207/// and a copy of the task's outcome on this side would be the second account that ADR 0052
208/// spent six days of silence proving is a defect.
209fn active_task(root: &Path) -> Option<TaskId> {
210    let text = std::fs::read_to_string(root.join(".ai/local/state/current.yaml")).ok()?;
211    let map = yaml::parse_mapping(&text).ok()?;
212    TaskId::parse(&map.get("id").and_then(yaml::scalar_string)?)
213}

215/// The URL of `origin`, which is what `mj_repository_id` writes into a shared record.
216///
217/// A subprocess, and deliberately not a read of `.git/config`: a linked worktree's config
218/// is the common one and an `includeIf` can put the remote somewhere else, so asking git is
219/// the only answer that is right in every checkout this tool runs in. One call per reading
220/// of a capability whose cache is two seconds wide.
221fn remote_url(root: &Path) -> Option<String> {
222    let out = crate::git::read_only(root)
223        .args(["config", "--get", "remote.origin.url"])
224        .output()
225        .ok()?;
226    if !out.status.success() {
227        return None;
228    }
229    let url = String::from_utf8_lossy(&out.stdout).trim().to_string();
230    (!url.is_empty()).then_some(url)
231}

233/// `(episode id, its timestamp in Unix seconds)` for every ledger line that carries one.
234///
235/// Selection is by the session id the writer stamped on each line, never by a time range.
236/// The ledger is one file per repository; a window of "everything after this episode
237/// opened" collects another worker's events, because two sessions write to one repository
238/// at once and nothing in a timestamp can tell them apart.
239fn ledger_stamps(root: &Path) -> Vec<(String, Option<i64>)> {
240    Ledger::of(root)
241        .read()
242        .events
243        .iter()
244        .filter_map(|line| {
245            let obj = line.as_object()?;
246            let session = obj.get("session")?.as_str()?.to_string();
247            let ts = obj
248                .get("ts")
249                .and_then(|v| v.as_str())
250                .and_then(crate::peers::epoch_seconds);
251            Some((session, ts))
252        })
253        .collect()
254}

256/// How this checkout spells each of the identities the audit found conflated, beside the
257/// canonical value this executable computes.
258///
259/// The point is not to choose a winner. It is to stop pretending the spellings are one
260/// value: a reader that knows a string is the *local* spelling can compare it with other
261/// local spellings and refuse to compare it with a shared one, which is exactly the
262/// comparison that made two records of one repository, written four minutes apart, look
263/// like records of two repositories.
264///
265/// ```
266/// use majordomus_cli::session::identities;
267///
268/// let dir = tempfile::tempdir().unwrap();
269/// let facets = identities(dir.path());
270/// let subjects: Vec<&str> = facets.iter().map(|f| f.subject.as_str()).collect();
271/// assert_eq!(subjects, ["repository", "checkout", "episode", "provider_session"]);
272///
273/// // a plain directory belongs to no git repository, and that is reported as absence
274/// assert!(facets[0].canonical.is_empty());
275/// // a checkout always has an identity: the question has to be answerable for a record
276/// // written on a disk that has since been unmounted
277/// assert_eq!(facets[1].canonical.len(), 32);
278/// ```
279pub fn identities(root: &Path) -> Vec<IdentityFacet> {
280    let git = crate::repository::git_identity(root);
281    let mut repository_spellings = vec![
282        RecordedSpelling {
283            spelling: Spelling::Local,
284            writer: "mj_git_repo_id (lib/common.sh)".to_string(),
285            value: git
286                .as_ref()
287                .map(|g| g.common_dir.display().to_string())
288                .unwrap_or_default(),
289        },
290        RecordedSpelling {
291            spelling: Spelling::Shared,
292            writer: "mj_repository_id (lib/common.sh)".to_string(),
293            value: remote_url(root).unwrap_or_default(),
294        },
295    ];
296    crate::order::canonical(&mut repository_spellings);
297
298    let mut checkout_spellings = vec![
299        RecordedSpelling {
300            spelling: Spelling::Local,
301            writer: "mj_record_front_matter `worktree:` (lib/common.sh)".to_string(),
302            value: root.display().to_string(),
303        },
304        RecordedSpelling {
305            spelling: Spelling::Shared,
306            writer: "mj_worktree_id `worktree_id:` (lib/common.sh)".to_string(),
307            value: String::new(),
308        },
309    ];
310    crate::order::canonical(&mut checkout_spellings);
311
312    let open = open_episodes(root);
313    let episode_spellings = open
314        .first()
315        .map(|e| {
316            vec![RecordedSpelling {
317                spelling: Spelling::Local,
318                writer: "session_id (state/sessions-open/<key>.yaml)".to_string(),
319                value: e.id.as_str().to_string(),
320            }]
321        })
322        .unwrap_or_default();
323    let provider_spellings = open
324        .first()
325        .and_then(|e| e.provider_session.clone())
326        .map(|p| {
327            vec![RecordedSpelling {
328                spelling: Spelling::Local,
329                writer: "provider_session (state/sessions-open/<key>.yaml)".to_string(),
330                value: p.as_str().to_string(),
331            }]
332        })
333        .unwrap_or_default();
334
335    vec![
336        IdentityFacet {
337            subject: "repository".to_string(),
338            canonical: git.map(|g| g.id).unwrap_or_default(),
339            spellings: repository_spellings,
340            note: "one value for every checkout of one repository. The local half writes a \
341                   path and the shared half writes a remote URL; neither can equal the other, \
342                   so a reader that compares them concludes two records of one repository are \
343                   about two repositories."
344                .to_string(),
345        },
346        IdentityFacet {
347            subject: "checkout".to_string(),
348            canonical: CheckoutId::of(root).as_str().to_string(),
349            spellings: checkout_spellings,
350            note: "one worktree. Two checkouts of one repository share a repository identity \
351                   and differ here, which is the first tier of every record resolution in the \
352                   tool: a record from another worktree is never silently offered as your \
353                   context."
354                .to_string(),
355        },
356        IdentityFacet {
357            subject: "episode".to_string(),
358            canonical: open
359                .first()
360                .map(|e| e.id.as_str().to_string())
361                .unwrap_or_default(),
362            spellings: episode_spellings,
363            note: "the canonical session identity and the only one. Minted at open, stamped \
364                   on every ledger line of the episode, and what `session_id:` names in the \
365                   closed record."
366                .to_string(),
367        },
368        IdentityFacet {
369            subject: "provider_session".to_string(),
370            canonical: String::new(),
371            spellings: provider_spellings,
372            note: "an external correlation id, never the session identity. It is how a \
373                   provider's hook finds the episode it opened. An episode opened by hand has \
374                   none; two providers may name their sessions alike; and a worker that \
375                   reconnects keeps its work while the value changes under it."
376                .to_string(),
377        },
378    ]
379}

385    #[test]
386    fn two_open_episodes_in_one_checkout_are_two_episodes() {
387        // The singleton defect, as a test: before `state/sessions-open/` existed, a second
388        // provider session opening an episode in one checkout was folded into the first,
389        // and either window's end event closed the episode for both.
390        let dir = tempfile::tempdir().expect("a temp dir");
391        let open = dir.path().join(store::OPEN_DIR);
392        std::fs::create_dir_all(&open).expect("the store");
393        for (key, id) in [
394            ("first", "s-20260910205542-e2a6"),
395            ("second", "s-20260910210510-8f20"),
396        ] {
397            std::fs::write(
398                open.join(format!("{key}.yaml")),
399                format!(
400                    "session_id: {id}\nstarted_at: 2026-09-10T20:55:42Z\nprovider_session: \"{key}\"\n"
401                ),
402            )
403            .expect("an open record");
404        }
405        let episodes = open_episodes(dir.path());
406        assert_eq!(episodes.len(), 2);
407        let ids: std::collections::BTreeSet<&str> =
408            episodes.iter().map(|e| e.id.as_str()).collect();
409        assert_eq!(ids.len(), 2, "two identities, not one");
410    }

412    #[test]
413    fn the_newest_ledger_line_of_an_episode_is_its_sign_of_life() {
414        let dir = tempfile::tempdir().expect("a temp dir");
415        let open = dir.path().join(store::OPEN_DIR);
416        std::fs::create_dir_all(&open).expect("the store");
417        std::fs::write(
418            open.join("k.yaml"),
419            "session_id: s-20260910205542-e2a6\nstarted_at: 2026-09-10T20:55:42Z\n",
420        )
421        .expect("an open record");
422        let ledger = dir.path().join(ledger::LEDGER_PATH);
423        std::fs::create_dir_all(ledger.parent().expect("a parent")).expect("the state dir");
424        std::fs::write(
425            &ledger,
426            "{\"ts\":\"2026-09-10T22:00:00Z\",\"event\":\"task.started\",\"session\":\"s-20260910205542-e2a6\"}\n\
427             {\"ts\":\"2026-09-11T09:00:00Z\",\"event\":\"task.started\",\"session\":\"s-20260910210510-8f20\"}\n",
428        )
429        .expect("a ledger");
430
431        let episodes = open_episodes(dir.path());
432        assert_eq!(episodes.len(), 1);
433        let opened = crate::peers::epoch_seconds("2026-09-10T20:55:42Z").expect("an instant");
434        let stamped = crate::peers::epoch_seconds("2026-09-10T22:00:00Z").expect("an instant");
435        assert_eq!(
436            episodes[0].last_sign_of_life,
437            Some(stamped),
438            "the later of the two, and never another episode's line"
439        );
440        assert!(stamped > opened);
441    }
