1//! The Cockpit's Trains page (ADR 0138): the open pull requests stacked on each other, as
2//! `integration.trains` answers them.
3//!
4//! The page is complete without JavaScript: every train is a table of its links in landing
5//! order, with every fact the capability carries about each one. `share/cockpit/trains.js`
6//! adds a second view of the same answer — a graph in the manner of the Neo4j Browser, with
7//! master as its anchor, a property panel, filters and search — and polls the same HTTP
8//! route the table was rendered from, so the drawing and the table cannot disagree about
9//! what a train is. Nothing here holds a train of its own and nothing here moves a ref.
10

23/// The capability the page renders, and the route the drawing polls.
24const CAPABILITY: &str = "integration.trains";
25const ROUTE: &str = "/api/v1/pull-requests/trains";
26
27/// How often the drawing asks again, in seconds. The answer is read from a recorded
28/// observation, so asking more often than the executor observes shows nothing new, and each
29/// answer decides every link's relation to master again, which costs seconds on a busy
30/// machine: once a minute, and never while the last ask is still out.
31const POLL_SECONDS: u32 = 60;
32
33/// `/cockpit/trains`.
34pub fn page(ctx: &Context) -> Page {
35    match ask(ctx, CAPABILITY, json!({})) {
36        Ok(view) => of_view(view),
37        Err(e) => failed(Area::Trains, "Trains", e),
38    }
39}

41/// The page of one answer of the capability: kept apart from the asking so that every shape
42/// of the answer can be drawn without a forge observation behind it.
43fn of_view(view: TrainsView) -> Page {
44    let trail = vec![
45        ("Cockpit", Some("/cockpit")),
46        ("Integration", Some("/cockpit/integration")),
47        ("Trains", None),
48    ];
49    let subtitle = "Open pull requests stacked on each other, in the order they land: each link's disposition, relation to master, prepared head and turn.";
50    let commands = card(
51        "Commands",
52        el("div")
53            .child(el("p").class("mj-prose").text(
54                "The page reads; nothing here pushes, fetches or moves a ref. The prepared heads under refs/prep/ are the executor's, pushed at each link's turn.",
55            ))
56            .child(pre(format!(
57                "majordomus prs refresh                  # observe the forge now; the one network read\n\
58                 majordomus prs trains                   # these trains, in landing order\n\
59                 majordomus prs trains --format json     # the same answer this page draws\n\
60                 majordomus prs explain <n>              # why one link is where it is\n\
61                 GET {ROUTE}"
62            ))),
63    );
64
65    if !view.observed {
66        return Page::new(
67            Area::Trains,
68            "Trains",
69            el("div")
70                .class("mj-grid")
71                .child(card(
72                    "No trains",
73                    el("div")
74                        .child(alert(
75                            "info",
76                            view.reason
77                                .clone()
78                                .unwrap_or_else(|| "nothing is observed".into()),
79                        ))
80                        .child(nothing(
81                            "Trains are read from a recorded forge observation, and this checkout has none. `majordomus prs refresh` records one.",
82                        )),
83                ))
84                .child(commands),
85        )
86        .subtitle(subtitle)
87        .trail(trail);
88    }
89
90    let source = format!("{CAPABILITY} · tallies");
91    let stats = el("div")
92        .class("mj-stats")
93        .child(statistic(
94            view.tallies.trains.to_string(),
95            "trains",
96            source.clone(),
97        ))
98        .child(statistic(
99            view.tallies.links.to_string(),
100            "links",
101            source.clone(),
102        ))
103        .child(statistic(
104            view.tallies.pending_prep.to_string(),
105            "prepared heads pending",
106            source.clone(),
107        ))
108        .child(statistic(
109            view.unattached_prep.len().to_string(),
110            "prepared heads no train holds",
111            format!("{CAPABILITY} · unattached_prep"),
112        ));
113
114    let observed_at = view.observed_at.clone().unwrap_or_default();
115    let identity = card_with(
116        "This answer",
117        link(format!("/cockpit/capabilities/{CAPABILITY}"), CAPABILITY),
118        facts(vec![
119            (
120                "Repository",
121                Node::Element(mono(view.repository.clone().unwrap_or_default())),
122            ),
123            (
124                "Base",
125                Node::Element(
126                    el("span")
127                        .child(mono(view.base.clone().unwrap_or_default()))
128                        .text(" at ")
129                        .child(mono(view.master_sha.clone().unwrap_or_default())),
130                ),
131            ),
132            (
133                "Observed",
134                Node::Element(
135                    el("span")
136                        .attr("data-mj-trains-observed", observed_at.clone())
137                        .text(observed_at),
138                ),
139            ),
140            (
141                "Asked",
142                Node::Element(
143                    el("span")
144                        .attr("data-mj-trains-asked", "")
145                        .text(format!(
146                            "when this page was rendered; the drawing asks {ROUTE} again every {POLL_SECONDS} s"
147                        )),
148                ),
149            ),
150        ]),
151    );
152
153    let mut grid = el("div")
154        .class("mj-grid")
155        .child(stats)
156        .child(identity)
157        .child(viewer(&view));
158    if !view.diagnostics.is_empty() {
159        grid = grid.child(card_with(
160            "Diagnostics",
161            badge("warn", view.diagnostics.len().to_string()),
162            el("div").children(
163                view.diagnostics
164                    .iter()
165                    .map(|d| alert("warn", d.clone()))
166                    .collect::<Vec<_>>(),
167            ),
168        ));
169    }
170    if view.trains.is_empty() {
171        grid = grid.child(card(
172            "Trains",
173            nothing("No open pull request is stacked on another, and none declares a dependency."),
174        ));
175    }
176    for t in &view.trains {
177        grid = grid.child(train_card(t));
178    }
179    if !view.unattached_prep.is_empty() {
180        let rows: Vec<El> = view
181            .unattached_prep
182            .iter()
183            .map(|p| {
184                row(vec![
185                    cell(mono(format!("refs/prep/{}", p.number))),
186                    cell(mono(short(&p.sha))),
187                    text_cell(if p.open {
188                        "open, stacked on nothing"
189                    } else {
190                        "not open"
191                    }),
192                ])
193            })
194            .collect();
195        grid = grid.child(card(
196            "Prepared heads no train holds",
197            table(&["Ref", "Commit", "Pull request"], rows),
198        ));
199    }
200    Page::new(Area::Trains, "Trains", grid.child(commands))
201        .subtitle(subtitle)
202        .trail(trail)
203        .script("trains.js")
204}

206/// The drawing's frame, its controls and its property panel. Empty until the script fills
207/// it; without the script the note under it says where the same facts are.
208fn viewer(view: &TrainsView) -> El {
209    let mut choose = el("select")
210        .class("mj-input")
211        .attr("data-mj-trains-train", "")
212        .attr("aria-label", "Show one train")
213        .child(el("option").attr("value", "").text("every train"));
214    for t in &view.trains {
215        choose = choose.child(el("option").attr("value", t.id.clone()).text(format!(
216            "{} ({} links)",
217            t.id,
218            t.links.len()
219        )));
220    }
221    let button = |data: &str, label: &str, aria: &str| {
222        el("button")
223            .class("mj-button mj-button--quiet")
224            .attr("type", "button")
225            .attr(data, "")
226            .attr("aria-label", aria)
227            .text(label)
228    };
229    let links: usize = view.trains.iter().map(|t| t.links.len()).sum();
230    el("section")
231        .class("mj-card")
232        .child(
233            el("div")
234                .class("mj-card-head")
235                .child(el("h2").class("mj-card-title").text("Graph"))
236                .child(
237                    el("div")
238                        .class("mj-graph-controls")
239                        .child(
240                            el("input")
241                                .class("mj-input")
242                                .attr("type", "search")
243                                .attr("data-mj-trains-search", "")
244                                .attr("aria-label", "Find a pull request by number, title or branch")
245                                .attr("placeholder", "Find #, title, branch"),
246                        )
247                        .child(choose)
248                        .child(button("data-mj-trains-zoom-in", "+", "Zoom in"))
249                        .child(button("data-mj-trains-zoom-out", "−", "Zoom out"))
250                        .child(button("data-mj-trains-fit", "Fit", "Fit the drawing to the frame"))
251                        .child(button(
252                            "data-mj-trains-layout",
253                            "Layered",
254                            "Switch between the force layout and the layered layout",
255                        ))
256                        .child(button("data-mj-trains-refresh", "Ask now", "Ask for the trains now")),
257                ),
258        )
259        .child(
260            // the drawing and its property panel side by side where there is room, the panel
261            // under the drawing where there is not: layout only, no colour and no size step
262            el("div")
263                .attr(
264                    "style",
265                    "display:flex;flex-wrap:wrap;gap:1rem;align-items:flex-start",
266                )
267                .child(frame(view, links))
268                .child(panel()),
269        )
270        .child(
271            el("p")
272                .class("mj-note")
273                .attr("id", "mj-trains-keys")
274                .text("With the drawing focused: the arrow keys move from link to link, Enter opens the selected link's page, + and − zoom, 0 fits, Escape clears the selection. Click a link to select it, drag to move it, scroll to zoom."),
275        )
276}

319/// One train as a table of its links, in landing order.
320fn train_card(t: &Train) -> El {
321    let rows: Vec<El> = t.links.iter().map(link_row).collect();
322    let mut body = el("div").child(table(
323        &[
324            "Turn",
325            "Pull request",
326            "Disposition",
327            "Required checks",
328            "Relation to master",
329            "Prepared head",
330            "Stands on",
331            "Carries",
332            "Branch",
333            "Title",
334        ],
335        rows,
336    ));
337    let edges: Vec<El> = t
338        .edges
339        .iter()
340        .map(|e| {
341            row(vec![
342                text_cell(format!("#{}", e.from)),
343                cell(mono(e.kind.label())),
344                text_cell(format!("#{}", e.to)),
345                text_cell(if e.satisfied {
346                    "satisfied — it landed"
347                } else {
348                    "not yet"
349                }),
350            ])
351        })
352        .collect();
353    body = body.child(details(
354        format!("Relationships ({})", t.edges.len()),
355        table(&["From", "Relationship", "To", "Satisfied"], edges),
356    ));
357    if !t.beneath.is_empty() {
358        let mut line = el("p")
359            .class("mj-prose")
360            .text("Beneath it, no longer open: ");
361        for (i, b) in t.beneath.iter().enumerate() {
362            if i > 0 {
363                line = line.text(", ");
364            }
365            line = match &b.url {
366                Some(u) => line.child(link(u.clone(), format!("#{}", b.number))),
367                None => line.child(mono(format!("#{}", b.number))),
368            };
369            line = line.text(format!(" ({})", word(&b.state)));
370        }
371        body = body.child(line);
372    }
373    let roots = numbers(&t.roots);
374    let tops = numbers(&t.tops);
375    el("section")
376        .class("mj-card")
377        .attr("id", t.id.clone())
378        .child(
379            el("div")
380                .class("mj-card-head")
381                .child(
382                    el("h2")
383                        .class("mj-card-title")
384                        .text(format!("{} — roots {roots}, tops {tops}", t.id)),
385                )
386                .child(match t.next {
387                    Some(n) => badge("ok", format!("next #{n}")),
388                    None => badge("unknown", "no link at its turn"),
389                }),
390        )
391        .child(body)
392}

394/// One link as a row of its train's table.
395fn link_row(l: &TrainLink) -> El {
396    let pr = match &l.url {
397        Some(u) => el("span")
398            .child(link(u.clone(), format!("#{}", l.number)))
399            .when(l.at_turn, |e| e.text(" ").child(tag("at its turn")))
400            .when(l.in_cycle, |e| e.text(" ").child(tag("in a cycle"))),
401        None => el("span").child(mono(format!("#{}", l.number))),
402    };
403    let relation = match l.behind {
404        Some(n) if l.relation_kind == "behind" => format!("behind by {n}"),
405        _ => l.relation_kind.replace('_', " "),
406    };
407    let relation = if l.carries_master {
408        format!("{relation} (carries master)")
409    } else {
410        relation
411    };
412    let prep = match &l.prep {
413        None => el("span").text("none"),
414        Some(p) => el("span").child(mono(short(&p.sha))).text(match p.state {
415            PrepState::Pushed => " pushed",
416            PrepState::Pending => " pending",
417        }),
418    };
419    let checks = match &l.checks_url {
420        Some(u) => el("span")
421            .child(word_badge(&word(&l.required_checks)))
422            .text(" ")
423            .child(link(u.clone(), "checks")),
424        None => el("span").child(word_badge(&word(&l.required_checks))),
425    };
426    row(vec![
427        text_cell(l.position.to_string()),
428        cell(pr),
429        cell(badge(
430            disposition_status(l.disposition),
431            l.disposition.as_str(),
432        )),
433        cell(checks),
434        text_cell(relation),
435        cell(prep),
436        text_cell(if l.on.is_empty() {
437            "master".to_string()
438        } else {
439            numbers(&l.on)
440        }),
441        text_cell(if l.carries.is_empty() {
442            "nothing".to_string()
443        } else {
444            numbers(&l.carries)
445        }),
446        cell(mono(l.head_ref.clone())),
447        text_cell(l.title.clone()),
448    ])
449}

451fn numbers(ns: &[u64]) -> String {
452    ns.iter()
453        .map(|n| format!("#{n}"))
454        .collect::<Vec<_>>()
455        .join(" ")
456}

458fn short(sha: &str) -> String {
459    sha.chars().take(10).collect()
