1//! GitHub, folded into signals: pull requests opened, merged and closed, and finished
2//! workflow runs.
3//!
4//! GitHub keeps the history, so the read is stateless: each pull request and each run says
5//! when it was opened, merged, closed or finished, and a signal is one of those moments
6//! after the cursor's second. Reaching GitHub is the network, so the fetch is bounded in
7//! time, shared by every reader for [`TTL`], and an unreachable or missing `gh` is a source
8//! state with its reason — never a quiet repository.
9//!
10//! A run that succeeded is news only on the trunk (master's checks, a deploy, a release);
11//! a pull request's green run is the expected and is counted as suppressed. A failed run is
12//! always news, and so is a cancelled one: a person or a newer push stopped it, and nobody
13//! learns which by its silence. A failed deploy workflow is asked for its jobs, because a
14//! Pages run fails when its publisher asks for help even after the deploy job went out:
15//! that is `deploy_needs_help`, not `deploy_failed`.
16

31/// How long one fetch is shared by every reader.
32pub(crate) const TTL: Duration = Duration::from_secs(30);
33
34/// How long one `gh` call may take before the source is reported as unreachable.
35pub(crate) const BUDGET: Duration = Duration::from_secs(4);
36
37/// How many pull requests and runs one fetch asks for.
38const WINDOW: usize = 40;
39
40/// The trunk: a run on it is news even when it succeeds.
41const TRUNK: &str = "master";
42
43/// What one fetch brought back: pull requests and runs as `gh` prints them.
44#[derive(Debug, Clone)]
45pub(crate) struct Fetched {
46    pub(crate) prs: Value,
47    pub(crate) runs: Value,
48}
49
50type Fetch = Result<Fetched, (MonitorSourceHealth, String)>;
51
52static CACHE: Mutex<Option<(PathBuf, Instant, Fetch)>> = Mutex::new(None);
53
54/// Serialises the tests that read or write [`CACHE`]: it holds one repository's fetch.
55#[cfg(test)]
56pub(crate) static CACHE_TESTS: Mutex<()> = Mutex::new(());

58/// Put `fetched` in the cache for `root`, as a fetch a moment ago would have: what a test of
59/// a reader of this source hands it instead of the network.
60#[cfg(test)]
61pub(crate) fn seed(root: &Path, fetched: Fetched) {
62    if let Ok(mut cache) = CACHE.lock() {
63        *cache = Some((root.to_path_buf(), Instant::now(), Ok(fetched)));
64    }
65}

67/// The pull requests and runs of the repository at `root`, from the cache when it is
68/// younger than [`TTL`].
69pub(crate) fn fetch(root: &Path) -> Fetch {
70    fetch_with("gh", root)
71}
72
73/// [`fetch`], asking `program` in place of `gh`: the seam a test puts its own program in.
74fn fetch_with(program: &str, root: &Path) -> Fetch {
75    if let Ok(cache) = CACHE.lock() {
76        if let Some((at_root, at, fetched)) = cache.as_ref() {
77            if at_root == root && at.elapsed() < TTL {
78                return fetched.clone();
79            }
80        }
81    }
82    let fetched = fetch_now(program, root);
83    if let Ok(mut cache) = CACHE.lock() {
84        *cache = Some((root.to_path_buf(), Instant::now(), fetched.clone()));
85    }
86    fetched
87}

89fn fetch_now(program: &str, root: &Path) -> Fetch {
90    let prs = run(
91        program,
92        root,
93        &[
94            "pr",
95            "list",
96            "--state",
97            "all",
98            "--limit",
99            &WINDOW.to_string(),
100            "--json",
101            "number,title,state,createdAt,mergedAt,closedAt,headRefName,author,url",
102        ],
103    )?;
104    let runs = run(
105        program,
106        root,
107        &[
108            "run",
109            "list",
110            "--limit",
111            &WINDOW.to_string(),
112            "--json",
113            "databaseId,workflowName,displayTitle,status,conclusion,event,headBranch,updatedAt,url",
114        ],
115    )?;
116    Ok(Fetched {
117        prs,
118        runs: with_deploy_jobs(program, root, runs),
119    })
120}

122/// `runs` with the jobs of every failed deploy run attached under `jobs`, each asked within
123/// [`BUDGET`]; a run whose jobs could not be read keeps its plain verdict.
124fn with_deploy_jobs(program: &str, root: &Path, mut runs: Value) -> Value {
125    for item in runs.as_array_mut().into_iter().flatten() {
126        let failed_deploy = text(item, "status") == "completed"
127            && verdict(&text(item, "conclusion")) == Verdict::Failed
128            && family(&text(item, "workflowName")) == "deploy";
129        let Some(id) = item["databaseId"].as_u64().filter(|_| failed_deploy) else {
130            continue;
131        };
132        if let Ok(view) = run(
133            program,
134            root,
135            &["run", "view", &id.to_string(), "--json", "jobs"],
136        ) {
137            item["jobs"] = view["jobs"].clone();
138        }
139    }
140    runs
141}

143/// One call of `program`, bounded by [`BUDGET`], its stdout parsed as JSON; every failure is
144/// reported as `gh`'s, because `gh` is the only program asked in earnest.
145fn run(program: &str, root: &Path, args: &[&str]) -> Result<Value, (MonitorSourceHealth, String)> {
146    let mut child = crate::process::retrying_busy(|| {
147        Command::new(program)
148            .args(args)
149            .current_dir(root)
150            .stdin(Stdio::null())
151            .stdout(Stdio::piped())
152            .stderr(Stdio::piped())
153            .spawn()
154    })
155    .map_err(|e| {
156        (
157            MonitorSourceHealth::Absent,
158            format!("gh could not be started: {e}"),
159        )
160    })?;
161    let deadline = Instant::now() + BUDGET;
162    let status = loop {
163        match child.try_wait() {
164            Ok(Some(status)) => break status,
165            // not finished, or not able to say: the budget decides either way
166            _ if Instant::now() < deadline => std::thread::sleep(Duration::from_millis(25)),
167            _ => {
168                let _ = child.kill();
169                let _ = child.wait();
170                return Err((
171                    MonitorSourceHealth::Error,
172                    format!("gh {} took longer than {}s", args[0], BUDGET.as_secs()),
173                ));
174            }
175        }
176    };
177    let mut out = String::new();
178    let mut err = String::new();
179    // both were piped above, so both are there to read
180    let _ = child.stdout.take().map(|mut s| s.read_to_string(&mut out));
181    let _ = child.stderr.take().map(|mut s| s.read_to_string(&mut err));
182    if !status.success() {
183        let reason = err
184            .lines()
185            .next()
186            .unwrap_or("no reason given")
187            .trim()
188            .to_string();
189        return Err((
190            MonitorSourceHealth::Error,
191            format!("gh {} failed: {reason}", args[0]),
192        ));
193    }
194    serde_json::from_str(&out).map_err(|e| {
195        (
196            MonitorSourceHealth::Error,
197            format!("gh {} printed no JSON: {e}", args[0]),
198        )
199    })
200}

202/// Fold what `gh` printed into signals after `after` (Unix seconds); without a position,
203/// only moments at or after `since` are shown.
204pub(crate) fn fold(
205    fetched: &Fetched,
206    after: Option<u64>,
207    since: u64,
208    reader: &Reader,
209) -> SourceRead<u64> {
210    let mut moments: Vec<(u64, Option<MonitorSignal>)> = Vec::new();
211    for pr in fetched.prs.as_array().into_iter().flatten() {
212        moments.extend(pull_request(pr, reader));
213    }
214    for run in fetched.runs.as_array().into_iter().flatten() {
215        if let Some(moment) = workflow_run(run, reader) {
216            moments.push(moment);
217        }
218    }
219    moments.sort_by_key(|(at, _)| *at);
220    let floor = after.unwrap_or_else(|| since.saturating_sub(1));
221    let mut read = SourceRead::default();
222    for (at, signal) in moments {
223        read.high = Some(read.high.map_or(at, |h: u64| h.max(at)));
224        if at <= floor {
225            continue;
226        }
227        match signal {
228            Some(signal) => read.signals.push((at, signal)),
229            None => read.suppressed += 1,
230        }
231    }
232    if let Some(after) = after {
233        read.high = Some(read.high.map_or(after, |h| h.max(after)));
234    }
235    read
236}

249/// The moments of one pull request: opened, and merged or closed.
250fn pull_request(pr: &Value, reader: &Reader) -> Vec<(u64, Option<MonitorSignal>)> {
251    let number = pr["number"].as_u64();
252    let title = text(pr, "title");
253    let branch = text(pr, "headRefName");
254    let author = pr["author"]["login"]
255        .as_str()
256        .unwrap_or_default()
257        .to_string();
258    let n = number.map(|n| format!("#{n}")).unwrap_or_default();
259    let make = |at: u64, kind: &str, severity, what: String| {
260        let mut s = MonitorSignal::new(at, MonitorSource::Github, kind, severity, &what);
261        s.pr = number;
262        s.branch = Some(branch.clone()).filter(|b| !b.is_empty());
263        s.actor = Some(author.clone()).filter(|a| !a.is_empty());
264        s.link = Some(text(pr, "url")).filter(|u| !u.is_empty());
265        s.for_you = reader.named_in(&title) || reader.named_in(&branch);
266        s
267    };
268    let mut moments = Vec::new();
269    if let Some(at) = time(pr, "createdAt") {
270        moments.push((
271            at,
272            Some(make(
273                at,
274                "pr_opened",
275                MonitorSeverity::Info,
276                format!("{n} opened: {title}"),
277            )),
278        ));
279    }
280    if let Some(at) = time(pr, "mergedAt") {
281        moments.push((
282            at,
283            Some(make(
284                at,
285                "pr_merged",
286                MonitorSeverity::Info,
287                format!("{n} merged: {title}"),
288            )),
289        ));
290    } else if text(pr, "state") == "CLOSED" {
291        if let Some(at) = time(pr, "closedAt") {
292            moments.push((
293                at,
294                Some(make(
295                    at,
296                    "pr_closed",
297                    MonitorSeverity::Info,
298                    format!("{n} closed: {title}"),
299                )),
300            ));
301        }
302    }
303    moments
304}

306/// The moment a workflow run finished, and whether it is news.
307fn workflow_run(run: &Value, reader: &Reader) -> Option<(u64, Option<MonitorSignal>)> {
308    if text(run, "status") != "completed" {
309        return None;
310    }
311    let at = time(run, "updatedAt")?;
312    let conclusion = text(run, "conclusion");
313    let workflow = text(run, "workflowName");
314    let branch = text(run, "headBranch");
315    let trunk = branch == TRUNK;
316    let verdict = verdict(&conclusion);
317    if verdict == Verdict::Succeeded && !trunk {
318        return Some((at, None));
319    }
320    let family = family(&workflow);
321    let failed_jobs = failed_jobs(run);
322    let deployed = family == "deploy" && verdict == Verdict::Failed && job_succeeded(run, "deploy");
323    let (outcome, severity) = match verdict {
324        Verdict::Failed if deployed => ("needs_help", MonitorSeverity::Attention),
325        Verdict::Failed => ("failed", MonitorSeverity::Attention),
326        Verdict::Cancelled if trunk => ("cancelled", MonitorSeverity::Attention),
327        Verdict::Cancelled => ("cancelled", MonitorSeverity::Info),
328        Verdict::Succeeded => ("succeeded", MonitorSeverity::Info),
329    };
330    let detail = if deployed {
331        format!(" (deployed; {} failed)", failed_jobs.join(", "))
332    } else {
333        String::new()
334    };
335    let mut s = MonitorSignal::new(
336        at,
337        MonitorSource::Github,
338        &format!("{family}_{outcome}"),
339        severity,
340        &format!(
341            "{workflow} {conclusion} on {branch}{detail}: {}",
342            text(run, "displayTitle")
343        ),
344    );
345    s.branch = Some(branch.clone()).filter(|b| !b.is_empty());
346    s.link = Some(text(run, "url")).filter(|u| !u.is_empty());
347    s.for_you = verdict != Verdict::Succeeded && reader.named_in(&branch);
348    Some((at, Some(s)))
349}
350
351/// What a finished run's conclusion says about it.
352#[derive(Debug, Clone, Copy, PartialEq, Eq)]
353enum Verdict {
354    Succeeded,
355    Cancelled,
356    Failed,
357}
358
359fn verdict(conclusion: &str) -> Verdict {
360    match conclusion {
361        "success" | "skipped" | "neutral" => Verdict::Succeeded,
362        "cancelled" => Verdict::Cancelled,
363        _ => Verdict::Failed,
364    }
365}

367/// The family a workflow's signals are named by: `release`, `deploy` or `run`.
368fn family(workflow: &str) -> &'static str {
369    let lower = workflow.to_lowercase();
370    if lower.contains("release") {
371        "release"
372    } else if lower.contains("pages") || lower.contains("deploy") {
373        "deploy"
374    } else {
375        "run"
376    }
377}

379/// The names of a run's jobs that failed, when its jobs were read.
380fn failed_jobs(run: &Value) -> Vec<String> {
381    run["jobs"]
382        .as_array()
383        .into_iter()
384        .flatten()
385        .filter(|j| verdict(j["conclusion"].as_str().unwrap_or("failure")) == Verdict::Failed)
386        .map(|j| text(j, "name"))
387        .collect()
388}
389
390fn job_succeeded(run: &Value, name: &str) -> bool {
391    run["jobs"]
392        .as_array()
393        .into_iter()
394        .flatten()
395        .any(|j| text(j, "name") == name && j["conclusion"].as_str() == Some("success"))
396}
