1//! The UI's icons: the kit's set, declared once in `site/data/kit/icons.toml` beside the
2//! site's templates and compiled into the crate by `majordomus generate`, so the site and
3//! the UI draw one icon set. Paths follow Lucide (ISC licence), drawn in `currentColor`.
4//!
5//! An icon carries no meaning of its own: the word beside it always says what it is, and a
6//! name the set does not hold draws nothing rather than a guess.
7
8use std::collections::BTreeMap;
9use std::sync::LazyLock;
10
11use crate::cockpit::html::{el, El};
12
13/// The one declaration of the icons, repository-relative.
14pub const SOURCE: &str = "site/data/kit/icons.toml";
15
16/// Its compiled copy, written by `majordomus generate`.
17pub const COMPILED: &str = "apps/majordomus-cli/src/ui/icons.toml";
18
19static ICONS: LazyLock<BTreeMap<String, String>> =
20    LazyLock::new(|| parse(include_str!("icons.toml")));
21
22/// The `name = '<markup>'` entries of the `[icons]` table: the only shape the file uses.
23fn parse(text: &str) -> BTreeMap<String, String> {
24    let mut out = BTreeMap::new();
25    let mut inside = false;
26    for line in text.lines() {
27        let t = line.trim();
28        if t.starts_with('[') {
29            inside = t == "[icons]";
30            continue;
31        }
32        if !inside || t.is_empty() || t.starts_with('#') {
33            continue;
34        }
35        if let Some((name, value)) = t.split_once(" = ") {
36            let markup = value.trim().trim_matches('\'');
37            out.insert(name.trim().to_string(), markup.to_string());
38        }
39    }
40    out
41}
42
43/// Whether the set holds `name`.
44#[cfg(test)]
45pub fn has(name: &str) -> bool {
46    ICONS.contains_key(name)
47}
48
49/// One icon as an inline SVG, hidden from assistive technology because the word beside it
50/// is what is read.
51pub fn icon(name: &str) -> El {
52    el("svg")
53        .class("mj-ui-icon")
54        .attr("viewBox", "0 0 24 24")
55        .attr("fill", "none")
56        .attr("stroke", "currentColor")
57        .attr("stroke-width", "1.75")
58        .attr("stroke-linecap", "round")
59        .attr("stroke-linejoin", "round")
60        .attr("aria-hidden", "true")
61        .attr("focusable", "false")
62        .raw(ICONS.get(name).cloned().unwrap_or_default())
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68
69    /// Only the `[icons]` table's `name = '<markup>'` entries are read: another table, a
70    /// comment, a blank line and a line of another shape are passed over.
71    #[test]
72    fn only_the_icons_tables_entries_are_read() {
73        let set =
74            parse("[other]\nfoo = 'x'\n[icons]\n# a comment\n\nnot an entry\nbook = '<path/>'\n");
75        assert_eq!(set.len(), 1, "{set:?}");
76        assert_eq!(set.get("book").map(String::as_str), Some("<path/>"));
77    }
78
79    /// The compiled copy is the declaration's: every name the declaration holds is here
80    /// with the same markup, so the site and the UI draw one set.
81    #[test]
82    fn the_compiled_set_is_the_declared_set() {
83        let declared = parse(include_str!("../../../../site/data/kit/icons.toml"));
84        assert!(!declared.is_empty(), "the declaration holds no icon");
85        assert_eq!(*ICONS, declared);
86    }
87
88    /// Every icon the UI's shell names exists in the set.
89    #[test]
90    fn every_icon_the_shell_names_exists() {
91        for s in crate::ui::SCREENS {
92            assert!(
93                has(s.icon),
94                "{} names icon {} the set does not hold",
95                s.route,
96                s.icon
97            );
98        }
99    }
100
101    /// A name the set does not hold draws an empty glyph, never another icon.
102    #[test]
103    fn an_unknown_name_draws_nothing() {
104        let svg = icon("no-such-icon").render();
105        assert!(svg.starts_with("<svg") && svg.ends_with("></svg>"), "{svg}");
106    }
