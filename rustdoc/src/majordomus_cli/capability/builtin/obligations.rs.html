1//! The `obligations` module: what a task owes before it may be called completed, and
2//! whether the evidence that discharged each one still describes this tree.
3//!
4//! Two halves, and the difference between them decides the design.
5//!
6//! **The vocabulary is shipped.** `share/obligations.yaml` is data the distribution
7//! carries beside `share/events.yaml` and `share/kinds.yaml`: a token, what discharges
8//! it, and the pathspecs its evidence is taken over. It is identical in every clone and
9//! in every installation, so it is read here the way `commands.list` reads the command
10//! registry — through [`crate::share::Share`] — and it answers in a checkout that has
11//! never run the lifecycle.
12//!
13//! **The closure is local.** The task record and the ledger live under
14//! `.ai/local/state/`, the half [`super::continuity`] documents: never tracked, never
15//! published, one file per checkout. A served instance answers about the checkout the
16//! process was started in and about no other, and a fresh clone has nothing to report —
17//! which is reported as nothing rather than as zero obligations discharged.
18//!
19//! **It is read, never written.** `majordomus evidence` records; this reads. A second
20//! writer for the same ledger would be a second account of events, which is the thing
21//! ADR 0030 refuses.
22//!
23//! **The judgement is the validator's, not a second opinion.** `mj_validate_obligations`
24//! in `lib/evidence.sh` decides whether a task may be completed; everything below
25//! reproduces its rule — a remote fact is labelled against the commit it was taken at, a
26//! tree-bound fact is re-hashed over the pathspecs its token declares, and the label is
27//! [`Divergence`], whose four words are `mj_git_label`'s. No fifth vocabulary is invented
28//! here, and where the shell is generous — a token that names neither inputs nor a remote
29//! fact can never go stale — this is generous in the same place rather than stricter in
30//! private.
31//!
32//! **Where the two can differ, and in which direction.** `mj_obligation_inputs_hash`
33//! interpolates the token's pathspecs unquoted, so the shell expands them as filename
34//! globs before `git ls-files` ever sees them; without `globstar`, `**` matches one level.
35//! `docs`, whose pathspecs include `.ai/repo/**/README.md`, therefore selects 255 files
36//! there and 261 here, and `implementation`, whose pathspec is `*`, selects 1915 there and
37//! 2480 here — a bare `*` expands to the top-level entries that are not dotfiles, so the
38//! whole of `.ai/` and `.github/` is outside the hash the shell takes. This reader passes
39//! the pathspecs to git verbatim, which is what the vocabulary's `inputs` plainly mean, so
40//! its selection is a superset and the disagreement can only run one way: it may call
41//! evidence stale that `finish` would still accept, and never the reverse. [`inputs_files`]
42//! is on every entry so the difference is measurable from the surface rather than argued
43//! about. The repair belongs to `lib/evidence.sh` — `set -f` around the `git ls-files`
44//! call, so that word splitting still happens and pathname expansion does not.
45//!
46//! [`inputs_files`]: ObligationClosure::inputs_files
47//!
48//! ```
49//! use majordomus_cli::capability::builtin::obligations;
50//!
51//! // two capabilities: what a token means, and where this checkout's task stands
52//! let m = obligations::module();
53//! let ids: Vec<&str> = m.capabilities.iter().map(|e| e.capability.id.as_str()).collect();
54//! assert_eq!(ids, ["obligations.vocabulary", "obligations.closure"]);
55//!
56//! // and both report; neither records. `majordomus evidence` is the writer.
57//! assert!(m.capabilities.iter().all(|e| e.capability.kind.is_read_only()));
58//! ```

109/// One token a task may declare in `requires`, as the distribution ships it.
110///
111/// `remote` and `inputs` are what decide how its evidence is judged, and they are the
112/// file's decision rather than this reader's: a fact the working tree cannot establish is
113/// bound to the commit it was taken at, and everything else is bound to the bytes of the
114/// files its pathspecs select.
115///
116/// ```
117/// use majordomus_cli::capability::builtin::obligations::Obligation;
118/// let push: Obligation = serde_json::from_str(
119///     r#"{"id":"push","title":"The commit reached the remote",
120///         "summary":"The branch's head exists on the remote it tracks.",
121///         "discharged_by":"git","remote":true}"#,
122/// )
123/// .unwrap();
124/// assert!(push.remote);
125/// assert!(push.inputs.is_empty(), "a remote fact is bound to a commit, not to a tree");
126/// ```
127#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
128pub struct Obligation {
129    /// The token, as a task's `requires` names it.
130    pub id: String,
131    /// The obligation as a heading.
132    pub title: String,
133    /// What a worker is being asked to have done.
134    pub summary: String,
135    /// The command that produces the evidence. `none` for a token held by another line of
136    /// the contract and listed so that a report can say it rather than leave a hole.
137    pub discharged_by: String,
138    /// The pathspecs the evidence is hashed over. Empty for a token whose fact is remote,
139    /// and empty for one bound to neither: see [`ObligationClosure::staleness`].
140    #[serde(default, skip_serializing_if = "Vec::is_empty")]
141    pub inputs: Vec<String>,
142    /// True when the fact cannot be established from the working tree alone — a push, a
143    /// publication, a deployment. Such evidence is bound to a commit, not to a tree.
144    pub remote: bool,
145    /// What the vocabulary says about the token beyond its summary.
146    #[serde(default, skip_serializing_if = "String::is_empty")]
147    pub note: String,
148}

150/// The vocabulary the distribution ships: every token there is.
151///
152/// `count` is measured from `obligations` rather than written down, so a token added to
153/// the file is counted by the same act that declares it.
154///
155/// ```
156/// use majordomus_cli::capability::builtin::obligations::Vocabulary;
157/// let v: Vocabulary = serde_json::from_str(
158///     r#"{"version":1,"source":"/opt/majordomus/share/obligations.yaml","count":1,
159///         "obligations":[{"id":"commit","title":"The work is committed",
160///                         "summary":"In the branch's history, not the working tree.",
161///                         "discharged_by":"git","remote":false}]}"#,
162/// )
163/// .unwrap();
164/// assert_eq!(v.count, v.obligations.len());
165/// assert_eq!(v.version, 1, "the only shape this reader accepts");
166/// ```
167#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
168pub struct Vocabulary {
169    /// The file's own format version. `1` is the only one this reader accepts.
170    pub version: u64,
171    /// Where it was read from, absolute: the distribution's share directory, which is not
172    /// necessarily inside the repository.
173    pub source: String,
174    /// How many tokens there are. No number anywhere is written down.
175    pub count: usize,
176    /// Every token, in the order the file declares them.
177    pub obligations: Vec<Obligation>,
178}

180/// `share/obligations.yaml` as written. Unknown fields are not refused: the vocabulary
181/// belongs to the shell tool, this is one of its readers, and `doctor` is what proves the
182/// file well-formed. A reader that refused a field it had not been taught would turn
183/// somebody else's addition into this process's failure.
184#[derive(Debug, Default, Deserialize)]
185struct VocabularyFile {
186    #[serde(default)]
187    version: u64,
188    #[serde(default)]
189    obligations: Vec<Obligation>,
190}

192/// Read the shipped vocabulary from `share`, the distribution this process located, or —
193/// when the caller has none, which is an index built without an application around it —
194/// the way every other reader locates one, from `root`. A distribution that ships no
195/// vocabulary declares no tokens, which is not an error but is reported as such by the
196/// closure's findings when a task names one.
197///
198/// The distribution is passed in rather than resolved here because the two answers differ:
199/// a process told which distribution to read (`--share`, or a released binary run inside a
200/// repository that is not its own) has already resolved one, and a second resolution from
201/// the repository root would find another distribution or, when the executable lives
202/// outside the repository, none.
203fn vocabulary_at(share: Option<&Path>, root: &Path) -> Result<Vocabulary, String> {
204    let share = crate::share::Share::locate(share, root)
205        .map_err(|e| format!("the distribution's share directory was not found: {e}"))?;
206    let path = share.dir().join(VOCABULARY_FILE);
207    let text = std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
208    let file: VocabularyFile =
209        yaml::parse_into(&text).map_err(|e| format!("{}: {e}", path.display()))?;
210    if file.version != 1 {
211        return Err(format!(
212            "{}: the obligation vocabulary must be version 1, and this one is {}",
213            path.display(),
214            file.version
215        ));
216    }
217    Ok(Vocabulary {
218        version: file.version,
219        source: path.to_string_lossy().to_string(),
220        count: file.obligations.len(),
221        obligations: file.obligations,
222    })
223}

227/// Where one obligation stands.
228///
229/// The words are the validator's verdicts, not a severity scale: `stale` is not a worse
230/// `owed`, it is evidence that was true and no longer describes what it proved.
231///
232/// ```
233/// use majordomus_cli::capability::builtin::obligations::ObligationState;
234/// // only one of the four lets a task be called completed
235/// let completable = |s: ObligationState| s == ObligationState::Discharged;
236/// assert!(completable(ObligationState::Discharged));
237/// assert!(!completable(ObligationState::Stale), "evidence that no longer describes the tree");
238/// assert!(!completable(ObligationState::Owed));
239/// assert!(!completable(ObligationState::Undeclared));
240/// ```
241#[derive(
242    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
243)]
244#[serde(rename_all = "snake_case")]
245pub enum ObligationState {
246    /// Declared, and no evidence for it was ever recorded.
247    Owed,
248    /// Evidence exists and still describes this tree, or this commit.
249    Discharged,
250    /// Evidence exists and no longer describes what it proved: the inputs changed, or the
251    /// commit it named is not this one.
252    Stale,
253    /// The task requires a token the shipped vocabulary does not declare. Nothing can
254    /// discharge it, because nothing knows what would.
255    Undeclared,
256}
257
258impl ObligationState {
259    /// The word as serialised, for a caller that tallies or renders states without
260    /// carrying a copy of the vocabulary: the tallies of a [`Closure`] are keyed by
261    /// exactly these strings, and so is anything that groups the entries by standing.
262    ///
263    /// ```
264    /// use majordomus_cli::capability::builtin::obligations::ObligationState;
265    /// assert_eq!(ObligationState::Stale.as_str(), "stale");
266    /// assert_eq!(ObligationState::Undeclared.as_str(), "undeclared");
267    /// ```
268    pub fn as_str(self) -> &'static str {
269        match self {
270            ObligationState::Owed => "owed",
271            ObligationState::Discharged => "discharged",
272            ObligationState::Stale => "stale",
273            ObligationState::Undeclared => "undeclared",
274        }
275    }
276}

278/// The `task.evidence` line that discharged an obligation, as the ledger holds it.
279///
280/// Nothing older is consulted: evidence is superseded by evidence, and the ledger keeps
281/// the history for a reader that wants it.
282/// ```
283/// use majordomus_cli::capability::builtin::obligations::Evidence;
284/// let e: Evidence = serde_json::from_str(
285///     r#"{"recorded_at":"2026-09-09T21:01:00Z","head":"f00ba7","branch":"master",
286///         "kind":"ci","command":"scripts/ci/reference-check","inputs_hash":"7f68b9"}"#,
287/// )
288/// .unwrap();
289/// assert_eq!(e.head, "f00ba7", "the envelope's commit is what a remote fact is judged against");
290/// assert!(e.artifact.is_empty(), "one of command and artifact; narrative is not evidence");
291/// ```
292#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
293pub struct Evidence {
294    /// When the line was appended.
295    pub recorded_at: String,
296    /// The commit the ledger's envelope stamped on it.
297    pub head: String,
298    /// The branch it was recorded on.
299    #[serde(default, skip_serializing_if = "String::is_empty")]
300    pub branch: String,
301    /// How it was taken: `test`, `build`, `ci`, `artifact` or `manual`.
302    #[serde(default, skip_serializing_if = "String::is_empty")]
303    pub kind: String,
304    /// The command that produced it. Narrative is not evidence, so one of this and
305    /// `artifact` is always present.
306    #[serde(default, skip_serializing_if = "String::is_empty")]
307    pub command: String,
308    /// The reference it points at, such as a published URL.
309    #[serde(default, skip_serializing_if = "String::is_empty")]
310    pub artifact: String,
311    /// What the command said, when its output was the point.
312    #[serde(default, skip_serializing_if = "String::is_empty")]
313    pub result: String,
314    /// The episode that recorded it, when one was open.
315    #[serde(default, skip_serializing_if = "String::is_empty")]
316    pub session: String,
317    /// The hash of the token's declared inputs at the moment it was taken. Empty for a
318    /// token that declares none.
319    #[serde(default, skip_serializing_if = "String::is_empty")]
320    pub inputs_hash: String,
321}

323/// One obligation the task declared, joined with what the vocabulary says about it and
324/// with the evidence that does or does not discharge it.
325///
326/// The vocabulary's fields are repeated here on purpose: a client asking what this task
327/// owes gets the token's title, its summary and the command that would discharge it in the
328/// same answer, and needs no second call to render a report.
329///
330/// ```
331/// use majordomus_cli::capability::builtin::obligations::{ObligationClosure, ObligationState};
332/// let owed: ObligationClosure = serde_json::from_str(
333///     r#"{"id":"pages","title":"The published site serves this commit","remote":true,
334///         "state":"owed","detail":"owed, and no evidence was recorded",
335///         "reproduce":"majordomus evidence --covers pages --command 'scripts/pages verify'"}"#,
336/// )
337/// .unwrap();
338/// assert_eq!(owed.state, ObligationState::Owed);
339/// assert!(owed.evidence.is_none() && owed.staleness.is_none(), "nothing to label");
340/// assert!(owed.reproduce.starts_with("majordomus evidence"), "a finding carries its repair");
341/// ```
342#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
343pub struct ObligationClosure {
344    /// The token, as the task's `requires` names it.
345    pub id: String,
346    /// From the vocabulary; the token itself when it declares none.
347    pub title: String,
348    /// From the vocabulary.
349    #[serde(default, skip_serializing_if = "String::is_empty")]
350    pub summary: String,
351    /// From the vocabulary: the command that produces the evidence.
352    #[serde(default, skip_serializing_if = "String::is_empty")]
353    pub discharged_by: String,
354    /// From the vocabulary: whether the fact is remote.
355    pub remote: bool,
356    /// From the vocabulary: the pathspecs the evidence is hashed over.
357    #[serde(default, skip_serializing_if = "Vec::is_empty")]
358    pub inputs: Vec<String>,
359    /// Where it stands.
360    pub state: ObligationState,
361    /// How far the evidence is from this checkout, in the repository's one staleness
362    /// vocabulary. `None` when there is no evidence to label.
363    ///
364    /// A remote fact is labelled against the commit it was taken at, because the site that
365    /// serves a commit goes on serving it while the tree moves underneath. A tree-bound
366    /// fact is labelled by re-hashing: equal hashes are `exact`, and a difference takes the
367    /// commit's label, which is `advanced` when the working tree has merely moved on and
368    /// `diverged` when the history it named is gone. A token that declares neither inputs
369    /// nor a remote fact is bound to nothing and stays `exact` once recorded — that is the
370    /// validator's behaviour, and it is reproduced rather than tightened here.
371    #[serde(default, skip_serializing_if = "Option::is_none")]
372    pub staleness: Option<Divergence>,
373    /// The line that discharged it, or `None`.
374    #[serde(default, skip_serializing_if = "Option::is_none")]
375    pub evidence: Option<Evidence>,
376    /// What the token's inputs hash to in this tree now — the other half of "stale against
377    /// what". `None` for a token that declares no inputs, and `None` when git could not be
378    /// asked which files they select.
379    #[serde(default, skip_serializing_if = "Option::is_none")]
380    pub inputs_hash_now: Option<String>,
381    /// How many tracked files that hash was taken over. Reported because the shell's
382    /// selection and this one are not always the same set — see the module header — and a
383    /// count is the cheapest way for a reader to see it rather than be told it.
384    #[serde(default, skip_serializing_if = "Option::is_none")]
385    pub inputs_files: Option<usize>,
386    /// One line: what this obligation's standing actually is, in the words the validator
387    /// uses when it refuses.
388    pub detail: String,
389    /// The command that would discharge it. Every finding here carries the way to act on
390    /// it, as `project.finding-carries-reproduce` asks.
391    pub reproduce: String,
392}

394/// What this checkout's active task owes, and how much of it is still true.
395///
396/// A clone that has never run the lifecycle answers this too, and answers it as absence:
397/// `present` false, no task, and a finding saying so. "Nothing owed" and "nothing to owe
398/// it" are different facts, and a served instance must not report the second as the first.
399///
400/// ```
401/// use majordomus_cli::capability::builtin::obligations::Closure;
402/// let fresh: Closure = serde_json::from_str(
403///     r#"{"present":false,"worktree":"/srv/clone","branch":"master","closed":false,
404///         "tallies":{},
405///         "findings":["no active task in this checkout (.ai/local/state/current.yaml); nothing owes anything here"]}"#,
406/// )
407/// .unwrap();
408/// assert!(!fresh.present && !fresh.closed, "nothing that does not exist is closed");
409/// assert!(fresh.task.is_none() && fresh.obligations.is_empty());
410/// assert_eq!(fresh.findings.len(), 1, "absence is reported, not implied");
411/// ```
412#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
413pub struct Closure {
414    /// Whether this checkout has a task to report about at all. False in a clone that has
415    /// never run the lifecycle, which is not a fault and is not "nothing owed".
416    pub present: bool,
417    /// The worktree this answer is about. Every reading below is scoped to it, and to no
418    /// other checkout of the same repository.
419    pub worktree: String,
420    /// The branch, or `DETACHED`.
421    pub branch: String,
422    /// The commit this checkout is on.
423    #[serde(default, skip_serializing_if = "String::is_empty")]
424    pub head: String,
425    /// `clean` or `dirty`.
426    #[serde(default, skip_serializing_if = "String::is_empty")]
427    pub working_tree: String,
428    /// The active task, or `None`. Its `requires` is the list the entries below expand.
429    #[serde(default, skip_serializing_if = "Option::is_none")]
430    pub task: Option<ActiveTask>,
431    /// True when the task declares obligations and every one of them is discharged and
432    /// current — that is, when `majordomus.obligation-closure` would not refuse
433    /// `finish --outcome completed`. False when anything is owed, stale or undeclared, and
434    /// false when there is no task: nothing that does not exist is closed.
435    pub closed: bool,
436    /// How many obligations stand where, by state word. Absent states are absent rather
437    /// than zero, so a reader never has to know the vocabulary to read the tallies.
438    pub tallies: BTreeMap<String, usize>,
439    /// Every obligation the task declared, in the order it declared them.
440    #[serde(default, skip_serializing_if = "Vec::is_empty")]
441    pub obligations: Vec<ObligationClosure>,
442    /// What a reader should know before trusting any of the above: an unreadable
443    /// vocabulary, a ledger that could not be read, a task that declares nothing. Empty is
444    /// the ordinary case.
445    #[serde(default, skip_serializing_if = "Vec::is_empty")]
446    pub findings: Vec<String>,
447}

449// ---------------------------------------------------------------- the input hash

451/// The hash `mj_inputs_hash` computes, reproduced.
452///
453/// The shell hashes each selected file, renders one `<path> <sha256>` line per file in
454/// git's own index order, and hashes that listing. Both halves matter: the order is git's,
455/// so two runs over one tree agree, and the listing carries the path, so moving a file
456/// changes the hash even when its bytes do not.
457///
458/// A file git tracks and the working tree does not hold is skipped rather than failed, for
459/// the same reason the shell skips it — `shasum` writes to stderr and `xargs` carries on —
460/// so a deleted-but-tracked file makes the hash differ rather than making the answer an
461/// error.
462///
463/// ```
464/// use majordomus_cli::capability::builtin::obligations::listing_hash;
465/// // the listing is what is hashed, and it is one line per file, path first
466/// assert_eq!(
467///     listing_hash("a.txt e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855\n"),
468///     listing_hash("a.txt e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855\n")
469/// );
470/// assert_ne!(listing_hash("a.txt 00\n"), listing_hash("b.txt 00\n"));
471/// ```
472pub fn listing_hash(listing: &str) -> String {
473    crate::policy::sha256_hex(listing)
474}
475
476/// The hash of one obligation's declared inputs in this tree and how many files it was
477/// taken over, or `None` when the token declares none — which is the empty string in the
478/// ledger and is compared as such.
479pub(crate) fn inputs_hash(root: &Path, specs: &[String]) -> Option<(String, usize)> {
480    if specs.is_empty() {
481        return None;
482    }
483    let refs: Vec<&str> = specs.iter().map(String::as_str).collect();
484    let files = git::ls_files_any(root, &refs).ok()?;
485    if files.is_empty() {
486        // the shell's empty answer: no file selected hashes to nothing at all, and the
487        // recorded value for such a token is the empty string
488        return Some((String::new(), 0));
489    }
490    let mut listing = String::new();
491    for file in &files {
492        let Ok(bytes) = std::fs::read(root.join(file)) else {
493            continue;
494        };
495        listing.push_str(file);
496        listing.push(' ');
497        listing.push_str(&crate::policy::sha256_bytes_hex(&bytes));
498        listing.push('\n');
499    }
500    Some((listing_hash(&listing), files.len()))
501}

505/// The newest `task.evidence` line per token for one task, read from this checkout's
506/// ledger. A line that is not JSON is skipped and counted, because a ledger that has grown
507/// one bad line still holds the rest.
508fn evidence_for(path: &Path, task: &str) -> (BTreeMap<String, Evidence>, usize) {
509    let Ok(text) = std::fs::read_to_string(path) else {
510        return (BTreeMap::new(), 0);
511    };
512    let mut out: BTreeMap<String, Evidence> = BTreeMap::new();
513    let mut skipped = 0usize;
514    for line in text.lines().filter(|l| !l.trim().is_empty()) {
515        let Ok(v) = serde_json::from_str::<serde_json::Value>(line) else {
516            skipped += 1;
517            continue;
518        };
519        let s = |k: &str| {
520            v.get(k)
521                .and_then(serde_json::Value::as_str)
522                .unwrap_or_default()
523                .to_string()
524        };
525        if s("event") != EVIDENCE_EVENT || s("task") != task {
526            continue;
527        }
528        let covers = s("covers");
529        if covers.is_empty() {
530            skipped += 1;
531            continue;
532        }
533        // last line wins: the file is append-only and ordered, so the last is the newest
534        out.insert(
535            covers,
536            Evidence {
537                recorded_at: s("ts"),
538                head: s("head"),
539                branch: s("branch"),
540                kind: s("kind"),
541                command: s("command"),
542                artifact: s("artifact"),
543                result: s("result"),
544                session: s("session"),
545                inputs_hash: s("inputs_hash"),
546            },
547        );
548    }
549    (out, skipped)
550}

554/// `mj_git_label`, whole: the branch decides first, then equality, then ancestry. A
555/// recorded branch this reader was not given is not a `different_context` — an older
556/// ledger line carries no branch, and calling somebody else's work on it would be worse
557/// than declining to.
558fn label(
559    root: &Path,
560    recorded_head: &str,
561    recorded_branch: &str,
562    head: Option<&str>,
563    branch: &str,
564) -> Divergence {
565    if !recorded_branch.is_empty() && recorded_branch != branch {
566        return Divergence::DifferentContext;
567    }
568    continuity::divergence(root, recorded_head, head)
569}

571/// The standing of one declared token, by the rule `mj_validate_obligations` applies.
572fn judge(
573    root: &Path,
574    token: &str,
575    declared: Option<&Obligation>,
576    evidence: Option<&Evidence>,
577    head: Option<&str>,
578    branch: &str,
579) -> ObligationClosure {
580    let Some(o) = declared else {
581        return ObligationClosure {
582            id: token.to_string(),
583            title: token.to_string(),
584            summary: String::new(),
585            discharged_by: String::new(),
586            remote: false,
587            inputs: Vec::new(),
588            state: ObligationState::Undeclared,
589            staleness: None,
590            evidence: evidence.cloned(),
591            inputs_hash_now: None,
592            inputs_files: None,
593            detail: format!(
594                "the task requires '{token}', which the shipped obligation vocabulary does not declare"
595            ),
596            reproduce: "majordomus evidence --help".into(),
597        };
598    };
599
600    let reproduce = format!(
601        "majordomus evidence --covers {} --command '{}'",
602        o.id, o.discharged_by
603    );
604    let measured = if o.remote {
605        // a remote fact is bound to a commit; re-hashing a tree it never described would
606        // be a number nobody could act on
607        None
608    } else {
609        inputs_hash(root, &o.inputs)
610    };
611    let hash_now = measured.as_ref().map(|(h, _)| h.clone());
612    let files_now = measured.as_ref().map(|(_, n)| *n);
613
614    let Some(ev) = evidence else {
615        return ObligationClosure {
616            id: o.id.clone(),
617            title: o.title.clone(),
618            summary: o.summary.clone(),
619            discharged_by: o.discharged_by.clone(),
620            remote: o.remote,
621            inputs: o.inputs.clone(),
622            state: ObligationState::Owed,
623            staleness: None,
624            evidence: None,
625            inputs_hash_now: hash_now,
626            inputs_files: files_now,
627            detail: "owed, and no evidence was recorded".into(),
628            reproduce,
629        };
630    };
631
632    let commit = label(root, &ev.head, &ev.branch, head, branch);
633    let short = |h: &str| h.chars().take(12).collect::<String>();
634
635    let (staleness, detail) = if o.remote {
636        let detail = match commit {
637            Divergence::Exact => "discharged at this commit".to_string(),
638            Divergence::Advanced => format!(
639                "the evidence names {}, and the branch has moved since; the fact it proved is about the older commit",
640                short(&ev.head)
641            ),
642            other => format!(
643                "the evidence was taken in a {} context ({}); it does not describe this branch",
644                other.as_str(),
645                short(&ev.head)
646            ),
647        };
648        (commit, detail)
649    } else if hash_now.as_deref().unwrap_or("") == ev.inputs_hash {
650        (
651            Divergence::Exact,
652            format!("discharged over inputs {}", short(&ev.inputs_hash)),
653        )
654    } else {
655        // the tree no longer hashes to what the evidence proved. Which of the four words
656        // that is, is the commit's answer — with `exact` ruled out, because a tree that
657        // hashes differently at the same commit has moved on inside it.
658        let moved = match commit {
659            Divergence::Exact => Divergence::Advanced,
660            other => other,
661        };
662        (
663            moved,
664            format!(
665                "the evidence was taken over inputs {} and this tree hashes to {} over {} file(s); it no longer describes what it proved",
666                short(&ev.inputs_hash),
667                short(hash_now.as_deref().unwrap_or("")),
668                files_now.unwrap_or(0)
669            ),
670        )
671    };
672
673    ObligationClosure {
674        id: o.id.clone(),
675        title: o.title.clone(),
676        summary: o.summary.clone(),
677        discharged_by: o.discharged_by.clone(),
678        remote: o.remote,
679        inputs: o.inputs.clone(),
680        state: if staleness == Divergence::Exact {
681            ObligationState::Discharged
682        } else {
683            ObligationState::Stale
684        },
685        staleness: Some(staleness),
686        evidence: Some(ev.clone()),
687        inputs_hash_now: hash_now,
688        inputs_files: files_now,
689        detail,
690        reproduce,
691    }
692}

701fn obligations_closure(ctx: &Context, _: Empty) -> Result<Closure, CapabilityError> {
702    let root = PathBuf::from(&ctx.index.repository.root);
703    let dir = root.join(STATE_DIR);
704
705    let (branch, head, working_tree) = match &ctx.index.repository.git {
706        GitState::Available(info) => (
707            info.branch.clone().unwrap_or_else(|| "DETACHED".into()),
708            info.head.clone(),
709            info.working_tree.clone(),
710        ),
711        GitState::Unavailable { .. } => ("DETACHED".into(), None, String::new()),
712    };
713
714    let mut findings = Vec::new();
715    let task = continuity::read_task(&dir.join("current.yaml"));
716
717    let vocabulary = match vocabulary_at(ctx.index.share.as_deref(), &root) {
718        Ok(v) => v.obligations,
719        Err(reason) => {
720            findings.push(format!(
721                "the obligation vocabulary could not be read, so no token can be resolved: {reason}"
722            ));
723            Vec::new()
724        }
725    };
726
727    let mut obligations = Vec::new();
728    if let Some(t) = &task {
729        if t.requires.is_empty() {
730            findings.push(format!(
731                "task {} declares no obligations, so this rule holds it to nothing beyond the rest of the contract",
732                t.id
733            ));
734        } else {
735            let (evidence, skipped) = evidence_for(&dir.join("ledger.jsonl"), &t.id);
736            if skipped > 0 {
737                findings.push(format!(
738                    "{skipped} ledger line(s) could not be read and were skipped; run `majordomus doctor`"
739                ));
740            }
741            for token in &t.requires {
742                obligations.push(judge(
743                    &root,
744                    token,
745                    vocabulary.iter().find(|o| &o.id == token),
746                    evidence.get(token),
747                    head.as_deref(),
748                    &branch,
749                ));
750            }
751        }
752    } else {
753        findings.push(format!(
754            "no active task in this checkout ({STATE_DIR}/current.yaml); nothing owes anything here"
755        ));
756    }
757
758    let mut tallies: BTreeMap<String, usize> = BTreeMap::new();
759    for o in &obligations {
760        *tallies.entry(o.state.as_str().to_string()).or_default() += 1;
761    }
762
763    Ok(Closure {
764        present: task.is_some(),
765        worktree: root.to_string_lossy().to_string(),
766        branch,
767        head: head.unwrap_or_default(),
768        working_tree,
769        closed: !obligations.is_empty()
770            && obligations
771                .iter()
772                .all(|o| o.state == ObligationState::Discharged),
773        task,
774        tallies,
775        obligations,
776        findings,
777    })
778}

780/// The module: the vocabulary every clone shares, and the closure only this checkout can
781/// answer.
782///
783/// Both are read-only and neither has a command-line projection, for the reason
784/// [`super::continuity`] states about the local half: `majordomus check` and
785/// `majordomus finish` are already the command line's answer to this question, and a
786/// second verb would be a second account of the same judgement.
787///
788/// ```
789/// use majordomus_cli::capability::builtin::obligations::module;
790/// let m = module();
791/// assert_eq!(m.id.as_str(), "obligations");
792/// // every projection is derived from the declaration; none is written anywhere else
793/// let routes: Vec<&str> = m
794///     .capabilities
795///     .iter()
796///     .filter_map(|e| e.capability.exposure.http.as_ref().map(|h| h.path.as_str()))
797///     .collect();
798/// assert_eq!(routes, ["/api/v1/obligations", "/api/v1/obligations/closure"]);
799/// ```
800pub fn module() -> ModuleDescriptor {
801    module! {
802        id: "obligations",
803        title: "Obligations",
804        description: "What a task owes before it may be called completed, and whether the evidence that discharged each obligation still describes this tree. The vocabulary is data the distribution ships and answers in any clone; the closure is read from the local half of the layer, which this process serves to the worker in front of it and never publishes. Read, never written: `majordomus evidence` records, and a second writer for one ledger would be a second account of the same events.",
805        stability: Stability::BehaviorallyVerified,
806        capabilities: [
807            capability! {
808                id: "obligations.vocabulary",
809                title: "Every obligation there is",
810                description: "The tokens a task may declare in `requires`: what each one asks of a worker, the command that discharges it, the pathspecs its evidence is hashed over, and whether its fact is remote and therefore bound to a commit rather than to a tree. Shipped data, identical in every clone, so this answers in a checkout that has never run the lifecycle.",
811                input: Empty,
812                output: Vocabulary,
813                stability: Stability::BehaviorallyVerified,
814                exposure: Exposure {
815                    mcp: Some(McpExposure {
816                        tool: Some("majordomus_obligations".into()),
817                        resource: Some(McpResource { uri: OBLIGATIONS_URI.into(), name: "obligations".into() }),
818                    }),
819                    http: get("/api/v1/obligations"),
820                    cli: None,
821                },
822                tags: ["obligations", "completion"],
823                // shipped data, immutable for the life of the process
824                cache: CachePolicy::Process { max_entries: 2, ttl_seconds: None },
825                handler: obligations_vocabulary,
826            },
827            capability! {
828                id: "obligations.closure",
829                title: "What this task still owes",
830                description: "Every obligation the active task declared, joined with what the vocabulary says about it and with the evidence that does or does not discharge it: what is owed, what is discharged, and what has gone stale — with the recorded input hash and the tree's current one, or the recorded commit and its label, so a reader can see against what. The judgement is the one `finish` applies, reproduced rather than re-decided, and the staleness words are the repository's only four. A checkout with no task reports that, rather than reporting nothing owed.",
831                input: Empty,
832                output: Closure,
833                stability: Stability::BehaviorallyVerified,
834                exposure: Exposure {
835                    mcp: Some(McpExposure {
836                        tool: Some("majordomus_obligation_closure".into()),
837                        resource: Some(McpResource { uri: CLOSURE_URI.into(), name: "obligation-closure".into() }),
838                    }),
839                    http: get("/api/v1/obligations/closure"),
840                    cli: None,
841                },
842                tags: ["obligations", "completion", "continuity"],
843                // Short-lived, for continuity's reason: the ledger and the task record are
844                // written by another process, and an answer cached for a minute would
845                // report an obligation that had since been discharged.
846                cache: CachePolicy::Process { max_entries: 2, ttl_seconds: Some(2) },
847                handler: obligations_closure,
848            },
849        ],
850    }

948    /// The four verdicts, decided the way `mj_validate_obligations` decides them. The
949    /// point of the test is that the two surfaces cannot disagree: a token with no
950    /// evidence is owed, a remote fact whose branch has moved on is stale, and a
951    /// tree-bound fact is judged by its hash and not by its commit.
952    #[test]
953    fn the_verdict_is_the_validators_verdict() {
954        let root = Path::new("/nonexistent-so-git-cannot-answer");
955        let remote = Obligation {
956            id: "push".into(),
957            title: "The commit reached the remote".into(),
958            summary: String::new(),
959            discharged_by: "git".into(),
960            inputs: Vec::new(),
961            remote: true,
962            note: String::new(),
963        };
964        let tree = Obligation {
965            id: "tests".into(),
966            title: "The cases were run".into(),
967            summary: String::new(),
968            discharged_by: "usecase impact".into(),
969            // a pathspec git is never asked about, because the root does not exist: the
970            // hash is `None`, which compares equal to the empty recorded hash
971            inputs: vec!["lib/**".into()],
972            remote: false,
973            note: String::new(),
974        };
975        let ev = |head: &str, branch: &str, hash: &str| Evidence {
976            recorded_at: "2026-09-09T00:00:00Z".into(),
977            head: head.into(),
978            branch: branch.into(),
979            kind: "manual".into(),
980            command: "git".into(),
981            artifact: String::new(),
982            result: String::new(),
983            session: String::new(),
984            inputs_hash: hash.into(),
985        };
986
987        // nothing recorded: owed, and the way to discharge it is in the answer
988        let owed = judge(root, "push", Some(&remote), None, Some("aaa"), "master");
989        assert_eq!(owed.state, ObligationState::Owed);
990        assert_eq!(owed.staleness, None);
991        assert_eq!(
992            owed.reproduce,
993            "majordomus evidence --covers push --command 'git'"
994        );
995
996        // a remote fact taken at this commit on this branch: discharged, exactly
997        let ok = judge(
998            root,
999            "push",
1000            Some(&remote),
1001            Some(&ev("aaa", "master", "")),
1002            Some("aaa"),
1003            "master",
1004        );
1005        assert_eq!(ok.state, ObligationState::Discharged);
1006        assert_eq!(ok.staleness, Some(Divergence::Exact));
1007
1008        // the same fact, recorded on another branch: not about this work at all
1009        let elsewhere = judge(
1010            root,
1011            "push",
1012            Some(&remote),
1013            Some(&ev("aaa", "other", "")),
1014            Some("aaa"),
1015            "master",
1016        );
1017        assert_eq!(elsewhere.state, ObligationState::Stale);
1018        assert_eq!(elsewhere.staleness, Some(Divergence::DifferentContext));
1019
1020        // a tree-bound fact whose recorded hash is not this tree's: stale, and the answer
1021        // carries both hashes so a reader can see against what
1022        let moved = judge(
1023            root,
1024            "tests",
1025            Some(&tree),
1026            Some(&ev("aaa", "master", "deadbeefdeadbeef")),
1027            Some("aaa"),
1028            "master",
1029        );
1030        assert_eq!(moved.state, ObligationState::Stale);
1031        assert!(moved.detail.contains("deadbeefdead"));
1032
1033        // a token nobody declares can be discharged by nothing, and says so
1034        let undeclared = judge(root, "invented", None, None, Some("aaa"), "master");
1035        assert_eq!(undeclared.state, ObligationState::Undeclared);
1036        assert!(undeclared.detail.contains("does not declare"));
1037    }
