1//! The forge adapter: the one place pull-request integration talks to the network.
2//!
3//! SECURITY.md declares it: `majordomus prs refresh` and `majordomus prs drain` — and
4//! nothing else — run the GitHub CLI (`gh`), `git fetch` and `git ls-remote` against this
5//! repository's own remote. The HTTP server, the MCP tools and the Cockpit never do: they render the last
6//! [`ForgeObservation`] recorded under `.ai/local/state/integration/`, with its age, so a
7//! page load can never reach the network. The observation is plain data, so every test
8//! builds one by hand and no test needs the forge.
9//!
10//! A required check the base binds to an app is only that app's check run, and `gh pr list`
11//! does not say which app wrote a check run. So when a context is bound, the refresh reads
12//! every open pull request's checks once more, from the forge's GraphQL rollup
13//! ([`WRITERS_QUERY`]), with the app of each ([`attributed_checks()`], [`attribute()`]). A
14//! base that binds nothing asks nothing more.
15//!
16//! Who may declare that one pull request replaces another is the forge's word too, and
17//! `gh pr list` does not carry it. So whenever a pull request is open, the refresh reads, with
18//! every open one, its author's association and the pull requests that mention it
19//! ([`DECLARATIONS_QUERY`], [`declarations()`], [`declare()`]): a declaration about an open pull
20//! request is read from that pull request's own cross-references, never from a bounded search
21//! of closed ones. One whose cross-references could not be read whole says so
22//! ([`CrossReferenceRead`]), and the classifier holds it and no other. A read the forge would
23//! not answer is not a hold: it fails the observation, and none is recorded from a part.
24

117/// Origin's branches from `git ls-remote --heads` output: name → tip. The module is private,
118/// so the examples here are text; the unit tests run the same assertions.
119///
120/// ```text
121/// use majordomus_cli::integration::forge::remote_heads_of;
122/// let heads = remote_heads_of("a1\trefs/heads/master\nb2\trefs/heads/feature/x\nc3\trefs/tags/v1\n");
123/// assert_eq!(heads.get("feature/x").map(String::as_str), Some("b2"));
124/// assert_eq!(heads.len(), 2, "a tag is not a branch");
125/// ```
126pub fn remote_heads_of(ls_remote: &str) -> BTreeMap<String, String> {
127    ls_remote
128        .lines()
129        .filter_map(|l| {
130            let (sha, name) = l.split_once('\t')?;
131            let name = name.trim().strip_prefix("refs/heads/")?;
132            (!sha.is_empty() && !name.is_empty()).then(|| (name.to_string(), sha.to_string()))
133        })
134        .collect()
135}

137/// The merged pull requests whose head branch origin still serves at the head that merged,
138/// from `gh pr list --state merged --json number,state,headRefName,headRefOid,isCrossRepository,mergedAt`.
139///
140/// A pull request counts only when the forge says it `MERGED` and when: it came from this
141/// repository (a fork's branch is not a branch here, whatever it is called), its branch is not
142/// the base, and origin's tip of that branch is exactly its merged head. One branch, one entry:
143/// when several merged pull requests left the same branch at the same head, the newest
144/// number names it. In branch order.
145///
146/// ```text
147/// use majordomus_cli::integration::forge::{merged_branches_of, remote_heads_of};
148/// let heads = remote_heads_of("aa\trefs/heads/fix/a\nbb\trefs/heads/fix/b\ncc\trefs/heads/master\n");
149/// let merged = serde_json::json!([
150///   {"number": 1, "state": "MERGED", "headRefName": "fix/a", "headRefOid": "aa", "isCrossRepository": false, "mergedAt": "t1"},
151///   {"number": 2, "state": "MERGED", "headRefName": "fix/b", "headRefOid": "b0", "isCrossRepository": false, "mergedAt": "t2"},
152///   {"number": 3, "state": "OPEN",   "headRefName": "fix/a", "headRefOid": "aa", "isCrossRepository": false}
153/// ]);
154/// let left = merged_branches_of(&merged, &heads, "master");
155/// assert_eq!(left.len(), 1, "fix/b moved after its merge, and an open one is not merged");
156/// assert_eq!((left[0].branch.as_str(), left[0].pr), ("fix/a", 1));
157/// ```
158pub fn merged_branches_of(
159    merged: &Value,
160    heads: &BTreeMap<String, String>,
161    base: &str,
162) -> Vec<MergedBranch> {
163    let mut by_branch: BTreeMap<String, MergedBranch> = BTreeMap::new();
164    for p in merged.as_array().into_iter().flatten() {
165        let s = |k: &str| p.get(k).and_then(Value::as_str).unwrap_or("");
166        let (Some(pr), "MERGED") = (p.get("number").and_then(Value::as_u64), s("state")) else {
167            continue;
168        };
169        let (branch, tip, merged_at) = (s("headRefName"), s("headRefOid"), s("mergedAt"));
170        let fork = p
171            .get("isCrossRepository")
172            .and_then(Value::as_bool)
173            .unwrap_or(true);
174        if fork || branch.is_empty() || branch == base || tip.is_empty() || merged_at.is_empty() {
175            continue;
176        }
177        if heads.get(branch).map(String::as_str) != Some(tip) {
178            continue;
179        }
180        let entry = MergedBranch {
181            branch: branch.to_string(),
182            tip: tip.to_string(),
183            pr,
184            merged_at: merged_at.to_string(),
185        };
186        match by_branch.get(branch) {
187            Some(seen) if seen.pr > pr => {}
188            _ => {
189                by_branch.insert(branch.to_string(), entry);
190            }
191        }
192    }
193    by_branch.into_values().collect()
194}

196/// A pull request that is no longer open, as the forge reported it.
197#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
198pub struct ResolvedPullRequest {
199    /// Whether the forge says it was merged, rather than closed unmerged. Evidence, and the
200    /// difference between a successor that released its hold (closed) and one merged somewhere
201    /// master does not contain. Never what makes a successor landed: git does.
202    pub merged: bool,
203    /// The commit its branch pointed at when it was merged or closed.
204    pub head_sha: String,
205    /// The body, for the supersessions it declares; never rendered.
206    #[serde(default)]
207    pub body: String,
208    /// Its merge commit, as the forge named it; empty when it has none or it was not read.
209    #[serde(default)]
210    pub merge_commit: String,
211    /// Its author's login; empty when unread. Evidence only.
212    #[serde(default)]
213    pub author: String,
214    /// The forge's word for what its author is to the repository; empty when unread.
215    #[serde(default)]
216    pub author_association: String,
217    /// Whether its head lives in a fork. A record that does not say is read as a fork.
218    #[serde(default = "unread_is_a_fork")]
219    pub cross_repository: bool,
220    /// The branch it asked to merge into; empty when unread. Evidence only.
221    #[serde(default)]
222    pub base_ref: String,
223    /// How many files the forge says it changes; zero when unread. A veto, never a proof: a
224    /// pull request that changes no file brought nothing to master, wherever its head now
225    /// points, so it is never a successor that landed.
226    #[serde(default)]
227    pub changed_files: u64,
228    /// Whether the reading said both where its head lives and how many files it changes.
229    /// The two defaults above are vetoes: read as facts, "a fork's" and "no file" each say a
230    /// successor closed unmerged brought nothing and release what it held. So a reading
231    /// that left either out is not one to decide by, and a record that does not say is not
232    /// whole.
233    #[serde(default)]
234    pub whole: bool,
235}

238/// which declares nothing.
239fn unread_is_a_fork() -> bool {
240    true

258/// [`resolved_of()`] reads, but for the author's association, which `gh pr view` does not

260pub const RESOLVED_FIELDS: &str =

263/// One pull request that is no longer open, from the source of a cross-reference
264/// ([`DECLARATIONS_QUERY`]) or from `gh pr view --json` output: `None` for an open one, or one
265/// without a head. What a reading does not say fails closed: no merge commit, no author, no
266/// association, a head read as a fork's and no changed file, so it declares nothing and
267/// never landed. A reading that does not say where the head lives, or how many files it
268/// changes, is not `whole`, and a successor read so is unread: it releases nothing.
269///
270/// The module is private, so the example is text; the unit tests run the same assertions.
271///
272/// ```text
273/// use majordomus_cli::integration::forge::resolved_of;
274/// let viewed = serde_json::json!({"number": 7, "state": "MERGED", "headRefOid": "h7",
275///     "mergeCommit": {"oid": "m7"}});
276/// let (number, read) = resolved_of(&viewed).unwrap();
277/// assert_eq!((number, read.merged, read.merge_commit.as_str()), (7, true, "m7"));
278/// assert!(read.cross_repository && read.author_association.is_empty(), "unread declares nothing");
279/// assert_eq!(read.changed_files, 0, "and changed nothing");
280/// assert!(!read.whole, "which is a default, not a reading: as a successor it is unread");
281/// assert!(resolved_of(&serde_json::json!({"number": 7, "state": "OPEN", "headRefOid": "h7"})).is_none());
282/// ```
283pub fn resolved_of(v: &Value) -> Option<(u64, ResolvedPullRequest)> {
284    let number = v.get("number")?.as_u64()?;
285    let merged = match v.get("state")?.as_str()? {
286        "MERGED" => true,
287        "CLOSED" => false,
288        _ => return None,
289    };
290    let head_sha = v.get("headRefOid")?.as_str()?.to_string();
291    if head_sha.is_empty() {
292        return None;
293    }
294    let text = |pointer: &str| {
295        v.pointer(pointer)
296            .and_then(Value::as_str)
297            .unwrap_or("")
298            .to_string()
299    };
300    let placed = v.get("isCrossRepository").and_then(Value::as_bool);
301    let files = v.get("changedFiles").and_then(Value::as_u64);
302    Some((
303        number,
304        ResolvedPullRequest {
305            merged,
306            head_sha,
307            body: text("/body"),
308            merge_commit: text("/mergeCommit/oid"),
309            author: text("/author/login"),
310            author_association: text("/authorAssociation"),
311            cross_repository: placed.unwrap_or(true),
312            base_ref: text("/baseRefName"),
313            changed_files: files.unwrap_or(0),
314            whole: placed.is_some() && files.is_some(),
315        },
316    ))
317}

319/// The pull requests no longer open that a supersession or a dependency involving an open one
320/// names: closed or merged ones whose body says they supersede an open one (`closed`: the
321/// sources of the open ones' cross-references, as [`declare()`] returned them), and the
322/// successors and dependencies open bodies name that are not open, each read with `view`. One
323/// `view` cannot read is left out, and the classifier says it is unread.
324pub fn resolved_for(
325    open: &[PullRequestObservation],
326    closed: &Value,
327    mut view: impl FnMut(u64) -> Option<Value>,
328) -> BTreeMap<u64, ResolvedPullRequest> {
329    use super::classify::declared_supersessions;
330    let numbers: BTreeSet<u64> = open.iter().map(|p| p.number).collect();
331    let mut resolved: BTreeMap<u64, ResolvedPullRequest> = closed
332        .as_array()
333        .into_iter()
334        .flatten()
335        .filter_map(resolved_of)
336        .filter(|(n, r)| {
337            !numbers.contains(n)
338                && declared_supersessions(&r.body)
339                    .supersedes
340                    .iter()
341                    .any(|t| t != n && numbers.contains(t))
342        })
343        .collect();
344    let named: BTreeSet<u64> = open
345        .iter()
346        .flat_map(|p| {
347            declared_supersessions(&p.body)
348                .superseded_by
349                .into_iter()
350                .chain(super::classify::declared_dependencies(&p.body))
351        })
352        .filter(|m| !numbers.contains(m) && !resolved.contains_key(m))
353        .collect();
354    for m in named {
355        if let Some((n, r)) = view(m).as_ref().and_then(resolved_of) {
356            if n == m {
357                resolved.insert(n, r);
358            }
359        }
360    }
361    resolved
362}

386fn gh(root: &Path, args: &[&str]) -> Result<(bool, String, String), ForgeError> {
387    let out = Command::new("gh")
388        .current_dir(root)
389        .args(args)
390        .output()
391        .map_err(|e| ForgeError(format!("the GitHub CLI (gh) could not run: {e}")))?;
392    Ok((
393        out.status.success(),
394        String::from_utf8_lossy(&out.stdout).into_owned(),
395        String::from_utf8_lossy(&out.stderr).into_owned(),
396    ))
397}

399/// `gh`, asked again on a transient failure ([`super::retry`]): every call here is a read,
400/// so asking twice is harmless. A refusal is returned as it came, for the caller to read.
401fn gh_retrying(root: &Path, args: &[&str]) -> Result<(bool, String, String), ForgeError> {
402    gh_retrying_unless(root, args, |_| false)
403}
404
405/// [`gh_retrying`], except that a failure `answered` holds is returned at the first attempt,
406/// as a refusal is, for a caller that has a better question to ask than the same one again.
407fn gh_retrying_unless(
408    root: &Path,
409    args: &[&str],
410    answered: fn(&str) -> bool,
411) -> Result<(bool, String, String), ForgeError> {
412    super::retry::forge(|| {
413        let (ok, out, err) = gh(root, args).map_err(|e| e.0)?;
414        if !ok && !answered(&err) && super::retry::transient(&err) {
415            Err(format!("gh {} failed: {}", args.join(" "), err.trim()))
416        } else {
417            Ok((ok, out, err))
418        }
419    })
420    .map_err(ForgeError)
421}

423/// Whether a failure is the forge giving up on a query before it was answered: the answer a
424/// page too large to compute in time gets, again, on every attempt. It comes in two forms. The
425/// gateway's HTTP 504; and GraphQL's own, an HTTP 200 whose error `gh` prints as "GraphQL:
426/// Something went wrong while executing your query. This may be the result of a timeout, or
427/// it could be a GitHub bug." `gh pr list` asks a hundred pull requests a page with their
428/// whole rollups, and on master's pages runs 40796031540, 40803295553 and 40823254353 all
429/// four attempts at that page were 504 (I4380); a smaller page is asked instead of the same
430/// page four times. [`super::retry::transient`] reads both as an outage (by "timeout"), so
431/// without this every one of them was the same question asked four times.
432///
433/// ```text
434/// gateway_timed_out("HTTP 504: 504 Gateway Timeout (https://api.github.com/graphql)") == true
435/// gateway_timed_out("GraphQL: Something went wrong while executing your query. \
436///     This may be the result of a timeout, or it could be a GitHub bug.") == true
437/// gateway_timed_out("HTTP 502: Bad Gateway") == false
438/// ```
439pub fn gateway_timed_out(message: &str) -> bool {
440    let m = message.to_ascii_lowercase();
441    m.contains("http 504")
442        || m.contains("504 gateway time")
443        || m.contains("may be the result of a timeout")
444}

446/// Whether the answer to `gh pr list`'s hundred-a-page read is one a smaller page answers: the
447/// forge giving up on the query ([`gateway_timed_out`]), or a body cut off on its way, which
448/// `gh` reports as "unexpected end of JSON input". An overloaded page of a hundred with their
449/// whole rollups came back cut off on pages runs 120850143313, 120870713946, 120881119536 and
450/// 120966409314 (2026-10-24), and that page was asked four times and never read in smaller
451/// ones; it is now read again in pages of [`OPEN_PAGE`] (I4380). Every other read keeps a
452/// cut-off body an outage and asks the same question again (I4222).
453///
454/// ```text
455/// list_needs_smaller_pages("unexpected end of JSON input") == true
456/// list_needs_smaller_pages("HTTP 504: 504 Gateway Timeout") == true
457/// list_needs_smaller_pages("HTTP 502: Bad Gateway") == false
458/// ```
459pub fn list_needs_smaller_pages(message: &str) -> bool {
460    gateway_timed_out(message)
461        || message
462            .to_ascii_lowercase()
463            .contains("unexpected end of json input")
464}

466/// The merge methods a repository's settings allow, from `GET repos/{repository}`, in the
467/// forge's words; `None` when the answer carries none of `allow_merge_commit`,
468/// `allow_squash_merge` and `allow_rebase_merge`. The forge leaves them out for a token that
469/// may not read them (the Pages workflow's), and the snapshot it published read that as
470/// "allows no merge commit (it allows: nothing)" while all three were allowed (I2966).
471fn merge_methods_of(settings: &Value) -> Option<Vec<String>> {
472    let mut read = false;
473    let mut methods = Vec::new();
474    for (key, word) in [
475        ("allow_merge_commit", "merge"),
476        ("allow_squash_merge", "squash"),
477        ("allow_rebase_merge", "rebase"),
478    ] {
479        if let Some(allowed) = settings.get(key).and_then(Value::as_bool) {
480            read = true;
481            if allowed {
482                methods.push(word.to_string());
483            }
484        }
485    }
486    read.then_some(methods)
487}

506/// The state word of one rollup entry: a check run (`status`/`conclusion`) or a commit
507/// status context (`state`).
508pub fn check_state(entry: &Value) -> CheckRunState {
509    let s = |k: &str| {
510        entry
511            .get(k)
512            .and_then(Value::as_str)
513            .unwrap_or("")
514            .to_ascii_uppercase()
515    };
516    if entry.get("__typename").and_then(Value::as_str) == Some("StatusContext") {
517        return match s("state").as_str() {
518            "SUCCESS" => CheckRunState::Passed,
519            "PENDING" | "EXPECTED" => CheckRunState::Pending,
520            _ => CheckRunState::Failed,
521        };
522    }
523    if s("status") != "COMPLETED" {
524        return CheckRunState::Pending;
525    }
526    match s("conclusion").as_str() {
527        "SUCCESS" => CheckRunState::Passed,
528        "SKIPPED" | "NEUTRAL" => CheckRunState::Skipped,
529        _ => CheckRunState::Failed,
530    }
531}

533/// One check from a rollup entry: `gh pr list --json statusCheckRollup` output, or a node of
534/// the GraphQL rollup [`WRITERS_QUERY`] reads. The two have one shape, but for the app: only
535/// the GraphQL node carries `checkSuite.app.databaseId`, so a check run read from `gh pr list`
536/// names no app, and neither does a status context, which no app writes.
537///
538/// The module is private, so the example is text; the unit tests run the same assertions.
539///
540/// ```text
541/// use majordomus_cli::integration::forge::check_of;
542/// let run = serde_json::json!({"__typename": "CheckRun", "name": "ci", "status": "COMPLETED",
543///     "conclusion": "SUCCESS", "completedAt": "2026-09-01T00:05:00Z",
544///     "checkSuite": {"app": {"databaseId": 15368}}});
545/// assert_eq!(check_of(&run).app_id, Some(15368));
546/// let listed = serde_json::json!({"__typename": "CheckRun", "name": "ci", "status": "COMPLETED"});
547/// assert_eq!(check_of(&listed).app_id, None, "gh pr list does not say who wrote it");
548/// ```
549pub fn check_of(entry: &Value) -> CheckObservation {
550    let text = |k: &str| entry.get(k).and_then(Value::as_str).unwrap_or("");
551    let kind = if text("__typename") == "StatusContext" {
552        CheckKind::StatusContext
553    } else {
554        CheckKind::CheckRun
555    };
556    CheckObservation {
557        name: entry
558            .get("name")
559            .or_else(|| entry.get("context"))
560            .and_then(Value::as_str)
561            .unwrap_or("")
562            .to_string(),
563        state: check_state(entry),
564        kind,
565        // the app of the suite the run belongs to: the one writer a forge read names
566        app_id: entry
567            .pointer("/checkSuite/app/databaseId")
568            .and_then(Value::as_u64),
569        completed_at: match text("completedAt") {
570            // a check run's zero time means it has not completed
571            "" | "0001-01-01T00:00:00Z" => text("startedAt")
572                .trim_start_matches("0001-01-01T00:00:00Z")
573                .to_string(),
574            t => t.to_string(),
575        },
576        // a status context has no run to time, and a run's zero time means it has not started
577        started_at: match (kind, text("startedAt")) {
578            (CheckKind::StatusContext, _) | (_, "0001-01-01T00:00:00Z") => String::new(),
579            (_, t) => t.to_string(),
580        },
581    }
582}

671/// Union two readings of the required checks: a context required by either is required,
672/// once for each app it is bound to. Two sources that bind one context to two apps require
673/// both apps' check runs, as the forge does; an unbound entry of a context is the same
674/// requirement as a bound one and is dropped beside it. In context order, then app order.
675fn union_checks(a: Vec<RequiredCheck>, b: Vec<RequiredCheck>) -> Vec<RequiredCheck> {
676    let all: BTreeSet<RequiredCheck> = a.into_iter().chain(b).collect();
677    let bound: BTreeSet<String> = all
678        .iter()
679        .filter(|c| c.app_id.is_some())
680        .map(|c| c.context.clone())
681        .collect();
682    all.into_iter()
683        .filter(|c| c.app_id.is_some() || !bound.contains(&c.context))
684        .collect()
685}
686
687/// Union two review requirements: the stricter of each.
688fn union_reviews(a: ReviewPolicy, b: ReviewPolicy) -> ReviewPolicy {
689    ReviewPolicy {
690        approvals: a.approvals.max(b.approvals),
691        code_owners: a.code_owners || b.code_owners,
692        dismiss_stale: a.dismiss_stale || b.dismiss_stale,
693    }
694}

696/// The app a required check is bound to, as the forge spelled it: `Some(None)` for a check
697/// bound to none (the field absent, `null`, or `-1`, the forge's word for any app),
698/// `Some(Some(id))` for a bound one, and `None` for a spelling that is neither — a string, a
699/// fraction, another negative number.
700///
701/// Unreadable is not unbound. Read as unbound, a binding the forge spelled another way would
702/// let any app's run of that name stand for the bound app's, which is the hole the binding
703/// exists to close; so a document that carries one is a requirement that was not read.
704fn bound_app(v: Option<&Value>) -> Option<Option<u64>> {
705    match v {
706        None | Some(Value::Null) => Some(None),
707        Some(Value::Number(n)) if n.as_i64() == Some(-1) => Some(None),
708        Some(Value::Number(n)) => n.as_u64().map(Some),
709        Some(_) => None,
710    }
711}

725pub fn rules_bind_readably(v: &Value) -> bool {

739/// The required checks and review requirement the rulesets that apply to a branch add,
740/// from `repos/{r}/rules/branches/{base}`: its `required_status_checks` rules (a context and
741/// the `integration_id` bound to it) and its `pull_request` rules.
742pub fn rules_of(v: &Value) -> (Vec<RequiredCheck>, ReviewPolicy) {
743    let mut checks = Vec::new();
744    let mut reviews = ReviewPolicy::default();
745    for rule in v.as_array().into_iter().flatten() {
746        let params = rule.get("parameters");
747        match rule.get("type").and_then(Value::as_str) {
748            Some("required_status_checks") => {
749                for c in params
750                    .and_then(|p| p.get("required_status_checks"))
751                    .and_then(Value::as_array)
752                    .into_iter()
753                    .flatten()
754                {
755                    if let Some(context) = c.get("context").and_then(Value::as_str) {
756                        checks.push(RequiredCheck {
757                            context: context.to_string(),
758                            app_id: bound_app(c.get("integration_id")).flatten(),
759                        });
760                    }
761                }
762            }
763            Some("pull_request") => {
764                let p = |k: &str| params.and_then(|p| p.get(k));
765                reviews = union_reviews(
766                    reviews,
767                    ReviewPolicy {
768                        approvals: p("required_approving_review_count")
769                            .and_then(Value::as_u64)
770                            .unwrap_or(0),
771                        code_owners: p("require_code_owner_review")
772                            .and_then(Value::as_bool)
773                            .unwrap_or(false),
774                        dismiss_stale: p("dismiss_stale_reviews_on_push")
775                            .and_then(Value::as_bool)
776                            .unwrap_or(false),
777                    },
778                );
779            }
780            _ => {}
781        }
782    }
783    (union_checks(checks, Vec::new()), reviews)
784}

805/// context the protection lists under `contexts` and again under `checks` is one check,
806/// bound to the app `checks` names for it.
807pub fn protection_of(v: &Value) -> (Vec<RequiredCheck>, ReviewPolicy) {

809        .pointer("/required_status_checks/contexts")
810        .and_then(Value::as_array)
811        .map(|a| {

827                })

831    let reviews = match v.get("required_pull_request_reviews") {

845                .unwrap_or(false),
846        },
847        None => ReviewPolicy::default(),
848    };
849    (union_checks(bound, listed), reviews)
850}

892/// One page of one pull request's rollup, as a writers answer gave it.
893#[derive(Debug, Clone, PartialEq, Eq)]
894pub struct RollupPage {
895    /// The head the rollup is of.
896    pub head: String,
897    /// The checks on this page, each with the app that wrote it where the forge named one.
898    pub checks: Vec<CheckObservation>,
899    /// Whether these are all of them: `false` when the forge says more contexts follow, and
900    /// when it listed contexts without saying whether more do.
901    pub complete: bool,
902    /// Where the next page of contexts starts, when the forge named a cursor.
903    pub cursor: Option<String>,
904}

906/// One pull request node of a writers answer, read as a [`RollupPage`]. A head without a
907/// rollup (`statusCheckRollup: null`) has no check, and that is all of them.
908///
909/// The module is private, so the example is text; the unit tests run the same assertions.
910///
911/// ```text
912/// use majordomus_cli::integration::forge::rollup_page_of;
913/// let node = serde_json::json!({"number": 1, "headRefOid": "h1", "statusCheckRollup": {"contexts": {
914///     "pageInfo": {"hasNextPage": true, "endCursor": "c1"},
915///     "nodes": [{"__typename": "CheckRun", "name": "ci", "status": "COMPLETED",
916///                "conclusion": "SUCCESS", "checkSuite": {"app": {"databaseId": 15368}}}]}}});
917/// let page = rollup_page_of(&node);
918/// assert_eq!((page.head.as_str(), page.checks[0].app_id), ("h1", Some(15368)));
919/// assert_eq!((page.complete, page.cursor.as_deref()), (false, Some("c1")));
920/// assert!(rollup_page_of(&serde_json::json!({"headRefOid": "h1", "statusCheckRollup": null})).complete);
921/// ```
922pub fn rollup_page_of(node: &Value) -> RollupPage {
923    let contexts = node.pointer("/statusCheckRollup/contexts");
924    let info = |k: &str| contexts.and_then(|c| c.pointer(k));
925    RollupPage {
926        head: node
927            .get("headRefOid")
928            .and_then(Value::as_str)
929            .unwrap_or("")
930            .to_string(),
931        checks: contexts
932            .and_then(|c| c.get("nodes"))
933            .and_then(Value::as_array)
934            .map(|a| a.iter().map(check_of).collect())
935            .unwrap_or_default(),
936        complete: contexts.is_none()
937            || info("/pageInfo/hasNextPage").and_then(Value::as_bool) == Some(false),
938        cursor: info("/pageInfo/endCursor")
939            .and_then(Value::as_str)
940            .map(str::to_string),
941    }
942}

1044/// let (head, checks) = whole.expect("two pages");
1045/// assert_eq!((head.as_str(), checks.len()), ("h1", 2));
1046/// ```
1047pub fn whole_rollup(
1048    mut ask: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1049) -> Result<Option<(String, Vec<CheckObservation>)>, ForgeError> {
1050    let mut read: Option<(String, Vec<CheckObservation>)> = None;
1051    let mut cursor: Option<String> = None;
1052    for _ in 0..CONTEXT_PAGES {
1053        let answer = ask(cursor.as_deref())?;
1054        let page = rollup_page_of(
1055            answer
1056                .pointer("/data/repository/pullRequest")
1057                .unwrap_or(&Value::Null),
1058        );
1059        let complete = page.complete;
1060        let (head, mut checks) = read
1061            .take()
1062            .unwrap_or_else(|| (page.head.clone(), Vec::new()));
1063        // a rollup is of one head: pages of two heads are not one rollup
1064        let same_head = head == page.head;
1065        checks.extend(page.checks);
1066        if complete || !same_head || page.cursor.is_none() {
1067            return Ok((complete && same_head).then_some((head, checks)));
1068        }
1069        read = Some((head, checks));
1070        cursor = page.cursor;
1071    }
1072    Ok(None)
1073}

1075/// The checks of the open pull requests `wanted` names, each read with the app that wrote
1076/// it. `page` answers one [`WRITERS_QUERY`] page after a cursor (`None` first); pages are
1077/// asked for until every wanted pull request was seen, the forge has no more, or `limit`
1078/// pull requests' worth of pages were asked for. A wanted one whose contexts did not fit its
1079/// page is read again whole with `rest` ([`whole_rollup()`]: its number, then the cursor). One
1080/// that stays unread is absent from the answer, and [`attribute()`] leaves it unattributed. A
1081/// read that fails fails the whole: a writer is read or the observation is not made.
1082///
1083/// The module is private, so the example is text; the unit tests run the same assertions.
1084///
1085/// ```text
1086/// use majordomus_cli::integration::forge::attributed_checks;
1087/// let page = serde_json::json!({"data": {"repository": {"pullRequests": {
1088///   "pageInfo": {"hasNextPage": false}, "nodes": [
1089///     {"number": 1, "headRefOid": "h1", "statusCheckRollup": null}]}}}});
1090/// let wanted = [1].into_iter().collect();
1091/// let read = attributed_checks(&wanted, 500, |_| Ok(page.clone()), |_, _| unreachable!()).unwrap();
1092/// assert_eq!(read[&1], ("h1".to_string(), vec![]));
1093/// ```
1094pub fn attributed_checks(
1095    wanted: &BTreeSet<u64>,
1096    limit: usize,
1097    mut page: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1098    mut rest: impl FnMut(u64, Option<&str>) -> Result<Value, ForgeError>,
1099) -> Result<AttributedChecks, ForgeError> {
1100    let mut read = AttributedChecks::new();
1101    let mut truncated: BTreeSet<u64> = BTreeSet::new();
1102    let mut cursor: Option<String> = None;
1103    for _ in 0..limit.div_ceil(WRITERS_PAGE) {
1104        if wanted
1105            .iter()
1106            .all(|n| read.contains_key(n) || truncated.contains(n))
1107        {
1108            break;
1109        }
1110        let answer = page(cursor.as_deref())?;
1111        read.extend(attributed_checks_of(&answer));
1112        truncated.extend(truncated_of(&answer));
1113        cursor = next_cursor(&answer);
1114        if cursor.is_none() {
1115            break;
1116        }
1117    }
1118    for number in truncated.intersection(wanted) {
1119        let whole = whole_rollup(|after| rest(*number, after))?;
1120        read.extend(whole.map(|rollup| (*number, rollup)));
1121    }
1122    Ok(read)
1123}

1125/// Give each observed pull request the checks read with their writers. Only a read of the

1127/// head (it moved between the two reads), keeps the checks `gh pr list` gave it, none of
1128/// which names an app, so a context bound to an app is `unknown` on it and never passed. A
1129/// head that was not observed is never matched, whatever the read says of it.

1146            .map(|(_, checks)| checks.clone());

1149        }
1150    }

1286/// let listed = open_pull_requests(500, |_| Ok(page.clone())).unwrap();
1287/// assert_eq!(listed[0]["number"], 1);
1288/// ```
1289pub fn open_pull_requests(
1290    limit: usize,
1291    mut page: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1292) -> Result<Vec<Value>, ForgeError> {
1293    let mut listed = Vec::new();
1294    let mut cursor: Option<String> = None;
1295    for _ in 0..limit.div_ceil(OPEN_PAGE) {
1296        let answer = page(cursor.as_deref())?;
1297        let nodes = answer
1298            .pointer("/data/repository/pullRequests/nodes")
1299            .and_then(Value::as_array)
1300            .ok_or_else(|| {
1301                ForgeError("the forge's page of open pull requests lists none".into())
1302            })?;
1303        listed.extend(nodes.iter().map(open_node_of));
1304        cursor = next_cursor(&answer);
1305        if cursor.is_none() {
1306            break;
1307        }
1308    }
1309    listed.truncate(limit);
1310    Ok(listed)
1311}

1491/// Every cross-reference to one pull request, read a page at a time with
1492/// [`DECLARATIONS_OF_QUERY`]: `ask` answers the page after a cursor (`None` first). Whole when
1493/// the forge says no more follow; truncated, with no source, when a page says more follow and
1494/// names no cursor, or [`REFERENCE_PAGES`] pages were not enough. `None` when the forge no
1495/// longer shows the pull request: nothing was read of it, and unread is never a shorter list.
1496/// A page the forge would not answer is the error it gave: a failed read is not a hold.
1497fn whole_references(
1498    mut ask: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1499) -> Result<Option<DeclarationRead>, ForgeError> {
1500    let mut sources: Vec<Value> = Vec::new();
1501    let mut cursor: Option<String> = None;
1502    let mut last: Option<ReferencePage> = None;
1503    for _ in 0..REFERENCE_PAGES {
1504        let answer = ask(cursor.as_deref())?;
1505        let Some(node) = answer
1506            .pointer("/data/repository/pullRequest")
1507            .filter(|node| node.is_object())
1508        else {
1509            return Ok(None);
1510        };
1511        let mut page = reference_page_of(node);
1512        sources.append(&mut page.sources);
1513        cursor = page.cursor.clone();
1514        let stop = page.complete || cursor.is_none();
1515        last = Some(page);
1516        if stop {
1517            break;
1518        }
1519    }
1520    // the last page read says whether more follow: one that does not is the whole
1521    Ok(last.map(|page| DeclarationRead::of(page, sources)))
1522}

1524/// What is declared about the open pull requests `wanted` names, and by whom. `page` answers
1525/// one [`DECLARATIONS_QUERY`] page after a cursor (`None` first); pages are asked for until
1526/// every wanted pull request was seen, the forge has no more, or `limit` pull requests' worth
1527/// of pages were asked for. A wanted one that was not read whole by then — its
1528/// cross-references did not fit its page, or no page listed it — is asked for alone with
1529/// `rest` (its number, then the cursor), [`REFERENCE_PAGES`] pages at most; one that still
1530/// does not fit is truncated, with no source at all, and held.
1531///
1532/// The two ways a read falls short are kept apart. A truncated read is that pull request's:
1533/// it is held and the others are decided. A read the forge would not answer — a page, or one
1534/// pull request's references — is an error, and the caller makes no observation from it:
1535/// what was not read is never an observation with a part missing. One the forge no longer
1536/// shows when asked for alone is unread — with the association a page gave it, or else absent
1537/// from the answer, which [`declare()`] leaves unread too.
1538///
1539/// The module is private, so the example is text; the unit tests run the same assertions.
1540///
1541/// ```text
1542/// use majordomus_cli::integration::forge::declarations;
1543/// use majordomus_cli::integration::CrossReferenceRead;
1544/// let page = serde_json::json!({"data": {"repository": {"pullRequests": {
1545///   "pageInfo": {"hasNextPage": false}, "nodes": [
1546///     {"number": 1, "authorAssociation": "OWNER", "isCrossRepository": false, "timelineItems": {
1547///        "pageInfo": {"hasNextPage": false}, "nodes": []}}]}}}});
1548/// let wanted = [1, 2].into_iter().collect();
1549/// let gone = |_: u64, _: Option<&str>| Ok(serde_json::json!({"data": {"repository": {"pullRequest": null}}}));
1550/// let read = declarations(&wanted, 500, |_| Ok(page.clone()), gone).unwrap();
1551/// assert_eq!((read[&1].association.as_str(), read[&1].read), ("OWNER", CrossReferenceRead::Whole));
1552/// assert!(!read.contains_key(&2), "one the forge no longer shows is unread, and #1 still is read");
1553/// let refused = |_: u64, _: Option<&str>| Err(ForgeError("HTTP 403".into()));
1554/// assert!(declarations(&wanted, 500, |_| Ok(page.clone()), refused).is_err(), "a failed read is an error");
1555/// ```
1556pub fn declarations(
1557    wanted: &BTreeSet<u64>,
1558    limit: usize,
1559    mut page: impl FnMut(Option<&str>) -> Result<Value, ForgeError>,
1560    mut rest: impl FnMut(u64, Option<&str>) -> Result<Value, ForgeError>,
1561) -> Result<Declarations, ForgeError> {
1562    let mut read = Declarations::new();
1563    let mut cursor: Option<String> = None;
1564    for _ in 0..limit.div_ceil(DECLARATIONS_PAGE) {
1565        if wanted.iter().all(|n| read.contains_key(n)) {
1566            break;
1567        }
1568        let answer = page(cursor.as_deref())?;
1569        read.extend(declarations_of(&answer));
1570        cursor = next_cursor(&answer);
1571        if cursor.is_none() {
1572            break;
1573        }
1574    }
1575    let whole = |read: &Declarations, n: &u64| {
1576        read.get(n)
1577            .is_some_and(|d| d.read == CrossReferenceRead::Whole)
1578    };
1579    let alone: Vec<u64> = wanted
1580        .iter()
1581        .copied()
1582        .filter(|n| !whole(&read, n))
1583        .collect();
1584    for number in alone {
1585        let entry = whole_references(|after| rest(number, after))?;
1586        // one the forge no longer shows is not a truncation: what a page said of its author
1587        // is kept, and its references are unread
1588        if let Some(kept) = read.get_mut(&number).filter(|_| entry.is_none()) {
1589            kept.read = CrossReferenceRead::Unread;
1590        }
1591        read.extend(entry.map(|entry| (number, entry)));
1592    }
1593    Ok(read)
1594}

1596/// Give each observed pull request what the declarations read said of it: its author's
1597/// association, and whether its cross-references were read whole. One the read does not name
1598/// keeps no association and `unread`, so it authorises nothing and is held. Returns the
1599/// sources of every named pull request's cross-references, in number order, as one array:
1600/// what [`resolved_for()`] reads the closed declarers from.
1601///
1602/// Where a head lives is half of who may declare, and `gh pr list` reads an absent answer as
1603/// "this repository". So an association is recorded only for a pull request this read placed:
1604/// one it says lives here (`Some(false)`), or one the list already calls a fork, which
1605/// declares nothing whatever its association. For one this read did not place, or placed in a
1606/// fork the list did not see, no association is recorded, and it authorises nothing. The
1607/// list's own flag is left as it was read.
1608///
1609/// A source that is open, says it supersedes the pull request, and is not among the listed
1610/// ones makes that pull request's references `unread`, whatever the read said: an open
1611/// successor the list did not carry is one nothing else would hold it for.
1612///
1613/// The module is private, so the example is text; the unit tests run the same assertions.
1614///
1615/// ```text
1616/// use majordomus_cli::integration::forge::{declare, pull_request_of, DeclarationRead, Declarations};
1617/// use majordomus_cli::integration::CrossReferenceRead;
1618/// let mut prs = vec![pull_request_of(&serde_json::json!({"number": 1})).unwrap(),
1619///                    pull_request_of(&serde_json::json!({"number": 2})).unwrap()];
1620/// let mut read = Declarations::new();
1621/// read.insert(1, DeclarationRead { association: "OWNER".into(), cross_repository: Some(false),
1622///     sources: vec![serde_json::json!({"number": 7})], read: CrossReferenceRead::Whole });
1623/// let sources = declare(&mut prs, &read);
1624/// assert_eq!(sources, serde_json::json!([{"number": 7}]));
1625/// assert_eq!(prs[0].cross_references, CrossReferenceRead::Whole);
1626/// assert_eq!(prs[1].cross_references, CrossReferenceRead::Unread, "not named: held");
1627/// ```
1628pub fn declare(prs: &mut [PullRequestObservation], read: &Declarations) -> Value {
1629    use super::classify::declared_supersessions;
1630    let listed: BTreeSet<u64> = prs.iter().map(|p| p.number).collect();
1631    let mut sources: BTreeMap<u64, &[Value]> = BTreeMap::new();
1632    for pr in prs.iter_mut() {
1633        if let Some(said) = read.get(&pr.number) {
1634            let placed = said.cross_repository == Some(false) || pr.cross_repository;
1635            pr.author_association = if placed {
1636                said.association.clone()
1637            } else {
1638                String::new()
1639            };
1640            // An open pull request that says it supersedes this one, and that the list of
1641            // open ones does not carry: the list was cut at its limit, or it was opened
1642            // between the two reads. Nothing else reads an open one's body, so its
1643            // declaration would hold nothing and this one could merge beside its successor.
1644            // What is known is that the list is not all that is open: unread, and held.
1645            let number = pr.number;
1646            let unlisted = said.sources.iter().any(|source| {
1647                source.get("state").and_then(Value::as_str) == Some("OPEN")
1648                    && source
1649                        .get("number")
1650                        .and_then(Value::as_u64)
1651                        .is_some_and(|m| m != number && !listed.contains(&m))
1652                    && source
1653                        .get("body")
1654                        .and_then(Value::as_str)
1655                        .is_none_or(|body| {
1656                            declared_supersessions(body).supersedes.contains(&number)
1657                        })
1658            });
1659            pr.cross_references = if unlisted {
1660                CrossReferenceRead::Unread
1661            } else {
1662                said.read
1663            };
1664            sources.insert(pr.number, &said.sources);
1665        }
1666    }
1667    Value::Array(sources.into_values().flatten().cloned().collect())
1668}

1674            root,
1675            &["repo", "view", "--json", "nameWithOwner,defaultBranchRef"],

1683            .pointer("/defaultBranchRef/name")

1693            root,
1694            &[
1695                "api",
1696                &format!("repos/{repository}/commits/{base}"),
1697                "--jq",
1698                "{sha: .sha}",
1699            ],
1700        )?
1701        .get("sha")
1702        .and_then(Value::as_str)
1703        .unwrap_or("")
1704        .to_string();

1706        // is a reading, not a failure; any other refusal leaves the requirement unread
1707        let (ok, out, err) = gh_retrying(
1708            root,
1709            &[
1710                "api",
1711                &format!("repos/{repository}/branches/{base}/protection"),
1712            ],
1713        )?;
1714        let protection: Option<(Vec<RequiredCheck>, ReviewPolicy, bool)> = if ok {
1715            // a binding the forge spelled unreadably leaves the requirement unread, as a
1716            // document that is not JSON does

1718                .ok()
1719                .filter(protection_binds_readably)
1720                .map(|v| {
1721                    let (checks, reviews) = protection_of(&v);
1722                    (checks, reviews, protection_requires_up_to_date(&v))
1723                })
1724        } else if err.contains("Branch not protected") || out.contains("Branch not protected") {
1725            Some((Vec::new(), ReviewPolicy::default(), false))
1726        } else {
1727            None
1728        };
1729        // the rulesets that apply to the base add requirements the protection does not list;
1730        // a ruleset read that fails leaves the requirement unread, as an unread protection does
1731        let (ok, out, _) = gh_retrying(
1732            root,
1733            &["api", &format!("repos/{repository}/rules/branches/{base}")],
1734        )?;
1735        let rules: Option<(Vec<RequiredCheck>, ReviewPolicy, bool)> = if ok {
1736            serde_json::from_str::<Value>(&out)
1737                .ok()
1738                .filter(rules_bind_readably)
1739                .map(|v| {
1740                    let (checks, reviews) = rules_of(&v);
1741                    (checks, reviews, rules_require_up_to_date(&v))
1742                })
1743        } else {
1744            None
1745        };

1747            (Some((pc, pr, ps)), Some((rc, rr, rs))) => (
1748                Some(union_checks(pc, rc)),
1749                Some(union_reviews(pr, rr)),
1750                Some(ps || rs),
1751            ),
1752            _ => (None, None, None),
1753        };
1754        let (owner, name) = repository
1755            .split_once('/')
1756            .unwrap_or((repository.as_str(), ""));
1757        let ask = |query: &str, typed: &str, after: Option<&str>| {
1758            gh_graphql(root, &writers_args(query, owner, name, typed, after))
1759        };
1760        let limit = OPEN_LIMIT.to_string();

1762            "pr",
1763            "list",
1764            "--state",
1765            "open",
1766            "--limit",
1767            &limit,
1768            "--json",
1769            "number,title,author,headRefName,headRefOid,baseRefName,isDraft,labels,createdAt,updatedAt,body,statusCheckRollup,reviewDecision,latestReviews,reviewRequests,autoMergeRequest,isCrossRepository",
1770        ];
1771        // `gh pr list` in its pages of a hundred, the one read every scripted forge answers;
1772        // a page the forge times out, or whose body is cut off on its way, is read again in
1773        // pages of OPEN_PAGE, each asked again on an outage as every read is, rather than the
1774        // same page four times
1775        let listed = gh_retrying_unless(root, &list_args, list_needs_smaller_pages)?;
1776        let list = open_list(&list_args, listed, &ask)?;
1777        // keyed by number, so the observation is in number order whatever the forge listed
1778        let mut pull_requests: Vec<PullRequestObservation> = list
1779            .as_array()
1780            .map(|a| {
1781                a.iter()
1782                    .filter_map(pull_request_of)
1783                    .map(|p| (p.number, p))
1784                    .collect::<std::collections::BTreeMap<_, _>>()
1785                    .into_values()
1786                    .collect()
1787            })
1788            .unwrap_or_default();
1789        // a context bound to an app is only that app's check run, and the list above does not
1790        // say who wrote one: the writers are read, or the observation is not made. A base that
1791        // binds nothing needs no writer and asks nothing more.
1792        let bound = required_checks
1793            .as_ref()
1794            .is_some_and(|r| r.iter().any(|c| c.app_id.is_some()));
1795        let wanted: BTreeSet<u64> = pull_requests.iter().map(|p| p.number).collect();

1797            let size = format!("n={WRITERS_PAGE}");

1799                &wanted,
1800                OPEN_LIMIT,
1801                |after| ask(WRITERS_QUERY, &size, after),

1804            attribute(&mut pull_requests, &read);
1805        }
1806        // who may declare a replacement, and what was declared: every open pull request is
1807        // read with its author's association and the pull requests that mention it. A
1808        // successor that landed is no longer listed among the open ones, and it is found
1809        // among the cross-references of the one it replaces, whenever it was closed: no
1810        // bounded search of closed pull requests decides. Nothing open, nothing asked; a read
1811        // that is partial holds the pull request it was about, and no other; a read that
1812        // fails makes no observation at all.
1813        let size = format!("n={DECLARATIONS_PAGE}");
1814        let read = declarations(
1815            &wanted,
1816            OPEN_LIMIT,
1817            |after| ask(DECLARATIONS_QUERY, &size, after),
1818            |number, after| ask(DECLARATIONS_OF_QUERY, &format!("number={number}"), after),
1819        )?;
1820        let sources = declare(&mut pull_requests, &read);
1821        let mut unreadable = None;
1822        let resolved = resolved_for(&pull_requests, &sources, |m| {
1823            match gh_retrying(
1824                root,
1825                &["pr", "view", &m.to_string(), "--json", RESOLVED_FIELDS],
1826            ) {
1827                Ok((true, out, _)) => serde_json::from_str(&out).ok(),
1828                // not a pull request, or refused: the successor is unread, never landed
1829                Ok(_) => None,
1830                Err(e) => {
1831                    unreadable.get_or_insert(e);
1832                    None
1833                }
1834            }
1835        });
1836        // an outage that outlasted the retries is a failed observation, as for the open list
1837        if let Some(e) = unreadable {
1838            return Err(e);
1839        }
1840        Ok(ForgeObservation {
1841            schema: OBSERVATION_SCHEMA,
1842            repository,
1843            base,
1844            base_sha,
1845            observed_at: crate::peers::rfc3339(std::time::SystemTime::now()),
1846            required_checks,
1847            review_policy,
1848            up_to_date_required,
1849            merge_methods,
1850            pull_requests,
1851            resolved,
1852            delete_branch_on_merge,
1853        })
1854    }

1947/// Remove the mirrors of pull requests this observation no longer names: merged, closed or
1948/// gone. Each one held the head the pull request had when it was last open, and a mirror
1949/// that is never refreshed again is only a stale answer waiting to be read. Best effort: a
1950/// mirror that cannot be removed is left, and nothing reads it.
1951fn prune_mirrors(root: &Path, obs: &ForgeObservation) {
1952    let keep: std::collections::BTreeSet<u64> = obs
1953        .pull_requests
1954        .iter()
1955        .map(|p| p.number)
1956        .chain(obs.resolved.keys().copied())
1957        .collect();
1958    // git that cannot run lists nothing, and nothing is removed
1959    let out = Command::new("git")
1960        .arg("-C")
1961        .arg(root)
1962        .args(["for-each-ref", "--format=%(refname)", PR_REF_PREFIX])
1963        .output()
1964        .map(|o| o.stdout)
1965        .unwrap_or_default();
1966    for r in String::from_utf8_lossy(&out).lines() {
1967        let stale = r
1968            .strip_prefix(PR_REF_PREFIX)
1969            .and_then(|n| n.parse::<u64>().ok())
1970            .is_some_and(|n| !keep.contains(&n));
1971        if stale {
1972            let _ = Command::new("git")
1973                .arg("-C")
1974                .arg(root)
1975                .args(["update-ref", "-d", r])
1976                .status();
1977        }
1978    }
1979}

1981/// Fetch the base and every observed head into this clone, the heads under
1982/// [`PR_REF_PREFIX`], and return the pull requests whose head the forge does not serve.
1983///
1984/// One `git fetch` when the forge serves everything. A single `refs/pull/<n>/head` it does
1985/// not serve fails that fetch for every ref in it, so on a failure `git ls-remote` asks which
1986/// of the heads origin serves and only those are fetched again. A head that cannot be fetched
1987/// stays unknown to the relation, which then says so, and its mirror is removed: the head it
1988/// held is an answer nobody can renew. The base is never optional — a base that cannot be
1989/// fetched fails the fetch, and so does a failure with every head served.
1990pub fn fetch(root: &Path, obs: &ForgeObservation) -> Result<Vec<u64>, ForgeError> {
1991    let base = format!("+refs/heads/{0}:refs/remotes/origin/{0}", obs.base);
1992    // a successor no longer open is fetched too, so git can say whether its head landed; the
1993    // forge keeps refs/pull/<n>/head for a closed pull request, and only one it read is here
1994    let numbers: BTreeSet<u64> = obs
1995        .pull_requests
1996        .iter()
1997        .map(|p| p.number)
1998        .chain(obs.resolved.keys().copied())
1999        .collect();
2000    prune_mirrors(root, obs);
2001    let refspecs = |heads: &mut dyn Iterator<Item = &u64>| -> Vec<String> {
2002        std::iter::once(base.clone())
2003            .chain(heads.map(|n| format!("+refs/pull/{n}/head:{PR_REF_PREFIX}{n}")))
2004            .collect()
2005    };
2006    let failed = match fetch_refspecs(root, &refspecs(&mut numbers.iter())) {
2007        Ok(()) => return Ok(Vec::new()),
2008        Err(e) => e,
2009    };
2010    let served = served_pull_heads(root, &numbers).map_err(|_| ForgeError(failed.clone()))?;
2011    let (kept, unserved): (Vec<u64>, Vec<u64>) = numbers.iter().partition(|n| served.contains(n));
2012    if unserved.is_empty() {
2013        // every head is there, so a missing one is not why the fetch failed
2014        return Err(ForgeError(failed));
2015    }
2016    for n in &unserved {
2017        let _ = Command::new("git")
2018            .arg("-C")
2019            .arg(root)
2020            .args(["update-ref", "-d", &format!("{PR_REF_PREFIX}{n}")])
2021            .status();
2022    }
2023    fetch_refspecs(root, &refspecs(&mut kept.iter())).map_err(ForgeError)?;
2024    Ok(unserved)
2025}

2027/// One `git fetch` of `refspecs` from origin. A fetch is a read: a dropped connection is
2028/// asked again, a refusal is not.
2029fn fetch_refspecs(root: &Path, refspecs: &[String]) -> Result<(), String> {
2030    super::retry::forge(|| {
2031        let out = Command::new("git")
2032            .arg("-C")
2033            .arg(root)
2034            .args(["fetch", "--quiet", "--no-tags", "origin"])
2035            .args(refspecs)
2036            .output()
2037            .map_err(|e| format!("git fetch could not run: {e}"))?;
2038        if out.status.success() {
2039            Ok(())

2049/// Which of `numbers` origin serves a `refs/pull/<n>/head` for, by `git ls-remote`.
2050fn served_pull_heads(root: &Path, numbers: &BTreeSet<u64>) -> Result<BTreeSet<u64>, String> {
2051    let wanted: Vec<String> = numbers
2052        .iter()
2053        .map(|n| format!("refs/pull/{n}/head"))
2054        .collect();
2055    super::retry::forge(|| {
2056        let mut ls = Command::new("git");
2057        ls.arg("-C")
2058            .arg(root)
2059            .args(["ls-remote", "origin"])
2060            .args(&wanted);
2061        output_of(ls, "git ls-remote").and_then(|out| stdout_of(out, "git ls-remote"))
2062    })
2063    .map(|listed| served_pull_heads_of(&listed))
2064}

2084/// The pull requests of `obs`, open and resolved, that have no mirror in this clone. After
2085/// [`fetch`] these are exactly the heads the forge does not serve: a served head is fetched
2086/// into its mirror, and an unserved one's mirror is removed.
2087pub fn unmirrored(root: &Path, obs: &ForgeObservation) -> Vec<u64> {
2088    let numbers: BTreeSet<u64> = obs
2089        .pull_requests
2090        .iter()
2091        .map(|p| p.number)
2092        .chain(obs.resolved.keys().copied())
2093        .collect();
2094    // git that cannot run lists nothing, and then every pull request is named
2095    let listed = Command::new("git")
2096        .arg("-C")
2097        .arg(root)
2098        .args(["for-each-ref", "--format=%(refname)", PR_REF_PREFIX])
2099        .output()
2100        .map(|o| o.stdout)
2101        .unwrap_or_default();
2102    let mirrored: BTreeSet<u64> = String::from_utf8_lossy(&listed)
2103        .lines()
2104        .filter_map(|r| r.strip_prefix(PR_REF_PREFIX).and_then(|n| n.parse().ok()))
2105        .collect();
2106    numbers.difference(&mirrored).copied().collect()
2107}

2110/// `refs/pull/<n>/head`: ls-remote matches a pattern by its tail, so a ref that only ends
2111/// in one is not the head.
2112fn served_pull_heads_of(listed: &str) -> BTreeSet<u64> {
2113    listed
2114        .lines()
2115        .filter_map(|l| l.split_once('\t').map(|(_, r)| r.trim()))
2116        .filter_map(|r| {
2117            r.strip_prefix("refs/pull/")?
2118                .strip_suffix("/head")?
2119                .parse()
2120                .ok()
2121        })
2122        .collect()
2123}
