1//! The bridge: a `majordomus mcp` process that found the repository's shared server
2//! already running forwards its client's stdio frames to that server's `/mcp` endpoint
3//! and writes the answers back, one HTTP request per message, and pings on the side so
4//! that the server keeps the session. It carries no index and no registry of its own,
5//! which is why it starts in milliseconds. The HTTP client below is the few lines a
6//! request needs (the server always answers with `Content-Length`, never chunked); no HTTP
7//! library is pulled in for it.
8//!
9//! The same client carries the mesh (ADR 0220, decision 1). An `https://` endpoint is
10//! reached over TLS — rustls, with the certificate verified against the roots compiled into
11//! the executable and the host name checked against it — and an `http://` endpoint only when
12//! every address its host resolves to is loopback, private, link-local or an overlay's
13//! (100.64.0.0/10, IPv6 unique-local): a public address is refused before a byte is sent,
14//! with the correction. Plain and TLS requests share one framing over one `Read + Write`.
15//!
16//! A message is sent a second time only when the server provably did not run it: the
17//! connection was never made, or the server answered 404 (the session is unknown) or 409
18//! (the lease is lost), both before it looks at the message. A message that was sent and
19//! never answered — a timeout, a dropped connection, a 5xx — may have run, so it is sent
20//! again only when it is idempotent: every method but `tools/call`, and a tool whose
21//! declared execution hints say so ([`idempotent`]). Anything else is the caller's to
22//! report as an outcome nobody knows ([`BridgeError::may_resend`]).

74/// How an endpoint's bytes travel, once [`transport`] has allowed it.
75///
76/// ```
77/// use majordomus_cli::mcp::bridge::{transport, Transport};
78/// // a hub by name over TLS; a seed on a private network in cleartext
79/// assert_eq!(transport("https://prismatic-majordomus.gigalixirapp.com").unwrap(), Transport::Tls);
80/// assert_eq!(transport("http://192.168.1.20:8742").unwrap(), Transport::Cleartext);
81/// assert_ne!(Transport::Tls, Transport::Cleartext);
82/// ```
83#[derive(Debug, Clone, Copy, PartialEq, Eq)]
84pub enum Transport {
85    /// `https://`: over TLS, the certificate and the host name verified.
86    Tls,
87    /// `http://` (or no scheme): in cleartext, to addresses that are not public.
88    Cleartext,
89}

91/// Whether cleartext may be sent to `ip`: loopback, RFC 1918 private, link-local, the
92/// 100.64.0.0/10 range an overlay (a tailnet) and carrier-grade NAT hand out, the unspecified
93/// address (which a connection reads as this machine), IPv6 unique-local (fc00::/7), and an
94/// IPv4-mapped IPv6 address by the IPv4 address it carries. Everything else is public.
95///
96/// ```
97/// use majordomus_cli::mcp::bridge::cleartext_allowed;
98/// assert!(cleartext_allowed("192.168.100.30".parse().unwrap()));
99/// assert!(cleartext_allowed("100.92.246.32".parse().unwrap()), "a tailnet address");
100/// assert!(cleartext_allowed("fd7a:115c:a1e0::1".parse().unwrap()), "unique-local");
101/// assert!(!cleartext_allowed("203.0.113.5".parse().unwrap()));
102/// assert!(!cleartext_allowed("2001:db8::1".parse().unwrap()));
103/// ```
104pub fn cleartext_allowed(ip: IpAddr) -> bool {
105    match ip {
106        IpAddr::V4(v4) => v4_cleartext_allowed(v4),
107        IpAddr::V6(v6) => {
108            if let Some(v4) = v6.to_ipv4_mapped() {
109                return v4_cleartext_allowed(v4);
110            }
111            let first = v6.segments()[0];
112            v6.is_loopback()
113                || v6.is_unspecified()
114                || (first & 0xfe00) == 0xfc00
115                || (first & 0xffc0) == 0xfe80
116        }
117    }
118}
119
120fn v4_cleartext_allowed(v4: Ipv4Addr) -> bool {
121    let [a, b, _, _] = v4.octets();
122    v4.is_loopback()
123        || v4.is_private()
124        || v4.is_link_local()
125        || v4.is_unspecified()
126        || (a == 100 && (b & 0xc0) == 64)
127}

147/// Split an authority (`host`, `host:port`, `[v6]` or `[v6]:port`) into the host — without
148/// the brackets — and the port it names, if any.
149pub(crate) fn split_authority(authority: &str) -> Result<(String, Option<u16>), String> {
150    let (host, port) = match authority.strip_prefix('[') {
151        Some(v6) => {
152            let (host, rest) = v6
153                .split_once(']')
154                .ok_or_else(|| format!("'{authority}' opens a bracket it does not close"))?;
155            match rest {
156                "" => (host, None),
157                _ => (
158                    host,
159                    Some(rest.strip_prefix(':').ok_or_else(|| {
160                        format!("'{authority}' has something other than a port after its address")
161                    })?),
162                ),
163            }
164        }
165        None => match authority.rsplit_once(':') {
166            Some((host, _)) if host.contains(':') => {
167                return Err(format!(
168                    "'{authority}' is an IPv6 address without brackets: write [address]:port"
169                ))
170            }
171            Some((host, port)) => (host, Some(port)),
172            None => (authority, None),
173        },
174    };
175    if host.is_empty() {
176        return Err(format!("'{authority}' names no host"));
177    }
178    let port = match port {
179        None => None,
180        Some(p) => Some(
181            p.parse::<u16>()
182                .ok()
183                .filter(|p| *p != 0)
184                .ok_or_else(|| format!("'{authority}' does not name a port from 1 to 65535"))?,
185        ),
186    };
187    Ok((host.to_string(), port))
188}

190/// Read `base_url` (`https://host[:port]`, `http://host[:port]`, or `host:port`, which is
191/// http) into an [`Endpoint`]. A path after the authority is not the endpoint's and is
192/// ignored, as it always was; any other scheme is refused.
193fn parse_endpoint(base_url: &str) -> std::io::Result<Endpoint> {
194    let invalid = |reason: String| std::io::Error::new(std::io::ErrorKind::InvalidInput, reason);
195    let (tls, rest) = if let Some(rest) = base_url.strip_prefix("https://") {
196        (true, rest)
197    } else if let Some(rest) = base_url.strip_prefix("http://") {
198        (false, rest)
199    } else if base_url.contains("://") {
200        return Err(invalid(format!(
201            "{base_url}: the scheme is neither http:// nor https://"
202        )));
203    } else {
204        (false, base_url)
205    };
206    let authority = rest.split('/').next().unwrap_or_default();
207    let (host, port) =
208        split_authority(authority).map_err(|e| invalid(format!("{base_url}: {e}")))?;
209    let port = port.unwrap_or(if tls { 443 } else { 80 });
210    Ok(Endpoint {
211        tls,
212        host,
213        port,
214        authority: authority.to_string(),
215    })
216}

218/// Every address `endpoint`'s host resolves to, in the resolver's order.
219fn resolve(endpoint: &Endpoint) -> std::io::Result<Vec<SocketAddr>> {
220    let addrs: Vec<SocketAddr> = (endpoint.host.as_str(), endpoint.port)
221        .to_socket_addrs()?
222        .collect();
223    some_address(addrs, &endpoint.authority)
224}
225
226/// `addrs`, unless the resolver answered with none: that is a failure naming `authority`, so
227/// that nothing indexes an empty answer.
228fn some_address(addrs: Vec<SocketAddr>, authority: &str) -> std::io::Result<Vec<SocketAddr>> {
229    if addrs.is_empty() {
230        return Err(std::io::Error::other(format!("{authority}: no address")));
231    }
232    Ok(addrs)
233}

235/// The refusal of cleartext to `endpoint`, when one of the addresses it resolves to is
236/// public: every one must be private for the request to be sent, because the resolver, not
237/// this client, picks which of them a later connection reaches.
238fn cleartext_refusal(endpoint: &Endpoint, addrs: &[SocketAddr]) -> Option<std::io::Error> {
239    let public = addrs.iter().find(|a| !cleartext_allowed(a.ip()))?;
240    Some(std::io::Error::new(
241        std::io::ErrorKind::PermissionDenied,
242        format!(
243            "{} was not dialed: {} is a public address, and the mesh sends nothing across a public network in cleartext; use https://{} (ADR 0220)",
244            endpoint.url(),
245            public.ip(),
246            endpoint.authority
247        ),
248    ))
249}

251/// How `base_url` would be reached, or why it would not be: [`Transport::Tls`] for an
252/// `https://` endpoint (nothing is resolved), [`Transport::Cleartext`] for an `http://` one
253/// whose every address is not public, and an error otherwise — `PermissionDenied` naming the
254/// public address, `InvalidInput` for an endpoint that cannot be read, or the resolver's own.
255/// The check [`request`] makes before it connects, offered to a self-check that dials nothing.
256///
257/// ```
258/// use majordomus_cli::mcp::bridge::{transport, Transport};
259/// assert_eq!(transport("https://hub.example:443").unwrap(), Transport::Tls);
260/// assert_eq!(transport("http://10.0.0.2:8742").unwrap(), Transport::Cleartext);
261/// let refused = transport("http://203.0.113.5:8791").unwrap_err();
262/// assert_eq!(refused.kind(), std::io::ErrorKind::PermissionDenied);
263/// assert!(refused.to_string().contains("203.0.113.5") && refused.to_string().contains("use https://"));
264/// ```
265pub fn transport(base_url: &str) -> std::io::Result<Transport> {
266    let endpoint = parse_endpoint(base_url)?;
267    if endpoint.tls {
268        return Ok(Transport::Tls);
269    }
270    let addrs = resolve(&endpoint)?;
271    match cleartext_refusal(&endpoint, &addrs) {
272        Some(refusal) => Err(refusal),
273        None => Ok(Transport::Cleartext),
274    }
275}

361/// which is what lets a caller send the same message again elsewhere.

366/// [`connect`] with the TLS configuration `tls` returns, asked for only when the endpoint is
367/// `https://`. The tests hand it one that trusts their own authority.
368fn connect_with(
369    base_url: &str,
370    timeout: Duration,
371    tls: impl FnOnce() -> std::io::Result<Arc<ClientConfig>>,
372) -> std::io::Result<(Stream, String)> {
373    let endpoint = parse_endpoint(base_url)?;
374    // the name the certificate is checked against, read before anything is resolved
375    let name = match endpoint.tls {
376        true => Some(server_name(&endpoint)?),
377        false => None,
378    };
379    let addrs = resolve(&endpoint)?;
380    if !endpoint.tls {
381        if let Some(refusal) = cleartext_refusal(&endpoint, &addrs) {
382            return Err(refusal);
383        }
384    }
385    // The connection shares the caller's budget when that is the shorter: a probe on a shell
386    // prompt that allows 250 ms must not wait 2 s on a server whose accept queue is full.
387    let socket = TcpStream::connect_timeout(&addrs[0], CONNECT_TIMEOUT.min(timeout))?;
388    let stream = match name {
389        Some(name) => Stream::Tls(Box::new(handshake(
390            socket,
391            &endpoint,
392            name,
393            tls()?,
394            timeout,
395        )?)),
396        None => Stream::Plain(socket),
397    };
398    Ok((stream, endpoint.authority))
399}

416/// Complete the TLS handshake on `socket` within `timeout` per read and write, so that a
417/// certificate that does not verify is refused here — naming the endpoint — before any byte
418/// of the request is written.
419fn handshake(
420    mut socket: TcpStream,
421    endpoint: &Endpoint,
422    name: ServerName<'static>,
423    config: Arc<ClientConfig>,
424    timeout: Duration,
425) -> std::io::Result<StreamOwned<ClientConnection, TcpStream>> {
426    bound(&socket, timeout)?;
427    let mut connection = ClientConnection::new(config, name).map_err(std::io::Error::other)?;
428    // `complete_io` reports a server that closes mid-handshake as `UnexpectedEof`, so the loop
429    // ends in a handshake or in an error, never in a spin
430    while connection.is_handshaking() {
431        connection.complete_io(&mut socket).map_err(|e| {
432            std::io::Error::new(
433                e.kind(),
434                format!("{}: the TLS handshake failed: {e}", endpoint.url()),
435            )
436        })?;
437    }
438    Ok(StreamOwned::new(connection, socket))
439}

441/// Bound every read and write on `socket` by `timeout`. A zero timeout is refused by the
442/// platform, and so here, rather than read as "no bound".
443fn bound(socket: &TcpStream, timeout: Duration) -> std::io::Result<()> {
444    socket.set_read_timeout(Some(timeout))?;
445    socket.set_write_timeout(Some(timeout))
446}

448/// Send one request on an open connection and read the reply. From the first byte written
449/// the server may have the request, so a failure here leaves its outcome unknown.

463/// The request and its reply over any stream: the one framing plain and TLS share.
464fn exchange_on<S: Read + Write>(
465    stream: &mut S,
466    host: &str,
467    method: &str,
468    path: &str,
469    headers: &[(&str, &str)],
470    body: Option<&str>,
471) -> std::io::Result<Reply> {
472    write_request(stream, host, method, path, headers, body)?;
473    let raw = read_reply(stream)?;
474    parse_reply(&raw)
475}

477/// Write one request: the head this client always sends, the caller's headers, the body.
478fn write_request<S: Write>(
479    stream: &mut S,
480    host: &str,
481    method: &str,
482    path: &str,
483    headers: &[(&str, &str)],
484    body: Option<&str>,
485) -> std::io::Result<()> {
486    let body = body.unwrap_or("");
487    let mut text = format!(
488        "{method} {path} HTTP/1.1\r\nHost: {host}\r\nConnection: close\r\nAccept: application/json\r\nContent-Type: application/json\r\nContent-Length: {}\r\n",
489        body.len()
490    );
491    for (k, v) in headers {
492        text.push_str(&format!("{k}: {v}\r\n"));
493    }
494    text.push_str("\r\n");
495    text.push_str(body);
496    stream.write_all(text.as_bytes())?;
497    stream.flush()
498}

500/// Read a reply to its end, bounded by [`MAX_REPLY_BYTES`]. A TLS server that closes the
501/// socket without saying so (no `close_notify`) ends the stream with `UnexpectedEof`; what it
502/// sent is the reply when its `Content-Length` is whole, and a truncation otherwise.
503fn read_reply<S: Read>(stream: &mut S) -> std::io::Result<Vec<u8>> {
504    let mut raw = Vec::new();
505    let mut chunk = vec![0u8; 64 * 1024];
506    loop {
507        match stream.read(&mut chunk) {
508            Ok(0) => break,
509            Ok(n) => {
510                raw.extend_from_slice(&chunk[..n]);
511                if raw.len() > MAX_REPLY_BYTES {
512                    return Err(std::io::Error::new(
513                        std::io::ErrorKind::InvalidData,
514                        format!(
515                            "the reply is over the {MAX_REPLY_BYTES}-byte bound and was not read"
516                        ),
517                    ));
518                }
519            }
520            Err(e) if e.kind() == std::io::ErrorKind::Interrupted => {}
521            Err(e) if e.kind() == std::io::ErrorKind::UnexpectedEof && whole(&raw) => break,
522            Err(e) => return Err(e),
523        }
524    }
525    Ok(raw)
526}

528/// Whether `raw` holds a whole reply: its head, and as many body bytes as its
529/// `Content-Length` says. Without a length nothing can say, and it is not whole.
530fn whole(raw: &[u8]) -> bool {
531    let Some(end) = raw.windows(4).position(|w| w == b"\r\n\r\n") else {
532        return false;
533    };
534    let head = String::from_utf8_lossy(&raw[..end]);
535    head.lines()
536        .filter_map(|l| l.split_once(':'))
537        .find(|(k, _)| k.trim().eq_ignore_ascii_case("content-length"))
538        .and_then(|(_, v)| v.trim().parse::<usize>().ok())
539        .is_some_and(|n| raw.len() - (end + 4) >= n)
540}

545/// What one reply read by [`request_json_within`] may cost: the bytes of it held, and the
546/// time it may take, between two reads and in all.
547///
548/// The two times are different defences. `idle` is what [`request`] bounds: a peer that
549/// goes quiet. `within` bounds a peer that never goes quiet — one byte every few seconds —
550/// which a bound between reads alone lets hold its caller for as long as it likes.
551///
552/// ```
553/// use std::time::Duration;
554/// use majordomus_cli::mcp::bridge::ReplyBound;
555///
556/// let bound = ReplyBound {
557///     bytes: 1024,
558///     idle: Duration::from_secs(5),
559///     within: Duration::from_secs(20),
560/// };
561/// assert!(bound.idle <= bound.within);
562/// ```
563#[derive(Debug, Clone, Copy, PartialEq, Eq)]
564pub struct ReplyBound {
565    /// The most bytes of the reply's JSON body held, without its insignificant whitespace.
566    pub bytes: usize,
567    /// The longest wait for the next bytes.
568    pub idle: Duration,
569    /// The longest wait for the whole reply, from the connection on.
570    pub within: Duration,
571}

573/// Why [`request_json_within`] stopped reading a reply: the typed half of the
574/// [`std::io::Error`] it returns, reached with [`ReplyRefused::of`]. Neither is a reply a
575/// caller can act on; both say the peer sent more, or more slowly, than it may (I2704).
576///
577/// ```
578/// use majordomus_cli::mcp::bridge::ReplyRefused;
579///
580/// let error = std::io::Error::new(
581///     std::io::ErrorKind::InvalidData,
582///     ReplyRefused::Oversized { bound: 10 },
583/// );
584/// assert_eq!(ReplyRefused::of(&error), Some(&ReplyRefused::Oversized { bound: 10 }));
585/// assert!(error.to_string().contains("10-byte bound"), "{error}");
586/// // an error of another cause is no refusal of the reply
587/// assert_eq!(ReplyRefused::of(&std::io::Error::other("refused")), None);
588/// ```
589#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
590pub enum ReplyRefused {
591    /// The reply's head, or its body without insignificant whitespace, is over the bound.
592    #[error("the reply is over the {bound}-byte bound and was not read")]
593    Oversized {
594        /// The bound it went over.
595        bound: usize,
596    },
597    /// The whole reply did not arrive within the time allowed for it.
598    #[error("the reply did not arrive whole within {millis} ms")]
599    Late {
600        /// The time allowed, in milliseconds.
601        millis: u64,
602    },
603}
604
605impl ReplyRefused {
606    /// The refusal an error of [`request_json_within`] carries, when that was its cause.
607    ///
608    /// ```
609    /// use majordomus_cli::mcp::bridge::ReplyRefused;
610    ///
611    /// let late = std::io::Error::new(
612    ///     std::io::ErrorKind::TimedOut,
613    ///     ReplyRefused::Late { millis: 250 },
614    /// );
615    /// assert_eq!(ReplyRefused::of(&late), Some(&ReplyRefused::Late { millis: 250 }));
616    /// // a timeout of another cause carries no refusal
617    /// let other = std::io::Error::from(std::io::ErrorKind::TimedOut);
618    /// assert_eq!(ReplyRefused::of(&other), None);
619    /// ```
620    pub fn of(error: &std::io::Error) -> Option<&ReplyRefused> {
621        error
622            .get_ref()
623            .and_then(|e| e.downcast_ref::<ReplyRefused>())
624    }
625
626    fn into_io(self) -> std::io::Error {
627        let kind = match self {
628            ReplyRefused::Oversized { .. } => std::io::ErrorKind::InvalidData,
629            ReplyRefused::Late { .. } => std::io::ErrorKind::TimedOut,
630        };
631        std::io::Error::new(kind, self)
632    }
633}

635/// One HTTP/1.1 request whose reply is a JSON document held at most at `bound.bytes` and
636/// read whole within `bound.within`: what a dialer asks of a peer it does not control.
637///
638/// The body is held without the whitespace JSON allows between tokens, which makes the
639/// bound the size of the document rather than of its formatting: the shared server
640/// pretty-prints every answer, and a peer that pads one costs time — bounded by `within` —
641/// never memory. The bound counts what is held, so the first reply refused is the one a
642/// byte over it, however it was formatted. The head is bounded too, at [`MAX_HEAD_BYTES`].
643/// A body that is not JSON loses its spaces: a caller that reads more than the status of
644/// a non-JSON answer uses [`request`].
645///
646/// ```
647/// use std::time::Duration;
648/// use majordomus_cli::mcp::bridge::{request_json_within, ReplyBound, ReplyRefused};
649///
650/// let bound = ReplyBound {
651///     bytes: 1024,
652///     idle: Duration::from_secs(1),
653///     within: Duration::from_secs(2),
654/// };
655/// // port 9 discards: nothing answers there, which is a failure to connect, not a refusal
656/// let error = request_json_within("http://127.0.0.1:9", "GET", "/", &[], None, bound)
657///     .unwrap_err();
658/// assert_eq!(ReplyRefused::of(&error), None);
659/// ```
660pub fn request_json_within(
661    base_url: &str,
662    method: &str,
663    path: &str,
664    headers: &[(&str, &str)],
665    body: Option<&str>,
666    bound: ReplyBound,
667) -> std::io::Result<Reply> {
668    let deadline = std::time::Instant::now() + bound.within;
669    let first = bound.idle.min(bound.within);
670    // connected and bounded in one step: a socket that cannot take its bound fails as a
671    // connection that could not be made
672    let (mut stream, host) = connect(base_url, first)
673        .and_then(|(stream, host)| self::bound(stream.socket(), first).map(|()| (stream, host)))?;
674    write_request(&mut stream, &host, method, path, headers, body)?;
675    let late = || {
676        ReplyRefused::Late {
677            millis: u64::try_from(bound.within.as_millis()).unwrap_or(u64::MAX),
678        }
679        .into_io()
680    };
681    let mut head: Vec<u8> = Vec::new();
682    let mut parsed: Option<ParsedHead> = None;
683    let mut held: Vec<u8> = Vec::new();
684    let mut json = Compact::default();
685    let mut chunk = vec![0u8; 16 * 1024];
686    loop {
687        if matches!(parsed, Some(ParsedHead { due: Some(0), .. })) {
688            break;
689        }
690        let left = deadline.saturating_duration_since(std::time::Instant::now());
691        if left.is_zero() {
692            return Err(late());
693        }
694        // when what is left of the whole is the shorter wait, a timeout is the deadline's
695        let deadline_wait = left <= bound.idle;
696        // a wait that cannot be set is an error like a read that failed, and is answered so
697        let read = stream
698            .socket()
699            .set_read_timeout(Some(left.min(bound.idle)))
700            .and_then(|()| stream.read(&mut chunk));
701        let n = match read {
702            Ok(0) => break,
703            Ok(n) => n,
704            Err(e)
705                if deadline_wait
706                    && matches!(
707                        e.kind(),
708                        std::io::ErrorKind::WouldBlock | std::io::ErrorKind::TimedOut
709                    ) =>
710            {
711                return Err(late())
712            }
713            Err(e) => return Err(e),
714        };
715        let mut bytes = &chunk[..n];
716        let rest;
717        if parsed.is_none() {
718            head.extend_from_slice(bytes);
719            let Some(end) = head.windows(4).position(|w| w == b"\r\n\r\n") else {
720                if head.len() > MAX_HEAD_BYTES {
721                    return Err(ReplyRefused::Oversized {
722                        bound: MAX_HEAD_BYTES,
723                    }
724                    .into_io());
725                }
726                continue;
727            };
728            rest = head.split_off(end + 4);
729            head.truncate(end);
730            let (status, headers) = parse_head(&String::from_utf8_lossy(&head))?;
731            let due = headers
732                .iter()
733                .find(|(k, _)| k == "content-length")
734                .and_then(|(_, v)| v.parse::<usize>().ok());
735            parsed = Some(ParsedHead {
736                status,
737                headers,
738                due,
739            });
740            bytes = &rest;
741        }
742        if let Some(ParsedHead { due: Some(due), .. }) = parsed.as_mut() {
743            let take = bytes.len().min(*due);
744            *due -= take;
745            bytes = &bytes[..take];
746        }
747        json.feed(bytes, &mut held, bound.bytes)?;
748    }
749    let head =
750        parsed.ok_or_else(|| std::io::Error::other("malformed HTTP response: no header end"))?;
751    Ok(Reply {
752        status: head.status,
753        headers: head.headers,
754        body: String::from_utf8_lossy(&held).into_owned(),
755    })
756}

758/// A reply's head once read: its status, its headers, and the body bytes its
759/// Content-Length says are still due, when it names one.
760struct ParsedHead {
761    status: u16,
762    headers: Vec<(String, String)>,
763    due: Option<usize>,
764}
765
766/// A JSON body held without its insignificant whitespace: every byte of a string is kept,
767/// the spaces, tabs and line ends between tokens are dropped, and nothing past the bound
768/// is held.
769#[derive(Debug, Default)]
770struct Compact {
771    in_string: bool,
772    escaped: bool,
773}
774
775impl Compact {
776    fn feed(&mut self, bytes: &[u8], held: &mut Vec<u8>, bound: usize) -> std::io::Result<()> {
777        for &b in bytes {
778            if self.in_string {
779                if self.escaped {
780                    self.escaped = false;
781                } else if b == b'\\' {
782                    self.escaped = true;
783                } else if b == b'"' {
784                    self.in_string = false;
785                }
786            } else if matches!(b, b' ' | b'\t' | b'\n' | b'\r') {
787                continue;
788            } else if b == b'"' {
789                self.in_string = true;
790            }
791            if held.len() >= bound {
792                return Err(ReplyRefused::Oversized { bound }.into_io());
793            }
794            held.push(b);
795        }
796        Ok(())
797    }
798}

800/// The status and the lowercased headers of a reply's head.
801fn parse_head(head: &str) -> std::io::Result<(u16, Vec<(String, String)>)> {
802    let mut lines = head.lines();
803    let status = lines
804        .next()
805        .and_then(|l| l.split_whitespace().nth(1))
806        .and_then(|s| s.parse::<u16>().ok())
807        .ok_or_else(|| std::io::Error::other("malformed HTTP response: no status line"))?;
808    let headers = lines
809        .filter_map(|l| {
810            l.split_once(':')
811                .map(|(k, v)| (k.trim().to_ascii_lowercase(), v.trim().to_string()))
812        })
813        .collect::<Vec<_>>();
814    Ok((status, headers))
815}
816
817fn parse_reply(raw: &[u8]) -> std::io::Result<Reply> {
818    let text = String::from_utf8_lossy(raw);
819    let (head, body) = text
820        .split_once("\r\n\r\n")
821        .ok_or_else(|| std::io::Error::other("malformed HTTP response: no header end"))?;
822    let (status, headers) = parse_head(head)?;
823    let length = headers
824        .iter()
825        .find(|(k, _)| k == "content-length")
826        .and_then(|(_, v)| v.parse::<usize>().ok());
827    let body = match length {
828        Some(n) if n <= body.len() => body[..n].to_string(),
829        _ => body.to_string(),
830    };
831    Ok(Reply {
832        status,
833        headers,
834        body,
835    })
836}

838/// Why the bridge could not get an answer.
839#[derive(Debug, Clone, thiserror::Error)]
840pub enum BridgeError {
841    #[error("the shared server at {url} is unreachable: {reason}")]
842    /// No connection could be made: nothing was sent, so nothing ran, and the server is gone.
843    Unreachable {
844        /// The server's URL.
845        url: String,
846        /// What happened.
847        reason: String,
848    },
849    #[error("the shared server at {url} did not answer: {reason}")]
850    /// The message was sent and no usable answer came back: a timeout, a dropped
851    /// connection, a server-side failure (5xx), an answer that is not JSON. The server
852    /// may have run it, and whether it did is unknown.
853    Unanswered {
854        /// The server's URL.
855        url: String,
856        /// What happened.
857        reason: String,
858    },
859    #[error("the shared server at {url} rejected the request ({status}): {body}")]
860    /// The server answered and refused.
861    Rejected {
862        /// The server's URL.
863        url: String,
864        /// The status code.
865        status: u16,
866        /// The body.
867        body: String,
868    },
869}
870
871impl BridgeError {
872    /// Whether this failure says the server is no longer the one to talk to, so that the
873    /// client should elect again: it could not be reached, it took the request and never
874    /// answered, it forgot the session twice running (404), or it said it had lost the lease
875    /// (409).
876    ///
877    /// Any other refusal comes from a server that is there and serving: it read the request
878    /// and would not take it — a request before `initialize`, a body too large. An election
879    /// cannot change that answer; it is relayed to the client.
880    ///
881    /// ```
882    /// use majordomus_cli::mcp::bridge::BridgeError;
883    /// let url = "http://127.0.0.1:1".to_string();
884    /// let refused = |status| BridgeError::Rejected { url: url.clone(), status, body: String::new() };
885    /// assert!(refused(404).calls_for_an_election());
886    /// assert!(refused(409).calls_for_an_election());
887    /// assert!(!refused(400).calls_for_an_election(), "a serving server's refusal is its answer");
888    /// assert!(!refused(413).calls_for_an_election(), "a message too large is too large anywhere");
889    /// assert!(BridgeError::Unanswered { url: url.clone(), reason: "timed out".into() }.calls_for_an_election());
890    /// ```
891    pub fn calls_for_an_election(&self) -> bool {
892        match self {
893            BridgeError::Unreachable { .. } | BridgeError::Unanswered { .. } => true,
894            BridgeError::Rejected { status, .. } => matches!(status, 404 | 409),
895        }
896    }
897
898    /// Whether `message` may be sent again after this failure without the risk of running
899    /// it twice: when the server provably did not run it (no connection, a lost session or
900    /// lease), or when running it twice changes nothing the first run did not
901    /// ([`idempotent`]).
902    ///
903    /// ```
904    /// use majordomus_cli::mcp::bridge::BridgeError;
905    /// use serde_json::json;
906    /// let url = "http://127.0.0.1:1".to_string();
907    /// let call = |tool: &str| json!({ "jsonrpc": "2.0", "id": 7, "method": "tools/call", "params": { "name": tool } });
908    /// let writer = call("majordomus_plan_transition");
909    /// let reader = call("majordomus_repository");
910    /// let lost = BridgeError::Unanswered { url: url.clone(), reason: "timed out".into() };
911    /// assert!(!lost.may_resend(&writer), "a write that may have run is not run again");
912    /// assert!(lost.may_resend(&reader), "a read is");
913    /// let never_sent = BridgeError::Unreachable { url: url.clone(), reason: "refused".into() };
914    /// assert!(never_sent.may_resend(&writer));
915    /// let too_large = BridgeError::Rejected { url, status: 413, body: String::new() };
916    /// assert!(!too_large.may_resend(&reader));
917    /// ```
918    pub fn may_resend(&self, message: &Value) -> bool {
919        match self {
920            BridgeError::Unreachable { .. } => true,
921            BridgeError::Rejected { status, .. } => matches!(status, 404 | 409),
922            BridgeError::Unanswered { .. } => idempotent(message),
923        }
924    }
925}

927/// Whether sending `message` twice changes nothing that sending it once did not.
928///
929/// Every MCP method but `tools/call` reads. A `tools/call` is idempotent when its tool's
930/// declared execution hints say so ([`crate::capability::ExecutionPolicy::hints`]), looked
931/// up in the compiled-in registry; a tool that registry does not know is assumed not to
932/// be, because a call wrongly withheld is reported while a call wrongly repeated may have
933/// written twice. A batch is idempotent when every message in it is.
934///
935/// ```
936/// use majordomus_cli::mcp::bridge::idempotent;
937/// use serde_json::json;
938/// let call = |tool: &str| json!({ "method": "tools/call", "params": { "name": tool } });
939/// assert!(idempotent(&json!({ "method": "resources/read", "params": { "uri": "majordomus://repository" } })));
940/// assert!(idempotent(&call("majordomus_repository")));
941/// assert!(!idempotent(&call("majordomus_plan_transition")));
942/// assert!(!idempotent(&call("no_such_tool")));
943/// assert!(!idempotent(&json!([call("majordomus_repository"), call("majordomus_plan_transition")])));
944/// ```
945pub fn idempotent(message: &Value) -> bool {
946    match message {
947        Value::Array(batch) => batch.iter().all(idempotent),
948        m if m.get("method").and_then(Value::as_str) == Some("tools/call") => m
949            .pointer("/params/name")
950            .and_then(Value::as_str)
951            .is_some_and(tool_is_idempotent),
952        _ => true,
953    }
954}
955
956fn tool_is_idempotent(name: &str) -> bool {
957    // composed only on a failure, never on the path of a message that was answered: the
958    // bridge keeps starting in milliseconds
959    crate::capability::CapabilityRegistry::builder()
960        .with_modules(crate::capability::builtin::modules())
961        .build()
962        .ok()
963        .and_then(|r| r.by_mcp_tool(name).map(|c| c.execution.hints().idempotent))
964        .unwrap_or(false)
965}

967/// A stdio session forwarded to a shared server.
968///
969/// Besides the frames, the bridge keeps two things it saw pass through: the client's
970/// `initialize`, so that a session can be re-opened without the client noticing, and the
971/// arguments of every accepted announcement the client has made, so that what the client
972/// said it was working on outlives the server it said it to. An announcement was the one
973/// fact that died with a server: on 2026-09-09 a session's intent vanished from the board
974/// when its transport was re-established, and eight peers could not see it for three hours.
975/// The bridge is the one place that sees both the announcement and the reconnection, so
976/// the bridge replays them — after a re-attach through [`Bridge::reinitialize`], and, for a
977/// takeover, through [`Bridge::announcements`] onto the new server's own board.
978///
979/// *Every* announcement, one per claim name, and not merely the last one: a client may
980/// hold several claims at once, and replaying only the most recent would restore one of
981/// them and quietly drop the rest — which is the same loss the replay exists to prevent,
982/// arriving by a different route.
983#[derive(Debug)]
984pub struct Bridge {
985    url: String,
986    session: Option<String>,
987    initialize: Option<Value>,
988    client: Option<ClientInfo>,
989    /// The arguments of the last accepted announcement per claim name, `""` for the
990    /// unnamed claim. A `BTreeMap` so a replay is in a stable order and re-announcing a
991    /// name replaces rather than appends, exactly as the board itself does.
992    announcements: BTreeMap<String, Value>,
993    /// How long one answer is waited for.
994    timeout: Duration,
995    /// Where the worker that started this bridge can be reached, read once from this
996    /// process's environment and carried in every `initialize` it forwards, so that the
997    /// board row of its session names it (I3956).
998    address: Option<PeerAddress>,
999}
1000
1001impl Bridge {
1002    /// A bridge to the server at `url`, with no session yet, waiting [`REQUEST_TIMEOUT`]
1003    /// for each answer.
1004    pub fn new(url: String) -> Self {
1005        Self::with_timeout(url, REQUEST_TIMEOUT)
1006    }
1007
1008    /// A bridge to the server at `url` that waits `timeout` for each answer. A call that
1009    /// outlasts it is [`BridgeError::Unanswered`]: it was sent, and whether it ran is unknown.
1010    ///
1011    /// ```
1012    /// use majordomus_cli::mcp::bridge::Bridge;
1013    /// use std::time::Duration;
1014    /// let bridge = Bridge::with_timeout("http://127.0.0.1:8741".into(), Duration::from_secs(5));
1015    /// assert_eq!(bridge.url(), "http://127.0.0.1:8741");
1016    /// ```
1017    pub fn with_timeout(url: String, timeout: Duration) -> Self {
1018        Bridge {
1019            url,
1020            session: None,
1021            initialize: None,
1022            client: None,
1023            announcements: BTreeMap::new(),
1024            timeout,
1025            address: PeerAddress::of_this_process(),
1026        }
1027    }
1028
1029    /// The arguments of every announcement the server accepted, one per claim name, in
1030    /// name order: what a takeover replays onto the new server's board.
1031    ///
1032    /// Empty when the client has announced nothing, which is every bridge that has only
1033    /// just been made — nothing has passed through it yet to be remembered.
1034    ///
1035    /// ```
1036    /// use majordomus_cli::mcp::bridge::Bridge;
1037    /// let bridge = Bridge::new("http://127.0.0.1:8741/mcp".into());
1038    /// assert_eq!(bridge.announcements().count(), 0);
1039    /// ```
1040    pub fn announcements(&self) -> impl Iterator<Item = &Value> {
1041        self.announcements.values()
1042    }
1043
1044    /// The server's URL.
1045    pub fn url(&self) -> &str {
1046        &self.url
1047    }
1048
1049    /// What the client said in `initialize`, once it has.
1050    pub fn client(&self) -> Option<&ClientInfo> {
1051        self.client.as_ref()
1052    }
1053
1054    /// Point the bridge at another server; the next message opens a session there.
1055    pub fn move_to(&mut self, url: String) {
1056        self.url = url;
1057        self.session = None;
1058    }
1059
1060    /// Forward one message; `None` when the server had nothing to say (a notification).
1061    pub fn handle(&mut self, message: &Value) -> Result<Option<Value>, BridgeError> {
1062        let stamped;
1063        let message = match &self.address {
1064            Some(address) if crate::http::mcp::is_initialize(message) => {
1065                stamped = stamp_address(message, address);
1066                &stamped
1067            }
1068            _ => message,
1069        };
1070        if crate::http::mcp::is_initialize(message) {
1071            let params = first_initialize_params(message);
1072            self.client = Some(ClientInfo::from_initialize(&params));
1073            self.initialize = Some(params);
1074            self.session = None;
1075        }
1076        let answer = match self.send(message)? {
1077            Sent::Answer(v) => v,
1078            Sent::SessionLost => {
1079                // the server forgot us (it restarted, or we were idle too long): open a
1080                // new session with the client's own initialize and try once more
1081                self.reinitialize()?;
1082                match self.send(message)? {
1083                    Sent::Answer(v) => v,
1084                    Sent::SessionLost => {
1085                        return Err(BridgeError::Rejected {
1086                            url: self.url.clone(),
1087                            status: 404,
1088                            body: "session lost twice in a row".into(),
1089                        })
1090                    }
1091                }
1092            }
1093        };
1094        if let Some(arguments) = announcement_in(message, answer.as_ref()) {
1095            let name = arguments
1096                .get("claim")
1097                .and_then(Value::as_str)
1098                .map(str::trim)
1099                .filter(|c| !c.is_empty())
1100                .unwrap_or("")
1101                .to_string();
1102            self.announcements.insert(name, arguments);
1103        }
1104        Ok(answer)
1105    }

1107    /// Open a new session with the initialize the client sent earlier. A no-op when it
1108    /// has not sent one yet.
1109    pub fn reinitialize(&mut self) -> Result<(), BridgeError> {
1110        let Some(params) = self.initialize.clone() else {
1111            return Ok(());
1112        };
1113        self.session = None;
1114        let init = json!({ "jsonrpc": "2.0", "id": "majordomus-bridge-reinitialize", "method": "initialize", "params": params });
1115        if let Sent::SessionLost = self.send(&init)? {
1116            return Err(BridgeError::Rejected {
1117                url: self.url.clone(),
1118                status: 404,
1119                body: "initialize was not accepted".into(),
1120            });
1121        }
1122        let _ = self.send(&json!({ "jsonrpc": "2.0", "method": "notifications/initialized" }))?;
1123        tracing::info!(url = %self.url, "session re-opened on the shared server");
1124        // what the client said it was working on is said again, to whoever serves now —
1125        // every claim it holds, because restoring one of several is still a loss
1126        for arguments in self.announcements.values().cloned().collect::<Vec<_>>() {
1127            let replay = json!({
1128                "jsonrpc": "2.0",
1129                "id": "majordomus-bridge-announce",
1130                "method": "tools/call",
1131                "params": { "name": crate::capability::builtin::peers::ANNOUNCE_TOOL, "arguments": arguments },
1132            });
1133            match self.send(&replay) {
1134                Ok(Sent::Answer(_)) => {
1135                    tracing::info!(url = %self.url, "the client's announcement was replayed to the server it is now attached to")
1136                }
1137                Ok(Sent::SessionLost) | Err(_) => {
1138                    tracing::warn!(url = %self.url, "the client's announcement could not be replayed; the board understates it until it announces again")
1139                }
1140            }
1141        }
1142        Ok(())
1143    }

1145    /// Keep the session alive.
1146    pub fn heartbeat(&mut self) -> Result<(), BridgeError> {
1147        if self.session.is_none() {
1148            return Ok(());
1149        }
1150        self.send(
1151            &json!({ "jsonrpc": "2.0", "id": "majordomus-bridge-heartbeat", "method": "ping" }),

1170    fn send(&mut self, message: &Value) -> Result<Sent, BridgeError> {
1171        let session = self.session.clone();
1172        let headers: Vec<(&str, &str)> = session
1173            .as_deref()
1174            .map(|s| vec![(SESSION_HEADER, s)])
1175            .unwrap_or_default();
1176        let body = message.to_string();
1177        // the server would refuse it unread; saying so here keeps the refusal from being
1178        // mistaken for a connection the server cut short while the body was still going out
1179        if body.len() > crate::http::server::MAX_BODY_BYTES {
1180            return Err(BridgeError::Rejected {
1181                url: self.url.clone(),
1182                status: 413,
1183                body: format!(
1184                    "the message is {} bytes, over the {} the server accepts; it was not sent",
1185                    body.len(),
1186                    crate::http::server::MAX_BODY_BYTES
1187                ),
1188            });
1189        }
1190        let (stream, host) =
1191            connect(&self.url, self.timeout).map_err(|e| BridgeError::Unreachable {
1192                url: self.url.clone(),
1193                reason: e.to_string(),
1194            })?;
1195        let reply = exchange(
1196            stream,
1197            &host,
1198            "POST",
1199            crate::http::mcp::PATH,
1200            &headers,
1201            Some(&body),
1202            self.timeout,
1203        )
1204        .map_err(|e| BridgeError::Unanswered {
1205            url: self.url.clone(),
1206            reason: e.to_string(),
1207        })?;
1208        if let Some(id) = reply.header(SESSION_HEADER) {
1209            self.session = Some(id.to_string());
1210        }
1211        match reply.status {
1212            200 => serde_json::from_str::<Value>(&reply.body)
1213                .map(|v| Sent::Answer(Some(v)))
1214                .map_err(|e| BridgeError::Unanswered {
1215                    url: self.url.clone(),
1216                    reason: format!("the answer is not JSON: {e}"),
1217                }),
1218            202 | 204 => Ok(Sent::Answer(None)),
1219            404 => Ok(Sent::SessionLost),
1220            // sent without a session by a bridge whose client did initialize (a re-attach
1221            // whose initialize failed): refused before the message was looked at, so it is
1222            // a session to open again, as a 404 is. A client that never initialized is told.
1223            400 if session.is_none()
1224                && self.initialize.is_some()
1225                && reply.body.contains("session_required") =>
1226            {
1227                Ok(Sent::SessionLost)
1228            }
1229            s if s >= 500 => Err(BridgeError::Unanswered {
1230                url: self.url.clone(),
1231                reason: format!("status {s}: {}", reply.body),
1232            }),
1233            s => Err(BridgeError::Rejected {
1234                url: self.url.clone(),
1235                status: s,
1236                body: reply.body,
1237            }),
1238        }
1239    }
1240}

1247/// The arguments of an announcement the server accepted: the message is a `tools/call` of
1248/// the announce tool, and the answer is a result rather than an error. A batch is not
1249/// looked into; a bridge that carried one announcement in a batch replays nothing, which
1250/// is what a bridge did for every announcement before this.
1251fn announcement_in(message: &Value, answer: Option<&Value>) -> Option<Value> {
1252    if message.get("method")?.as_str()? != "tools/call" {
1253        return None;
1254    }
1255    let params = message.get("params")?;
1256    if params.get("name")?.as_str()? != crate::capability::builtin::peers::ANNOUNCE_TOOL {
1257        return None;
1258    }
1259    let answer = answer?;
1260    if answer.get("error").is_some() || answer["result"]["isError"] == true {
1261        return None;
1262    }
1263    Some(
1264        params
1265            .get("arguments")
1266            .cloned()
1267            .unwrap_or_else(|| json!({})),
1268    )
1269}

1430    /// A server that answers each connection with the next scripted reply and records what
1431    /// it received: the request line, the session header, and the body.
1432    fn scripted(replies: Vec<Scripted>) -> (String, std::thread::JoinHandle<Vec<Value>>) {
1433        use std::io::BufRead;
1434        let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
1435        let url = format!("http://{}", listener.local_addr().unwrap());
1436        let server = std::thread::spawn(move || {
1437            let mut seen = Vec::new();
1438            for (status, body, session) in replies {
1439                let (stream, _) = listener.accept().unwrap();
1440                stream
1441                    .set_read_timeout(Some(Duration::from_secs(30)))
1442                    .unwrap();
1443                let mut reader = std::io::BufReader::new(stream.try_clone().unwrap());
1444                let mut line = String::new();
1445                reader.read_line(&mut line).unwrap();
1446                let (mut length, mut header) = (0usize, Value::Null);
1447                loop {
1448                    let mut h = String::new();
1449                    reader.read_line(&mut h).unwrap();
1450                    let h = h.trim_end();
1451                    if h.is_empty() {
1452                        break;
1453                    }
1454                    let (k, v) = h.split_once(':').unwrap();
1455                    if k.eq_ignore_ascii_case("content-length") {
1456                        length = v.trim().parse().unwrap();
1457                    }
1458                    if k.eq_ignore_ascii_case(SESSION_HEADER) {
1459                        header = json!(v.trim());
1460                    }
1461                }
1462                let mut raw = vec![0u8; length];
1463                reader.read_exact(&mut raw).unwrap();
1464                let message: Value = serde_json::from_slice(&raw).unwrap_or(Value::Null);
1465                seen.push(
1466                    json!({ "line": line.trim_end(), "session": header, "message": message }),
1467                );
1468                let mut head = format!(
1469                    "HTTP/1.1 {status} X\r\nConnection: close\r\nContent-Length: {}\r\n",
1470                    body.len()
1471                );
1472                if let Some(s) = session {
1473                    head.push_str(&format!("{SESSION_HEADER}: {s}\r\n"));
1474                }
1475                let mut stream = stream;
1476                stream
1477                    .write_all(format!("{head}\r\n{body}").as_bytes())
1478                    .unwrap();
1479            }
1480            seen
1481        });
1482        (url, server)
1483    }

1814        let ca_key = rcgen::KeyPair::generate().unwrap();

1816        ca.is_ca = rcgen::IsCa::Ca(rcgen::BasicConstraints::Unconstrained);
1817        ca.distinguished_name
1818            .push(rcgen::DnType::CommonName, "majordomus test authority");

1820            rcgen::KeyUsagePurpose::KeyCertSign,
1821            rcgen::KeyUsagePurpose::DigitalSignature,
1822        ];

1824        let issuer = rcgen::Issuer::new(ca, ca_key);
1825        let leaf_key = rcgen::KeyPair::generate().unwrap();

1827            rcgen::CertificateParams::new(names.iter().map(|n| n.to_string()).collect::<Vec<_>>())

1835                .unwrap()

1852    /// A TLS server on `bind` answering `connections` connections with one JSON body each,
1853    /// closing with `close_notify` or without. A connection whose handshake the client
1854    /// refused is heard as an empty request.
1855    fn tls_server(
1856        bind: &str,
1857        config: Arc<ServerConfig>,
1858        connections: usize,
1859        close_notify: bool,
1860    ) -> (SocketAddr, std::thread::JoinHandle<Vec<Heard>>) {
1861        let listener = std::net::TcpListener::bind(bind).unwrap();
1862        let addr = listener.local_addr().unwrap();
1863        let server = std::thread::spawn(move || {
1864            let mut heard = Vec::new();
1865            for _ in 0..connections {
1866                let (socket, _) = listener.accept().unwrap();
1867                socket
1868                    .set_read_timeout(Some(Duration::from_secs(30)))
1869                    .unwrap();
1870                let mut tls =
1871                    StreamOwned::new(ServerConnection::new(config.clone()).unwrap(), socket);
1872                let mut raw = Vec::new();
1873                let mut buf = [0u8; 4096];
1874                while !whole(&raw) {
1875                    match tls.read(&mut buf) {
1876                        Ok(0) | Err(_) => break,
1877                        Ok(n) => raw.extend_from_slice(&buf[..n]),
1878                    }
1879                }
1880                let sni = tls.conn.server_name().map(str::to_string);
1881                if whole(&raw) {
1882                    let body = r#"{"accepted":true,"candidates":[]}"#;
1883                    let _ = tls.write_all(
1884                        format!("HTTP/1.1 200 OK\r\nConnection: close\r\nContent-Length: {}\r\n\r\n{body}", body.len())
1885                            .as_bytes(),
1886                    );
1887                    if close_notify {
1888                        tls.conn.send_close_notify();
1889                    }
1890                    let _ = tls.flush();
1891                }
1892                heard.push(Heard {
1893                    request: String::from_utf8_lossy(&raw).into_owned(),
1894                    sni,
1895                });
1896            }
1897            heard
1898        });
1899        (addr, server)
1900    }
