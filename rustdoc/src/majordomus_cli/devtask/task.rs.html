1//! One issue read as an executable development task.
2//!
3//! Four groups, one per owner of the facts in it — see the [module documentation] for the
4//! table — and every field of every group attested. Nothing is read here: [`DevTaskModel`]
5//! carries the already-derived plan, the already-parsed record, the already-derived git
6//! trace and the already-resolved session records, and [`DevTask::build`] is a pure
7//! function of them. That is what lets the whole model be unit-tested without a repository
8//! and what keeps the derivations single: this file composes, it does not compute.
9//!
10//! [module documentation]: super

22/// The adapter that owns the projection of this model onto GitHub.
23///
24/// Named, not reimplemented. This executable makes no network call —
25/// `test/cases/08_no_forbidden_constructs.sh` is the proof — so the live state of the

34/// Everything [`DevTask::build`] needs, gathered by the caller that has a repository.
35#[derive(Debug)]
36pub struct DevTaskModel<'a> {
37    /// The derived plan: the one derivation of status, waves, dependents and findings.
38    pub plan: &'a Plan,
39    /// The canonical record, when the model declares this issue.
40    pub record: Option<RecordRef<'a>>,
41    /// What git says about the branches and commits that realised it, when git was
42    /// consulted. `None` is not "nothing realised it": it is "git was not asked", and the
43    /// execution group says which.
44    pub trace: Option<&'a IssueTrace>,
45    /// The session records whose branch names this issue, in canonical order. Resolved by
46    /// the caller because the rule belongs to [`crate::worktree::state::issue_of`] and is
47    /// not restated here.
48    pub sessions: Vec<SessionRef>,
49    /// The repository the external projection targets, as `project.yaml` declares it.
50    pub repository: Option<&'a str>,
51}

53/// The canonical record: where it is, and what it says.
54#[derive(Debug, Clone, Copy)]
55pub struct RecordRef<'a> {
56    /// Repository-relative path of the file the record was read from.
57    pub path: &'a str,
58    /// The parsed YAML, whole. Key *presence* here is what makes a field explicit, which is
59    /// the one thing the flattened plan cannot tell a reader.
60    pub metadata: &'a Value,
61}

63/// One session record that names this issue, and how strongly.
64///
65/// The two ways a session reaches an issue are not equally good, and the difference is the
66/// whole reason this carries a flag rather than being a bare id:
67///
68/// * the record's own `issues` key. `share/schemas/majordomus/session-record` describes it
69///   as "the issues it moved, by id — derived from the ledger's own events for this
70///   episode; never authored", so it is a canonical link that a machine worked out from
71///   events it can read. That is `FieldProvenance::Derived`.
72/// * the branch name. A record always declares the branch it was written on, and
73///   [`crate::worktree::state::issue_of`] reads an issue id out of one. It is a rule about a
74///   name and it can be wrong — a branch renamed, an id that is a prefix of another — so it
75///   is `FieldProvenance::Inferred`.
76///
77/// Reporting both under one provenance would make the weaker of them look like the
78/// stronger, which is the defect this whole module exists to prevent.
79#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
80pub struct SessionRef {
81    /// The episode's id.
82    pub session_id: String,
83    /// The branch it was written on.
84    pub branch: String,
85    /// When the record says it was written.
86    pub created_at: String,
87    /// Repository-relative path of the record.
88    pub path: String,
89    /// True when the record's own `issues` key names this issue: a canonical link, not a
90    /// guess about a branch name.
91    pub declares_issue: bool,
92}

96/// What a person authored in the canonical record. Every field is `FieldProvenance::Explicit`

115    pub milestone: AttestedText,
116    /// `p0` … `p3`.
117    pub priority: AttestedText,
118    /// The execution profile it is worked under.
119    pub profile: AttestedText,
120    /// Whether it may run beside another issue of its wave.

129    pub cancelled: AttestedText,

131    pub created_at: AttestedText,
132    /// When it last moved.
133    pub updated_at: AttestedText,
134    /// When execution began.
135    pub started_at: AttestedText,

189/// Local execution state: what actually happened in this checkout's git history and session
190/// records. Derived where the relation was declared, inferred where it was not, and the
191/// fields say which.
192#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
193pub struct DevTaskExecution {
194    /// Whether git was consulted at all. False makes every git-derived field below
195    /// `unknown` rather than empty, which is the difference between "nothing realised this"
196    /// and "nobody looked".
197    pub git_consulted: bool,
198    /// The branches whose name carries this issue's id, local first.
199    pub branches: AttestedList,
200    /// Of those, the ones that reached the trunk.
201    pub merged_branches: AttestedList,
202    /// How many distinct commits those branches hold that the trunk did not.
203    pub commits: AttestedCount,
204    /// The trunk every branch was measured against.
205    pub trunk: AttestedText,
206    /// True when every branch's commits could be derived. False when one reached the trunk
207    /// without a merge commit of its own, so part of the work cannot be told from the
208    /// trunk's — reported rather than counted as zero.
209    pub commits_complete: AttestedText,
210    /// The session records whose own `issues` key names this issue. Derived: the key is
211    /// worked out from the episode's ledger events, not authored and not guessed.
212    pub sessions: AttestedList,
213    /// The session records that reach this issue only through their branch name, and which
214    /// therefore may be wrong. Never overlaps [`DevTaskExecution::sessions`]: a session that
215    /// declared the issue is reported there and not here, so the weaker relation never
216    /// stands in for the stronger one.
217    pub sessions_by_branch: AttestedList,
218}

274/// One issue as an executable development task: what was authored, where it sits, what has
275/// happened to it, where its external projection stands, and whether a worker may start it.
276#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
277pub struct DevTask {
278    /// The issue id, as asked for.
279    pub issue: String,
280    /// Whether the canonical model declares it. False is answered rather than refused, with
281    /// every canonical field `unknown`: a typo that read as "nothing has been authored" is
282    /// the one answer this must never give.
283    pub declared: bool,
284    /// Repository-relative path of the canonical record, when there is one.
285    #[serde(skip_serializing_if = "Option::is_none")]
286    pub record: Option<String>,
287    /// What a person authored.
288    pub declaration: DevTaskDeclaration,
289    /// Where the plan graph puts it.
290    pub position: DevTaskPosition,
291    /// What happened locally.
292    pub execution: DevTaskExecution,
293    /// Where the external projection stands.
294    pub synchronisation: DevTaskSynchronisation,
295    /// Whether it can be worked on, and what is in the way.
296    pub readiness: ReadinessVerdict,
297    /// Everything wrong with its records.
298    pub diagnostics: Vec<DevTaskDiagnostic>,
299    /// How much of the answer was authored and how much a machine worked out.
300    pub attestation: AttestationTally,
301}

303/// Is the key present in the record, and does it carry anything? A key present and null is
304/// as good as absent: a reader cannot act on it, and calling it explicit would be the
305/// defect this module exists to prevent.
306fn present<'a>(meta: Option<&'a Value>, key: &str) -> Option<&'a Value> {
307    match meta?.get(key) {
308        None | Some(Value::Null) => None,
309        Some(Value::String(s)) if s.is_empty() => None,
310        Some(v) => Some(v),
311    }
312}

314/// One scalar of the record, attested by whether the record carries the key.
315fn authored(record: Option<RecordRef<'_>>, key: &str) -> AttestedText {
316    let source = source_of(record, key);
317    match present(record.map(|r| r.metadata), key) {
318        Some(v) => AttestedText::explicit(crate::plan::scalar(Some(v)), source),
319        None if record.is_none() => AttestedText::unknown(
320            source,
321            "the canonical model declares no issue with this id".to_string(),
322        ),
323        None => AttestedText::unknown(source, format!("the record declares no `{key}`")),
324    }
325}
326
327/// One list of the record, attested by whether the record carries the key — so an authored
328/// empty list is explicit and an absent key is unknown.
329fn authored_list(record: Option<RecordRef<'_>>, key: &str) -> AttestedList {
330    let source = source_of(record, key);
331    match record.map(|r| r.metadata).and_then(|m| m.get(key)) {
332        Some(Value::Array(items)) => AttestedList::explicit(
333            items.iter().map(|v| crate::plan::scalar(Some(v))).collect(),
334            source,
335        ),
336        // A scalar where a list was expected is one item, which is what the flattener
337        // produces for `scope: apps/` too; the plan reads it the same way.
338        Some(other) if !other.is_null() => {
339            AttestedList::explicit(vec![crate::plan::scalar(Some(other))], source)
340        }
341        _ if record.is_none() => {
342            AttestedList::unknown(source, "the canonical model declares no issue with this id")
343        }
344        _ => AttestedList::unknown(source, format!("the record declares no `{key}`")),
345    }
346}

348/// Where a reader goes to check one authored field.
349fn source_of(record: Option<RecordRef<'_>>, key: &str) -> String {
350    match record {
351        Some(r) => format!("{}#{key}", r.path),
352        None => format!(".ai/repo/project/issues#{key}"),
353    }
354}

356/// The `covers` token of every evidence entry that carries one, and the `commit` of every
357/// entry that carries one. Read from the record rather than from the plan, because the plan
358/// counts them and this names them.
359fn evidence_field(record: Option<RecordRef<'_>>, key: &str) -> AttestedList {
360    let source = source_of(record, &format!("evidence[].{key}"));
361    let Some(RecordRef { metadata, .. }) = record else {
362        return AttestedList::unknown(source, "the canonical model declares no issue with this id");
363    };
364    match metadata.get("evidence") {
365        Some(Value::Array(items)) => AttestedList::explicit(
366            items
367                .iter()
368                .filter_map(|e| e.get(key))
369                .map(|v| crate::plan::scalar(Some(v)))
370                .collect(),
371            source,
372        ),
373        _ => AttestedList::unknown(source, "the record carries no `evidence` entries"),
374    }
375}

377impl DevTask {
378    /// The capability id every field of the derived half points a reader at.
379    const PLAN: &'static str = "plan.model";
380    /// The capability id the git-derived half points a reader at.
381    const TRACE: &'static str = "trace.issue";
382
383    /// Compose one development task out of derivations that already exist.
384    ///
385    /// Pure: every input is a value the caller already has, and the only work done here is
386    /// attesting each field and deriving the readiness. A repository is never touched.
387    #[allow(clippy::too_many_lines)] // one composition, field by field, with each attested
388    pub fn build(model: &DevTaskModel<'_>, issue: &str) -> DevTask {
389        let record = model.record;
390        let declared = record.is_some();
391        let derived = model.plan.issue(issue);
392        let milestone = derived.and_then(|i| model.plan.milestone(&i.milestone));
393
394        let declaration = DevTaskDeclaration {
395            title: authored(record, "title"),
396            objective: authored(record, "objective"),
397            why: authored(record, "why"),
398            current_state: authored(record, "current_state"),
399            desired_state: authored(record, "desired_state"),
400            completion: authored(record, "completion"),
401            milestone: authored(record, "milestone"),
402            priority: authored(record, "priority"),
403            profile: authored(record, "profile"),
404            parallel_safe: authored(record, "parallel_safe"),
405            owner: authored(record, "owner"),
406            risk: authored(record, "risk"),
407            slug: authored(record, "slug"),
408            cancelled: authored(record, "cancelled"),
409            created_at: authored(record, "created_at"),
410            updated_at: authored(record, "updated_at"),
411            started_at: authored(record, "started_at"),
412            verified_at: authored(record, "verified_at"),
413            completed_at: authored(record, "completed_at"),
414            scope: authored_list(record, "scope"),
415            non_scope: authored_list(record, "non_scope"),
416            depends_on: authored_list(record, "depends_on"),
417            acceptance_criteria: authored_list(record, "acceptance_criteria"),
418            validation: authored_list(record, "validation"),
419            evidence_required: authored_list(record, "evidence_required"),
420            evidence_present: evidence_field(record, "covers"),
421            evidence_commits: evidence_field(record, "commit"),
422        };
423
424        let position = Self::position(model.plan, derived, milestone);
425        let execution = Self::execution(model.trace, &model.sessions);
426        let synchronisation = Self::synchronisation(model.repository);
427        let readiness = Self::readiness(
428            model.plan,
429            derived,
430            &declaration,
431            milestone.map(|m| m.id.as_str()),
432        );
433        let diagnostics = Self::diagnostics(model.plan, issue);
434
435        let mut task = DevTask {
436            issue: issue.to_string(),
437            declared,
438            record: record.map(|r| r.path.to_string()),
439            declaration,
440            position,
441            execution,
442            synchronisation,
443            readiness,
444            diagnostics,
445            attestation: AttestationTally::default(),
446        };
447        task.attestation = task.tally();
448        task
449    }

451    fn position(
452        plan: &Plan,
453        derived: Option<&PlanIssue>,
454        milestone: Option<&crate::plan::PlanMilestone>,
455    ) -> DevTaskPosition {
456        let absent = "the canonical model declares no issue with this id";
457        let Some(i) = derived else {
458            return DevTaskPosition {
459                status: AttestedText::unknown(Self::PLAN, absent),
460                wave: AttestedCount::unknown(Self::PLAN, absent),
461                blocked_by: AttestedList::unknown(Self::PLAN, absent),
462                dependents: AttestedList::unknown(Self::PLAN, absent),
463                transitive_dependents: AttestedCount::unknown(Self::PLAN, absent),
464                milestone_status: AttestedText::unknown(Self::PLAN, absent),
465                milestone_rank: AttestedCount::unknown(Self::PLAN, absent),
466                milestone_blocked_by: AttestedList::unknown(Self::PLAN, absent),
467                evidence_have: AttestedCount::unknown(Self::PLAN, absent),
468                evidence_need: AttestedCount::unknown(Self::PLAN, absent),
469            };
470        };
471        let cyclic = plan
472            .findings
473            .iter()
474            .any(|f| f.code == "cycle" && f.message.split_whitespace().any(|w| w == i.id));
475        DevTaskPosition {
476            status: AttestedText::derived(&i.status, Self::PLAN),
477            wave: if cyclic {
478                AttestedCount::unknown(
479                    Self::PLAN,
480                    "a dependency cycle prevents it from ever having a wave",
481                )
482            } else {
483                AttestedCount::derived(i.wave, Self::PLAN)
484            },
485            blocked_by: AttestedList::derived(i.blocked_by.clone(), Self::PLAN),
486            dependents: AttestedList::derived(i.dependents.clone(), Self::PLAN),
487            transitive_dependents: AttestedCount::derived(
488                u32::try_from(super::graph::transitive_dependents(&plan.issues, &i.id).len())
489                    .unwrap_or(u32::MAX),
490                Self::PLAN,
491            ),
492            milestone_status: milestone.map_or_else(
493                || {
494                    AttestedText::unknown(
495                        Self::PLAN,
496                        "the issue names no milestone the model declares",
497                    )
498                },
499                |m| AttestedText::derived(&m.status, Self::PLAN),
500            ),
501            milestone_rank: milestone.map_or_else(
502                || {
503                    AttestedCount::unknown(
504                        Self::PLAN,
505                        "the issue names no milestone the model declares",
506                    )
507                },
508                |m| AttestedCount::derived(m.rank, Self::PLAN),
509            ),
510            milestone_blocked_by: milestone.map_or_else(
511                || {
512                    AttestedList::unknown(
513                        Self::PLAN,
514                        "the issue names no milestone the model declares",
515                    )
516                },
517                |m| AttestedList::derived(m.blocked_by.clone(), Self::PLAN),
518            ),
519            evidence_have: AttestedCount::derived(i.evidence_have, Self::PLAN),
520            evidence_need: AttestedCount::derived(i.evidence_need, Self::PLAN),
521        }
522    }

524    fn execution(trace: Option<&IssueTrace>, sessions: &[SessionRef]) -> DevTaskExecution {
525        let unasked = "git was not consulted for this answer";
526        let declared_field = AttestedList::derived(
527            sessions
528                .iter()
529                .filter(|s| s.declares_issue)
530                .map(|s| s.session_id.clone())
531                .collect(),
532            "kind:session#issues",
533        );
534        let branch_field = AttestedList::inferred(
535            sessions
536                .iter()
537                .filter(|s| !s.declares_issue)
538                .map(|s| s.session_id.clone())
539                .collect(),
540            "kind:session#branch",
541            "the record's own `issues` key does not name this issue; the link is the \
542             branch-name rule of worktree::state::issue_of, which a rename can break",
543        );
544        let Some(t) = trace else {
545            return DevTaskExecution {
546                git_consulted: false,
547                branches: AttestedList::unknown(Self::TRACE, unasked),
548                merged_branches: AttestedList::unknown(Self::TRACE, unasked),
549                commits: AttestedCount::unknown(Self::TRACE, unasked),
550                trunk: AttestedText::unknown(Self::TRACE, unasked),
551                commits_complete: AttestedText::unknown(Self::TRACE, unasked),
552                sessions: declared_field,
553                sessions_by_branch: branch_field,
554            };
555        };
556        DevTaskExecution {
557            git_consulted: true,
558            branches: AttestedList::derived(
559                t.branches.iter().map(|b| b.name.clone()).collect(),
560                Self::TRACE,
561            ),
562            merged_branches: AttestedList::derived(
563                t.branches
564                    .iter()
565                    .filter(|b| b.merge_commit.is_some())
566                    .map(|b| b.name.clone())
567                    .collect(),
568                Self::TRACE,
569            ),
570            commits: AttestedCount::derived(
571                u32::try_from(t.commits).unwrap_or(u32::MAX),
572                Self::TRACE,
573            ),
574            trunk: t.trunk.as_ref().map_or_else(
575                || AttestedText::unknown(Self::TRACE, "this repository has no trunk ref"),
576                |trunk| AttestedText::derived(trunk, Self::TRACE),
577            ),
578            commits_complete: AttestedText::derived(t.complete.to_string(), Self::TRACE),
579            sessions: declared_field,
580            sessions_by_branch: branch_field,
581        }
582    }

585        let network = format!(
586            "a live GitHub state; this executable makes no network call, so only \
587             `{PROJECTION_ADAPTER} --check` can answer it"
588        );
589        DevTaskSynchronisation {
590            adapter: PROJECTION_ADAPTER.to_string(),
591            repository: repository.map_or_else(

596                    // `project/milestones/*.yaml` and one for `project/issues/*.yaml` and
597                    // none for `project/project.yaml`, so the `project` kind has no objects
598                    // and `Plan::project` is blank for every reader of it. Saying the file
599                    // is silent would be this module inventing a fact about a file it did
600                    // not read, which is the whole thing it exists not to do.
601                    AttestedText::unknown(

604                    )
605                },
606                |r| AttestedText::explicit(r, ".ai/repo/project/project.yaml#repository"),
607            ),
608            state: AttestedText::unknown(PROJECTION_ADAPTER, network.clone()),
609            external_id: AttestedText::unknown(PROJECTION_ADAPTER, network),
610            reconcile_with: RECONCILE_WITH.iter().map(|c| (*c).to_string()).collect(),
611            writable: false,
612        }

615    fn readiness(
616        plan: &Plan,
617        derived: Option<&PlanIssue>,
618        declaration: &DevTaskDeclaration,
619        milestone: Option<&str>,
620    ) -> ReadinessVerdict {
621        let Some(i) = derived else {
622            return ReadinessVerdict {
623                state: TaskReadiness::Undeclared,
624                canonical_status: String::new(),
625                reason: "the canonical model declares no issue with this id".into(),
626                startable: false,
627                terminal: false,
628                blockers: vec![Blocker::new(
629                    BlockerKind::Reference,
630                    "",
631                    "no canonical record; author one under .ai/repo/project/issues/ \
632                     before anything can be executed against it",
633                )],
634            };
635        };
636        let completion_recorded = !declaration.completed_at.text().is_empty();
637        let evidence_covered = i.evidence_have >= i.evidence_need;
638        let state = TaskReadiness::derive(
639            &i.status,
640            &i.blocked_by,
641            completion_recorded,
642            evidence_covered,
643        );
644        let mut blockers: Vec<Blocker> = Vec::new();
645        for b in &i.blocked_by {
646            if let Some(m) = b.strip_prefix(MILESTONE_BLOCKER) {
647                blockers.push(Blocker::new(
648                    BlockerKind::Milestone,
649                    m,
650                    format!("the milestone {m} waits on an outcome of its own"),
651                ));
652            } else {
653                blockers.push(Blocker::new(
654                    BlockerKind::Issue,
655                    b,
656                    format!("{b} is not DONE"),
657                ));
658            }
659        }
660        // A dependency the model does not declare cannot appear in `blocked_by` — the plan
661        // drops the edge and reports `unknown_dependency` — so it is named here from the
662        // declaration, or it would be a blocker nobody could see from this surface.
663        for d in &declaration.depends_on.values {
664            if plan.issue(d).is_none() && d != &i.id {
665                blockers.push(Blocker::new(
666                    BlockerKind::Reference,
667                    d,
668                    format!(
669                        "{d} is declared as a dependency and the model has no such issue; \
670                         fix the reference or author the record"
671                    ),
672                ));
673            }
674        }
675        // A cycle is reported once, about the whole graph; an issue in it or downstream of
676        // it can never become ready, and that is a blocker of its own kind.
677        for f in &plan.findings {
678            if f.code == "cycle" && f.message.split_whitespace().any(|w| w == i.id) {
679                blockers.push(Blocker::new(
680                    BlockerKind::Cycle,
681                    &i.id,
682                    "a dependency cycle prevents it from ever becoming ready".to_string(),
683                ));
684            }
685        }
686        if state == TaskReadiness::CompletionBlocked {
687            let have = &declaration.evidence_present.values;
688            for need in &declaration.evidence_required.values {
689                if !have.contains(need) {
690                    blockers.push(Blocker::new(
691                        BlockerKind::Evidence,
692                        need,
693                        format!(
694                            "completion is recorded and no evidence entry covers `{need}`; \
695                             record it with `majordomus evidence`"
696                        ),
697                    ));
698                }
699            }
700        }
701        let reason = match state {
702            TaskReadiness::Ready => {
703                "no dependency and no gate holds it; nobody has started it".into()
704            }
705            TaskReadiness::Blocked => format!(
706                "{} of its dependencies {} not DONE",
707                blockers
708                    .iter()
709                    .filter(|b| b.kind == BlockerKind::Issue)
710                    .count(),
711                if blockers
712                    .iter()
713                    .filter(|b| b.kind == BlockerKind::Issue)
714                    .count()
715                    == 1
716                {
717                    "is"
718                } else {
719                    "are"
720                }
721            ),
722            TaskReadiness::Waiting => format!(
723                "every dependency is satisfied; the milestone {} waits on an outcome of its own",
724                milestone.unwrap_or("it belongs to")
725            ),
726            TaskReadiness::InProgress => "execution began; the record carries started_at".into(),
727            TaskReadiness::Review => {
728                "implementation is declared finished and completion is not recorded".into()
729            }
730            TaskReadiness::CompletionBlocked => {
731                "completion is recorded and the evidence the record requires is not all there"
732                    .into()
733            }
734            TaskReadiness::Complete => {
735                "completion is recorded and every required evidence token is present".into()
736            }
737            TaskReadiness::Cancelled => "withdrawn".into(),
738            TaskReadiness::Undeclared => format!(
739                "the plan assigned the status `{}`, which its own vocabulary does not declare",
740                i.status
741            ),
742        };
743        ReadinessVerdict {
744            state,
745            canonical_status: i.status.clone(),
746            reason,
747            startable: state.is_startable(),
748            terminal: state.is_terminal(),
749            blockers,
750        }
751    }
