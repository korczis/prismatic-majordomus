1//! Canonical presentation order.
2//!
3//! One collection has one order, and every interface that shows it shows that order. A
4//! sort written beside a renderer is a second opinion about the same collection, and two
5//! opinions drift: the command line alphabetises, the Cockpit groups, the generated index
6//! keeps whatever the walk produced, and a reader who compares them concludes that the
7//! surfaces disagree about the repository rather than about sorting.
8//!
9//! The order is a total order over four parts, most significant first:
10//!
11//! 1. **group** — the semantic bucket the item is filed under, itself compared naturally;
12//!    an item with no group sorts after every grouped item, because an ungrouped tail is
13//!    legible and an ungrouped head hides the groups.
14//! 2. **rank** — an explicit position inside the group, ascending, for the collections
15//!    whose domain genuinely has one (a lifecycle runs setup before conclude, whatever
16//!    those words sort as). It defaults to [`OrderKey::UNRANKED`], so a collection that
17//!    has no such semantics never mentions it.
18//! 3. **label** — what a person reads, compared naturally: digit runs by value, so
19//!    `item-2` precedes `item-10`, and letters case-insensitively, so `Alpha` and `alpha`
20//!    stay adjacent.
21//! 4. **identity** — the canonical id, compared naturally, as the final tie-breaker.
22//!
23//! The fourth part is what makes the order total rather than merely tidy. Two items with
24//! the same label must not exchange places because an unrelated item was added, because a
25//! different iterator was used, or because one machine's filesystem enumerates differently
26//! from another's; identities are unique, so the comparison always ends there.
27//!
28//! A numeric sample is ordered here too, by [`numeric`]: nobody reads it item by item, but a
29//! median read off it depends on its order, and a float has exactly one order worth having.
30//!
31//! ```
32//! use majordomus_cli::order::{canonical, OrderKey, Ordered};
33//!
34//! struct Recipe {
35//!     group: &'static str,
36//!     name: &'static str,
37//! }
38//!
39//! impl Ordered for Recipe {
40//!     fn order_key(&self) -> OrderKey<'_> {
41//!         OrderKey::grouped(self.group, self.name, self.name)
42//!     }
43//! }
44//!
45//! let mut recipes = vec![
46//!     Recipe { group: "quality", name: "test-10" },
47//!     Recipe { group: "development", name: "serve" },
48//!     Recipe { group: "quality", name: "test-2" },
49//! ];
50//! canonical(&mut recipes);
51//! let order: Vec<_> = recipes.iter().map(|r| r.name).collect();
52//! assert_eq!(order, ["serve", "test-2", "test-10"]);
53//! ```
54

57/// What a collection's items answer so that the collection can be put in canonical order.
58///
59/// Implemented on the domain type, once, next to the type — not on a view of it beside a
60/// renderer. Every projection then sorts by calling [`canonical`], and none of them holds
61/// an opinion of its own about the sequence.
62pub trait Ordered {
63    /// The four parts this item is ordered by. Borrowed from the item: building a key
64    /// allocates nothing, so sorting a large collection costs the comparisons and no more.
65    fn order_key(&self) -> OrderKey<'_>;
66}
67
68/// A reference orders as the item it points at, so a projection can put a borrowed view of
69/// a collection in canonical order without cloning every item into a vector first.
70impl<T: Ordered + ?Sized> Ordered for &T {
71    fn order_key(&self) -> OrderKey<'_> {
72        (**self).order_key()
73    }
74}

76/// The four parts of the canonical order, most significant first. See the module
77/// documentation for what each part means and why the last one is not optional.
78#[derive(Debug, Clone, Copy, PartialEq, Eq)]
79pub struct OrderKey<'a> {
80    /// The semantic bucket, or `None` for an item that is not filed under one. `None`
81    /// sorts after every named group.
82    pub group: Option<&'a str>,
83    /// The explicit position inside the group, ascending. [`OrderKey::UNRANKED`] when the
84    /// collection has no such semantics, which is most of them.
85    pub rank: i64,
86    /// What a person reads.
87    pub label: &'a str,
88    /// The canonical id: unique, and therefore the tie-breaker that makes the order total.
89    pub identity: &'a str,
90}
91
92impl<'a> OrderKey<'a> {
93    /// The rank of an item whose collection has no explicit ordering semantics. Zero, so
94    /// that a collection where only some items are ranked puts those first, which is what
95    /// an explicit rank is for.
96    pub const UNRANKED: i64 = 0;
97
98    /// An item with no group and no explicit rank: ordered by its label, then its identity.
99    pub fn plain(label: &'a str, identity: &'a str) -> Self {
100        Self {
101            group: None,
102            rank: Self::UNRANKED,
103            label,
104            identity,
105        }
106    }
107
108    /// An item filed under a group, ordered inside it by its label, then its identity.
109    pub fn grouped(group: &'a str, label: &'a str, identity: &'a str) -> Self {
110        Self {
111            group: Some(group),
112            rank: Self::UNRANKED,
113            label,
114            identity,
115        }
116    }
117
118    /// The same key with an explicit position inside its group.
119    pub fn ranked(self, rank: i64) -> Self {
120        Self { rank, ..self }
121    }
122
123    /// The same key ranked by `n` descending: the largest first. What "newest first" is for
124    /// a collection whose items carry a counter that only grows — a pull request's number, a
125    /// commit's time — so the order is stated once, here, instead of as a `Reverse` beside
126    /// every renderer. A value beyond `i64::MAX` ranks with it, first.
127    ///
128    /// ```
129    /// use majordomus_cli::order::OrderKey;
130    ///
131    /// let newer = OrderKey::plain("b", "b").descending(12);
132    /// let older = OrderKey::plain("a", "a").descending(9);
133    /// assert!(newer < older, "the larger number comes first, whatever the label says");
134    /// assert_eq!(newer.rank, -12);
135    /// ```
136    pub fn descending(self, n: u64) -> Self {
137        self.ranked(i64::try_from(n).unwrap_or(i64::MAX).saturating_neg())
138    }
139}
140
141impl Ord for OrderKey<'_> {
142    fn cmp(&self, other: &Self) -> Ordering {
143        group_cmp(self.group, other.group)
144            .then_with(|| self.rank.cmp(&other.rank))
145            .then_with(|| natural_cmp(self.label, other.label))
146            .then_with(|| natural_cmp(self.identity, other.identity))
147    }
148}
149
150impl PartialOrd for OrderKey<'_> {
151    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
152        Some(self.cmp(other))
153    }
154}

156/// Put a collection in canonical order, in place.
157///
158/// The sort is stable, but nothing relies on that: the identity part of the key makes the
159/// order total, so the result does not depend on the order the items arrived in. That is
160/// the property [`is_canonical`] checks and the permutation tests below prove.
161pub fn canonical<T: Ordered>(items: &mut [T]) {
162    items.sort_by(|a, b| a.order_key().cmp(&b.order_key()));
163}
164
165/// Put a collection of plain strings in canonical order, in place.
166///
167/// A `Vec<String>` has no group, no rank and no identity apart from the string itself, so it
168/// cannot implement [`Ordered`] without a newtype nobody wants. It still has an order, and
169/// before this existed every caller picked one of two: `.sort()`, which is `Ord for str` and
170/// puts `case-107` before `case-9`, or a hand-written `sort_by(|a, b| natural_cmp(a, b))`,
171/// which is this function spelled out again at the call site. The second is the duplication
172/// `project.canonical-order` exists to prevent, and the first is a different order from the
173/// one every other collection in the crate is shown in.
174///
175/// ```
176/// use majordomus_cli::order::canonical_strings;
177///
178/// let mut cases = vec!["case-107".to_string(), "case-9".to_string(), "case-10".to_string()];
179/// canonical_strings(&mut cases);
180/// assert_eq!(cases, ["case-9", "case-10", "case-107"]);
181/// ```
182pub fn canonical_strings<S: AsRef<str>>(items: &mut [S]) {
183    items.sort_by(|a, b| natural_cmp(a.as_ref(), b.as_ref()));
184}

186/// Put a numeric sample in ascending order, in place.
187///
188/// A sample is not a collection a person reads item by item: it is ordered so that its
189/// order statistics — a median, a quantile, the bounds of a bootstrap interval — can be read
190/// off it. It still has exactly one order, IEEE 754's total order, in which every value,
191/// both zeros and a NaN included, has one fixed place, so the result never depends on the
192/// order the values arrived in. It is written here, once, instead of as a comparator beside
193/// every piece of arithmetic that needs one.
194///
195/// ```
196/// use majordomus_cli::order::numeric;
197///
198/// let mut sample = vec![0.5, 0.0, 0.25, -0.0];
199/// numeric(&mut sample);
200/// assert_eq!(sample, [0.0, 0.0, 0.25, 0.5]);
201/// assert!(sample[0].is_sign_negative(), "-0 has its own place, before +0");
202/// ```
203pub fn numeric(values: &mut [f64]) {
204    values.sort_by(f64::total_cmp);
205}
206
207/// Put a set of numeric samples in ascending order, in place: value by value in
208/// [`numeric`]'s order, and a sample that is a prefix of another before it. Once each sample
209/// is itself in [`numeric`] order, this is what makes a computation that draws among the
210/// samples independent of the order they were listed in.
211///
212/// ```
213/// use majordomus_cli::order::numeric_samples;
214///
215/// let mut samples = vec![vec![0.3, 0.4], vec![0.1], vec![0.3]];
216/// numeric_samples(&mut samples);
217/// assert_eq!(samples, [vec![0.1], vec![0.3], vec![0.3, 0.4]]);
218/// ```
219pub fn numeric_samples(samples: &mut [Vec<f64>]) {
220    samples.sort_by(|a, b| {
221        a.iter()
222            .zip(b.iter())
223            .map(|(x, y)| x.total_cmp(y))
224            .find(|o| o.is_ne())
225            .unwrap_or_else(|| a.len().cmp(&b.len()))
226    });
227}

229/// Compare two versions by precedence: the dot-separated numbers by value, then a release
230/// above any of its pre-releases (`1.0.0-rc.1` before `1.0.0`), pre-releases compared
231/// naturally (`rc.2` before `rc.10`). A leading `v` is not part of the version, so `v0.27.0`
232/// and `0.27.0` are equal. A string that is not a version precedes every version and equals
233/// every other non-version: it has no precedence to compare.
234///
235/// Equal is the right answer for one version spelled two ways — "is this release published
236/// yet" asks precedence, not spelling. [`versions`] adds the tie-breaker a sort needs.
237///
238/// ```
239/// use majordomus_cli::order::version_cmp;
240/// use std::cmp::Ordering;
241///
242/// assert_eq!(version_cmp("v0.9.1", "v0.10.0"), Ordering::Less);
243/// assert_eq!(version_cmp("v1.0.0-rc.1", "v1.0.0"), Ordering::Less);
244/// assert_eq!(version_cmp("v0.27.0", "0.27.0"), Ordering::Equal);
245/// assert_eq!(version_cmp("next", "v0.1.0"), Ordering::Less);
246/// ```
247pub fn version_cmp(a: &str, b: &str) -> Ordering {
248    match (version_parts(a), version_parts(b)) {
249        (Some(a), Some(b)) => a
250            .numbers
251            .cmp(&b.numbers)
252            .then_with(|| match (a.pre, b.pre) {
253                (Some(x), Some(y)) => natural_cmp(x, y),
254                (Some(_), None) => Ordering::Less,
255                (None, Some(_)) => Ordering::Greater,
256                (None, None) => Ordering::Equal,
257            }),
258        (Some(_), None) => Ordering::Greater,
259        (None, Some(_)) => Ordering::Less,
260        (None, None) => Ordering::Equal,
261    }
262}

264/// Whether `text` is a version [`version_cmp`] can order: dot-separated whole numbers, an
265/// optional leading `v` and an optional `-pre-release`.
266///
267/// ```
268/// use majordomus_cli::order::is_version;
269///
270/// assert!(is_version("v0.27.0") && is_version("1.0.0-rc.1"));
271/// assert!(!is_version("vnext") && !is_version("v") && !is_version("v1..2"));
272/// ```
273pub fn is_version(text: &str) -> bool {
274    version_parts(text).is_some()
275}

277/// Put a collection in version order, lowest first, in place: by [`version_cmp`] of what
278/// `version` reads from each item, then naturally by that text, so that one version spelled
279/// two ways still has one fixed place.
280///
281/// ```
282/// use majordomus_cli::order::versions;
283///
284/// let mut tags = vec![("v0.10.0", "c"), ("v0.10.0-rc.1", "b"), ("v0.9.0", "a")];
285/// versions(&mut tags, |t| t.0);
286/// assert_eq!(tags.iter().map(|t| t.1).collect::<String>(), "abc");
287/// ```
288pub fn versions<T>(items: &mut [T], version: impl Fn(&T) -> &str) {
289    items.sort_by(|a, b| {
290        let (a, b) = (version(a), version(b));
291        version_cmp(a, b).then_with(|| natural_cmp(a, b))
292    });
293}

295/// A version's parts: the numbers, and the pre-release when there is one.
296struct VersionParts<'a> {
297    numbers: Vec<u64>,
298    pre: Option<&'a str>,
299}
300
301fn version_parts(text: &str) -> Option<VersionParts<'_>> {
302    let v = text.strip_prefix('v').unwrap_or(text);
303    let (core, pre) = match v.split_once('-') {
304        Some((core, pre)) => (core, Some(pre)),
305        None => (v, None),
306    };
307    let numbers = core
308        .split('.')
309        .map(|n| n.parse::<u64>().ok())
310        .collect::<Option<Vec<u64>>>()?;
311    Some(VersionParts { numbers, pre })
312}

314/// Is this collection already in canonical order? What a validator asks of a projection it
315/// did not build, and what a test asks of an interface's output.
316pub fn is_canonical<T: Ordered>(items: &[T]) -> bool {
317    items
318        .windows(2)
319        .all(|pair| pair[0].order_key() <= pair[1].order_key())
320}

322/// A group compares naturally against another group; the absence of a group sorts last.
323fn group_cmp(a: Option<&str>, b: Option<&str>) -> Ordering {
324    match (a, b) {
325        (Some(a), Some(b)) => natural_cmp(a, b),
326        (Some(_), None) => Ordering::Less,
327        (None, Some(_)) => Ordering::Greater,
328        (None, None) => Ordering::Equal,
329    }
330}

332/// A bare string is ordered by itself.
333///
334/// A list of paths, ids or names is a collection like any other, and it had no way to reach
335/// the canonical order without a comparator written beside it. This is that way: one
336/// implementation, so `canonical` orders a `Vec<String>` the same way it orders everything
337/// else — naturally, and identically on every machine, which `sort()` does not promise
338/// because it compares bytes and `natural_cmp` compares what a person reads.
339///
340/// ```
341/// use majordomus_cli::order::canonical;
342///
343/// let mut paths = vec!["a/item-10".to_string(), "a/item-2".to_string()];
344/// canonical(&mut paths);
345/// assert_eq!(paths, ["a/item-2", "a/item-10"]);
346/// ```
347impl Ordered for String {
348    fn order_key(&self) -> OrderKey<'_> {
349        OrderKey::plain(self, self)
350    }
351}

353/// Compare two human-facing strings the way a person reads them.
354///
355/// A run of ASCII digits compares by value, so `item-2` precedes `item-10` instead of
356/// following it. Everything else compares by Unicode scalar, case-insensitively for ASCII
357/// letters first so that `Alpha` and `alpha` sort together, and case-sensitively after, so
358/// that two strings differing only in case still have a fixed order rather than comparing
359/// equal and letting the caller's input order decide.
360///
361/// Separators are not normalised. `a-b` and `a_b` are different strings and stay different:
362/// folding them together would make two distinct identities compare equal, which is exactly
363/// the tie this order exists to break.
364///
365/// ```
366/// use majordomus_cli::order::natural_cmp;
367/// use std::cmp::Ordering;
368///
369/// assert_eq!(natural_cmp("item-2", "item-10"), Ordering::Less);
370/// assert_eq!(natural_cmp("item-002", "item-2"), Ordering::Less);
371/// assert_eq!(natural_cmp("Alpha", "beta"), Ordering::Less);
372/// ```
373pub fn natural_cmp(a: &str, b: &str) -> Ordering {
374    let folded = fold_cmp(a, b);
375    if folded != Ordering::Equal {
376        return folded;
377    }
378    // Same word to a reader; still two different strings. Compare the raw scalars so that
379    // the pair has one fixed order on every machine.
380    exact_cmp(a, b)
381}
382
383/// The reader's comparison: digit runs by value, letters folded to lower case.
384fn fold_cmp(a: &str, b: &str) -> Ordering {
385    let (mut a, mut b) = (a.chars().peekable(), b.chars().peekable());
386    loop {
387        match (a.peek().copied(), b.peek().copied()) {
388            (None, None) => return Ordering::Equal,
389            (None, Some(_)) => return Ordering::Less,
390            (Some(_), None) => return Ordering::Greater,
391            (Some(x), Some(y)) => {
392                if x.is_ascii_digit() && y.is_ascii_digit() {
393                    let ordering = digits_cmp(&mut a, &mut b);
394                    if ordering != Ordering::Equal {
395                        return ordering;
396                    }
397                    continue;
398                }
399                let ordering = fold(x).cmp(&fold(y));
400                if ordering != Ordering::Equal {
401                    return ordering;
402                }
403                a.next();
404                b.next();
405            }
406        }
407    }
408}

410/// Compare the digit run at the head of each iterator by the number it spells, consuming
411/// both runs. A longer run of significant digits is the larger number; equal numbers are
412/// separated by how they were written, so that `002` and `2` have a fixed order.
413fn digits_cmp<I, J>(a: &mut std::iter::Peekable<I>, b: &mut std::iter::Peekable<J>) -> Ordering
414where
415    I: Iterator<Item = char>,
416    J: Iterator<Item = char>,
417{
418    let left = take_digits(a);
419    let right = take_digits(b);
420    let (lt, rt) = (left.trim_start_matches('0'), right.trim_start_matches('0'));
421    lt.len()
422        .cmp(&rt.len())
423        .then_with(|| lt.cmp(rt))
424        // Equal value, different spelling: the one written with more leading zeros first.
425        // Arbitrary, and fixed — which is the whole requirement.
426        .then_with(|| right.len().cmp(&left.len()))
427}
428
429fn take_digits<I: Iterator<Item = char>>(chars: &mut std::iter::Peekable<I>) -> String {
430    let mut run = String::new();
431    while let Some(c) = chars.peek().copied() {
432        if !c.is_ascii_digit() {
433            break;
434        }
435        run.push(c);
436        chars.next();
437    }
438    run
439}

441/// Scalar-by-scalar comparison, used only to separate two strings a reader would call the
442/// same word.
443fn exact_cmp(a: &str, b: &str) -> Ordering {
444    a.cmp(b)
445}
446
447fn fold(c: char) -> char {
448    c.to_ascii_lowercase()
449}

638        // Antisymmetry and transitivity over every pair and triple: a comparator that fails
639        // either can make sort() produce a different answer for a different input order.
640        for a in FIXTURE.iter() {
641            for b in FIXTURE.iter() {

662    /// Every permutation of a slice, smallest first. Seven items is 5040 orders, which is
663    /// cheap and exhaustive; a random shuffle would prove less and flake more.
664    fn permutations<T: Copy>(items: &[T]) -> Vec<Vec<T>> {
665        let mut out = Vec::new();
666        let mut current: Vec<T> = items.to_vec();
667        let n = current.len();
668        let mut counters = vec![0usize; n];
669        out.push(current.clone());
670        let mut i = 0;
671        while i < n {
672            if counters[i] < i {
673                let j = if i % 2 == 0 { 0 } else { counters[i] };
674                current.swap(j, i);
675                out.push(current.clone());
676                counters[i] += 1;
677                i = 0;
678            } else {
679                counters[i] = 0;
680                i += 1;
681            }
682        }
683        out
684    }
