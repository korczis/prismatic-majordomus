1//! Serving the static surfaces of a resolved topology.
2//!
3//! One arm in the router asks this: does a static surface own the path? Every discovered
4//! `StaticDirectory` is answered here, and a surface that did not exist when this file was
5//! written is served because it was discovered, not because an arm was added for it. The
6//! routes the executable answers itself — the JSON index, the OpenAPI document, the Swagger
7//! shell, MCP, the Cockpit — are not dispatched here: they differ in behaviour rather than
8//! in data, and a variant per existing arm would be the match again with more ceremony
9//! (ADR 0013).
10//!
11//! The boundary this module defends: only a resolved surface's own directory is reachable,
12//! a request may not walk out of it, and only a closed set of media types is answered.
13
14use std::collections::BTreeMap;
15use std::path::{Path, PathBuf};
16use std::sync::Mutex;
17
18use super::model::{SurfaceKind, Topology};
19use crate::http::router::Response;

21/// The media types a generated report may be served as. A file whose extension is not here
22/// is not answered: a static surface is a directory of documents, never a file server.
23const MEDIA_TYPES: &[(&str, &str)] = &[
24    ("html", "text/html; charset=utf-8"),
25    ("css", "text/css; charset=utf-8"),
26    ("js", "text/javascript; charset=utf-8"),
27    ("mjs", "text/javascript; charset=utf-8"),
28    ("json", "application/json"),
29    ("map", "application/json"),
30    ("svg", "image/svg+xml"),
31    ("txt", "text/plain; charset=utf-8"),
32    ("xml", "application/xml"),
33];
34
35/// The static surfaces of a topology, ready to answer.
36///
37/// Built once at startup from the resolved topology: the router holds it and asks it before
38/// its own routes, so a mount that a surface owns is never shadowed by a later arm.
39#[derive(Debug)]
40pub struct StaticSurfaces {
41    mounts: Vec<Mounted>,
42    /// The mounts the executable answers itself. The application is mounted at `/` and
43    /// therefore owns every path nothing else claims — including the ones the router
44    /// answers — so precedence is resolved from the topology here rather than left to the
45    /// order the router happens to consult its arms in.
46    native: Vec<String>,
47    cache: Mutex<BTreeMap<PathBuf, Option<(&'static str, String)>>>,
48}
49
50#[derive(Debug)]
51struct Mounted {
52    id: String,
53    prefix: String,
54    root: PathBuf,
55    index: String,
56}
57
58impl StaticSurfaces {
59    /// Take every static surface of `topology` whose directory exists, resolved against
60    /// `root`. A surface whose producer has not run is not mounted: the validator is where
61    /// that becomes a finding, and a router that answered 500 for it would be worse.
62    pub fn new(topology: &Topology, root: &Path) -> Self {
63        let mut mounts = Vec::new();
64        let mut native = Vec::new();
65        for surface in &topology.surfaces {
66            if surface.kind != SurfaceKind::StaticDirectory {
67                native.push(surface.mount.as_str().to_string());
68                continue;
69            }
70            let Some(artifact) = surface.artifact.as_ref() else {
71                native.push(surface.mount.as_str().to_string());
72                continue;
73            };
74            let dir = root.join(artifact);
75            let Ok(canonical) = dir.canonicalize() else {
76                // Declared but not built. Its mount stays reserved rather than falling to
77                // whoever holds the prefix above it: the application mounted at the root
78                // would otherwise answer /docs with a page of its own the moment the
79                // documentation had not been generated, which is the one thing the
80                // reserved namespaces exist to prevent. The router answers the unbuilt
81                // mount itself, naming the command that builds it.
82                native.push(surface.mount.as_str().to_string());
83                continue;
84            };
85            mounts.push(Mounted {
86                id: surface.id.clone(),
87                prefix: surface.mount.prefix(),
88                root: canonical,
89                index: surface.index.clone().unwrap_or_else(|| "index.html".into()),
90            });
91        }
92        // most specific first, so a surface inside another's prefix is consulted first; the
93        // validator refuses that arrangement, and the order does not depend on it
94        mounts.sort_by(|a, b| b.prefix.len().cmp(&a.prefix.len()).then(a.id.cmp(&b.id)));
95        native.sort();
96        StaticSurfaces {
97            mounts,
98            native,
99            cache: Mutex::new(BTreeMap::new()),
100        }
101    }
102
103    /// The ids mounted, in the order they are consulted.
104    pub fn ids(&self) -> Vec<&str> {
105        self.mounts.iter().map(|m| m.id.as_str()).collect()
106    }
107
108    /// Is anything mounted?
109    pub fn is_empty(&self) -> bool {
110        self.mounts.is_empty()
111    }
112
113    /// Does a static surface own this path — and is it not one the executable answers?
114    pub fn owns(&self, path: &str) -> bool {
115        !self.reserved(path) && self.mounts.iter().any(|m| owns(m, path))
116    }
117
118    /// Is this path one a native route of the topology owns? Those are answered by the
119    /// router itself, and the application's catch-all mount may not take them.
120    fn reserved(&self, path: &str) -> bool {
121        self.native.iter().any(|mount| {
122            let bare = mount.trim_end_matches('/');
123            !bare.is_empty() && (path == bare || path.starts_with(&format!("{bare}/")))
124        })
125    }
126
127    /// Answer a request a static surface owns, or `None` when none does.
128    ///
129    /// A directory answers with its index; a path that walks out of the surface's root, or
130    /// names a media type the set does not hold, is a 404 rather than a hint about what is
131    /// on the disk.
132    pub fn handle(&self, method: &str, path: &str) -> Option<Response> {
133        if self.reserved(path) {
134            return None; // the executable answers this one itself
135        }
136        let mounted = self.mounts.iter().find(|m| owns(m, path))?;
137        if method != "GET" && method != "HEAD" {
138            return Some(Response::new(
139                405,
140                "application/json",
141                r#"{"error":"method_not_allowed","message":"a static surface answers GET"}"#,
142            ));
143        }
144        // exactly one separator is removed: `//` is a different URL, and collapsing it here
145        // would hide an empty segment that safe_relative exists to refuse
146        let after = path
147            .strip_prefix(mounted.prefix.trim_end_matches('/'))
148            .unwrap_or("");
149        let relative = after.strip_prefix('/').unwrap_or(after);
150        let relative = if relative.is_empty() || relative.ends_with('/') {
151            format!("{relative}{}", mounted.index)
152        } else {
153            relative.to_string()
154        };
155        let Some(safe) = safe_relative(&relative) else {
156            return Some(not_found(&mounted.id));
157        };
158        let file = mounted.root.join(&safe);
159        match self.read(&file, &mounted.root) {
160            Some((media_type, body)) => Some(Response::new(200, media_type, body)),
161            None => Some(not_found(&mounted.id)),
162        }
163    }
164
165    /// Read a file once and keep it: a generated report is rebuilt by its producer, not by
166    /// a request, so the bytes cannot change under a running server without one.
167    fn read(&self, file: &Path, root: &Path) -> Option<(&'static str, String)> {
168        if let Ok(cache) = self.cache.lock() {
169            if let Some(hit) = cache.get(file) {
170                return hit.clone();
171            }
172        }
173        let loaded = load(file, root);
174        if let Ok(mut cache) = self.cache.lock() {
175            cache.insert(file.to_path_buf(), loaded.clone());
176        }
177        loaded
178    }
179}
180
181fn owns(mounted: &Mounted, path: &str) -> bool {
182    let bare = mounted.prefix.trim_end_matches('/');
183    if bare.is_empty() {
184        return path.starts_with('/');
185    }
186    path == bare || path.starts_with(&mounted.prefix)
187}
188
189fn load(file: &Path, root: &Path) -> Option<(&'static str, String)> {
190    let media_type = media_type(file)?;
191    // the relative path is already known to hold no `..`; canonicalising both ends closes
192    // the door a symbolic link inside the directory would otherwise open
193    let resolved = file.canonicalize().ok()?;
194    if !resolved.starts_with(root) {
195        tracing::warn!(path = %file.display(), "a static surface's file resolves outside its root; refused");
196        return None;
197    }
198    let body = std::fs::read_to_string(&resolved).ok()?;
199    Some((media_type, body))
200}

202/// The media type of a file, or `None` when the set does not hold it.
203///
204/// ```
205/// use majordomus_cli::web::serve::media_type;
206/// use std::path::Path;
207/// assert_eq!(media_type(Path::new("a/index.html")), Some("text/html; charset=utf-8"));
208/// assert_eq!(media_type(Path::new("a/results.json")), Some("application/json"));
209/// assert_eq!(media_type(Path::new("a/binary.wasm")), None);
210/// assert_eq!(media_type(Path::new("noextension")), None);
211/// ```
212pub fn media_type(file: &Path) -> Option<&'static str> {
213    let extension = file.extension()?.to_str()?.to_ascii_lowercase();
214    MEDIA_TYPES
215        .iter()
216        .find(|(e, _)| *e == extension)
217        .map(|(_, t)| *t)
218}

220/// A request path's relative part, refused before the filesystem is touched when it holds
221/// anything but the characters a generated report uses, an empty segment, `.` or `..`.
222///
223/// ```
224/// use majordomus_cli::web::serve::safe_relative;
225/// assert_eq!(safe_relative("index.html").as_deref(), Some("index.html"));
226/// assert_eq!(safe_relative("coverage/index.html").as_deref(), Some("coverage/index.html"));
227/// assert_eq!(safe_relative("../../etc/passwd"), None);
228/// assert_eq!(safe_relative("a//b"), None);
229/// assert_eq!(safe_relative("a/./b"), None);
230/// ```
231pub fn safe_relative(relative: &str) -> Option<String> {
232    if relative.is_empty() || relative.len() > 300 {
233        return None;
234    }
235    if !relative
236        .bytes()
237        .all(|b| b.is_ascii_alphanumeric() || b"._-/".contains(&b))
238    {
239        return None;
240    }
241    let segments: Vec<&str> = relative.split('/').collect();
242    if segments
243        .iter()
244        .any(|s| s.is_empty() || *s == "." || *s == "..")
245    {
246        return None;
247    }
248    Some(segments.join("/"))
249}
250
251fn not_found(surface: &str) -> Response {
252    Response::new(
253        404,
254        "application/json",
255        format!(
256            r#"{{"error":"not_found","message":"the surface '{surface}' does not hold that path"}}"#
257        ),
258    )
259}

382    /// The documentation is a static surface like the application, and until its producer
383    /// has run it is a mount with nothing behind it. The application holds `/`, so the one
