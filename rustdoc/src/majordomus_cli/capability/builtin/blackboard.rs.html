1//! The `blackboard` module: who works on what in this repository, as one projection (ADR 0176).
2//!
3//! Three records already know it, each about its own half: the board of every checkout
4//! (`peers.list`, gathered per ADR 0044) knows which clients are attached and what they
5//! announced; the mesh journal (ADR 0067) knows the sessions, claims, reviews and handovers of
6//! every linked machine, this repository's board peers among them as the `board-pN` sessions
7//! their own server projected; the monitor (ADR 0133) knows what happened, folded, with a
8//! cursor. This module joins them once, so that no surface joins them again. It writes nothing
9//! and keeps nothing.
10//!
11//! The join is the whole of what is new here, and it is deliberately narrow: a board peer and
12//! a mesh session are one worker only when an explicit identifier says so (see [`join`]). A
13//! peer id such as `p1` is a position on one board, handed out again on every reconnect and by
14//! every other board, so it is never an identity on its own.

434/// The mesh sessions the board peer's own server projected for it.
435///
436/// The server holding a checkout's board writes each attached peer into its own mesh stream
437/// as the session `board-<peer>`, with the checkout's identity and the client's name
438/// (`Cooperation::project_board`). A peer id is a position on that server's board, handed out
439/// again by every server and after every restart, so the join is bound to the stream the
440/// board's server names (`BoardView::stream`, which carries the process instance): a session of
441/// that stream naming the same board session, checkout and client is that peer. A board whose
442/// server names no stream joins nothing. More than one fitting session is a candidate each.
443pub(crate) fn join<'s>(
444    peer: &Peer,
445    checkout: &str,
446    stream: Option<&str>,
447    sessions: &[&'s SessionView],
448) -> Vec<&'s SessionView> {
449    let Some(stream) = stream else {
450        return Vec::new();
451    };
452    let name = format!("board-{}", peer.id);
453    // cut exactly as the projection cut it: bytes on a character boundary, controls removed
454    let client = crate::mesh::cooperation::clip(&peer.client.name, 64);
455    sessions
456        .iter()
457        .copied()
458        .filter(|s| {
459            s.state == SessionState::Active
460                && s.stream.as_str() == stream
461                && s.info.session == name
462                && s.info.checkout.as_deref() == Some(checkout)
463                && s.info.client == client
464        })
465        .collect()
466}

468fn board_key(checkout: &str, peer: &Peer) -> String {
469    format!("board:{checkout}/{}", peer.id)
470}
471
472fn mesh_key(session: &SessionView) -> String {
473    format!("mesh:{}", session.key)
474}

476/// Join the reads into the blackboard.
477pub(crate) fn project(
478    reads: &Reads<'_>,
479    at: String,
480    checkout: String,
481    head: Option<String>,
482) -> Blackboard {
483    let mut sources = Vec::new();
484    let unread: Vec<&str> = reads
485        .board
486        .boards
487        .iter()
488        .filter_map(|b| b.reason.as_deref())
489        .collect();
490    sources.push(BlackboardSource {
491        source: "board".into(),
492        state: if reads.board.complete {
493            BlackboardSourceState::Ok
494        } else {
495            BlackboardSourceState::Partial
496        },
497        reason: (!unread.is_empty()).then(|| unread.join("; ")),
498    });
499    let (mesh_state, mesh_source) = match &reads.mesh {
500        Ok(state) => (Some(*state), (BlackboardSourceState::Ok, None)),
501        Err(MeshAbsence::Undeclared(why)) => (None, (BlackboardSourceState::Skipped, Some(why.clone()))),
502        Err(MeshAbsence::Unreachable(why)) => (
503            None,
504            (
505                BlackboardSourceState::Absent,
506                Some(format!("{why}: other machines and the journal's claims, reviews and handovers are not covered")),
507            ),
508        ),
509    };
510    sources.push(BlackboardSource {
511        source: "mesh".into(),
512        state: mesh_source.0,
513        reason: mesh_source.1,
514    });
515    // who is here is the boards' and the mesh's to say: the monitor's sources are what
516    // happened, and one that has recorded nothing yet is not a worker nobody could ask
517    let complete = sources.iter().all(|s| {
518        matches!(
519            s.state,
520            BlackboardSourceState::Ok | BlackboardSourceState::Skipped
521        )
522    });
523    for s in &reads.monitor.sources {
524        sources.push(monitor_source(s));
525    }
526
527    let active: Vec<&SessionView> = mesh_state
528        .map(|m| {
529            m.sessions
530                .iter()
531                .filter(|s| s.state == SessionState::Active)
532                .collect()
533        })
534        .unwrap_or_default();
535    let claims_of = |session: &SessionView| -> Vec<BlackboardClaim> {
536        mesh_state
537            .map(|m| {
538                m.claims
539                    .iter()
540                    .filter(|c| c.session == session.key && c.state.is_live())
541                    .map(|c| BlackboardClaim {
542                        name: None,
543                        intent: c.intent.clone(),
544                        scope: c.scope.clone(),
545                        mode: c.mode,
546                        issue: c.issue.clone(),
547                        record: Some(c.key.clone()),
548                        conflicted: matches!(c.state, ClaimState::Conflicted(_)),
549                    })
550                    .collect()
551            })
552            .unwrap_or_default()
553    };
554    let mut workers = Vec::new();
555    let mut relations = Vec::new();
556    // session key → the worker that holds it
557    let mut held: BTreeMap<String, String> = BTreeMap::new();
558    for peer in &reads.board.peers {
559        let Some(checkout) = peer.checkout.as_ref() else {
560            continue;
561        };
562        let key = board_key(&checkout.id, peer);
563        let stream = reads
564            .board
565            .boards
566            .iter()
567            .find(|b| b.checkout.id == checkout.id)
568            .and_then(|b| b.stream.as_deref());
569        let fits = join(peer, &checkout.id, stream, &active);
570        let joined = (fits.len() == 1).then(|| fits[0]);
571        for s in &fits {
572            relations.push(BlackboardRelation {
573                kind: BlackboardRelationKind::SameWorker,
574                evidence: if joined.is_some() {
575                    BlackboardEvidence::Verified
576                } else {
577                    BlackboardEvidence::Candidate
578                },
579                from: key.clone(),
580                to: mesh_key(s),
581                record: s.key.clone(),
582            });
583        }
584        if let Some(s) = joined {
585            held.insert(s.key.clone(), key.clone());
586        }
587        let beat = joined
588            .and_then(|s| reads.beats.get(&s.runtime))
589            .and_then(|b| b.1);

731/// A board peer's claims: its announcements, and the live claims its joined session holds
732/// beside them — an exclusive `mesh.claim` taken through its MCP session, say. The advisory
733/// claims its server projected from the announcements are the announcements themselves and
734/// are not counted twice; a projected claim is one that carries exactly what the projection
735/// writes (`Cooperation::project_board`): advisory, the announcement's scope and clipped
736/// intent, no issue. Each announcement stands for one such claim. Any other claim of the
737/// session, a same-scope advisory claim with its own intent or issue among them, is kept.
738fn claims_of_peer(peer: &Peer, session: Vec<BlackboardClaim>) -> Vec<BlackboardClaim> {
739    let normal = |scope: &[String]| -> Vec<String> {
740        scope
741            .iter()
742            .map(|p| p.trim().trim_start_matches("./").to_string())
743            .filter(|p| !p.is_empty())
744            .collect()
745    };
746    let mut claims: Vec<BlackboardClaim> = peer
747        .claims
748        .iter()
749        .map(|c| BlackboardClaim {
750            name: c.name.clone(),
751            intent: Some(c.intent.clone()).filter(|i| !i.is_empty()),
752            scope: c.scope.clone(),
753            mode: ClaimMode::Advisory,
754            issue: None,
755            record: None,
756            conflicted: false,
757        })
758        .collect();
759    // what the projection wrote for each announcement, each matched at most once
760    let mut projected: Vec<Option<(Vec<String>, Option<String>)>> = peer
761        .claims
762        .iter()
763        .map(|c| {
764            let intent = crate::mesh::cooperation::clip(&c.intent, 512);
765            Some((normal(&c.scope), Some(intent).filter(|i| !i.is_empty())))
766        })
767        .collect();
768    claims.extend(session.into_iter().filter(|c| {
769        if c.mode != ClaimMode::Advisory || c.conflicted || c.issue.is_some() {
770            return true;
771        }
772        let this = (normal(&c.scope), c.intent.clone());
773        match projected.iter_mut().find(|p| p.as_ref() == Some(&this)) {
774            Some(slot) => {
775                *slot = None;
776                false
777            }
778            None => true,
779        }
780    }));
781    claims
782}

802/// Every pair of claims of two different workers whose scopes meet, confirmed first, at most
803/// [`MAX_CONFLICTS`], with how many there are in all. A worker and a candidate session of its
804/// own are not two workers for this: the session's claims are the projection of the peer's.
805fn conflicts_among(
806    workers: &[BlackboardWorker],
807    candidates: &BTreeMap<String, String>,
808) -> (usize, Vec<BlackboardConflict>) {
809    let same = |a: &str, b: &str| {
810        candidates.get(a).is_some_and(|w| w == b) || candidates.get(b).is_some_and(|w| w == a)
811    };
812    let mut confirmed = Vec::new();
813    let mut potential = Vec::new();
814    for (i, first) in workers.iter().enumerate() {
815        for second in workers.iter().skip(i + 1) {
816            if same(&first.key, &second.key) {
817                continue;
818            }
819            for a in &first.claims {
820                for b in &second.claims {
821                    let paths: Vec<OverlapPath> = a
822                        .scope
823                        .iter()
824                        .flat_map(|x| {
825                            b.scope
826                                .iter()
827                                .filter(|y| claims_meet(x, y))
828                                .map(|y| OverlapPath {
829                                    yours: x.clone(),
830                                    theirs: y.clone(),
831                                })
832                        })
833                        .collect();
834                    if paths.is_empty() {
835                        continue;
836                    }
837                    let both_exclusive =
838                        a.mode == ClaimMode::Exclusive && b.mode == ClaimMode::Exclusive;
839                    let certainty = if both_exclusive || a.conflicted || b.conflicted {
840                        BlackboardCertainty::Confirmed
841                    } else {
842                        BlackboardCertainty::Potential
843                    };
844                    let conflict = BlackboardConflict {
845                        certainty,
846                        first: claim_ref(first, a),
847                        second: claim_ref(second, b),
848                        paths,
849                    };
850                    match certainty {
851                        BlackboardCertainty::Confirmed => confirmed.push(conflict),
852                        BlackboardCertainty::Potential => potential.push(conflict),
853                    }
854                }
855            }
856        }
857    }
858    let total = confirmed.len() + potential.len();
859    confirmed.extend(potential);
860    confirmed.truncate(MAX_CONFLICTS);
861    (total, confirmed)
862}

913/// The blackboard over `board`: the signals and the mesh read, and every source joined.
914fn read(
915    ctx: &Context,
916    input: BlackboardInput,
917    board: &PeerList,
918) -> Result<Blackboard, CapabilityError> {
919    let wanted = input.signals.unwrap_or(DEFAULT_SIGNALS).min(MAX_SIGNALS);
920    // no signals asked for is no monitor read: the journal is read once less, which is most
921    // of what the answer costs
922    let monitor = if wanted == 0 {
923        MonitorPage::default()
924    } else {
925        super::monitor::monitor(
926            ctx,
927            MonitorInput {
928                after: None,
929                limit: Some(wanted),
930                sources: Some(input.sources.unwrap_or_else(|| DEFAULT_SOURCES.into())),
931                reader: input.reader,
932                since: None,
933            },
934        )?
935    };
936    let cooperation = ctx.mesh.cooperation();
937    let state = cooperation.as_ref().map(|c| c.state());
938    let beats: HashMap<String, (StreamLiveness, Option<u64>)> = cooperation
939        .as_ref()
940        .map(|c| {
941            c.streams()
942                .into_iter()
943                .map(|(stream, liveness, beat)| (stream.runtime_key(), (liveness, beat)))
944                .collect()
945        })
946        .unwrap_or_default();
947    let machines = machines(ctx.mesh.nodes(), ctx.mesh.status().identity);
948    let reason = ctx
949        .mesh
950        .cooperation_status()
951        .reason
952        .unwrap_or_else(|| "the mesh is not running in this server".to_string());
953    let absence = absence(super::mesh::resolved_declaration(ctx).declaration, reason);
954    let root = PathBuf::from(&ctx.index.repository.root);
955    let root = root.canonicalize().unwrap_or(root);
956    let head = crate::worktree::git::head_of(&root).ok().flatten();
957    Ok(project(
958        &Reads {
959            board,
960            mesh: state.as_ref().ok_or(absence),
961            machines: &machines,
962            beats: &beats,
963            monitor: &monitor,
964        },
965        rfc3339(SystemTime::now()),
966        repository::identity(&root),
967        head,
968    ))
969}
