1//! Evidence subjects: every declaration a verdict can be asked about, what it is made of,
2//! and the tests it reaches.
3//!
4//! A *subject* is a declaration a person or a page asks "is this proven?" about: a claim of
5//! `docs/CLAIMS.yaml`, a rule of the layer, a feature, a public command, a builtin capability
6//! of the executable, an MCP tool (an alias of its capability) or a use case. Its key is
7//! `<kind>:<id>` — `claim:scope-integrity`, `command:commit plan`, `mcp:majordomus_get`.
8//!
9//! # Derived only
10//!
11//! Nothing lists a subject, a member or a route by hand. [`index`] reads the declarations
12//! that already exist and nothing else:
13//!
14//! * the claims, through the evidence join ([`crate::evidence::report`]) with an empty
15//!   ledger, so a claim's test identity is the one grammar [`TestId::of`] owns;
16//! * the rules, through [`crate::rules::definitions`];
17//! * the features, through the product model;
18//! * the use cases and the public shell commands, from the index;
19//! * the documented command-line examples ([`crate::cli::EXAMPLES`]);
20//! * the builtin capabilities of the registry, with the file each was composed in;
21//! * the first `# majordomus-covers:` and the first `# majordomus-negative:` line of every
22//!   case under `test/cases/`, and the integration test binaries of the crate.
23//!
24//! Adding a declaration adds its subject, and removing it removes it. The committed index,
25//! [`PATH`], carries structure only — no ledger row, no commit — so recording a run never
26//! makes it stale, and `majordomus generate --check` refuses a copy that differs from this
27//! derivation.
28//!
29//! # The shell derivations it replaces or holds equal
30//!
31//! Before this index, six shell programs derived "which tests prove this subject":
32//!
33//! * the command pages (`scripts/generate-site-data`, `commands.json`) read the first
34//!   covers and negative header of every case — the index reads the same first line, and
35//!   case 507 holds the two equal;
36//! * the command-coverage doctrine (`lib/commands.sh`) reads the same first line;
37//! * the `command-furnished` gate and the use-case impact trace match a command as a whole
38//!   word of *any* header line, so they would read a second header of a kind where the index
39//!   reads only the first; a prefixed word such as `capability:commit.plan` is a command for
40//!   none of them — case 507 holds every case to one header of each kind, and every header
41//!   to naming the same public commands under both readings, so they cannot yet disagree;
42//! * the capability pages (`scripts/lib/executable-site.jq`) give every capability the
43//!   claims implemented in its module's file — the index gives `capability:<id>` the same
44//!   claim members, and case 507 holds the two equal;
45//! * the doctrine pages (`doctrines.json`) name the first test of each rule — case 507
46//!   holds every one of them to a route or a mechanism of `rule:<id>`.
47//!
48//! The command words are the command graph's ([`crate::command_graph`]) as well: the public
49//! commands of the shell tool and the runnable commands of the executable, which the
50//! documented examples name one for one. Case 507 holds the command subjects, and the
51//! programs each names, equal to `majordomus commands graph`.
52//!
53//! # Example
54//!
55//! ```
56//! use majordomus_cli::evidence::subject::{self, SubjectKind};
57//! use majordomus_cli::synthetic::SyntheticRepository;
58//!
59//! let repo = SyntheticRepository::small().unwrap();
60//! let ctx = repo.context().unwrap();
61//! let s = subject::index(&ctx.index, &ctx.product, &ctx.registry);
62//!
63//! // an MCP tool is an alias of the capability that declares it, never a subject of its own
64//! let tool = s.get("mcp:majordomus_capabilities").unwrap();
65//! let capability = tool.alias_of.as_deref().unwrap();
66//! assert_eq!(SubjectKind::parse(capability).unwrap().0, SubjectKind::Capability);
67//! assert_eq!(tool.members, vec![capability.to_string()]);
68//! assert!(tool.page.is_none(), "an alias has no page of its own");
69//! ```

136/// What kind of declaration a subject is. The declaration order is the order the committed
137/// index lists its kinds in.
138///
139/// ```
140/// use majordomus_cli::evidence::subject::SubjectKind;
141///
142/// let key = SubjectKind::Command.key("commit plan");
143/// assert_eq!(key, "command:commit plan");
144/// assert_eq!(SubjectKind::parse(&key), Some((SubjectKind::Command, "commit plan")));
145/// assert_eq!(serde_json::to_value(SubjectKind::UseCase).unwrap(), "use_case");
146/// ```
147#[derive(
148    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
149)]
150#[serde(rename_all = "snake_case")]
151#[schemars(rename = "EvidenceSubjectKind")]
152pub enum SubjectKind {
153    /// A claim of `docs/CLAIMS.yaml`.
154    Claim,
155    /// A rule of the layer, by id without its version.
156    Rule,
157    /// A feature under `.ai/repo/features/`.
158    Feature,
159    /// A public shell command or a native command path, its words joined by one space.
160    Command,
161    /// A builtin capability of the executable.
162    Capability,
163    /// An MCP tool name: an alias of its capability.
164    Mcp,
165    /// A use case under `.ai/repo/use-cases/`.
166    UseCase,
167}
168
169impl SubjectKind {
170    /// Every kind, in declaration order.
171    pub const ALL: [SubjectKind; 7] = [
172        SubjectKind::Claim,
173        SubjectKind::Rule,
174        SubjectKind::Feature,
175        SubjectKind::Command,
176        SubjectKind::Capability,
177        SubjectKind::Mcp,
178        SubjectKind::UseCase,
179    ];
180
181    /// The word a key starts with: the kind's serialised name.
182    ///
183    /// ```
184    /// use majordomus_cli::evidence::subject::SubjectKind;
185    /// assert_eq!(SubjectKind::Claim.prefix(), "claim");
186    /// assert_eq!(SubjectKind::UseCase.prefix(), "use_case");
187    /// ```
188    pub fn prefix(self) -> &'static str {
189        match self {
190            SubjectKind::Claim => "claim",
191            SubjectKind::Rule => "rule",
192            SubjectKind::Feature => "feature",
193            SubjectKind::Command => "command",
194            SubjectKind::Capability => "capability",
195            SubjectKind::Mcp => "mcp",
196            SubjectKind::UseCase => "use_case",
197        }
198    }
199
200    /// The key of the subject of this kind with this id: `<prefix>:<id>`.
201    ///
202    /// ```
203    /// use majordomus_cli::evidence::subject::SubjectKind;
204    /// assert_eq!(SubjectKind::Rule.key("project.scope"), "rule:project.scope");
205    /// ```
206    pub fn key(self, id: &str) -> String {
207        format!("{}:{id}", self.prefix())
208    }
209
210    /// Read a key back into its kind and its id, or `None` when it is not a subject key.
211    ///
212    /// The key splits at its **first** `:`, so an id may carry colons and spaces of its own;
213    /// the prefix must be a kind's, and the id must not be empty.
214    ///
215    /// ```
216    /// use majordomus_cli::evidence::subject::SubjectKind;
217    /// assert_eq!(
218    ///     SubjectKind::parse("command:commit plan"),
219    ///     Some((SubjectKind::Command, "commit plan"))
220    /// );
221    /// assert_eq!(SubjectKind::parse("gate:x"), None, "not a kind");
222    /// assert_eq!(SubjectKind::parse("claim:"), None, "no id");
223    /// assert_eq!(SubjectKind::parse("nonsense"), None);
224    /// ```
225    pub fn parse(key: &str) -> Option<(SubjectKind, &str)> {
226        let (prefix, id) = key.split_once(':')?;
227        if id.is_empty() {
228            return None;
229        }
230        let kind = SubjectKind::ALL
231            .into_iter()
232            .find(|k| k.prefix() == prefix)?;
233        Some((kind, id))
234    }
235
236    /// The site section a subject of this kind has its page in, or `None` when the kind has
237    /// no page of its own (a rule, an MCP alias).
238    ///
239    /// ```
240    /// use majordomus_cli::evidence::subject::SubjectKind;
241    /// assert_eq!(SubjectKind::UseCase.section(), Some("use-cases"));
242    /// assert_eq!(SubjectKind::Mcp.section(), None);
243    /// ```
244    pub fn section(self) -> Option<&'static str> {
245        match self {
246            SubjectKind::Claim => Some("claims"),
247            SubjectKind::Rule => None,
248            SubjectKind::Feature => Some("features"),
249            SubjectKind::Command => Some("commands"),
250            SubjectKind::Capability => Some("capabilities"),
251            SubjectKind::Mcp => None,
252            SubjectKind::UseCase => Some("use-cases"),
253        }
254    }
255}

272    Claim,
273    /// A test a rule's enforcement block names.
274    Rule,
275    /// A case whose first covers header names the subject.
276    Behaviour,
277    /// A case whose first negative header names the subject.
278    Negative,

280    Example,
281    /// A use case's own scenario, which the ledger does not record.
282    Scenario,
283}

285/// What kind of test a route reaches, read off the test's identity and never declared.
286///
287/// ```
288/// use majordomus_cli::evidence::subject::Category;
289/// assert_eq!(Category::of("suite:07_scope"), Some(Category::E2e));
290/// assert_eq!(Category::of("crate:why"), Some(Category::Integration));
291/// assert_eq!(Category::of("lib/x.sh"), None);
292/// ```
293#[derive(
294    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
295)]
296#[serde(rename_all = "snake_case")]
297#[schemars(rename = "EvidenceCategory")]
298pub enum Category {
299    /// A behavioural case of the suite, driving the tools end to end.
300    E2e,
301    /// An integration test binary of the crate.
302    Integration,
303    /// A unit test inside the library (no runner records one yet).
304    Unit,
305    /// A documentation example (no runner records one yet).
306    Doctest,
307    /// A use case's scenario.
308    Scenario,
309}
310
311impl Category {
312    /// Every category, in declaration order.
313    pub const ALL: [Category; 5] = [
314        Category::E2e,
315        Category::Integration,
316        Category::Unit,
317        Category::Doctest,
318        Category::Scenario,
319    ];
320
321    /// The category of a test identity, or `None` when no runner's prefix is on it. The
322    /// first matching prefix wins, so a library unit test is never read as an integration
323    /// binary.
324    ///
325    /// ```
326    /// use majordomus_cli::evidence::subject::Category;
327    /// assert_eq!(Category::of("crate:lib::evidence::x"), Some(Category::Unit));
328    /// assert_eq!(Category::of("crate:cli_examples"), Some(Category::Integration));
329    /// assert_eq!(Category::of("doc:x"), Some(Category::Doctest));
330    /// ```
331    pub fn of(test: &str) -> Option<Category> {
332        if test.starts_with("suite:") {
333            Some(Category::E2e)
334        } else if test.starts_with("crate:lib::") {
335            Some(Category::Unit)
336        } else if test.starts_with("crate:") {
337            Some(Category::Integration)
338        } else if test.starts_with("doc:") {
339            Some(Category::Doctest)
340        } else {
341            None
342        }
343    }
344}

412/// Which program answers to a command word: the shell tool, or the Rust executable. A word
413/// both answer to names both, in this order.
414///
415/// These are two of the command graph's [`crate::command_graph::Origin`]s, read from the

431    /// `cli::validate` gives every runnable command an example and refuses one on a group,
432    /// so these are exactly the executable's runnable commands.
433    Native,
434    /// The shell tool: a public command of its registry.
435    Shell,
436}

583    /// let ctx = repo.context().unwrap();

585    /// let paths: Vec<&str> = pages.iter().map(|p| p.path.as_str()).collect();
586    /// assert!(paths.windows(2).all(|w| w[0] < w[1]), "ordered by path, each once");
587    /// ```
588    pub fn pages(&self) -> Vec<Page> {
589        let mut by_path: BTreeMap<String, Page> = BTreeMap::new();
590        for (key, e) in &self.subjects {
591            if let Some(path) = &e.page {
592                by_path.entry(path.clone()).or_insert_with(|| Page {
593                    kind: e.kind,
594                    subject: key.clone(),
595                    path: path.clone(),
596                });
597            }
598        }
599        by_path.into_values().collect()
600    }
601
602    /// Every page path more than one subject claims, with those subjects' keys. Empty on a
603    /// sound index; [`artifact`] refuses to write one that is not.
604    ///
605    /// ```
606    /// use majordomus_cli::evidence::subject;
607    /// use majordomus_cli::synthetic::SyntheticRepository;
608    ///
609    /// let repo = SyntheticRepository::small().unwrap();
610    /// let ctx = repo.context().unwrap();
611    /// let s = subject::index(&ctx.index, &ctx.product, &ctx.registry);
612    /// assert!(s.duplicate_pages().is_empty());
613    /// ```
614    pub fn duplicate_pages(&self) -> Vec<(String, Vec<String>)> {
615        let mut by_path: BTreeMap<&str, Vec<String>> = BTreeMap::new();
616        for (key, e) in &self.subjects {
617            if let Some(path) = &e.page {
618                by_path.entry(path.as_str()).or_default().push(key.clone());
619            }
620        }
621        by_path
622            .into_iter()
623            .filter(|(_, keys)| keys.len() > 1)
624            .map(|(path, keys)| (path.to_string(), keys))
625            .collect()
626    }
627
628    /// Every route a subject reaches: its own routes, then each member's, depth first in
629    /// member order, each subject walked once and each route listed once.
630    ///
631    /// This is the one walk of the member graph; the structural finding of a feature is
632    /// read off it, and a surface that needs what a subject reaches calls it rather than
633    /// walking the members again.
634    ///
635    /// ```
636    /// use majordomus_cli::evidence::subject::{self, Via};
637    /// use majordomus_cli::synthetic::SyntheticRepository;
638    ///
639    /// let repo = SyntheticRepository::small().unwrap();
640    /// let ctx = repo.context().unwrap();
641    /// let s = subject::index(&ctx.index, &ctx.product, &ctx.registry);
642    /// // an alias reaches what its capability reaches, and nothing the tree does not carry
643    /// let reached = s.reach("mcp:majordomus_capabilities").unwrap();
644    /// assert!(reached.iter().all(|r| r.via != Via::Scenario));
645    /// assert!(s.reach("feature:no-such").is_err());
646    /// ```
647    pub fn reach(&self, key: &str) -> Result<Vec<&Route>, UnknownSubject> {
648        let (key, entry) = self
649            .subjects
650            .get_key_value(key)
651            .ok_or_else(|| UnknownSubject {
652                key: key.to_string(),
653            })?;
654        let mut walked: BTreeSet<&str> = BTreeSet::new();
655        let mut listed: BTreeSet<&Route> = BTreeSet::new();
656        let mut out = Vec::new();
657        self.walk(key, entry, &mut walked, &mut listed, &mut out);
658        Ok(out)
659    }
660
661    /// One step of [`SubjectIndex::reach`]. A member is always a key of the index, because
662    /// `derive` adds only members that exist; one that did not would be passed over.
663    fn walk<'s>(
664        &'s self,
665        key: &'s str,
666        entry: &'s SubjectEntry,
667        walked: &mut BTreeSet<&'s str>,
668        listed: &mut BTreeSet<&'s Route>,
669        out: &mut Vec<&'s Route>,
670    ) {
671        if !walked.insert(key) {
672            return;
673        }
674        for route in &entry.routes {
675            if listed.insert(route) {
676                out.push(route);
677            }
678        }
679        for (member, e) in entry
680            .members
681            .iter()
682            .filter_map(|m| self.subjects.get_key_value(m))
683        {
684            self.walk(member, e, walked, listed, out);
685        }
686    }
687
688    /// The committed index as a document: the header, the kinds, the categories, the pages,
689    /// every subject by key and every test nothing reaches, in that order. Structure only:
690    /// no commit, no ledger row and no time.
691    ///
692    /// ```
693    /// use majordomus_cli::evidence::subject::{self, SCHEMA};
694    /// use majordomus_cli::synthetic::SyntheticRepository;
695    ///
696    /// let repo = SyntheticRepository::small().unwrap();
697    /// let ctx = repo.context().unwrap();
698    /// let doc = subject::index(&ctx.index, &ctx.product, &ctx.registry).document("0.0.0");
699    /// assert_eq!(doc["schema"], SCHEMA);
700    /// assert_eq!(doc["generator"]["version"], "0.0.0");
701    /// let members: Vec<&String> = doc.as_object().unwrap().keys().collect();
702    /// assert_eq!(members.first().map(|k| k.as_str()), Some("schema"));
703    /// assert_eq!(members.last().map(|k| k.as_str()), Some("excluded"));
704    /// ```
705    pub fn document(&self, version: &str) -> Value {
706        let kinds: Vec<Value> = SubjectKind::ALL
707            .into_iter()
708            .map(|k| match k.section() {
709                Some(section) => json!({ "kind": k, "section": section }),
710                None => json!({ "kind": k }),
711            })
712            .collect();
713        json!({
714            "schema": SCHEMA,
715            "generated": crate::generate::json_banner(SOURCE),
716            "generator": { "id": "majordomus-cli", "version": version },
717            "route": ROUTE,
718            "kinds": kinds,
719            "categories": Category::ALL,
720            "pages": self.pages(),
721            "subjects": self.subjects,
722            "excluded": self.excluded,
723        })
724    }
725}

935/// The words of the first line of `text` that starts with `header`, `none` dropped. The
936/// first line decides, as the command pages read it; a later line of the same kind names
937/// nothing.
938fn header_words(text: &str, header: &str) -> Vec<String> {
939    text.lines()
940        .find_map(|line| line.strip_prefix(header))
941        .map(|rest| {
942            rest.split_whitespace()
943                .filter(|w| *w != "none")
944                .map(str::to_string)
945                .collect()
946        })
947        .unwrap_or_default()
948}

950/// The regular files of `root/dir` whose repository path `TestId::of` accepts, with that
951/// path. A directory that is not there holds none.
952fn test_files(root: &Path, dir: &str) -> Vec<(String, TestId, PathBuf)> {
953    let Ok(entries) = std::fs::read_dir(root.join(dir)) else {
954        return Vec::new();
955    };
956    entries
957        .flatten()
958        .filter(|e| e.path().is_file())
959        .filter_map(|e| {
960            // read lossily rather than dropped: the suite's own glob runs a file whatever
961            // bytes its name holds, so a test the runner sees is never one the index hides
962            let name = e.file_name().to_string_lossy().into_owned();
963            let path = format!("{dir}/{name}");
964            let test = TestId::of(&path)?;
965            Some((path, test, e.path()))
966        })
967        .collect()
968}

1014/// A subject while it is being derived: its kind and id as it was opened with, and every
1015/// collection a set, so that the result does not depend on the order the declarations came
1016/// in.
1017struct Draft {
1018    kind: SubjectKind,
1019    id: String,
1020    status: Option<String>,
1021    alias_of: Option<String>,
1022    programs: BTreeSet<Program>,
1023    members: BTreeSet<String>,
1024    routes: BTreeSet<Route>,
1025    mechanisms: BTreeSet<String>,
1026}

1028/// The draft of the subject of this kind with this id, opened empty the first time it is
1029/// asked for. Every draft is opened here, so its key, kind and id always agree.
1030fn draft_of<'d>(
1031    drafts: &'d mut BTreeMap<String, Draft>,
1032    kind: SubjectKind,
1033    id: &str,
1034) -> &'d mut Draft {
1035    drafts.entry(kind.key(id)).or_insert_with(|| Draft {
1036        kind,
1037        id: id.to_string(),
1038        status: None,
1039        alias_of: None,
1040        programs: BTreeSet::new(),
1041        members: BTreeSet::new(),
1042        routes: BTreeSet::new(),
1043        mechanisms: BTreeSet::new(),
1044    })
1045}
1046
1047impl Draft {
1048    /// Two declarations of one subject keep the smaller status, whichever came first.
1049    fn status(&mut self, status: &str) {
1050        let keep = match self.status.take() {
1051            Some(held) if held.as_str() <= status => held,
1052            _ => status.to_string(),
1053        };
1054        self.status = Some(keep);
1055    }
1056}

1058/// Derive the index. Pure: no I/O, and the result does not depend on the order of any
1059/// input vector.
1060pub(crate) fn derive(d: &Declarations) -> SubjectIndex {
1061    let mut drafts: BTreeMap<String, Draft> = BTreeMap::new();
1062
1063    // ---- leaves: claims, rules, commands
1064    for c in &d.claims {
1065        let inputs: Vec<String> = [
1066            c.source.clone(),
1067            c.implementation.clone(),
1068            c.test_path
1069                .as_deref()
1070                .and_then(TestId::of)
1071                .map(|t| t.source()),
1072        ]
1073        .into_iter()
1074        .flatten()
1075        .collect();
1076        let draft = draft_of(&mut drafts, SubjectKind::Claim, &c.id);
1077        draft.status(&c.status);
1078        draft.routes.insert(Route {
1079            via: Via::Claim,
1080            test: c.test.clone(),
1081            path: c.test_path.clone(),
1082            category: c.test.as_deref().and_then(Category::of),
1083            inputs: Some(inputs),
1084        });
1085    }
1086    for r in &d.rules {
1087        let draft = draft_of(&mut drafts, SubjectKind::Rule, &r.id);
1088        draft.status(&r.status);
1089        for p in &r.tests {
1090            match TestId::of(p) {
1091                Some(t) => {
1092                    let test = t.as_string();
1093                    draft.routes.insert(Route {
1094                        via: Via::Rule,
1095                        category: Category::of(&test),
1096                        test: Some(test),
1097                        path: Some(p.clone()),
1098                        inputs: Some(vec![t.source(), r.path.clone()]),
1099                    });
1100                }
1101                None => {
1102                    draft.mechanisms.insert(p.clone());
1103                }
1104            }
1105        }
1106    }
1107
1108    let shell: BTreeSet<&str> = d.commands.iter().map(String::as_str).collect();
1109    let examples: BTreeSet<&str> = d.examples.iter().map(String::as_str).collect();
1110    let native: BTreeSet<&str> = examples
1111        .iter()
1112        .copied()
1113        .chain(d.capabilities.iter().filter_map(|c| c.cli.as_deref()))
1114        .collect();
1115    // the example route exists only where the binary that runs the examples does: a route
1116    // to a test the repository does not carry would read `not_run` forever
1117    let example_route = TestId::of(CLI_EXAMPLES)
1118        .filter(|t| d.crate_tests.contains(&t.as_string()))
1119        .map(|t| {
1120            let test = t.as_string();
1121            Route {
1122                via: Via::Example,
1123                category: Category::of(&test),
1124                test: Some(test),
1125                path: Some(t.source()),
1126                inputs: None,
1127            }
1128        });
1129    for w in shell.union(&native) {
1130        let draft = draft_of(&mut drafts, SubjectKind::Command, w);
1131        if native.contains(w) {
1132            draft.programs.insert(Program::Native);
1133        }
1134        if shell.contains(w) {
1135            draft.programs.insert(Program::Shell);
1136        }
1137        if let Some(route) = example_route.as_ref().filter(|_| examples.contains(w)) {
1138            draft.routes.insert(route.clone());
1139        }
1140    }
1141
1142    // ---- capabilities and their aliases
1143    for c in &d.capabilities {
1144        let key = SubjectKind::Capability.key(&c.id);
1145        let draft = draft_of(&mut drafts, SubjectKind::Capability, &c.id);
1146        if let Some(cli) = &c.cli {
1147            draft.members.insert(SubjectKind::Command.key(cli));
1148        }
1149        // a claim implemented in the file the capability's module is composed in belongs to
1150        // every capability of that module, and a claim implemented anywhere else to none
1151        for claim in &d.claims {
1152            if claim.implementation.as_deref() == Some(c.source.as_str()) {
1153                draft.members.insert(SubjectKind::Claim.key(&claim.id));
1154            }
1155        }
1156        if let Some(tool) = &c.mcp_tool {
1157            let alias = draft_of(&mut drafts, SubjectKind::Mcp, tool);
1158            alias.members.insert(key.clone());
1159            alias.alias_of = match alias.alias_of.take() {
1160                Some(held) if held <= key => Some(held),
1161                _ => Some(key),
1162            };
1163        }
1164    }
1165
1166    // ---- the case headers: a command word routes to its command, a `capability:<id>` word
1167    // to that capability, and every other prefixed word to nothing
1168    for case in &d.cases {
1169        for (via, words) in [
1170            (Via::Behaviour, &case.covers),
1171            (Via::Negative, &case.negative),
1172        ] {
1173            for w in words {
1174                let key = if w.contains(':') {
1175                    match SubjectKind::parse(w) {
1176                        Some((SubjectKind::Capability, _)) => w.clone(),
1177                        _ => continue,
1178                    }
1179                } else {
1180                    SubjectKind::Command.key(w)
1181                };
1182                if let Some(draft) = drafts.get_mut(&key) {
1183                    draft.routes.insert(Route {
1184                        via,
1185                        test: Some(case.test.clone()),
1186                        path: Some(case.path.clone()),
1187                        category: Category::of(&case.test),
1188                        inputs: None,
1189                    });
1190                }
1191            }
1192        }
1193    }
1194
1195    // ---- composites: use cases, then features. A member is a key that exists; a
1196    // reference that resolves to nothing is the product validation's to report.
1197    let exists = |drafts: &BTreeMap<String, Draft>, kind: SubjectKind, id: &str| {
1198        let key = kind.key(id);
1199        drafts.contains_key(&key).then_some(key)
1200    };
1201    for u in &d.use_cases {
1202        let members: BTreeSet<String> = [
1203            (SubjectKind::Command, &u.commands),
1204            (SubjectKind::Rule, &u.doctrines),
1205            (SubjectKind::Claim, &u.claims),
1206            (SubjectKind::Mcp, &u.mcp_tools),
1207        ]
1208        .into_iter()
1209        .flat_map(|(kind, ids)| ids.iter().map(move |id| (kind, id)))
1210        .filter_map(|(kind, id)| exists(&drafts, kind, id))
1211        .collect();
1212        let draft = draft_of(&mut drafts, SubjectKind::UseCase, &u.id);
1213        if let Some(status) = &u.status {
1214            draft.status(status);
1215        }
1216        draft.members.extend(members);
1217        draft.routes.insert(Route {
1218            via: Via::Scenario,
1219            test: None,
1220            path: Some(u.path.clone()),
1221            category: Some(Category::Scenario),
1222            inputs: None,
1223        });
1224    }
1225    for f in &d.features {
1226        let members: BTreeSet<String> = [
1227            (SubjectKind::Claim, &f.claims),
1228            (SubjectKind::Rule, &f.rules),
1229            (SubjectKind::Command, &f.commands),
1230            (SubjectKind::UseCase, &f.use_cases),
1231        ]
1232        .into_iter()
1233        .flat_map(|(kind, ids)| ids.iter().map(move |id| (kind, id)))
1234        .filter_map(|(kind, id)| exists(&drafts, kind, id))
1235        .collect();
1236        let draft = draft_of(&mut drafts, SubjectKind::Feature, &f.id);
1237        draft.status(&f.status);
1238        draft.members.extend(members);
1239    }
1240
1241    // ---- the entries
1242    let mut subjects: BTreeMap<String, SubjectEntry> = BTreeMap::new();
1243    for (key, draft) in drafts {
1244        let mut findings = Vec::new();
1245        if draft.programs.len() > 1 {
1246            findings.push(SubjectFinding {
1247                code: COMMAND_IN_TWO_PROGRAMS.to_string(),
1248                severity: Severity::Warning,
1249                subject: key.clone(),
1250                message: format!(
1251                    "the word `{}` is a public command of the shell tool and a command path \
1252                     of the executable: its behaviour and negative routes come from the shell \
1253                     tool's case headers, its example route from the executable's documented \
1254                     examples, and one page shows both",
1255                    draft.id
1256                ),
1257            });
1258        }
1259        let page = page_of(draft.kind, &draft.id);
1260        let entry = SubjectEntry {
1261            kind: draft.kind,
1262            id: draft.id,
1263            status: draft.status,
1264            page,
1265            alias_of: draft.alias_of,
1266            programs: draft.programs.into_iter().collect(),
1267            members: draft.members.into_iter().collect(),
1268            routes: draft.routes.into_iter().collect(),
1269            mechanisms: draft.mechanisms.into_iter().collect(),
1270            findings,
1271        };
1272        subjects.insert(key, entry);
1273    }
1274
1275    // ---- every test a runner owns that no route reaches
1276    let reached: BTreeSet<&str> = subjects
1277        .values()
1278        .flat_map(|e| e.routes.iter().filter_map(|r| r.test.as_deref()))
1279        .collect();
1280    let excluded: Vec<String> = d
1281        .cases
1282        .iter()
1283        .map(|c| c.test.as_str())
1284        .chain(d.crate_tests.iter().map(String::as_str))
1285        .filter(|t| !reached.contains(t))
1286        .map(str::to_string)
1287        .collect::<BTreeSet<_>>()
1288        .into_iter()
1289        .collect();
1290
1291    let mut index = SubjectIndex { subjects, excluded };
1292
1293    // ---- the structural finding, read off the one walk
1294    let without_evidence: BTreeSet<String> = index
1295        .subjects
1296        .iter()
1297        .filter(|(_, e)| e.kind == SubjectKind::Feature)
1298        .filter(|(key, _)| {
1299            index
1300                .reach(key)
1301                .map(|routes| routes.iter().all(|r| r.test.is_none()))
1302                .unwrap_or(false)
1303        })
1304        .map(|(key, _)| key.clone())
1305        .collect();
1306    for (key, e) in index
1307        .subjects
1308        .iter_mut()
1309        .filter(|(key, _)| without_evidence.contains(key.as_str()))
1310    {
1311        let severity = if e.status.as_deref() == Some(crate::product::STABLE) {
1312            Severity::Error
1313        } else {
1314            Severity::Warning
1315        };
1316        e.findings.push(SubjectFinding {
1317            code: FEATURE_WITHOUT_EVIDENCE.to_string(),
1318            severity,
1319            subject: key.clone(),
1320            message: format!(
1321                "the feature `{}` reaches no test a runner drives through its claims, rules, \
1322                 commands or use cases",
1323                e.id
1324            ),
1325        });
1326    }
1327    index
1328}

1389/// A subject's verdict over its parts: the one aggregation
1390/// ([`crate::evidence::freshness::aggregate`]) over the declared order, and whether any part
1391/// can carry proof. A failing part makes the subject failing; otherwise the weakest part
1392/// that can carry proof decides; otherwise the weakest of all; and no part reads `no_test`.
1393///
1394/// ```
1395/// use majordomus_cli::evidence::subject::verdict;
1396/// use majordomus_cli::evidence::ProofState::{Failing, NoTest, NotRun, Proven};
1397///
1398/// assert_eq!(verdict(&[(Failing, true), (NotRun, true)]), (Failing, true));
1399/// assert_eq!(verdict(&[(Proven, true), (NotRun, false)]), (Proven, true));
1400/// assert_eq!(verdict(&[(NotRun, false)]), (NotRun, false));
1401/// assert_eq!(verdict(&[]), (NoTest, false));
1402/// ```
1403pub fn verdict(parts: &[(ProofState, bool)]) -> (ProofState, bool) {
1404    match aggregate(parts, ProofState::Failing) {
1405        None => (ProofState::NoTest, false),
1406        Some(s) => (s, parts.iter().any(|(_, carries)| *carries)),
1407    }
1408}

1547/// Judges many subjects at one presented revision with one set of comparisons.
1548///
1549/// **Precondition:** `claims` is [`crate::evidence::report_at`]`(index, ledger, presented,
1550/// uncommitted)` and `rules` is [`crate::rules::report`]`(index, ledger)`, for the same
1551/// index, the same ledger and the same `uncommitted`. At a presented commit the ledger is
1552/// [`crate::evidence::freshness::ledger_at`] of it and `uncommitted` is
1553/// [`crate::evidence::freshness::uncommitted`]; for the working tree it is empty.
1554///
1555/// The rules report judges at the working tree, so when the presented tree is not clean a
1556/// rule-derived `proven` is re-judged at the presented revision and reads `inputs
1557/// unchanged`, as every other route does. At a presented commit, every route the judge
1558/// computes passes through the monotone rule over the working ledger's uncommitted run of
1559/// its test, as the claim report applies it, so a subject never reads greener than
1560/// `evidence show --presented`.
1561///
1562/// ```
1563/// use majordomus_cli::evidence::subject::{self, Judge};
1564/// use majordomus_cli::evidence::{report_at, Ledger, Presented};
1565/// use majordomus_cli::synthetic::SyntheticRepository;
1566///
1567/// let repo = SyntheticRepository::small().unwrap();
1568/// let ctx = repo.context().unwrap();
1569/// let s = subject::index(&ctx.index, &ctx.product, &ctx.registry);
1570/// let ledger = Ledger::empty();
1571/// let claims = report_at(&ctx.index, &ledger, &Presented::WorkingTree, &[]);
1572/// let rules = majordomus_cli::rules::report(&ctx.index, &ledger);
1573/// let presented = Presented::WorkingTree;
1574/// let mut judge = Judge::new(&s, &claims, &rules, &ledger, &[], repo.root(), &presented);
1575/// let key = s.subjects.keys().find(|k| k.starts_with("capability:")).unwrap().clone();
1576/// let answer = judge.subject(&key).unwrap();
1577/// assert_eq!(answer.presented.revision, "working_tree");
1578/// assert!(judge.subject("gate:x").is_err());
1579/// ```

1582    claims: BTreeMap<&'a str, &'a ClaimProof>,
1583    rules: BTreeMap<&'a str, &'a RuleProof>,
1584    revision: &'a PresentedRevision,
1585    ledger: &'a Ledger,
1586    uncommitted: &'a [Execution],

1589    presented_tree: TreeState,
1590    adjust: Option<&'a Adjust<'a>>,

1690    /// The monotone rule over the working ledger's uncommitted run of `test`, at a presented
1691    /// commit, built exactly as the claim report builds it.
1692    fn uncommitted_weakens(
1693        &mut self,
1694        judgement: Judgement,
1695        test: &str,
1696        execution: Option<&Execution>,
1697    ) -> Judgement {
1698        let (Some(p), Some(e)) = (self.presented.commit(), execution) else {
1699            return judgement;
1700        };
1701        if judgement.state >= ProofState::Stale {
1702            return judgement;
1703        }
1704        let uncommitted = self.uncommitted;
1705        let Some(u) = uncommitted.iter().find(|u| u.test == test) else {
1706            return judgement;
1707        };
1708        let record = Supplementary {
1709            execution: u,
1710            named: UNCOMMITTED_RUN,
1711            after_evidence: self.contains(&u.commit, &e.commit),
1712            in_presented: self.contains(p, &u.commit),
1713        };
1714        weakened_by(judgement, &[record])
1715    }

1717    /// A recorded run judged at the presented revision, as a behaviour route is.
1718    fn at_presented(&mut self, route: &Route, execution: Option<&Execution>) -> Judgement {
1719        let source = route
1720            .path
1721            .as_deref()
1722            .and_then(TestId::of)
1723            .map(|t| t.source())
1724            .unwrap_or_default();
1725        let (recorded, cmp, moved) = match execution {
1726            None => (Recorded::NotRun, None, false),
1727            Some(e) => {
1728                let moved = e.outcome.proves() && e.digest_matches(self.root) == Some(false);
1729                (Recorded::Ran(e), Some(self.comparison(&e.commit)), moved)
1730            }
1731        };
1732        freshness(
1733            recorded,
1734            route.inputs.as_deref(),
1735            &source,
1736            moved,
1737            cmp.as_ref(),
1738            self.presented_tree,
1739        )
1740    }

1743    /// away from the state its own row gave.
1744    fn route(&mut self, entry: &SubjectEntry, route: &Route) -> (RouteProof, bool) {
1745        let test = route.test.as_deref();
1746        let mut carries = true;
1747        let mut reproduce = test.and(route.path.as_deref()).and_then(TestId::of);
1748        let (judgement, execution) = match route.via {
1749            Via::Claim => match self.claims.get(entry.id.as_str()) {
1750                Some(c) => {
1751                    carries = !matches!(c.status.as_str(), "planned" | "rejected");
1752                    let j = Judgement {
1753                        state: c.state,
1754                        changed: c.changed.clone(),
1755                        detail: c.detail.clone(),
1756                    };
1757                    (j, c.execution.clone())
1758                }
1759                None => {
1760                    carries = !matches!(entry.status.as_deref(), Some("planned" | "rejected"));
1761                    (missing("claim"), None)
1762                }
1763            },
1764            Via::Rule => {
1765                let proof = self.rules.get(entry.id.as_str()).and_then(|r| {
1766                    r.tests
1767                        .iter()
1768                        .find(|t| Some(t.path.as_str()) == route.path.as_deref())
1769                });
1770                match proof {
1771                    None => (missing("rule"), None),
1772                    Some(t) => {
1773                        let e = t.execution.clone();
1774                        let j = if self.presented_tree != TreeState::Clean
1775                            && t.state == ProofState::Proven
1776                        {
1777                            // the presented-tree cap: the rules report judged a clean
1778                            // working tree, and the presented tree is weaker
1779                            match &e {
1780                                Some(e) => self.at_presented(route, Some(e)),
1781                                None => Judgement {
1782                                    state: ProofState::Stale,
1783                                    changed: Vec::new(),
1784                                    detail: Some(NO_EXECUTION_DETAIL.to_string()),
1785                                },
1786                            }
1787                        } else {
1788                            Judgement {
1789                                state: t.state,
1790                                changed: Vec::new(),
1791                                detail: None,
1792                            }
1793                        };
1794                        (j, e)
1795                    }
1796                }
1797            }
1798            Via::Behaviour | Via::Negative | Via::Example => {
1799                let e = test.and_then(|t| self.ledger.latest(t)).cloned();
1800                (self.at_presented(route, e.as_ref()), e)
1801            }
1802            Via::Scenario => {
1803                carries = false;
1804                reproduce = None;
1805                let mut j = freshness(
1806                    Recorded::NotRun,
1807                    None,
1808                    route.path.as_deref().unwrap_or_default(),
1809                    false,
1810                    None,
1811                    self.presented_tree,
1812                );
1813                j.detail = Some(SCENARIO_DETAIL.to_string());
1814                (j, None)
1815            }
1816        };
1817        let own = judgement.state;
1818        let judgement = match (route.via, test) {
1819            (Via::Claim | Via::Scenario, _) | (_, None) => judgement,
1820            (_, Some(t)) => self.uncommitted_weakens(judgement, t, execution.as_ref()),
1821        };
1822        let judgement = self.adjusted(route, execution.as_ref(), judgement);
1823        let proof = RouteProof {
1824            route: route.clone(),
1825            state: judgement.state,
1826            carries_proof: carries,
1827            changed: judgement.changed,
1828            detail: judgement.detail,
1829            execution,
1830            reproduce: reproduce.map(|t| t.reproduce()),
1831        };
1832        (proof, own != judgement.state)
1833    }
1834
1835    /// The adjuster, when one is set and the route names a test: the route keeps the weaker
1836    /// of the two states, and the adjusted reasons only when they are what weakened it.
1837    fn adjusted(
1838        &self,
1839        route: &Route,
1840        execution: Option<&Execution>,
1841        judgement: Judgement,
1842    ) -> Judgement {
1843        let Some(adjust) = self.adjust else {
1844            return judgement;
1845        };
1846        let Some(id) = route
1847            .test
1848            .as_ref()
1849            .and(route.path.as_deref())
1850            .and_then(TestId::of)
1851        else {
1852            return judgement;
1853        };
1854        let state = judgement.state;
1855        let adjusted = adjust(&id, execution, judgement.clone());
1856        let weaker = aggregate(
1857            &[(state, true), (adjusted.state, true)],
1858            ProofState::Failing,
1859        )
1860        .unwrap_or(state);
1861        if weaker == state {
1862            judgement
1863        } else {
1864            Judgement {
1865                state: weaker,
1866                ..adjusted
1867            }
1868        }
1869    }

1873    /// Everything the repository can say about the subject `key` at the presented revision.
1874    ///
1875    /// ```
1876    /// use majordomus_cli::evidence::subject::{self, Judge};
1877    /// use majordomus_cli::evidence::{report_at, Ledger, Presented};
1878    /// use majordomus_cli::synthetic::SyntheticRepository;
1879    ///
1880    /// let repo = SyntheticRepository::small().unwrap();
1881    /// let ctx = repo.context().unwrap();
1882    /// let s = subject::index(&ctx.index, &ctx.product, &ctx.registry);
1883    /// let ledger = Ledger::empty();
1884    /// let claims = report_at(&ctx.index, &ledger, &Presented::WorkingTree, &[]);
1885    /// let rules = majordomus_cli::rules::report(&ctx.index, &ledger);
1886    /// let p = Presented::WorkingTree;
1887    /// let mut judge = Judge::new(&s, &claims, &rules, &ledger, &[], repo.root(), &p);
1888    /// let err = judge.subject("feature:no-such").unwrap_err();
1889    /// assert!(err.to_string().contains("feature:no-such"));
1890    /// ```
1891    pub fn subject(&mut self, key: &str) -> Result<SubjectEvidence, UnknownSubject> {
1892        let subjects = self.subjects;
1893        let entry = subjects.get(key)?;
1894        let (routes, v) = self.evaluate(key, entry);
1895        let members: Vec<MemberProof> = entry
1896            .members
1897            .iter()
1898            .filter_map(|m| {
1899                let kind = SubjectKind::parse(m)?.0;
1900                let (state, carries_proof, rule_state) = self.verdict_of(m)?;
1901                Some(MemberProof {
1902                    subject: m.clone(),
1903                    kind,
1904                    state,
1905                    carries_proof,
1906                    rule_state,
1907                })
1908            })
1909            .collect();
1910        let mut totals: BTreeMap<String, usize> = BTreeMap::new();
1911        let counted = routes.iter().map(|r| r.state).chain(
1912            members
1913                .iter()
1914                .filter(|_| entry.kind != SubjectKind::Rule)
1915                .map(|m| m.state),
1916        );
1917        for state in counted {
1918            *totals.entry(state.label().to_string()).or_insert(0) += 1;
1919        }
1920        let subject_findings = self.findings(key, entry, &routes);
1921        Ok(SubjectEvidence {
1922            subject: key.to_string(),
1923            kind: entry.kind,
1924            id: entry.id.clone(),
1925            page: entry.page.clone(),
1926            presented: self.revision.clone(),
1927            verdict: v.0,
1928            meaning: v.0.meaning().to_string(),
1929            carries_proof: v.1,
1930            rule_state: v.2,
1931            totals,
1932            routes,
1933            members,
1934            mechanisms: entry.mechanisms.clone(),
1935            subject_findings,
1936        })
1937    }

1939    /// A member's verdict, memoised: members form a DAG, and each is judged once.
1940    fn verdict_of(&mut self, key: &str) -> Option<Verdict> {
1941        if let Some(v) = self.memo.get(key) {
1942            return Some(*v);
1943        }
1944        let subjects = self.subjects;
1945        let entry = subjects.subjects.get(key)?;
1946        Some(self.evaluate(key, entry).1)
1947    }
1948
1949    /// A subject's own judged routes and its verdict, memoised by key.
1950    fn evaluate(&mut self, key: &str, entry: &'a SubjectEntry) -> (Vec<RouteProof>, Verdict) {
1951        let mut routes = Vec::new();
1952        let mut moved = Vec::new();
1953        for r in &entry.routes {
1954            let (proof, was_moved) = self.route(entry, r);
1955            if was_moved {
1956                moved.push(proof.state);
1957            }
1958            routes.push(proof);
1959        }
1960        let v = if entry.kind == SubjectKind::Rule {
1961            self.rule_verdict(entry, &moved)
1962        } else {
1963            let mut parts: Vec<(ProofState, bool)> =
1964                routes.iter().map(|r| (r.state, r.carries_proof)).collect();
1965            for m in &entry.members {
1966                if let Some((state, carries, _)) = self.verdict_of(m) {
1967                    parts.push((state, carries));
1968                }
1969            }
1970            let (state, carries) = verdict(&parts);
1971            (state, carries, None)
1972        };
1973        self.memo.insert(key.to_string(), v);
1974        (routes, v)
1975    }

1977    /// A rule subject: its rule proof through [`RuleState::as_proof`], capped by the
1978    /// presented tree, and re-aggregated only with the routes something moved.
1979    fn rule_verdict(&self, entry: &SubjectEntry, moved: &[ProofState]) -> Verdict {
1980        let Some(proof) = self.rules.get(entry.id.as_str()) else {
1981            return (ProofState::NoTest, false, None);
1982        };
1983        let mut s = proof.state;
1984        if self.presented_tree != TreeState::Clean && s == RuleState::Proven {
1985            s = RuleState::InputsUnchanged;
1986        }
1987        let (mut p, carries) = s.as_proof(proof.rule.class);
1988        if !moved.is_empty() {
1989            let parts: Vec<(ProofState, bool)> = std::iter::once((p, true))
1990                .chain(moved.iter().map(|m| (*m, true)))
1991                .collect();
1992            p = aggregate(&parts, ProofState::Failing).unwrap_or(p);
1993        }
1994        (p, carries, Some(s))
1995    }

1997    /// The entry's structural findings, then one `dangling_member` per direct rule member
1998    /// that names a missing path, then one `freshness_unknown` per evidence commit of its own
1999    /// passing routes that git could not compare.
2000    fn findings(
2001        &mut self,
2002        key: &str,
2003        entry: &SubjectEntry,
2004        routes: &[RouteProof],
2005    ) -> Vec<SubjectFinding> {
2006        let mut findings = entry.findings.clone();
2007        for m in &entry.members {
2008            let Some((SubjectKind::Rule, id)) = SubjectKind::parse(m) else {
2009                continue;
2010            };
2011            if self
2012                .rules
2013                .get(id)
2014                .is_some_and(|r| r.state == RuleState::Dangling)
2015            {
2016                findings.push(SubjectFinding {
2017                    code: DANGLING_MEMBER.to_string(),
2018                    severity: Severity::Warning,
2019                    subject: key.to_string(),
2020                    message: format!("the rule `{id}` names a path that is not in the tree"),
2021                });
2022            }
2023        }
2024        let commits: BTreeSet<&str> = routes
2025            .iter()
2026            .filter_map(|r| r.execution.as_ref())
2027            .filter(|e| e.outcome.proves())
2028            .map(|e| e.commit.as_str())
2029            .collect();
2030        for commit in commits {
2031            let cmp = self.comparison(commit);
2032            if cmp.containment == Containment::CommitUnknown || cmp.changed.is_none() {
2033                let short: String = commit.chars().take(12).collect();
2034                findings.push(SubjectFinding {
2035                    code: FRESHNESS_UNKNOWN.to_string(),
2036                    severity: Severity::Warning,
2037                    subject: key.to_string(),
2038                    message: format!(
2039                        "git could not compare the evidence commit {short} with the presented \
2040                         revision, so the routes recorded on it read stale; not knowing is \
2041                         not proof"
2042                    ),
2043                });
2044            }
2045        }
2046        findings
2047    }

2050/// One subject, judged at the presented revision from the two reports of the given ledger.
2051///
2052/// The caller passes the ledger and its uncommitted executions as `evidence show
2053/// --presented` reads them: the checkout's ledger and nothing for the working tree, the
2054/// commit's ledger and [`crate::evidence::freshness::uncommitted`] for a presented commit.
2055/// An unknown key costs no git work. No adjuster is set; a caller that needs one builds the
2056/// [`Judge`] itself.
2057///
2058/// ```
2059/// use majordomus_cli::evidence::subject;
2060/// use majordomus_cli::evidence::{Ledger, Presented};
2061/// use majordomus_cli::synthetic::SyntheticRepository;
2062///
2063/// let repo = SyntheticRepository::small().unwrap();
2064/// let ctx = repo.context().unwrap();
2065/// let (ledger, p) = (Ledger::empty(), Presented::WorkingTree);
2066/// let ask = |key: &str| {
2067///     subject::answer(&ctx.index, &ctx.product, &ctx.registry, &ledger, &[], &p, key)
2068/// };
2069/// assert!(ask("nonsense").is_err());
2070/// let tool = ask("mcp:majordomus_capabilities").unwrap();
2071/// assert_eq!(tool.members.len(), 1, "an alias is made of its capability");
2072/// ```
2073pub fn answer(
2074    index: &Index,
2075    product: &ProductModel,
2076    registry: &CapabilityRegistry,
2077    ledger: &Ledger,
2078    uncommitted: &[Execution],
2079    presented: &Presented,
2080    key: &str,
2081) -> Result<SubjectEvidence, UnknownSubject> {
2082    let s = self::index(index, product, registry);
2083    s.get(key)?;
2084    let claims = crate::evidence::report_at(index, ledger, presented, uncommitted);
2085    let rules = crate::rules::report(index, ledger);
2086    let root = PathBuf::from(&index.repository.root);
2087    Judge::new(&s, &claims, &rules, ledger, uncommitted, &root, presented).subject(key)
2088}
