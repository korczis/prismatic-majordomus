1//! The plan: milestones, issues, the two graphs between them, and everything derived from
2//! them — execution waves, issue status, milestone status, the roadmap order, the active
3//! milestone, the next executable issue and every validation finding.
4//!
5//! **This is a second reader of one set of semantics, not a second set.** `lib/project.awk`
6//! is the incumbent engine: the shell tool, the site generator and the GitHub adapter all
7//! call it, and it has been the only implementation since the model existed. This module
8//! derives the same values in Rust so that the capability registry can answer them — an
9//! agent asking the shared MCP server what to work on next could not be told otherwise —
10//! and the two are held to byte equality by `test/cases/99_plan_capabilities.sh`, which
11//! runs both over every record in this repository and over a fixture built to hit the
12//! edges. Where the two disagree the awk is right; this file is the one that changes.
13//!
14//! The order of derivation below follows `lib/project.awk` step for step, including the
15//! order findings are produced in, because that order is observable: a reader diffing the
16//! two engines diffs a sequence, not a set. Comments name the awk block each step mirrors.
17//!
18//! Nothing here reads a file. Milestones and issues are ordinary declarative objects of the
19//! layer — kinds `milestone` and `issue`, whole-YAML metadata — already discovered, parsed
20//! and schema-validated by the time the index holds them, so building the plan is a walk of
21//! values already in memory. That is what lets it be built on demand, inside the handler of
22//! a plan capability, rather than eagerly when a [`crate::capability::Context`] is composed:
23//! `App::load` has a stated budget and 200 project records have no business inside it.
24//!
25//! One question is not the records' to answer: which release carried a milestone. That is
26//! [`released_in`], the only function here that asks git anything, and only a reader that
27//! wants the answer (the roadmap) pays for it.

66/// One YAML scalar as the flattener would have written it: the text an awk comparison sees.
67///
68/// `lib/yaml.sh` flattens every value to a string, so `order: 3` is `"3"`, `cancelled: true`
69/// is `"true"` and a missing key is `""`. The index holds the same records as typed JSON,
70/// so the comparisons this module makes have to be made against the same text the awk
71/// compares — otherwise `parallel_safe: false` would be a bool here and the string `"false"`
72/// there, and only one of them would be right.
73///
74/// ```
75/// use majordomus_cli::plan::scalar;
76/// use serde_json::json;
77/// assert_eq!(scalar(Some(&json!(3))), "3");
78/// assert_eq!(scalar(Some(&json!(true))), "true");
79/// assert_eq!(scalar(Some(&json!("x"))), "x");
80/// assert_eq!(scalar(Some(&json!(null))), "");
81/// assert_eq!(scalar(None), "");
82/// ```
83pub fn scalar(v: Option<&Value>) -> String {
84    match v {
85        None | Some(Value::Null) => String::new(),
86        Some(Value::String(s)) => s.clone(),
87        Some(Value::Bool(b)) => b.to_string(),
88        Some(Value::Number(n)) => n.to_string(),
89        Some(other) => other.to_string(),
90    }
91}

93fn field(meta: &Value, key: &str) -> String {
94    scalar(meta.get(key))
95}

97/// A list field as the flattener's `key.0`, `key.1` … would have carried it: every item as a
98/// scalar, in the file's order. A scalar where a list was expected is one item, which is
99/// what the flattener produces for `scope: apps/` too.
100fn list(meta: &Value, key: &str) -> Vec<String> {
101    match meta.get(key) {
102        None | Some(Value::Null) => Vec::new(),
103        Some(Value::Array(items)) => items.iter().map(|v| scalar(Some(v))).collect(),
104        Some(other) => vec![scalar(Some(other))],
105    }
106}

108/// The `covers` token of every evidence entry that carries one, in file order. The awk
109/// counts the `evidence.N.covers` keys, so an entry without one is not evidence.
110fn evidence_covers(meta: &Value) -> Vec<String> {
111    match meta.get("evidence") {
112        Some(Value::Array(items)) => items
113            .iter()
114            .filter_map(|e| e.get("covers"))
115            .map(|v| scalar(Some(v)))
116            .collect(),
117        _ => Vec::new(),
118    }
119}

121/// The latest non-empty `recorded_at` among a record's `evidence` entries, compared as text
122/// (ISO 8601 stamps sort by time), or `None` when no entry carries one.
123fn last_evidence_stamp(meta: &Value) -> Option<String> {
124    match meta.get("evidence") {
125        Some(Value::Array(items)) => items
126            .iter()
127            .filter_map(|e| e.get("recorded_at"))
128            .map(|v| scalar(Some(v)))
129            .filter(|s| !s.is_empty())
130            .max(),
131        _ => None,
132    }
133}

142/// The first instant a stamp could only have been written with its seal (ADR 0097).
143///
144/// A stamp older than this may be excused by its record's `unsealed_stamps` list, which
145/// names the stamps recorded before the transition wrote a seal. A stamp at or after
146/// it never is: the transition that wrote it wrote the seal in the same write, so a stamp
147/// without one was written by something else. `SEALED_FROM` in `lib/project.awk`.
148pub const SEALED_FROM: &str = "2026-10-01T00:00:00Z";
149
150/// The seal of one transition: the proof, tracked in the record beside the stamp, that the
151/// stamp was written by the transition that appends its event.
152///
153/// The ledger that holds the event is checkout-local and never tracked, so no gate reading
154/// a clone can compare a stamp with it. The seal is the tracked half: the SHA-256 of the
155/// event, the issue and the stamp, written by [`transition`] in the same write as the stamp
156/// and read by [`Plan::build`], which gives a stamp without its seal no effect on status and
157/// reports it as `stamp_without_event`. `mj_pj_seal_rows` in `lib/project.sh` computes the
158/// same digest for the awk engine.
159///
160/// ```
161/// use majordomus_cli::plan::{seal, Transition};
162/// let s = seal(Transition::Done, "I0001", "2026-09-16T12:00:00Z");
163/// assert!(s.starts_with("sha256:") && s.len() == 7 + 64);
164/// // A seal proves one stamp of one move of one issue, and nothing else.
165/// assert_ne!(s, seal(Transition::Start, "I0001", "2026-09-16T12:00:00Z"));
166/// assert_ne!(s, seal(Transition::Done, "I0002", "2026-09-16T12:00:00Z"));
167/// assert_ne!(s, seal(Transition::Done, "I0001", "2026-09-16T12:00:01Z"));
168/// ```
169pub fn seal(transition: Transition, issue: &str, stamp: &str) -> String {
170    format!(
171        "sha256:{}",
172        crate::policy::sha256_hex(&format!(
173            "majordomus.plan-event/v1\n{}\n{issue}\n{stamp}\n",
174            transition.event()
175        ))
176    )
177}

257
258    /// The stamp a move writes, as the record declares it.
259    fn stamp(&self, t: Transition) -> &str {
260        match t {
261            Transition::Start => &self.started_at,
262            Transition::Verify => &self.verified_at,
263            Transition::Done => &self.completed_at,
264        }
265    }
266
267    /// The seal the record carries beside that stamp.
268    fn sealed_with(&self, t: Transition) -> &str {
269        match t {
270            Transition::Start => &self.started_event,
271            Transition::Verify => &self.verified_event,
272            Transition::Done => &self.completed_event,
273        }
274    }
275
276    /// What a transition writes: the stamp and its seal, together.
277    fn stamp_with_seal(&mut self, t: Transition, now: &str) {
278        let s = seal(t, &self.id, now);
279        let (stamp, sealed) = match t {
280            Transition::Start => (&mut self.started_at, &mut self.started_event),
281            Transition::Verify => (&mut self.verified_at, &mut self.verified_event),
282            Transition::Done => (&mut self.completed_at, &mut self.completed_event),
283        };
284        *stamp = now.to_string();
285        *sealed = s;
286    }
287
288    /// Is the stamp of this move a fact: absent, sealed by its event, or recorded before
289    /// seals existed and named by the record's `unsealed_stamps`? Anything else was written
290    /// by something other than the transition.
291    fn proven(&self, t: Transition) -> bool {
292        let stamp = self.stamp(t);
293        stamp.is_empty()
294            || self.sealed_with(t) == seal(t, &self.id, stamp)
295            || (stamp < SEALED_FROM
296                && self
297                    .unsealed_stamps
298                    .contains(&format!("{} {stamp}", t.field())))
299    }
300}

531    }

534    pub fn warnings(&self) -> usize {
535        self.findings.iter().filter(|f| f.level == "WARN").count()
536    }

538    /// Whether this repository holds a canonical project model at all.
539    pub fn is_present(&self) -> bool {
540        !self.milestones.is_empty() || !self.issues.is_empty()
541    }
542
543    /// The milestone ids in roadmap order: rank, then order, then id. Derived from the
544    /// graph; no list of versions is maintained anywhere.
545    pub fn roadmap(&self) -> Vec<&PlanMilestone> {
546        let mut out: Vec<&PlanMilestone> = self.milestones.iter().collect();
547        out.sort_by(|a, b| {
548            (a.rank, a.order, &a.id)
549                .partial_cmp(&(b.rank, b.order, &b.id))
550                .unwrap_or(std::cmp::Ordering::Equal)
551        });
552        out
553    }
554
555    /// The one issue a worker should take now: the lowest-wave READY issue of the active
556    /// milestone, highest priority first, then id.
557    ///
558    /// The active milestone can have nothing ready while another one does — a milestone
559    /// waiting on its own acceptance evidence, for instance — so the search widens to the
560    /// whole plan rather than answering "none" and sending a worker away from executable
561    /// work. `mj_pj_next_ready` in `lib/project.sh` does exactly this.
562    pub fn next_ready(&self, milestone: Option<&str>) -> Option<&PlanIssue> {
563        let scope = milestone.unwrap_or(self.project.active_milestone.as_str());
564        self.ready_ranked(Some(scope))
565            .or_else(|| self.ready_ranked(None))
566    }
567
568    fn ready_ranked(&self, milestone: Option<&str>) -> Option<&PlanIssue> {
569        self.issues
570            .iter()
571            .filter(|i| i.status == "READY")
572            .filter(|i| milestone.is_none_or(|m| m.is_empty() || i.milestone == m))
573            .min_by_key(|i| (i.wave, priority_rank(&i.priority), i.id.clone()))
574    }

583    /// The plan as it will read once `transition` has been written to `id`.
584    ///
585    /// The same records, the same [`derive`], one timestamp changed: the status a caller is
586    /// told after a write is computed by the one derivation every reader uses, never
587    /// announced by the writer. This matters because the resulting status is not a function
588    /// of the transition alone — a `done` whose required evidence is absent stays VERIFY,
589    /// and a record marked `cancelled` stays CANCELLED whatever is written to it.
590    ///
591    /// Nothing is read from disk and nothing is written: this is the projection of a write,
592    /// and [`transition`] is what performs one.
593    ///
594    /// ```no_run
595    /// # use majordomus_cli::index::Index;
596    /// use majordomus_cli::plan::{Plan, Transition};
597    /// # fn demo(index: &Index) {
598    /// let before = Plan::build(index);
599    /// let after = Plan::after(index, "I0001", Transition::Done, "2026-09-11T12:00:00Z");
600    /// // The status is derived both times, so an issue whose evidence is incomplete is
601    /// // still VERIFY afterwards and the caller can see that before writing anything.
602    /// assert_eq!(before.issue("I0001").is_some(), after.issue("I0001").is_some());
603    /// # }
604    /// ```
605    pub fn after(index: &Index, id: &str, transition: Transition, now: &str) -> Plan {
606        let (header, milestones, mut issues) = raws(index);
607        if let Some(r) = issues.iter_mut().find(|r| r.id == id) {
608            r.stamp_with_seal(transition, now);
609        }
610        derive(header, milestones, issues)
611    }
612}

614/// Every record the plan is derived from, as its file declares it.
615///
616/// Split out of [`Plan::build`] so that [`Plan::after`] derives from the same collection
617/// rather than a second reading of the index.
618fn raws(index: &Index) -> (PlanProject, Vec<PlanMilestoneRaw>, Vec<PlanIssueRaw>) {
619    let mut milestones: Vec<PlanMilestoneRaw> = Vec::new();
620    let mut issues: Vec<PlanIssueRaw> = Vec::new();
621    let mut header = PlanProject {
622        name: String::new(),
623        repository: String::new(),
624        default_branch: String::new(),
625        active_milestone: String::new(),
626    };
627    for o in &index.objects {
628        match o.kind.as_str() {
629            MILESTONE => milestones.push(PlanMilestoneRaw::of(&o.identity, &o.metadata)),
630            ISSUE => issues.push(PlanIssueRaw::of(&o.identity, &o.metadata)),
631            PROJECT if o.provenance.path.ends_with("project/project.yaml") => {
632                header.name = field(&o.metadata, "name");
633                header.repository = field(&o.metadata, "repository");
634                header.default_branch = field(&o.metadata, "default_branch");
635            }
636            _ => {}
637        }
638    }
639    // The awk reads the files in `LC_ALL=C sort` order of their names, and the loader
640    // refuses a record whose id disagrees with its file name, so id order is file order.
641    milestones.sort_by(|a, b| a.id.cmp(&b.id));
642    issues.sort_by(|a, b| a.id.cmp(&b.id));
643    (header, milestones, issues)
644}

646/// A priority's rank, lowest first: the order [`Plan::next_ready`] takes issues in, and the
647/// one every reader that ranks issues the way the plan does uses (I2917).
648pub(crate) fn priority_rank(p: &str) -> u8 {
649    match p {
650        "p0" => 0,
651        "p1" => 1,
652        "p2" => 2,
653        _ => 3,
654    }

680/// Does path `a` contain path `b`, or the reverse? Trailing slashes and `./` are normalised
681/// so that a hand-written scope entry cannot silently miss an overlap. `overlap()` in
682/// `lib/project.awk`.
683///
684/// ```
685/// use majordomus_cli::plan::overlap;
686/// assert!(overlap("apps/cli", "apps/cli/src/main.rs"));
687/// assert!(overlap("./lib/", "lib"));
688/// assert!(!overlap("lib", "libexec"));
689/// ```
690pub fn overlap(a: &str, b: &str) -> bool {
691    let norm = |s: &str| {
692        let s = s.strip_prefix("./").unwrap_or(s);
693        s.trim_end_matches('/').to_string()
694    };
695    let (a, b) = (norm(a), norm(b));
696    a == b || b.starts_with(&format!("{a}/")) || a.starts_with(&format!("{b}/"))
697}

699#[allow(clippy::too_many_lines)] // one derivation, in the order lib/project.awk derives it
700fn derive(mut header: PlanProject, mraw: Vec<PlanMilestoneRaw>, iraw: Vec<PlanIssueRaw>) -> Plan {
701    let mut v = PlanFindings(Vec::new());
702    let iids: Vec<String> = iraw.iter().map(|i| i.id.clone()).collect();
703    let mids: Vec<String> = mraw.iter().map(|m| m.id.clone()).collect();
704    let iseen: BTreeSet<&str> = iids.iter().map(String::as_str).collect();
705    let mseen: BTreeSet<&str> = mids.iter().map(String::as_str).collect();
706    let ix = |id: &str| -> Option<usize> { iraw.iter().position(|r| r.id == id) };
707
708    // --- the event behind every stamp. A stamp its transition did not write is not a fact
709    //     about the issue: it moves no status, and it is a failure by name (ADR 0097).
710    let proven: Vec<[bool; 3]> = iraw.iter().map(|r| MOVES.map(|t| r.proven(t))).collect();
711    // `MOVES` lists the moves in declaration order, so a move's discriminant is its column.
712    let stamped =
713        |n: usize, t: Transition| -> bool { !iraw[n].stamp(t).is_empty() && proven[n][t as usize] };
714
715    // --- local facts: is this issue DONE on its own terms?
716    let mut covered = vec![true; iraw.len()];
717    let mut done = vec![false; iraw.len()];
718    for (n, r) in iraw.iter().enumerate() {
719        covered[n] = r
720            .evidence_required
721            .iter()
722            .all(|need| r.evidence_have.contains(need));
723        done[n] = !r.cancelled && stamped(n, Transition::Done) && covered[n];
724    }
725
726    // --- edges, and the errors visible in them
727    let mut edge: BTreeSet<(String, String)> = BTreeSet::new();
728    let mut dependents: Vec<Vec<String>> = vec![Vec::new(); iraw.len()];
729    let mut edges: Vec<PlanEdge> = Vec::new();
730    for (n, r) in iraw.iter().enumerate() {
731        let mut seen_here: BTreeSet<&str> = BTreeSet::new();
732        for d in &r.depends_on {
733            if d == &r.id {
734                v.fail("self_dependency", &r.id, "depends on itself".into());
735                continue;
736            }
737            if !iseen.contains(d.as_str()) {
738                v.fail(
739                    "unknown_dependency",
740                    &r.id,
741                    format!("depends on {d}, which is not an issue"),
742                );
743                continue;
744            }
745            if !seen_here.insert(d.as_str()) {
746                v.warn(
747                    "duplicate_dependency",
748                    &r.id,
749                    format!("names {d} more than once"),
750                );
751                continue;
752            }
753            edge.insert((r.id.clone(), d.clone()));
754            if let Some(dn) = ix(d) {
755                dependents[dn].push(r.id.clone());
756            }
757            edges.push(PlanEdge {
758                from: d.clone(),
759                to: r.id.clone(),
760            });
761        }
762        if r.milestone.is_empty() {
763            v.fail("no_milestone", &r.id, "names no milestone".into());
764        } else if !mseen.contains(r.milestone.as_str()) {
765            v.fail(
766                "unknown_milestone",
767                &r.id,
768                format!("names milestone {}, which does not exist", r.milestone),
769            );
770        }
771        if r.acceptance_criteria == 0 && !r.cancelled {
772            v.fail(
773                "no_acceptance",
774                &r.id,
775                "has no acceptance criteria; an issue without one is a placeholder".into(),
776            );
777        }
778        if r.validation == 0 && !r.cancelled {
779            v.fail("no_validation", &r.id, "names no validation command".into());
780        }
781        if r.evidence_required.is_empty() && !r.cancelled {
782            v.warn(
783                "no_evidence_required",
784                &r.id,
785                "requires no evidence; nothing gates its completion".into(),
786            );
787        }
788        for (k, t) in MOVES.iter().enumerate() {
789            if !proven[n][k] {
790                v.fail(
791                    "stamp_without_event",
792                    &r.id,
793                    format!(
794                        "{} {} has no {} event: {} is absent or does not seal it, and only `majordomus plan {}` writes the stamp",
795                        t.field(),
796                        r.stamp(*t),
797                        t.event(),
798                        t.seal_field(),
799                        t.verb()
800                    ),
801                );
802            }
803        }
804        if stamped(n, Transition::Done) && !covered[n] && !r.cancelled {
805            let miss: Vec<&str> = r
806                .evidence_required
807                .iter()
808                .filter(|need| !r.evidence_have.contains(need))
809                .map(String::as_str)
810                .collect();
811            v.warn(
812                "evidence_missing",
813                &r.id,
814                format!(
815                    "completed_at is set but evidence is missing for {}; status stays VERIFY",
816                    miss.join(",")
817                ),
818            );
819        }
820    }
821
822    // --- waves: Kahn layering. A node enters a wave only once every dependency has left,
823    //     so its wave is one past the longest path to it. Nodes that never leave are in or
824    //     downstream of a cycle, and are reported rather than silently given a wave.
825    let mut left = vec![true; iraw.len()];
826    let mut nleft = iraw.len();
827    let mut wave: Vec<Option<u32>> = vec![None; iraw.len()];
828    let mut wavelines: Vec<PlanWave> = Vec::new();
829    let mut layer: u32 = 0;
830    while nleft > 0 {
831        let mut front: Vec<usize> = Vec::new();
832        for (n, r) in iraw.iter().enumerate() {
833            if !left[n] {
834                continue;
835            }
836            let blocked = r.depends_on.iter().any(|d| {
837                d != &r.id && iseen.contains(d.as_str()) && ix(d).is_some_and(|dn| left[dn])
838            });
839            if !blocked {
840                front.push(n);
841            }
842        }
843        if front.is_empty() {
844            let stuck: Vec<&str> = (0..iraw.len())
845                .filter(|n| left[*n])
846                .map(|n| iids[n].as_str())
847                .collect();
848            v.fail(
849                "cycle",
850                "graph",
851                format!(
852                    "a dependency cycle prevents these issues from ever becoming ready: {}",
853                    stuck.join(" ")
854                ),
855            );
856            break;
857        }
858        let mut line: Vec<String> = Vec::new();
859        for n in front {
860            wave[n] = Some(layer);
861            left[n] = false;
862            nleft -= 1;
863            line.push(iids[n].clone());
864        }
865        wavelines.push(PlanWave {
866            wave: layer,
867            issues: line,
868        });
869        layer += 1;
870    }
871
872    // --- issue status
873    let mut status: Vec<String> = vec![String::new(); iraw.len()];
874    let mut blocked_by: Vec<Vec<String>> = vec![Vec::new(); iraw.len()];
875    for (n, r) in iraw.iter().enumerate() {
876        // Every declared dependency, not every edge: a duplicate is a warning, and the awk
877        // walks the declaration here, so a record naming one twice blocks on it twice.
878        let bb: Vec<String> = r
879            .depends_on
880            .iter()
881            .filter(|d| {
882                *d != &r.id && iseen.contains(d.as_str()) && ix(d).is_some_and(|dn| !done[dn])
883            })
884            .cloned()
885            .collect();
886        let st = if r.cancelled {
887            "CANCELLED"
888        } else if done[n] {
889            "DONE"
890        } else if stamped(n, Transition::Done) || stamped(n, Transition::Verify) {
891            "VERIFY"
892        } else if stamped(n, Transition::Start) {
893            "ACTIVE"
894        } else if !bb.is_empty() {
895            "BLOCKED"
896        } else {
897            "READY"
898        };
899        if !bb.is_empty() && matches!(st, "DONE" | "ACTIVE" | "VERIFY") {
900            v.fail(
901                "premature_execution",
902                &r.id,
903                format!("is {st} while {} is not DONE", bb.join(",")),
904            );
905        }
906        status[n] = st.to_string();
907        blocked_by[n] = bb;
908        if !left[n] && wave[n].is_none() && st != "CANCELLED" {
909            wave[n] = Some(0);
910        }
911    }
912
913    // --- the milestone graph: validated before anything derives from it, under the same
914    //     rules the issue graph lives under, one level up.
915    let mut medge: BTreeSet<(String, String)> = BTreeSet::new();
916    let mut mdependents: Vec<Vec<String>> = vec![Vec::new(); mraw.len()];
917    let mut medges: Vec<PlanEdge> = Vec::new();
918    let mx = |id: &str| -> Option<usize> { mraw.iter().position(|r| r.id == id) };
919    for r in &mraw {
920        for d in &r.depends_on {
921            if d == &r.id {
922                v.fail("milestone_self_dependency", &r.id, "requires itself".into());
923                continue;
924            }
925            if !mseen.contains(d.as_str()) {
926                v.fail(
927                    "milestone_unknown_dependency",
928                    &r.id,
929                    format!("requires {d}, which is not a milestone here"),
930                );
931                continue;
932            }
933            if medge.contains(&(r.id.clone(), d.clone())) {
934                v.warn(
935                    "milestone_duplicate_dependency",
936                    &r.id,
937                    format!("requires {d} more than once"),
938                );
939                continue;
940            }
941            medge.insert((r.id.clone(), d.clone()));
942            if let Some(dn) = mx(d) {
943                mdependents[dn].push(r.id.clone());
944            }
945            medges.push(PlanEdge {
946                from: d.clone(),
947                to: r.id.clone(),
948            });
949        }
950    }
951
952    // --- roadmap rank: Kahn layering over the milestone graph. Rank is what orders the
953    //     roadmap; `order` only breaks ties inside one rank.
954    let mut mleft = vec![true; mraw.len()];
955    let mut mnleft = mraw.len();
956    let mut rank: Vec<u32> = vec![0; mraw.len()];
957    let mut mlayer: u32 = 0;
958    while mnleft > 0 {
959        let mut front: Vec<usize> = Vec::new();
960        for (n, r) in mraw.iter().enumerate() {
961            if !mleft[n] {
962                continue;
963            }
964            let blocked = r.depends_on.iter().any(|d| {
965                medge.contains(&(r.id.clone(), d.clone())) && mx(d).is_some_and(|dn| mleft[dn])
966            });
967            if !blocked {
968                front.push(n);
969            }
970        }
971        if front.is_empty() {
972            break;
973        }
974        for n in front {
975            rank[n] = mlayer;
976            mleft[n] = false;
977            mnleft -= 1;
978        }
979        mlayer += 1;
980    }
981    if mnleft > 0 {
982        let cyc: Vec<&str> = (0..mraw.len())
983            .filter(|n| mleft[*n])
984            .map(|n| mids[n].as_str())
985            .collect();
986        v.fail(
987            "milestone_cycle",
988            &cyc.join(" "),
989            "milestone dependencies form a cycle; no roadmap order exists".into(),
990        );
991        for n in 0..mraw.len() {
992            if mleft[n] {
993                rank[n] = 0;
994            }
995        }
996    }
997
998    // --- milestone status
999    let mut mstatus: Vec<String> = vec![String::new(); mraw.len()];
1000    let mut counts: Vec<PlanCounts> = Vec::with_capacity(mraw.len());
1001    let mut mempty = vec![false; mraw.len()];
1002    for (n, r) in mraw.iter().enumerate() {
1003        let c = tally(&iraw, &status, &r.id);
1004        let (nd, nr, na, nv) = (
1005            *c.by_status.get("DONE").unwrap_or(&0),
1006            *c.by_status.get("READY").unwrap_or(&0),
1007            *c.by_status.get("ACTIVE").unwrap_or(&0),
1008            *c.by_status.get("VERIFY").unwrap_or(&0),
1009        );
1010        let mcov = r
1011            .evidence_required
1012            .iter()
1013            .all(|need| r.evidence_have.contains(need));
1014        let (tot, req) = (c.total, c.required);
1015        let ms = if r.cancelled {
1016            "CANCELLED"
1017        } else if !r.superseded_by.is_empty() {
1018            "SUPERSEDED"
1019        } else if tot == 0 && !r.evidence_required.is_empty() && mcov {
1020            "DONE"
1021        } else if tot == 0 {
1022            "PLANNED"
1023        } else if req > 0 && nd == req && mcov {
1024            "DONE"
1025        } else if req > 0 && nd == req {
1026            "VERIFY"
1027        } else if na > 0 || nv > 0 || nd > 0 {
1028            "ACTIVE"
1029        } else if nr == 0 {
1030            "BLOCKED"
1031        } else {
1032            "PLANNED"
1033        };
1034        mstatus[n] = ms.to_string();
1035        mempty[n] = tot == 0;
1036        if req > 0 && nd == req && !mcov {
1037            let miss: Vec<&str> = r
1038                .evidence_required
1039                .iter()
1040                .filter(|need| !r.evidence_have.contains(need))
1041                .map(String::as_str)
1042                .collect();
1043            v.warn(
1044                "milestone_evidence_missing",
1045                &r.id,
1046                format!(
1047                    "every issue is DONE but milestone evidence is missing for {}; status stays VERIFY",
1048                    miss.join(",")
1049                ),
1050            );
1051        }
1052        counts.push(c);
1053    }
1054
1055    // --- the gate. A milestone whose required outcomes are not yet real cannot be
1056    //     executable, whatever its own issues say. Run in rank order, so a dependency is
1057    //     always decided before the milestone requiring it and the block cascades.
1058    let mut mblocked: Vec<Vec<String>> = vec![Vec::new(); mraw.len()];
1059    for r in 0..=mlayer {
1060        for (n, m) in mraw.iter().enumerate() {
1061            if rank[n] != r {
1062                continue;
1063            }
1064            let block: Vec<String> = m
1065                .depends_on
1066                .iter()
1067                .filter(|d| medge.contains(&(m.id.clone(), (*d).clone())))
1068                .filter(|d| mx(d).is_some_and(|dn| mstatus[dn] != "DONE"))
1069                .cloned()
1070                .collect();
1071            mblocked[n] = block.clone();
1072            if block.is_empty() {
1073                continue;
1074            }
1075            if mstatus[n] == "CANCELLED" || mstatus[n] == "SUPERSEDED" {
1076                continue;
1077            }
1078            if matches!(mstatus[n].as_str(), "DONE" | "VERIFY" | "ACTIVE") {
1079                v.fail(
1080                    "milestone_premature",
1081                    &m.id,
1082                    format!("is {} while {} is not DONE", mstatus[n], block.join(",")),
1083                );
1084            }
1085            mstatus[n] = "BLOCKED".into();
1086        }
1087    }
1088
1089    for (n, m) in mraw.iter().enumerate() {
1090        if !mempty[n] {
1091            continue;
1092        }
1093        if matches!(mstatus[n].as_str(), "DONE" | "CANCELLED" | "SUPERSEDED") {
1094            continue;
1095        }
1096        if !mblocked[n].is_empty() {
1097            continue;
1098        }
1099        v.warn(
1100            "empty_milestone",
1101            &m.id,
1102            "is reachable and has no issues; it is an outcome nobody is executing".into(),
1103        );
1104    }
1105
1106    // --- the gate reaches the work. An issue inside a milestone the gate has blocked is not
1107    //     executable, whatever its own dependencies say: the outcome it belongs to is not
1108    //     reachable yet.
1109    for (n, r) in iraw.iter().enumerate() {
1110        if r.milestone.is_empty() {
1111            continue;
1112        }
1113        let Some(mn) = mx(&r.milestone) else { continue };
1114        if mblocked[mn].is_empty() {
1115            continue;
1116        }
1117        if status[n] == "CANCELLED" || status[n] == "DONE" {
1118            continue;
1119        }
1120        if status[n] == "ACTIVE" || status[n] == "VERIFY" {
1121            v.fail(
1122                "premature_execution",
1123                &r.id,
1124                format!(
1125                    "is {} while its milestone {} waits on {}",
1126                    status[n],
1127                    r.milestone,
1128                    mblocked[mn].join(",")
1129                ),
1130            );
1131        }
1132        status[n] = "BLOCKED".into();
1133        blocked_by[n].push(format!("milestone:{}", r.milestone));
1134    }
1135
1136    // counts follow the statuses, so a blocked milestone cannot report ready work
1137    for (n, m) in mraw.iter().enumerate() {
1138        if mblocked[n].is_empty() {
1139            continue;
1140        }
1141        counts[n] = tally(&iraw, &status, &m.id);
1142    }
1143
1144    // --- scope conflict: two issues that could run together but touch the same paths.
1145    //     Reported conservatively — an uncertain overlap serialises rather than races.
1146    for a in 0..iraw.len() {
1147        for b in (a + 1)..iraw.len() {
1148            if !matches!(status[a].as_str(), "READY" | "ACTIVE") {
1149                continue;
1150            }
1151            if !matches!(status[b].as_str(), "READY" | "ACTIVE") {
1152                continue;
1153            }
1154            if wave[a] != wave[b] {
1155                continue;
1156            }
1157            'pair: for x in &iraw[a].scope {
1158                for y in &iraw[b].scope {
1159                    if overlap(x, y) {
1160                        v.warn(
1161                            "scope_conflict",
1162                            &iraw[a].id,
1163                            format!(
1164                                "shares {x} with {}; they may not run concurrently",
1165                                iraw[b].id
1166                            ),
1167                        );
1168                        break 'pair;
1169                    }
1170                }
1171            }
1172        }
1173    }
1174
1175    // --- the active milestone: the lowest-ranked unblocked milestone that is ACTIVE, else
1176    //     the lowest-ranked unblocked one not finished.
1177    let mut best: Option<usize> = None;
1178    for pass in 1..=2 {
1179        for n in 0..mraw.len() {
1180            if mstatus[n] == "DONE" || mstatus[n] == "CANCELLED" {
1181                continue;
1182            }
1183            if pass == 1 && mstatus[n] != "ACTIVE" {
1184                continue;
1185            }
1186            if !mblocked[n].is_empty() {
1187                continue;
1188            }
1189            let o = i64::from(rank[n]) * 1_000_000 + order_of(&mraw[n].order);

1330/// What [`released_in`] answers for a milestone no release carried, or not yet.
1331pub const UNRELEASED: &str = "unreleased";

1333/// The release that carried a milestone: the first release record
1334/// (`.ai/repo/releases/v*.yaml`, the tracked statement of a tag), in version order, whose
1335/// `commit:` contains the merge of its last accepted issue, and [`UNRELEASED`] for a
1336/// milestone that is not DONE or that no record's commit contains yet. A milestone DONE with
1337/// no issue was finished by its own evidence: the commit that wrote the latest `recorded_at`
1338/// of that evidence stands in for the merge.
1339///
1340/// Records and not tags, because derived data must not depend on refs outside the tree: the
1341/// answer is a function of the commit, and it changes only when a record lands. Releases
1342/// before the first record have none, so work finished before it reads as the earliest
1343/// release this layer records.
1344///
1345/// The merge is found through the commit that wrote the issue's `completed_at` stamp — the
1346/// oldest commit whose diff of the record adds or removes that text — and failing that
1347/// through the latest merge whose message names the issue. A release commit containing
1348/// either contains the merge that brought it to the trunk. This is `mj_pj_m_released_in` in
1349/// `lib/project.sh`, command for command, so the two engines agree
1350/// (`test/cases/99_plan_capabilities.sh`).
1351///
1352/// ```no_run
1353/// # use majordomus_cli::index::Index;
1354/// use majordomus_cli::plan::{released_in, Plan, UNRELEASED};
1355/// # fn demo(index: &Index) {
1356/// let plan = Plan::build(index);
1357/// for m in &plan.milestones {
1358///     let release = released_in(index, &plan, &m.id);
1359///     assert!(release == UNRELEASED || release.starts_with('v'));
1360/// }
1361/// # }
1362/// ```
1363pub fn released_in(index: &Index, plan: &Plan, milestone: &str) -> String {
1364    let root = std::path::Path::new(&index.repository.root);
1365    let Some(m) = plan.milestone(milestone).filter(|m| m.status == "DONE") else {
1366        return UNRELEASED.to_string();
1367    };
1368    let last = m.last_accepted.as_str();
1369    let git = |args: &[&str]| -> Option<std::process::Output> {
1370        crate::git::read_only(root)
1371            .args(args)
1372            .output()
1373            .ok()
1374            .filter(|o| o.status.success())
1375    };
1376    let last_line = |args: &[&str]| -> Option<String> {
1377        git(args).and_then(|o| {
1378            String::from_utf8_lossy(&o.stdout)
1379                .lines()
1380                .last()
1381                .map(str::to_string)
1382        })
1383    };
1384    let record = |kind: &str, id: &str| {
1385        index
1386            .objects
1387            .iter()
1388            .find(|o| o.kind == kind && o.identity == id)
1389    };
1390    let sha = if last.is_empty() {
1391        // DONE with no issue: its own evidence finished it, so the commit that wrote the
1392        // latest `recorded_at` of that evidence is the one a release has to contain
1393        record(MILESTONE, &m.id).and_then(|o| {
1394            let stamp = last_evidence_stamp(&o.metadata)?;
1395            last_line(&[
1396                "log",
1397                "--format=%H",
1398                &format!("-S{stamp}"),
1399                "--",
1400                &o.provenance.path,
1401            ])
1402        })
1403    } else {
1404        let stamp = plan
1405            .issue(last)
1406            .map(|i| i.completed_at.as_str())
1407            .unwrap_or_default();
1408        let grep = format!("--grep=(^|[^A-Za-z0-9]){last}([^A-Za-z0-9]|$)");
1409        record(ISSUE, last)
1410            .and_then(|o| {
1411                last_line(&[
1412                    "log",
1413                    "--format=%H",
1414                    &format!("-S{stamp}"),
1415                    "--",
1416                    &o.provenance.path,
1417                ])
1418            })
1419            .or_else(|| last_line(&["log", "--merges", "-1", "--format=%H", "-E", &grep]))
1420    };
1421    let Some(sha) = sha else {
1422        return UNRELEASED.to_string();
1423    };
1424    release_records(root)
1425        .into_iter()
1426        .find(|(_, commit)| git(&["merge-base", "--is-ancestor", &sha, commit]).is_some())
1427        .map_or_else(|| UNRELEASED.to_string(), |(tag, _)| tag)
1428}

1430/// The release records under `<root>/.ai/repo/releases/`, `(tag, commit)` in ascending
1431/// version order: every record whose top-level `tag:` is a release tag and that names a
1432/// `commit:`. Read line by line, the way `mj_pj_release_records` reads them. Collected into
1433/// a set keyed by the version first, so the set is the order and the directory walk is not.
1434fn release_records(root: &std::path::Path) -> Vec<(String, String)> {
1435    let dir = root.join(crate::distribution::release::DIR);
1436    let records: BTreeSet<((u64, u64, u64), String, String)> = std::fs::read_dir(dir)
1437        .into_iter()
1438        .flatten()
1439        .flatten()
1440        .map(|e| e.path())
1441        .filter(|p| p.extension().is_some_and(|x| x == "yaml"))
1442        .filter_map(|p| {
1443            let text = std::fs::read_to_string(p).ok()?;
1444            let top = |key: &str| {
1445                text.lines()
1446                    .find_map(|l| l.strip_prefix(key))
1447                    .map(|v| v.trim().replace('"', ""))
1448            };
1449            let tag = top("tag:")?;
1450            let commit = top("commit:").filter(|c| !c.is_empty())?;
1451            Some((release_key(&tag)?, tag, commit))
1452        })
1453        .collect();
1454    records
1455        .into_iter()
1456        .map(|(_, tag, commit)| (tag, commit))
1457        .collect()
1458}

1460/// `vMAJOR.MINOR.PATCH` as numbers, and nothing else: a pre-release or a named tag is not a
1461/// release.
1462fn release_key(tag: &str) -> Option<(u64, u64, u64)> {
1463    let mut parts = tag.strip_prefix('v')?.split('.');
1464    let mut num = || -> Option<u64> {
1465        let p = parts
1466            .next()
1467            .filter(|p| !p.is_empty() && p.bytes().all(|b| b.is_ascii_digit()))?;
1468        p.parse().ok()
1469    };
1470    let key = (num()?, num()?, num()?);
1471    parts.next().is_none().then_some(key)
1472}

1476fn order_of(raw: &str) -> i64 {
1477    raw.trim().parse::<i64>().unwrap_or(0)

1482/// before the gate and once after it, because the gate changes statuses underneath it.
1483fn tally(iraw: &[PlanIssueRaw], status: &[String], milestone: &str) -> PlanCounts {

1487        .collect();

1491            continue;

1493        total += 1;
1494        *by.entry(status[n].clone()).or_insert(0) += 1;
1495    }
1496    let cancelled = *by.get("CANCELLED").unwrap_or(&0);
1497    PlanCounts {
1498        total,
1499        required: total - cancelled,
1500        by_status: by,
1501    }

1713// ---------------------------------------------------------------- transitions
1714//
1715// The write half of the plan, and the first mutating capability of the development
1716// lifecycle. Everything above this line derives; everything below it changes one field of
1717// one record and records that it did.
1718//
1719// `lib/plan.sh`'s `mj_plan_transition` has been the only implementation of these three
1720// moves. This is the second, and the contract it is held to is byte equality of the file it
1721// writes — `test/cases/133_plan_transition.sh` performs the same transition through both
1722// engines over the same fixture and diffs the record. The guards are stated here in the
1723// same order the shell states them, because the *first* refusal a caller meets is part of
1724// the behaviour: an issue that is both blocked and missing its evidence must be told about
1725// the blocker, which is the thing it can act on.
1726
1727/// One lifecycle move.
1728///
1729/// Three, not four: `evidence` appends a record rather than moving an issue, and it belongs
1730/// with the record it proves rather than with the moves it unblocks.
1731///
1732/// ```
1733/// use majordomus_cli::plan::Transition;
1734///
1735/// // The wire spelling is the one a caller types on the command line.
1736/// let t: Transition = serde_json::from_str("\"done\"").unwrap();
1737/// assert_eq!(t, Transition::Done);
1738/// assert_eq!(serde_json::to_string(&Transition::Start).unwrap(), "\"start\"");
1739///
1740/// // Each move owns the field it stamps and the event it records; nothing else decides
1741/// // either, so a new move cannot be added without saying both.
1742/// for (t, field, event) in [
1743///     (Transition::Start, "started_at", "plan_start"),
1744///     (Transition::Verify, "verified_at", "plan_verify"),
1745///     (Transition::Done, "completed_at", "plan_done"),
1746/// ] {
1747///     assert_eq!(t.field(), field);

1754// cannot mean two things: the move an issue makes is named for the plan it belongs to.

1757    /// Execution began. Refused unless the issue is READY.
1758    Start,
1759    /// Implementation is complete and the evidence is still outstanding.
1760    Verify,
1761    /// Complete. Refused while anything it waits on is unfinished, or while any evidence
1762    /// its own record requires is absent.
1763    Done,
1764}

1837/// Why a transition did not happen.
1838///
1839/// Each variant carries what the caller needs to act, never only that it failed: which
1840/// status the issue is actually in, which dependency holds it, which evidence token is
1841/// missing. The messages are the shell's messages.
1842///
1843/// ```
1844/// use majordomus_cli::plan::TransitionError;
1845///
1846/// // A refusal says what is in the way, because that is the thing the caller can change.
1847/// let refused = TransitionError::Refused("I0001 is DONE, not READY".into());
1848/// assert_eq!(refused.to_string(), "I0001 is DONE, not READY");
1849///
1850/// // A missing issue is a different answer from an illegal move, and callers map them to
1851/// // different statuses: not_found rather than refused.
1852/// let missing = TransitionError::NoSuchIssue("I9999".into());
1853/// assert_eq!(missing.to_string(), "no issue 'I9999'");
1854/// assert_ne!(missing, refused);
1855/// ```
1856#[derive(Debug, Clone, PartialEq, Eq)]
1857pub enum TransitionError {
1858    /// No issue of that id.
1859    NoSuchIssue(String),
1860    /// The move is illegal from the status the issue is in.
1861    Refused(String),
1862    /// The record could not be rewritten.
1863    Write {
1864        /// The record.
1865        path: std::path::PathBuf,
1866        /// What the filesystem said.
1867        reason: String,
1868    },
1869}
1870
1871impl std::fmt::Display for TransitionError {
1872    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
1873        match self {
1874            Self::NoSuchIssue(id) => write!(f, "no issue '{id}'"),
1875            Self::Refused(m) => f.write_str(m),
1876            Self::Write { path, reason } => write!(f, "{}: {reason}", path.display()),
1877        }
1878    }
1879}

1883/// Whether `transition` is legal for `issue` as the plan reads now, and why not when it is
1884/// not.
1885///
1886/// Separate from the write so that a surface can offer or withhold a control without
1887/// performing it: the Cockpit disables a button by asking this, and never by reimplementing
1888/// the rule.
1889///
1890/// ```no_run
1891/// # use majordomus_cli::index::Index;
1892/// use majordomus_cli::plan::{check, Plan, Transition, TransitionError};
1893/// # fn demo(index: &Index) {
1894/// let plan = Plan::build(index);
1895/// let now = "2026-09-11T12:00:00Z";
1896/// // Offer the control only when the model would accept it, and show the reason when not.
1897/// let offered = check(index, &plan, "I0001", Transition::Start, now).is_ok();
1898///
1899/// // An issue the plan does not hold is not found, never merely refused: the two are
1900/// // different answers and a surface maps them to different statuses.
1901/// let missing = check(index, &plan, "no-such-issue", Transition::Start, now);
1902/// assert_eq!(missing, Err(TransitionError::NoSuchIssue("no-such-issue".into())));
1903///
1904/// // Asking twice never changes anything: this is the question, not the move.
1905/// assert_eq!(offered, check(index, &plan, "I0001", Transition::Start, now).is_ok());
1906/// # }
1907/// ```
1908pub fn check(
1909    index: &Index,
1910    plan: &Plan,
1911    id: &str,
1912    transition: Transition,
1913    now: &str,
1914) -> Result<(), TransitionError> {
1915    let issue = plan
1916        .issue(id)
1917        .ok_or_else(|| TransitionError::NoSuchIssue(id.to_string()))?;
1918    match transition {
1919        Transition::Start => {
1920            if issue.status != "READY" {
1921                let waiting = if issue.blocked_by.is_empty() {
1922                    String::new()
1923                } else {
1924                    format!(" (waiting on {})", issue.blocked_by.join(" "))
1925                };
1926                return Err(TransitionError::Refused(format!(
1927                    "{id} is {}, not READY{waiting}",
1928                    issue.status
1929                )));
1930            }
1931            // where the policy requires binding, a transition is held to it as a task is
1932            if let Some(why) = crate::intent_binding::start_refusal(index, plan, id) {
1933                return Err(TransitionError::Refused(why));
1934            }
1935        }
1936        Transition::Verify => {
1937            if !matches!(issue.status.as_str(), "ACTIVE" | "VERIFY") {
1938                return Err(TransitionError::Refused(format!(
1939                    "{id} is {}; only an ACTIVE issue can move to VERIFY",
1940                    issue.status
1941                )));
1942            }
1943        }
1944        Transition::Done => {
1945            // The blocker first: an issue that is both blocked and short of evidence is
1946            // told about the blocker, because that is the one it cannot act on alone.
1947            if !issue.blocked_by.is_empty() {
1948                return Err(TransitionError::Refused(format!(
1949                    "{id} cannot be DONE while {} is not DONE",
1950                    issue.blocked_by.join(" ")
1951                )));
1952            }
1953            // Done ends the lifecycle; it does not replace it. An issue is DONE only after
1954            // it was started, so READY, BLOCKED, CANCELLED and DONE itself are refused: the
1955            // stamp would otherwise claim an execution no `plan_start` ever recorded, and a
1956            // cancelled issue would be given a completion the plan had given up on.
1957            if !matches!(issue.status.as_str(), "ACTIVE" | "VERIFY") {
1958                return Err(TransitionError::Refused(format!(
1959                    "{id} is {}; only an ACTIVE or VERIFY issue can move to DONE",
1960                    issue.status
1961                )));
1962            }
1963            // Whether the evidence is complete is not a count — a record may carry two
1964            // entries covering one token and none covering another — and the rule for it
1965            // already exists inside `derive`. So rather than restate it, ask what the plan
1966            // would say: write the stamp in memory and see whether the issue reaches DONE.
1967            // An issue that stays VERIFY with `completed_at` set is exactly the record
1968            // `derive` reports as "completed_at is set but evidence is missing".
1969            let after = Plan::after(index, id, transition, now);
1970            if after.issue(id).is_some_and(|i| i.status != "DONE") {
1971                return Err(TransitionError::Refused(format!(
1972                    "{id} has evidence for {} of {} requirement(s) (run: majordomus plan evidence {id} --covers <token> ...)",
1973                    issue.evidence_have, issue.evidence_need
1974                )));
1975            }
1976        }
1977    }
1978    Ok(())
1979}

1981/// Perform one transition: check it, stamp the record, append the event.
1982///
1983/// The order is the shell's and is not an implementation detail. The guard runs against the
1984/// plan as it reads now; the record is rewritten; only then is the event appended. A ledger
1985/// line therefore never describes a change that did not reach the file, which is the
1986/// property that lets the record be read as history rather than as intent.
1987///
1988/// `path` is the issue's own file, as the index recorded it. It is passed in rather than
1989/// derived from the id here, because the index already knows where every record lives and a
1990/// second convention for locating one is how two readers come to disagree about which file
1991/// is the record.
1992///
1993/// ```no_run
1994/// # use std::path::Path;
1995/// # use majordomus_cli::index::Index;
1996/// # use majordomus_cli::git::GitState;
1997/// use majordomus_cli::ledger::{self, Vocabulary};
1998/// use majordomus_cli::plan::{transition, Plan, Transition};
1999/// # fn demo(index: &Index, root: &Path, record: &Path, git: &GitState) {
2000/// let plan = Plan::build(index);
2001/// let vocabulary = Vocabulary::load(&root.join("share/events.yaml")).unwrap();
2002/// let at = ledger::now();
2003/// // The guard runs first, the record is rewritten, and only then is the event appended:
2004/// // a ledger line never describes a change that did not reach the file.
2005/// transition(root, record, index, &plan, "I0001", Transition::Start, &at, &vocabulary, git)
2006///     .expect("a READY issue may start");
2007/// # }
2008/// ```
2009#[allow(clippy::too_many_arguments)]
2010pub fn transition(
2011    root: &std::path::Path,
2012    path: &std::path::Path,
2013    index: &Index,
2014    plan: &Plan,
2015    id: &str,
2016    transition: Transition,
2017    now: &str,
2018    vocabulary: &crate::ledger::Vocabulary,
2019    git: &crate::git::GitState,
2020) -> Result<(), TransitionError> {
2021    check(index, plan, id, transition, now)?;
2022    let text = std::fs::read_to_string(path).map_err(|e| TransitionError::Write {
2023        path: path.to_path_buf(),
2024        reason: e.to_string(),
2025    })?;
2026    // The fields in the order the shell writes them: the move's own stamp, its seal, then
2027    // `updated_at`. A record whose `updated_at` preceded its `completed_at` would be a
2028    // record that says it was finished before it was last touched.
2029    // The seal is written with the stamp, in the same write, because it is the only part of
2030    // the event a clone carries: the ledger line below stays in this checkout (ADR 0097).
2031    let text = set_field(&text, transition.field(), now);
2032    let text = set_field(&text, transition.seal_field(), &seal(transition, id, now));
2033    let text = set_field(&text, "updated_at", now);
2034    std::fs::write(path, text).map_err(|e| TransitionError::Write {
2035        path: path.to_path_buf(),
2036        reason: e.to_string(),
2037    })?;
2038    crate::ledger::append(
2039        root,
2040        vocabulary,
2041        git,
2042        now,
2043        transition.event(),
2044        &[("issue", id.to_string())],
2045    )
2046    .map_err(|e| TransitionError::Write {
2047        path: root.join(".ai/local/state/ledger.jsonl"),
2048        reason: e.to_string(),
2049    })?;
2050    Ok(())
2051}

2053/// Set one top-level scalar field of a record, exactly as `mj_pj_set_field` in
2054/// `lib/project.sh` does.
2055///
2056/// The rule is small and every part of it is load-bearing:
2057///
2058/// * a line beginning `key:` is replaced in place, keeping its position in the file, so a
2059///   record's field order survives every write;
2060/// * otherwise the field is inserted immediately *before* `evidence:`, because evidence is
2061///   a block and a scalar appended after it would be read as one of its entries;
2062/// * with no `evidence:` line, the field is appended at the end.
2063///
2064/// Only the first `evidence:` line is an anchor, and matching is on the line's start, which
2065/// is what `grep -qE "^$key:"` and the awk beside it do. A nested key of the same name is
2066/// indented and is therefore not matched — the same blindness, deliberately, because the
2067/// two engines must be blind in the same places.
2068fn set_field(text: &str, key: &str, value: &str) -> String {
2069    let prefix = format!("{key}:");
2070    let line = format!("{key}: {value}");
2071    if text.lines().any(|l| l.starts_with(&prefix)) {
2072        let mut out = String::with_capacity(text.len() + line.len());
2073        for l in text.lines() {
2074            if l.starts_with(&prefix) {
2075                out.push_str(&line);
2076            } else {
2077                out.push_str(l);
2078            }
2079            out.push('\n');
2080        }
2081        return out;
2082    }
2083    let mut out = String::with_capacity(text.len() + line.len() + 1);
2084    let mut inserted = false;
2085    for l in text.lines() {
2086        if !inserted && l.starts_with("evidence:") {
2087            out.push_str(&line);
2088            out.push('\n');
2089            inserted = true;
2090        }
2091        out.push_str(l);
2092        out.push('\n');
2093    }
2094    if !inserted {
2095        out.push_str(&line);
2096        out.push('\n');
2097    }
2098    out
2099}
