1//! Each task's delivery economics, folded from every checkout's ledger (I3376).
2//!
3//! The owner's report of 2026-10-19 asks, for each task, its time, its interventions, its CI
4//! reruns, its spend and whether its result was accepted (T3). The ledger already records a
5//! task's steps — `task.started`, every `task.refused`, `task.finished` — and the episodes that
6//! carried it, but it is per checkout and untracked, and the one reader that folded it
7//! ([`crate::intent_realization::gather`]) read only the root checkout's. This module reads the
8//! ledger of every checkout `git worktree list` reports, with the archives `ledger.rotated`
9//! left beside it, and answers one record per task.
10//!
11//! Wall time, refusals and episodes are measured. Interventions, CI reruns, spend and
12//! acceptance are not yet observed by anything; each is present on every record as
13//! [`TaskEconomicsFieldStatus::NotObserved`] with the reason and the issue that will supply it,
14//! so later issues fill a field without changing the response type.
15//!
16//! ```
17//! use majordomus_cli::economics::tasks::{fold, TaskEconomicsQuery, TaskEconomicsSource};
18//! let line = |ts: &str, event: &str, extra: &str| {
19//!     format!(r#"{{"ts":"{ts}","event":"{event}","head":"h","branch":"b","by":"me","session":"s-1","task_id":"t-1"{extra}}}"#)
20//! };
21//! let text = [
22//!     line("2026-10-21T10:00:00Z", "task.started", r#","issue":"I3376""#),
23//!     line("2026-10-21T10:30:00Z", "task.refused", r#","outcome":"completed","unmet":1"#),
24//!     line("2026-10-21T11:00:00Z", "task.finished", r#","outcome":"completed""#),
25//! ]
26//! .join("\n");
27//! let source = TaskEconomicsSource::parse("repo", "ledger.jsonl", &text);
28//! let list = fold(vec![source], &TaskEconomicsQuery::default());
29//! assert_eq!(list.count, 1);
30//! let t = &list.tasks[0];
31//! assert_eq!(t.wall_seconds, Some(3600));
32//! assert_eq!(t.refusals, 1);
33//! assert_eq!(t.issue.as_deref(), Some("I3376"));
34//! ```

52/// Which tasks to answer. A window filters by each task's `task.started` time, so a task
53/// with no start line in the ledgers read is answered only when no window is given.
54///
55/// ```
56/// use majordomus_cli::economics::tasks::TaskEconomicsQuery;
57/// let q: TaskEconomicsQuery = serde_json::from_str(r#"{"since":"2026-10-21T00:00:00Z"}"#).unwrap();
58/// assert_eq!(q.since.as_deref(), Some("2026-10-21T00:00:00Z"));
59/// assert!(q.until.is_none());
60/// assert_eq!(serde_json::from_str::<TaskEconomicsQuery>("{}").unwrap(), TaskEconomicsQuery::default());
61/// // a misspelt bound is refused, never read as no bound
62/// assert!(serde_json::from_str::<TaskEconomicsQuery>(r#"{"after":"x"}"#).is_err());
63/// ```
64#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize, JsonSchema)]
65#[serde(deny_unknown_fields, default)]
66pub struct TaskEconomicsQuery {
67    /// Answer only tasks started at or after this time (`YYYY-MM-DDTHH:MM:SSZ`).
68    #[serde(skip_serializing_if = "Option::is_none")]
69    pub since: Option<String>,
70    /// Answer only tasks started before this time (`YYYY-MM-DDTHH:MM:SSZ`).
71    #[serde(skip_serializing_if = "Option::is_none")]
72    pub until: Option<String>,
73}

100/// One task's delivery economics: who carried it, when, how often `finish` refused it, and
101/// the fields nothing observes yet, each saying why it has no value.
102///
103/// ```
104/// use majordomus_cli::economics::tasks::{
105///     fold, TaskEconomicsRecord, TaskEconomicsSource, TaskEconomicsState,
106/// };
107/// let text = r#"{"ts":"2026-10-21T10:00:00Z","event":"task.started","head":"h","branch":"b",
108///     "by":"me","session":"s-1","task_id":"t-1","issue":"I3376"}"#;
109/// let source = TaskEconomicsSource::parse("repo", "ledger.jsonl", &text.replace('\n', ""));
110/// let list = fold(vec![source], &Default::default());
111/// let r: &TaskEconomicsRecord = &list.tasks[0];
112/// assert_eq!((r.task_id.as_str(), r.branch.as_str()), ("t-1", "b"));
113/// assert_eq!(r.state, TaskEconomicsState::Open);
114/// assert_eq!(r.outcome, "active");
115/// assert_eq!(r.wall_seconds, None, "an open task with no refusal has no end");
116/// assert_eq!(r.issue.as_deref(), Some("I3376"));
117/// assert!(r.spend.reason.contains("I3379"));
118/// ```
119#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
120pub struct TaskEconomicsRecord {
121    /// The task's id, as `start` wrote it.
122    pub task_id: String,
123    /// The issue the task was started under, when it named one.
124    #[serde(skip_serializing_if = "Option::is_none")]
125    pub issue: Option<String>,
126    /// The intent the task was started under, when it named one.
127    #[serde(skip_serializing_if = "Option::is_none")]
128    pub intent: Option<String>,
129    /// The checkout whose ledger carries the task.
130    pub checkout: String,
131    /// The branch the task started on.
132    pub branch: String,
133    /// `finished` once a `task.finished` line is read, `open` otherwise.
134    pub state: TaskEconomicsState,
135    /// The outcome as recorded: the finish's outcome, or `active` / `handed_over` while open.
136    pub outcome: String,
137    /// The time of the task's first `task.started` line.
138    #[serde(skip_serializing_if = "Option::is_none")]
139    pub started_at: Option<String>,
140    /// The time of the task's last `task.finished` line, or of its last `task.refused` line
141    /// while it has none.
142    #[serde(skip_serializing_if = "Option::is_none")]
143    pub ended_at: Option<String>,
144    /// Seconds from `started_at` to `ended_at`; absent while either is.
145    #[serde(skip_serializing_if = "Option::is_none")]
146    pub wall_seconds: Option<i64>,
147    /// How many times `finish` refused the task.
148    pub refusals: u32,
149    /// The episodes that carried the task, each with the span of its lines about it.
150    pub episodes: Vec<TaskEconomicsEpisode>,
151    /// Human interventions during the task.
152    pub interventions: TaskEconomicsField,
153    /// CI runs of the task's work that were rerun.
154    pub ci_reruns: TaskEconomicsField,
155    /// What the task cost.
156    pub spend: TaskEconomicsField,
157    /// Whether the task's result was accepted.
158    pub accepted: TaskEconomicsField,
159}

298/// The entries of one checkout's ledgers, in the order they were written, and what was read.

306/// assert_eq!(s.entries[0].event, "task.started");
307/// ```

309pub struct TaskEconomicsSource {
310    /// The checkout the entries come from.
311    pub checkout: String,
312    /// The entries, archives first, then the live ledger.
313    pub entries: Vec<Entry>,
314    /// The files read.
315    pub ledgers: Vec<TaskEconomicsLedger>,

317
318impl TaskEconomicsSource {
319    /// A source holding one ledger file's text.
320    ///
321    /// ```
322    /// use majordomus_cli::economics::tasks::TaskEconomicsSource;
323    /// let s = TaskEconomicsSource::parse("repo", "ledger.jsonl", "not json\n\n");
324    /// assert_eq!((s.ledgers[0].lines, s.ledgers[0].skipped), (0, 1));
325    /// ```
326    pub fn parse(checkout: &str, path: &str, text: &str) -> Self {
327        let mut s = TaskEconomicsSource {
328            checkout: checkout.to_string(),
329            ..Default::default()
330        };
331        s.push(path, text);
332        s
333    }
334
335    /// Append one more ledger file's text to this source: its entries after the ones already
336    /// held, and one more [`TaskEconomicsLedger`] counting its lines.
337    ///
338    /// ```
339    /// use majordomus_cli::economics::tasks::TaskEconomicsSource;
340    /// let line = |id: &str| {
341    ///     format!(r#"{{"ts":"2026-10-21T10:00:00Z","event":"task.started","head":"h","branch":"b","by":"me","task_id":"{id}"}}"#)
342    /// };
343    /// let mut s = TaskEconomicsSource::parse("repo", "ledger.1.jsonl.archived", &line("old"));
344    /// s.push("ledger.jsonl", &format!("{}\nbroken\n", line("new")));
345    /// assert_eq!(s.ledgers.len(), 2);
346    /// assert_eq!((s.ledgers[1].lines, s.ledgers[1].skipped), (1, 1));
347    /// assert_eq!(s.entries[1].payload["task_id"], "new", "appended after the archive");
348    /// ```
349    pub fn push(&mut self, path: &str, text: &str) {
350        let (mut lines, mut skipped) = (0, 0);
351        for line in text.lines().filter(|l| !l.trim().is_empty()) {
352            match serde_json::from_str::<Entry>(line) {
353                Ok(e) => {
354                    self.entries.push(e);
355                    lines += 1;
356                }
357                Err(_) => skipped += 1,
358            }
359        }
360        self.ledgers.push(TaskEconomicsLedger {
361            checkout: self.checkout.clone(),
362            path: path.to_string(),
363            lines,
364            skipped,
365        });
366    }
367}

369fn payload(e: &Entry, key: &str) -> String {
370    e.payload
371        .get(key)
372        .and_then(|v| v.as_str())
373        .unwrap_or_default()
374        .to_string()
375}

384fn in_window(started_at: Option<&str>, q: &TaskEconomicsQuery) -> bool {
385    if q.since.is_none() && q.until.is_none() {
386        return true;
387    }
388    let Some(at) = started_at.and_then(crate::peers::epoch_seconds) else {
389        return false;
390    };
391    let after = |b: &Option<String>| b.as_deref().and_then(crate::peers::epoch_seconds);
392    after(&q.since).is_none_or(|s| at >= s) && after(&q.until).is_none_or(|u| at < u)
393}

395/// One record per distinct task id across the sources, filtered by the window.
396///
397/// Each source is folded on its own with [`tasks_from_ledger`], which names the episodes and
398/// the recorded outcome; the times, the refusals and each episode's span are read from the
399/// same lines here. A task id met in two sources is one task.
400///
401/// ```
402/// use majordomus_cli::economics::tasks::{fold, TaskEconomicsQuery, TaskEconomicsSource};
403/// let line = |ts: &str, event: &str, id: &str| {
404///     format!(r#"{{"ts":"{ts}","event":"{event}","head":"h","branch":"b","by":"me","session":"s-1","task_id":"{id}"}}"#)
405/// };
406/// let a = TaskEconomicsSource::parse("a", "ledger.jsonl", &line("2026-10-21T10:00:00Z", "task.started", "t-1"));
407/// let b = TaskEconomicsSource::parse("b", "ledger.jsonl", &[
408///     line("2026-10-21T11:00:00Z", "task.finished", "t-1"),
409///     line("2026-10-22T09:00:00Z", "task.started", "t-2"),
410/// ].join("\n"));
411/// let all = fold(vec![a.clone(), b.clone()], &TaskEconomicsQuery::default());
412/// assert_eq!(all.count, 2, "t-1 met in two checkouts is one task");
413/// assert_eq!(all.tasks[0].wall_seconds, Some(3600));
414/// let window = TaskEconomicsQuery { since: Some("2026-10-22T00:00:00Z".into()), until: None };
415/// let late = fold(vec![a, b], &window);
416/// assert_eq!(late.tasks.iter().map(|t| t.task_id.as_str()).collect::<Vec<_>>(), ["t-2"]);
417/// ```
418pub fn fold(sources: Vec<TaskEconomicsSource>, q: &TaskEconomicsQuery) -> TaskEconomicsList {
419    let mut ledgers = Vec::new();
420    let mut records: BTreeMap<String, TaskEconomicsRecord> = BTreeMap::new();
421    for source in sources {
422        ledgers.extend(source.ledgers.iter().cloned());
423        let units = tasks_from_ledger(&source.entries, &BTreeMap::new());
424        for unit in units {
425            let lines: Vec<&Entry> = source
426                .entries
427                .iter()
428                .filter(|e| e.event.starts_with("task.") && payload(e, "task_id") == unit.id)
429                .collect();
430            let started = lines.iter().find(|e| e.event == "task.started");
431            let finished = lines.iter().rev().find(|e| e.event == "task.finished");
432            let refused: Vec<&&Entry> =
433                lines.iter().filter(|e| e.event == "task.refused").collect();
434            let r = records.entry(unit.id.clone()).or_insert_with(|| TaskEconomicsRecord {
435                task_id: unit.id.clone(),
436                issue: None,
437                intent: None,
438                checkout: source.checkout.clone(),
439                branch: started.map(|e| e.branch.clone()).unwrap_or_default(),
440                state: TaskEconomicsState::Open,
441                outcome: String::new(),
442                started_at: None,
443                ended_at: None,
444                wall_seconds: None,
445                refusals: 0,
446                episodes: Vec::new(),
447                interventions: not_observed(
448                    "nothing records a human intervention per task yet; I3377 supplies it",
449                ),
450                ci_reruns: not_observed(
451                    "no CI run is joined to a task yet; I3378 supplies the rerun count",
452                ),
453                spend: not_observed("no token or cost reading is joined to a task yet; I3379 supplies it"),
454                accepted: not_observed(
455                    "acceptance of a task's result is not recorded yet; I3380 and I3381 supply it",
456                ),
457            });
458            if let Some(s) = started {
459                let issue = payload(s, "issue");
460                let intent = payload(s, "intent");
461                if r.issue.is_none() && !issue.is_empty() {
462                    r.issue = Some(issue);
463                }
464                if r.intent.is_none() && !intent.is_empty() {
465                    r.intent = Some(intent);
466                }
467                if r.started_at.as_deref().is_none_or(|at| s.ts.as_str() < at) {
468                    r.started_at = Some(s.ts.clone());
469                }
470            }
471            r.refusals += refused.len() as u32;
472            if let Some(f) = finished {
473                r.state = TaskEconomicsState::Finished;
474                r.ended_at = Some(f.ts.clone());
475            } else if r.state == TaskEconomicsState::Open {
476                if let Some(last) = refused.last() {
477                    r.ended_at = Some(last.ts.clone());
478                }
479            }
480            r.outcome = if unit.outcome.is_empty() {
481                "active".to_string()
482            } else {
483                unit.outcome.clone()
484            };
485            for ep in &unit.episodes {
486                let mine: Vec<&&Entry> = lines
487                    .iter()
488                    .filter(|e| e.session.as_deref() == Some(ep.session.as_str()))
489                    .collect();
490                // an episode is listed once, across checkouts, spanning its first and last line
491                let known = r.episodes.iter().any(|x| x.session == ep.session);
492                let episode =
493                    mine.first()
494                        .zip(mine.last())
495                        .filter(|_| !known)
496                        .map(|(first, last)| TaskEconomicsEpisode {
497                            session: ep.session.clone(),
498                            provider: ep.provider.clone(),
499                            first_at: first.ts.clone(),
500                            last_at: last.ts.clone(),
501                            lines: mine.len(),
502                        });
503                r.episodes.extend(episode);
504            }
505        }
506    }
507    let mut tasks: Vec<TaskEconomicsRecord> = records
508        .into_values()
509        .filter(|r| in_window(r.started_at.as_deref(), q))
510        .map(|mut r| {
511            r.wall_seconds = match (&r.started_at, &r.ended_at) {
512                (Some(a), Some(b)) => crate::peers::epoch_seconds(b)
513                    .zip(crate::peers::epoch_seconds(a))
514                    .map(|(b, a)| b - a),
515                _ => None,
516            };
517            r
518        })
519        .collect();
520    crate::order::canonical(&mut tasks);
521    TaskEconomicsList {
522        count: tasks.len(),
523        window: q.clone(),
524        tasks,
525        ledgers,
526        unread: Vec::new(),
527    }
528}

530/// The ledgers of one checkout: its rotated archives, oldest first, then its live ledger.
531/// `None` when it holds neither.
532///
533/// ```
534/// use majordomus_cli::economics::tasks::{read_checkout, STATE_DIR};
535/// let dir = tempfile::tempdir().unwrap();
536/// assert!(read_checkout(dir.path(), "repo").is_none(), "no ledger at all");
537/// let state = dir.path().join(STATE_DIR);
538/// std::fs::create_dir_all(&state).unwrap();
539/// std::fs::write(state.join("ledger.20261020T000000Z.jsonl.archived"), "x\n").unwrap();
540/// std::fs::write(state.join("ledger.jsonl"), "y\n").unwrap();
541/// let s = read_checkout(dir.path(), "repo").unwrap();
542/// let paths: Vec<&str> = s.ledgers.iter().map(|l| l.path.as_str()).collect();
543/// assert_eq!(
544///     paths,
545///     [".ai/local/state/ledger.20261020T000000Z.jsonl.archived", ".ai/local/state/ledger.jsonl"],
546///     "archives first, then the live ledger"
547/// );
548/// ```
549pub fn read_checkout(dir: &Path, checkout: &str) -> Option<TaskEconomicsSource> {
550    let state = dir.join(STATE_DIR);
551    let mut archives: Vec<String> = std::fs::read_dir(&state)
552        .map(|rd| {
553            rd.filter_map(|e| e.ok())
554                .map(|e| e.file_name().to_string_lossy().into_owned())
555                .filter(|n| n.starts_with("ledger.") && n.ends_with(ARCHIVE_SUFFIX))
556                .collect()
557        })
558        .unwrap_or_default();
559    // oldest first: an archive's name carries its rotation stamp, which natural order reads
560    crate::order::canonical_strings(&mut archives);
561    let mut source = TaskEconomicsSource {
562        checkout: checkout.to_string(),
563        ..Default::default()
564    };
565    for name in archives
566        .iter()
567        .map(String::as_str)
568        .chain(std::iter::once(LEDGER))
569    {
570        let Ok(text) = std::fs::read_to_string(state.join(name)) else {
571            continue;
572        };
573        source.push(&format!("{STATE_DIR}/{name}"), &text);
574    }
575    (!source.ledgers.is_empty()).then_some(source)
576}

578/// Every task of every checkout of the repository at `root`.
579///
580/// The checkouts are the ones `git worktree list` reports; when git cannot be asked, `root`
581/// alone is read. A checkout git marks prunable, or one holding no ledger, is listed as not
582/// read with the reason.
583///
584/// ```
585/// use majordomus_cli::economics::tasks::{tasks, TaskEconomicsQuery, STATE_DIR};
586/// let dir = tempfile::tempdir().unwrap();
587/// let state = dir.path().join(STATE_DIR);
588/// std::fs::create_dir_all(&state).unwrap();
589/// let line = r#"{"ts":"2026-10-21T10:00:00Z","event":"task.started","head":"h","branch":"b","by":"me","task_id":"t-1"}"#;
590/// std::fs::write(state.join("ledger.jsonl"), format!("{line}\n")).unwrap();
591/// // not a git repository: the directory alone is read
592/// let list = tasks(dir.path(), &TaskEconomicsQuery::default());
593/// assert_eq!(list.count, 1);
594/// assert_eq!(list.tasks[0].checkout, dir.path().display().to_string());
595/// assert!(list.unread.is_empty());
596/// ```
597pub fn tasks(root: &Path, q: &TaskEconomicsQuery) -> TaskEconomicsList {
598    let records = crate::worktree::topology::read(root).unwrap_or_default();
599    let mut sources = Vec::new();
600    let mut unread = Vec::new();
601    let checkouts: Vec<(std::path::PathBuf, Option<String>)> = if records.is_empty() {
602        vec![(root.to_path_buf(), None)]
603    } else {
604        records
605            .iter()
606            .filter(|r| !r.bare)
607            .map(|r| (r.path.clone(), r.prunable.clone()))
608            .collect()
609    };
610    for (dir, prunable) in checkouts {
611        let name = dir.display().to_string();
612        if let Some(why) = prunable {
613            unread.push(TaskEconomicsUnread {
614                checkout: name,
615                reason: format!("git marks it prunable: {why}"),
616            });
617            continue;
618        }
619        match read_checkout(&dir, &name) {
620            Some(s) => sources.push(s),
621            None => unread.push(TaskEconomicsUnread {
622                checkout: name,
623                reason: format!("no ledger under {STATE_DIR}"),
624            }),
625        }
626    }
627    let mut list = fold(sources, q);
628    list.unread = unread;
629    list
630}
