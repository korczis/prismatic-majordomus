1//! Routing and binding, transport-neutral: a request in, a response out.
2//!
3//! The router names no surface. It asks [`super::surfaces::Served`] — the web topology this
4//! process resolved, narrowed to what it can answer — who owns a path, and dispatches to
5//! what that surface bound. A route under `/api/v1/` is a capability with an HTTP exposure;
6//! `/`, `/openapi.json`, `/swagger`, `/cockpit` and `/mcp` are routes the executable answers
7//! itself; `/docs/` and every generated report are directories a producer wrote. All of them
8//! are declared once, in `crate::web::discover`, and reach the router, the home page, the
9//! `web.surfaces` capability and the publication from there.

29/// A request as the router sees it: method, path without query, decoded query pairs,
30/// headers, and the body.
31#[derive(Debug, Clone)]
32pub struct Request {
33    /// `GET`, `POST`, ... as received (a `HEAD` arrives as `GET`).
34    pub method: String,
35    /// The path without the query string.
36    pub path: String,
37    /// The query pairs, percent-decoded, in order.
38    pub query: Vec<(String, String)>,
39    /// Header names and values as received; looked up case-insensitively.
40    pub headers: Vec<(String, String)>,
41    /// The body, raw.
42    pub body: Vec<u8>,
43    /// The address the request came from, when it came over a socket. A request built in
44    /// this process (a test, the Cockpit composing a call) has none, and is this machine's.
45    pub remote: Option<std::net::IpAddr>,
46}
47
48impl Request {
49    /// Split a request target (`/api/v1/objects?kind=rule`) into path and decoded pairs.
50    ///
51    /// ```
52    /// use majordomus_cli::http::Request;
53    /// let r = Request::parse_target("GET", "/api/v1/search?query=Git%20is&limit=2", vec![]);
54    /// assert_eq!(r.path, "/api/v1/search");
55    /// assert_eq!(r.query, vec![("query".to_string(), "Git is".to_string()), ("limit".to_string(), "2".to_string())]);
56    /// ```
57    pub fn parse_target(method: &str, target: &str, body: Vec<u8>) -> Self {
58        let (path, query) = match target.split_once('?') {
59            Some((p, q)) => (p.to_string(), q),
60            None => (target.to_string(), ""),
61        };
62        let query = query
63            .split('&')
64            .filter(|s| !s.is_empty())
65            .map(|pair| {
66                let (k, v) = pair.split_once('=').unwrap_or((pair, ""));
67                (percent_decode(k), percent_decode(v))
68            })
69            .collect();
70        Request {
71            method: method.to_string(),
72            path,
73            query,
74            headers: Vec::new(),
75            body,
76            remote: None,
77        }
78    }
79
80    /// The request that carries `input` to a capability's route the way the binding says:
81    /// `GET` puts every top-level property that is not `null` on the query string (a
82    /// scalar as its text, anything else as JSON), `POST` sends the input as the body.
83    /// The inverse of the router's binding, so a benchmark case reaches the route as a
84    /// client would send it.
85    ///
86    /// ```
87    /// use majordomus_cli::capability::HttpMethod;
88    /// use majordomus_cli::http::Request;
89    /// use serde_json::json;
90    /// let r = Request::bind(HttpMethod::Get, "/api/v1/search", &json!({ "query": "a b", "limit": 2, "kind": null }));
91    /// assert_eq!(r.path, "/api/v1/search");
92    /// assert_eq!(r.query, vec![("query".to_string(), "a b".to_string()), ("limit".to_string(), "2".to_string())]);
93    /// let p = Request::bind(HttpMethod::Post, "/api/v1/peers/announce", &json!({ "intent": "x" }));
94    /// assert_eq!(p.body, br#"{"intent":"x"}"#);
95    /// ```
96    pub fn bind(method: HttpMethod, path: &str, input: &Value) -> Self {
97        match method {
98            HttpMethod::Get => Request {
99                method: "GET".into(),
100                path: path.into(),
101                query: input
102                    .as_object()
103                    .map(|m| {
104                        m.iter()
105                            .filter(|(_, v)| !v.is_null())
106                            .map(|(k, v)| {
107                                let text = match v {
108                                    Value::String(s) => s.clone(),
109                                    other => other.to_string(),
110                                };
111                                (k.clone(), text)
112                            })
113                            .collect()
114                    })
115                    .unwrap_or_default(),
116                headers: Vec::new(),
117                body: Vec::new(),
118                remote: None,
119            },
120            HttpMethod::Post => Request {
121                method: "POST".into(),
122                path: path.into(),
123                query: Vec::new(),
124                headers: Vec::new(),
125                body: input.to_string().into_bytes(),
126                remote: None,
127            },
128        }
129    }
130
131    /// The request target this request would be sent as: the path with the query string,
132    /// percent-encoded.
133    ///
134    /// ```
135    /// use majordomus_cli::capability::HttpMethod;
136    /// use majordomus_cli::http::Request;
137    /// let r = Request::bind(HttpMethod::Get, "/api/v1/search", &serde_json::json!({ "query": "a b&c" }));
138    /// assert_eq!(r.target(), "/api/v1/search?query=a%20b%26c");
139    /// ```
140    pub fn target(&self) -> String {
141        if self.query.is_empty() {
142            return self.path.clone();
143        }
144        let pairs: Vec<String> = self
145            .query
146            .iter()
147            .map(|(k, v)| format!("{}={}", percent_encode(k), percent_encode(v)))
148            .collect();
149        format!("{}?{}", self.path, pairs.join("&"))
150    }
151
152    /// The same request with its headers.
153    pub fn with_headers(mut self, headers: Vec<(String, String)>) -> Self {
154        self.headers = headers;
155        self
156    }
157
158    /// The same request, as having come from `address`.
159    ///
160    /// ```
161    /// use majordomus_cli::http::Request;
162    /// let r = Request::parse_target("GET", "/", vec![]).with_remote(Some("10.0.0.2".parse().unwrap()));
163    /// assert_eq!(r.remote, Some("10.0.0.2".parse().unwrap()));
164    /// ```
165    pub fn with_remote(mut self, address: Option<std::net::IpAddr>) -> Self {
166        self.remote = address;
167        self
168    }
169
170    /// Did this request come from another host?
171    ///
172    /// Loopback is this machine, in either family and in the IPv4-mapped IPv6 form a
173    /// dual-stack socket reports. A request with no address was built in this process.
174    ///
175    /// ```
176    /// use majordomus_cli::http::Request;
177    /// let r = |a: &str| Request::parse_target("POST", "/", vec![]).with_remote(Some(a.parse().unwrap()));
178    /// assert!(!r("127.0.0.1").is_remote());
179    /// assert!(!r("::1").is_remote());
180    /// assert!(!r("::ffff:127.0.0.1").is_remote());
181    /// assert!(r("192.168.100.30").is_remote());
182    /// assert!(r("100.92.246.32").is_remote());
183    /// assert!(r("::ffff:10.0.0.2").is_remote());
184    /// assert!(!Request::parse_target("POST", "/", vec![]).is_remote());
185    /// ```
186    pub fn is_remote(&self) -> bool {
187        self.remote
188            .is_some_and(|ip| !ip.to_canonical().is_loopback())
189    }
190
191    /// A header value, by case-insensitive name.
192    ///
193    /// ```
194    /// use majordomus_cli::http::Request;
195    /// let r = Request::parse_target("POST", "/mcp", vec![])
196    ///     .with_headers(vec![("Mcp-Session-Id".into(), "abc".into())]);
197    /// assert_eq!(r.header("mcp-session-id"), Some("abc"));
198    /// assert_eq!(r.header("x-none"), None);
199    /// ```
200    pub fn header(&self, name: &str) -> Option<&str> {
201        self.headers
202            .iter()
203            .find(|(k, _)| k.eq_ignore_ascii_case(name))
204            .map(|(_, v)| v.as_str())
205    }
206}

208/// What a response carries. Text for everything this projection computes; bytes for a file
209/// it serves from disk, which is not always UTF-8 and must not be repaired into it.
210#[derive(Debug, Clone, PartialEq, Eq)]
211pub enum Body {
212    /// A string this process produced.
213    Text(String),
214    /// Bytes read from somewhere else, passed through unchanged.
215    Bytes(Vec<u8>),
216}
217
218impl Body {
219    /// The bytes on the wire.
220    pub fn as_bytes(&self) -> &[u8] {
221        match self {
222            Body::Text(t) => t.as_bytes(),
223            Body::Bytes(b) => b,
224        }
225    }
226
227    /// How many bytes it is.
228    pub fn len(&self) -> usize {
229        self.as_bytes().len()
230    }
231
232    /// Is it empty?
233    pub fn is_empty(&self) -> bool {
234        self.len() == 0
235    }
236
237    /// The body as text: borrowed when it is text, and lossy when it is not, which only a
238    /// diagnostic or a test ever asks for.
239    ///
240    /// ```
241    /// use majordomus_cli::http::router::Body;
242    /// assert_eq!(Body::Text("a".into()).text(), "a");
243    /// assert_eq!(Body::Bytes(vec![0xff]).text(), "\u{fffd}");
244    /// ```
245    pub fn text(&self) -> std::borrow::Cow<'_, str> {
246        match self {
247            Body::Text(t) => std::borrow::Cow::Borrowed(t),
248            Body::Bytes(b) => String::from_utf8_lossy(b),
249        }
250    }
251}
252
253impl From<String> for Body {
254    fn from(value: String) -> Self {
255        Body::Text(value)
256    }
257}
258
259impl From<&str> for Body {
260    fn from(value: &str) -> Self {
261        Body::Text(value.to_string())
262    }
263}
264
265impl From<Vec<u8>> for Body {
266    fn from(value: Vec<u8>) -> Self {
267        Body::Bytes(value)
268    }
269}
270
271impl PartialEq<str> for Body {
272    fn eq(&self, other: &str) -> bool {
273        self.as_bytes() == other.as_bytes()
274    }
275}
276
277impl PartialEq<&str> for Body {
278    fn eq(&self, other: &&str) -> bool {
279        self.as_bytes() == other.as_bytes()
280    }
281}
282
283impl std::fmt::Display for Body {
284    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
285        f.write_str(&self.text())
286    }
287}

344/// Routes requests to the surfaces this process serves. Cheap to clone: every worker
345/// thread holds one, and the resolution behind it is shared.
346pub struct Router {
347    /// The repository as it is now. Every request takes the current context from here
348    /// before it is answered, which is what keeps a server that has run for hours from
349    /// serving the repository it started in.
350    live: Arc<Live>,
351    version: &'static str,
352    /// The OpenAPI document, rendered once per generation of the repository: it is a
353    /// projection of the registry, and the registry is a projection of the index.
354    openapi: Arc<Memo<Result<String, String>>>,
355    /// MCP over HTTP at the mount its surface declares, when this router serves a shared

360    ui: Option<Arc<Ui>>,
361    /// The resolved surfaces and their handlers, and the context narrowed to them. Built
362    /// on first use, because the builder learns what this process offers after `new`, and

365    /// Which git repository the served checkout belongs to, asked once: the index route
366    /// answers it on every probe, and a probe must not cost a subprocess.

393/// What one router serves: the bound surfaces, and the context every capability call is
394/// given so that `web.surfaces` answers with what this process actually serves.
395struct Resolution {
396    served: Served,
397    ctx: Arc<Context>,
398}

400/// The surfaces a router serves, held for as long as its reader needs them.
401///
402/// A resolution belongs to one generation of the repository and is replaced when the
403/// repository moves under the process, so it cannot be lent out as a bare reference tied
404/// to the router. This holds the generation alive while the caller reads it, and derefs to
405/// the surfaces themselves so that a caller writes `router.served()?.summary(url)` exactly
406/// as it did when the index was immutable.
407///
408/// ```
409/// # use majordomus_cli::http::Router;
410/// # use majordomus_cli::http::router::ServedRef;
411/// # fn example(router: &Router) {
412/// // the surfaces, read through the generation that resolved them
413/// let served: ServedRef = router.served().expect("a topology that resolves");
414/// assert!(served.topology().surfaces.iter().any(|s| s.id == "home"));
415/// # }
416/// ```
417pub struct ServedRef {
418    resolution: Arc<Result<Resolution, String>>,
419}
420
421impl std::ops::Deref for ServedRef {
422    type Target = Served;
423
424    fn deref(&self) -> &Served {
425        match &*self.resolution {
426            Ok(resolution) => &resolution.served,
427            // `served()` is the only constructor and it builds one only from `Ok`
428            Err(reason) => unreachable!("a resolved surface set that did not resolve: {reason}"),
429        }
430    }
431}
432
433impl std::fmt::Debug for ServedRef {
434    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
435        f.debug_struct("ServedRef").finish()
436    }
437}

522    fn resolution_of(&self, view: crate::live::View) -> Arc<Result<Resolution, String>> {
523        self.served.get_or_init(view.generation, || {
524            let root = std::path::Path::new(&view.ctx.index.repository.root);
525            Served::resolve(&view.ctx.web, root, self.runtime())
526                .map(|served| Resolution {
527                    ctx: Arc::new(view.ctx.with_web(served.shared())),
528                    served,
529                })
530                // the reason alone: `served()` puts it back into the same error, and a
531                // message that names its own kind twice reads as a bug in the tool
532                .map_err(|e| match e {
533                    crate::error::Error::InvalidSurface { reason, .. } => reason,
534                    other => other.to_string(),
535                })
536        })
537    }

539    /// The surfaces this router serves, for a caller that wants to describe them: the
540    /// server's startup line, and the tests.
541    ///
542    /// The answer is the resolution, not a fresh discovery — a router that could not
543    /// resolve its topology never answered a request either, and says the same thing here
544    /// that it says to a client.
545    pub fn served(&self) -> crate::error::Result<ServedRef> {
546        let resolution = self.resolution();
547        match &*resolution {
548            Ok(_) => Ok(ServedRef { resolution }),
549            Err(reason) => Err(crate::error::Error::InvalidSurface {
550                surface: "topology".into(),
551                reason: reason.clone(),
552            }),
553        }
554    }

556    /// Does this request open the live channel, and may it?
557    ///
558    /// The server asks before it reads a body or writes a response, because an upgrade is
559    /// the one thing a router cannot answer: it hands the socket over. `None` means the
560    /// request is not for the live channel at all and is routed normally.
561    pub fn websocket(&self, req: &Request) -> Option<Result<events::Accepted, Response>> {
562        if req.path != events::PATH {
563            return None;
564        }
565        if !super::ws::is_upgrade(req) {
566            return None;
567        }
568        // the upgrade is answered before `handle`, so it is refused here for the reason
569        // `handle` refuses every other request a DNS-rebinding page sends: otherwise a page
570        // whose domain was made to resolve to this machine follows every execution
571        if let Some(refusal) = Self::rebound(req) {
572            return Some(Err(refusal));
573        }
574        // the live channel carries every execution of this process, and those are not the
575        // repository: a caller on another host never follows them (ADR 0220 decision 4, I3842)
576        if req.is_remote() {
577            return Some(Err(Response::error(
578                403,
579                "forbidden",
580                "the live channel answers only a caller on this machine",
581            )));
582        }
583        // a process whose topology does not resolve serves nothing, the live channel
584        // included: the same answer every other path gets
585        let resolution = self.resolution();
586        let ctx = match &*resolution {
587            Ok(resolution) => &resolution.ctx,
588            Err(reason) => return Some(Err(Response::error(500, "internal", reason))),
589        };
590        // the blackboard on the same channel: a read of `blackboard.snapshot`, followed only
591        // where that capability is served (a remote caller was refused above, I3842)
592        if let Some(follow) = events::accept_follow(ctx, req) {
593            let admitted = ctx.registry.get("blackboard.snapshot").is_some();
594            if !admitted {
595                let e = CapabilityError::remote("blackboard.snapshot");
596                return Some(Err(error_response(403, e.code(), &e.to_string())));
597            }
598            return Some(follow);
599        }
600        Some(events::accept(
601            ctx.executions.store(),
602            req,
603            &snapshot_route(ctx),
604        ))
605    }

607    fn openapi(&self) -> Response {
608        let view = self.live.view();
609        let rendered = self.openapi.get_or_init(view.generation, || {
610            let cases = CaseContext {
611                index: &view.ctx.index,
612            };
613            openapi::document(&view.ctx.registry, self.version, Some(&cases))
614                .map(|d| openapi::render(&d))
615        });
616        match &*rendered {
617            Ok(text) => Response::new(200, "application/json", text.clone()),
618            Err(e) => error_response(500, "internal", e),
619        }
620    }

622    /// Answer one request: the surface that owns its path, or nothing.
623    ///
624    /// A request that carries an `Origin` header came from a page in a browser. A browser
625    /// cannot read a cross-origin response without the headers this server never sends,
626    /// so a read is already contained; a request that changes something is not, and one
627    /// from another origin is refused before it reaches a handler. A client that is not a
628    /// browser sends no `Origin` and is unaffected.
629    pub fn handle(&self, req: &Request) -> Response {
630        if let Some(refusal) = Self::rebound(req) {
631            return refusal;
632        }
633        if let Some(refusal) = Self::local_only_page(req) {
634            return refusal;
635        }
636        if req.method != "GET" && req.method != "HEAD" {
637            if let Some(refusal) = self.foreign_origin(req) {
638                return refusal;
639            }
640        }
641        // a page of the UI is drawn from the generation there is and never waits for the next
642        // one; every other request waits for the generation its caller saw
643        let page = req.method == "GET"
644            && (req.path == crate::ui::PREFIX
645                || req.path.starts_with(&format!("{}/", crate::ui::PREFIX)));
646        let held = if page {
647            self.resolution_without_waiting()
648        } else {
649            self.resolution()
650        };
651        let resolution = match &*held {
652            Ok(resolution) => resolution,
653            Err(reason) => return error_response(500, "internal", reason),
654        };
655        let Some((surface, bound)) = resolution.served.owner(&req.path) else {
656            return error_response(
657                404,
658                "not_found",
659                &format!(
660                    "no surface owns {}; what this process serves is listed at / and at {}web/surfaces",
661                    req.path,
662                    crate::capability::model::HttpExposure::PREFIX
663                ),
664            );
665        };
666        // a surface owns its mount and everything under it, which is what a directory and a
667        // prefix want and what a single route does not: `/swagger/anything` is not the
668        // Swagger UI, and answering it as though it were would invent a route nothing
669        // declared. The surfaces that answer one path say so here, once.
670        let exact = req.path == surface.mount.as_str() || req.path == surface.mount.prefix();
671        // the home page is not among them: its mount is the root, which owns every path
672        // nothing else claims, and it answers those with a 404 that names what is served
673        let single_path = matches!(
674            bound,
675            Bound::Route(Native::OpenApi | Native::Swagger | Native::Mcp | Native::Events)
676        );
677        if single_path && !exact {
678            return error_response(
679                404,
680                "not_found",
681                &format!(
682                    "{} is the whole of the '{}' surface; nothing is served under it. What this process serves is listed at / and at {}web/surfaces",
683                    surface.mount,
684                    surface.id,
685                    crate::capability::model::HttpExposure::PREFIX
686                ),
687            );
688        }
689        match bound {
690            Bound::Directory(files) => {
691                if req.method != "GET" {
692                    return error_response(
693                        405,
694                        "method_not_allowed",
695                        &format!(
696                            "{} is a generated directory; it is read with GET",
697                            surface.mount
698                        ),
699                    );
700                }
701                files.respond(&req.path)
702            }
703            Bound::Route(Native::Home) => self.home(req, resolution),
704            Bound::Route(Native::OpenApi) => self.openapi(),
705            Bound::Route(Native::Swagger) => {
706                let checkout = self.checkout(&resolution.ctx);
707                swagger_response(&checkout.as_home().line(&checkout.repository))
708            }
709            Bound::Route(Native::Mcp) => match &self.mcp {
710                Some(endpoint) => endpoint.handle(req),
711                None => error_response(
712                    500,
713                    "internal",
714                    "the MCP surface is served with no endpoint behind it",
715                ),
716            },
717            // the upgrade itself never reaches here: the server answers it before routing,
718            // because handing the socket over is something only the server can do. What
719            // reaches here is a request that did not ask to be upgraded, and the reply says
720            // what this path is and where the same events are readable over HTTP.
721            Bound::Route(Native::Events) => match events::accept(
722                resolution.ctx.executions.store(),
723                req,
724                &snapshot_route(&resolution.ctx),
725            ) {
726                Ok(_) => Response::error(
727                    500,
728                    "internal",
729                    "a WebSocket upgrade reached the router; the server answers it before routing",
730                ),
731                Err(response) => response,
732            },
733            Bound::Route(Native::Cockpit) => match &self.cockpit {
734                Some(cockpit) => cockpit.handle(req),
735                None => error_response(
736                    500,
737                    "internal",
738                    "the Cockpit surface is served with no Cockpit behind it",
739                ),
740            },
741            Bound::Route(Native::Ui) => match &self.ui {
742                Some(ui) => {
743                    Ui::warm_once(ui);
744                    ui.handle(req)
745                }
746                None => error_response(
747                    500,
748                    "internal",
749                    "the UI surface is served with no UI behind it",
750                ),
751            },
752            Bound::Route(Native::Api) => {
753                let prefix = crate::capability::model::HttpExposure::PREFIX;
754                if req.path == prefix || req.path == prefix.trim_end_matches('/') {
755                    return self.routes(&resolution.ctx);
756                }
757                self.capability(req, &resolution.ctx)
758            }
759        }
760    }

762    /// `/`: the home page for a browser, the same topology as JSON for everything else.
763    ///
764    /// The root surface owns every path nothing else claims, which is how a request for a
765    /// path no surface declares reaches the most useful 404 this server can give: one from
766    /// a process that knows what it does serve.
767    fn home(&self, req: &Request, resolution: &Resolution) -> Response {
768        if req.path != "/" {
769            return error_response(
770                404,
771                "not_found",
772                &format!(
773                    "no surface owns {}; what this process serves is listed at / and at {}web/surfaces",
774                    req.path,
775                    crate::capability::model::HttpExposure::PREFIX
776                ),
777            );
778        }
779        if req.method != "GET" {
780            return error_response(405, "method_not_allowed", "the home page is read with GET");
781        }
782        let topology = resolution.served.topology();
783        let checkout = self.checkout(&resolution.ctx);
784        if prefers_html(req) {
785            let identity = home::Identity {
786                version: self.version,
787                summary: crate::about::SUMMARY,
788                repository: repository_name(&resolution.ctx.index.repository.root),
789                revision: match &resolution.ctx.index.repository.git {
790                    crate::git::GitState::Available(info) => info.head.as_deref(),
791                    crate::git::GitState::Unavailable { .. } => None,
792                },
793                capabilities: resolution.ctx.registry.summary().total,
794                checkout: checkout.as_home(),
795            };
796            let ready = |id: &str| {
797                if resolution.served.ready(id) {
798                    home::Availability::Ready
799                } else {
800                    home::Availability::NotBuilt
801                }
802            };
803            return Response::new(
804                200,
805                "text/html; charset=utf-8",
806                home::page(topology, &identity, &ready),
807            );
808        }
809        let surfaces: Vec<Value> = topology
810            .surfaces
811            .iter()
812            .map(|s| {
813                json!({
814                    "id": s.id,
815                    "title": s.title,
816                    "path": s.mount.as_str(),
817                    "category": s.category.to_string(),
818                    "visibility": s.visibility.to_string(),
819                    "kind": s.kind.to_string(),
820                    "ready": resolution.served.ready(&s.id),
821                })
822            })
823            .collect();
824        let git = checkout.git.as_ref();
825        // whether this process is still the checkout's server. A server whose lease was
826        // taken over keeps serving the peers it already has, so it goes on answering here,
827        // and every other field is as true of it as of the current server. This is the one
828        // that tells them apart; `lease::probe` reads it under the same name, which is why
829        // the name is the constant and not a literal.
830        let mut index = json!({
831                "name": "majordomus",
832                "version": self.version,
833                // the commit this executable was built from, the value `health.live` and
834                // `health.ready` answer under the same names: a checkout's server that
835                // answers from another commit than the checkout's HEAD is serving another
836                // tree, and a reader can only say so if the server names its own
837                "commit": crate::COMMIT,
838                "dirty": crate::DIRTY,
839                // and whether that code still exists: null while the file this process was
840                // started from is the file on disk, the reason once it has been replaced —
841                // the same answer `health.ready` gives, so a client reading only this
842                // document can tell a current server from one serving yesterday's code
843                "stale": crate::lease::serving_replaced_code(),
844                "description": crate::about::SUMMARY,
845                "reference": crate::about::REFERENCE_URL,
846                // the repository's name and not its path: this answer is served to whoever
847                // can reach the socket, and where the checkout sits on the host is of no
848                // use to them and of some use to somebody else. The HTML page has always
849                // withheld it; the two projections of one surface now agree.
850                "repository": repository_name(&resolution.ctx.index.repository.root),
851                // which repository, without saying where it is: a second process that
852                // already knows the root computes the same value and knows this server is
853                // its own (crate::repository::identity)
854                "repository_id": checkout.id,
855                // and which git repository that checkout belongs to: one value for every
856                // worktree of it, so that a reader can tell two servers of one repository
857                // from the servers of two (crate::repository::git_identity)
858                "git_repository_id": git.map(|g| g.id.clone()),
859                "linked_worktree": git.is_some_and(|g| g.linked),
860                // the branch and commit the served answers were read from, as the index
861                // read them: two servers of one repository can serve two trees (I1031)
862                "branch": checkout.branch,
863                "head": checkout.head,
864                "surfaces": surfaces,
865        });
866        index[crate::lease::LEASEHOLDER_KEY] = json!(!crate::lease::was_lost());
867        json_response(200, &index)
868    }

893    /// The refusal for a state-changing request from another origin, when there is one.
894    /// The allowed origins are this server's own: whatever host the request was addressed
895    /// to. Nothing else is configured, because nothing else should be able to change this
896    /// process from a browser.
897    fn foreign_origin(&self, req: &Request) -> Option<Response> {
898        let origin = req.header("origin")?;
899        // `null` is what a sandboxed frame or a `file://` page sends; neither is this server
900        let host = req.header("host").unwrap_or_default();
901        let allowed = [format!("http://{host}"), format!("https://{host}")];
902        // The origin matching the Host header is only this server's own origin when the Host
903        // cannot be chosen by a page: a DNS-rebinding page names its own domain in both and
904        // matches. So the request must also be addressed by an IP literal or `localhost`,
905        // which no rebinding domain is.
906        if allowed.iter().any(|a| a == origin) && Self::host_is_literal(host) {
907            return None;
908        }
909        tracing::warn!(
910            origin = %crate::redaction::redact_secrets(origin).text,
911            method = %req.method,
912            path = %crate::redaction::redact_secrets(&req.path).text,
913            "a state-changing request from another origin was refused"
914        );
915        Some(error_response(
916            403,
917            "forbidden",
918            &format!(
919                "a {} from origin '{origin}' is refused: this server accepts a state-changing request from a browser only from its own origin",
920                req.method
921            ),
922        ))
923    }

974    /// Whether a `Host` header names this machine by an address rather than by a domain a
975    /// third party could point at it: an IPv4 or bracketed IPv6 literal, `localhost` or a
976    /// `*.localhost` name, each with an optional port.
977    /// The refusal for a request a DNS-rebinding page sent, when this is one.
978    ///
979    /// A page on another domain cannot read this server's answers, until its domain is made
980    /// to resolve to this machine: then the browser on this machine sends the page's requests
981    /// here, from loopback, naming the page's domain as the host. Nothing else sends a
982    /// loopback request under a name that is neither an address nor `localhost`, so that
983    /// request is refused before any surface answers it, reads included. A request from
984    /// another host is judged by ADR 0126, and one that names no host is a program's.
985    fn rebound(req: &Request) -> Option<Response> {
986        let from_loopback = req.remote.is_some_and(|ip| ip.to_canonical().is_loopback());
987        let host = req.header("host")?;
988        if !from_loopback || Self::host_is_literal(host) {
989            return None;
990        }
991        tracing::warn!(host = host, path = %req.path, "a loopback request under a foreign host name was refused");
992        Some(error_response(
993            403,
994            "forbidden",
995            &format!(
996                "a request from this machine addressed to '{host}' is refused: this server answers loopback requests only under an address or `localhost`, so a page whose domain was made to resolve here reads nothing"
997            ),
998        ))
999    }

1001    fn host_is_literal(host: &str) -> bool {
1002        let name = if let Some(rest) = host.strip_prefix('[') {
1003            match rest.split_once(']') {
1004                Some((v6, tail)) if tail.is_empty() || tail.starts_with(':') => {
1005                    return v6.parse::<std::net::Ipv6Addr>().is_ok();
1006                }
1007                _ => return false,
1008            }
1009        } else {
1010            host.rsplit_once(':').map_or(host, |(name, port)| {
1011                if port.parse::<u16>().is_ok() {
1012                    name
1013                } else {
1014                    host
1015                }
1016            })
1017        };
1018        name.parse::<std::net::Ipv4Addr>().is_ok()
1019            || name.eq_ignore_ascii_case("localhost")
1020            || name.to_ascii_lowercase().ends_with(".localhost")
1021    }

1023    /// The capability mount itself: every route under it, from the registry that declares
1024    /// them.
1025    ///
1026    /// It exists so that the surface the home page offers is a page a person can open
1027    /// rather than a prefix that answers 404, and it is the registry's own list rather
1028    /// than a second one — the schemas, the parameters and the examples stay in the
1029    /// OpenAPI document, which is generated from the same descriptors.
1030    fn routes(&self, ctx: &Arc<Context>) -> Response {
1031        // Ordered as routes, then serialised — the same order the Cockpit's table and the
1032        // OpenAPI document read in. The previous sort compared `Value::to_string()`, which
1033        // is the JSON encoding of the path, quotes included, and had no method tiebreak.
1034        let mut declared: Vec<_> = ctx
1035            .registry
1036            .iter()
1037            .filter_map(|c| c.exposure.http.as_ref().map(|http| (c, http)))
1038            .collect();
1039        declared.sort_by(|(a, ha), (b, hb)| {
1040            crate::order::natural_cmp(&ha.path, &hb.path)
1041                .then_with(|| crate::order::natural_cmp(ha.method.as_str(), hb.method.as_str()))
1042                .then_with(|| crate::order::natural_cmp(a.id.as_str(), b.id.as_str()))
1043        });
1044        let routes: Vec<Value> = declared
1045            .into_iter()
1046            .map(|(c, http)| {
1047                json!({
1048                    "method": http.method.as_str(),
1049                    "path": http.path,
1050                    "capability": c.id.as_str(),
1051                    "title": c.title,
1052                })
1053            })
1054            .collect();
1055        json_response(
1056            200,
1057            &json!({
1058                "description": "Every capability with an HTTP exposure, as the registry declares it.",
1059                "openapi": swagger::SPEC_PATH,
1060                "swagger": swagger::SWAGGER_PATH,
1061                "routes": routes,
1062            }),
1063        )
1064    }

1066    fn capability(&self, req: &Request, ctx: &Arc<Context>) -> Response {
1067        let Some(method) = HttpMethod::parse(&req.method) else {
1068            return error_response(
1069                405,
1070                "method_not_allowed",
1071                &format!("method {} is not served", req.method),
1072            );
1073        };
1074        let Some(c) = ctx.registry.by_http(method, &req.path) else {
1075            let other_method = [HttpMethod::Get, HttpMethod::Post]
1076                .into_iter()
1077                .filter(|m| *m != method)
1078                .any(|m| ctx.registry.by_http(m, &req.path).is_some());
1079            return if other_method {
1080                error_response(
1081                    405,
1082                    "method_not_allowed",
1083                    &format!(
1084                        "{} {} is not a route; another method is",
1085                        req.method, req.path
1086                    ),
1087                )
1088            } else {
1089                error_response(
1090                    404,
1091                    "not_found",
1092                    &format!(
1093                        "no route {} {}; the routes are listed at /openapi.json",
1094                        req.method, req.path
1095                    ),
1096                )
1097            };
1098        };
1099        if req.is_remote() && !c.execution.admits_remote() {
1100            let e = CapabilityError::remote(c.id.as_str());
1101            return error_response(403, e.code(), &e.to_string());
1102        }
1103        let input = match method {
1104            HttpMethod::Get => {
1105                let (props, _) = c.input.properties();
1106                let mut obj = serde_json::Map::new();
1107                for (k, raw) in &req.query {
1108                    let property = props
1109                        .iter()
1110                        .find(|(n, _)| n == k)
1111                        .map(|(_, s)| s.clone())
1112                        .unwrap_or(Value::Null);
1113                    match crate::capability::schema::coerce(&property, raw) {
1114                        Ok(v) => {
1115                            obj.insert(k.clone(), v);
1116                        }
1117                        Err(e) => {
1118                            return error_response(
1119                                400,
1120                                "invalid_input",
1121                                &format!("parameter '{k}': {e}"),
1122                            )
1123                        }
1124                    }
1125                }
1126                Value::Object(obj)
1127            }
1128            HttpMethod::Post => {
1129                if req.body.iter().all(u8::is_ascii_whitespace) {
1130                    json!({})
1131                } else {
1132                    match serde_json::from_slice::<Value>(&req.body) {
1133                        Ok(v) if v.is_object() => v,
1134                        Ok(_) => {
1135                            return error_response(
1136                                400,
1137                                "invalid_input",
1138                                "the body must be a JSON object",
1139                            )
1140                        }
1141                        Err(e) => {
1142                            return error_response(
1143                                400,
1144                                "invalid_input",
1145                                &format!("the body is not JSON: {e}"),
1146                            )
1147                        }
1148                    }
1149                }
1150            }
1151        };
1152        // one line per request is debug: a client that does not drain stderr must not be
1153        // able to wedge the workers on a full pipe at the default level
1154        tracing::debug!(capability_id = %c.id, route = %format!("{} {}", req.method, req.path), "http");
1155        match ctx.execute(c.id.as_str(), input) {
1156            Ok(v) => json_response(200, &v),
1157            // the word is the error's own, the same one an MCP tool result carries; only
1158            // the status is this transport's
1159            Err(e) => failed(&e),
1160        }
1161    }
1162}

1164/// A capability's error as an HTTP response: the status is this transport's, the code and
1165/// the message are the error's own.
1166fn failed(e: &CapabilityError) -> Response {
1167    let status = match e {
1168        CapabilityError::InvalidInput(_) => 400,
1169        CapabilityError::NotFound(_) => 404,
1170        CapabilityError::Refused(_) => 422,
1171        CapabilityError::Internal(_) => 500,
1172        CapabilityError::Forbidden(_) => 403,
1173    };
1174    let (CapabilityError::InvalidInput(m)
1175    | CapabilityError::NotFound(m)
1176    | CapabilityError::Refused(m)
1177    | CapabilityError::Internal(m)
1178    | CapabilityError::Forbidden(m)) = e;
1179    error_response(status, e.code(), m)
1180}

1182/// Does this request come from something that would rather have a page than a document?
1183/// Only an explicit `text/html` in `Accept` counts: a client that sends none, or `*/*`,
1184/// gets the JSON index, which is what every existing caller does.
1185///
1186/// ```
1187/// use majordomus_cli::http::{router::prefers_html, Request};
1188/// let plain = Request::parse_target("GET", "/", vec![]);
1189/// assert!(!prefers_html(&plain));
1190/// let browser = plain.clone().with_headers(vec![
1191///     ("Accept".into(), "text/html,application/xhtml+xml,*/*;q=0.8".into()),
1192/// ]);
1193/// assert!(prefers_html(&browser));
1194/// ```
1195pub fn prefers_html(req: &Request) -> bool {
1196    req.header("accept").is_some_and(|accept| {
1197        accept.split(',').any(|part| {

1247/// Percent-encode everything but the unreserved characters of RFC 3986.
1248pub fn percent_encode(s: &str) -> String {
1249    let mut out = String::with_capacity(s.len());
1250    for b in s.bytes() {
1251        match b {
1252            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~' => {
1253                out.push(b as char)
1254            }
1255            _ => out.push_str(&format!("%{b:02X}")),
1256        }
1257    }
1258    out
1259}

1261/// `%XX` and `+` decoding; a malformed escape is kept as it is.
1262pub fn percent_decode(s: &str) -> String {
1263    let bytes = s.as_bytes();
1264    let mut out = Vec::with_capacity(bytes.len());
1265    let mut i = 0;
1266    while i < bytes.len() {
1267        match bytes[i] {
1268            b'+' => out.push(b' '),
1269            b'%' if i + 2 < bytes.len() => {
1270                match u8::from_str_radix(
1271                    std::str::from_utf8(&bytes[i + 1..i + 3]).unwrap_or("zz"),
1272                    16,
1273                ) {
1274                    Ok(b) => {
1275                        out.push(b);
1276                        i += 2;
1277                    }
1278                    Err(_) => out.push(b'%'),
1279                }
1280            }
1281            b => out.push(b),
1282        }
1283        i += 1;
1284    }
1285    String::from_utf8_lossy(&out).into_owned()
1286}
