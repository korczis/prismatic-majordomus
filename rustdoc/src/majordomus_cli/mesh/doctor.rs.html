1//! The mesh self-check: every prerequisite proved on this machine alone, no second node
2//! required. Deterministic checks in a fixed order, each with its own verdict, so "why
3//! is the mesh not working" has an answer that names the broken link instead of a
4//! shrug — and, when a check fails or limits cooperation, what that means and what to do.
5//! Read-only toward the repository; the sockets it probes are ephemeral and closed before
6//! it answers.
7//!
8//! What this cannot see is the running server: whether its cooperation thread beats and
9//! whether its peers answer. That is `mesh verify`, which asks the server.
10//!
11//! ```
12//! use majordomus_cli::mesh::doctor::doctor;
13//!
14//! // No declaration is the default posture, and the self-check says so and runs on.
15//! let report = doctor(None);
16//! let declaration = report.checks.iter().find(|c| c.check == "declaration").unwrap();
17//! assert!(declaration.ok);
18//! assert!(declaration.detail.contains("default posture"));
19//! ```
20

32/// One check's verdict.
33#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
34pub struct DoctorCheck {
35    /// What was checked.
36    pub check: String,
37    /// Whether it holds.
38    pub ok: bool,
39    /// The evidence: a path, an address, an error.
40    pub detail: String,
41    /// What a failure or a limitation means for the mesh.
42    #[serde(default, skip_serializing_if = "Option::is_none")]
43    pub impact: Option<String>,
44    /// What to do about it.
45    #[serde(default, skip_serializing_if = "Option::is_none")]
46    pub remediation: Option<String>,
47}
48
49impl DoctorCheck {
50    fn pass(check: &str, detail: impl Into<String>) -> Self {
51        DoctorCheck {
52            check: check.into(),
53            ok: true,
54            detail: detail.into(),
55            impact: None,
56            remediation: None,
57        }
58    }
59
60    fn fail(check: &str, detail: impl Into<String>, impact: &str, remediation: &str) -> Self {
61        DoctorCheck {
62            check: check.into(),
63            ok: false,
64            detail: detail.into(),
65            impact: Some(impact.into()),
66            remediation: Some(remediation.into()),
67        }
68    }
69}

71/// The whole self-check.
72#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
73pub struct MeshDoctorReport {
74    /// Whether every check holds.
75    pub ok: bool,
76    /// The checks, in the order they ran.
77    pub checks: Vec<DoctorCheck>,
78}

80/// Run the self-check against the declaration as parsed (or its absence, or its error).
81/// Deterministic order, no second node required, no repository writes.
82pub fn doctor(declaration: Option<Result<MeshConfig, MeshError>>) -> MeshDoctorReport {
83    doctor_at(declaration, None)
84}

86/// [`doctor`] for the repository at `root`: adds whether the repository has a mesh
87/// identity cooperation can match on.
88///
89/// Discovery works without one — two nodes can hear each other while belonging to
90/// different repositories — but a link does not, so a report that omits this check can say
91/// every prerequisite holds while cooperation is unreachable. `root` is optional because
92/// the caller may have no repository at hand, and then the check is not run rather than
93/// failed.
94///
95/// ```
96/// use majordomus_cli::mesh::doctor::doctor_at;
97///
98/// // a directory holding no history has no identity to match on, and the check says so
99/// let nowhere = tempfile::tempdir().unwrap();
100/// let report = doctor_at(None, Some(nowhere.path()));
101/// let repository = report.checks.iter().find(|c| c.check == "repository").unwrap();
102/// assert!(!repository.ok);
103/// assert!(repository.remediation.as_deref().unwrap().contains("cooperation.repository"));
104///
105/// // with no root to ask about, the question is not asked at all
106/// assert!(doctor_at(None, None).checks.iter().all(|c| c.check != "repository"));
107/// ```
108pub fn doctor_at(
109    declaration: Option<Result<MeshConfig, MeshError>>,
110    root: Option<&Path>,
111) -> MeshDoctorReport {
112    let mut checks = Vec::new();
113
114    // The declaration: present, parseable, and what it says.
115    match &declaration {
116        None => checks.push(DoctorCheck::pass(
117            "declaration",
118            "absent: the mesh is off, which is the default posture",
119        )),
120        Some(Err(e)) => checks.push(DoctorCheck::fail(
121            "declaration",
122            e.to_string(),
123            "the mesh stays off: no discovery socket opens and no link is made",
124            "fix the declaration under .ai/repo/mesh/ (majordomus check names the schema violation), then restart the server",
125        )),
126        Some(Ok(config)) => checks.push(DoctorCheck::pass(
127            "declaration",
128            format!(
129                "{}: enabled={}, multicast={}, broadcast={}, rendezvous endpoints={}, trust={} ({} allowed key(s)), cooperation={} (heartbeat {}s, expiry {}s, {} seed(s))",
130                config.id,
131                config.enabled,
132                config.multicast.enabled,
133                config.broadcast.mode != super::config::BroadcastMode::Disabled,
134                config.rendezvous.endpoints.len(),
135                config.trust.policy.as_str(),
136                config.trust.allow.len(),
137                config.cooperation.enabled,
138                config.cooperation.heartbeat_seconds,
139                config.cooperation.expiry_seconds,
140                config.cooperation.seeds.len()
141            ),
142        )),
143    }
144
145    // The identity: where it lives, and whether what is there loads.
146    let identity = match default_identity_path() {
147        None => {
148            checks.push(DoctorCheck::fail(
149                "identity",
150                "no HOME and no XDG_STATE_HOME: this process has nowhere to keep a node identity",
151                "the mesh cannot activate: a node without a key cannot be told apart from an impostor",
152                "run the server with HOME or XDG_STATE_HOME set",
153            ));
154            None
155        }
156        Some(path) if path.is_file() => match NodeIdentity::load_or_create(&path) {
157            Ok(identity) => {
158                checks.push(DoctorCheck::pass(
159                    "identity",
160                    format!("{} at {}", identity.public.node_id, path.display()),
161                ));
162                Some(identity)
163            }
164            Err(e) => {
165                checks.push(DoctorCheck::fail(
166                    "identity",
167                    e.to_string(),
168                    "the mesh cannot activate with a broken identity, and replacing it silently would make this machine a new node every peer distrusts",
169                    "restore the file from a backup, or remove it deliberately — a new key is a new node, and every allowlist naming the old key must be updated",
170                ));
171                None
172            }
173        },
174        Some(path) => {
175            checks.push(DoctorCheck::pass(
176                "identity",
177                format!("absent; created at first activation at {}", path.display()),
178            ));
179            None
180        }
181    };
182
183    // Trust: under an allowlist-shaped declaration, is this machine's own key listed? A
184    // fleet whose declaration forgot a machine links around it.
185    if let (Some(Ok(config)), Some(identity)) = (&declaration, &identity) {
186        if config.enabled && !config.trust.allow.is_empty() {
187            let listed = config.trust.allow.contains(&identity.public.public_key);
188            checks.push(if listed {
189                DoctorCheck::pass("trust", "this machine's key is on the allowlist")
190            } else {
191                DoctorCheck::fail(
192                    "trust",
193                    format!(
194                        "this machine's key {} is not among the {} allowed",
195                        identity.public.public_key,
196                        config.trust.allow.len()
197                    ),
198                    "the other machines of the fleet refuse this one's links as untrusted; this machine can still dial them if it trusts their keys",
199                    "add the key above to trust.allow in the mesh declaration and commit it",
200                )
201            });
202        }
203    }
204
205    // The repository's mesh identity: what links are matched on.
206    if let Some(root) = root {
207        let declared = declaration
208            .as_ref()
209            .and_then(|d| d.as_ref().ok())
210            .and_then(|c| c.cooperation.repository.clone());
211        checks.push(match super::repository::resolve(root, declared.as_deref()) {
212            Ok(identity) => DoctorCheck::pass(
213                "repository",
214                format!(
215                    "{} from {:?}: {}",
216                    identity.id, identity.basis, identity.detail
217                ),
218            ),
219            Err(e) => DoctorCheck::fail(
220                "repository",
221                e.to_string(),
222                "cooperation stays off: without a repository identity no peer can be matched to this repository",
223                "declare cooperation.repository in the mesh declaration, or fetch the full history (git fetch --unshallow)",
224            ),
225        });
226    }
227
228    // A UDP socket at all.
229    checks.push(match UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)) {
230        Ok(socket) => DoctorCheck::pass(
231            "udp",
232            format!(
233                "ephemeral bind ok ({})",
234                socket
235                    .local_addr()
236                    .map(|a| a.to_string())
237                    .unwrap_or_else(|_| "?".into())
238            ),
239        ),
240        Err(e) => DoctorCheck::fail(
241            "udp",
242            format!("cannot bind an ephemeral UDP socket: {e}"),
243            "multicast and broadcast discovery cannot run; seeds and rendezvous still can",
244            "allow UDP for this process (a sandbox or a firewall is refusing it), or declare cooperation.seeds",
245        ),
246    });
247
248    // Multicast capability, against the declared (or default) group.
249    let group_text = declaration
250        .as_ref()
251        .and_then(|d| d.as_ref().ok())
252        .map(|c| c.multicast.group.clone())
253        .unwrap_or_else(|| super::config::DEFAULT_GROUP.into());
254    checks.push(match multicast_probe(&group_text) {
255        Ok(()) => DoctorCheck::pass(
256            "multicast",
257            format!("joined and left {group_text} on an ephemeral socket"),
258        ),
259        Err(e) => DoctorCheck::fail(
260            "multicast",
261            e.to_string(),
262            "runtimes on this segment will not hear each other by multicast",
263            "check the interface's multicast route, or declare cooperation.seeds or rendezvous endpoints instead",
264        ),
265    });
266
267    // Broadcast capability.
268    checks.push(
269        match UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)).and_then(|s| s.set_broadcast(true)) {
270            Ok(()) => DoctorCheck::pass("broadcast", "the broadcast flag sets"),
271            Err(e) => DoctorCheck::fail(
272                "broadcast",
273                format!("cannot enable broadcast: {e}"),
274                "the broadcast fallback cannot run",
275                "leave broadcast disabled and rely on multicast, seeds or rendezvous",
276            ),
277        },
278    );
279
280    // The discovery protocol, end to end in memory: sign, encode, parse, verify.
281    checks.push(match protocol_probe() {
282        Ok(detail) => DoctorCheck::pass("protocol", detail),
283        Err(e) => DoctorCheck::fail(
284            "protocol",
285            e,
286            "no advertisement this executable writes would verify anywhere",
287            "this is a defect of the executable: reinstall it and report the detail",
288        ),
289    });
290
291    // The link protocol, end to end in memory: a hello signed, verified, refused when
292    // tampered — the handshake's own cryptography, with no peer.
293    checks.push(match link_probe() {
294        Ok(detail) => DoctorCheck::pass("link", detail),
295        Err(e) => DoctorCheck::fail(
296            "link",
297            e,
298            "no link this executable opens would be admitted anywhere",
299            "this is a defect of the executable: reinstall it and report the detail",
300        ),
301    });
302
303    MeshDoctorReport {
304        ok: checks.iter().all(|c| c.ok),
305        checks,
306    }
307}

309fn multicast_probe(group_text: &str) -> Result<(), MeshError> {
310    let group: Ipv4Addr = group_text
311        .parse()
312        .map_err(|_| MeshError::Provider(format!("'{group_text}' is not an IPv4 group")))?;
313    if !group.is_multicast() {
314        return Err(MeshError::Provider(format!(
315            "{group} is not a multicast address"
316        )));
317    }
318    let socket = UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0))
319        .map_err(|e| MeshError::Provider(format!("cannot bind: {e}")))?;
320    socket
321        .join_multicast_v4(&group, &Ipv4Addr::UNSPECIFIED)
322        .map_err(|e| MeshError::Provider(format!("cannot join {group}: {e}")))?;
323    let _ = socket.leave_multicast_v4(&group, &Ipv4Addr::UNSPECIFIED);
324    Ok(())
325}

327fn protocol_probe() -> Result<String, String> {
328    let identity = NodeIdentity::ephemeral().map_err(|e| e.to_string())?;
329    let envelope = protocol::advertise_as(
330        &identity,
331        "0000000000000001",
332        1,
333        &["127.0.0.1:1".into()],
334        &["http".into()],
335        &[],
336        "self-check",
337    );
338    let bytes = protocol::encode(&envelope).map_err(|e| e.to_string())?;
339    let parsed = protocol::parse(&bytes).map_err(|r| format!("a fresh envelope refused: {r}"))?;
340    if parsed.adv.node_id() != Some(identity.public.node_id.clone()) {
341        return Err("the parsed envelope proves a different node".into());
342    }
343    Ok(format!(
344        "sign → encode ({} bytes) → parse → verify, protocol {}",
345        bytes.len(),
346        protocol::PROTOCOL_VERSION
347    ))
348}

350fn link_probe() -> Result<String, String> {
351    use super::link::{sign, verify_signed, Domain, LINK_PROTOCOL_MAX, LINK_PROTOCOL_MIN};
352    let identity = NodeIdentity::ephemeral().map_err(|e| e.to_string())?;
353    let mut signed = sign(
354        &identity,
355        Domain::Hello,
356        serde_json::json!({"probe": super::link::fresh_token()}),
357    );
358    if !verify_signed(&identity.public.public_key, Domain::Hello, &signed) {
359        return Err("a fresh hello does not verify".into());
360    }
361    signed.body = serde_json::json!({"probe": "tampered"});
362    if verify_signed(&identity.public.public_key, Domain::Hello, &signed) {
363        return Err("a tampered hello verifies".into());
364    }
365    Ok(format!(
366        "hello signed → verified → tampered copy refused, link protocol {LINK_PROTOCOL_MIN}..{LINK_PROTOCOL_MAX}"
367    ))
368}

374    #[test]
375    fn the_self_check_runs_without_a_declaration_and_without_a_network() {
376        let report = doctor(None);
377        let names: Vec<&str> = report.checks.iter().map(|c| c.check.as_str()).collect();
378        assert_eq!(
379            names,
380            vec![
381                "declaration",
382                "identity",
383                "udp",
384                "multicast",
385                "broadcast",
386                "protocol",
387                "link"
388            ]
389        );
390        for check in ["protocol", "link"] {
391            let c = report.checks.iter().find(|c| c.check == check).unwrap();
392            assert!(c.ok, "{}", c.detail);
393        }
394    }

396    #[test]
397    fn a_parsed_declaration_is_summarised_in_the_verdict() {
398        let config: MeshConfig = serde_json::from_value(serde_json::json!({
399            "schema": "mesh/v1", "kind": "mesh-declaration", "id": "docs", "enabled": true
400        }))
401        .unwrap();
402        let report = doctor(Some(Ok(config)));
403        let declaration = report
404            .checks
405            .iter()
406            .find(|c| c.check == "declaration")
407            .unwrap();
408        assert!(declaration.ok);
409        assert!(
410            declaration.detail.contains("enabled=true"),
411            "{}",
412            declaration.detail
413        );
414        assert!(declaration.detail.contains("trust=deny_unknown"));
415        assert!(declaration.detail.contains("cooperation=true"));
416    }

418    #[test]
419    fn a_broken_declaration_is_one_failed_check_with_its_impact_and_remedy() {
420        let report = doctor(Some(Err(MeshError::Config("bad".into()))));
421        let declaration = report
422            .checks

424            .find(|c| c.check == "declaration")

428        assert!(!report.ok);
429    }

436            .arg(dir.path())
