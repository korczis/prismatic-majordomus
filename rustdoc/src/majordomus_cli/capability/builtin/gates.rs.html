1//! The `gates` module: completion as a state the repository decides.
2//!
3//! Two capabilities, and the split is the same one [`super::obligations`] makes for the
4//! same reason.
5//!
6//! **The model is declared.** `.ai/repo/ci/gates.yaml` is this repository's CI model: every
7//! validation gate, the job that runs it, the command it runs, and the path classes that
8//! decide which change selects which gate. `gates.model` reads it and answers in any
9//! checkout that has one, whether or not the lifecycle has ever run.
10//!
11//! **The completion is local.** Whether *this* task may be called finished depends on the
12//! task record and the ledger under `.ai/local/state/`, which is one checkout's own state
13//! and is never tracked or published. `gates.completion` answers about the checkout the
14//! process was started in and about no other.
15//!
16//! **It is read, never written.** `majordomus evidence --gate` records a run; this judges
17//! the record. A second writer for the same ledger would be a second account of events,
18//! which is what ADR 0030 refuses.
19//!
20//! **One judgement, and every surface reads it.** The shell tool's `check` and `finish`
21//! call this executable rather than re-deriving the answer, the HTTP route serves the same
22//! execution, MCP exposes the same tool, and the Cockpit renders the same document. That is
23//! the property the module exists for: a worker cannot get a different verdict by asking a
24//! different surface, and no surface can be persuaded that "done" is true because an agent
25//! said so.
26//!
27//! ```
28//! use majordomus_cli::capability::builtin::gates::module;
29//!
30//! // the whole module, as every projection reads it: two capabilities, and neither has a
31//! // command-line arm of its own because `check` and `finish` are the command line's answer
32//! let m = module();
33//! let ids: Vec<&str> = m.capabilities.iter().map(|c| c.capability.id.as_str()).collect();
34//! assert_eq!(ids, ["gates.model", "gates.completion"]);
35//! for entry in &m.capabilities {
36//!     assert!(entry.capability.exposure.cli.is_none());
37//!     assert!(entry.capability.exposure.http.is_some());
38//! }
39//! ```

67/// The CI model this repository declares: what may refuse a change, and what selects each.
68///
69/// Two things about this report are worth knowing before reading one. `count` is measured
70/// from `gates` rather than written down, so the two can never disagree the way a
71/// hand-maintained total does. And `unreadable` is how a reader tells "this repository
72/// declares no CI model" from "the call failed": a repository of the layer is not obliged
73/// to declare gates, so the empty model is an answer, and it says why it is empty.
74///
75/// ```
76/// use majordomus_cli::capability::builtin::gates::GateModelReport;
77/// use serde_json::json;
78///
79/// // the shape every surface serves: `unreadable` is skipped when the model was read
80/// let r: GateModelReport = serde_json::from_value(json!({
81///     "version": 1,
82///     "source": ".ai/repo/ci/gates.yaml",
83///     "count": 0,
84///     "on_demand": [],
85///     "gates": [],
86///     "classes": [],
87/// })).unwrap();
88/// assert_eq!(r.count, r.gates.len(), "the total is measured, never declared");
89/// assert!(r.unreadable.is_none(), "read, and it declares nothing");
90///
91/// // and the other empty answer, which a reader must not confuse with that one
92/// let absent: GateModelReport = serde_json::from_value(json!({
93///     "version": 0,
94///     "source": ".ai/repo/ci/gates.yaml",
95///     "count": 0,
96///     "on_demand": [],
97///     "gates": [],
98///     "classes": [],
99///     "unreadable": "no such file",
100/// })).unwrap();
101/// assert_eq!(absent.unreadable.as_deref(), Some("no such file"));
102/// ```
103pub struct GateModelReport {
104    /// The schema version the file states.
105    pub version: u32,
106    /// Where it was read from, repository-relative.
107    pub source: String,
108    /// How many gates it declares; measured, not written down.
109    pub count: usize,
110    /// The gates whose runner cannot be had on demand, so no routine plan selects them.
111    /// A gate that never runs protects nothing, and this is where that is said out loud.
112    pub on_demand: Vec<String>,
113    /// Every gate, in declaration order, with the pathspecs its evidence is taken over.
114    pub gates: Vec<GateModelEntry>,
115    /// Every path class, in declaration order.
116    pub classes: Vec<model::GateClass>,
117    #[serde(default, skip_serializing_if = "Option::is_none")]
118    /// Why there is no model to report, when there is none. A repository of the layer does
119    /// not have to declare a CI model, and a reader is entitled to the difference between
120    /// "this repository declares no gates" and "this call failed".
121    pub unreadable: Option<String>,
122}

124/// ```
125/// use majordomus_cli::capability::builtin::gates::GateModelEntry;
126/// use majordomus_cli::gates::GateDecl;
127///
128/// let decl: GateDecl = serde_json::from_value(serde_json::json!({
129///     "id": "site-check",
130///     "job": "site",
131///     "runs": "scripts/site-check",
132///     "summary": "the site is whole"
133/// }))
134/// .unwrap();
135/// let entry = GateModelEntry { declared: decl, inputs: vec!["site/**".into()] };
136/// // the declaration as the file carries it, plus the paths its evidence is taken over —
137/// // derived from the classes that select it, never written down a second time
138/// assert_eq!(entry.declared.id, "site-check");
139/// assert_eq!(entry.inputs, ["site/**"]);
140/// ```
141#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
142/// One gate of the model, with what the model implies about it.
143pub struct GateModelEntry {
144    #[serde(flatten)]
145    /// The declaration as the file carries it.
146    pub declared: model::GateDecl,
147    /// The pathspecs a run of this gate is evidence over, derived from the classes that
148    /// select it. A change to any of them makes a recorded run stale.
149    pub inputs: Vec<String>,
150}

154/// ```
155/// use majordomus_cli::capability::builtin::gates::CompletionInput;
156///
157/// // the default asks about the active task's own change set
158/// let mine = CompletionInput::default();
159/// assert!(mine.changed.is_none() && mine.base.is_none());
160///
161/// // and a caller may name the change instead, which is how a test drives it
162/// let named = CompletionInput {
163///     changed: Some(vec!["docs/CLI.md".into()]),
164///     ..Default::default()

166/// assert_eq!(named.changed.unwrap(), ["docs/CLI.md"]);
167/// ```
168#[derive(Debug, Clone, Default, Serialize, Deserialize, JsonSchema)]
169#[serde(deny_unknown_fields)]
170/// Which change to judge completion over.
171pub struct CompletionInput {
172    #[serde(default, skip_serializing_if = "Option::is_none")]
173    /// The changed paths, repository-relative. Absent means the task's own change set:
174    /// everything between the commit the task started at and the working tree.
175    pub changed: Option<Vec<String>>,
176    #[serde(default, skip_serializing_if = "Option::is_none")]
177    /// The commit to compare with, when the task's own starting commit is not wanted.
178    pub base: Option<String>,
179    #[serde(default, skip_serializing_if = "Option::is_none")]
180    /// Also plan the gates the model marks on-demand. Off by default, because a plan that
181    /// selects a gate whose runner is unavailable is a plan whose verdict never arrives.
182    pub on_demand: Option<bool>,
183}

217/// The vocabulary the distribution ships, read the way every other reader locates it.
218fn vocabulary(ctx: &Context) -> Result<Vec<Obligation>, String> {
219    let root = PathBuf::from(&ctx.index.repository.root);
220    let share = crate::share::Share::locate(ctx.index.share.as_deref(), &root)
221        .map_err(|e| format!("no distribution to read the obligation vocabulary from: {e}"))?;
222    let path = share.dir().join("obligations.yaml");
223    let text = std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
224    #[derive(Deserialize)]
225    struct File {
226        obligations: Vec<Obligation>,
227    }
228    let file: File =
229        crate::metadata::yaml::parse_into(&text).map_err(|e| format!("{}: {e}", path.display()))?;
230    Ok(file.obligations)
231}

233/// The CI model, or an empty one and the reason there is none.
234///
235/// A repository of the layer does not have to declare gates, and every reader of this module
236/// already says so in its own way: `lib/gates.sh` skips with "this repository declares no
237/// readable CI model" rather than failing. Refusing the call instead would make a capability
238/// that cannot answer about most repositories — and the generic tests that replay every
239/// capability's benchmark cases against a fixture are exactly where that showed.
240fn load_model(root: &Path) -> (model::GateModel, Option<String>) {
241    match model::GateModel::load(root) {
242        Ok(m) => (m, None),
243        Err(reason) => (
244            model::GateModel {
245                version: 0,
246                gates: Vec::new(),
247                classes: Vec::new(),
248            },
249            Some(reason),
250        ),
251    }
252}
253
254fn gates_model(ctx: &Context, _: Empty) -> Result<GateModelReport, CapabilityError> {
255    let root = PathBuf::from(&ctx.index.repository.root);
256    let (m, unreadable) = load_model(&root);
257    Ok(GateModelReport {
258        version: m.version,
259        source: model::MODEL_PATH.to_string(),
260        count: m.gates.len(),
261        on_demand: m
262            .gates
263            .iter()
264            .filter(|g| g.on_demand)
265            .map(|g| g.id.clone())
266            .collect(),
267        gates: m
268            .gates
269            .iter()
270            .map(|g| GateModelEntry {
271                inputs: m.inputs_of(&g.id),
272                declared: g.clone(),
273            })
274            .collect(),
275        classes: m.classes.clone(),
276        unreadable,
277    })
278}

280fn gates_completion(ctx: &Context, input: CompletionInput) -> Result<Completion, CapabilityError> {
281    let root = PathBuf::from(&ctx.index.repository.root);
282    let mut findings: Vec<String> = Vec::new();
283
284    let (m, unreadable) = load_model(&root);
285    if let Some(reason) = unreadable {
286        // no model is not no answer: every gate is unknown, said out loud, and `finishable`
287        // is left to the obligations rather than granted by a silence
288        findings.push(format!(
289            "this repository declares no readable CI model at {}, so no gate can be judged              here — unknown, never a pass: {reason}",
290            model::MODEL_PATH
291        ));
292    }
293    let task = super::continuity::read_task(&gates::task_path(&root));
294    if task.is_none() {
295        findings.push(format!(
296            "no active task in this checkout ({}); the gates are reported against the \
297             working tree and nothing is attributed to a task",
298            gates::task_path(&root)
299                .strip_prefix(&root)
300                .unwrap_or(&gates::task_path(&root))
301                .display()
302        ));
303    }
304
305    // the change set: what was asked for, or the task's own — from its starting commit to
306    // the working tree, the uncommitted half included
307    let changed = match input.changed {
308        Some(list) => list,
309        None => {
310            let base = input
311                .base
312                .clone()
313                .or_else(|| task.as_ref().map(|t| t.head.clone()));
314            match gates::changed_paths(&root, base.as_deref()) {
315                Ok(paths) => paths,
316                Err(reason) => {
317                    findings.push(format!(
318                        "the change set could not be read from git, so applicability is \
319                         computed over nothing and every gate is reported as it stands: {reason}"
320                    ));
321                    Vec::new()
322                }
323            }
324        }
325    };
326
327    let vocab = match vocabulary(ctx) {
328        Ok(v) => v,
329        Err(reason) => {
330            findings.push(format!(
331                "the obligation vocabulary could not be read, so no obligation can be \
332                 implied from the change: {reason}"
333            ));
334            Vec::new()
335        }
336    };
337
338    let (runs, skipped) = match &task {
339        Some(t) => judge::runs_for(&gates::ledger_path(&root), &t.id),
340        None => (BTreeMap::new(), 0),
341    };
342    if skipped > 0 {
343        findings.push(format!(
344            "{skipped} ledger line(s) could not be read and were skipped; run `majordomus doctor`"
345        ));
346    }
347
348    // one hash per gate, over the files that select it. Taken here rather than inside the
349    // judgement so that a test drives the same function with hashes of its own.
350    let mut hashes: BTreeMap<String, Option<String>> = BTreeMap::new();
351    for gate in &m.gates {
352        let specs = m.inputs_of(&gate.id);
353        let hash = super::obligations::inputs_hash(&root, &specs).map(|(h, _)| h);
354        // a token with no inputs hashes to the empty string in the ledger, and is compared
355        // as such; `None` here means the hash could not be taken at all
356        hashes.insert(
357            gate.id.clone(),
358            if specs.is_empty() {
359                Some(String::new())
360            } else {
361                hash
362            },
363        );
364    }
365
366    // The obligation half of the done invariant is `obligations.closure`'s judgement, asked
367    // for through the same executor MCP and the HTTP routes use, so this report cannot
368    // disagree with `majordomus check` about whether an obligation stands. Nothing is
369    // re-judged here; the state word comes back and is translated once.
370    let mut standing: BTreeMap<String, gates::ObligationStanding> = BTreeMap::new();
371    let mut closure_reachable = false;
372    match ctx.execute("obligations.closure", serde_json::json!({})) {
373        Ok(value) => match serde_json::from_value::<super::obligations::Closure>(value) {
374            Ok(closure) => {
375                closure_reachable = true;
376                for o in closure.obligations {
377                    standing.insert(
378                        o.id.clone(),
379                        gates::ObligationStanding {
380                            state: o.state.as_str().to_string(),
381                            detail: o.detail,
382                            reproduce: o.reproduce,
383                        },
384                    );
385                }
386            }
387            Err(e) => findings.push(format!(
388                "obligations.closure answered something this report cannot read, so every \
389                 obligation question is unknown rather than passing: {e}"
390            )),
391        },
392        Err(e) => findings.push(format!(
393            "obligations.closure could not be executed, so every obligation question is \
394             unknown rather than passing: {e}"
395        )),
396    }
397
398    Ok(gates::complete(
399        &m,
400        &changed,
401        task.as_ref(),
402        &vocab,
403        &runs,
404        &hashes,
405        &standing,
406        closure_reachable,
407        input.on_demand.unwrap_or(false),
408        &crate::peers::rfc3339(std::time::SystemTime::now()),
409        findings,
410    ))
411}

415/// The `gates` module: the CI model every clone shares, and the completion state only this
416/// checkout can answer.
417///
418/// Neither has a command-line projection of its own, for the reason [`super::obligations`]
419/// states about the local half: `majordomus check` and `majordomus finish` are already the
420/// command line's answer to this question, and they read this module rather than deriving a
421/// second one. What is new is that they now have a gate half to read.
422///
423/// ```
424/// use majordomus_cli::capability::builtin::gates::module;
425/// let m = module();
426/// assert_eq!(m.id.as_str(), "gates");
427/// // every projection is derived from the declaration and written nowhere else
428/// let routes: Vec<&str> = m
429///     .capabilities
430///     .iter()
431///     .filter_map(|e| e.capability.exposure.http.as_ref().map(|h| h.path.as_str()))
432///     .collect();
433/// assert_eq!(routes, ["/api/v1/gates", "/api/v1/gates/completion"]);
434/// ```
435pub fn module() -> ModuleDescriptor {
436    module! {
437        id: "gates",
438        title: "Completion gates",
439        description: "Completion as a state the repository decides rather than a claim a worker makes: the validation gates this repository declares in its CI model, which of them a task's own change set selects, what each one last reported, whether that verdict still describes the tree it was taken over, and therefore whether the task may be called finished. A gate that has never reported is `queued` and not `pass`, a run whose files have changed since is `stale` and refuses exactly as a failure does, and a gate the change cannot affect is `exempt` rather than unknown.",
440        stability: Stability::BehaviorallyVerified,
441        capabilities: [
442            capability! {
443                id: "gates.model",
444                title: "The CI model",
445                description: "Every validation gate this repository declares — the job that runs it, the command it runs, what it proves, whether every plan selects it and whether its runner can be had on demand at all — with the path classes that decide which change selects which gate, and, derived from those classes, the pathspecs each gate's evidence is taken over.",
446                input: Empty,
447                output: GateModelReport,
448                stability: Stability::BehaviorallyVerified,
449                exposure: Exposure {
450                    mcp: Some(McpExposure {
451                        tool: Some("majordomus_gates".into()),
452                        resource: Some(McpResource { uri: GATES_URI.into(), name: "gates".into() }),
453                    }),
454                    http: get("/api/v1/gates"),
455                    cli: None,
456                },
457                tags: ["gates", "ci", "completion", "introspection"],
458                cache: CachePolicy::Process { max_entries: 4, ttl_seconds: Some(5) },
459                handler: gates_model,
460            },
461            capability! {
462                id: "gates.completion",
463                title: "Whether this task may be called finished",
464                description: "The active task's own change set — from the commit it started at to the working tree, uncommitted files included — put through the CI model: which gates it selects and why, what each one last reported and over which files, which verdicts have gone stale because the tree moved underneath them, which required gates have never reported at all, and which obligations the change implies whether or not the task declared them. `finishable` is false only when a required gate is known to be failing, stale or blocked by one that is; absence of a verdict is reported as unverified rather than silently accepted.",
465                input: CompletionInput,
466                output: Completion,
467                stability: Stability::BehaviorallyVerified,
468                exposure: Exposure {
469                    mcp: Some(McpExposure {
470                        tool: Some("majordomus_completion".into()),
471                        resource: Some(McpResource { uri: COMPLETION_URI.into(), name: "completion".into() }),
472                    }),
473                    http: get("/api/v1/gates/completion"),
474                    cli: None,
475                },
476                tags: ["gates", "completion", "tasks", "evidence"],
477                cache: CachePolicy::Disabled,
478                handler: gates_completion,
479            },
480        ],
481    }
482}

485mod tests {

488    #[test]
489    fn the_module_declares_what_it_claims() {
490        let m = module();
491        assert_eq!(m.id.as_str(), "gates");
492        assert_eq!(m.capabilities.len(), 2);
493        for e in &m.capabilities {
494            assert_eq!(e.capability.id.namespace(), "gates");
495            assert!(
496                e.capability.exposure.mcp.is_some() && e.capability.exposure.http.is_some(),
497                "{} must be readable on every surface without one recomputing it",
498                e.capability.id
499            );
500        }
501    }
502
503    #[test]
504    fn the_completion_answer_is_never_cached() {
505        // a completion state cached for even a few seconds is a state that can say
506        // `finishable` about a tree that has already changed, which is the defect
507        let m = module();
508        let c = m
509            .capabilities
510            .iter()
511            .find(|e| e.capability.id.as_str() == "gates.completion")
512            .expect("declared");
513        assert!(matches!(c.capability.cache, CachePolicy::Disabled));
514    }
