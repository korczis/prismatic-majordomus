1//! The completion policy: `share/completion.yaml`, read as the one definition of done.
2//!
3//! # Why the definition of done is data
4//!
5//! Before this file, "done" was stated in places that could not disagree loudly:
6//! `share/obligations.yaml` named the tokens a task owes, `.ai/repo/ci/gates.yaml` named the
7//! gates, and `gates::done` carried twenty-two questions as a Rust literal that no reader
8//! outside the executable could see, and that put no question in any stage of the work.
9//! Each was right about its own half and the only reconciliation was a reader's.
10//!
11//! The policy names the questions, the stage each belongs to, and the *source* each one is
12//! answered from; the sources are the older declarations and the closing readings. Nothing
13//! is judged here — this module reads the declaration, resolves every source it names
14//! against the vocabulary, and refuses a policy that names something this executable
15//! cannot answer. The judgement is [`super::done`]'s, which asks exactly these questions in
16//! this order; the stage is [`super::stage`]'s; the report is `gates.completion`'s, and the
17//! policy itself is served by `gates.policy`.
18//!
19//! ```
20//! use majordomus_cli::gates::{CompletionPolicy, QuestionSource};
21//!
22//! let text = "\
23//! version: 1
24//! stages:
25//!   - id: build
26//!     title: Build
27//!     summary: The work exists.
28//!   - id: ship
29//!     title: Ship
30//!     summary: It is out.
31//! questions:
32//!   - id: committed
33//!     stage: build
34//!     question: Is it committed?
35//!     source: obligation:commit
36//!     remediation: git commit
37//!   - id: ci
38//!     stage: ship
39//!     question: Did every gate pass?
40//!     source: gates
41//!     remediation: majordomus evidence --run-gates
42//! ";
43//! let policy = CompletionPolicy::parse(text, "share/completion.yaml").unwrap();
44//! assert_eq!(policy.stages.len(), 2);
45//! assert_eq!(policy.questions[0].source, "obligation:commit", "carried as the file states it");
46//! assert_eq!(policy.questions[0].kind(), QuestionSource::Obligation("commit".into()));
47//! assert_eq!(policy.questions[1].kind(), QuestionSource::Gates);
48//!
49//! // a token the vocabulary lacks is a defect of the distribution, named by question
50//! let problems = policy.validate(&["push".into()]);
51//! assert!(problems.iter().any(|p| p.contains("committed") && p.contains("commit")));
52//! // a gate the repository never declares is not a problem; it is a question never asked
53//! assert!(policy.validate(&["commit".into()]).is_empty());
54//! ```

64/// One stage of the lifecycle, in the order the policy states them. A stage carries no
65/// question of its own: the questions name it by `id`, and
66/// [`CompletionPolicy::questions_of`] gathers them back, so a stage is a heading over
67/// whatever the policy hangs under it.
68///
69/// ```
70/// use majordomus_cli::gates::StageDecl;
71///
72/// let stage = StageDecl {
73///     id: "commit".into(),
74///     title: "Commit and push".into(),
75///     summary: "The work is in the branch's history.".into(),
76/// };
77/// // the same shape the file carries, so a report and the policy agree byte for byte
78/// let json = serde_json::to_value(&stage).unwrap();
79/// assert_eq!(json["id"], "commit");
80/// assert_eq!(serde_json::from_value::<StageDecl>(json).unwrap(), stage);
81/// ```
82#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
83pub struct StageDecl {
84    /// The stage's identity, stable across reports.
85    pub id: String,
86    /// The stage as a heading.
87    pub title: String,
88    /// What it means for the stage to be complete.
89    pub summary: String,
90}

92/// Where a question's answer is taken from.
93///
94/// ```
95/// use majordomus_cli::gates::QuestionSource;
96/// assert_eq!(QuestionSource::parse("obligation:tests").unwrap(), QuestionSource::Obligation("tests".into()));
97/// assert_eq!(QuestionSource::parse("gate:release-check").unwrap(), QuestionSource::Gate("release-check".into()));
98/// assert_eq!(QuestionSource::parse("gates").unwrap(), QuestionSource::Gates);
99/// assert_eq!(QuestionSource::parse("closing:debt").unwrap(), QuestionSource::ClosingDebt);
100/// assert!(QuestionSource::parse("magic").is_err(), "a source nothing answers is refused");
101/// assert_eq!(QuestionSource::Elsewhere("x y".into()).to_string(), "elsewhere:x y");
102/// ```
103#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
104#[serde(tag = "kind", content = "name", rename_all = "kebab-case")]
105pub enum QuestionSource {
106    /// A token of `share/obligations.yaml`, judged by `obligations.closure`.
107    Obligation(String),
108    /// The aggregate of every gate the change selects.
109    Gates,
110    /// One gate of the CI model, by id.
111    Gate(String),
112    /// The change set itself: whether a test path is among the touched files.
113    ChangeTestPath,
114    /// `convergence.report`: whether any branch, worktree or pull request is held where its
115    /// work can be lost.
116    Convergence,
117    /// The debt baselines at the commit the task started at and now (`super::closing`).
118    ClosingDebt,
119    /// The branches and worktrees created since the task started that remain.
120    ClosingAccumulated,
121    /// The pull requests opened since the task started that are still open.
122    ClosingBacklog,
123    /// Nothing reachable from the report; the command that answers it.
124    Elsewhere(String),
125}
126
127impl QuestionSource {
128    /// Parse the `source:` field of a question. A named form (`obligation:`, `gate:`,
129    /// `elsewhere:`) carries what follows the colon with its whitespace trimmed and must
130    /// name something; a bare word is one of the fixed sources; anything else is refused
131    /// with the list of what would have been accepted, so the fix is one edit of the file.
132    ///
133    /// ```
134    /// use majordomus_cli::gates::QuestionSource;
135    ///
136    /// let parsed = QuestionSource::parse(" gate: release-check ").unwrap();
137    /// assert_eq!(parsed, QuestionSource::Gate("release-check".into()));
138    /// assert_eq!(QuestionSource::parse("closing:backlog").unwrap(), QuestionSource::ClosingBacklog);
139    /// assert!(QuestionSource::parse("obligation:").unwrap_err().contains("names nothing"));
140    /// assert!(QuestionSource::parse("nope").unwrap_err().contains("obligation:<token>"));
141    /// ```
142    pub fn parse(text: &str) -> Result<Self, String> {
143        let text = text.trim();
144        if let Some(rest) = text.strip_prefix("obligation:") {
145            return Self::named(rest, "obligation").map(Self::Obligation);
146        }
147        if let Some(rest) = text.strip_prefix("gate:") {
148            return Self::named(rest, "gate").map(Self::Gate);
149        }
150        if let Some(rest) = text.strip_prefix("elsewhere:") {
151            return Self::named(rest, "elsewhere").map(Self::Elsewhere);
152        }
153        match text {
154            "gates" => Ok(Self::Gates),
155            "change:test-path" => Ok(Self::ChangeTestPath),
156            "convergence" => Ok(Self::Convergence),
157            "closing:debt" => Ok(Self::ClosingDebt),
158            "closing:accumulated" => Ok(Self::ClosingAccumulated),
159            "closing:backlog" => Ok(Self::ClosingBacklog),
160            other => Err(format!(
161                "source '{other}' is not one this executable answers (obligation:<token>, \
162                 gates, gate:<id>, change:test-path, convergence, closing:debt, \
163                 closing:accumulated, closing:backlog, elsewhere:<command>)"
164            )),
165        }
166    }
167
168    fn named(rest: &str, kind: &str) -> Result<String, String> {
169        let rest = rest.trim();
170        if rest.is_empty() {
171            return Err(format!("source '{kind}:' names nothing"));
172        }
173        Ok(rest.to_string())
174    }
175}

177impl std::fmt::Display for QuestionSource {
178    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
179        match self {
180            Self::Obligation(t) => write!(f, "obligation:{t}"),
181            Self::Gates => write!(f, "gates"),
182            Self::Gate(g) => write!(f, "gate:{g}"),
183            Self::ChangeTestPath => write!(f, "change:test-path"),
184            Self::Convergence => write!(f, "convergence"),
185            Self::ClosingDebt => write!(f, "closing:debt"),
186            Self::ClosingAccumulated => write!(f, "closing:accumulated"),
187            Self::ClosingBacklog => write!(f, "closing:backlog"),
188            Self::Elsewhere(c) => write!(f, "elsewhere:{c}"),
189        }
190    }
191}

193/// One question of the policy, as declared.
194///
195/// `source` is carried as the text the file states (`obligation:tests`, `gates`,
196/// `closing:debt`), never as a structure: the policy file, `gates.policy` and the
197/// questions of the completion report then state the same bytes for the same question,
198/// and a reader compares them by eye. [`Self::kind`]
199/// parses it; [`CompletionPolicy::parse`] has already refused a text nothing answers.
200///
201/// ```
202/// use majordomus_cli::gates::{QuestionDecl, QuestionSource};
203///
204/// let decl = QuestionDecl {
205///     id: "pushed".into(),
206///     stage: "commit".into(),
207///     question: "Has the commit reached the remote?".into(),
208///     source: "obligation:push".into(),
209///     remediation: "git push".into(),
210/// };
211/// assert_eq!(decl.kind(), QuestionSource::Obligation("push".into()));
212/// // the source is carried as text, and serialises as the text the file stated
213/// assert_eq!(serde_json::to_value(&decl).unwrap()["source"], "obligation:push");
214/// ```
215#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
216pub struct QuestionDecl {
217    /// The question's identity, stable across reports.
218    pub id: String,
219    /// The stage it belongs to.
220    pub stage: String,
221    /// The question, as a person would ask it.
222    pub question: String,
223    /// What answers it, as the policy states it.
224    pub source: String,
225    /// What would settle it, as a command.
226    pub remediation: String,
227}
228
229impl QuestionDecl {
230    /// The source, parsed. A text the parser refuses — impossible for a policy that came
231    /// through [`CompletionPolicy::parse`] — is answered by nothing, which is what
232    /// [`QuestionSource::Elsewhere`] means.
233    ///
234    /// ```
235    /// use majordomus_cli::gates::{QuestionDecl, QuestionSource};
236    ///
237    /// let mut decl = QuestionDecl {
238    ///     id: "ci".into(),
239    ///     stage: "gates".into(),
240    ///     question: "Did every gate pass?".into(),
241    ///     source: "gates".into(),
242    ///     remediation: "majordomus evidence --run-gates".into(),
243    /// };
244    /// assert_eq!(decl.kind(), QuestionSource::Gates);
245    /// // a text the parser refuses is answered by nothing, and says so rather than failing
246    /// decl.source = "oracle".into();
247    /// assert_eq!(decl.kind(), QuestionSource::Elsewhere("oracle".into()));
248    /// ```
249    pub fn kind(&self) -> QuestionSource {
250        QuestionSource::parse(&self.source)
251            .unwrap_or_else(|_| QuestionSource::Elsewhere(self.source.clone()))
252    }
253}

255/// The policy, as read: the lifecycle's stages in order and every question with the
256/// source that answers it. It is the one definition of done, and every projection of that
257/// definition — the completion report's questions and stage, the outline `gates.policy`
258/// serves — is derived from a value of this type rather than restated beside it. A value only exists
259/// once [`Self::parse`] has accepted the file, so a policy in hand is structurally sound:
260/// every question names a declared stage and a source this executable answers.
261///
262/// ```
263/// use majordomus_cli::gates::CompletionPolicy;
264///
265/// let text = "\
266/// version: 1
267/// stages:
268///   - id: build
269///     title: Build
270///     summary: The work exists.
271///   - id: ship
272///     title: Ship
273///     summary: It is out.
274/// questions:
275///   - id: committed
276///     stage: build
277///     question: Is it committed?
278///     source: obligation:commit
279///     remediation: git commit
280///   - id: changelog
281///     stage: ship
282///     question: Is the release record current?
283///     source: gate:release-check
284///     remediation: scripts/ci/release-check
285/// ";
286/// let policy = CompletionPolicy::parse(text, "inline").unwrap();
287/// assert_eq!(policy.version, 1);
288/// assert_eq!(policy.source, "inline");
289/// assert_eq!(policy.stages[1].title, "Ship");
290/// assert_eq!(policy.questions_of("build").count(), 1);
291/// assert_eq!(policy.outline(), "- Build: committed\n- Ship: changelog\n");
292/// ```
293#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
294pub struct CompletionPolicy {
295    /// The format version the file states.
296    pub version: u32,
297    /// Where it was read from.
298    pub source: String,
299    /// The lifecycle, in order.
300    pub stages: Vec<StageDecl>,
301    /// Every question, in declaration order.
302    pub questions: Vec<QuestionDecl>,
303}

321impl CompletionPolicy {
322    /// Parse the policy text. `source` is what the report names as where it came from,
323    /// and every refusal starts with it. A file of another version, a question whose
324    /// source nothing answers, a stage declared twice or never declared, an empty question
325    /// and a policy with no question at all are all refused here, so that no later reader
326    /// meets them.
327    ///
328    /// ```
329    /// use majordomus_cli::gates::CompletionPolicy;
330    ///
331    /// let text = "\
332    /// version: 1
333    /// stages:
334    ///   - id: build
335    ///     title: Build
336    ///     summary: The work exists.
337    /// questions:
338    ///   - id: committed
339    ///     stage: build
340    ///     question: Is it committed?
341    ///     source: '  obligation:commit  '
342    ///     remediation: git commit
343    /// ";
344    /// let policy = CompletionPolicy::parse(text, "share/completion.yaml").unwrap();
345    /// assert_eq!(policy.questions[0].source, "obligation:commit", "trimmed as it is read");
346    ///
347    /// // a source nothing answers is refused by question, and the refusal names the file
348    /// let err = CompletionPolicy::parse(&text.replace("obligation:commit", "oracle"), "t").unwrap_err();
349    /// assert!(err.starts_with("t: question 'committed'"), "{err}");
350    /// assert!(CompletionPolicy::parse("version: 2\nstages: []\nquestions: []\n", "t")
351    ///     .unwrap_err()
352    ///     .contains("version 1"));
353    /// ```
354    pub fn parse(text: &str, source: &str) -> Result<Self, String> {
355        let raw: RawFile = yaml::parse_into(text).map_err(|e| format!("{source}: {e}"))?;
356        if raw.version != 1 {
357            return Err(format!(
358                "{source}: the completion policy must be version 1, and this one is {}",
359                raw.version
360            ));
361        }
362        let mut questions = Vec::with_capacity(raw.questions.len());
363        for q in raw.questions {
364            QuestionSource::parse(&q.source)
365                .map_err(|e| format!("{source}: question '{}': {e}", q.id))?;
366            questions.push(QuestionDecl {
367                id: q.id,
368                stage: q.stage,
369                question: q.question,
370                source: q.source.trim().to_string(),
371                remediation: q.remediation,
372            });
373        }
374        let policy = Self {
375            version: raw.version,
376            source: source.to_string(),
377            stages: raw.stages,
378            questions,
379        };
380        let structural = policy.structural_problems();
381        if !structural.is_empty() {
382            return Err(format!("{source}: {}", structural.join("; ")));
383        }
384        Ok(policy)
385    }

387    /// Read the policy from a distribution's `share/` directory: the file [`POLICY_FILE`]
388    /// under it, parsed with the path as its source. A directory without the file is an
389    /// error naming the path, since a distribution without a definition of done is not
390    /// one this executable can judge completion against.
391    ///
392    /// ```
393    /// use majordomus_cli::gates::{CompletionPolicy, POLICY_FILE};
394    ///
395    /// let share = tempfile::tempdir().unwrap();
396    /// assert!(CompletionPolicy::load(share.path()).unwrap_err().contains(POLICY_FILE));
397    ///
398    /// std::fs::write(
399    ///     share.path().join(POLICY_FILE),
400    ///     "version: 1\nstages:\n  - id: a\n    title: Alpha\n    summary: s\nquestions:\n  - id: q\n    stage: a\n    question: Is it?\n    source: gates\n    remediation: run it\n",
401    /// )
402    /// .unwrap();
403    /// let policy = CompletionPolicy::load(share.path()).unwrap();
404    /// assert_eq!(policy.source, format!("share/{POLICY_FILE}"), "named as the distribution names it");
405    /// assert!(!policy.source.starts_with('/'), "never by where this machine keeps it");
406    /// assert_eq!(policy.questions[0].id, "q");
407    /// ```
408    pub fn load(share_dir: &std::path::Path) -> Result<Self, String> {
409        let path = share_dir.join(POLICY_FILE);
410        let text =
411            std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
412        // the source is named as the distribution names it, never by where this machine keeps
413        // it: every surface that serves the policy (MCP, HTTP, the completion report) carries
414        // this string, and an absolute path would publish the reader's home directory
415        Self::parse(&text, &format!("share/{POLICY_FILE}"))
416    }
417
418    /// The problems a policy has on its own: a duplicate id, a question naming a stage the
419    /// policy does not declare, an empty question, no question at all. A stage no question
420    /// names is not a problem: it is declared so the lifecycle reads whole, and the fold
421    /// reports it `empty`, which neither blocks nor completes.
422    fn structural_problems(&self) -> Vec<String> {
423        let mut out = Vec::new();
424        let mut seen = std::collections::BTreeSet::new();
425        for s in &self.stages {
426            if !seen.insert(s.id.as_str()) {
427                out.push(format!("stage '{}' is declared twice", s.id));
428            }
429        }
430        let mut qseen = std::collections::BTreeSet::new();
431        for q in &self.questions {
432            if !qseen.insert(q.id.as_str()) {
433                out.push(format!("question '{}' is declared twice", q.id));
434            }
435            if !seen.contains(q.stage.as_str()) {
436                out.push(format!(
437                    "question '{}' names stage '{}', which the policy does not declare",
438                    q.id, q.stage
439                ));
440            }
441            if q.question.trim().is_empty() || q.remediation.trim().is_empty() {
442                out.push(format!(
443                    "question '{}' must carry a question and a remediation",
444                    q.id
445                ));
446            }
447        }
448        if self.questions.is_empty() {
449            out.push("the policy declares no question".into());
450        }
451        out
452    }

454    /// The problems a policy has against the vocabulary it is shipped beside: an
455    /// obligation token `share/obligations.yaml` does not declare. Both files travel
456    /// together, so a token one names and the other lacks is a defect of the distribution,
457    /// and each problem names the question so the fix is one edit.
458    ///
459    /// A gate the *repository's* CI model does not declare is not a problem: the model is
460    /// the repository's, the policy is shipped, and a question a repository never asks is
461    /// answered `exempt` by name in the report. [`Self::unanswered_gates`] lists those.
462    ///
463    /// ```
464    /// use majordomus_cli::gates::CompletionPolicy;
465    ///
466    /// let text = "\
467    /// version: 1
468    /// stages:
469    ///   - id: build
470    ///     title: Build
471    ///     summary: The work exists.
472    ///   - id: ship
473    ///     title: Ship
474    ///     summary: It is out.
475    /// questions:
476    ///   - id: committed
477    ///     stage: build
478    ///     question: Is it committed?
479    ///     source: obligation:commit
480    ///     remediation: git commit
481    ///   - id: changelog
482    ///     stage: ship
483    ///     question: Is the release record current?
484    ///     source: gate:release-check
485    ///     remediation: scripts/ci/release-check
486    /// ";
487    /// let policy = CompletionPolicy::parse(text, "t").unwrap();
488    /// assert!(policy.validate(&["commit".into(), "push".into()]).is_empty());
489    ///
490    /// let problems = policy.validate(&["push".into()]);
491    /// assert_eq!(problems.len(), 1);
492    /// assert!(problems[0].contains("question 'committed'") && problems[0].contains("'commit'"));
493    /// ```
494    pub fn validate(&self, obligations: &[String]) -> Vec<String> {
495        let mut out = self.structural_problems();
496        for q in &self.questions {
497            if let QuestionSource::Obligation(token) = q.kind() {
498                if !obligations.contains(&token) {
499                    out.push(format!(
500                        "question '{}' is answered by obligation '{token}', which \
501                         share/obligations.yaml does not declare",
502                        q.id
503                    ));
504                }
505            }
506        }
507        out
508    }

510    /// The questions answered by a gate this repository's CI model does not declare, as
511    /// `question:gate` pairs: what the policy would ask and the repository never does.
512    ///
513    /// ```
514    /// use majordomus_cli::gates::CompletionPolicy;
515    ///
516    /// let text = "\
517    /// version: 1
518    /// stages:
519    ///   - id: build
520    ///     title: Build
521    ///     summary: The work exists.
522    ///   - id: ship
523    ///     title: Ship
524    ///     summary: It is out.
525    /// questions:
526    ///   - id: committed
527    ///     stage: build
528    ///     question: Is it committed?
529    ///     source: obligation:commit
530    ///     remediation: git commit
531    ///   - id: changelog
532    ///     stage: ship
533    ///     question: Is the release record current?
534    ///     source: gate:release-check
535    ///     remediation: scripts/ci/release-check
536    /// ";
537    /// let policy = CompletionPolicy::parse(text, "t").unwrap();
538    /// assert!(policy.unanswered_gates(&["release-check".into()]).is_empty());
539    /// // a repository whose model declares no such gate is told which question it never asks
540    /// assert_eq!(policy.unanswered_gates(&["rust-check".into()]), ["changelog:release-check"]);
541    /// ```
542    pub fn unanswered_gates(&self, gates: &[String]) -> Vec<String> {
543        self.questions
544            .iter()
545            .filter_map(|q| match q.kind() {
546                QuestionSource::Gate(g) if !gates.contains(&g) => Some(format!("{}:{g}", q.id)),
547                _ => None,
548            })
549            .collect()
550    }

553    /// no question names yields nothing, and so does a stage id the policy never declared;

556    /// ```
557    /// use majordomus_cli::gates::CompletionPolicy;
558    ///
559    /// let text = "\
560    /// version: 1
561    /// stages:
562    ///   - id: build
563    ///     title: Build
564    ///     summary: The work exists.
565    ///   - id: ship
566    ///     title: Ship
567    ///     summary: It is out.
568    /// questions:
569    ///   - id: committed
570    ///     stage: build
571    ///     question: Is it committed?
572    ///     source: obligation:commit
573    ///     remediation: git commit
574    ///   - id: changelog
575    ///     stage: ship
576    ///     question: Is the release record current?
577    ///     source: gate:release-check
578    ///     remediation: scripts/ci/release-check
579    /// ";
580    /// let policy = CompletionPolicy::parse(text, "t").unwrap();
581    /// let ids: Vec<&str> = policy.questions_of("ship").map(|q| q.id.as_str()).collect();
582    /// assert_eq!(ids, ["changelog"]);
583    /// assert_eq!(policy.questions_of("nowhere").count(), 0);
584    /// ```
585    pub fn questions_of<'a>(
586        &'a self,
587        stage: &'a str,
588    ) -> impl Iterator<Item = &'a QuestionDecl> + 'a {
589        self.questions.iter().filter(move |q| q.stage == stage)
590    }

592    /// The outline: the definition of done in a dozen lines rather than a page. One line
593    /// per stage that asks anything, in the policy's order, its questions after a colon; a
594    /// stage no question names is left out, because a heading over nothing tells a reader
595    /// nothing. Plain list items, so that the outline can be quoted into any Markdown
596    /// document as it is.
597    ///
598    /// ```
599    /// use majordomus_cli::gates::CompletionPolicy;
600    /// let p = CompletionPolicy::parse("version: 1\nstages:\n  - id: a\n    title: Alpha\n    summary: s\nquestions:\n  - id: q\n    stage: a\n    question: Is it?\n    source: gates\n    remediation: run it\n", "t").unwrap();
601    /// assert_eq!(p.outline(), "- Alpha: q\n");
602    /// ```
603    pub fn outline(&self) -> String {
604        let mut out = String::new();
605        for s in &self.stages {
606            let ids: Vec<&str> = self.questions_of(&s.id).map(|q| q.id.as_str()).collect();
607            if ids.is_empty() {
608                continue;
609            }
610            out.push_str(&format!("- {}: {}\n", s.title, ids.join(", ")));
611        }
612        out
613    }
614}
