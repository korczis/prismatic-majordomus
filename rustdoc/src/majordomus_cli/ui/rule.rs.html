1//! One rule in full, for the Governance screen's inspector (`/ui/governance?rule=<id>`): what
2//! it states, how it is enforced, every test that proves it with the state of its proof, the
3//! gates that run them, what is still missing to prove it, and the rules it depends on and
4//! that depend on it, each a link to its own inspector.
5//!
6//! Read from `rules.show`, the capability that joins a rule to its proof; a rule the index
7//! does not hold is its refusal, said as such.
8

19/// A rule id as the index writes them (`project.worktree-topology`, `majordomus.adr-integrity@1`):
20/// letters, digits, `.`, `-`, `_` and `@`, and nothing a link could carry anything else in.
21pub fn is_rule_id(id: &str) -> bool {
22    !id.is_empty()
23        && id.len() <= 128
24        && id
25            .bytes()
26            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_' | b'@'))
27}

29/// The rule a query names, when it names one.
30pub fn picked(query: &[(String, String)]) -> Option<String> {
31    query
32        .iter()
33        .find(|(k, _)| k == "rule")
34        .map(|(_, v)| v.trim().to_string())
35        .filter(|v| is_rule_id(v))
36}

38/// Where a rule's inspector is.
39pub fn href(id: &str) -> String {
40    format!("{}?rule={id}#rule", super::governance::ROUTE)
41}
42
43/// The id of a rule as the report or the index gives it: the rule object's `id`, or the
44/// string itself when the report gives one.
45pub fn id_of(rule: &Value) -> String {
46    match rule {
47        Value::String(s) => s.clone(),
48        other => text(other, "id"),
49    }
50}

52/// Ask `rules.show` for `id`.
53pub fn ask(ctx: &Context, id: &str) -> Answer {
54    ask_input(ctx, "rules.show", serde_json::json!({ "rule": id }))
55}

57fn chips(ids: Vec<String>) -> El {
58    if ids.is_empty() {
59        return el("span").class("mj-ui-dim").text("none");
60    }
61    el("span")
62        .class("mj-ui-badges")
63        .children(ids.into_iter().map(|id| {
64            // a dependency is written with its version (`x@1`); the inspector takes either
65            el("a")
66                .class("mj-ui-chip mj-ui-chip--link")
67                .attr("href", href(&id))
68                .child(el("code").class("mj-ui-chip-value").text(id))
69        }))
70}
71
72fn strings(v: &Value, k: &str) -> Vec<String> {
73    v.get(k)
74        .and_then(Value::as_array)
75        .map(|a| {
76            a.iter()
77                .filter_map(|x| {
78                    x.as_str()
79                        .map(str::to_string)
80                        .or_else(|| x.get("id").and_then(Value::as_str).map(str::to_string))
81                })
82                .collect()
83        })
84        .unwrap_or_default()
85}

87/// The inspector of the rule `answer` is about.
88pub fn inspector(answer: &Answer, id: &str) -> El {
89    let body = unanswered(answer).unwrap_or_else(|| {
90        // `unanswered` has spoken for an answer that failed; this one answered
91        let a = answer.ok().cloned().unwrap_or(Value::Null);
92        let proof = a.get("proof").cloned().unwrap_or(Value::Null);
93        let rule = proof.get("rule").cloned().unwrap_or(Value::Null);
94        let enforcement = rule.get("enforcement").cloned().unwrap_or(Value::Null);
95        let fact = |label: &str, value: El| {
96            el("div")
97                .class("mj-ui-fact")
98                .child(el("dt").text(label))
99                .child(el("dd").child(value))
100        };
101        let tests = proof
102            .get("tests")
103            .and_then(Value::as_array)
104            .cloned()
105            .unwrap_or_default()
106            .iter()
107            .map(|t| {
108                let path = text(t, "path");
109                // a test the report found on disk is an object of the index: its technical
110                // view; one it did not find is its path, said as text
111                let test = if path.is_empty() {
112                    none()
113                } else if t.get("present").and_then(Value::as_bool) == Some(true) {
114                    object(&format!("majordomus://test/{path}"), path)
115                } else {
116                    el("span")
117                        .class("mj-ui-mono")
118                        .text(format!("{path} (not found)"))
119                };
120                el("tr")
121                    .child(el("td").child(test))
122                    .child(el("td").child(state(t, "state")))
123                    .child(el("td").class("mj-ui-dim").text(value(t, "meaning")))
124            })
125            .collect::<Vec<_>>();
126        let missing = text(&a, "missing");
127        el("section")
128            .class("mj-ui-detail")
129            .attr("id", "rule")
130            .attr("aria-label", format!("Rule {id}"))
131            .child(
132                el("div")
133                    .class("mj-ui-detail-head")
134                    .child(
135                        el("p")
136                            .class("mj-ui-mono mj-ui-dim")
137                            .text(value(&rule, "identity")),
138                    )
139                    .child(
140                        el("h3")
141                            .class("mj-ui-detail-title")
142                            .text(value(&rule, "title")),
143                    )
144                    .child(
145                        el("p")
146                            .class("mj-ui-badges")
147                            .child(state(&rule, "class"))
148                            .child(state(&rule, "status"))
149                            .child(state(&proof, "state"))
150                            .child(
151                                el("a")
152                                    .class("mj-ui-more")
153                                    .attr("href", super::governance::ROUTE)
154                                    .text("Close"),
155                            ),
156                    ),
157            )
158            .child(el("p").class("mj-ui-lead").text(text(&rule, "statement")))
159            .child(el("p").class("mj-ui-line").text(text(&proof, "meaning")))
160            .child(
161                el("dl")
162                    .class("mj-ui-facts")
163                    .child(fact(
164                        "Enforced",
165                        el("span").class("mj-ui-mono").text(format!(
166                            "{} · {}",
167                            Some(text(&enforcement, "mode"))
168                                .filter(|s| !s.is_empty())
169                                .unwrap_or_else(|| "not said".into()),
170                            Some(text(&enforcement, "validator"))
171                                .filter(|s| !s.is_empty())
172                                .unwrap_or_else(|| "no validator".into())
173                        )),
174                    ))
175                    .child(fact("Gates", {
176                        // each gate leads to its row in the CI model
177                        let g = strings(&proof, "gates");
178                        if g.is_empty() {
179                            none()
180                        } else {
181                            el("span").class("mj-ui-badges").children(
182                                g.into_iter()
183                                    .map(|id| code_link(super::delivery::gate_href(&id), id)),
184                            )
185                        }
186                    }))
187                    .child(fact("Depends on", chips(strings(&a, "depends_on"))))
188                    .child(fact("Required by", chips(strings(&a, "required_by"))))
189                    .child(fact("Declared in", {
190                        // the rule's own record, which the index holds: rules.show found it
191                        object_or(&text(&rule, "uri"), &text(&rule, "path"))
192                    })),
193            )
194            .child(el("h4").class("mj-ui-panel-title").text("What proves it"))
195            .child(table(
196                &["Test", "State", "What that means"],
197                tests,
198                "It names no test.",
199            ))
200            .when(!missing.is_empty(), |s| {
201                s.child(el("h4").class("mj-ui-panel-title").text("To prove it"))
202                    .child(el("code").class("mj-ui-mono").text(missing))
203            })
204    });
205    panel(
206        "Rule",
207        "One rule of the repository, with what proves it and what is still missing.",
208        body,
209        &[answer],
210    )
211}

219    fn answer(value: Result<Value, String>) -> Answer {
220        Answer {
221            source: Source::of("rules.show"),

225            took: Duration::from_millis(1),

230    #[test]
231    fn a_rule_shows_its_proof_and_leads_to_the_rules_around_it() {
232        let a = answer(Ok(serde_json::json!({
233            "proof": {
234                "rule": { "id": "project.a", "identity": "project.a@1", "title": "A holds", "statement": "A is true.",
235                          "class": "blocking", "status": "active", "path": ".ai/repo/rules/project/a.v1.md",
236                          "uri": "majordomus://rule/project.a@1",
237                          "enforcement": { "mode": "dispatched", "validator": "adr" } },
238                "tests": [{ "path": "test/cases/96_a.sh", "present": true, "state": "not_run", "meaning": "never run" },
239                          { "path": "test/cases/97_gone.sh", "present": false, "state": "missing" }],
240                "gates": ["shell-suite"], "state": "not_run", "meaning": "No run is recorded."
241            },
242            "depends_on": ["project.b@1"], "required_by": [{ "id": "project.c" }],
243            "missing": "Run and record: bash test/run.sh 96_a"
244        })));
245        let html = inspector(&a, "project.a").render();
246        for want in [
247            "id=\"rule\"",
248            "A is true.",
249            "test/cases/96_a.sh",
250            "shell-suite",
251            "/ui/governance?rule=project.b@1#rule",
252            "/ui/governance?rule=project.c#rule",
253            "bash test/run.sh 96_a",
254            // every entity it names is a link: the gate to its row in the CI model, the test
255            // and the declaration to their technical views
256            "href=\"/ui/delivery#gate-shell-suite\"",
257            "href=\"/cockpit/object?uri=majordomus%3A%2F%2Ftest%2Ftest%2Fcases%2F96_a.sh\"",
258            "href=\"/cockpit/object?uri=majordomus%3A%2F%2Frule%2Fproject.a%401\"",
259            // a test the report did not find is never a link to a page that would answer 404
260            "test/cases/97_gone.sh (not found)",
261        ] {
262            assert!(html.contains(want), "no {want}: {html}");
263        }
264        assert!(!html.contains("97_gone.sh\""), "{html}");
265    }

267    /// A rule its index says little about still opens: no dependency either way, no
268    /// enforcement named, no gate, each said rather than left blank.
269    #[test]
270    fn a_rule_that_says_little_says_what_it_lacks() {
271        let html = inspector(
272            &answer(Ok(serde_json::json!({
273                "proof": { "rule": { "id": "project.z", "title": "Z" }, "tests": [], "state": "not_run" }
274            }))),
275            "project.z",
276        )
277        .render();
278        for want in ["not said", "no validator", "It names no test.", ">none<"] {
279            assert!(html.contains(want), "no {want}: {html}");
280        }
281    }

283    #[test]
284    fn an_unknown_rule_is_its_refusal_and_a_hostile_id_is_never_asked() {
285        let refused = inspector(&answer(Err("not found: no rule nope".into())), "nope").render();
286        assert!(refused.contains("no rule nope") && !refused.contains("id=\"rule\""));
287        let q = |v: &str| vec![("rule".to_string(), v.to_string())];
288        assert_eq!(
289            picked(&q("majordomus.adr-integrity@1")),
290            Some("majordomus.adr-integrity@1".into())
291        );
292        assert_eq!(picked(&q("x\" onclick=\"y")), None);
293        assert_eq!(id_of(&serde_json::json!({ "id": "p.q" })), "p.q");
294        assert_eq!(id_of(&serde_json::json!("p.r")), "p.r");
295    }
