1//! The `guard` module: whether a write to a file of this checkout is covered by a claim
2//! (I2735). The judgement is [`crate::guard`]'s, shared with `majordomus guard hook`, which a
3//! provider's pre-tool hook runs without building the index; this projects the same answer
4//! to MCP and HTTP, reading the mesh's claims from this process when it is the server and
5//! from the checkout's server otherwise.

19/// Which write to judge.
20#[derive(Debug, Serialize, Deserialize, JsonSchema)]
21#[serde(deny_unknown_fields)]
22pub struct GuardEditInput {
23    /// The path about to be written: absolute, or relative to the repository root.
24    pub path: String,
25    /// The writing session's id on the mesh (`claude-code-<session id>` for Claude Code);
26    /// without it only the active task's scope can cover the write.
27    #[serde(default, skip_serializing_if = "Option::is_none")]
28    pub session: Option<String>,
29}

52fn guard_edit(ctx: &Context, input: GuardEditInput) -> Result<EditVerdict, CapabilityError> {
53    let root = std::path::PathBuf::from(&ctx.index.repository.root);
54    let claims = || match ctx.mesh.cooperation() {
55        Some(c) => ClaimSource::Claims(c.state().claims),
56        None if ctx.mesh.decided() => {
57            let why = ctx.mesh.cooperation_status().reason.unwrap_or_default();
58            ClaimSource::Unavailable(format!("this server does not run the mesh: {why}"))
59        }
60        None => server_claims(&root),
61    };
62    Ok(judge(&root, &input.path, input.session.as_deref(), claims))
63}

65/// The module.
66pub fn module() -> ModuleDescriptor {
67    module! {
68        id: "guard",
69        title: "Edit guard",
70        description: "Whether a write to a file of this checkout is covered by a claim: the active task's scope, or a live mesh claim of the writing session. The judgement a provider's pre-tool hook asks before every edit (majordomus guard hook), switched by session.edits_require_claim.",
71        stability: Stability::Experimental,
72        capabilities: [
73            capability! {
74                id: "guard.edit",
75                title: "Judge one edit",
76                description: "Whether the given session may write the given path now: allowed when the policy's session.edits_require_claim is off, the path is outside the checkout or its local state, the repository declares no enabled mesh, the active task's scope covers it, or a live mesh claim of the session covers it; refused as unclaimed otherwise, with the remedy. What cannot be read — the policy, the mesh — is let through and named, never refused.",
77                input: GuardEditInput,
78                output: EditVerdict,
79                stability: Stability::Experimental,
80                exposure: Exposure { mcp: mcp("majordomus_guard_edit"), http: get("/api/v1/guard/edit"), cli: None },
81                tags: ["guard", "mesh", "coordination"],
82                handler: guard_edit,
83            },
84        ],
85    }
86}

108    /// A context over a synthetic repository whose policy asks every edit for a claim and
109    /// whose mesh is declared enabled, so that only the claims can decide.
110    fn on_the_mesh() -> (crate::synthetic::SyntheticRepository, Context) {
111        let repo = crate::synthetic::SyntheticRepository::small().unwrap();
112        let index = repo.index().unwrap();
113        let registry = crate::capability::CapabilityRegistry::builder()
114            .with_modules(vec![module()])
115            .with_index(&index)
116            .build()
117            .expect("the registry builds");
118        let ctx = Context::new(std::sync::Arc::new(index), std::sync::Arc::new(registry));
119        let policy = "version: 1\nsession:\n  edits_require_claim: true\n";
120        std::fs::write(repo.root().join(".ai/repo/policy.yaml"), policy).unwrap();
121        std::fs::create_dir_all(repo.root().join(".ai/repo/mesh")).unwrap();
122        std::fs::write(
123            repo.root().join(".ai/repo/mesh/mesh.yaml"),
124            "enabled: true\n",
125        )
126        .unwrap();
127        (repo, ctx)
128    }

130    fn edit(ctx: &Context, path: &str) -> EditVerdict {
131        let input = serde_json::json!({ "path": path, "session": "claude-code-s1" });
132        serde_json::from_value(ctx.execute("guard.edit", input).expect("an answer"))
133            .expect("a verdict")
134    }

136    #[test]
137    fn a_server_running_the_mesh_answers_from_its_own_claims() {
138        use crate::mesh::config::{CooperationConfig, TrustConfig};
139        use crate::mesh::cooperation::{CheckoutFacts, Cooperation, CooperationSetup};
140        use crate::mesh::identity::NodeIdentity;
141        use crate::mesh::link::HttpTransport;
142        use crate::mesh::repository::of_root_commits;
143        use std::sync::Arc;
144
145        let (_repo, ctx) = on_the_mesh();
146        let cooperation = Cooperation::new(CooperationSetup {
147            identity: Arc::new(NodeIdentity::ephemeral().unwrap()),
148            runtime: "0000000000000001".into(),
149            repository: of_root_commits(&["root".into()]),
150            endpoints: vec!["127.0.0.1:9".into()],
151            version: "test".into(),
152            config: CooperationConfig::default(),
153            trust: TrustConfig::default(),
154            journal_path: None,
155            // the one peer store, reached through the runtime that holds it
156            registry: Arc::clone(ctx.mesh.registry()),
157            transport: Arc::new(HttpTransport),
158            board: None,
159            checkout: CheckoutFacts::default(),
160        })
161        .unwrap();
162        ctx.mesh.attach_cooperation(Arc::clone(&cooperation));
163        let v = edit(&ctx, "docs/a.md");
164        assert!(!v.allowed, "the running mesh holds no claim of the session");
165        assert_eq!(v.ground, "unclaimed");
166        cooperation.stop();
167    }
