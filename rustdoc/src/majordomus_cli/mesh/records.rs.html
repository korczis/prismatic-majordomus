1//! Typed coordination records in the mesh journal (I4665).
2//!
3//! The owner decided on 2026-11-20 that the typed records agents coordinate through live in
4//! the mesh journal, the one coordination mechanism, and not in a second blackboard beside
5//! it. A record is one event kind, `record_written`, carried, signed and replicated like a
6//! claim or a handover; everything else here is the fold that reads it and the admission
7//! that refuses a write before it is signed.
8//!
9//! What a record is:
10//!
11//! - **Typed**: a [`RecordKind`] from a closed set — observation, hypothesis, plan, decision,
12//!   blocker, handoff, outcome — about one task (a task, intent or issue identifier) and a
13//!   scope of repository paths.
14//! - **Authored by the journal**: its author is the `<stream>/<session>` key of the event
15//!   that created it. The stream is the signature's, so no caller can name another runtime;
16//!   within a runtime, a session over MCP writes for itself alone.
17//! - **Versioned**: each write carries a revision. Only the author's session revises its
18//!   record, and only at the revision it read (compare-and-swap); another session supersedes
19//!   it with a new record that links back, and never overwrites it. A runtime that restarts
20//!   is a new stream: its earlier records are superseded, not revised.
21//! - **Evidenced or asserted**: the evidence state is derived, never written — a record is
22//!   an assertion unless one of its sources is retrievable (a file at a commit, or a tool's
23//!   observed result).
24//! - **Leased**: a record with an expiry is stale after it; a record whose author's session
25//!   closed or whose runtime stopped beating is stale too. Stale records are flagged, never
26//!   deleted: the journal is append-only, and every revision stays in the record's history.
27//!
28//! An older runtime relays `record_written` opaquely, as it relays every kind it does not
29//! know, so replication stays compatible; it simply does not fold the records.
30//!
31//! The lifecycle, on one journal: a record written, revised by its author, superseded by
32//! another session, and folded with its history kept.
33//!
34//! ```
35//! use std::sync::Arc;
36//! use majordomus_cli::mesh::identity::NodeIdentity;
37//! use majordomus_cli::mesh::journal::{EventBody, Journal, StreamLiveness};
38//! use majordomus_cli::mesh::records::{RecordBody, RecordKind};
39//! use majordomus_cli::mesh::state::fold;
40//!
41//! let j = Journal::open(Arc::new(NodeIdentity::ephemeral().unwrap()), "0000000000000001",
42//!     "repo".into(), None).unwrap();
43//! let plan = RecordBody::new("r-1", "s1", RecordKind::Plan, "I1", &["docs"], "write it");
44//! j.append_own(EventBody::RecordWritten { record: plan.clone() }).unwrap();
45//! let revised = RecordBody { revision: 2, title: "write it twice".into(), ..plan };
46//! j.append_own(EventBody::RecordWritten { record: revised }).unwrap();
47//! let key = format!("{}/r-1", j.own_stream());
48//! let mut other = RecordBody::new("r-2", "s2", RecordKind::Plan, "I1", &["docs"], "no");
49//! other.supersedes = Some(key.clone());
50//! j.append_own(EventBody::RecordWritten { record: other }).unwrap();
51//!
52//! let state = fold(&j.events(), &|_| StreamLiveness::Own);
53//! let first = state.records.iter().find(|r| r.key == key).unwrap();
54//! assert_eq!((first.revision, first.history.len()), (2, 2), "every revision kept");
55//! assert_eq!(first.superseded_by.len(), 1, "superseded, never overwritten");
56//! ```

90/// What a record is about. The set is closed: a peer that writes another word is refused
91/// at ingest, never guessed at.
92///
93/// ```
94/// use majordomus_cli::mesh::records::RecordKind;
95/// assert_eq!(serde_json::to_value(RecordKind::Decision).unwrap(), "decision");
96/// assert!(serde_json::from_str::<RecordKind>("\"rumour\"").is_err());
97/// assert!(RecordKind::Plan.contends() && !RecordKind::Observation.contends());
98/// ```
99#[derive(
100    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
101)]
102#[serde(rename_all = "snake_case")]
103#[schemars(rename = "MeshRecordKind")]
104pub enum RecordKind {
105    /// Something seen.
106    Observation,
107    /// Something believed and not yet shown.
108    Hypothesis,
109    /// What a session intends to do.
110    Plan,
111    /// What a session decided.
112    Decision,
113    /// What stops the work.
114    Blocker,
115    /// The work handed on: objective, current state, evidence, blockers, next action.
116    Handoff,
117    /// What the work ended with.
118    Outcome,
119}
120
121impl RecordKind {
122    /// The kind's wire word, written out rather than derived from the variant name: it is
123    /// protocol, and renaming a variant must not change what another runtime reads.
124    ///
125    /// ```
126    /// use majordomus_cli::mesh::records::RecordKind;
127    /// assert_eq!(RecordKind::Handoff.as_str(), "handoff");
128    /// assert_eq!(serde_json::to_value(RecordKind::Handoff).unwrap(), "handoff");
129    /// ```
130    pub fn as_str(self) -> &'static str {
131        match self {
132            RecordKind::Observation => "observation",
133            RecordKind::Hypothesis => "hypothesis",
134            RecordKind::Plan => "plan",
135            RecordKind::Decision => "decision",
136            RecordKind::Blocker => "blocker",
137            RecordKind::Handoff => "handoff",
138            RecordKind::Outcome => "outcome",
139        }
140    }
141
142    /// Whether two live records of this kind by different authors contend: two plans or two
143    /// decisions for one task, or over meeting scopes, are a conflict to report. Findings
144    /// (observations, hypotheses) may disagree without contending; they are superseded.
145    ///
146    /// ```
147    /// use majordomus_cli::mesh::records::RecordKind;
148    /// assert!(RecordKind::Decision.contends());
149    /// assert!(!RecordKind::Hypothesis.contends(), "findings disagree without contending");
150    /// ```
151    pub fn contends(self) -> bool {
152        matches!(self, RecordKind::Plan | RecordKind::Decision)
153    }
154}

177/// What a record cites, with its version: a file at a commit, a tool's observed result, or
178/// another record at a revision.
179///
180/// ```
181/// use majordomus_cli::mesh::records::RecordSource;
182/// let file = RecordSource::File { path: "src/a.rs".into(), commit: "8df53bc26e9".into(),
183///     lines: Some("10-20".into()) };
184/// assert!(file.validate().is_ok() && file.is_retrievable());
185/// let link = RecordSource::Record { record: "x/r-1".into(), revision: 2 };
186/// assert!(!link.is_retrievable(), "a link to a record is provenance, not evidence");
187/// ```
188#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
189#[serde(tag = "type", rename_all = "snake_case")]
190#[schemars(rename = "MeshRecordSource")]
191pub enum RecordSource {
192    /// A repository file at a commit, optionally a line range (`12` or `12-40`).
193    File {
194        /// Repository-relative path.
195        path: String,
196        /// The commit it was read at: 7 to 64 lowercase hex.
197        commit: String,
198        /// The lines: `N` or `N-M`.
199        #[serde(default, skip_serializing_if = "Option::is_none")]
200        lines: Option<String>,
201    },
202    /// A tool run and what it observed.
203    Tool {
204        /// The command or tool call, one line.
205        command: String,
206        /// What it answered: an exit code, a digest, a short excerpt.
207        result: String,
208        /// When it was observed, RFC 3339 UTC.
209        #[serde(default, skip_serializing_if = "Option::is_none")]
210        observed_at: Option<String>,
211    },
212    /// Another record at a revision.
213    Record {
214        /// The record's key, `<stream>/<id>`.
215        record: String,
216        /// The revision cited.
217        revision: u32,
218    },
219}
220
221impl RecordSource {
222    /// Whether the source can be retrieved and checked by someone else: a file at a commit,
223    /// or a tool's result. A record citing only links or nothing is an assertion.
224    ///
225    /// ```
226    /// use majordomus_cli::mesh::records::RecordSource;
227    /// let ran = RecordSource::Tool { command: "cargo test".into(), result: "exit 0".into(),
228    ///     observed_at: None };
229    /// assert!(ran.is_retrievable());
230    /// let blank = RecordSource::Tool { command: "cargo test".into(), result: " ".into(),
231    ///     observed_at: None };
232    /// assert!(!blank.is_retrievable(), "a tool source without its result proves nothing");
233    /// ```
234    pub fn is_retrievable(&self) -> bool {
235        match self {
236            RecordSource::File { .. } | RecordSource::Tool { .. } => self.validate().is_ok(),
237            RecordSource::Record { .. } => false,
238        }
239    }
240
241    /// Check the source's bounds, on the writer and on every reader: a file names a
242    /// repository-relative path and a 7-64 hex commit, a line range is `N` or `N-M` from 1, a
243    /// tool names its command and its result, and a cited record is a key at a revision.
244    ///
245    /// ```
246    /// use majordomus_cli::mesh::records::RecordSource;
247    /// let file = |lines: &str| RecordSource::File { path: "src/a.rs".into(),
248    ///     commit: "8df53bc".into(), lines: Some(lines.into()) };
249    /// assert!(file("3-9").validate().is_ok());
250    /// assert!(file("9-3").validate().unwrap_err().contains("lines"));
251    /// ```
252    pub fn validate(&self) -> Result<(), String> {
253        match self {
254            RecordSource::File {
255                path,
256                commit,
257                lines,
258            } => {
259                paths(std::slice::from_ref(path), true)?;
260                let hex = (7..=64).contains(&commit.len())
261                    && commit
262                        .bytes()
263                        .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b));
264                if !hex {
265                    return Err(format!(
266                        "source commit '{commit}' is not 7 to 64 lowercase hex"
267                    ));
268                }
269                if let Some(range) = lines {
270                    line_range(range)?;
271                }
272                Ok(())
273            }
274            RecordSource::Tool {
275                command,
276                result,
277                observed_at,
278            } => {
279                if command.trim().is_empty() || result.trim().is_empty() {
280                    return Err("a tool source names its command and its result".into());
281                }
282                line("source command", command, 512)?;
283                text("source result", result, 2048)?;
284                if let Some(at) = observed_at {
285                    rfc3339("source observed_at", at)?;
286                }
287                Ok(())
288            }
289            RecordSource::Record { record, revision } => {
290                record_key(record)?;
291                if *revision == 0 {
292                    return Err("a cited record's revision starts at 1".into());
293                }
294                Ok(())
295            }
296        }
297    }
298}

300fn line_range(range: &str) -> Result<(), String> {
301    let bad = || format!("source lines '{range}' are not N or N-M with 1 <= N <= M");
302    let (a, b) = match range.split_once('-') {
303        Some((a, b)) => (a, b),
304        None => (range, range),
305    };
306    let (a, b): (u32, u32) = (a.parse().map_err(|_| bad())?, b.parse().map_err(|_| bad())?);
307    if a == 0 || b < a {
308        return Err(bad());
309    }
310    Ok(())
311}

313fn rfc3339(name: &str, value: &str) -> Result<u64, String> {
314    crate::peers::parse_rfc3339(value)
315        .ok_or_else(|| format!("{name} '{value}' is not RFC 3339 UTC (YYYY-MM-DDTHH:MM:SSZ)"))
316}

318/// A record key, `<stream>/<id>`: the stream parsed, the id an identifier.
319fn record_key(key: &str) -> Result<(), String> {
320    let ok = key.split_once('/').is_some_and(|(stream, id)| {
321        StreamId::parse(stream).is_some() && ident("record", id).is_ok()
322    });
323    if ok {
324        Ok(())
325    } else {
326        Err(format!("'{key}' is not a record key, <stream>/<id>"))
327    }
328}

330/// What a handoff must carry. Every field is required: a handoff that leaves one out is
331/// refused, because the next worker cannot ask the one who wrote it.
332///
333/// ```
334/// use majordomus_cli::mesh::records::Handoff;
335/// let mut h = Handoff { objective: "ship".into(), current_state: "built".into(),
336///     evidence: vec!["cargo test: ok".into()], blockers: Some(vec![]),
337///     next_action: "open the PR".into() };
338/// assert!(h.validate().is_ok());
339/// h.evidence.clear();
340/// assert!(h.validate().unwrap_err().contains("evidence"));
341/// ```
342#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
343#[schemars(rename = "MeshHandoff")]
344pub struct Handoff {
345    /// What the work is for.
346    pub objective: String,
347    /// Where it stands now.
348    pub current_state: String,
349    /// What shows it: commands run with their results, commits, records. At least one.
350    #[serde(default)]
351    pub evidence: Vec<String>,
352    /// What blocks it; `[]` says nothing does. Required: silence is not "no blockers".
353    #[serde(default, skip_serializing_if = "Option::is_none")]
354    pub blockers: Option<Vec<String>>,
355    /// What the next worker does first.
356    pub next_action: String,
357}
358
359impl Handoff {
360    /// Check that every part is present and bounded: the three paragraphs non-empty, at
361    /// least one evidence item, and the blockers stated (`[]` for none) rather than omitted.
362    ///
363    /// ```
364    /// use majordomus_cli::mesh::records::Handoff;
365    /// let h = Handoff { objective: "ship".into(), current_state: "built".into(),
366    ///     evidence: vec!["cargo test: ok".into()], blockers: None,
367    ///     next_action: "open the PR".into() };
368    /// assert!(h.validate().unwrap_err().contains("blockers"), "silence is not none");
369    /// assert!(Handoff { blockers: Some(vec![]), ..h }.validate().is_ok());
370    /// ```
371    pub fn validate(&self) -> Result<(), String> {
372        for (name, value) in [
373            ("objective", &self.objective),
374            ("current_state", &self.current_state),
375            ("next_action", &self.next_action),
376        ] {
377            if value.trim().is_empty() {
378                return Err(format!("a handoff carries its {name}"));
379            }
380            text(name, value, HANDOFF_TEXT_MAX)?;
381        }
382        if self.evidence.is_empty() {
383            return Err("a handoff carries its evidence: at least one item".into());
384        }
385        let Some(blockers) = &self.blockers else {
386            return Err("a handoff says its blockers: [] when nothing blocks".into());
387        };
388        for (name, items) in [("evidence", &self.evidence), ("blockers", blockers)] {
389            if items.len() > MAX_HANDOFF_ITEMS {
390                return Err(format!(
391                    "a handoff lists at most {MAX_HANDOFF_ITEMS} {name}"
392                ));
393            }
394            for item in items {
395                if item.trim().is_empty() {
396                    return Err(format!("a handoff's {name} item is empty"));
397                }
398                line(name, item, HANDOFF_ITEM_MAX)?;
399            }
400        }
401        Ok(())
402    }
403}

405/// A record as the journal carries it: one revision of it. The author is not in it — the
406/// event's stream and the `session` within it are — and neither is its evidence state, which
407/// is derived from its sources.
408///
409/// ```
410/// use majordomus_cli::mesh::journal::EventBody;
411/// use majordomus_cli::mesh::records::{RecordBody, RecordKind};
412/// let body = RecordBody::new("r-1", "s1", RecordKind::Blocker, "I1", &["ci"], "CI is red");
413/// let wire = serde_json::to_value(EventBody::RecordWritten { record: body }).unwrap();
414/// assert_eq!(wire["kind"], "record_written");
415/// assert_eq!(wire["record"]["kind"], "blocker", "the record's own kind, inside it");
416/// ```
417#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
418#[schemars(rename = "MeshRecordBody")]
419pub struct RecordBody {
420    /// The record's id within its stream; its key is `<stream>/<id>`.
421    pub id: String,
422    /// The writing session within the stream.
423    pub session: String,
424    /// The revision this write makes, from 1.
425    pub revision: u32,
426    /// What the record is.
427    pub kind: RecordKind,
428    /// The task, intent or issue it is about.
429    pub task: String,
430    /// The repository paths it is about; inside the writing session's claims.
431    pub scope: Vec<String>,
432    /// Where it stands.
433    #[serde(default)]
434    pub status: RecordStatus,
435    /// One line.
436    pub title: String,
437    /// The text, up to [`RECORD_BODY_MAX`] bytes.
438    #[serde(default, skip_serializing_if = "Option::is_none")]
439    pub body: Option<String>,
440    /// What it cites, with versions.
441    #[serde(default, skip_serializing_if = "Vec::is_empty")]
442    pub sources: Vec<RecordSource>,
443    /// A handoff's parts; present exactly when the kind is `handoff`.
444    #[serde(default, skip_serializing_if = "Option::is_none")]
445    pub handoff: Option<Handoff>,
446    /// The record this one supersedes, `<stream>/<id>`.
447    #[serde(default, skip_serializing_if = "Option::is_none")]
448    pub supersedes: Option<String>,
449    /// When it stops holding, RFC 3339 UTC: stale after it on the reader's clock.
450    #[serde(default, skip_serializing_if = "Option::is_none")]
451    pub expires_at: Option<String>,
452    /// The writer's retry key: a write repeated with the same key and content is answered
453    /// with the first.
454    #[serde(default, skip_serializing_if = "Option::is_none")]
455    pub idempotency: Option<String>,
456    /// When it was written, RFC 3339, as its author's clock said; informational.
457    #[serde(default, skip_serializing_if = "Option::is_none")]
458    pub written_at: Option<String>,
459}

461impl RecordBody {
462    /// A first revision: open, citing nothing, superseding nothing, with no lease. The
463    /// constructor of a test or a script; admission builds what a session writes.
464    ///
465    /// ```
466    /// use majordomus_cli::mesh::records::{RecordBody, RecordKind, RecordStatus};
467    /// let r = RecordBody::new("r-1", "s1", RecordKind::Plan, "I1", &["docs"], "write it");
468    /// assert_eq!((r.revision, r.status), (1, RecordStatus::Open));
469    /// assert!(r.validate().is_ok());
470    /// ```
471    pub fn new(
472        id: &str,
473        session: &str,
474        kind: RecordKind,
475        task: &str,
476        scope: &[&str],
477        title: &str,
478    ) -> Self {
479        RecordBody {
480            id: id.into(),
481            session: session.into(),
482            revision: 1,
483            kind,
484            task: task.into(),
485            scope: scope.iter().map(|p| p.to_string()).collect(),
486            status: RecordStatus::Open,
487            title: title.into(),
488            body: None,
489            sources: Vec::new(),
490            handoff: None,
491            supersedes: None,
492            expires_at: None,
493            idempotency: None,
494            written_at: None,
495        }
496    }
497
498    /// Check every bound; run on the writer before signing and on every reader at ingest,
499    /// so that nothing a peer sends is stored unchecked and nothing signed here is refused
500    /// there. A handoff must carry its parts, and only a handoff may.
501    ///
502    /// ```
503    /// use majordomus_cli::mesh::records::{RecordBody, RecordKind};
504    /// let r = RecordBody::new("r-1", "s1", RecordKind::Handoff, "I1", &["docs"], "over");
505    /// assert!(r.validate().unwrap_err().contains("handoff carries"));
506    /// let escaping = RecordBody::new("r-1", "s1", RecordKind::Plan, "I1", &["../etc"], "x");
507    /// assert!(escaping.validate().is_err(), "a scope is repository-relative");
508    /// ```
509    pub fn validate(&self) -> Result<(), String> {
510        ident("record", &self.id)?;
511        ident("session", &self.session)?;
512        if self.revision == 0 {
513            return Err("a record's revision starts at 1".into());
514        }
515        if self.task.trim().is_empty() {
516            return Err("a record names its task, intent or issue".into());
517        }
518        line("task", &self.task, RECORD_TASK_MAX)?;
519        paths(&self.scope, true)?;
520        if self.title.trim().is_empty() {
521            return Err("a record has a title".into());
522        }
523        line("title", &self.title, RECORD_TITLE_MAX)?;
524        if let Some(body) = &self.body {
525            text("body", body, RECORD_BODY_MAX)?;
526        }
527        if self.sources.len() > MAX_RECORD_SOURCES {
528            return Err(format!(
529                "a record cites at most {MAX_RECORD_SOURCES} sources"
530            ));
531        }
532        for source in &self.sources {
533            source.validate()?;
534        }
535        match (self.kind, &self.handoff) {
536            (RecordKind::Handoff, Some(h)) => h.validate()?,
537            (RecordKind::Handoff, None) => return Err(
538                "a handoff carries objective, current_state, evidence, blockers and next_action"
539                    .into(),
540            ),
541            (_, Some(_)) => return Err("only a handoff carries handoff parts".into()),
542            (_, None) => {}
543        }
544        if let Some(key) = &self.supersedes {
545            record_key(key)?;
546        }
547        if let Some(at) = &self.expires_at {
548            rfc3339("expires_at", at)?;
549        }
550        if let Some(key) = &self.idempotency {
551            ident("idempotency", key)?;
552        }
553        if let Some(at) = &self.written_at {
554            line("written_at", at, 64)?;
555        }
556        let size = serde_json::to_vec(self)
557            .map(|v| v.len())
558            .unwrap_or(usize::MAX);
559        if size > RECORD_MAX_BYTES {
560            return Err(format!(
561                "a record of {size} bytes exceeds {RECORD_MAX_BYTES}"
562            ));
563        }
564        Ok(())
565    }

567    /// The digest of what the record says, leaving out who wrote it, which revision and
568    /// when: two writes with equal digests say the same thing. An idempotent retry is
569    /// recognised by it.
570    ///
571    /// ```
572    /// use majordomus_cli::mesh::records::{RecordBody, RecordKind};
573    /// let a = RecordBody::new("r-1", "s1", RecordKind::Plan, "I1", &["docs"], "write it");
574    /// let b = RecordBody { id: "r-2".into(), session: "s2".into(), revision: 3, ..a.clone() };
575    /// assert_eq!(a.content_digest(), b.content_digest(), "who and which revision aside");
576    /// let c = RecordBody { title: "write less".into(), ..a.clone() };
577    /// assert_ne!(a.content_digest(), c.content_digest());
578    /// ```
579    pub fn content_digest(&self) -> String {
580        use sha2::{Digest, Sha256};
581        // what it says, and nothing of who wrote it, which revision or when
582        let value = serde_json::json!({
583            "kind": self.kind,
584            "task": self.task,
585            "scope": self.scope,
586            "status": self.status,
587            "title": self.title,
588            "body": self.body,
589            "sources": self.sources,
590            "handoff": self.handoff,
591            "supersedes": self.supersedes,
592            "expires_at": self.expires_at,
593        });
594        let hash = Sha256::digest(canonical_json(&value));
595        hash.iter().take(16).map(|b| format!("{b:02x}")).collect()
596    }

908    created: BTreeMap<String, u64>,

912impl RecordFold {
913    /// Apply one `record_written` event. `closed_at` is the fold's map of session closes so
914    /// far: a revision after its author's session closed is not the author's.
915    pub(crate) fn apply(
916        &mut self,
917        event: &MeshEvent,
918        record: RecordBody,
919        closed_at: &BTreeMap<String, u64>,
920    ) {
921        let stream = &event.stream;
922        let key = format!("{stream}/{}", record.id);
923        let author = format!("{stream}/{}", record.session);
924        let revision = RecordRevision {
925            revision: record.revision,
926            event: event.id(),
927            lamport: event.lamport,
928            status: record.status,
929            title: record.title.clone(),
930            sources: record.sources.clone(),
931            evidence: record.evidence(),
932            digest: record.content_digest(),
933            idempotency: record.idempotency.clone(),
934            written_at: record.written_at.clone(),
935        };
936        let mut ignore = |reason: String| {
937            self.ignored.push(IgnoredRecordWrite {
938                event: event.id(),
939                record: key.clone(),
940                reason,
941            })
942        };
943        match self.records.get_mut(&key) {
944            None if record.revision == 1 => {
945                self.created.insert(key.clone(), event.lamport);
946                self.records.insert(
947                    key.clone(),
948                    RecordView {
949                        key,
950                        author,
951                        runtime: stream.runtime_key(),
952                        repository: event.repo.clone(),
953                        kind: record.kind,
954                        task: record.task,
955                        scope: record.scope,
956                        status: record.status,
957                        title: record.title,
958                        body: record.body,
959                        sources: record.sources,
960                        handoff: record.handoff,
961                        evidence: revision.evidence,
962                        revision: 1,
963                        expires_at: record.expires_at,
964                        supersedes: record.supersedes,
965                        superseded_by: Vec::new(),
966                        author_state: AuthorState::Unknown,
967                        stale: Vec::new(),
968                        history: vec![revision],
969                    },
970                );
971            }
972            None => ignore(format!(
973                "revision {} of a record this runtime never saw created",
974                record.revision
975            )),
976            Some(_) if record.revision == 1 => ignore("a second creation of one record".into()),
977            Some(view) => {
978                if view.author != author {
979                    ignore(format!(
980                        "written by {author}, and only its author {} revises it",
981                        view.author
982                    ));
983                } else if record.revision != view.revision + 1 {
984                    ignore(format!(
985                        "revision {} does not follow revision {}",
986                        record.revision, view.revision
987                    ));
988                } else if view.kind != record.kind {
989                    ignore(format!(
990                        "a {} cannot be revised into a {}",
991                        view.kind.as_str(),
992                        record.kind.as_str()
993                    ));
994                } else if closed_at
995                    .get(&author)
996                    .is_some_and(|closed| self.created.get(&view.key).is_some_and(|c| c < closed))
997                {
998                    ignore("its author's session closed after creating it".into());
999                } else {
1000                    view.revision = record.revision;
1001                    view.task = record.task;
1002                    view.scope = record.scope;
1003                    view.status = record.status;
1004                    view.title = record.title;
1005                    view.body = record.body;
1006                    view.sources = record.sources;
1007                    view.handoff = record.handoff;
1008                    view.evidence = revision.evidence;
1009                    view.expires_at = record.expires_at;
1010                    view.supersedes = record.supersedes;
1011                    view.history.push(revision);
1012                }
1013            }
1014        }
1015    }

1017    /// The records with their links, their authors' standing and what contends, and the
1018    /// writes the fold ignored.
1019    pub(crate) fn finish(
1020        mut self,
1021        sessions: &BTreeMap<String, SessionView>,
1022    ) -> (
1023        Vec<RecordView>,
1024        Vec<RecordConflict>,
1025        Vec<IgnoredRecordWrite>,
1026    ) {
1027        let links: Vec<(String, String)> = self
1028            .records
1029            .values()
1030            .filter_map(|r| r.supersedes.clone().map(|old| (old, r.key.clone())))
1031            .collect();
1032        for (old, new) in links {
1033            if old == new {
1034                continue;
1035            }
1036            // each record names one predecessor, so a link is pushed once; a predecessor not
1037            // heard here (yet) is a dangling link the superseding record still shows
1038            if let Some(view) = self.records.get_mut(&old) {
1039                view.superseded_by.push(new);
1040            }
1041        }
1042        for view in self.records.values_mut() {
1043            view.author_state = match sessions.get(&view.author).map(|s| s.state) {
1044                Some(SessionState::Active) => AuthorState::Active,
1045                Some(SessionState::Closed) => AuthorState::Closed,
1046                Some(SessionState::Expired) => AuthorState::Expired,
1047                None => AuthorState::Unknown,
1048            };
1049            match view.author_state {
1050                AuthorState::Active => {}
1051                AuthorState::Closed => view.stale.push("its author's session closed".into()),
1052                AuthorState::Expired => view
1053                    .stale
1054                    .push("its author's runtime stopped beating".into()),
1055                AuthorState::Unknown => view
1056                    .stale
1057                    .push("its author's session is not known here".into()),
1058            }
1059        }
1060        let records: Vec<RecordView> = self.records.into_values().collect();
1061        let conflicts = conflicts(&records);
1062        (records, conflicts, self.ignored)
1063    }
1064}

1066/// Mark every record whose expiry has passed at `now` (seconds since the epoch) stale. A
1067/// reading of the reader's clock, like a stream's liveness; nothing is removed.
1068pub(crate) fn judge_leases(records: &mut [RecordView], now: u64) -> bool {
1069    let mut changed = false;
1070    for view in records.iter_mut() {
1071        let Some(at) = view.expires_at.as_deref() else {
1072            continue;
1073        };
1074        if crate::peers::parse_rfc3339(at).is_some_and(|expiry| expiry <= now) {
1075            let why = format!("its lease expired at {at}");
1076            if !view.stale.contains(&why) {
1077                view.stale.push(why);
1078                changed = true;
1079            }
1080        }
1081    }
1082    changed
1083}

1085/// The contending pairs among live records, in key order.
1086pub(crate) fn conflicts(records: &[RecordView]) -> Vec<RecordConflict> {
1087    let live: Vec<&RecordView> = records
1088        .iter()
1089        .filter(|r| r.is_live() && r.kind.contends())
1090        .collect();
1091    let mut out = Vec::new();
1092    for (i, first) in live.iter().enumerate() {
1093        for second in &live[i + 1..] {
1094            if first.kind != second.kind || first.author == second.author {
1095                continue;
1096            }
1097            let paths = meeting(&first.scope, &second.scope);
1098            let reason = if first.task == second.task {
1099                ConflictReason::SameTask
1100            } else if !paths.is_empty() {
1101                ConflictReason::OverlappingScope
1102            } else {
1103                continue;
1104            };
1105            out.push(RecordConflict {
1106                kind: first.kind,
1107                first: first.key.clone(),
1108                second: second.key.clone(),
1109                reason,
1110                paths,
1111            });
1112        }
1113    }
1114    out
1115}

1118    let mut out = Vec::new();
1119    for x in a {
1120        for y in b {
1121            if crate::peers::claims_meet(x, y) {
1122                out.push((x.clone(), y.clone()));
1123            }
1124        }
1125    }

1139    let (p, c) = (
1140        path.trim().trim_end_matches('/'),
1141        claimed.trim().trim_end_matches('/'),
1142    );
1143    !p.is_empty() && !c.is_empty() && (p == c || p.starts_with(&format!("{c}/")))
1144}

1210/// Why a write was refused: every refusal fails closed and names what to do.
1211#[derive(Debug, Clone, PartialEq, Eq)]
1212pub(crate) enum RecordRefusal {

1214    Invalid(String),
1215    /// A path lies outside the writing session's live claims.
1216    OutOfScope(String),
1217    /// The record is another session's.
1218    NotOwn(String),
1219    /// The revision read is not the current one.
1220    Stale {
1221        /// The record.
1222        record: String,
1223        /// Its current revision.
1224        current: u32,
1225        /// Why.
1226        why: String,
1227    },
1228    /// The record or the one superseded is not known here.
1229    NotFound(String),
1230    /// The retry key was used for other content.
1231    IdempotencyConflict(String),
1232}

1238    Append(Box<RecordBody>),

1242        key: String,
1243        /// The revision first written.
1244        revision: RecordRevision,
1245    },

1248/// Decide a write of `session` of the runtime whose stream is `stream` against the folded
1249/// `records` and `claims`, with `fresh_id` the id a new record takes. Pure: the caller holds
1250/// the admission lock and appends what comes back.
1251pub(crate) fn admit(
1252    records: &[RecordView],
1253    claims: &[ClaimView],
1254    stream: &StreamId,
1255    session: &str,
1256    write: RecordWrite,
1257    fresh_id: &str,
1258    written_at: String,
1259) -> Result<Admission, RecordRefusal> {
1260    let author = format!("{stream}/{session}");
1261    let Some(kind) = write.kind else {
1262        return Err(RecordRefusal::Invalid(
1263            "a record names its kind: observation, hypothesis, plan, decision, blocker, handoff or outcome".into(),
1264        ));
1265    };
1266    let target = match &write.record {
1267        Some(key) => Some(
1268            records
1269                .iter()
1270                .find(|r| &r.key == key)
1271                .ok_or_else(|| RecordRefusal::NotFound(format!("no record {key} here")))?,
1272        ),
1273        None => None,
1274    };
1275    let (id, revision) = match target {
1276        Some(view) => (
1277            view.key.rsplit('/').next().unwrap_or_default().to_string(),
1278            view.revision + 1,
1279        ),
1280        None => (fresh_id.to_string(), 1),
1281    };
1282    let body = RecordBody {
1283        id,
1284        session: session.to_string(),
1285        revision,
1286        kind,
1287        task: write.task,
1288        scope: write.scope,
1289        status: write.status.unwrap_or_default(),
1290        title: write.title,
1291        body: write.body,
1292        sources: write.sources,
1293        handoff: write.handoff,
1294        supersedes: write.supersedes,
1295        expires_at: write.expires_at,
1296        idempotency: write.idempotency,
1297        written_at: Some(written_at),
1298    };
1299    body.validate().map_err(RecordRefusal::Invalid)?;
1300
1301    // a retry: the same key from the same author answers what it answered the first time
1302    if let Some(retry) = &body.idempotency {
1303        let digest = body.content_digest();
1304        let earlier = records
1305            .iter()
1306            .filter(|r| r.author == author)
1307            .flat_map(|r| r.history.iter().map(move |h| (r, h)))
1308            .find(|(_, h)| h.idempotency.as_deref() == Some(retry.as_str()));
1309        if let Some((view, first)) = earlier {
1310            let same_target = write.record.as_deref().is_none_or(|k| k == view.key);
1311            if first.digest == digest && same_target {
1312                return Ok(Admission::Repeat {
1313                    key: view.key.clone(),
1314                    revision: first.clone(),
1315                });
1316            }
1317            return Err(RecordRefusal::IdempotencyConflict(format!(
1318                "retry key '{retry}' already wrote revision {} of {} with other content",
1319                first.revision, view.key
1320            )));
1321        }
1322    }
1323
1324    if let Some(view) = target {
1325        if view.author != author {
1326            return Err(RecordRefusal::NotOwn(format!(
1327                "{} is {}'s record; only its author revises it — write a new record with supersedes: {} instead",
1328                view.key, view.author, view.key
1329            )));
1330        }
1331        let Some(read) = write.revision else {
1332            return Err(RecordRefusal::Invalid(format!(
1333                "a revision names the revision it read: revision {} is current",
1334                view.revision
1335            )));
1336        };
1337        if read != view.revision {
1338            return Err(RecordRefusal::Stale {
1339                record: view.key.clone(),
1340                current: view.revision,
1341                why: format!("revision {read} was read and {} is current", view.revision),
1342            });
1343        }
1344        if !view.superseded_by.is_empty() {
1345            return Err(RecordRefusal::Stale {
1346                record: view.key.clone(),
1347                current: view.revision,
1348                why: format!("it is superseded by {}", view.superseded_by.join(", ")),
1349            });
1350        }
1351        if view.kind != kind {
1352            return Err(RecordRefusal::Invalid(format!(
1353                "a {} cannot be revised into a {}; write a new record",
1354                view.kind.as_str(),
1355                kind.as_str()
1356            )));
1357        }
1358        if view.author_state != AuthorState::Active {
1359            return Err(RecordRefusal::NotOwn(format!(
1360                "{}'s author session is no longer active; supersede it",
1361                view.key
1362            )));
1363        }
1364    } else if write.revision.is_some() {
1365        return Err(RecordRefusal::Invalid(
1366            "a revision is given only with the record it revises".into(),
1367        ));
1368    }
1369
1370    if let Some(old) = &body.supersedes {
1371        if target.is_some_and(|t| &t.key == old) || !records.iter().any(|r| &r.key == old) {
1372            return Err(RecordRefusal::NotFound(format!(
1373                "supersedes {old}, which is not another record known here"
1374            )));
1375        }
1376    }
1377
1378    // the scope: every path inside a live claim of this very session
1379    let held: Vec<&ClaimView> = claims
1380        .iter()
1381        .filter(|c| c.session == author && c.state.is_live())
1382        .collect();
1383    let outside: Vec<&String> = body
1384        .scope
1385        .iter()
1386        .filter(|p| !held.iter().any(|c| c.scope.iter().any(|s| within(p, s))))
1387        .collect();
1388    if !outside.is_empty() {
1389        return Err(RecordRefusal::OutOfScope(format!(
1390            "{} lie outside every live claim of session {author}; claim them first (mesh.claim, or announce them)",
1391            outside.iter().map(|p| p.as_str()).collect::<Vec<_>>().join(", ")
1392        )));
1393    }
1394    Ok(Admission::Append(Box::new(body)))
1395}

1498/// The records of `records` the query admits, with the conflicts that name one of them and
1499/// the ignored writes about them (all of them when nothing narrows), under the whole fold's
1500/// digest.
1501///
1502/// ```
1503/// use std::sync::Arc;
1504/// use majordomus_cli::mesh::identity::NodeIdentity;
1505/// use majordomus_cli::mesh::journal::{EventBody, Journal, StreamLiveness};
1506/// use majordomus_cli::mesh::records::{select, RecordBody, RecordKind, RecordQuery};
1507/// use majordomus_cli::mesh::state::fold;
1508/// let j = Journal::open(Arc::new(NodeIdentity::ephemeral().unwrap()), "0000000000000001",
1509///     "repo".into(), None).unwrap();
1510/// for (id, kind) in [("r-1", RecordKind::Plan), ("r-2", RecordKind::Blocker)] {
1511///     j.append_own(EventBody::RecordWritten { record:
1512///         RecordBody::new(id, "s1", kind, "I1", &["docs"], id) }).unwrap();
1513/// }
1514/// let s = fold(&j.events(), &|_| StreamLiveness::Own);
1515/// let q = RecordQuery { kind: Some(RecordKind::Blocker), ..RecordQuery::default() };
1516/// let list = select(&s.records, &s.record_conflicts, &s.records_ignored, &s.digest, &q);
1517/// assert_eq!(list.records.len(), 1);
1518/// assert_eq!(list.digest, s.digest, "the whole fold's");
1519/// ```
1520pub fn select(
1521    records: &[RecordView],
1522    conflicts: &[RecordConflict],
1523    ignored: &[IgnoredRecordWrite],
1524    digest: &str,
1525    query: &RecordQuery,
1526) -> RecordList {
1527    let admitted: Vec<RecordView> = records
1528        .iter()
1529        .filter(|r| query.record.as_deref().is_none_or(|k| r.key == k))
1530        .filter(|r| query.task.as_deref().is_none_or(|t| r.task == t))
1531        .filter(|r| query.kind.is_none_or(|k| r.kind == k))
1532        .filter(|r| {
1533            query.author.as_deref().is_none_or(|a| {
1534                r.author == a || r.author.rsplit('/').next().is_some_and(|s| s == a)
1535            })
1536        })
1537        .filter(|r| {
1538            query
1539                .path
1540                .as_deref()
1541                .is_none_or(|p| r.scope.iter().any(|s| crate::peers::claims_meet(s, p)))
1542        })
1543        .filter(|r| query.live.is_none_or(|live| r.is_live() == live))
1544        .cloned()
1545        .collect();
1546    let keys: std::collections::BTreeSet<&str> = admitted.iter().map(|r| r.key.as_str()).collect();
1547    let narrowed = *query != RecordQuery::default();
1548    RecordList {
1549        active: true,
1550        reason: None,
1551        digest: digest.to_string(),
1552        conflicts: conflicts
1553            .iter()
1554            .filter(|c| keys.contains(c.first.as_str()) || keys.contains(c.second.as_str()))
1555            .cloned()
1556            .collect(),
1557        ignored: ignored
1558            .iter()
1559            .filter(|i| !narrowed || keys.contains(i.record.as_str()))
1560            .cloned()
1561            .collect(),
1562        records: admitted,
1563    }
1564}
