1//! The executor: one merge at a time, each against a master observed a moment before.
2//!
3//! # The invariant, structurally
4//!
5//! [`step`] takes no plan. It observes the forge, builds the queue, picks the first ready
6//! pull request, observes the forge *again*, rebuilds the queue, and merges only if the
7//! second decision names the same master and head as the first and still says `ready`.
8//! [`drain`] is a loop over `step` and nothing else, so no candidate list survives a merge:
9//! there is no variable for it to survive in.
10//!
11//! # What it will not do
12//!
13//! It never passes `--admin`, never merges a pull request whose required checks are not
14//! all passed on its current head, never merges one that does not contain the current
15//! master, never rewrites a branch — the one push it makes is a fast-forward of the head it
16//! observed, leased on that head — and never closes a pull request — closure is
17//! [`super::drain::cleanup`]'s, which demands stronger evidence and an explicit `--apply`.
18//! Only one executor per base branch runs at a time ([`IntegrationLease`]).
19//!
20//! # The trail comes first
21//!
22//! Every act is appended to the repository's audit trail ([`record`]) before it is taken: a
23//! merge after `merge_attempted`, a refresh push after `refresh_attempted`, a closure after
24//! `close_attempted`. When that line cannot be written, the act is not taken — the step says
25//! [`DrainStepOutcome::TrailUnwritable`] and the drain stops — because an act the trail
26//! cannot name is one nobody can audit afterwards. The trail is one file under the common
27//! git directory ([`super::events_path`]), so every worktree of the repository writes and
28//! reads the same one.

52/// still holds the refresh pipeline, counted from the `refreshed` event. A forge creates no
53/// check run for an aggregate job until every job it needs has finished, so for most of a CI

60/// The one executor of a base branch: an exclusive file under the *common* git directory,
61/// so every worktree of the repository contends on the same file and no other repository
62/// sees it. Read-only observers never take it.
63///
64/// Holding it is holding an exclusive `flock` on that file's descriptor, for the life of
65/// this value. The kernel decides who holds it, so executors started at the same instant
66/// cannot both win, and a holder that ends — even by a crash — releases it at once. The
67/// holder record written into the file says who holds it; it decides nothing.
68pub struct IntegrationLease {
69    path: PathBuf,
70    token: String,
71    /// The repository whose trail records the lease's release.
72    root: PathBuf,
73    /// The open lock file: closing it gives the `flock` back.
74    file: fs::File,
75    /// The repository-wide claim taken with it, released after the lease
76    /// ([`super::exclusive`]); `None` where only this clone is guarded.
77    mesh: Option<super::exclusive::MeshClaim>,
78}

130/// An exclusive `flock` on `file`, without waiting: `false` when another descriptor holds it.
131/// Released when the file is closed.
132fn try_lock_exclusive(file: &fs::File) -> std::io::Result<bool> {
133    use std::os::unix::io::AsRawFd;
134    // SAFETY: the descriptor belongs to `file`, which is open for the whole call.
135    if unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX | libc::LOCK_NB) } == 0 {
136        return Ok(true);
137    }
138    // held by another descriptor is an answer; anything else is an error
139    let e = std::io::Error::last_os_error();
140    (e.kind() == std::io::ErrorKind::WouldBlock)
141        .then_some(false)
142        .ok_or(e)
143}

145/// Whether `path` still names the file `file` has open.
146fn names_file(path: &Path, file: &fs::File) -> bool {
147    use std::os::unix::fs::MetadataExt;
148    match (fs::metadata(path), file.metadata()) {
149        (Ok(a), Ok(b)) => a.dev() == b.dev() && a.ino() == b.ino(),
150        _ => false,
151    }
152}

154/// How often a kept-alive lease renews its record: well inside [`LEASE_STALE_AFTER`].
155pub const KEEP_ALIVE_EVERY: Duration = Duration::from_secs(60);
156/// How promptly a keep-alive notices it was asked to stop.
157const KEEP_ALIVE_POLL: Duration = Duration::from_millis(200);
158
159/// Replace the holder record through the locked file, never the path: a path written to
160/// after something removed it would be a new, unlocked file a second executor could take.
161fn write_record(file: &fs::File, token: &str) -> std::io::Result<()> {
162    use std::os::unix::fs::FileExt;
163    // a write like any of the trail's, and like them it can fail (tests walk each failure)
164    if let Some(why) = injected_failure() {
165        return Err(std::io::Error::other(why));
166    }
167    file.set_len(0)
168        .and_then(|()| file.write_all_at(token.as_bytes(), 0))
169}

171/// A lease's record kept fresh by a thread, until this is dropped.
172pub struct LeaseKeepAlive {
173    stop: std::sync::Arc<std::sync::atomic::AtomicBool>,
174    thread: Option<std::thread::JoinHandle<()>>,
175}

177impl Drop for LeaseKeepAlive {
178    fn drop(&mut self) {
179        self.stop.store(true, std::sync::atomic::Ordering::SeqCst);
180        let _ = self.thread.take().map(std::thread::JoinHandle::join);
181    }
182}

185    /// The lease file of a base branch.
186    pub fn path_for(common: &Path, base: &str) -> PathBuf {
187        common
188            .join("majordomus/locks")
189            .join(format!("integration-{}.lock", base.replace('/', "-")))

192    /// Take the lease, or say who holds it. Another executor that holds the `flock` is a
193    /// refusal, however old its record; a record whose holder no longer holds the lock — it
194    /// stopped without releasing — is taken over at once. Taking it is recorded
195    /// (`lease_acquired`); a lease the trail cannot record is given back at once and refused.
196    pub fn acquire(root: &Path, base: &str) -> Result<Self, String> {
197        Self::acquire_holding(root, base, None)
198    }
199
200    /// [`acquire`](Self::acquire), holding `mesh` — the claim taken for it across machines —
201    /// which its record names and which is released after the lease. A lease refused here
202    /// drops the claim, which releases it.
203    pub(crate) fn acquire_holding(
204        root: &Path,
205        base: &str,
206        mesh: Option<super::exclusive::MeshClaim>,
207    ) -> Result<Self, String> {
208        let path = Self::path_for(&common_dir(root)?, base);
209        if let Some(dir) = path.parent() {
210            fs::create_dir_all(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
211        }
212        let holder = LeaseHolder {
213            pid: std::process::id(),
214            host: host(),
215            base: base.to_string(),
216            since: now_secs(),
217            mesh_claim: mesh.as_ref().map(|m| m.key().to_string()),
218        };
219        let token = serde_json::to_string(&holder).map_err(|e| e.to_string())?;
220        // A holder that releases unlinks the file before it closes it, so an executor that
221        // opened the old file in between locks a file nobody can find any more: it must see
222        // that the path still names the file it locked, or open the path again.
223        for _ in 0..8 {
224            let (file, locked) = fs::OpenOptions::new()
225                .read(true)
226                .write(true)
227                .create(true)
228                .truncate(false)
229                .open(&path)
230                .and_then(|file| try_lock_exclusive(&file).map(|locked| (file, locked)))
231                .map_err(|e| format!("{}: {e}", path.display()))?;
232            if !locked {
233                let held = fs::read_to_string(&path).unwrap_or_default();
234                return Err(format!(
235                    "another integration executor holds {}: {}",
236                    path.display(),
237                    held.trim()
238                ));
239            }
240            // a path that names another file by now is opened again
241            if names_file(&path, &file) {
242                let taken = IntegrationEvent {
243                    detail: token.clone(),
244                    ..IntegrationEvent::of(IntegrationAction::LeaseAcquired)
245                };
246                let kept = write_record(&file, &token)
247                    .map_err(|e| format!("its holder record could not be written: {e}"))
248                    .and_then(|()| {
249                        record(root, taken)
250                            .map_err(|e| format!("the trail could not record it: {e}"))
251                    });
252                if let Err(e) = kept {
253                    // unlinked while still locked, so nobody takes the record this leaves
254                    let _ = fs::remove_file(&path);
255                    return Err(format!("the integration lease was given back: {e}"));
256                }
257                return Ok(IntegrationLease {
258                    path,
259                    token,
260                    root: root.to_path_buf(),
261                    file,
262                    mesh,
263                });
264            }
265        }
266        Err(format!("{}: could not be taken", path.display()))
267    }

269    /// Mark the lease alive: a long drain renews it between steps. Refused when the lease is
270    /// no longer this executor's — the path names another file, or the record another
271    /// holder — and the drain stops on that, as on any systemic failure: an executor that
272    /// lost its lease must not act as if it held one.
273    pub fn renew(&self) -> Result<(), String> {
274        if !names_file(&self.path, &self.file) {
275            return Err(format!(
276                "the integration lease was lost: {} no longer names the file this executor locked",
277                self.path.display()
278            ));
279        }
280        let held = fs::read_to_string(&self.path).unwrap_or_default();
281        if held.trim() != self.token {
282            return Err(format!(
283                "the integration lease was lost: {} names another holder: {}",
284                self.path.display(),
285                held.trim()
286            ));
287        }
288        write_record(&self.file, &self.token).map_err(|e| format!("{}: {e}", self.path.display()))
289    }
290
291    /// Keep the record fresh while one long act runs — a refresh's derive can outlast what an
292    /// observer calls stale — until the returned guard is dropped. It renews only while the
293    /// lease is still this executor's, and never acts on anything.
294    pub fn keep_alive(&self) -> LeaseKeepAlive {
295        self.keep_alive_every(KEEP_ALIVE_EVERY)
296    }
297
298    /// [`keep_alive`](Self::keep_alive) at a chosen interval.
299    pub(crate) fn keep_alive_every(&self, every: Duration) -> LeaseKeepAlive {
300        let poll = KEEP_ALIVE_POLL.min(every);
301        let stop = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
302        let thread = self.file.try_clone().ok().and_then(|file| {
303            let (path, token, stop) = (self.path.clone(), self.token.clone(), stop.clone());
304            std::thread::Builder::new()
305                .name("majordomus-integration-keep-alive".into())
306                .spawn(move || {
307                    use std::sync::atomic::Ordering;
308                    let mut waited = Duration::ZERO;
309                    while !stop.load(Ordering::SeqCst) {
310                        std::thread::sleep(poll);
311                        waited += poll;
312                        if waited < every {
313                            continue;
314                        }
315                        waited = Duration::ZERO;
316                        let mine = names_file(&path, &file)
317                            && fs::read_to_string(&path).is_ok_and(|c| c.trim() == token);
318                        if !mine {
319                            return;
320                        }
321                        let _ = write_record(&file, &token);
322                    }
323                })
324                .ok()
325        });
326        LeaseKeepAlive { stop, thread }
327    }

329    /// The repository-wide claim this lease holds, when it holds one.
330    pub fn mesh_claim(&self) -> Option<&super::exclusive::MeshClaim> {
331        self.mesh.as_ref()
332    }

334    /// Who holds the lease of `base` now, read without taking it: what an observer shows.
335    /// `None` when nobody does.
336    pub fn read(root: &Path, base: &str) -> Result<Option<IntegrationLeaseState>, String> {
337        let path = Self::path_for(&common_dir(root)?, base);
338        let Ok(text) = fs::read_to_string(&path) else {
339            return Ok(None);
340        };
341        let age = fs::metadata(&path)
342            .and_then(|m| m.modified())
343            .ok()
344            .and_then(|m| SystemTime::now().duration_since(m).ok())
345            .unwrap_or_default();
346        Ok(Some(IntegrationLeaseState {
347            holder: serde_json::from_str(text.trim()).ok(),
348            renewed_seconds_ago: age.as_secs(),
349            stale: age > LEASE_STALE_AFTER,
350        }))
351    }
352}

366impl Drop for IntegrationLease {
367    fn drop(&mut self) {
368        // unlinked while the lock is held, closed after: see `acquire`. Only while the path
369        // still names the file this lease locked: two holders' records can read the same
370        // (one process, one second), so the record alone cannot say whose file it is.
371        if names_file(&self.path, &self.file)
372            && fs::read_to_string(&self.path).is_ok_and(|c| c.trim() == self.token)
373        {
374            let _ = fs::remove_file(&self.path);
375            let released = IntegrationEvent {
376                detail: self.token.clone(),
377                ..IntegrationEvent::of(IntegrationAction::LeaseReleased)
378            };
379            if let Err(e) = record(&self.root, released) {
380                // a destructor cannot refuse; it says so where a person reads
381                eprintln!(
382                    "majordomus: the integration lease was released, but the trail could not record it: {e}"
383                );
384            }
385        }
386    }
387}

503/// Why an act failed, as one class a person and a retry policy can act on. Every failed act
504/// on the trail carries one (`class`), and the drain decides from it whether the next
505/// candidate may be tried ([`FailureClass::recoverable`]) or the drain must stop.
506#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
507#[serde(rename_all = "snake_case")]
508pub enum FailureClass {
509    /// Master or the head moved since the decision.
510    Stale,
511    /// The merge of master conflicts.
512    Conflict,
513    /// A required check failed that had passed.
514    NewFailingCheck,
515    /// An approval was withdrawn.
516    ReviewRevoked,
517    /// The forge or the network failed in a way that passes.
518    Transient,
519    /// The branch protection refused it.
520    PolicyViolation,
521    /// What landed could not be verified.
522    VerificationFailed,
523    /// The forge or git could not be read.
524    Unreadable,
525}
526
527impl std::fmt::Display for FailureClass {
528    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
529        // the wire word, as the trail writes it
530        f.write_str(&super::classify::word(self))
531    }
532}

534impl FailureClass {
535    /// Whether the failure is this candidate's, so the drain may go on with the next one;
536    /// otherwise it is the repository's or the executor's, and the drain stops.
537    pub fn recoverable(self) -> bool {
538        !matches!(
539            self,
540            FailureClass::VerificationFailed | FailureClass::Unreadable
541        )
542    }
543
544    /// The class of a merge the forge refused, from its own words.
545    pub fn of_merge_refusal(reason: &str) -> FailureClass {
546        let r = reason.to_ascii_lowercase();
547        if super::retry::transient(reason) {
548            FailureClass::Transient
549        } else if r.contains("conflict") {
550            FailureClass::Conflict
551        } else if r.contains("head branch was modified") || r.contains("head sha") {
552            FailureClass::Stale
553        } else if r.contains("review") {
554            FailureClass::ReviewRevoked
555        } else if r.contains("check") && (r.contains("fail") || r.contains("expected")) {
556            FailureClass::NewFailingCheck
557        } else {
558            // a protection rule, a missing permission, a setting: the forge's policy said no
559            FailureClass::PolicyViolation
560        }
561    }
562
563    /// The class of a refresh that failed, from its words.
564    pub fn of_refresh_failure(reason: &str) -> FailureClass {
565        let r = reason.to_ascii_lowercase();
566        if super::retry::transient(reason) {
567            FailureClass::Transient
568        } else if r.contains("conflict") {
569            FailureClass::Conflict
570        } else if r.contains("rejected")
571            || r.contains("fetch first")
572            || r.contains("non-fast-forward")
573            || r.contains("moved since it was observed")
574        {
575            FailureClass::Stale
576        } else {
577            FailureClass::PolicyViolation
578        }
579    }
580}

582/// One entry of the audit trail.
583#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
584pub struct IntegrationEvent {

662/// Append one event to the repository's audit trail, append-only JSON lines. The event as
663/// written, or why it could not be: a caller about to act on the forge does not act then.
664pub fn record(root: &Path, mut event: IntegrationEvent) -> Result<IntegrationEvent, String> {
665    if event.at.is_empty() {
666        event.at = crate::peers::rfc3339(SystemTime::now());
667    }
668    if event.actor.is_empty() {
669        event.actor = actor();
670    }
671    // The detail is as often relayed as composed: the standard error of a `git` or a `gh`
672    // that failed, naming the remote in whatever form it was configured. The trail is
673    // served, so a credential in that URL is removed here, at the one place every event
674    // passes, rather than trusted to have been anonymised by the tool that printed it.
675    event.detail = crate::session::public_text(&event.detail);
676    if event.class.is_none() {
677        // every failed act names its class, decided once, here, from what it says
678        event.class = match event.action {
679            IntegrationAction::StaleDecision => Some(FailureClass::Stale),
680            IntegrationAction::MergeFailed => Some(FailureClass::of_merge_refusal(&event.detail)),
681            IntegrationAction::RefreshFailed | IntegrationAction::RepairRefused => {
682                Some(FailureClass::of_refresh_failure(&event.detail))
683            }
684            IntegrationAction::VerificationFailed => Some(FailureClass::VerificationFailed),
685            IntegrationAction::ObserveFailed => Some(FailureClass::Transient),
686            _ => None,
687        };
688    }
689    if let Some(why) = injected_failure() {
690        return Err(why);
691    }
692    let path = events_path(root)?;
693    let dir = path.parent().unwrap_or(&path);
694    fs::create_dir_all(dir)
695        .and_then(|()| serde_json::to_string(&event).map_err(std::io::Error::other))
696        .and_then(|line| {
697            fs::OpenOptions::new()
698                .create(true)
699                .append(true)
700                .open(&path)
701                .and_then(|mut f| writeln!(f, "{line}"))
702        })
703        .map_err(at(&path))?;
704    Ok(event)
705}

773/// What one step did.
774#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
775#[serde(rename_all = "snake_case", tag = "outcome")]
776pub enum DrainStepOutcome {
777    /// Nothing is ready.
778    Idle {
779        /// Why the queue did not yield a merge.
780        why: String,
781    },
782    /// Dry run: this would have been merged.
783    WouldMerge {
784        /// The pull request.
785        pr: u64,
786    },
787    /// Dry run: master would have been brought into this pull request.
788    WouldRefresh {
789        /// The pull request.
790        pr: u64,
791    },
792    /// Master was brought into the branch; its required checks now run on the new head.
793    Refreshed {
794        /// The pull request.
795        pr: u64,
796        /// The head before.
797        head_before: String,
798        /// The head after.
799        head_after: String,
800    },
801    /// A refreshed pull request is waiting for its checks; refreshing another now would be
802    /// wasted, because merging the first re-stales it.
803    AwaitingChecks {
804        /// The pull request whose checks are running.
805        pr: u64,
806    },
807    /// Bringing master in failed. Recoverable classes let the drain go on: the queue holds
808    /// this candidate back (`executor_refresh_failed`) until its head or master moves.
809    RefreshFailed {
810        /// The pull request.
811        pr: u64,
812        /// Why.
813        reason: String,
814        /// Why, as a class.
815        class: FailureClass,
816    },
817    /// The decision went stale between planning and acting; nothing was merged.
818    StaleDecision {
819        /// The pull request.
820        pr: u64,
821        /// What moved.
822        what: String,
823    },
824    /// Merged and verified.
825    Merged {
826        /// The pull request.
827        pr: u64,
828        /// Master before.
829        master_before: String,
830        /// Master after.
831        master_after: String,
832    },
833    /// The forge refused the merge. Recoverable classes let the drain go on: the queue holds
834    /// this candidate back (`executor_merge_refused`) until its head or master moves.
835    MergeRefused {
836        /// The pull request.
837        pr: u64,
838        /// The forge's words.
839        reason: String,
840        /// The forge's words, as a class.
841        class: FailureClass,
842    },
843    /// Merged, but what landed could not be verified: the drain stops.
844    VerificationFailed {
845        /// The pull request.
846        pr: u64,
847        /// What did not hold.
848        reason: String,
849    },
850    /// Nothing is merged: a merge of an earlier drain could not be verified, and no drain
851    /// merges until a person has looked (`prs drain --resume-after-failure`).
852    Halted {

854        pr: Option<u64>,
855        /// What did not hold then.
856        reason: String,

858    /// The act was not taken, because the trail could not record it first: nothing reached

862        /// The pull request.
863        pr: u64,
864        /// The event that could not be written: `merge_attempted` or `refresh_attempted`.

866        /// Why the trail refused it.
867        reason: String,
868    },

899/// A source of queues and a merger. The command line uses the forge and git; the tests use
900/// a scripted repository, which is how the re-plan-after-every-merge property is proved
901/// without a network.
902pub trait Integrator {
903    /// Observe the forge now and build the queue against the master observed.
904    fn observe(&mut self) -> Result<IntegrationQueue, String>;
905    /// Merge one pull request, requiring the forge's head to still be `head_sha`.
906    fn merge(&mut self, pr: u64, head_sha: &str, method: &str) -> Result<(), String>;
907    /// After a merge, or after a merge whose answer was lost: whether the forge shows the
908    /// pull request merged, and that what landed is this merge — of `at.head_sha` onto
909    /// `at.master_sha` — and nothing else.
910    fn verify(&mut self, pr: u64, at: &EvaluatedAgainst, method: &str)
911        -> Result<Landed, NotLanded>;
912    /// Bring master into a pull request's branch — a merge commit with the derived driver
913    /// and a fresh derive, pushed as a fast-forward of the observed head, leased on it.
914    /// Returns the new head. Never a rewrite: the push is refused if the branch moved. The
915    /// executor's refresh (`prs drain --refresh`) and a person's repair (`prs repair`) are
916    /// this one act.
917    fn refresh_branch(&mut self, a: &PullRequestAssessment, base: &str) -> Result<String, String>;
918    /// Close a pull request with a comment saying why.
919    fn close(&mut self, pr: u64, head_sha: &str, comment: &str) -> Result<(), String>;
920}

922/// One integration step: observe, decide, observe again, act only on an unchanged
923/// decision, verify. `dry_run` stops after the first decision.
924pub fn step(
925    root: &Path,
926    integrator: &mut dyn Integrator,
927    dry_run: bool,
928    allow_refresh: bool,
929) -> Result<DrainStepOutcome, String> {
930    let first = integrator.observe()?;
931    if !dry_run {
932        // what became or stopped being actionable since the trail's last word: the wait
933        // of every pull request is folded from these, so a dry run leaves them out too
934        super::wait::record_transitions(root, &first)?;
935    }
936    let Some(candidate) = first.next_merge.and_then(|n| first.get(n).cloned()) else {
937        if allow_refresh {
938            if let Some(outcome) = refresh_step(root, integrator, &first, dry_run)? {
939                return Ok(outcome);
940            }
941        }
942        let why = if first.next_refresh.is_empty() {
943            format!(
944                "nothing is ready among {} open pull request(s)",
945                first.tallies.open
946            )
947        } else {
948            format!(
949                "nothing is ready; {} pull request(s) need master brought in first: {}",
950                first.next_refresh.len(),
951                first
952                    .next_refresh
953                    .iter()
954                    .map(|n| format!("#{n}"))
955                    .collect::<Vec<_>>()
956                    .join(" ")
957            )
958        };
959        if !dry_run {
960            record(root, event(IntegrationAction::Idle, None, why.clone()))?;
961        }
962        return Ok(DrainStepOutcome::Idle { why });
963    };
964    if dry_run {
965        return Ok(DrainStepOutcome::WouldMerge {
966            pr: candidate.number,
967        });
968    }
969    let mut selected = event(
970        IntegrationAction::Selected,
971        Some(&candidate),
972        "the first ready pull request",
973    );
974    selected.passed_over = others(&first, PullRequestDisposition::Ready, candidate.number);
975    record(root, selected)?;
976    // the decision is re-taken from a new observation; acting on the first one would be
977    // acting on a picture of the repository that may already be wrong
978    let second = integrator.observe()?;
979    let fresh = second.get(candidate.number);
980    let stale = match fresh {
981        None => Some("it is no longer open".to_string()),
982        Some(a) if a.evaluated_against.master_sha != candidate.evaluated_against.master_sha => {
983            Some(format!(
984                "master moved from {} to {}",
985                candidate.evaluated_against.master_sha, a.evaluated_against.master_sha
986            ))
987        }
988        Some(a) if a.evaluated_against.head_sha != candidate.evaluated_against.head_sha => {
989            Some(format!(
990                "its head moved from {} to {}",
991                candidate.evaluated_against.head_sha, a.evaluated_against.head_sha
992            ))
993        }
994        Some(a) if a.disposition != PullRequestDisposition::Ready => {
995            Some(format!("it is {} now", a.disposition.as_str()))
996        }
997        Some(_) => None,
998    };
999    if let Some(what) = stale {
1000        record(
1001            root,
1002            event(
1003                IntegrationAction::StaleDecision,
1004                Some(&candidate),
1005                what.clone(),
1006            ),
1007        )?;
1008        return Ok(DrainStepOutcome::StaleDecision {
1009            pr: candidate.number,
1010            what,
1011        });
1012    }
1013    // no merge commit allowed holds every pull request, so a ready one has one; never another
1014    let Some(method) = second.policy.merge_method.clone() else {
1015        let what = "the repository allows no merge commit".to_string();
1016        record(
1017            root,
1018            event(
1019                IntegrationAction::StaleDecision,
1020                Some(&candidate),
1021                what.clone(),
1022            ),
1023        )?;
1024        return Ok(DrainStepOutcome::StaleDecision {
1025            pr: candidate.number,
1026            what,
1027        });
1028    };
1029    // on the trail before it reaches the forge: a merge the trail cannot name is not asked for
1030    let attempted = event(
1031        IntegrationAction::MergeAttempted,
1032        Some(&candidate),
1033        format!("--{method}"),
1034    )
1035    .with_evidence(&candidate);
1036    if let Err(reason) = record(root, attempted) {
1037        return Ok(DrainStepOutcome::TrailUnwritable {
1038            pr: candidate.number,
1039            unrecorded: IntegrationAction::MergeAttempted,
1040            reason,
1041        });
1042    }
1043    let at = candidate.evaluated_against.clone();
1044    if let Err(reason) = integrator.merge(candidate.number, &at.head_sha, &method) {
1045        // A merge whose answer was lost may have landed: it is asked whether it did, never
1046        // asked to merge again (retry::forge never retries a merge).
1047        if super::retry::transient(&reason) {
1048            match integrator.verify(candidate.number, &at, &method) {
1049                Ok(landed) => {
1050                    return merged(
1051                        root,
1052                        &candidate,
1053                        landed,
1054                        format!("merged, although the forge's answer was lost: {reason}"),
1055                    )
1056                }
1057                Err(NotLanded::Unproved(why)) => return unverified(root, &candidate, why),
1058                Err(NotLanded::NotMerged(_)) => {}
1059            }
1060        }
1061        record(
1062            root,
1063            event(
1064                IntegrationAction::MergeFailed,
1065                Some(&candidate),
1066                reason.clone(),
1067            ),
1068        )?;
1069        let class = FailureClass::of_merge_refusal(&reason);
1070        return Ok(DrainStepOutcome::MergeRefused {
1071            pr: candidate.number,
1072            reason,
1073            class,
1074        });
1075    }
1076    match integrator.verify(candidate.number, &at, &method) {
1077        Ok(landed) => merged(root, &candidate, landed, "merged and verified".into()),
1078        Err(e) => unverified(root, &candidate, e.to_string()),
1079    }
1080}

1087    detail: String,

1121/// A merge the trail says was asked for, and nothing says how it ended: the executor
1122/// stopped between asking and verifying. It is verified now — asked whether it landed,
1123/// never asked again — and its end recorded, before anything else is decided.
1124pub fn reconcile(
1125    root: &Path,
1126    integrator: &mut dyn Integrator,
1127) -> Result<Option<DrainStepOutcome>, String> {
1128    let trail = events(root);
1129    let Some(i) = trail
1130        .iter()
1131        .rposition(|e| e.action == IntegrationAction::MergeAttempted)
1132    else {
1133        return Ok(None);
1134    };
1135    let asked = &trail[i];
1136    let ended = trail[i + 1..].iter().any(|e| {
1137        e.pr == asked.pr
1138            && matches!(
1139                e.action,
1140                IntegrationAction::MergeSucceeded
1141                    | IntegrationAction::MergeFailed
1142                    | IntegrationAction::VerificationFailed
1143            )
1144    });
1145    let (Some(pr), Some(master_sha), Some(head_sha)) = (
1146        asked.pr,
1147        asked.master_before.clone(),
1148        asked.head_sha.clone(),
1149    ) else {
1150        return Ok(None);
1151    };
1152    if ended {
1153        return Ok(None);
1154    }
1155    // decided when the merge was asked for: the attempt's own moment
1156    let at = EvaluatedAgainst {
1157        master_sha,
1158        head_sha,
1159        observed_at: asked.at.clone(),
1160    };
1161    let method = asked
1162        .detail
1163        .strip_prefix("--")
1164        .unwrap_or("merge")
1165        .to_string();
1166    let of = |action, detail: String| IntegrationEvent {
1167        pr: Some(pr),
1168        master_before: Some(at.master_sha.clone()),
1169        head_sha: Some(at.head_sha.clone()),
1170        reasons: asked.reasons.clone(),
1171        detail,
1172        ..IntegrationEvent::of(action)
1173    };
1174    Ok(Some(match integrator.verify(pr, &at, &method) {
1175        Ok(landed) => {
1176            let mut e = of(
1177                IntegrationAction::MergeSucceeded,
1178                "reconciled: the merge the last executor asked for landed".into(),
1179            );
1180            e.master_after = Some(landed.master_after.clone());
1181            e.merge_commit = landed.merge_commit;
1182            record(root, e)?;
1183            DrainStepOutcome::Merged {
1184                pr,
1185                master_before: at.master_sha.clone(),
1186                master_after: landed.master_after,
1187            }
1188        }
1189        Err(NotLanded::NotMerged(why)) => {
1190            let reason =
1191                format!("reconciled: the merge the last executor asked for never landed: {why}");
1192            // nothing the forge refused: the change is decided afresh, never held back for it
1193            let mut e = of(IntegrationAction::MergeFailed, reason.clone());
1194            e.class = Some(FailureClass::Transient);
1195            record(root, e)?;
1196            DrainStepOutcome::MergeRefused {
1197                pr,
1198                reason,
1199                class: FailureClass::Transient,
1200            }
1201        }
1202        Err(NotLanded::Unproved(why)) => {
1203            record(root, of(IntegrationAction::VerificationFailed, why.clone()))?;
1204            DrainStepOutcome::VerificationFailed { pr, reason: why }
1205        }
1206    }))
1207}

1209/// The merge that stops every drain until a person acknowledges it (owner decision D7):
1210/// the last `verification_failed` of the trail, unless a `failure_acknowledged` follows it.
1211pub fn halted_by(root: &Path) -> Option<(Option<u64>, String)> {

1213        .into_iter()
1214        .rev()
1215        .find(|e| {
1216            matches!(
1217                e.action,
1218                IntegrationAction::VerificationFailed | IntegrationAction::FailureAcknowledged
1219            )
1220        })
1221        .filter(|e| e.action == IntegrationAction::VerificationFailed)
1222        .map(|e| (e.pr, e.detail))

1225/// How many verified merges the trail must hold, since the last merge that could not be
1226/// verified, before a continuous drain may start (ADR 0101 §13).
1227///
1228/// The rollout is staged and the trail is its record: a dry run first, then one bounded
1229/// merge (`--max 1`), then small bounded drains, and continuous mode only once this many
1230/// merges have been made and proved where they landed. A merge that could not be verified
1231/// starts the count again, because the record it ends was the evidence.
1232pub const ROLLOUT_MERGES_BEFORE_CONTINUOUS: usize = 5;
1233
1234/// The verified merges recorded since the last `verification_failed`, or in the whole trail
1235/// when none failed: the record continuous mode is unlocked by.
1236///
1237/// `merge_succeeded` is recorded only for a merge proved where it landed, so the count is
1238/// of proved merges and not of attempts.
1239pub fn verified_merges_since_failure(events: &[IntegrationEvent]) -> usize {
1240    events
1241        .iter()
1242        .rev()
1243        .take_while(|e| e.action != IntegrationAction::VerificationFailed)
1244        .filter(|e| e.action == IntegrationAction::MergeSucceeded)
1245        .count()
1246}
1247
1248/// Why a continuous drain may not start yet, or nothing when the trail's record allows it.
1249pub fn continuous_refused(events: &[IntegrationEvent]) -> Option<String> {
1250    let record = verified_merges_since_failure(events);
1251    (record < ROLLOUT_MERGES_BEFORE_CONTINUOUS).then(|| {
1252        format!(
1253            "a continuous drain needs {ROLLOUT_MERGES_BEFORE_CONTINUOUS} verified merges on the \
1254             trail since the last one that could not be verified, and it holds {record}: \
1255             land them with bounded drains first (`prs drain --max 1`, then `--max 3`) \
1256             (ADR 0101 §13)"
1257        )
1258    })
1259}

1272/// The refresh half of a step, when nothing is ready. Pipeline depth one: while a pull
1273/// request the executor refreshed waits for its checks, no other is refreshed — merging the
1274/// first would put the second behind again and waste its CI run.
1275///
1276/// Only a head the executor pushed holds the pipeline — one a `refreshed` event, or the
1277/// `repaired` event of `prs repair`, names as `head_after` — and only while its required check is *pending*, or *missing* for
1278/// less than [`REFRESHED_HEAD_REPORTS_WITHIN`] since that event: an aggregate check is not
1279/// created until the jobs it needs finish, and a check that never reports must not hold every
1280/// refresh forever. A check running on a head the author pushed is not the executor's run to
1281/// wait for.
1282fn refresh_step(
1283    root: &Path,
1284    integrator: &mut dyn Integrator,
1285    first: &IntegrationQueue,
1286    dry_run: bool,
1287) -> Result<Option<DrainStepOutcome>, String> {
1288    // each pushed head with the moment it was pushed; on a trail line without a readable
1289    // time a pending check still holds, and a missing one does not
1290    let mut pushed: BTreeMap<(u64, String), Option<i64>> = BTreeMap::new();
1291    for e in events(root)
1292        .into_iter()
1293        // `prs repair` pushes master into a branch the same way a refresh does, so its head
1294        // holds the pipeline too: another refresh now would waste that run as well
1295        .filter(|e| {
1296            matches!(
1297                e.action,
1298                IntegrationAction::Refreshed | IntegrationAction::Repaired
1299            )
1300        })
1301    {
1302        if let Some(key) = e.pr.zip(e.head_after) {
1303            let at = crate::peers::epoch_seconds(&e.at);
1304            let slot = pushed.entry(key).or_insert(at);
1305            *slot = (*slot).max(at);
1306        }
1307    }
1308    let now = SystemTime::now()
1309        .duration_since(UNIX_EPOCH)
1310        .map_or(0, |d| d.as_secs() as i64);
1311    let within = REFRESHED_HEAD_REPORTS_WITHIN.as_secs() as i64;
1312    if let Some(waiting) = first.assessments.iter().find(|a| {
1313        a.disposition == PullRequestDisposition::WaitingForChecks
1314            && matches!(a.relation, super::RelationToMaster::UpToDate { .. })
1315            && pushed
1316                .get(&(a.number, a.evaluated_against.head_sha.clone()))
1317                .is_some_and(|at| {
1318                    a.required_checks == super::RequiredCheckState::Pending
1319                        || (a.required_checks == super::RequiredCheckState::Missing
1320                            && at.is_some_and(|t| now - t < within))
1321                })
1322    }) {
1323        return Ok(Some(DrainStepOutcome::AwaitingChecks {
1324            pr: waiting.number,
1325        }));
1326    }
1327    let Some(candidate) = first
1328        .next_refresh
1329        .first()
1330        .and_then(|n| first.get(*n))
1331        .cloned()
1332    else {
1333        return Ok(None);
1334    };
1335    if dry_run {
1336        return Ok(Some(DrainStepOutcome::WouldRefresh {
1337            pr: candidate.number,
1338        }));
1339    }
1340    let mut selected = event(
1341        IntegrationAction::RefreshSelected,
1342        Some(&candidate),
1343        "the first pull request that needs master",
1344    )
1345    .with_evidence(&candidate);
1346    selected.passed_over = others(
1347        first,
1348        PullRequestDisposition::NeedsRefresh,
1349        candidate.number,
1350    );
1351    record(root, selected)?;
1352    let second = integrator.observe()?;
1353    let fresh = second.get(candidate.number);
1354    let stale = match fresh {
1355        None => Some("it is no longer open".to_string()),
1356        Some(a) if a.evaluated_against != candidate.evaluated_against => Some(format!(
1357            "master or its head moved (now master {}, head {})",
1358            a.evaluated_against.master_sha, a.evaluated_against.head_sha
1359        )),
1360        Some(a) if a.disposition != PullRequestDisposition::NeedsRefresh => {
1361            Some(format!("it is {} now", a.disposition.as_str()))
1362        }
1363        Some(_) => None,
1364    };
1365    if let Some(what) = stale {
1366        record(
1367            root,
1368            event(
1369                IntegrationAction::StaleDecision,
1370                Some(&candidate),
1371                what.clone(),
1372            ),
1373        )?;
1374        return Ok(Some(DrainStepOutcome::StaleDecision {
1375            pr: candidate.number,
1376            what,
1377        }));
1378    }
1379    // on the trail before anything is pushed: a push the trail cannot name is not made
1380    let attempted = event(
1381        IntegrationAction::RefreshAttempted,
1382        Some(&candidate),
1383        format!("master {} into {}", second.master_sha, candidate.head_ref),
1384    );
1385    if let Err(reason) = record(root, attempted) {
1386        return Ok(Some(DrainStepOutcome::TrailUnwritable {
1387            pr: candidate.number,
1388            unrecorded: IntegrationAction::RefreshAttempted,
1389            reason,
1390        }));
1391    }
1392    match integrator.refresh_branch(&candidate, &second.base) {
1393        Ok(head_after) => {
1394            let mut e = event(
1395                IntegrationAction::Refreshed,
1396                Some(&candidate),
1397                format!("new head {head_after}"),
1398            );
1399            e.master_after = Some(second.master_sha.clone());
1400            e.head_after = Some(head_after.clone());
1401            record(root, e)?;
1402            Ok(Some(DrainStepOutcome::Refreshed {
1403                pr: candidate.number,
1404                head_before: candidate.evaluated_against.head_sha.clone(),
1405                head_after,
1406            }))
1407        }
1408        Err(reason) => {
1409            record(
1410                root,
1411                event(
1412                    IntegrationAction::RefreshFailed,
1413                    Some(&candidate),
1414                    reason.clone(),
1415                ),
1416            )?;
1417            let class = FailureClass::of_refresh_failure(&reason);
1418            Ok(Some(DrainStepOutcome::RefreshFailed {
1419                pr: candidate.number,
1420                reason,
1421                class,
1422            }))
1423        }
1424    }
1425}

1440impl DrainReport {

1446            Some(DrainStepOutcome::Idle { .. } | DrainStepOutcome::Halted { .. }) => None,

1448        }

1452/// Merge until `max` merges, an idle queue, or a systemic failure. A stale decision or a
1453/// refused merge is candidate-specific: the next step re-plans from a new observation,
1454/// bounded by `max_steps` so a flapping queue cannot spin.
1455pub fn drain(
1456    root: &Path,
1457    integrator: &mut dyn Integrator,
1458    max: usize,
1459    dry_run: bool,
1460    allow_refresh: bool,
1461) -> Result<DrainReport, String> {
1462    drain_until(root, integrator, max, dry_run, allow_refresh, None)
1463}

1465/// [`drain`], asked before every step whether to stop: a signal lets the step in progress
1466/// finish and starts no other.
1467pub fn drain_until(
1468    root: &Path,
1469    integrator: &mut dyn Integrator,
1470    max: usize,
1471    dry_run: bool,
1472    allow_refresh: bool,
1473    stop: Option<&std::sync::atomic::AtomicBool>,
1474) -> Result<DrainReport, String> {
1475    let mut report = DrainReport {
1476        dry_run,
1477        steps: Vec::new(),
1478        merged: Vec::new(),
1479        stopped: String::new(),
1480    };
1481    if !dry_run {
1482        // an earlier executor's merge that was asked for and never ended is ended first
1483        if let Some(outcome) = reconcile(root, integrator)? {
1484            if let DrainStepOutcome::Merged { pr, .. } = &outcome {
1485                report.merged.push(*pr);
1486            }
1487            report.steps.push(outcome);
1488        }
1489        if let Some((pr, reason)) = halted_by(root) {
1490            report.stopped = format!(
1491                "{} could not be verified after merging ({reason}); nothing merges until a \
1492                 person has looked and run `prs drain --resume-after-failure`",
1493                pr.map_or_else(|| "a merge".to_string(), |n| format!("#{n}"))
1494            );
1495            report.steps.push(DrainStepOutcome::Halted { pr, reason });
1496            return Ok(report);
1497        }
1498    }
1499    let max_steps = max.saturating_mul(3).max(3);
1500    while report.merged.len() < max {
1501        if stop.is_some_and(|s| s.load(std::sync::atomic::Ordering::SeqCst)) {
1502            report.stopped = "asked to stop; the step in progress finished first".into();
1503            break;
1504        }
1505        if report.steps.len() >= max_steps {
1506            report.stopped = format!("{max_steps} steps without reaching {max} merge(s)");
1507            break;
1508        }
1509        let outcome = step(root, integrator, dry_run, allow_refresh)?;
1510        if let DrainStepOutcome::Merged { pr, .. } = &outcome {
1511            report.merged.push(*pr);
1512        }
1513        let stop = ends_drain(&outcome);
1514        report.steps.push(outcome);
1515        if let Some(why) = stop {
1516            report.stopped = why;
1517            return Ok(report);
1518        }
1519    }
1520    if report.stopped.is_empty() {
1521        report.stopped = format!("{max} merge(s), the bound asked for");
1522    }
1523    Ok(report)
1524}

1526/// Why a drain ends after a step that came out as `outcome`, or `None` when it goes on.
1527pub(crate) fn ends_drain(outcome: &DrainStepOutcome) -> Option<String> {
1528    match outcome {
1529        DrainStepOutcome::Idle { why } => Some(why.clone()),
1530        DrainStepOutcome::WouldMerge { pr } => Some(format!(
1531            "dry run: #{pr} would be merged; nothing after it is planned, because its merge would change the queue"
1532        )),
1533        DrainStepOutcome::Merged { .. } => None,
1534        DrainStepOutcome::StaleDecision { .. } => None,
1535        // the candidate's own failure: the queue holds it back and the next is tried. An
1536        // outage is not the candidate's: asking the forge to merge again at once would
1537        // only hammer it, so the drain ends and a later one (or the next cycle) asks
1538        DrainStepOutcome::MergeRefused { pr, reason, class } => {
1539            (*class == FailureClass::Transient || !class.recoverable()).then(|| {
1540                format!("the forge refused #{pr} ({reason}); nothing further is attempted")
1541            })
1542        }
1543        DrainStepOutcome::WouldRefresh { pr } => Some(format!(
1544            "dry run: master would be brought into #{pr}; its checks must then pass before it can merge"
1545        )),
1546        DrainStepOutcome::Refreshed { pr, .. } => Some(format!(
1547            "#{pr} now contains master; its required checks run on the new head, and the next drain merges it when they pass"
1548        )),
1549        DrainStepOutcome::AwaitingChecks { pr } => Some(format!(
1550            "#{pr} contains master and its checks are running; nothing else is refreshed meanwhile"
1551        )),
1552        DrainStepOutcome::RefreshFailed { pr, reason, class } => (!class.recoverable())
1553            .then(|| format!("bringing master into #{pr} failed: {reason}")),
1554        DrainStepOutcome::VerificationFailed { pr, reason } => {
1555            Some(format!("#{pr} could not be verified after merging: {reason}"))
1556        }
1557        DrainStepOutcome::TrailUnwritable {
1558            pr,
1559            unrecorded,
1560            reason,
1561        } => Some(format!(
1562            "the trail could not record {unrecorded}, so nothing was done for #{pr} ({reason}); nothing further is attempted while the trail cannot be written"
1563        )),
1564        // a halt comes from the start of a drain, before any step; said as it is
1565        DrainStepOutcome::Halted { reason, .. } => Some(reason.clone()),
1566    }
1567}

1601/// Drain, wait, and drain again, until `stop` is set, a cycle bound is reached, a merge
1602/// could not be verified, or the repository cannot be read. Every cycle is a [`drain`], so
1603/// every action is preceded by a fresh observation and no plan outlives a merge; the wait
1604/// is taken in one-second slices so a stop is honoured within a second of the step in
1605/// progress finishing. `sleep` takes the waits (a test records them); `on_cycle` hears each
1606/// How many consecutive cycles a continuous drain waits out a transient failure to observe
1607/// the forge — a timeout, a 5xx, a rate limit — before it ends: long enough for a blip,
1608/// short enough that an outage is not hidden for hours.
1609pub const CONTINUOUS_TRANSIENT_LIMIT: usize = 3;
1610
1611/// Wait one interval, a second at a time, until it passes or a stop is asked for.
1612fn wait_interval(
1613    interval: Duration,
1614    stop: &std::sync::atomic::AtomicBool,
1615    sleep: &mut dyn FnMut(Duration),
1616) {
1617    let mut left = interval;
1618    while !left.is_zero() && !stop.load(std::sync::atomic::Ordering::SeqCst) {
1619        let slice = left.min(Duration::from_secs(1));
1620        sleep(slice);
1621        left = left.saturating_sub(slice);
1622    }
1623}

1625/// cycle's report as it ends. The caller holds the lease for the whole run.
1626pub fn continuous(
1627    root: &Path,
1628    integrator: &mut dyn Integrator,
1629    opts: ContinuousOptions,
1630    stop: &std::sync::atomic::AtomicBool,
1631    sleep: &mut dyn FnMut(Duration),
1632    on_cycle: &mut dyn FnMut(usize, &DrainReport),
1633) -> ContinuousReport {
1634    use std::sync::atomic::Ordering;
1635    let mut out = ContinuousReport {
1636        cycles: 0,
1637        merged: Vec::new(),
1638        stopped: String::new(),
1639        failure: None,
1640    };
1641    let started = IntegrationEvent {
1642        detail: format!(
1643            "every {} s, at most {} merge(s) a cycle, refresh allowed: {}",
1644            opts.interval.as_secs(),
1645            opts.max_per_cycle,
1646            opts.allow_refresh
1647        ),
1648        ..IntegrationEvent::of(IntegrationAction::ContinuousStarted)
1649    };
1650    if let Err(e) = record(root, started) {
1651        out.stopped = "the trail could not record the start; nothing is attempted".into();
1652        out.failure = Some(e);
1653        return out;
1654    }
1655    let mut outages = 0usize;
1656    loop {
1657        if stop.load(Ordering::SeqCst) {
1658            out.stopped = "asked to stop; the step in progress finished first".into();
1659            break;
1660        }
1661        let report = match drain_until(
1662            root,
1663            integrator,
1664            opts.max_per_cycle,
1665            false,
1666            opts.allow_refresh,
1667            Some(stop),
1668        ) {
1669            Ok(r) => {
1670                outages = 0;
1671                r
1672            }
1673            // a short outage of the forge is waited out, a few cycles at most, and said on the
1674            // trail; anything else, or an outage that outlasts them, ends the run
1675            Err(e) if super::retry::transient(&e) && outages < CONTINUOUS_TRANSIENT_LIMIT => {
1676                outages += 1;
1677                let _ = record(
1678                    root,
1679                    IntegrationEvent {
1680                        detail: format!(
1681                            "outage {outages} of at most {CONTINUOUS_TRANSIENT_LIMIT}: {e}"
1682                        ),
1683                        ..IntegrationEvent::of(IntegrationAction::ObserveFailed)
1684                    },
1685                );
1686                wait_interval(opts.interval, stop, sleep);
1687                continue;
1688            }
1689            Err(e) => {
1690                out.stopped =
1691                    "the repository could not be read; nothing further is attempted".into();
1692                out.failure = Some(e);
1693                break;
1694            }
1695        };
1696        out.cycles += 1;
1697        out.merged.extend(report.merged.iter().copied());
1698        on_cycle(out.cycles, &report);
1699        if let Some(DrainStepOutcome::VerificationFailed { pr, reason }) = report
1700            .steps
1701            .iter()
1702            .find(|s| matches!(s, DrainStepOutcome::VerificationFailed { .. }))
1703        {
1704            out.stopped = format!(
1705                "#{pr} could not be verified after merging ({reason}); a person looks before anything else merges"
1706            );
1707            break;
1708        }
1709        if report
1710            .steps
1711            .iter()
1712            .any(|s| matches!(s, DrainStepOutcome::Halted { .. }))
1713        {
1714            out.stopped = report.stopped.clone();
1715            break;
1716        }
1717        if let Some(DrainStepOutcome::TrailUnwritable { reason, .. }) = report
1718            .steps
1719            .iter()
1720            .find(|s| matches!(s, DrainStepOutcome::TrailUnwritable { .. }))
1721        {
1722            out.stopped = report.stopped.clone();
1723            out.failure = Some(reason.clone());
1724            break;
1725        }
1726        if opts.cycles.is_some_and(|n| out.cycles >= n) {
1727            out.stopped = format!("{} cycle(s), the bound asked for", out.cycles);
1728            break;
1729        }
1730        wait_interval(opts.interval, stop, sleep);
1731    }
1732    let stopped = IntegrationEvent {
1733        detail: out.stopped.clone(),
1734        ..IntegrationEvent::of(IntegrationAction::ContinuousStopped)
1735    };
1736    if let Err(e) = record(root, stopped) {
1737        // the first failure is the one that stopped the drain; this one is said after it
1738        out.failure.get_or_insert(e);
1739    }
1740    out
1741}

1743/// Ask this process to stop at the next safe point on SIGINT or SIGTERM: the returned flag
1744/// is set by the first signal, and a second one ends the process at once the way the signal
1745/// would have. Installed once per process; the handler does only async-signal-safe work.
1746pub fn stop_on_signals() -> &'static std::sync::atomic::AtomicBool {
1747    signals::install()
1748}
1749
1750#[cfg(unix)]
1751mod signals {
1752    use std::sync::atomic::{AtomicBool, Ordering};
1753    use std::sync::Once;
1754
1755    static STOP: AtomicBool = AtomicBool::new(false);
1756    static INSTALL: Once = Once::new();
1757
1758    pub fn install() -> &'static AtomicBool {
1759        INSTALL.call_once(|| {
1760            let handler = on_signal as extern "C" fn(libc::c_int) as libc::sighandler_t;
1761            for signal in [libc::SIGTERM, libc::SIGINT] {
1762                // SAFETY: the handler only touches an atomic and, on a second signal, calls
1763                // async-signal-safe libc functions
1764                unsafe { libc::signal(signal, handler) };
1765            }
1766        });
1767        &STOP
1768    }
1769
1770    extern "C" fn on_signal(signal: libc::c_int) {
1771        if STOP.swap(true, Ordering::SeqCst) {
1772            // the second signal: the person means it
1773            // SAFETY: restoring the default disposition and re-raising are async-signal-safe
1774            unsafe {
1775                libc::signal(signal, libc::SIG_DFL);
1776                libc::raise(signal);
1777            }
1778        }
1779    }
1780}

1784    use std::sync::atomic::AtomicBool;
1785
1786    static STOP: AtomicBool = AtomicBool::new(false);

1789        &STOP
1790    }

1807    if out.status.success() {

1813            format!(
1814                "gh {} failed ({}) and said nothing",
1815                args.iter().take(3).copied().collect::<Vec<_>>().join(" "),
1816                out.status
1817            )
1818        } else {
1819            said
1820        })
1821    }

1824/// What landed on `master`, proved from git alone: the decision's master is on master's
1825/// first-parent line, and the commit right after it there is this merge — its first parent
1826/// that master and, for a merge commit, its second parent the head that was decided on. A
1827/// merge that landed after another one, onto a master nobody tested together with it, is
1828/// refused by name: it is the one thing a merge-after-decision can do wrong that the forge
1829/// would still call merged. The merge commit, or `None` for a rebase merge, which leaves no
1830/// single commit to name.
1831pub(crate) fn landing(
1832    root: &Path,
1833    at: &EvaluatedAgainst,
1834    method: &str,
1835    master: &str,
1836) -> Result<Option<String>, String> {
1837    let git = |args: &[&str]| -> Option<String> {
1838        Command::new("git")
1839            .arg("-C")
1840            .arg(root)
1841            .args(args)
1842            .output()
1843            .ok()
1844            .filter(|out| out.status.success())
1845            .map(|out| String::from_utf8_lossy(&out.stdout).trim().to_string())
1846    };
1847    let contains = |a: &str, b: &str| {
1848        Command::new("git")
1849            .arg("-C")
1850            .arg(root)
1851            .args(["merge-base", "--is-ancestor", a, b])
1852            .status()
1853            .is_ok_and(|s| s.success())
1854    };
1855    if !contains(&at.master_sha, master) {
1856        return Err(format!(
1857            "master {master} does not contain {}, the master the merge was decided against",
1858            at.master_sha
1859        ));
1860    }
1861    if method == "rebase" {
1862        // a rebase leaves copies of the head's commits, not the head and not one commit
1863        return Ok(None);
1864    }
1865    let after = git(&[
1866        "rev-list",
1867        "--first-parent",
1868        "--reverse",
1869        &format!("{}..{master}", at.master_sha),
1870    ])
1871    .unwrap_or_default();
1872    let Some(merge) = after.lines().next().map(str::to_string) else {
1873        return Err(format!(
1874            "master is still {master}: nothing landed on the master the merge was decided against"
1875        ));
1876    };
1877    if git(&["rev-parse", &format!("{merge}^1")]).as_deref() != Some(at.master_sha.as_str()) {
1878        return Err(format!(
1879            "{} is not on master's first-parent line: the commit after it there is {merge}",
1880            at.master_sha
1881        ));
1882    }
1883    if method == "merge" {
1884        let second = git(&["rev-parse", &format!("{merge}^2")]);
1885        if second.as_deref() != Some(at.head_sha.as_str()) {
1886            return Err(format!(
1887                "unexpected_master: the commit merged onto {} is {merge}, a merge of {}, not \
1888                 of {}: another change landed between the decision and this merge",
1889                at.master_sha,
1890                second.as_deref().unwrap_or("nothing"),
1891                at.head_sha
1892            ));
1893        }
1894    }
1895    Ok(Some(merge))
1896}

1898/// Why the merge of `master` in the scratch worktree `dir` stopped, from the paths it left
1899/// unmerged, each named as authored or derived by master's own `.gitattributes`
1900/// ([`super::relation::derived_paths`]); `None` when git names none, or cannot say which is
1901/// which. An authored conflict is the owner's to settle; a derived one means this clone has no
1902/// `merge.derived` driver.
1903fn unmerged_paths(root: &Path, dir: &Path, master: &str) -> Option<String> {
1904    let listed = Command::new("git")
1905        .arg("-C")
1906        .arg(dir)
1907        .args(["diff", "--name-only", "-z", "--diff-filter=U"])
1908        .output()
1909        .map(|o| o.stdout)
1910        .unwrap_or_default();
1911    let paths: Vec<String> = String::from_utf8_lossy(&listed)
1912        .split('\0')
1913        .filter(|p| !p.is_empty())
1914        .map(str::to_string)
1915        .collect();
1916    Some(paths)
1917        .filter(|paths| !paths.is_empty())
1918        .and_then(|paths| {
1919            super::relation::derived_paths(root, master, &paths)
1920                .ok()
1921                .map(|derived| (paths, derived))
1922        })
1923        .map(|(paths, derived)| {
1924            let named: Vec<String> = paths
1925                .iter()
1926                .map(|p| {
1927                    format!(
1928                        "{p} ({})",
1929                        ["authored", "derived"][usize::from(derived.contains(p))]
1930                    )
1931                })
1932                .collect();
1933            format!(
1934                "the merge of master conflicts on {} path(s): {}; an authored path is the \
1935                 owner's to settle, a derived one needs this clone's merge.derived driver \
1936                 (just derive-merge-driver)",
1937                named.len(),
1938                named.join(", ")
1939            )
1940        })
1941}

2080    fn refresh_branch(&mut self, a: &PullRequestAssessment, base: &str) -> Result<String, String> {
2081        // a refresh runs the repository's derive, which can outlast what an observer calls a
2082        // stale lease: the record is kept fresh for as long as it runs
2083        let _alive = self.lease.map(IntegrationLease::keep_alive);
2084        let root = self.root;
2085        let common = common_dir(root)?;
2086        let dir = common
2087            .join("majordomus/integration")
2088            .join(format!("pr-{}", a.number));
2089        let head = a.evaluated_against.head_sha.clone();
2090        let master = a.evaluated_against.master_sha.clone();
2091        // a scratch worktree of our own: the person's checkouts are never touched. One left
2092        // by an interrupted act is removed first; git's complaint that there is none is not
2093        // the person's to read, so it is captured rather than inherited
2094        let _ = Command::new("git")
2095            .arg("-C")
2096            .arg(root)
2097            .args(["worktree", "remove", "--force"])
2098            .arg(&dir)
2099            .output();
2100        let git_in = |args: &[&str]| -> Result<String, String> {
2101            let out = Command::new("git")
2102                .arg("-C")
2103                .arg(&dir)
2104                .args(args)
2105                .output()
2106                .map_err(|e| e.to_string())?;
2107            if out.status.success() {
2108                Ok(String::from_utf8_lossy(&out.stdout).trim().to_string())
2109            } else {
2110                Err(format!(
2111                    "git {}: {}",
2112                    args.join(" "),
2113                    String::from_utf8_lossy(&out.stderr).trim()
2114                ))
2115            }
2116        };
2117        let add = Command::new("git")
2118            .arg("-C")
2119            .arg(root)
2120            .args(["worktree", "add", "--detach"])
2121            .arg(&dir)
2122            .arg(&head)
2123            .output()
2124            .map_err(|e| e.to_string())?;
2125        if !add.status.success() {
2126            return Err(format!(
2127                "git worktree add: {}",
2128                String::from_utf8_lossy(&add.stderr).trim()
2129            ));
2130        }
2131        let result = (|| -> Result<String, String> {
2132            // rerere off: a resolution recorded during some other merge is a memory of what
2133            // one person decided about two other commits, and replaying it here would resolve
2134            // paths nobody decided on
2135            if let Err(e) = git_in(&[
2136                "-c",
2137                "rerere.enabled=false",
2138                "merge",
2139                "--no-commit",
2140                "--no-ff",
2141                &master,
2142            ]) {
2143                let why = unmerged_paths(root, &dir, &master)
2144                    .unwrap_or(format!("the merge of master conflicts after all: {e}"));
2145                let _ = git_in(&["merge", "--abort"]);
2146                return Err(why);
2147            }
2148            // the derived artifacts of the merge result, regenerated rather than resolved
2149            let target = root.join("apps/majordomus-cli/target");
2150            let derive = Command::new(dir.join("scripts/derive"))
2151                .current_dir(&dir)
2152                .env_remove("MAJORDOMUS_SHARE")
2153                .env("CARGO_TARGET_DIR", &target)
2154                .output()
2155                .map_err(|e| format!("scripts/derive could not run: {e}"))?;
2156            if !derive.status.success() {
2157                let _ = git_in(&["merge", "--abort"]);
2158                return Err(format!(
2159                    "scripts/derive failed: {}",
2160                    String::from_utf8_lossy(&derive.stderr)
2161                        .lines()
2162                        .rev()
2163                        .take(3)
2164                        .collect::<Vec<_>>()
2165                        .join(" | ")
2166                ));
2167            }
2168            git_in(&["add", "-A"])?;
2169            let message = format!("Merge {base} into {} with its derived artifacts regenerated\n\nBrought in by the integrator (majordomus prs) so that the required checks run against master {master}.", a.head_ref);
2170            let commit = Command::new("git")
2171                .arg("-C")
2172                .arg(&dir)
2173                .args(["commit", "--no-edit", "-m", &message])
2174                .env_remove("MAJORDOMUS_SHARE")
2175                .output()
2176                .map_err(|e| e.to_string())?;
2177            if !commit.status.success() {
2178                let _ = git_in(&["merge", "--abort"]);
2179                return Err(format!(
2180                    "the merge commit was refused: {}",
2181                    String::from_utf8_lossy(&commit.stderr)
2182                        .lines()
2183                        .rev()
2184                        .take(3)
2185                        .collect::<Vec<_>>()
2186                        .join(" | ")
2187                ));
2188            }
2189            let new_head = git_in(&["rev-parse", "HEAD"])?;
2190            // what is pushed descends from the head that was decided on, or nothing is pushed
2191            if git_in(&["merge-base", "--is-ancestor", &head, &new_head]).is_err() {
2192                return Err(format!(
2193                    "the merge {new_head} does not descend from the observed head {head}; nothing was pushed"
2194                ));
2195            }
2196            // the branch must still be the head that was decided on: one its author moved since
2197            // — forward, sideways or back, or deleted — is refused here. A listing that fails is
2198            // its own refusal, in git's words, so an outage is classed as one (transient) and
2199            // never reported as a branch that moved.
2200            let listed = git_in(&["ls-remote", "origin", &format!("refs/heads/{}", a.head_ref)])
2201                .map_err(|e| format!("the branch could not be listed: {e}"))?;
2202            let remote = listed.split_whitespace().next().unwrap_or("nothing");
2203            if remote != head {
2204                return Err(format!(
2205                    "the branch moved since it was observed: {remote} is not {head}; nothing was pushed"
2206                ));
2207            }
2208            // A plain push, never forced: git accepts only a fast-forward, so no history is
2209            // ever overwritten. Between the check above and this push the author may still act.
2210            // Moving the branch forward or sideways then makes this push no fast-forward, and
2211            // git refuses it. Rewinding it to an ancestor of the observed head is the one move
2212            // that slips through: the push is then a fast-forward that restores the commits the
2213            // author removed. That window is accepted, and it never rewrites anything.
2214            let push = Command::new("git")
2215                .arg("-C")
2216                .arg(&dir)
2217                .args([
2218                    "push",
2219                    "origin",
2220                    &format!("{new_head}:refs/heads/{}", a.head_ref),
2221                ])
2222                .env_remove("MAJORDOMUS_SHARE")
2223                .output()
2224                .map_err(|e| e.to_string())?;
2225            if !push.status.success() {
2226                return Err(format!(
2227                    "the push was refused: {}",
2228                    String::from_utf8_lossy(&push.stderr).trim()
2229                ));
2230            }
2231            Ok(new_head)
2232        })();
2233        let _ = Command::new("git")
2234            .arg("-C")
2235            .arg(root)
2236            .args(["worktree", "remove", "--force"])
2237            .arg(&dir)
2238            .output();
2239        result
2240    }

2265/// Close pull requests whose work is provably on master already, and nothing else.
2266///
2267/// Only [`PullRequestDisposition::Redundant`] — the head is an ancestor of master, merging it
2268/// changes no file, or every commit is on master as an equal patch — and
2269/// [`PullRequestDisposition::Superseded`] — a declared successor landed, and the comment names
2270/// it — qualify. A pull request whose merge would change only derived
2271/// artifacts ([`PullRequestDisposition::PossiblyRedundant`]) is listed for a person and never closed:
2272/// "the generated output differs" is not proof that the authored change landed. Age,
2273/// shared paths and similar titles are not evidence of anything here. The queue is
2274/// observed afresh before closing, and each closure names the master that proved it.
2275pub fn cleanup(
2276    root: &Path,
2277    integrator: &mut dyn Integrator,
2278    apply: bool,
2279) -> Result<Vec<CleanupItem>, String> {
2280    let queue = integrator.observe()?;
2281    let mut items = Vec::new();
2282    for a in &queue.assessments {
2283        let Some(planned) = cleanup_action(a.disposition) else {
2284            continue;
2285        };
2286        let action = if apply && planned == WOULD_CLOSE {
2287            close_superseded(root, integrator, a)?
2288        } else {
2289            planned.to_string()
2290        };
2291        items.push(cleanup_item(a, action));
2292    }
2293    Ok(items)
2294}

2335/// Close one superseded pull request, as a merge is taken: observed again first, closed only
2336/// if the second decision still says superseded against the same master and head, recorded
2337/// before the forge hears of it, and refused by the forge side if the head moved meanwhile.
2338/// The comment names the reason that decided it, the base, the master and the head.
2339fn close_superseded(
2340    root: &Path,
2341    integrator: &mut dyn Integrator,
2342    a: &PullRequestAssessment,
2343) -> Result<String, String> {
2344    let second = integrator.observe()?;
2345    let again = second.get(a.number);
2346    let stale = match again {
2347        None => Some(format!("#{} is no longer open", a.number)),
2348        Some(b) if b.evaluated_against != a.evaluated_against => Some(format!(
2349            "master or the head moved: decided on master {} head {}, now master {} head {}",
2350            a.evaluated_against.master_sha,
2351            a.evaluated_against.head_sha,
2352            b.evaluated_against.master_sha,
2353            b.evaluated_against.head_sha
2354        )),
2355        Some(b) if b.disposition != a.disposition => Some(format!(
2356            "#{} is {} now, not {}",
2357            a.number,
2358            b.disposition.as_str(),
2359            a.disposition.as_str()
2360        )),
2361        Some(_) => None,
2362    };
2363    if let Some(what) = stale {
2364        record(
2365            root,
2366            event(IntegrationAction::StaleDecision, Some(a), what.clone()),
2367        )?;
2368        return Ok(format!("stale_decision: {what}"));
2369    }
2370    let deciding = a
2371        .reasons
2372        .first()
2373        .map_or(String::from("superseded"), ToString::to_string);
2374    let body = format!(
2375        "Closed by `majordomus prs cleanup`: its work is already on `{}`.\n\nEvidence: {deciding} (master {}, head {}).",
2376        second.base, a.evaluated_against.master_sha, a.evaluated_against.head_sha
2377    );
2378    // a declared successor that landed is named first: the reader learns where the work went
2379    let body = match a.superseded_by {
2380        Some(by) => format!("Superseded by #{by}, which landed.\n\n{body}"),
2381        None => body,
2382    };
2383    // on the trail before the forge hears of it: a closure the trail cannot name is not made,
2384    // and nothing after it is attempted
2385    record(
2386        root,
2387        event(IntegrationAction::CloseAttempted, Some(a), body.clone()),
2388    )
2389    .map_err(|e| {
2390        format!(
2391            "#{} was not closed: the trail could not record it first: {e}",
2392            a.number
2393        )
2394    })?;
2395    match integrator.close(a.number, &a.evaluated_against.head_sha, &body) {
2396        Ok(()) => {
2397            record(

2413/// A branch a merged pull request left on origin, as cleanup reports it. Reported, never
2414/// deleted: the forge's `delete_branch_on_merge` decides deletion (owner decision D4), so the
2415/// executor holds no branch-deleting write at all.
2416#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
2417pub struct LeftBranch {
2418    /// The branch on origin.
2419    pub branch: String,
2420    /// Its tip: the head that merged.
2421    pub tip: String,
2422    /// The pull request that merged it.
2423    pub pr: u64,
2424    /// When that pull request merged.
2425    pub merged_at: String,
2426    /// `left_for_a_person`, or `kept: checked out at <path>` when a worktree of this
2427    /// repository has the branch checked out: somebody may still be standing on it.
2428    pub action: String,
2429    /// What would clear it, in the setting's own terms.
2430    pub next_step: String,
2431}
2432
2433/// The last report of the branches merged pull requests left on origin, as `prs cleanup`
2434/// read it. Written by `prs cleanup` alone — the one place that asks the forge for them —
2435/// and rendered offline, with its age, by every surface that cannot reach the network.
2436#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
2437pub struct LeftBranchReport {
2438    /// When cleanup read origin and the merged pull requests.
2439    pub read_at: String,
2440    /// The base the report excludes.
2441    pub base: String,
2442    /// The forge's `delete_branch_on_merge`, from the observation; `None` when unread.
2443    pub delete_branch_on_merge: Option<bool>,
2444    /// The branches, or `None` when origin or the merged pull requests could not be read:
2445    /// unread is not "nothing left behind".
2446    pub branches: Option<Vec<LeftBranch>>,
2447    /// The command that reads it afresh.
2448    pub refreshed_by: String,
2449}
2450
2451/// Where [`LeftBranchReport`] is kept, beside the observation.
2452pub const LEFT_BRANCHES_FILE: &str = "left-branches.json";
2453
2454/// The branches merged pull requests left behind, each with what would clear it.
2455///
2456/// `merged` are same-repository heads origin serves at the exact head that merged
2457/// ([`super::forge::merged_branches_of`]), so a branch whose tip moved after its merge is
2458/// never among them; the base is excluded here too. One checked out in a worktree is listed
2459/// as kept, with the path. `None` when they could not be read.
2460pub fn left_branches(
2461    merged: Option<&[super::forge::MergedBranch]>,
2462    base: &str,
2463    delete_branch_on_merge: Option<bool>,
2464    checked_out: &BTreeMap<String, PathBuf>,
2465) -> Option<Vec<LeftBranch>> {
2466    let next_step = |branch: &str| match delete_branch_on_merge {
2467        Some(true) => format!(
2468            "the forge deletes merged branches, and this one outlived its merge (merged before \
2469             the setting, or restored): git push origin --delete {branch}"
2470        ),
2471        Some(false) => format!(
2472            "the forge keeps merged branches: enable delete_branch_on_merge, and delete this one \
2473             with git push origin --delete {branch}"
2474        ),
2475        None => format!(
2476            "the forge's delete_branch_on_merge was not read: majordomus prs refresh, or delete \
2477             it with git push origin --delete {branch}"
2478        ),
2479    };
2480    Some(
2481        merged?
2482            .iter()
2483            .filter(|m| m.branch != base)
2484            .map(|m| LeftBranch {
2485                branch: m.branch.clone(),
2486                tip: m.tip.clone(),
2487                pr: m.pr,
2488                merged_at: m.merged_at.clone(),
2489                action: match checked_out.get(&m.branch) {
2490                    Some(path) => format!("kept: checked out at {}", path.display()),
2491                    None => "left_for_a_person".into(),
2492                },
2493                next_step: next_step(&m.branch),
2494            })
2495            .collect(),
2496    )
2497}

2499/// Read origin and the merged pull requests now ([`super::forge::merged_branches`]), against
2500/// the recorded observation's base and setting and the worktrees this repository has
2501/// registered, and record the report beside the observation. `None` when nothing was ever
2502/// observed here. Network and a write: only `prs cleanup` calls it, after observing. A
2503/// worktree list git cannot give marks nothing as kept; nothing here deletes a branch.
2504pub fn report_left_branches(root: &Path) -> Result<Option<LeftBranchReport>, String> {
2505    let Some(obs) = super::load_observation(root)? else {
2506        return Ok(None);
2507    };
2508    let checked_out: BTreeMap<String, PathBuf> = crate::worktree::topology::read(root)
2509        .unwrap_or_default()
2510        .into_iter()
2511        .filter_map(|w| w.branch.map(|b| (b, w.path)))
2512        .collect();
2513    let merged = super::forge::merged_branches(root, &obs.base);
2514    let report = LeftBranchReport {
2515        read_at: crate::peers::rfc3339(SystemTime::now()),
2516        base: obs.base.clone(),
2517        delete_branch_on_merge: obs.delete_branch_on_merge,
2518        branches: left_branches(
2519            merged.as_deref(),
2520            &obs.base,
2521            obs.delete_branch_on_merge,
2522            &checked_out,
2523        ),
2524        refreshed_by: "majordomus prs cleanup".into(),
2525    };
2526    // a record of strings, numbers and options: serializing it cannot fail
2527    let text = serde_json::to_string_pretty(&report).expect("a branch report serializes");
2528    super::write_atomic(&super::state_path(root, LEFT_BRANCHES_FILE), &(text + "\n"))?;
2529    Ok(Some(report))
2530}

2532/// The last recorded [`LeftBranchReport`], read offline; `None` when `prs cleanup` never
2533/// recorded one here or the record does not parse.
2534pub fn recorded_left_branches(root: &Path) -> Option<LeftBranchReport> {
2535    let text = fs::read_to_string(super::state_path(root, LEFT_BRANCHES_FILE)).ok()?;
2536    serde_json::from_str(&text).ok()
2537}
