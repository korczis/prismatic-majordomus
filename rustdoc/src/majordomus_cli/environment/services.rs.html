1//! The local services of a repository: what this executable serves, where, and whether
2//! anything is answering there now.
3//!
4//! Paths are not written here. Each descriptor names the constant that already decides its
5//! route — [`crate::cockpit::PREFIX`], [`crate::capability::HttpExposure::PREFIX`],
6//! [`crate::http::swagger::SWAGGER_PATH`], [`crate::web::discover::DOCS_MOUNT`] and the
7//! rest of what [`crate::http::openapi::infrastructure_routes`] lists — so a route that
8//! moves moves here too, and `every_infrastructure_route_is_described` fails the build if
9//! a new one is added without a name for it. The base address is the shared server's lease, which is
10//! where a running server publishes it and the only place it is true.
11//!
12//! Availability is decided by a connection attempt with a hard budget and no name
13//! resolution ([`super::probe`]); a service that did not answer in time is `unknown`,
14//! never "down".
15

26/// How long the connection attempt to a published address may take. Loopback either
27/// answers in microseconds or is not there; anything slower is a machine under load, and
28/// the snapshot says `unknown` rather than making a shell prompt wait for it.
29pub const PROBE_BUDGET: Duration = Duration::from_millis(80);
30
31/// One service this executable serves, and the constant that decides where.
32struct Descriptor {
33    id: &'static str,
34    title: &'static str,
35    path: &'static str,
36}
37
38/// Every service, in the order a snapshot reports them: the human surfaces first, the
39/// machine ones after.
40fn descriptors() -> [Descriptor; 8] {
41    [
42        Descriptor {
43            id: "cockpit",
44            title: "Cockpit",
45            path: cockpit::PREFIX,
46        },
47        Descriptor {
48            id: "docs",
49            title: "Documentation",
50            path: DOCS_MOUNT,
51        },
52        Descriptor {
53            id: "swagger",
54            title: "Swagger UI",
55            path: swagger::SWAGGER_PATH,
56        },
57        Descriptor {
58            id: "api",
59            title: "HTTP API",
60            // The prefix every capability route lives under; the routes themselves are the
61            // registry's and are not repeated anywhere. Without the trailing slash, because
62            // the surface is mounted without it (`web::discover::native_all` trims it) and
63            // this list is compared against what the server actually serves.
64            path: HttpExposure::PREFIX.trim_end_matches('/'),
65        },
66        Descriptor {
67            id: "events",
68            title: "Execution events",
69            // The stream a Cockpit or a terminal follows an execution over. It arrived with
70            // the execution control plane, after this list was written, which is exactly the
71            // gap `every_infrastructure_route_is_described` exists to close.
72            path: crate::http::events::PATH,
73        },
74        Descriptor {
75            id: "openapi",
76            title: "OpenAPI document",
77            path: swagger::SPEC_PATH,
78        },
79        Descriptor {
80            id: "mcp",
81            title: "MCP over HTTP",
82            path: mcp::PATH,
83        },
84        Descriptor {
85            id: "index",
86            title: "Home page",
87            path: "/",
88        },
89    ]
90}

92/// The services of the repository at `root`, with the address a running server published.
93///
94/// `probe` decides whether the address is contacted at all: a resolution that may not
95/// spend the time reports every service `unknown`, which is what it knows.
96pub fn resolve(root: &Path, local_half: &str, probe: bool) -> Vec<ServiceState> {
97    let base = published_url(root, local_half);
98    // One connection attempt for the whole server, not one per route: they are all the
99    // same socket, and seven probes would cost seven times as much to learn one thing.
100    let availability = match (&base, probe) {
101        (Some(url), true) => super::probe::reachable(url, PROBE_BUDGET),
102        (Some(_), false) => ServiceAvailability::Unknown,
103        (None, _) => ServiceAvailability::NotRunning,
104    };
105    descriptors()
106        .into_iter()
107        .map(|d| ServiceState {
108            id: d.id.into(),
109            title: d.title.into(),
110            path: d.path.into(),
111            url: base.as_ref().map(|base| join(base, d.path)),
112            availability,
113        })
114        .collect()
115}

117/// The address the repository's shared server published, from its lease. Reads one small
118/// file and contacts nothing; [`crate::lease::probe`] is the version that checks whether
119/// the server is really there, and it costs an HTTP round trip.
120pub fn published_url(root: &Path, local_half: &str) -> Option<String> {
121    crate::lease::LeaseFile::read(&crate::lease::lease_file(root, local_half))
122        .document()?
123        .url
124        .clone()
125}

127/// A base address and a route, joined without a doubled or missing slash.
128///
129/// ```
130/// use majordomus_cli::environment::services::join;
131/// assert_eq!(join("http://127.0.0.1:8741", "/cockpit"), "http://127.0.0.1:8741/cockpit");
132/// assert_eq!(join("http://127.0.0.1:8741/", "/"), "http://127.0.0.1:8741/");
133/// assert_eq!(join("http://127.0.0.1:8741", "/api/v1/"), "http://127.0.0.1:8741/api/v1/");
134/// ```
135pub fn join(base: &str, path: &str) -> String {
136    format!("{}{}", base.trim_end_matches('/'), path)
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142    use crate::http::openapi::infrastructure_routes;
143
144    /// The reason the paths are constants rather than strings: a route added to the
145    /// server's own infrastructure list without a name here would be a service the
146    /// Cockpit, the banner and the documentation never mention. This is what notices.
147    #[test]
148    fn every_infrastructure_route_is_described() {
149        let described: Vec<&str> = descriptors().iter().map(|d| d.path).collect();
150        for route in infrastructure_routes() {
151            assert!(
152                described.contains(&route.path.as_str()),
153                "the server serves {} and no service describes it",
154                route.path
155            );
156        }
157    }
158
159    #[test]
160    fn every_service_has_a_distinct_id_and_a_path_the_router_could_serve() {
161        let mut ids: Vec<&str> = descriptors().iter().map(|d| d.id).collect();
162        let count = ids.len();
163        ids.sort();
164        ids.dedup();
165        assert_eq!(ids.len(), count, "two services share an id");
166        for d in descriptors() {
167            assert!(d.path.starts_with('/'), "{} has a relative path", d.id);
168            assert!(!d.title.is_empty());
169        }
170    }
171
172    #[test]
173    fn without_a_lease_nothing_is_running_and_no_url_is_invented() {
174        let dir = tempfile::tempdir().expect("a temporary directory");
175        let services = resolve(dir.path(), ".ai/local", true);
176        assert!(!services.is_empty());
177        for s in &services {
178            assert_eq!(s.availability, ServiceAvailability::NotRunning);
179            assert_eq!(s.url, None, "{} invented an address", s.id);
180        }
181    }
182
183    #[test]
184    fn a_lease_of_another_schema_is_not_read() {
185        let dir = tempfile::tempdir().expect("a temporary directory");
186        let path = dir.path().join(".ai/local").join(crate::lease::LEASE_PATH);
187        std::fs::create_dir_all(path.parent().expect("a parent")).expect("a directory");
188        std::fs::write(
189            &path,
190            r#"{"schema":"something/else","url":"http://127.0.0.1:1"}"#,
191        )
192        .expect("a lease");
193        assert_eq!(published_url(dir.path(), ".ai/local"), None);
194    }
195
196    #[test]
197    fn a_published_address_becomes_every_services_url() {
198        let dir = tempfile::tempdir().expect("a temporary directory");
199        let path = dir.path().join(".ai/local").join(crate::lease::LEASE_PATH);
200        std::fs::create_dir_all(path.parent().expect("a parent")).expect("a directory");
201        std::fs::write(
202            &path,
203            format!(
204                r#"{{"schema":"{}","url":"http://127.0.0.1:8741"}}"#,
205                crate::lease::SCHEMA
206            ),
207        )
208        .expect("a lease");
209        // No probe: this asserts the addresses, not what answers at them.
210        let services = resolve(dir.path(), ".ai/local", false);
211        let cockpit = services
212            .iter()
213            .find(|s| s.id == "cockpit")
214            .expect("the cockpit");
215        assert_eq!(
216            cockpit.url.as_deref(),
217            Some("http://127.0.0.1:8741/cockpit")
218        );
219        assert_eq!(
220            cockpit.availability,
221            ServiceAvailability::Unknown,
222            "a resolution that did not probe knows nothing, and says so"
223        );
224    }
