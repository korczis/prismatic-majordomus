1//! The human-readable benchmark report, `docs/generated/economics.md`: a rendering of one
2//! [`EconomicsSummary`] and nothing else. Every number in it is a field of the summary,
3//! spelled for a reader; the report computes none, so it cannot disagree with the CLI, the
4//! API, MCP, the Cockpit or the site, which render the same summary.
5//!
6//! ```
7//! use majordomus_cli::economics::{report, summarize};
8//! let dir = tempfile::tempdir().unwrap();
9//! let text = report::markdown(&summarize(dir.path(), &Default::default()));
10//! assert!(text.contains("No verified total-token-savings claim is available"));
11//! ```

24/// A signed whole number, with no sign at all when it rounds to zero: a figure that is,
25/// to the precision shown, no difference never reads as `+0` or `-0`.
26fn signed_count(v: f64) -> String {
27    if !v.is_finite() {
28        return "n/a".into();
29    }
30    let whole = v.round();
31    if whole == 0.0 {
32        "0".into()
33    } else {
34        format!("{whole:+.0}")
35    }
36}

38/// A figure in its unit, as every surface spells it. A `ratio` is a signed percentage with
39/// one decimal ([`percent_text`]); any other unit is a signed whole number followed by the
40/// unit. Neither carries a sign when it rounds to zero, and a value that is not a number
41/// prints `n/a`, so no surface ever shows `-0.0%` or a figure nobody computed.
42///
43/// Every ratio the calculator derives from a pair is a reduction, `1 - treatment /
44/// control`: positive means Majordomus used less, and a negative reduction means
45/// Majordomus used more. The spelling keeps that sign and never flips it.
46///
47/// ```
48/// use majordomus_cli::economics::report::amount;
49/// assert_eq!(amount(0.2, "ratio"), "+20.0%");
50/// assert_eq!(amount(-0.1, "ratio"), "-10.0%", "a negative reduction stays negative");
51/// assert_eq!(amount(-0.0004, "ratio"), "0.0%", "never -0.0%");
52/// assert_eq!(amount(-1500.4, "tokens"), "-1500 tokens");
53/// assert_eq!(amount(0.3, "tokens"), "0 tokens");
54/// assert_eq!(amount(f64::NAN, "ratio"), "n/a");
55/// ```
56pub fn amount(v: f64, unit: &str) -> String {
57    if unit == "ratio" {
58        percent_text(v)
59    } else {
60        format!("{} {unit}", signed_count(v))
61    }
62}

64/// A metric's value as the report spells it: [`amount`] in the metric's unit, and an
65/// absent value the words `not measured`, so that an unmeasured metric never reads as a
66/// change of zero.
67///
68/// ```
69/// use majordomus_cli::economics::{model::EconomicsMetric, report};
70/// let m: EconomicsMetric = serde_json::from_value(serde_json::json!({
71///     "id": "effective_token_reduction", "title": "Effective token reduction",
72///     "class": "derived", "unit": "ratio", "formula": "1 - treatment / control",
73///     "value": 0.2, "status": "preliminary", "n": 3, "suite": "pilot"
74/// }))
75/// .unwrap();
76/// assert_eq!(report::value(&m), "+20.0%");
77/// assert_eq!(report::value(&EconomicsMetric { value: None, ..m }), "not measured");
78/// ```
79pub fn value(m: &EconomicsMetric) -> String {
80    match m.value {
81        None => "not measured".into(),
82        Some(v) => amount(v, &m.unit),
83    }
84}

86/// An interval as every surface spells it: the confidence level ([`level_text`], so that
87/// `9750` reads `97.5%` and is never rounded into a level the methodology did not
88/// declare), then both bounds in the metric's unit ([`amount`]). The bounds of a ratio are
89/// reductions, like the value they bracket.
90///
91/// ```
92/// use majordomus_cli::economics::{model::EconomicsInterval, report::interval_text};
93/// let i = EconomicsInterval {
94///     level_bp: 9750, low: -0.05, high: 0.3, method: "percentile bootstrap".into(),
95///     resamples: 2000, seed: 7,
96/// };
97/// assert_eq!(interval_text(&i, "ratio"), "97.5%: -5.0% to +30.0%");
98/// let tokens = EconomicsInterval { level_bp: 9500, low: 1200.0, high: 4800.0, ..i };
99/// assert_eq!(interval_text(&tokens, "tokens"), "95%: +1200 tokens to +4800 tokens");
100/// ```
101pub fn interval_text(i: &EconomicsInterval, unit: &str) -> String {
102    format!(
103        "{}: {} to {}",
104        level_text(i.level_bp),
105        amount(i.low, unit),
106        amount(i.high, unit)
107    )
108}

110/// An enum of the summary as a person reads it: its serialised name ([`wire`], the one the
111/// API and MCP answer with) with each underscore a space, so `no_evidence` reads `no
112/// evidence`. A surface never spells a state from `Debug`, which would print
113/// `NoEvidence`, a word the API never uses.
114///
115/// ```
116/// use majordomus_cli::economics::model::{EconomicsFreshness, EconomicsPairStatus};
117/// use majordomus_cli::economics::report::words;
118/// assert_eq!(words(&EconomicsFreshness::NoEvidence), "no evidence");
119/// assert_eq!(words(&EconomicsPairStatus::ControlFailed), "control failed");
120/// assert_eq!(words(&EconomicsPairStatus::Valid), "valid");
121/// ```
122pub fn words<T: Serialize>(value: &T) -> String {
123    wire(value).replace('_', " ")
124}

126fn cell(s: &str) -> String {
127    s.replace('|', "\\|").replace('\n', " ")
128}

130/// The whole report for one summary, as Markdown. It reads nothing but `s`, so the same
131/// summary always renders to the same bytes. When no methodology is declared (`present` is
132/// false) the report stops after the verdict, its unmet thresholds and the records that
133/// could not be read, because every later section would describe evidence that cannot
134/// exist. Free text placed in a table cell has its pipes escaped and its line breaks
135/// flattened, so a reason can never split a row. Every per-pair and aggregate token figure
136/// is a reduction, spelled by [`amount`]: a negative reduction means Majordomus used more
137/// tokens, and no figure is negated on the way to the page.
138///
139/// ```
140/// use majordomus_cli::economics::{report, summarize};
141/// let dir = tempfile::tempdir().unwrap();
142/// let s = summarize(dir.path(), &Default::default());
143/// let text = report::markdown(&s);
144/// assert!(text.starts_with("# Token economics: benchmark report"));
145/// assert!(text.contains(&s.verdict.statement));
146/// assert!(!s.present && !text.contains("## Metrics"));
147/// ```
148pub fn markdown(s: &EconomicsSummary) -> String {
149    let mut o = String::new();
150    let _ = writeln!(o, "# Token economics: benchmark report\n");
151    let _ = writeln!(o, "Generated from recorded evidence by `majordomus generate economics`. Every figure below is computed by one calculator (`apps/majordomus-cli/src/economics`) from the raw runs under `.ai/repo/benchmarks/economics/runs/`; nothing here is typed by hand.\n");
152    let _ = writeln!(o, "## Verdict\n");
153    let _ = writeln!(o, "{}\n", s.verdict.statement);
154    if !s.verdict.unmet.is_empty() {
155        let _ = writeln!(o, "The publication rule is not met:\n");
156        for u in &s.verdict.unmet {
157            let _ = writeln!(o, "- {u}");
158        }
159        let _ = writeln!(o);
160    }
161    if !s.diagnostics.is_empty() {
162        let _ = writeln!(o, "## Records that could not be read\n");
163        let _ = writeln!(o, "Nothing in them is counted anywhere below.\n");
164        for d in &s.diagnostics {
165            let _ = writeln!(o, "- {d}");
166        }
167        let _ = writeln!(o);
168    }
169    if !s.present {
170        return o;
171    }
172    let _ = writeln!(o, "## Question and methodology\n");
173    if let Some(q) = &s.question {
174        let _ = writeln!(o, "{q}\n");
175    }
176    let _ = writeln!(
177        o,
178        "Methodology version {}; primary metric `{}`. The methodology is `.ai/repo/benchmarks/economics/methodology.yaml`.\n",
179        s.methodology.unwrap_or(0),
180        s.primary_metric.clone().unwrap_or_default()
181    );
182    let _ = writeln!(o, "### Control and treatment\n");
183    for v in &s.variants {
184        let _ = writeln!(
185            o,
186            "- **{}** ({}, `{}`): {}",
187            v.title, v.role, v.id, v.description
188        );
189        if !v.excludes.is_empty() {
190            let _ = writeln!(o, "  Excludes: {}.", v.excludes.join(", "));
191        }
192    }
193    let _ = writeln!(o);
194    if let Some(p) = &s.publication {
195        let _ = writeln!(o, "### When a total-token claim may be published\n");
196        let _ = writeln!(
197            o,
198            "At least {} valid matched pairs; at least {} task categories with {} valid pairs each; at least {} repetitions of every task and variant; at least {} in 10000 attempted pairs valid; a bootstrap interval no wider than {} in 10000; evidence {}; no run recorded from a working tree with uncommitted changes.\n",
199            p.min_valid_pairs,
200            p.min_categories,
201            p.min_pairs_per_category,
202            p.min_repetitions,
203            p.min_valid_pair_rate_bp,
204            p.max_interval_width_bp,
205            if p.require_current { "current" } else { "of any age" }
206        );
207    }
208    let _ = writeln!(o, "## Evidence\n");
209    let _ = writeln!(o, "| suite | kind | freshness | runs | valid pairs | attempted | control failed | treatment failed | both failed | other | revisions | harness | models |");
210    let _ = writeln!(o, "|---|---|---|---|---|---|---|---|---|---|---|---|---|");
211    for v in &s.suites {
212        let _ = writeln!(
213            o,
214            "| {} | {} | {} | {} | {} | {} | {} | {} | {} | {} | {} | {} | {} |",
215            v.id,
216            v.kind,
217            words(&v.freshness),
218            v.runs,
219            v.pairs.valid,
220            v.pairs.attempted,
221            v.pairs.control_failed,
222            v.pairs.treatment_failed,
223            v.pairs.both_failed,
224            v.pairs.other,
225            v.revisions
226                .iter()
227                .map(|r| format!("`{}`", &r[..r.len().min(12)]))
228                .collect::<Vec<_>>()
229                .join(" "),
230            v.harnesses.join(", "),
231            v.models_reported.join(", ")
232        );
233    }
234    let _ = writeln!(o);
235    for v in s.suites.iter().filter(|v| v.freshness_detail.is_some()) {
236        let _ = writeln!(
237            o,
238            "- `{}`: {}",
239            v.id,
240            v.freshness_detail.as_deref().unwrap_or("")
241        );
242    }
243    let _ = writeln!(o, "\n## Metrics\n");
244    let _ = writeln!(
245        o,
246        "| metric | value | status | class | n | interval | what it is not |"
247    );
248    let _ = writeln!(o, "|---|---|---|---|---|---|---|");
249    for m in &s.metrics {
250        let class = match m.inputs {
251            Some(i) => format!("{} from {}", m.class.word(), i.word()),
252            None => m.class.word().to_string(),
253        };
254        let interval = m
255            .interval
256            .as_ref()
257            .map(|i| interval_text(i, &m.unit))
258            .unwrap_or_else(|| "—".into());
259        let _ = writeln!(
260            o,
261            "| `{}` | {} | {} | {} | {} | {} | {} |",
262            m.id,
263            value(m),
264            words(&m.status),
265            class,
266            m.n,
267            interval,
268            cell(m.not.as_deref().unwrap_or(""))
269        );
270    }
271    let _ = writeln!(o, "\nFormulas and warnings:\n");
272    for m in &s.metrics {
273        let _ = writeln!(o, "- `{}`: {}.", m.id, m.formula);
274        if let Some(i) = &m.interval {
275            let _ = writeln!(
276                o,
277                "  - interval: {}; {} resamples, seed {}.",
278                i.method, i.resamples, i.seed
279            );
280        }
281        for w in &m.warnings {
282            let _ = writeln!(o, "  - {w}");
283        }
284    }
285    let _ = writeln!(o);
286    if !s.segments.is_empty() {
287        let _ = writeln!(o, "## Segments\n");
288        let _ = writeln!(o, "Median per-pair total-token reduction over valid pairs, by segment: positive means Majordomus used fewer tokens, and a negative reduction means Majordomus used more tokens.\n");
289        let _ = writeln!(
290            o,
291            "| dimension | segment | valid pairs | median reduction | lowest | highest |"
292        );
293        let _ = writeln!(o, "|---|---|---|---|---|---|");
294        for g in &s.segments {
295            if let Some(d) = &g.token_reduction {
296                let _ = writeln!(
297                    o,
298                    "| {} | {} | {} | {} | {} | {} |",
299                    g.dimension,
300                    g.value,
301                    g.n,
302                    percent_text(d.median),
303                    percent_text(d.min),
304                    percent_text(d.max)
305                );
306            }
307        }
308        let _ = writeln!(o);
309    }
310    let live: Vec<_> = s.pairs.iter().collect();
311    if !live.is_empty() {
312        let _ = writeln!(o, "## Pairs\n");
313        let _ = writeln!(o, "Every declared pair, valid or not. Tokens are the provider-reported totals of every session of the run.\n");
314        let _ = writeln!(o, "| task | category | rep | status | control tokens | treatment tokens | token reduction | cost reduction | first-request overhead | reasons |");
315        let _ = writeln!(o, "|---|---|---|---|---|---|---|---|---|---|");
316        for p in live {
317            let _ = writeln!(
318                o,
319                "| {} | {} | {} | {} | {} | {} | {} | {} | {} | {} |",
320                p.task,
321                p.category,
322                p.repetition,
323                words(&p.status),
324                p.control_usage
325                    .as_ref()
326                    .map(|u| u.total.to_string())
327                    .unwrap_or_else(|| "—".into()),
328                p.treatment_usage
329                    .as_ref()
330                    .map(|u| u.total.to_string())
331                    .unwrap_or_else(|| "—".into()),
332                p.token_reduction
333                    .map(percent_text)
334                    .unwrap_or_else(|| "—".into()),
335                p.cost_reduction
336                    .map(percent_text)
337                    .unwrap_or_else(|| "—".into()),
338                p.first_request_overhead
339                    .map(|x| signed_count(x as f64))
340                    .unwrap_or_else(|| "—".into()),
341                cell(&p.reasons.join("; "))
342            );
343        }
344        let _ = writeln!(o, "\n\"Token reduction\" is `1 - treatment / control` on total tokens, and \"cost reduction\" the same on the harness's cost projection: positive means Majordomus used fewer tokens, and a negative reduction means Majordomus used more tokens. \"First-request overhead\" is not a reduction: it is the treatment's first-request input minus the control's, so positive means Majordomus added tokens before the model acted. An excluded pair shows its reductions so that the outlier rule's effect is visible; only valid pairs enter the metrics.\n");
345        let valid = s
346            .pairs
347            .iter()
348            .filter(|p| p.status == EconomicsPairStatus::Valid)
349            .count();
350        let _ = writeln!(
351            o,
352            "{valid} of {} declared pair(s) are valid.\n",
353            s.pairs.len()
354        );
355    }
356    if let Some(c) = &s.context {
357        let _ = writeln!(o, "## Context selection (deterministic)\n");
358        let _ = writeln!(
359            o,
360            "At revision `{}`, {} seeds (the issues of this repository's plan) were compiled by `majordomus devcontext` under its default budget of {} (bytes-over-four) tokens. Counted with {} ({}): {} candidate tokens (files the compiler judged relevant), {} selected; {} tokens reached in all. The compiler's own estimate of the selected tokens was {}; {} seed(s) exceed the budget when counted.\n",
361            &c.revision[..c.revision.len().min(12)],
362            c.seeds,
363            c.budget_tokens,
364            c.tokenizer.encoding,
365            c.tokenizer.implementation,
366            c.candidate_tokens,
367            c.selected_tokens,
368            c.considered_tokens,
369            c.estimated_selected_tokens,
370            c.seeds_over_budget_counted
371        );
372        let _ = writeln!(o, "This is context *selection*, not total token savings: it says what the compiler put in front of a worker out of what it found relevant, not what a session consumed.\n");
373        let unmeasured = s
374            .metrics
375            .iter()
376            .find(|m| m.id == CONTEXT_REDUCTION)
377            .filter(|m| m.status == EconomicsMetricStatus::NotMeasured);
378        if let Some(m) = unmeasured {
379            let _ = writeln!(
380                o,
381                "No figure is taken from this record: `{}` is not measured ({}).\n",
382                m.id,
383                m.warnings.join("; ")
384            );
385        }
386    }
387    if !s.history.is_empty() {
388        let _ = writeln!(o, "## History\n");
389        let _ = writeln!(
390            o,
391            "| suite | methodology | revision | recorded | metric | value | n |"
392        );
393        let _ = writeln!(o, "|---|---|---|---|---|---|---|");
394        for h in &s.history {
395            let _ = writeln!(
396                o,
397                "| {} | {} | `{}` | {} | `{}` | {} | {} |",
398                h.suite,
399                h.methodology,
400                &h.revision[..h.revision.len().min(12)],
401                h.at,
402                h.metric,
403                percent_text(h.value),
404                h.n
405            );
406        }
407        let _ = writeln!(o);
408    }
409    let _ = writeln!(o, "## Hypotheses\n");
410    let _ = writeln!(o, "Stated before any evidence existed, so that the evidence can refute them. They are not results.\n");
411    for h in &s.hypotheses {
412        let _ = writeln!(o, "- `{}` ({}): {}", h.id, h.status, h.statement);
413    }
414    let _ = writeln!(o, "\n## Reproduce\n");
415    let _ = writeln!(o, "```sh");
416    let _ = writeln!(o, "majordomus economics references        # every task fails on its start and passes on its reference; no model");
417    let _ = writeln!(
418        o,
419        "majordomus economics measure            # the context suite; deterministic, no model"
420    );
421    let _ = writeln!(o, "majordomus economics run --suite pilot  # live sessions; needs a Claude Code login and spends usage");
422    let _ = writeln!(
423        o,
424        "majordomus economics summary            # recompute every figure from the recorded runs"
425    );
426    let _ = writeln!(o, "majordomus economics explain effective_token_reduction");
427    let _ = writeln!(o, "```\n");
428    let _ = writeln!(o, "## Limitations\n");
429    let _ = writeln!(o, "- One fixture repository, small by design; a large repository is where context selection matters most, and it is not measured here.");
430    let _ = writeln!(o, "- One harness (Claude Code) and the model each suite names. Tokens are the provider's; counts from different providers are never compared directly.");
431    let _ = writeln!(o, "- Sessions are stochastic. A pair is one sample of each arm; only the distribution over many pairs says anything.");
432    let _ = writeln!(
433        o,
434        "- Cost is the harness's own projection at run time, not a bill."
435    );
436    let _ = writeln!(
437        o,
438        "- Context reduction is measured against what the compiler itself judged relevant."
439    );
440    let _ = writeln!(o, "- The context suite counts the corpus at the revision it recorded, not at the current one: a later change to the plan or to the files it selects from moves nothing until the suite is measured again.");
441    let _ = writeln!(o, "- The treatment is Majordomus as a user who installs only the shell tool gets it, and its generated bootstrap advises commands that installation cannot run (`majordomus worktree`). What a session spends on them is part of what that user gets, and it is measured as such, not corrected for.");
442    o
443}
