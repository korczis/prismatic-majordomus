1//! The one execution path.
2//!
3//! Every call of an executable capability — from the stdio MCP session, from `/mcp`, from
4//! an HTTP route, from the command line, from the Cockpit or from a benchmark runner —
5//! goes through [`CapabilityExecutor::execute`]: registry lookup, counters, the cache the
6//! capability's own policy asks for, then the handler. Transport adapters convert protocol
7//! to JSON and back and own nothing else, so instrumentation and caching apply to every
8//! transport at once and cannot drift apart. A second execution path is the defect this
9//! module exists to make impossible.
10//!
11//! # The cache key
12//!
13//! Process memory, bounded per capability by its [`CachePolicy`], keyed
14//! by three things: the canonical id, the input **normalised** (object keys sorted at every
15//! level, so `{"a":1,"b":2}` and `{"b":2,"a":1}` are one entry), and the registry
16//! fingerprint — which hashes every descriptor and every declarative object's content. Two
17//! processes over different repository states therefore never share an entry, and a changed
18//! layer never answers from an old one.
19//!
20//! Errors are never cached. Commands are never cached, and the registry refuses a
21//! descriptor that asks for it, so that is a build-time refusal rather than a runtime one.
22//!
23//! Two policies narrow the key further, for answers the registry fingerprint cannot see
24//! move. [`CachePolicy::Generation`] adds the context's [`Context::reading`]: an answer
25//! read off the index, including the git state it was read at, is kept for as long as that
26//! reading is the one asked. [`CachePolicy::Checkout`] adds that and the fingerprint of the
27//! checkout, taken again on every call (`capability::checkout`): HEAD, the staging index,
28//! the refs, and every changed or untracked path of the work tree with its size and
29//! modification time. A checkout git cannot fingerprint is not cached.
30//!
31//! # One handler per key at a time
32//!
33//! Concurrent calls that miss on the same key do not each run the handler: the first runs
34//! it, the others wait for it and are answered from what it stored. A screen that asks the
35//! same source from two panels at once, or two browsers that open one page together, pay
36//! for one computation. A handler that fails stores nothing, so a caller that waited on it
37//! runs the handler itself.
38//!
39//! # Switching it off
40//!
41//! `MAJORDOMUS_CAPABILITY_CACHE=off` in a process's environment makes every call run its
42//! handler and store nothing, whatever the capability's policy: the way to settle a doubt
43//! about a cached answer by asking without the cache (`project.cache-is-invisible`).
44//!
45//! # Invisibility
46//!
47//! A cached answer is byte-identical to a computed one; nothing in a response says which it
48//! was. `project.cache-is-invisible` requires it, and [`canonical_json`] is what makes the
49//! key well-defined enough for it to be true.
50//!
51//! ```
52//! use majordomus_cli::capability::executor::canonical_json;
53//!
54//! // the same input written two ways is one cache entry, at every level of nesting
55//! let one = serde_json::json!({ "b": 2, "a": { "y": 1, "x": 0 } });
56//! let other = serde_json::json!({ "a": { "x": 0, "y": 1 }, "b": 2 });
57//! assert_eq!(canonical_json(&one), canonical_json(&other));
58//!
59//! // and a different input is a different entry
60//! assert_ne!(canonical_json(&one), canonical_json(&serde_json::json!({ "b": 3 })));
61//! ```

75/// The executor of one registry.
76#[derive(Debug, Default)]
77pub struct CapabilityExecutor {
78    cache: Mutex<Cache>,
79    /// The handlers this executor ran: cache misses and uncached calls. A measurement of
80    /// this executor reads it; the process-wide counters also count every other executor
81    /// in the process, which a test running beside another reads as its own.
82    invocations: AtomicU64,
83    /// How many calls of a capability that writes the repository have succeeded here.
84    writes: AtomicU64,
85}
86
87#[derive(Debug, Default)]
88struct Cache {
89    entries: HashMap<CacheKey, CacheEntry>,
90    /// Insertion order per capability, for eviction.
91    order: HashMap<String, VecDeque<CacheKey>>,
92    /// The keys whose handler is running now, each with the lock its waiters queue on.
93    flights: HashMap<CacheKey, Arc<Mutex<()>>>,
94}
95
96#[derive(Debug, Clone, PartialEq, Eq, Hash)]
97struct CacheKey {
98    id: String,
99    input: String,
100    fingerprint: String,
101    /// What the policy adds to the registry fingerprint: empty for
102    /// [`CachePolicy::Process`], the context's reading for [`CachePolicy::Generation`], and
103    /// the reading with the checkout's fingerprint for [`CachePolicy::Checkout`].
104    state: String,
105}

107/// The environment switch that turns the capability cache off for a process.
108pub const SWITCH: &str = "MAJORDOMUS_CAPABILITY_CACHE";

110/// Whether this process was told to run every capability without its cache: `SWITCH`
111/// set to `off`.
112///
113/// ```
114/// use majordomus_cli::capability::executor::{bypassed, SWITCH};
115/// assert_eq!(bypassed(), std::env::var_os(SWITCH).is_some_and(|v| v == "off"));
116/// ```
117pub fn bypassed() -> bool {
118    std::env::var_os(SWITCH).is_some_and(|v| v == "off")
119}

133    /// Execute a capability by id with a JSON input. The one place a handler runs.
134    pub fn execute(&self, ctx: &Context, id: &str, input: Value) -> Result<Value, CapabilityError> {
135        let policy = ctx.registry.get(id).map(|c| c.cache);
136        self.run(ctx, id, input, policy)
137    }
138
139    /// The same call with the cache stepped over: the handler runs, whatever is stored.
140    ///
141    /// One caller: the execution engine. A cached answer is indistinguishable from a fresh
142    /// one — that is the point of the cache and the rule `project.cache-is-invisible` says
143    /// so — but an *execution* is not an answer. It is a record of work happening, watched
144    /// while it happens, and an execution answered from a cache would report no steps, no
145    /// progress and no output of its own while claiming to have succeeded: a story of work
146    /// that did not occur. So an execution always runs the handler, and nothing it produces
147    /// is put into the cache either, because the cache holds what calls returned and this
148    /// call is a different kind of thing.
149    pub fn execute_uncached(
150        &self,
151        ctx: &Context,
152        id: &str,
153        input: Value,
154    ) -> Result<Value, CapabilityError> {
155        self.run(ctx, id, input, Some(CachePolicy::Disabled))
156    }
157
158    fn run(
159        &self,
160        ctx: &Context,
161        id: &str,
162        input: Value,
163        policy: Option<CachePolicy>,
164    ) -> Result<Value, CapabilityError> {
165        Counters::bump(&COUNTERS.executions);
166        let policy = policy.ok_or_else(|| CapabilityError::NotFound(format!("capability {id}")))?;
167        let policy = if bypassed() {
168            CachePolicy::Disabled
169        } else {
170            policy
171        };
172        let Some(key) = key_of(ctx, id, &input, policy) else {
173            return self.invoke(ctx, id, input);
174        };
175        let ttl = policy.ttl_seconds();
176        if let Some(value) = self.lookup(&key, ttl) {
177            return Ok(value);
178        }
179        // a miss: take the key's flight, so that one handler runs for it at a time, and look
180        // again once it is ours — a caller that waited is answered by the one it waited for
181        let flight = {
182            let mut cache = lock(&self.cache);
183            Arc::clone(cache.flights.entry(key.clone()).or_default())
184        };
185        let _flight = flight.lock().unwrap_or_else(|e| e.into_inner());
186        if let Some(value) = self.peek(&key, ttl) {
187            return Ok(value);
188        }
189        let value = self.invoke(ctx, id, input);
190        let mut cache = lock(&self.cache);
191        if cache
192            .flights
193            .get(&key)
194            .is_some_and(|f| Arc::ptr_eq(f, &flight))
195        {
196            cache.flights.remove(&key);
197        }
198        let value = value?;
199        let max_entries = policy.max_entries().unwrap_or(1);
200        let Cache { entries, order, .. } = &mut *cache;
201        let order = order.entry(key.id.clone()).or_default();
202        if !entries.contains_key(&key) {
203            order.push_back(key.clone());
204            while order.len() > max_entries {
205                if let Some(old) = order.pop_front() {
206                    entries.remove(&old);
207                    Counters::bump(&COUNTERS.cache_evictions);
208                }
209            }
210        }
211        entries.insert(
212            key,
213            CacheEntry {
214                value: value.clone(),
215                stored: Instant::now(),
216            },
217        );
218        Ok(value)
219    }
220
221    /// The stored answer for `key` when it is still fresh, counted as a hit or a miss.
222    fn lookup(&self, key: &CacheKey, ttl: Option<u64>) -> Option<Value> {
223        let _guard = perf::phase(Phase::CacheLookup);
224        let mut cache = lock(&self.cache);
225        let fresh = cache
226            .entries
227            .get(key)
228            .map(|e| ttl.is_none_or(|ttl| e.stored.elapsed() < Duration::from_secs(ttl)));
229        match fresh {
230            Some(true) => {
231                Counters::bump(&COUNTERS.cache_hits);
232                return cache.entries.get(key).map(|e| e.value.clone());
233            }
234            Some(false) => {
235                cache.entries.remove(key);
236                Counters::bump(&COUNTERS.cache_evictions);
237                Counters::bump(&COUNTERS.cache_misses);
238            }
239            None => Counters::bump(&COUNTERS.cache_misses),
240        }
241        None
242    }
243
244    /// The stored answer for `key` once its flight is ours: what the handler a caller waited
245    /// for stored. Counted as a hit, since no handler runs for it.
246    fn peek(&self, key: &CacheKey, ttl: Option<u64>) -> Option<Value> {
247        let cache = lock(&self.cache);
248        let value = cache
249            .entries
250            .get(key)
251            .filter(|e| ttl.is_none_or(|ttl| e.stored.elapsed() < Duration::from_secs(ttl)))
252            .map(|e| e.value.clone());
253        if value.is_some() {
254            Counters::bump(&COUNTERS.cache_hits);
255        }
256        value
257    }
258
259    /// Run the handler: counted, timed, and its writes noted.
260    fn invoke(&self, ctx: &Context, id: &str, input: Value) -> Result<Value, CapabilityError> {
261        Counters::bump(&COUNTERS.handler_invocations);
262        self.invocations.fetch_add(1, Ordering::Relaxed);
263        let value = {
264            let _guard = perf::phase(Phase::HandlerExecution);
265            ctx.registry.dispatch(ctx, id, input)?
266        };
267        // counted on the one path every transport and every execution shares, so a write
268        // made from a page, a tool, a route or an execution is seen by `crate::live` alike
269        if ctx
270            .registry
271            .get(id)
272            .is_some_and(|c| c.execution.effect == crate::capability::Effect::RepositoryMutation)
273        {
274            self.writes.fetch_add(1, Ordering::SeqCst);
275        }
276        Ok(value)
277    }
278
279    /// How many calls of a capability whose effect is a repository mutation have succeeded
280    /// through this executor.
281    ///
282    /// A write to a tracked file moves no git control file, so a long-lived process that
283    /// follows the repository by those files would go on answering from the picture it had
284    /// before its own write. The executor outlives a reload, so this number is what
285    /// `crate::live` compares: when it has moved, the process wrote, and the next reading
286    /// is a new generation.
287    ///
288    /// ```
289    /// use majordomus_cli::capability::executor::CapabilityExecutor;
290    /// let executor = CapabilityExecutor::new();
291    /// assert_eq!(executor.writes(), 0, "an executor that has run nothing has written nothing");
292    /// ```
293    pub fn writes(&self) -> u64 {
294        self.writes.load(Ordering::SeqCst)
295    }
296
297    /// How many entries the cache holds, all capabilities together.
298    pub fn cached_entries(&self) -> usize {
299        lock(&self.cache).entries.len()
300    }
301
302    /// How many handlers this executor has run — its cache misses and uncached calls, never
303    /// another executor's. Dropping the cache does not reset it.
304    ///
305    /// ```
306    /// use majordomus_cli::capability::executor::CapabilityExecutor;
307    /// let e = CapabilityExecutor::default();
308    /// assert_eq!(e.handler_invocations(), 0);
309    /// ```
310    pub fn handler_invocations(&self) -> u64 {
311        self.invocations.load(Ordering::Relaxed)
312    }
313
314    /// Drop every cached entry.
315    pub fn clear(&self) {
316        let mut cache = lock(&self.cache);
317        cache.entries.clear();
318        cache.order.clear();
319    }
320}

322/// The key a call is kept under, or `None` when its policy keeps nothing — or keeps it per
323/// checkout and the checkout has no fingerprint.
324fn key_of(ctx: &Context, id: &str, input: &Value, policy: CachePolicy) -> Option<CacheKey> {
325    let state = match policy {
326        CachePolicy::Disabled => return None,
327        CachePolicy::Process { .. } => String::new(),
328        CachePolicy::Generation { .. } => format!("reading {}", ctx.reading),
329        CachePolicy::Checkout { .. } => {
330            let root = std::path::Path::new(&ctx.index.repository.root);
331            let checkout = super::checkout::fingerprint(root)?;
332            format!("reading {} checkout {checkout}", ctx.reading)
333        }
334    };
335    Some(CacheKey {
336        id: id.to_string(),
337        input: canonical_json(input),
338        fingerprint: ctx.registry.fingerprint().to_string(),
339        state,
340    })
341}

343fn lock(m: &Mutex<Cache>) -> std::sync::MutexGuard<'_, Cache> {
344    m.lock().unwrap_or_else(|e| e.into_inner())
345}

347/// The input as one canonical string: object keys sorted at every level, no whitespace,
348/// so that two inputs equal as data are equal as keys whatever order a client wrote them.
349///
350/// ```
351/// use majordomus_cli::capability::executor::canonical_json;
352/// use serde_json::json;
353/// assert_eq!(canonical_json(&json!({"b": 2, "a": {"d": 1, "c": [3, {"z": 0, "y": 1}]}})),
354///            canonical_json(&json!({"a": {"c": [3, {"y": 1, "z": 0}], "d": 1}, "b": 2})));
355/// assert_ne!(canonical_json(&json!([1, 2])), canonical_json(&json!([2, 1])), "arrays are ordered");
356/// ```
357pub fn canonical_json(v: &Value) -> String {
358    fn write(v: &Value, out: &mut String) {
359        match v {
360            Value::Object(m) => {
361                let mut keys: Vec<&String> = m.keys().collect();
362                keys.sort();
363                out.push('{');
364                for (i, k) in keys.iter().enumerate() {
365                    if i > 0 {
366                        out.push(',');
367                    }
368                    out.push_str(&serde_json::to_string(k).unwrap_or_default());
369                    out.push(':');
370                    write(&m[*k], out);
371                }
372                out.push('}');
373            }
374            Value::Array(a) => {
375                out.push('[');
376                for (i, x) in a.iter().enumerate() {
377                    if i > 0 {
378                        out.push(',');
379                    }
380                    write(x, out);
381                }
382                out.push(']');
383            }
384            other => out.push_str(&other.to_string()),
385        }
386    }
387    let mut out = String::new();
388    write(v, &mut out);
389    out
390}
