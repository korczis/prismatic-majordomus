1//! `majordomus models`: the terminal rendering of the model catalogue capabilities.
2//! Both subcommands execute the capability the CLI path names in-process — the
3//! catalogue is a file of the share directory, so the local answer is the truth — and
4//! own the rendering and nothing else: the filtering, the routing and every reason
5//! live in the capability, and `--format json` prints the same answer the HTTP route
6//! and the MCP tool serve.
7
8use serde_json::{json, Value};
9
10use crate::app::App;
11use crate::capability::CapabilityError;
12use crate::cli::{ModelsArgs, ModelsCommand, ModelsListArgs, ModelsRouteArgs, OutputFormat};
13use crate::error::{Error, Result};

15/// Run `majordomus models`.
16pub fn run(args: ModelsArgs) -> Result<u8> {
17    match args.command {
18        ModelsCommand::List(args) => list(args),
19        ModelsCommand::Route(args) => route(args),
20    }
21}

23fn execute(repo: &crate::cli::RepoArgs, path: &[&str], input: Value) -> Result<Value> {
24    let app = App::load(repo)?;
25    let ctx = &app.context;
26    let words: Vec<String> = path.iter().map(|w| w.to_string()).collect();
27    let id = ctx
28        .registry
29        .by_cli(&words)
30        .map(|c| c.id.as_str())
31        .ok_or_else(|| Error::Protocol {
32            reason: format!(
33                "no capability is exposed as `majordomus {}`",
34                path.join(" ")
35            ),
36        })?;
37    ctx.execute(id, input).map_err(map)
38}

40fn list(args: ModelsListArgs) -> Result<u8> {
41    let value = execute(
42        &args.repo,
43        &["models", "list"],
44        json!({ "vendor": args.vendor, "capability": args.capability, "id": args.id }),
45    )?;
46    match args.format {
47        OutputFormat::Json => println!(
48            "{}",
49            serde_json::to_string_pretty(&value).unwrap_or_default()
50        ),
51        OutputFormat::Text => {
52            let mut out = String::new();
53            for v in value["vendors"].as_array().into_iter().flatten() {
54                let credential = match v["credential_configured"] {
55                    Value::Bool(true) => "credential configured",
56                    Value::Bool(false) => "credential not configured",
57                    _ => "no credential declared",
58                };
59                out.push_str(&format!(
60                    "vendor     {}  {}  {}  {credential}\n",
61                    v["id"].as_str().unwrap_or("?"),
62                    v["title"].as_str().unwrap_or(""),
63                    v["inference"].as_str().unwrap_or("remote"),
64                ));
65            }
66            out.push_str(&format!("models     {} model(s)\n", value["count"]));
67            for m in value["models"].as_array().into_iter().flatten() {
68                let caps: Vec<&str> = m["capabilities"]
69                    .as_array()
70                    .into_iter()
71                    .flatten()
72                    .filter_map(Value::as_str)
73                    .collect();
74                out.push_str(&format!(
75                    "{}  {}  {}  ctx {}  [{}]{}\n",
76                    m["id"].as_str().unwrap_or("?"),
77                    m["vendor"].as_str().unwrap_or("?"),
78                    m["status"].as_str().unwrap_or("available"),
79                    m["context_window"]
80                        .as_u64()
81                        .map(|c| c.to_string())
82                        .unwrap_or_else(|| "?".into()),
83                    caps.join(","),
84                    m["note"]
85                        .as_str()
86                        .map(|n| format!("  — {n}"))
87                        .unwrap_or_default(),
88                ));
89            }
90            for d in value["diagnostics"].as_array().into_iter().flatten() {
91                out.push_str(&format!("finding    {}\n", d.as_str().unwrap_or("")));
92            }
93            println!("{}", out.trim_end());
94        }
95    }
96    Ok(0)
97}

99fn route(args: ModelsRouteArgs) -> Result<u8> {
100    let value = execute(
101        &args.repo,
102        &["models", "route"],
103        json!({
104            "require": args.require,
105            "min_context": args.min_context,
106            "vendor": args.vendor,
107            "local_only": args.local_only,
108            "model": args.model,
109        }),
110    )?;
111    match args.format {
112        OutputFormat::Json => println!(
113            "{}",
114            serde_json::to_string_pretty(&value).unwrap_or_default()
115        ),
116        OutputFormat::Text => {
117            let mut out = String::new();
118            match value["selected"].as_str() {
119                Some(id) => {
120                    out.push_str(&format!("selected   {id}\n"));
121                    if let Some(reason) = value["reason"].as_str() {
122                        out.push_str(&format!("reason     {reason}\n"));
123                    }
124                }
125                None => out.push_str("selected   nothing qualifies\n"),
126            }
127            let fallbacks: Vec<&str> = value["fallbacks"]
128                .as_array()
129                .into_iter()
130                .flatten()
131                .filter_map(Value::as_str)
132                .collect();
133            if !fallbacks.is_empty() {
134                out.push_str(&format!("fallbacks  {}\n", fallbacks.join(" → ")));
135            }
136            for e in value["excluded"].as_array().into_iter().flatten() {
137                out.push_str(&format!(
138                    "excluded   {}  — {}\n",
139                    e["model"].as_str().unwrap_or("?"),
140                    e["reason"].as_str().unwrap_or(""),
141                ));
142            }
143            println!("{}", out.trim_end());
144        }
145    }
146    Ok(0)
147}

149fn map(e: CapabilityError) -> Error {
150    Error::Protocol {
151        reason: e.to_string(),
152    }
