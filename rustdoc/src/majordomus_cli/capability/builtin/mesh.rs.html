1//! The `mesh` module: this process's mesh runtime, projected. Discovery (`mesh.status`,
2//! `mesh.nodes`, `mesh.identity`, `mesh.doctor`, `mesh.register`) reads the one
3//! [`crate::mesh::MeshRegistry`]; cooperation (`mesh.cooperation`, `mesh.peers`,
4//! `mesh.peer`, `mesh.state`, `mesh.events`, `mesh.verify` and the commands) reads and
5//! writes through the one [`crate::mesh::cooperation::Cooperation`] runtime. No capability
6//! holds state of its own, so the CLI, HTTP, OpenAPI, MCP and the Cockpit cannot disagree
7//! about a peer, a claim or a link (ADR 0050, ADR 0067).
8//!
9//! Discovery grants nothing. Cooperation is admitted per link — same repository, trusted
10//! key, compatible protocol — and what it carries is metadata about work: sessions,
11//! claims, handovers, reviews. There is no remote execution here for a linked peer to gain.

96            path: None,

100        });

103        return Ok(MeshIdentityReport {
104            path: Some(path.display().to_string()),
105            present: false,
106            identity: None,
107            error: None,

111        Ok(identity) => Ok(MeshIdentityReport {
112            path: Some(path.display().to_string()),
113            present: true,
114            identity: Some(identity.public),
115            error: None,
116        }),
117        Err(e) => Ok(MeshIdentityReport {
118            path: Some(path.display().to_string()),
119            present: true,
120            identity: None,
121            error: Some(e.to_string()),
122        }),
123    }

141/// The name a machine's allowlist entry carries: its node id, its host name and its
142/// platform, the shape a person writes by hand ("641bdb94 — a MacBook (macos aarch64)").
143fn machine_label(identity: &PublicIdentity) -> String {
144    let host = std::process::Command::new("hostname")

148        .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string())
149        .filter(|h| !h.is_empty())
150        .unwrap_or_else(|| "this machine".into());
151    format!(
152        "{} — {host} ({} {})",
153        identity.display_name,
154        std::env::consts::OS,
155        std::env::consts::ARCH

159fn mesh_enrol(ctx: &Context, _: Empty) -> Result<MeshEnrolment, CapabilityError> {
160    let root = std::path::Path::new(&ctx.index.repository.root);
161    let object = ctx
162        .index
163        .objects
164        .iter()
165        .find(|o| o.kind == crate::mesh::KIND)
166        .ok_or_else(|| {
167            CapabilityError::NotFound(
168                "this repository declares no mesh: `majordomus init --extend` seeds .ai/repo/mesh/mesh.yaml".into(),
169            )
170        })?;
171    let relative = object.provenance.path.clone();
172    let path = root.join(&relative);
173    let identity_path = default_identity_path().ok_or_else(|| {
174        CapabilityError::Refused(
175            "no HOME and no XDG_STATE_HOME: nowhere to keep this machine's node identity".into(),
176        )
177    })?;
178    let identity = NodeIdentity::load_or_create(&identity_path)
179        .map_err(|e| CapabilityError::Refused(e.to_string()))?;
180    let key = identity.public.public_key.clone();
181    // read, edit and write as one step: the index has read and parsed this declaration
182    // already, so in practice only the write can fail, and every failure names the file
183    let added = std::fs::read_to_string(&path)
184        .map_err(io_reason)
185        .and_then(|text| crate::mesh::enrol::enrol(&text, &key, &machine_label(&identity.public)))
186        .and_then(|edited| write_enrolled(&path, edited))
187        .map_err(|e| CapabilityError::Internal(format!("{relative}: {e}")))?;
188    Ok(if added {
189        MeshEnrolment {
190            next: format!(
191                "commit {relative}, so every machine that pulls it trusts this one; restart this checkout's server (`majordomus serve stop`, then `majordomus serve ensure`) to open the mesh here"
192            ),
193            path: relative,
194            key,
195            added: true,
196        }
197    } else {
198        MeshEnrolment {
199            path: relative,
200            key,
201            added: false,
202            next: "nothing: this machine's key is listed already".into(),
203        }
204    })
205}

207/// Write an enrolled declaration over `path`: whether a key was added, `false` when the
208/// declaration listed it already and nothing was written.
209fn write_enrolled(path: &std::path::Path, edited: Option<String>) -> Result<bool, String> {
210    match edited {
211        None => Ok(false),
212        Some(text) => std::fs::write(path, text).map(|()| true).map_err(io_reason),
213    }

232fn mesh_doctor(ctx: &Context, _: Empty) -> Result<MeshDoctorReport, CapabilityError> {
233    let root = std::path::Path::new(&ctx.index.repository.root);
234    // The runtime's verdict is reported only where a shared server made one: in the
235    // command line's process nothing activates the mesh, and that absence is not news.
236    let runtime = ctx.mesh.decided().then(|| ctx.mesh.status());
237    let resolved = resolved_declaration(ctx);
238    let source = crate::mesh::doctor::declaration_source(&resolved);
239    let mut report =
240        crate::mesh::doctor::doctor_at(resolved.declaration, Some(root), runtime.as_ref());
241    if let Some(check) = source {
242        report.checks.push(check);
243    }
244    // the journal is judged only where a runtime holds one
245    if let Some(cooperation) = ctx.mesh.cooperation() {
246        report.checks.push(crate::mesh::doctor::journal_check(
247            &cooperation.journal().tallies(),
248        ));
249    }
250    // one conjunction over every check, so neither added check's failure can be lost
251    report.ok = report.checks.iter().all(|c| c.ok);
252    Ok(report)
253}

487fn rank(l: StreamLiveness) -> u8 {
488    match l {
489        StreamLiveness::Own => 0,
490        StreamLiveness::Live => 1,
491        StreamLiveness::Expired => 2,
492    }

495/// Build the peer tree from a running cooperation, narrowed and viewed as `query` asks. The
496/// summary view keeps every runtime and only the active sessions, each with its claims
497/// counted rather than listed and its intent cut; the full view is every session with every
498/// claim. Either way the digest is the whole fold's.
499pub fn peer_tree(c: &Cooperation, query: &Query) -> PeerTree {
500    let status = c.status();
501    let whole = c.state();
502    let state = page::narrow(&whole, query);
503    let summary = query.view() == View::Summary;
504    let links: BTreeMap<String, PeerView> = status
505        .peers
506        .iter()
507        .map(|p| (p.runtime.clone(), p.clone()))
508        .collect();
509    let own = status.runtime.clone().unwrap_or_default();
510    let own_node = own.get(..32).unwrap_or_default().to_string();
511
512    // Runtimes: this one, every linked one, and every one the journal has heard of.
513    let mut runtimes: BTreeMap<String, RuntimeNode> = BTreeMap::new();
514    let mut touch = |key: &str, liveness: StreamLiveness, age: Option<u64>| {
515        let node = runtimes
516            .entry(key.to_string())
517            .or_insert_with(|| RuntimeNode {
518                runtime: key.to_string(),
519                this_runtime: key == own,
520                liveness,
521                last_beat_ms: age,
522                link: None,
523                newest_claim: None,
524                sessions: Vec::new(),
525            });
526        if rank(liveness) < rank(node.liveness) {
527            node.liveness = liveness;
528        }
529        if let Some(a) = age {
530            node.last_beat_ms = Some(node.last_beat_ms.map_or(a, |b| b.min(a)));
531        }
532    };
533    for (stream, liveness, age) in c.streams() {
534        touch(&stream.runtime_key(), liveness, age);
535    }
536    for key in links.keys() {
537        touch(key, StreamLiveness::Expired, None);
538    }
539    for (key, node) in runtimes.iter_mut() {
540        node.link = links.get(key).cloned();
541    }
542    for claim in state.claims.iter().filter(|cl| cl.state.is_live()) {
543        // every claim's runtime is a node: each stream of the journal was touched above
544        runtimes.entry(claim.runtime.clone()).and_modify(|runtime| {
545            let newer = runtime
546                .newest_claim
547                .as_ref()
548                .is_none_or(|b| claim.acquired_lamport > b.acquired_lamport);
549            if newer {
550                runtime.newest_claim = Some(ClaimBrief {
551                    key: claim.key.clone(),
552                    session: claim.session.clone(),
553                    issue: claim.issue.clone(),
554                    intent: claim.intent.as_deref().map(page::cut),
555                    paths: claim.scope.len(),
556                    mode: claim.mode,
557                    acquired_lamport: claim.acquired_lamport,
558                });
559            }
560        });
561    }
562    for session in &state.sessions {
563        if summary && session.state != SessionState::Active {
564            continue;
565        }
566        let claims: Vec<ClaimView> = state
567            .claims
568            .iter()
569            .filter(|cl| cl.session == session.key)
570            .cloned()
571            .collect();
572        let claims_held = claims.iter().filter(|cl| cl.state.is_live()).count();
573        let mut info = session.info.clone();
574        if summary {
575            info.intent = info.intent.as_deref().map(page::cut);
576        }
577        let reviews = state
578            .reviews
579            .iter()
580            .filter(|r| r.session == session.key)
581            .map(|r| r.key.clone())
582            .collect();
583        if let Some(runtime) = runtimes.get_mut(&session.runtime) {
584            runtime.sessions.push(SessionNode {
585                key: session.key.clone(),
586                info,
587                state: session.state,
588                claims: if summary { Vec::new() } else { claims },
589                claims_held,
590                reviews,
591            });
592        }
593    }
594
595    let own_name = status
596        .peers
597        .iter()
598        .find(|p| p.local)
599        .map(|p| p.name.clone());
600    let mut machines: BTreeMap<String, MachineNode> = BTreeMap::new();
601    for (key, runtime) in runtimes {
602        let node = key.get(..32).unwrap_or_default().to_string();
603        let name = runtime
604            .link
605            .as_ref()
606            .map(|l| l.name.clone())
607            .or_else(|| (node == own_node).then(|| own_name.clone()).flatten())
608            .unwrap_or_else(|| node.chars().take(8).collect());
609        machines
610            .entry(node.clone())
611            .or_insert_with(|| MachineNode {
612                node: node.clone(),
613                name,
614                local: node == own_node,

629        refused: status.refused,
630        digest: Some(whole.digest),
631    }

862/// The answer of `mesh.events`.
863#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
864pub struct EventList {
865    /// Whether cooperation runs.
866    pub active: bool,
867    /// How many events this answer carries.
868    pub count: usize,
869    /// The `after` of the next page of a walk by stamp: the stamp of the last event this
870    /// page carries, or the `after` that was asked when it carries none. A page never ends
871    /// inside a run of equal stamps (I2140).
872    pub lamport: u64,
873    /// The `cursor` of the next page: pass it back as `cursor` and the walk resumes exactly
874    /// after what this page returned (I2733). Absent from an answer to a walk by `after`.
875    #[serde(default, skip_serializing_if = "Option::is_none")]
876    pub cursor: Option<String>,
877    /// The events, in Lamport order.
878    pub events: Vec<MeshEvent>,
879}
880
881fn mesh_events(ctx: &Context, input: MeshEventsInput) -> Result<EventList, CapabilityError> {
882    let Some(c) = ctx.mesh.cooperation() else {
883        return Ok(EventList {
884            active: false,
885            count: 0,
886            lamport: 0,
887            cursor: None,
888            events: Vec::new(),
889        });
890    };
891    let limit = input.limit.unwrap_or(100).clamp(1, 1000) as usize;
892    if input.after.is_some() && input.cursor.is_some() {
893        return Err(CapabilityError::InvalidInput(
894            "after and cursor are two different walks: pass one of them".into(),
895        ));
896    }
897    let after = input.after.unwrap_or(0);
898    let (events, cursor) = if input.newest == Some(true) {
899        // the whole tail above `after`, in order, and then its end: the journal is in memory
900        // and bounded by its own retention, so this is a walk and not a second index
901        let head = c.journal().head_cursor();
902        let mut all = c.journal().events_after(after, usize::MAX);
903        let from = all.len().saturating_sub(limit);
904        all.drain(..from);
905        (all, Some(head))
906    } else if input.after.is_some() {
907        (c.journal().events_after(after, limit), None)
908    } else {
909        let from = match &input.cursor {
910            Some(text) => Some(text.parse::<Cursor>().map_err(cursor_refusal)?),
911            None => None,
912        };
913        let (events, next) = c
914            .journal()
915            .events_since(from.as_ref(), limit)
916            .map_err(cursor_refusal)?;
917        (events, Some(next))
918    };
919    Ok(EventList {
920        active: true,
921        count: events.len(),
922        lamport: events.last().map_or(after, |e| e.lamport),
923        cursor: cursor.map(|c| c.to_string()),
924        events,
925    })
926}
