1//! Navigation, derived. The sidebar's catalogues are not a list in this file: the
2//! capability modules come from the registry, the object kinds come from `repository.info`,
3//! and the graphs come from the derivations. A module added to `compose_modules!`, a kind
4//! added to `share/kinds.yaml`, a graph added to the derivation table — each appears here
5//! with no edit to the Cockpit.
6//!
7//! The kinds are *asked for* rather than read: ADR 0012 says no page reads the index
8//! directly, because a reader that steps around the executor is outside the cache, the
9//! counters and the validation every other caller passes through.
10//!
11//! What *is* written here is the areas: Overview, Capabilities, Commands, Executions,
12//! Objects, Directories, Graphs, Continuity, Worktrees, Health, Quality, Artifacts,
13//! Design, API. Those are concepts rather than
14//! entities, they change when the Cockpit's own shape changes, and deriving them from
15//! anything would be deriving them from a list of exactly themselves.
16//!
17//! Every section is presented in the canonical order (`crate::order`). The order a section
18//! is *built* in is an accident of its source — the order the modules were composed in, the
19//! order the graph derivations are declared in — and an accident is not a reading order.
20//! The order is applied to the sections as a whole, so a section added later is ordered
21//! without being told to be.
22//!
23//! One section is grouped rather than flat: the capability modules sit under the
24//! operational area they serve. That area is not written here either. A feature declares
25//! the modules it is built from and the areas it serves; the product model resolves the two
26//! into an area per module, and this file asks for it. A module no feature places is shown
27//! under no heading, which the canonical order puts last, and the product model reports it.

34/// One of the Cockpit's areas. A page declares the area it belongs to and the navigation
35/// marks it.

40    /// The capability explorer and the runner.

47    Directories,
48    /// The derived graphs.
49    Graphs,

51    Executions,
52    /// What this checkout's lifecycle is holding.
53    Continuity,
54    /// The branch-to-worktree topology of the repository.
55    Worktrees,
56    /// The discovered nodes of the mesh, and the machinery that observes them.
57    Mesh,
58    /// The declared model catalogue and its routing.
59    Models,
60    /// Token economics, measured.
61    Economics,
62    /// The health report.
63    Health,
64    /// What the crate's own public surface is held to.
65    Quality,
66    /// What the generator writes.
67    Artifacts,
68    /// What the public contract did since the last release, and the version it requires.
69    Release,
70    /// The design system: what every surface of this tool is rendered with.
71    Design,
72    /// The HTTP and MCP surfaces.
73    Api,
74    /// A page that belongs to no area (search results, an error).
75    None,
76}

200/// One entry.
201#[derive(Debug, Clone)]
202pub struct Item {
203    /// What the reader sees.
204    pub label: String,
205    /// Where it goes.
206    pub href: String,
207    /// The area it belongs to.
208    pub area: Area,
209    /// The heading it sits under inside its section, when the section is grouped. Derived,
210    /// never written here: the capability modules are grouped by the operational area the
211    /// features that name them serve.
212    pub group: Option<String>,
213    /// How many things are behind it, when the number is a fact and not decoration.
214    pub count: Option<usize>,
215    /// Whether this is the page being shown.
216    pub current: bool,
217}

219/// A heading and its entries.
220#[derive(Debug, Clone)]
221pub struct Section {
222    /// The heading.
223    pub title: String,
224    /// The entries, in a deterministic order.

226}

241/// What the layer holds, as `repository.info` answers it: the object count and the kinds
242/// with theirs.
243///
244/// A capability that cannot answer leaves the navigation without its object counts rather
245/// than without a navigation: the sidebar is how a person reaches the page that would say
246/// what went wrong, so it is the last thing that should fail with the thing it reports on.
247/// The two fields of `repository.info` this file reads; the rest of the report is the
248/// health page's subject, and taking only these keeps the navigation's dependency on the
249/// capability to what it actually shows.
250#[derive(Default, serde::Deserialize)]
251struct Held {
252    objects: usize,
253    kinds: std::collections::BTreeMap<String, usize>,
254}
255
256fn held(ctx: &Context) -> Held {
257    let Ok(value) = ctx.execute("repository.info", serde_json::json!({})) else {
258        return Held::default();
259    };
260    serde_json::from_value::<Held>(value).unwrap_or_default()
261}

263/// Build the navigation for a request: the areas, then the catalogues derived from the
264/// registry, the index and the graph derivations. `here` is the request path, so the
265/// current entry can be marked without a page saying which it is.
266pub fn build(ctx: &Context, here: &str) -> Navigation {
267    let summary = ctx.registry.summary();
268    // What the layer holds is asked for, not opened. `repository.info` answers how many
269    // objects there are and which kinds they have, and asking it is what keeps the
270    // navigation a projection instead of a second reader of the index (ADR 0012). One
271    // call answers both catalogues below.
272    let held = held(ctx);
273    // the counts are facts of this context, decided per area: how many things are behind
274    // an entry is not part of what an area is
275    let count = |a: Area| -> Option<usize> {
276        match a {
277            Area::Capabilities => Some(summary.total),
278            Area::Objects => Some(held.objects),
279            Area::Graphs => Some(graph::ids().len()),
280            Area::Api => Some(summary.http_routes),
281            _ => None,
282        }
283    };
284    let areas = Section {
285        title: "Cockpit".into(),
286        items: areas()
287            .iter()
288            .map(|a| item(a.label, a.href, a.area, count(a.area), here))
289            .collect(),
290    };
291
292    // the executable's own modules: what a capability belongs to, from the registry
293    let modules = Section {
294        title: "Capability modules".into(),
295        items: ctx
296            .registry
297            .modules()
298            .filter(|m| m.source != ModuleSource::Declarative)
299            .map(|m| Item {
300                label: m.title.clone(),
301                href: format!(
302                    "/cockpit/capabilities?module={}",
303                    percent_encode(m.id.as_str())
304                ),
305                area: Area::Capabilities,
306                // The module's own area, derived by the product model from the features
307                // that name it, shown under the name the Why catalogue gives it. Neither
308                // the grouping nor the heading is written in this file.
309                group: ctx
310                    .product
311                    .module_area(m.id.as_str())
312                    .and_then(|id| ctx.why.areas().iter().find(|a| a.id == id))
313                    .map(|a| a.title.clone()),
314                count: Some(m.capabilities),
315                current: false,
316            })
317            .collect(),
318    };
319
320    // the kinds of the layer: what the repository declares, not what this code knows
321    let kinds = Section {
322        title: "Object kinds".into(),
323        items: held
324            .kinds
325            .iter()
326            .map(|(kind, count)| Item {
327                label: kind.to_string(),
328                href: crate::entity::kind_route(kind),
329                area: Area::Objects,
330                group: None,
331                count: Some(*count),
332                current: false,
333            })
334            .collect(),
335    };
336
337    let graphs = Section {
338        title: "Graphs".into(),
339        items: graph::ids()
340            .into_iter()
341            .map(|id| Item {
342                label: id.to_string(),
343                href: format!("/cockpit/graphs/{id}"),
344                area: Area::Graphs,
345                group: None,
346                count: None,
347                current: here == format!("/cockpit/graphs/{id}"),
348            })
349            .collect(),
350    };
351
352    Navigation {
353        sections: [areas, modules, kinds, graphs]
354            .into_iter()
355            .filter(|s| !s.items.is_empty())
356            .map(alphabetical)
357            .collect(),
358    }
359}

361/// One section's entries in the canonical order: by label, case-folded so that `API` sits
362/// with `Artifacts` rather than ahead of every lowercase kind, digit runs by value, and the
363/// href behind it to break a tie between two entries a reader would call the same.
364///
365/// The comparator is `crate::order`'s, not this file's. It was this file's, it was the only
366/// case-folded comparator in the crate, and every other surface sorted by raw bytes instead.
367fn alphabetical(mut section: Section) -> Section {
368    crate::order::canonical(&mut section.items);
369    section
370}
371
372impl crate::order::Ordered for Item {
373    fn order_key(&self) -> crate::order::OrderKey<'_> {
374        match &self.group {
375            Some(group) => crate::order::OrderKey::grouped(group, &self.label, &self.href),
376            None => crate::order::OrderKey::plain(&self.label, &self.href),
377        }
378    }
379}
380
381fn item(label: &str, href: &str, area: Area, count: Option<usize>, here: &str) -> Item {
382    Item {
383        label: label.into(),
384        href: href.into(),
385        area,
386        group: None,
387        count,
388        current: here == href,
389    }
390}

401    #[test]
402    fn the_module_catalogue_is_the_registrys_and_not_a_list_here() {
403        let repo = repository();
404        let ctx = repo.context().expect("a context");
405        let nav = build(&ctx, "/cockpit");
406        let modules = nav
407            .sections()
408            .iter()
409            .find(|s| s.title == "Capability modules")
410            .expect("a module section");
411        let labels: Vec<&str> = modules.items.iter().map(|i| i.label.as_str()).collect();
412        let expected = ctx
413            .registry
414            .modules()
415            .filter(|m| m.source != ModuleSource::Declarative)
416            .count();
417        assert_eq!(labels.len(), expected);
418        assert!(expected > 0, "the builtin registry composes modules");
419    }

421    #[test]
422    fn a_grouped_section_keeps_its_groups_contiguous_and_the_ungrouped_last() {
423        let repo = repository();
424        let ctx = repo.context().expect("a context");
425        let nav = build(&ctx, "/cockpit");
426        for section in nav.sections() {
427            let mut seen: Vec<&str> = Vec::new();
428            let mut ungrouped = false;
429            for item in &section.items {
430                match item.group.as_deref() {
431                    Some(g) => {
432                        assert!(
433                            !ungrouped,
434                            "a grouped entry follows an ungrouped one in '{}': the canonical \
435                             order puts the ungrouped tail last",
436                            section.title
437                        );
438                        if seen.last().copied() != Some(g) {
439                            assert!(
440                                !seen.contains(&g),
441                                "the group '{g}' appears twice in '{}': its members are not \
442                                 contiguous, so one pass cannot render its heading once",
443                                section.title
444                            );
445                            seen.push(g);
446                        }
447                    }
448                    None => ungrouped = true,
449                }
450            }
451        }
452    }
