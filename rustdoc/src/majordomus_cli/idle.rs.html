1//! When a server that no client owns is idle, and when it leaves (I4007).
2//!
3//! A server `serve ensure` starts for a checkout carries an idle life: the declared
4//! `server.idle_seconds:` in `.ai/repo/policy.yaml` (900 when nothing is declared), passed to
5//! it as `serve --idle`. It is meant to end when nobody uses it. On 2026-10-22 fifteen of them
6//! had run for up to eight hours on one Mac with no peer attached, each holding a port, a
7//! lease and 60-450 MB, because the idle timer counted *every* HTTP request as use — and the
8//! machine's servers talk to each other all the time. Every server on a machine hears the
9//! mesh's multicast and links to the others (`POST /api/v1/mesh/link/hello`, `.../sync`), a
10//! board gather from any checkout reads every sibling's board (`GET /api/v1/peers?
11//! checkouts=this`), and `serve status`, `serve ensure` and the gather probe every lease's
12//! server (`GET /`). Fifteen servers kept each other alive.
13//!
14//! Idle is decided here from state the server already holds, never from a guess:
15//!
16//! - no MCP session attached and no row on this checkout's peer board;
17//! - no execution queued, starting, running or cancelling;
18//! - no live `/events` channel open;
19//! - no request from a client answered since the last look: a request another Majordomus
20//!   process sends to keep the machine's servers in touch ([`counts_as_use`]) is not one.
21//!
22//! The lease is not part of the question. Every `serve` that is still running holds its own
23//! checkout's lease — the election ends any other at once — and a server whose lease was
24//! taken over ends when its last session has been handed on (I2127). A server started without
25//! an idle life (plain `serve`, a hosted deployment, the server `majordomus mcp` hosts for its
26//! client's stdio session) never idle-exits: its owner is whoever started it.
27//!
28//! When the bound is reached the server logs why ([`IdleExit`]), runs its ordered stop —
29//! sessions and episodes closed, listeners closed, mesh drained, lease released — and `serve`
30//! returns 0. That is the server leaving on its own, which ADR 0200 permits; nothing here
31//! signals or kills another process.
32//!
33//! The lifecycle, as the serve loop runs it once a second:
34//!
35//! ```
36//! use std::time::{Duration, Instant};
37//! use majordomus_cli::idle::{counts_as_use, Activity, IdleTimer};
38//!
39//! let start = Instant::now();
40//! let mut timer = IdleTimer::new(Duration::from_secs(3), start);
41//! let at = |s| start + Duration::from_secs(s);
42//! // every second another server's mesh sync arrives: it is not a client
43//! let sync = counts_as_use("POST", "/api/v1/mesh/link/sync", false);
44//! let look = Activity { client_request: sync, ..Activity::default() };
45//! assert!(timer.observe(&look, at(1)).is_none());
46//! assert!(timer.observe(&look, at(2)).is_none());
47//! // three idle seconds: the server logs why, stops in order and exits 0
48//! let why = timer.observe(&look, at(3)).unwrap();
49//! assert!(why.to_string().contains("machine traffic"));
50//! ```

52use std::fmt;
53use std::time::{Duration, Instant};

55/// The header this executable's own probes carry (`lease::probe` and every caller of it), so
56/// that a server tells "another Majordomus process asked whether I am here" apart from a
57/// person or a client reading its home page. A probe from an older executable lacks it and
58/// still counts as use, which errs on the side of keeping a server.
59pub const PROBE_HEADER: &str = "X-Majordomus-Probe";
60
61/// The paths of the mesh's runtime-to-runtime transport: a link's hello and sync, and a
62/// registration with a rendezvous hub. Only other runtimes send these.
63const MESH_TRANSPORT: [&str; 3] = [
64    crate::mesh::link::HELLO_PATH,
65    crate::mesh::link::SYNC_PATH,
66    crate::mesh::rendezvous::REGISTER_PATH,
67];
68
69/// Whether an HTTP request is a client using this server, or another Majordomus process
70/// keeping the machine's servers in touch — which must not keep an idle server alive.
71///
72/// Machine traffic, and nothing else, is:
73///
74/// - the mesh transport: `POST /api/v1/mesh/link/hello`, `.../link/sync`, `/api/v1/mesh/register`;
75/// - a probe: `GET` or `HEAD /` carrying [`PROBE_HEADER`];
76/// - a sibling's board gather: `GET /api/v1/peers?checkouts=this`, the one question a server
77///   asks another checkout's server for its own board.
78///
79/// `target` is the request target as received, path and query.
80///
81/// ```
82/// use majordomus_cli::idle::counts_as_use;
83/// // another runtime linking, a probe, a sibling reading this board: not use
84/// assert!(!counts_as_use("POST", "/api/v1/mesh/link/sync", false));
85/// assert!(!counts_as_use("GET", "/", true));
86/// assert!(!counts_as_use("GET", "/api/v1/peers?checkouts=this", false));
87/// // a person on the home page, a client of the REST API, an MCP request: use
88/// assert!(counts_as_use("GET", "/", false));
89/// assert!(counts_as_use("GET", "/api/v1/peers", false));
90/// assert!(counts_as_use("POST", "/mcp", false));
91/// assert!(counts_as_use("GET", "/api/v1/mesh/state", false));
92/// ```
93pub fn counts_as_use(method: &str, target: &str, probe_header: bool) -> bool {
94    let (path, query) = match target.split_once('?') {
95        Some((path, query)) => (path, Some(query)),
96        None => (target, None),
97    };
98    if MESH_TRANSPORT.contains(&path) {
99        return false;
100    }
101    let read = method.eq_ignore_ascii_case("GET") || method.eq_ignore_ascii_case("HEAD");
102    if read && path == "/" && probe_header {
103        return false;
104    }
105    let sibling_board = read
106        && path == "/api/v1/peers"
107        && query.is_some_and(|q| q.split('&').any(|pair| pair == "checkouts=this"));
108    !sibling_board
109}

111/// What a server holds at one look: the state idleness is decided from.
112///
113/// ```
114/// use majordomus_cli::idle::Activity;
115/// assert_eq!(Activity::default().busy_because(), None);
116/// let busy = Activity { executions: 1, ..Activity::default() };
117/// assert_eq!(busy.busy_because().as_deref(), Some("1 execution(s) queued or running"));
118/// ```
119#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
120pub struct Activity {
121    /// MCP sessions attached, or rows on this checkout's peer board, whichever is more.
122    pub peers: usize,
123    /// Executions queued, starting, running or cancelling.
124    pub executions: usize,
125    /// Live `/events` channels open.
126    pub channels: usize,
127    /// Whether a client's request was answered since the last look ([`counts_as_use`]).
128    pub client_request: bool,
129}
130
131impl Activity {
132    /// What keeps the server in use, in words; `None` when nothing does.
133    ///
134    /// ```
135    /// use majordomus_cli::idle::Activity;
136    /// let watched = Activity { peers: 1, channels: 1, ..Activity::default() };
137    /// assert_eq!(
138    ///     watched.busy_because().as_deref(),
139    ///     Some("1 peer(s) attached, 1 live channel(s) open")
140    /// );
141    /// assert_eq!(Activity::default().busy_because(), None);
142    /// ```
143    pub fn busy_because(&self) -> Option<String> {
144        let mut why = Vec::new();
145        if self.peers > 0 {
146            why.push(format!("{} peer(s) attached", self.peers));
147        }
148        if self.executions > 0 {
149            why.push(format!(
150                "{} execution(s) queued or running",
151                self.executions
152            ));
153        }
154        if self.channels > 0 {
155            why.push(format!("{} live channel(s) open", self.channels));
156        }
157        if self.client_request {
158            why.push("a client request answered".to_string());
159        }
160        (!why.is_empty()).then(|| why.join(", "))
161    }
162
163    /// Whether nothing keeps the server in use.
164    ///
165    /// ```
166    /// use majordomus_cli::idle::Activity;
167    /// assert!(Activity::default().is_idle());
168    /// assert!(!Activity { peers: 1, ..Activity::default() }.is_idle());
169    /// ```
170    pub fn is_idle(&self) -> bool {
171        self.busy_because().is_none()
172    }
173}

175/// The idle clock of one server: reset by every look that finds it in use, and due when it
176/// has been idle for the whole bound. Deterministic: the caller passes the time.
177///
178/// ```
179/// use std::time::{Duration, Instant};
180/// use majordomus_cli::idle::{Activity, IdleTimer};
181/// let start = Instant::now();
182/// let mut timer = IdleTimer::new(Duration::from_secs(10), start);
183/// // idle, but not for long enough
184/// assert!(timer.observe(&Activity::default(), start + Duration::from_secs(9)).is_none());
185/// // a peer attached resets the clock
186/// let busy = Activity { peers: 1, ..Activity::default() };
187/// assert!(timer.observe(&busy, start + Duration::from_secs(9)).is_none());
188/// assert!(timer.observe(&Activity::default(), start + Duration::from_secs(18)).is_none());
189/// // ten idle seconds after the peer left, the server is due to leave
190/// let due = timer.observe(&Activity::default(), start + Duration::from_secs(19)).unwrap();
191/// assert_eq!(due.idle_for, Duration::from_secs(10));
192/// ```
193#[derive(Debug, Clone, Copy)]
194pub struct IdleTimer {
195    bound: Duration,
196    since: Instant,
197}
198
199impl IdleTimer {
200    /// A clock with `bound`, idle from `now` (a server is idle from the moment it starts).
201    ///
202    /// ```
203    /// use std::time::{Duration, Instant};
204    /// use majordomus_cli::idle::{Activity, IdleTimer};
205    /// let start = Instant::now();
206    /// let mut timer = IdleTimer::new(Duration::from_secs(2), start);
207    /// // nobody ever came: due two seconds after the start
208    /// assert!(timer.observe(&Activity::default(), start + Duration::from_secs(2)).is_some());
209    /// ```
210    pub fn new(bound: Duration, now: Instant) -> Self {
211        Self { bound, since: now }
212    }
213
214    /// Look once: `Some` when the server has been idle for the whole bound.
215    ///
216    /// ```
217    /// use std::time::{Duration, Instant};
218    /// use majordomus_cli::idle::{Activity, IdleTimer};
219    /// let start = Instant::now();
220    /// let mut timer = IdleTimer::new(Duration::from_secs(5), start);
221    /// // an execution queued or running holds the clock at every look
222    /// let running = Activity { executions: 1, ..Activity::default() };
223    /// assert!(timer.observe(&running, start + Duration::from_secs(60)).is_none());
224    /// assert!(timer.observe(&Activity::default(), start + Duration::from_secs(64)).is_none());
225    /// let due = timer.observe(&Activity::default(), start + Duration::from_secs(65)).unwrap();
226    /// assert_eq!(due.idle_for, Duration::from_secs(5));
227    /// ```
228    pub fn observe(&mut self, activity: &Activity, now: Instant) -> Option<IdleExit> {
229        if !activity.is_idle() {
230            self.since = now;
231            return None;
232        }
233        let idle_for = now.saturating_duration_since(self.since);
234        (idle_for >= self.bound).then_some(IdleExit {
235            idle_for,
236            bound: self.bound,
237        })
238    }
239}

241/// Why an idle server leaves: what the log line says.
242///
243/// ```
244/// use std::time::Duration;
245/// use majordomus_cli::idle::IdleExit;
246/// let why = IdleExit { idle_for: Duration::from_secs(900), bound: Duration::from_secs(900) };
247/// assert!(why.to_string().starts_with("idle for 900 s (bound 900 s)"));
248/// ```
249#[derive(Debug, Clone, Copy, PartialEq, Eq)]
250pub struct IdleExit {
251    /// How long nothing used the server.
252    pub idle_for: Duration,
253    /// The idle life it was given.
254    pub bound: Duration,
255}
256
257impl fmt::Display for IdleExit {
258    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
259        write!(
260            f,
261            "idle for {} s (bound {} s): no peer attached, no execution queued or running, \
262             no live channel, no client request; machine traffic (mesh links, probes, sibling \
263             board reads) is not use",
264            self.idle_for.as_secs(),
265            self.bound.as_secs()
266        )
267    }
268}

274    #[test]
275    fn machine_traffic_is_not_use() {
276        for (method, target, probe) in [
277            ("POST", "/api/v1/mesh/link/hello", false),
278            ("POST", "/api/v1/mesh/link/sync", false),
279            ("POST", "/api/v1/mesh/register", false),
280            ("GET", "/", true),
281            ("HEAD", "/", true),
282            ("GET", "/api/v1/peers?checkouts=this", false),
283            ("GET", "/api/v1/peers?format=json&checkouts=this", false),
284        ] {
285            assert!(
286                !counts_as_use(method, target, probe),
287                "{method} {target} (probe header {probe}) was counted as use"
288            );
289        }
290    }

293    fn a_client_is_use() {
294        for (method, target, probe) in [
295            ("GET", "/", false),
296            ("GET", "/cockpit", false),
297            ("POST", "/mcp", false),
298            ("GET", "/api/v1/peers", false),
299            ("GET", "/api/v1/peers?checkouts=repository", false),
300            ("POST", "/api/v1/executions/start", false),

313    fn each_kind_of_activity_keeps_the_server() {

346    #[test]
347    fn the_clock_is_due_only_after_a_whole_idle_bound() {
348        let t0 = Instant::now();
349        let bound = Duration::from_secs(5);
350        let mut timer = IdleTimer::new(bound, t0);
351        assert!(timer.observe(&Activity::default(), t0).is_none());
352        assert!(timer
353            .observe(&Activity::default(), t0 + Duration::from_millis(4_999))
354            .is_none());
355        let due = timer
356            .observe(&Activity::default(), t0 + bound)
357            .expect("due at the bound");
358        assert_eq!(
359            due,
360            IdleExit {
361                idle_for: bound,
362                bound
363            }
364        );
365    }

367    #[test]
368    fn a_running_execution_holds_the_clock_back() {
369        let t0 = Instant::now();
370        let mut timer = IdleTimer::new(Duration::from_secs(1), t0);
371        let running = Activity {
372            executions: 1,
373            ..Activity::default()
374        };
375        for s in 0..10 {
376            assert!(timer
377                .observe(&running, t0 + Duration::from_secs(s))
378                .is_none());
379        }
380        // it finished at 9 s: idle from then, due one second later and not before
381        assert!(timer
382            .observe(&Activity::default(), t0 + Duration::from_millis(9_500))
383            .is_none());
384        assert!(timer
385            .observe(&Activity::default(), t0 + Duration::from_secs(10))
386            .is_some());
387    }
