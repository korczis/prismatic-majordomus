1//! Navigation, derived. The sidebar's catalogues are not a list in this file: the
2//! capability modules come from the registry, the object kinds come from `repository.info`,
3//! and the graphs come from the derivations. A module added to `compose_modules!`, a kind
4//! added to `share/kinds.yaml`, a graph added to the derivation table — each appears here
5//! with no edit to the Cockpit.
6//!
7//! The kinds are *asked for* rather than read: ADR 0012 says no page reads the index
8//! directly, because a reader that steps around the executor is outside the cache, the
9//! counters and the validation every other caller passes through.
10//!
11//! What *is* written here is the areas: Overview, Capabilities, Commands, Executions,
12//! Objects, Directories, Graphs, Continuity, Worktrees, Integration, Health, Quality,
13//! Artifacts, Design, API. Those are concepts rather than
14//! entities, they change when the Cockpit's own shape changes, and deriving them from
15//! anything would be deriving them from a list of exactly themselves.
16//!
17//! Every section is presented in the canonical order (`crate::order`). The order a section
18//! is *built* in is an accident of its source — the order the modules were composed in, the
19//! order the graph derivations are declared in — and an accident is not a reading order.
20//! The order is applied to the sections as a whole, so a section added later is ordered
21//! without being told to be.
22//!
23//! One section is grouped rather than flat: the capability modules sit under the
24//! operational area they serve. That area is not written here either. A feature declares
25//! the modules it is built from and the areas it serves; the product model resolves the two
26//! into an area per module, and this file asks for it. A module no feature places is shown
27//! under no heading, which the canonical order puts last, and the product model reports it.

224/// One entry.
225#[derive(Debug, Clone)]
226pub struct Item {
227    /// What the reader sees.
228    pub label: String,
229    /// Where it goes.
230    pub href: String,
231    /// The area it belongs to.
232    pub area: Area,
233    /// The heading it sits under inside its section, when the section is grouped. Derived,
234    /// never written here: the capability modules are grouped by the operational area the
235    /// features that name them serve.
236    pub group: Option<String>,
237    /// How many things are behind it, when the number is a fact and not decoration: `None`
238    /// for an entry that carries no count, and [`Count::Unknown`] for one whose count the
239    /// capability asked for it did not answer.
240    pub count: Option<Count>,
241    /// Whether this is the page being shown.
242    pub current: bool,
243}

245/// A count an entry shows: a number the capability answered, or the fact that it did not.
246/// A capability that failed is never a count of zero.
247#[derive(Debug, Clone, Copy, PartialEq, Eq)]
248pub enum Count {
249    /// The number.
250    Known(usize),
251    /// The capability behind it did not answer.
252    Unknown,
253}

277/// What the layer holds, as `repository.info` answers it: the object count and the kinds
278/// with theirs.
279///
280/// A capability that cannot answer leaves the navigation without its object counts rather
281/// than without a navigation: the sidebar is how a person reaches the page that would say
282/// what went wrong, so it is the last thing that should fail with the thing it reports on.
283/// The two fields of `repository.info` this file reads; the rest of the report is the
284/// health page's subject, and taking only these keeps the navigation's dependency on the
285/// capability to what it actually shows.
286#[derive(serde::Deserialize)]
287struct Held {
288    objects: usize,
289    kinds: std::collections::BTreeMap<String, usize>,
290}
291
292/// `None` when `repository.info` did not answer, or answered something this file cannot
293/// read: the object count is then unknown, never zero.
294fn held(ctx: &Context) -> Option<Held> {
295    let value = ctx.execute("repository.info", serde_json::json!({})).ok()?;
296    serde_json::from_value::<Held>(value).ok()
297}

299/// Build the navigation for a request: the areas, then the catalogues derived from the
300/// registry, the index and the graph derivations. `here` is the request path, so the
301/// current entry can be marked without a page saying which it is.
302pub fn build(ctx: &Context, here: &str) -> Navigation {
303    let summary = ctx.registry.summary();
304    // What the layer holds is asked for, not opened. `repository.info` answers how many
305    // objects there are and which kinds they have, and asking it is what keeps the
306    // navigation a projection instead of a second reader of the index (ADR 0012). One
307    // call answers both catalogues below.
308    let held = held(ctx);
309    build_with(ctx, here, held.as_ref(), &summary)
310}
311
312/// [`build`], over what `repository.info` answered (`None`: it did not).
313fn build_with(
314    ctx: &Context,
315    here: &str,
316    held: Option<&Held>,
317    summary: &crate::capability::registry::Summary,
318) -> Navigation {
319    // the counts are facts of this context, decided per area: how many things are behind
320    // an entry is not part of what an area is
321    let count = |a: Area| -> Option<Count> {
322        match a {
323            Area::Capabilities => Some(Count::Known(summary.total)),
324            Area::Objects => Some(held.map_or(Count::Unknown, |h| Count::Known(h.objects))),
325            Area::Graphs => Some(Count::Known(graph::ids().len())),
326            Area::Api => Some(Count::Known(summary.http_routes)),
327            _ => None,
328        }
329    };
330    let areas = Section {
331        title: "Cockpit".into(),
332        items: areas()
333            .iter()
334            .map(|a| item(a.label, a.href, a.area, count(a.area), here))
335            .collect(),
336    };
337
338    // the executable's own modules: what a capability belongs to, from the registry
339    let modules = Section {
340        title: "Capability modules".into(),
341        items: ctx
342            .registry
343            .modules()
344            .filter(|m| m.source != ModuleSource::Declarative)
345            .map(|m| Item {
346                label: m.title.clone(),
347                href: format!(
348                    "/cockpit/capabilities?module={}",
349                    percent_encode(m.id.as_str())
350                ),
351                area: Area::Capabilities,
352                // The module's own area, derived by the product model from the features
353                // that name it, shown under the name the Why catalogue gives it. Neither
354                // the grouping nor the heading is written in this file.
355                group: ctx
356                    .product
357                    .module_area(m.id.as_str())
358                    .and_then(|id| ctx.why.areas().iter().find(|a| a.id == id))
359                    .map(|a| a.title.clone()),
360                count: Some(Count::Known(m.capabilities)),
361                current: false,
362            })
363            .collect(),
364    };
365
366    // the kinds of the layer: what the repository declares, not what this code knows
367    let kinds = Section {
368        title: "Object kinds".into(),
369        items: held
370            .into_iter()
371            .flat_map(|h| &h.kinds)
372            .map(|(kind, count)| Item {
373                label: kind.to_string(),
374                href: crate::entity::kind_route(kind),
375                area: Area::Objects,
376                group: None,
377                count: Some(Count::Known(*count)),
378                current: false,
379            })
380            .collect(),
381    };
382
383    let graphs = Section {
384        title: "Graphs".into(),
385        items: graph::ids()
386            .into_iter()
387            .map(|id| Item {
388                label: id.to_string(),
389                href: format!("/cockpit/graphs/{id}"),
390                area: Area::Graphs,
391                group: None,
392                count: None,
393                current: here == format!("/cockpit/graphs/{id}"),
394            })
395            .collect(),
396    };
397
398    Navigation {
399        sections: [areas, modules, kinds, graphs]
400            .into_iter()
401            .filter(|s| !s.items.is_empty())
402            .map(alphabetical)
403            .collect(),
404    }
405}

407/// One section's entries in the canonical order: by label, case-folded so that `API` sits
408/// with `Artifacts` rather than ahead of every lowercase kind, digit runs by value, and the
409/// href behind it to break a tie between two entries a reader would call the same.
410///
411/// The comparator is `crate::order`'s, not this file's. It was this file's, it was the only
412/// case-folded comparator in the crate, and every other surface sorted by raw bytes instead.
413fn alphabetical(mut section: Section) -> Section {
414    crate::order::canonical(&mut section.items);
415    section
416}

418impl crate::order::Ordered for Item {
419    fn order_key(&self) -> crate::order::OrderKey<'_> {
420        match &self.group {
421            Some(group) => crate::order::OrderKey::grouped(group, &self.label, &self.href),
422            None => crate::order::OrderKey::plain(&self.label, &self.href),
423        }
424    }
425}
426
427fn item(label: &str, href: &str, area: Area, count: Option<Count>, here: &str) -> Item {
428    Item {
429        label: label.into(),
430        href: href.into(),
431        area,
432        group: None,
433        count,
434        current: here == href,
435    }
436}

467    #[test]
468    fn a_grouped_section_keeps_its_groups_contiguous_and_the_ungrouped_last() {
469        let repo = repository();
470        let ctx = repo.context().expect("a context");
471        let nav = build(&ctx, "/cockpit");
472        for section in nav.sections() {
473            let mut seen: Vec<&str> = Vec::new();
474            let mut ungrouped = false;
475            for item in &section.items {
476                match item.group.as_deref() {
477                    Some(g) => {
478                        assert!(
479                            !ungrouped,
480                            "a grouped entry follows an ungrouped one in '{}': the canonical \
481                             order puts the ungrouped tail last",
482                            section.title
483                        );
484                        if seen.last().copied() != Some(g) {
485                            assert!(
486                                !seen.contains(&g),
487                                "the group '{g}' appears twice in '{}': its members are not \
488                                 contiguous, so one pass cannot render its heading once",
489                                section.title
490                            );
491                            seen.push(g);
492                        }
493                    }
494                    None => ungrouped = true,
495                }
496            }
497        }
498    }
