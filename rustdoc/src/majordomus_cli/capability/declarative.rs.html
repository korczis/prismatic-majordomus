1//! The declarative source: every object of the index is a resource capability. Its id is
2//! `<kind>.<identity>`, its MCP resource is the object's URI, it has no HTTP or CLI
3//! exposure of its own (it is read through `objects.get`), and its provenance is the
4//! object's. Nothing here reads a file: the index already did, and reported what it
5//! could not read.
6
7use super::model::{
8    Availability, BenchmarkPolicy, CachePolicy, Capability, CapabilityId, CapabilityKind, Exposure,
9    McpExposure, McpResource, ModuleId, Provenance, Stability, Visibility, WaiverReason,
10};
11use super::schema::CanonicalSchema;
12use crate::capability::builtin::ObjectView;
13use crate::model::Object;
14
15/// The object view's schema, derived once: every declarative resource shares it.
16fn object_view_schema() -> CanonicalSchema {
17    static SCHEMA: std::sync::OnceLock<CanonicalSchema> = std::sync::OnceLock::new();
18    SCHEMA
19        .get_or_init(CanonicalSchema::of::<ObjectView>)
20        .clone()
21}
22
23/// The resource capability of one object of the index.
24pub fn capability_of(object: &Object) -> Capability {
25    let kind = CapabilityKind::Resource;
26    let exposure = Exposure {
27        mcp: Some(McpExposure {
28            tool: None,
29            resource: Some(McpResource {
30                uri: object.uri.clone(),
31                name: object.identity.clone(),
32            }),
33        }),
34        http: None,
35        cli: None,
36    };
37    Capability {
38        availability: Availability::classify(kind, &exposure),
39        execution: crate::capability::ExecutionPolicy::classify(kind),
40        visibility: Visibility::classify(&exposure),
41        id: CapabilityId::unchecked(&format!("{}.{}", object.kind, object.identity)),
42        module: ModuleId::unchecked(&object.kind),
43        kind,
44        title: object
45            .title
46            .clone()
47            .unwrap_or_else(|| object.identity.clone()),
48        description: object.description.clone().unwrap_or_default(),
49        input: CanonicalSchema::empty(),
50        output: object_view_schema(),
51        provenance: Provenance::Declarative {
52            path: object.provenance.path.clone(),
53            directory: object.provenance.directory.clone(),
54            source_class: object.provenance.source_class.clone(),
55            section: object.provenance.section.clone(),
56            media_type: object.media_type.to_string(),
57            member: object.provenance.member.clone(),
58        },
59        exposure,
60        stability: Stability::Implemented,
61        tags: object.tags().iter().map(|t| t.to_string()).collect(),
62        benchmark: BenchmarkPolicy::Waived {
63            reason: WaiverReason::NotExecutable,
64        },
65        cache: CachePolicy::Disabled,
66    }
67}
