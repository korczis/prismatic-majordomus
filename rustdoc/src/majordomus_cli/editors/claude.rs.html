1//! Claude Code: what it reads from a project, and the projection of the capabilities that
2//! declare a skill onto it.
3//!
4//! Verified against Claude Code 2.1 — <https://code.claude.com/docs/en/skills> (a project
5//! skill is `.claude/skills/<name>/SKILL.md`; its front matter carries `name`, `description`,
6//! `argument-hint` and `disable-model-invocation`; `$ARGUMENTS` is replaced by the text typed
7//! after the name; custom commands and skills are one mechanism), `/docs/en/mcp` (a project's
8//! servers are `.mcp.json`) and `/docs/en/memory` (`CLAUDE.md`).
9//!
10//! A skill written here is an adapter and nothing else: it names the capability, says where
11//! the typed text goes, and tells the agent to call it once. What the capability does, what
12//! it accepts and what it answers are its declaration's to say, and this file repeats none of
13//! it beyond the title and description the declaration itself carries (ADR 0144, ADR 0140).
14

27/// Where Claude Code discovers a project's skills.
28pub const SKILLS_DIR: &str = ".claude/skills";

30/// The most a skill's description may hold before Claude Code truncates it in its listing.
31const DESCRIPTION_BUDGET: usize = 1024;

33/// The path of the skill a command is projected to.
34pub fn skill_path(alias: &str) -> String {
35    format!("{SKILLS_DIR}/{alias}/SKILL.md")
36}

38/// The skill of one capability: front matter Claude Code requires, the banner, and the
39/// instruction to call the capability once with the typed text in its declared field.
40///
41/// A capability that changes something is invoked by a person only
42/// (`disable-model-invocation: true`): the model does not decide on its own to write the
43/// repository because a conversation resembled the description.
44pub fn command_artifact(command: &CommandRef, version: &str) -> Artifact {
45    let path = skill_path(&command.alias);
46    let summary = format!(
47        "{}. Runs the Majordomus capability {} with the text you type as its `{}`.",
48        command.title.trim_end_matches('.'),
49        command.capability,
50        command.argument
51    );
52    let description: String = summary.chars().take(DESCRIPTION_BUDGET).collect();
53    let quoted = serde_json::to_string(&description).unwrap_or_else(|_| "\"\"".into());
54    let hint =
55        serde_json::to_string(&format!("<{}>", command.argument)).unwrap_or_else(|_| "\"\"".into());
56    let [a, b, c] = banner_lines(&command.source, version);
57    let mut front = format!(
58        "---\nname: {}\ndescription: {quoted}\nargument-hint: {hint}\n",
59        command.alias
60    );
61    if command.changes_something {
62        front.push_str("disable-model-invocation: true\n");
63    }
64    front.push_str("---\n");
65
66    let mut ways = String::new();
67    if let Some(tool) = &command.tool {
68        ways.push_str(&format!(
69            "- With the Majordomus MCP server attached, call the tool `{tool}` with that text as `{}`.\n",
70            command.argument
71        ));
72    }
73    if let Some(cli) = &command.cli {
74        ways.push_str(&format!(
75            "- Otherwise run `majordomus {cli}` from the repository root, giving that text as its `{}` (`majordomus {cli} --help` shows how).\n",
76            command.argument
77        ));
78    }
79    let once = if command.changes_something {
80        "Call it once. It changes the repository: if the call is refused or its outcome is not \
81         known, report that as it is and do not try again by another way.\n"
82    } else {
83        "Call it once and report what it answers.\n"
84    };
85    let content = format!(
86        "{front}<!-- {a}\n     {b}\n     {c} -->\n\n\
87         # {title}\n\n\
88         Invoke the Majordomus capability `{id}`. Its `{arg}` is the text below, whole and \
89         verbatim — do not rephrase, shorten, translate or complete it:\n\n\
90         $ARGUMENTS\n\n\
91         If no text was given, ask for it and stop; do not invent one.\n\n\
92         {ways}\n\
93         {once}\n\
94         Then report what the capability answered, as it answered.\n\n\
95         What `{id}` does, what it accepts and what it answers are defined once, in its \
96         declaration: `majordomus capabilities describe {id}`, or the MCP tool's own description. \
97         This file only makes it reachable by name and says nothing the declaration does not.\n",
98        title = command.title,
99        id = command.capability,
100        arg = command.argument,
101    );
102    Artifact::verbatim(
103        path,
104        format!("{}command", super::DOCUMENT_PREFIX),
105        ArtifactFormat::Markdown,
106        None,
107        command.source.clone(),
108        content,
109    )
110}

112/// What a domain's description says in the listing at most: Claude Code shows the start of
113/// it beside every other skill, and fifty-odd domains must not crowd the listing out.
114const DOMAIN_DESCRIPTION_BUDGET: usize = 300;
115
116/// The description of a domain skill: the module and its capabilities' own titles, cut at a
117/// word so that what the listing shows is what the registry says.
118fn domain_description(domain: &DomainRef) -> String {
119    let titles: Vec<&str> = domain
120        .entries
121        .iter()
122        .map(|e| e.title.trim_end_matches('.'))
123        .collect();
124    let whole = format!(
125        "Majordomus {}: {}.",
126        domain.module.replace('_', " "),
127        titles.join("; ")
128    );
129    if whole.chars().count() <= DOMAIN_DESCRIPTION_BUDGET {
130        return whole;
131    }
132    let cut: String = whole.chars().take(DOMAIN_DESCRIPTION_BUDGET - 1).collect();
133    let at = cut.rfind(' ').unwrap_or(cut.len());
134    format!("{}…", cut[..at].trim_end_matches([';', ',', ' ']))
135}

137/// The skill of one domain (ADR 0165): the module's capabilities as a table the agent chooses
138/// from, the request typed after the name, and the rule that a capability that changes
139/// something runs only when the person asked for that change. The model may load it on its
140/// own, because loading it changes nothing; calling a capability is what the rule governs.
141pub fn domain_artifact(domain: &DomainRef, version: &str) -> Artifact {
142    let path = skill_path(&domain.alias);
143    // a JSON string's rendering cannot fail, so there is no fallback to take
144    let quoted = serde_json::Value::String(domain_description(domain)).to_string();
145    let source = format!("module:{}", domain.module);
146    let [a, b, c] = banner_lines(&source, version);
147    let cell = |s: &str| s.replace('|', "\\|").replace('\n', " ");
148    let mut rows = String::new();
149    for e in &domain.entries {
150        rows.push_str(&format!(
151            "| `{}` | {} | `{}` | {} | {} |\n",
152            e.capability,
153            cell(&e.title),
154            e.tool,
155            e.cli
156                .as_ref()
157                .map(|c| format!("`majordomus {c}`"))
158                .unwrap_or_else(|| "—".into()),
159            if e.changes_something {
160                "changes"
161            } else {
162                "reads"
163            },
164        ));
165    }
166    let content = format!(
167        "---\nname: {alias}\ndescription: {quoted}\nargument-hint: \"[request]\"\n---\n\
168         <!-- {a}\n     {b}\n     {c} -->\n\n\
169         # Majordomus `{module}`\n\n\
170         The capabilities the Majordomus registry declares in its module `{module}`. Choose the \
171         one that answers the request below — or, without one, the person's last request about \
172         `{module}` — call it through its MCP tool, and report what it answers, as it answered:\n\n\
173         $ARGUMENTS\n\n\
174         | Capability | What it is | MCP tool | Command line | Effect |\n\
175         |---|---|---|---|---|\n\
176         {rows}\n\
177         A capability that *changes* something runs only when the person asked for that change: \
178         call it once, and if it is refused or its outcome is not known, report that as it is \
179         and do not try again by another way. One that *reads* may be called whenever the \
180         request needs it.\n\n\
181         Without the Majordomus MCP server attached, run the command line form from the \
182         repository root; `--help` shows its arguments. What each capability accepts and \
183         answers is defined once, in its declaration: `majordomus capabilities describe <id>`, \
184         or the MCP tool's own description. This file is derived from the registry and says \
185         nothing the declarations do not.\n",
186        alias = domain.alias,
187        module = domain.module,
188    );
189    Artifact::verbatim(
190        path,
191        format!("{}domain", super::DOCUMENT_PREFIX),
192        ArtifactFormat::Markdown,
193        None,
194        source,
195        content,
196    )
197}

199/// The domains Claude Code is given: every derived domain whose name no command alias and no
200/// workflow skill of the layer already has. A name both want stays theirs, and the domain is
201/// reported by [`name_findings`].
202pub(crate) fn reachable_domains(
203    domains: Vec<DomainRef>,
204    commands: &[CommandRef],
205    skills: &[super::SkillRef],
206) -> Vec<DomainRef> {
207    domains
208        .into_iter()
209        .filter(|d| !commands.iter().any(|c| c.alias == d.alias))
210        .filter(|d| !skills.iter().any(|s| s.id == d.alias))
211        .collect()
212}

214/// A command whose name a workflow skill of the layer already has, where both products that
215/// read `.agents/skills/` and Claude Code would show two things under one name.
216pub(crate) fn name_findings(
217    commands: &[CommandRef],
218    skills: &[super::SkillRef],
219) -> Vec<EditorDiagnostic> {
220    commands
221        .iter()
222        .filter(|c| skills.iter().any(|s| s.id == c.alias))
223        .map(|c| EditorDiagnostic {
224            severity: Severity::Warning,
225            code: "command-shares-a-skill-name".into(),
226            adapter: Some(ID.into()),
227            path: Some(skill_path(&c.alias)),
228            message: format!(
229                "the capability {} is invoked as /{}, and a workflow skill of the same name exists under .ai/repo/skills/",
230                c.capability, c.alias
231            ),
232            remedy: "rename the skill, or give the capability another alias in its declaration".into(),
233        })
234        .collect()
235}

237/// The workflow skills Claude Code is given a pointer to: every skill of the layer with a
238/// name both products accept, except one whose name a command already has.
239pub(crate) fn workflow_skills<'a>(
240    commands: &[CommandRef],
241    skills: &'a [super::SkillRef],
242) -> Vec<&'a super::SkillRef> {
243    skills
244        .iter()
245        .filter(|s| super::is_skill_name(&s.id))
246        .filter(|s| !commands.iter().any(|c| c.alias == s.id))
247        .collect()
248}

250/// What Claude Code does with each capability here.
251pub fn adapter(
252    commands: &[CommandRef],
253    domains: &[DomainRef],
254    skills: &[super::SkillRef],
255) -> EditorAdapter {
256    EditorAdapter {
257        id: ID.into(),
258        title: "Claude Code".into(),
259        verified_against: VERIFIED_AGAINST.into(),
260        capabilities: vec![
261            EditorCapability {
262                id: "commands".into(),
263                support: EditorSupport::Supported,
264                artifacts: commands.iter().map(|c| skill_path(&c.alias)).collect(),
265                evidence: "code.claude.com/docs/en/skills".into(),
266                note: "One skill per capability that declares an alias: /<alias> <text> calls the capability once with the text in its declared field. A capability that changes something is invoked by a person only. Every other capability is reached by its MCP tool, which needs no file.".into(),
267            },
268            EditorCapability {
269                id: "domains".into(),
270                support: EditorSupport::Supported,
271                artifacts: domains.iter().map(|d| skill_path(&d.alias)).collect(),
272                evidence: "code.claude.com/docs/en/skills".into(),
273                note: "One skill per module of the registry with a public MCP tool (ADR 0165): /mj-<module> lists the module's capabilities, and the model may load it when a request is about that module. A capability that changes something runs only on the person's request.".into(),
274            },
275            EditorCapability {
276                id: "skills".into(),
277                support: EditorSupport::Supported,
278                artifacts: workflow_skills(commands, skills)
279                    .iter()
280                    .map(|s| skill_path(&s.id))
281                    .collect(),
282                evidence: "code.claude.com/docs/en/skills".into(),
283                note: "One pointer per workflow skill of the layer, so that /<id> reaches the definition under .ai/repo/skills/; Claude Code reads .claude/skills/ and not .agents/skills/. A skill whose name a command has is not written twice.".into(),
284            },
285            EditorCapability {
286                id: "instructions".into(),
287                support: EditorSupport::Supported,
288                artifacts: vec!["CLAUDE.md".into()],
289                evidence: "code.claude.com/docs/en/memory".into(),
290                note: "Claude Code reads the provider bootstrap CLAUDE.md, which `majordomus generate providers` already derives from the policy; nothing is added here.".into(),
291            },
292            EditorCapability {
293                id: "mcp".into(),
294                support: EditorSupport::Supported,
295                artifacts: vec![".mcp.json".into()],
296                evidence: "code.claude.com/docs/en/mcp".into(),
297                note: "The project's .mcp.json starts bin/majordomus-mcp; a person approves a project server once, in their own client.".into(),
298            },
299        ],
300    }

398    fn domain(module: &str, titles: &[(&str, bool)]) -> DomainRef {
399        DomainRef {
400            module: module.into(),
401            alias: format!("mj-{}", module.replace('_', "-")),
402            entries: titles
403                .iter()
404                .enumerate()
405                .map(|(i, (title, changes))| crate::editors::DomainEntry {
406                    capability: format!("{module}.c{i}"),
407                    title: (*title).into(),
408                    tool: format!("majordomus_{module}_c{i}"),
409                    cli: (i == 0).then(|| format!("{module} c{i}")),
410                    changes_something: *changes,
411                })
412                .collect(),
413            sha256: "0".repeat(64),
414        }
415    }
