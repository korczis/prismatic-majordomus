1//! What a run is made of, beyond the executions it recorded.
2//!
3//! A recording reads the reports a run wrote: the suite's TSV, `cargo test`'s output, a
4//! coverage summary. Every entry of a run belongs to exactly one of those reports, and
5//! [`EvidenceProducer`] is the word that names which. Not everything a report holds is
6//! something a claim can name: the crate's own unit-test binary and its doctests ran, and no
7//! claim of `docs/CLAIMS.yaml` can point at one of their tests yet. Such an entry is listed as
8//! an [`EvidenceDropped`], with the report it came from and the reason, rather than ignored,
9//! because a recording that silently threw part of a run away reads as a run that was
10//! smaller than it was.
11//!
12//! # What a run is made of
13//!
14//! A producer word reads back as the producer it names and nothing else reads at all; a
15//! `cargo test` output with the crate's unit tests and one integration binary yields one
16//! binary a claim can name and one dropped entry that says why the other is not one:
17//!
18//! ```
19//! use majordomus_cli::evidence::{read_crate_output, EvidenceProducer, Outcome};
20//!
21//! for p in [EvidenceProducer::Suite, EvidenceProducer::Crate, EvidenceProducer::Coverage] {
22//!     assert_eq!(EvidenceProducer::parse(p.as_str()), Some(p));
23//! }
24//! assert_eq!(EvidenceProducer::parse("cargo"), None, "a word no report is named by");
25//!
26//! let out = "     Running unittests src/lib.rs (target/debug/deps/majordomus_cli-1)\n\
27//!            test result: ok. 7 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; \
28//!            finished in 0.50s\n\
29//!                 Running tests/alpha.rs (target/debug/deps/alpha-1)\n\
30//!            test result: ok. 3 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; \
31//!            finished in 1.40s\n";
32//! let read = read_crate_output(out);
33//! assert_eq!(read.binaries.len(), 1);
34//! assert_eq!(read.binaries[0].name, "alpha");
35//! assert_eq!(read.binaries[0].outcome, Outcome::Pass);
36//!
37//! assert_eq!(read.dropped.len(), 1, "the unit tests ran, and are listed rather than ignored");
38//! let dropped = &read.dropped[0];
39//! assert_eq!(dropped.producer, EvidenceProducer::Crate);
40//! assert_eq!(dropped.what, "unittests src/lib.rs");
41//! assert!(dropped.reason.contains("no claim can name"), "{}", dropped.reason);
42//! ```
43//!
44//! # A recording, as a typed run
45//!
46//! Every recording returns an [`EvidenceRunRecord`]: its commit, the tree each report's run
47//! measured, its totals, what was absent, dropped or unknown, and the executions, each of
48//! which maps back to the record's id:
49//!
50//! ```
51//! use majordomus_cli::evidence::{record, EvidenceProducer, EvidenceRunRecord, Origin, RecordRequest, TreeState};
52//! use std::process::Command;
53//!
54//! let root = tempfile::tempdir().unwrap();
55//! let git = |args: &[&str]| {
56//!     Command::new("git").arg("-C").arg(root.path()).args(args).output().unwrap()
57//! };
58//! git(&["init", "-q"]);
59//! git(&["config", "user.email", "t@example.com"]);
60//! git(&["config", "user.name", "t"]);
61//! std::fs::create_dir_all(root.path().join("test/cases")).unwrap();
62//! std::fs::write(root.path().join("test/cases/01_a.sh"), "echo a\n").unwrap();
63//! git(&["add", "-A"]);
64//! git(&["commit", "-qm", "init"]);
65//!
66//! let reports = tempfile::tempdir().unwrap();
67//! let tsv = reports.path().join("run.tsv");
68//! std::fs::write(&tsv, "01_a\tok\t1\tparallel\n").unwrap();
69//! let out = reports.path().join("run.json");
70//! let req = RecordRequest {
71//!     suite: Some(tsv),
72//!     run_record: Some(out.clone()),
73//!     ..RecordRequest::new(Origin::Local)
74//! };
75//! let run = record(root.path(), &req).unwrap().run_record;
76//!
77//! // no measurement came with the report, so the run's tree is not vouched for
78//! assert_eq!(run.working_tree, TreeState::Unknown);
79//! assert_eq!(run.totals.executions, 1);
80//! assert_eq!(run.absent[0].producer, EvidenceProducer::Crate);
81//! let written: EvidenceRunRecord =
82//!     serde_json::from_str(&std::fs::read_to_string(&out).unwrap()).unwrap();
83//! assert_eq!(written, run);
84//! assert_eq!(EvidenceRunRecord::id_of(&run.executions[0]), run.id);
85//! ```

100/// Which report an entry of a run belongs to: the suite's TSV, `cargo test`'s output, or a
101/// coverage summary.
102///
103/// The serde word is the word a person types and a document carries (`suite`, `crate`,
104/// `coverage`), and [`EvidenceProducer::parse`] reads exactly those three back. A word that
105/// names no report is not a producer, so a misspelled one is refused where it is read
106/// rather than guessed at.
107///
108/// ```
109/// use majordomus_cli::evidence::EvidenceProducer;
110///
111/// assert_eq!(EvidenceProducer::parse("crate"), Some(EvidenceProducer::Crate));
112/// assert_eq!(EvidenceProducer::Coverage.as_str(), "coverage");
113/// assert_eq!(serde_json::to_value(EvidenceProducer::Suite).unwrap(), "suite");
114/// assert_eq!(EvidenceProducer::parse("Suite"), None, "the word is exact");
115/// ```
116#[derive(
117    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
118)]
119#[serde(rename_all = "snake_case")]
120pub enum EvidenceProducer {
121    /// The suite runner's TSV report (`MJ_TEST_REPORT` of `test/run.sh`).
122    Suite,
123    /// `cargo test`'s output: one result line per test binary.
124    Crate,
125    /// A coverage summary (`scripts/rust-coverage --summary-json`).
126    Coverage,
127}
128
129impl EvidenceProducer {
130    /// The producer a word names, or `None` when it names none.
131    ///
132    /// The three words are exact: no case folding and no trimming, because every caller
133    /// reads a word a program wrote and a near-miss is a defect to name, not to absorb.
134    ///
135    /// ```
136    /// use majordomus_cli::evidence::EvidenceProducer;
137    /// assert_eq!(EvidenceProducer::parse("suite"), Some(EvidenceProducer::Suite));
138    /// assert_eq!(EvidenceProducer::parse(" suite"), None);
139    /// assert_eq!(EvidenceProducer::parse(""), None);
140    /// ```
141    pub fn parse(word: &str) -> Option<EvidenceProducer> {
142        match word {
143            "suite" => Some(EvidenceProducer::Suite),
144            "crate" => Some(EvidenceProducer::Crate),
145            "coverage" => Some(EvidenceProducer::Coverage),
146            _ => None,
147        }
148    }
149
150    /// The word this producer is written as: the serde word, and what
151    /// [`EvidenceProducer::parse`] reads back.
152    ///
153    /// ```
154    /// use majordomus_cli::evidence::EvidenceProducer;
155    /// assert_eq!(EvidenceProducer::Crate.as_str(), "crate");
156    /// assert_eq!(
157    ///     EvidenceProducer::parse(EvidenceProducer::Suite.as_str()),
158    ///     Some(EvidenceProducer::Suite)
159    /// );
160    /// ```
161    pub fn as_str(self) -> &'static str {
162        match self {
163            EvidenceProducer::Suite => "suite",
164            EvidenceProducer::Crate => "crate",
165            EvidenceProducer::Coverage => "coverage",
166        }
167    }
168}

170/// Something a report held that no claim can name yet, listed instead of ignored.
171///
172/// `cargo test` runs the crate's own unit-test binary and its doctests beside the
173/// integration binaries under `tests/`. They ran, and a claim names a test by its path, so
174/// none of their tests is anything a claim can point at until a runner records them one by
175/// one. A recording lists each such entry with the report it came from ([`producer`]), what
176/// it was ([`what`]) and why it is not recorded ([`reason`]), so that the part of a run
177/// nobody can use yet is visible rather than silently smaller.
178///
179/// [`producer`]: EvidenceDropped::producer
180/// [`what`]: EvidenceDropped::what
181/// [`reason`]: EvidenceDropped::reason
182///
183/// ```
184/// use majordomus_cli::evidence::{EvidenceDropped, EvidenceProducer};
185///
186/// let d: EvidenceDropped = serde_json::from_value(serde_json::json!({
187///     "producer": "crate",
188///     "what": "doc-tests majordomus_cli",
189///     "reason": "doctests, which no claim can name until a runner records them one by one",
190/// }))
191/// .unwrap();
192/// assert_eq!(d.producer, EvidenceProducer::Crate);
193/// assert_eq!(serde_json::to_value(&d).unwrap()["producer"], "crate");
194/// ```
195#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
196pub struct EvidenceDropped {
197    /// The report the entry was read from.
198    pub producer: EvidenceProducer,
199    /// What the entry was, in the report's own words: `unittests src/lib.rs`,
200    /// `doc-tests <crate>`, the path of a binary that is not under `tests/`, or `a result
201    /// line` when a result belonged to no binary at all.
202    pub what: String,
203    /// Why it is listed rather than recorded. It carries no counts.
204    pub reason: String,
205}

210/// One recording, as a typed run: what it recorded, against what, measured how, and what
211/// the reports held that it could not record.
212///
213/// Every `evidence record` returns one and writes it with `--run-record <file>`. It is
214/// supplementary: a verdict is derived from the tracked ledger only, and a run record can
215/// weaken one only through the one monotone rule ([`weakened_by_records`]).
216///
217/// ```
218/// use majordomus_cli::evidence::{EvidenceRunRecord, LedgerTarget, TreeState};
219/// let r: EvidenceRunRecord = serde_json::from_value(serde_json::json!({
220///     "schema": 1, "id": "local:0123456789ab:20260926T120000Z", "origin": "local",
221///     "commit": "0123456789abcdef0123456789abcdef01234567", "working_tree": "clean",
222///     "recorded_at": "2026-09-26T12:00:00Z", "ledger": "repo"
223/// })).unwrap();
224/// assert_eq!((r.working_tree, r.ledger), (TreeState::Clean, LedgerTarget::Repo));
225/// assert!(r.executions.is_empty() && r.coverage.is_none());
226/// ```
227#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
228pub struct EvidenceRunRecord {
229    /// [`RUN_RECORD_SCHEMA`].
230    pub schema: u32,
231    /// [`run_id`] of the recording.
232    pub id: String,
233    /// Where the run happened.
234    pub origin: Origin,
235    /// The recorder's HEAD, which every measurement given equals.
236    pub commit: String,
237    /// The weakest tree over the reports given: `unknown` for a report with no measurement.
238    pub working_tree: TreeState,
239    /// RFC 3339 UTC, the `at` of every execution.
240    pub recorded_at: String,
241    /// The ledger the executions were merged into.
242    pub ledger: LedgerTarget,
243    /// The CI run, when there was one.
244    #[serde(default, skip_serializing_if = "Option::is_none")]
245    pub run: Option<RunRef>,
246    /// Each report's measurement, ordered suite, crate, coverage, each naming its producer.
247    #[serde(default)]
248    pub provenance: Vec<EvidenceProvenance>,
249    /// Counts over the executions.
250    #[serde(default)]
251    pub totals: EvidenceRunTotals,
252    /// The producers that gave nothing, and why.
253    #[serde(default)]
254    pub absent: Vec<EvidenceAbsent>,
255    /// What the reports held that no claim can name yet.
256    #[serde(default)]
257    pub dropped: Vec<EvidenceDropped>,
258    /// Tests the reports named that this repository does not have.
259    #[serde(default)]
260    pub unknown: Vec<String>,
261    /// The coverage measurement, bound to its commit.
262    #[serde(default, skip_serializing_if = "Option::is_none")]
263    pub coverage: Option<EvidenceCoverage>,
264    /// The executions recorded.
265    #[serde(default)]
266    pub executions: Vec<Execution>,
267}

278/// Counts over a run's executions: how many, of each outcome, of each runner. A zero is
279/// absent rather than 0.
280///
281/// ```
282/// use majordomus_cli::evidence::EvidenceRunTotals;
283/// let t = EvidenceRunTotals::default();
284/// assert_eq!(serde_json::to_value(&t).unwrap()["executions"], 0);
285/// ```
286#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
287pub struct EvidenceRunTotals {
288    /// How many executions the run recorded.
289    pub executions: usize,
290    /// By outcome word, as [`crate::evidence::outcome_counts`] counts them.
291    pub outcomes: BTreeMap<String, usize>,
292    /// By runner prefix (`suite`, `crate`).
293    pub runners: BTreeMap<String, usize>,
294}
295
296impl EvidenceRunTotals {
297    /// The totals of `executions`: how many there are, how many of each outcome word, and
298    /// how many of each runner, with a count nothing reached left out rather than written 0.
299    ///
300    /// ```
301    /// use majordomus_cli::evidence::EvidenceRunTotals;
302    /// assert_eq!(EvidenceRunTotals::of(&[]), EvidenceRunTotals::default());
303    /// ```
304    pub fn of(executions: &[Execution]) -> EvidenceRunTotals {
305        let mut runners = BTreeMap::new();
306        for e in executions {
307            *runners.entry(e.runner.prefix().to_string()).or_insert(0) += 1;
308        }
309        EvidenceRunTotals {
310            executions: executions.len(),
311            outcomes: outcome_counts(executions),
312            runners,
313        }
314    }
315}

332/// The id of a recording: `<origin>:<provider>:<run id>:<attempt>` for a CI run, else
333/// `<origin>:<commit[..12]>:<at without '-' and ':'>`.
334///
335/// ```
336/// use majordomus_cli::evidence::{run_id, Origin};
337/// let c = "0123456789abcdef0123456789abcdef01234567";
338/// assert_eq!(run_id(Origin::Local, None, c, "2026-09-26T12:00:00Z"),
339///     "local:0123456789ab:20260926T120000Z");
340/// ```
341pub fn run_id(origin: Origin, run: Option<&RunRef>, commit: &str, at: &str) -> String {
342    match run {
343        Some(r) => format!(
344            "{}:{}:{}:{}",
345            origin_word(origin),
346            r.provider,
347            r.id,
348            r.attempt
349        ),
350        None => format!(
351            "{}:{}:{}",
352            origin_word(origin),
353            commit.get(..12).unwrap_or(commit),
354            at.replace(['-', ':'], "")
355        ),
356    }
357}

392/// A judgement, weakened by the run records that ran the same test: the second source of
393/// the one monotone rule, [`crate::evidence::freshness::weakened_by`].
394///
395/// It holds no predicate of its own. Each execution of a record that ran `execution`'s test
396/// becomes one [`Supplementary`], placed by `contains(descendant, ancestor)`: after the
397/// evidence when the record's commit contains the evidence commit, and in the presented
398/// history when the presented commit contains the record's. The most recent record comes
399/// first. The result is the input state or `stale`, never anything stronger.
400///
401/// ```
402/// use majordomus_cli::evidence::{weaken_by_records, EvidenceRunRecord, Execution, Judgement, ProofState};
403/// use majordomus_cli::git::Containment;
404/// let exec = |outcome: &str, commit: &str| -> Execution { serde_json::from_value(serde_json::json!({
405///     "test": "suite:01_a", "runner": "suite", "source": "test/cases/01_a.sh",
406///     "outcome": outcome, "seconds": 1, "commit": commit, "working_tree": "clean",
407///     "digest": "sha256:0", "at": "2026-09-26T12:00:00Z", "origin": "ci",
408///     "command": "bash test/run.sh 01_a"
409/// })).unwrap() };
410/// let (e, r) = ("e".repeat(40), "f".repeat(40));
411/// let evidence = exec("pass", &e);
412/// let record: EvidenceRunRecord = serde_json::from_value(serde_json::json!({
413///     "schema": 1, "id": "ci:x", "origin": "ci", "commit": r, "working_tree": "clean",
414///     "recorded_at": "2026-09-26T12:00:00Z", "ledger": "repo",
415///     "executions": [exec("fail", &r)]
416/// })).unwrap();
417/// let proven = Judgement { state: ProofState::Proven, changed: vec![], detail: None };
418/// let j = weaken_by_records(proven, Some(&evidence), &[record], Some(&r), |_, _| Containment::Contains);
419/// assert_eq!(j.state, ProofState::Stale);
420/// ```
421pub fn weaken_by_records(
422    judgement: Judgement,
423    execution: Option<&Execution>,
424    records: &[EvidenceRunRecord],
425    presented_commit: Option<&str>,
426    contains: impl Fn(&str, &str) -> Containment,
427) -> Judgement {
428    let (Some(execution), Some(presented)) = (execution, presented_commit) else {
429        return judgement;
430    };
431    // newest first: the canonical order, which is oldest first, read backwards
432    let mut ordered: Vec<&EvidenceRunRecord> = records.iter().collect();
433    crate::order::canonical(&mut ordered);
434    ordered.reverse();
435    let named: Vec<String> = ordered
436        .iter()
437        .map(|r| format!("the {} run {}", origin_word(r.origin), r.id))
438        .collect();
439    let mut supplementaries = Vec::new();
440    for (r, name) in ordered.iter().zip(&named) {
441        for x in r.executions.iter().filter(|x| x.test == execution.test) {
442            supplementaries.push(Supplementary {
443                execution: x,
444                named: name,
445                after_evidence: contains(&r.commit, &execution.commit),
446                in_presented: contains(presented, &r.commit),
447            });
448        }
449    }
450    weakened_by(judgement, &supplementaries)
451}

453/// [`weaken_by_records`] against a checkout: the presented commit is the one `presented`
454/// names, or HEAD for the working tree (none when HEAD is unborn), and containment is
455/// [`crate::git::contains`] in `root`.
456///
457/// ```
458/// use majordomus_cli::evidence::{weakened_by_records, Judgement, Presented, ProofState};
459/// let dir = tempfile::tempdir().unwrap();
460/// let proven = Judgement { state: ProofState::Proven, changed: vec![], detail: None };
461/// let j = weakened_by_records(proven.clone(), None, &[], dir.path(), &Presented::WorkingTree);
462/// assert_eq!(j, proven);
463/// ```
464pub fn weakened_by_records(
465    judgement: Judgement,
466    execution: Option<&Execution>,
467    records: &[EvidenceRunRecord],
468    root: &Path,
469    presented: &Presented,
470) -> Judgement {
471    let presented_commit = match presented {
472        Presented::Commit { commit, .. } => Some(commit.clone()),
473        Presented::WorkingTree => match crate::git::inspect(root) {
474            crate::git::GitState::Available(i) => i.head,
475            crate::git::GitState::Unavailable { .. } => None,
476        },
477    };
478    weaken_by_records(
479        judgement,
480        execution,
481        records,
482        presented_commit.as_deref(),
483        |d, a| crate::git::contains(root, d, a),
484    )
485}

587    /// Containment in a line of history E < R < P: a later commit contains every earlier one.
588    fn linear(d: &str, a: &str) -> Containment {
589        let pos = |c: &str| [E, R, P].iter().position(|x| *x == c);
590        match (pos(d), pos(a)) {
591            (Some(d), Some(a)) if d >= a => Containment::Contains,
592            (Some(_), Some(_)) => Containment::DoesNotContain,
593            _ => Containment::CommitUnknown,
594        }
595    }

597    fn weaken(state: ProofState, records: &[EvidenceRunRecord]) -> Judgement {
598        let evidence = exec("suite:01_a", "pass", E, "clean", "local");
599        weaken_by_records(judged(state), Some(&evidence), records, Some(P), linear)
600    }

702    /// The newest recording is the one named, whatever order the records arrived in; two
703    /// recorded in the same second are told apart by their ids in the canonical order, so
704    /// `ci:x:1:10` is later than `ci:x:1:9` and not, as a byte comparison has it, earlier.
705    #[test]
706    fn the_newest_failing_record_is_the_one_named() {
707        let failing = |id: &str, at: &str| {
708            record_of(
709                id,
710                at,
711                R,
712                vec![exec("suite:01_a", "fail", R, "clean", "ci")],
713            )
714        };
715        let named = |records: &[EvidenceRunRecord]| {
716            let detail = weaken(ProofState::Proven, records).detail.unwrap();
717            detail.split(" recorded on ").next().unwrap().to_string()
718        };
719        let older = failing("ci:x:1:1", "2026-09-27T00:00:00Z");
720        let newer = failing("ci:x:2:1", "2026-09-28T00:00:00Z");
721        assert_eq!(
722            named(&[older.clone(), newer.clone()]),
723            "the ci run ci:x:2:1"
724        );
725        assert_eq!(named(&[newer, older]), "the ci run ci:x:2:1");
726        let nine = failing("ci:x:1:9", "2026-09-27T00:00:00Z");
727        let ten = failing("ci:x:1:10", "2026-09-27T00:00:00Z");
728        assert_eq!(named(&[ten.clone(), nine.clone()]), "the ci run ci:x:1:10");
729        assert_eq!(named(&[nine, ten]), "the ci run ci:x:1:10");
730    }
