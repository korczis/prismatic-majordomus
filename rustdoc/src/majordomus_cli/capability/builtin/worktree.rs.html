1//! The `worktree` module: the branch-to-worktree topology of this repository, projected.
2//!
3//! Four questions, all read-only, all answered by [`crate::worktree::WorktreeService`] —
4//! the same service the command line renders and the git hooks ask. Declaring them here is
5//! what puts them on MCP (as tools and as a resource), on HTTP, in the OpenAPI document, in
6//! the Swagger UI, in the Cockpit and in the generated reference, without any of those
7//! carrying a route, a schema or a sentence of their own.
8//!
9//! Creating, migrating, repairing and removing are deliberately *not* here. A capability that
10//! writes the repository must declare `Effect::RepositoryMutation`, and the set of those is
11//! pinned by a test in `builtin`; moving worktrees on disk is not a thing the shared MCP
12//! server takes on. They are command-line operations, they call the same service, and the
13//! Cockpit names the exact command for each.
14

42pub struct StatusInput {
43    #[serde(default, skip_serializing_if = "Option::is_none")]
44    /// A directory inside one of this repository's worktrees; the answer is about that
45    /// worktree. Relative to the primary checkout when relative (`.` is the primary

49}

87fn refused(e: WorktreeError) -> CapabilityError {
88    match e.exit_code() {
89        crate::worktree::EXIT_MISSING => CapabilityError::NotFound(e.to_string()),
90        crate::worktree::EXIT_REFUSED => CapabilityError::Refused(e.to_string()),
91        _ => CapabilityError::Internal(e.to_string()),
92    }
93}

95/// The one service, over the repository this process serves.
96pub fn service_of(ctx: &Context) -> Result<WorktreeService, CapabilityError> {
97    WorktreeService::open(Path::new(&ctx.index.repository.root)).map_err(refused)
98}
99
100/// The service over a directory the caller named, refused unless it is inside a worktree
101/// of the repository this process serves.
102fn service_at(ctx: &Context, path: Option<&str>) -> Result<WorktreeService, CapabilityError> {
103    let own = service_of(ctx)?;
104    let Some(path) = path else {
105        return Ok(own);
106    };
107    // relative to the primary checkout, never to this process's working directory
108    let path = {
109        let p = Path::new(path);
110        if p.is_absolute() {
111            p.to_path_buf()
112        } else {
113            Path::new(&ctx.index.repository.root).join(p)
114        }
115    };
116    let other = WorktreeService::open(&path).map_err(refused)?;
117    if !other
118        .identity()
119        .git_common_dir()
120        .same_as(own.identity().git_common_dir())
121    {
122        return Err(CapabilityError::Refused(
123            WorktreeError::NotThisRepository {
124                path,
125                git_common_dir: own.identity().git_common_dir().path.clone(),
126            }
127            .to_string(),
128        ));
129    }
130    Ok(other)
131}

133// ---------------------------------------------------------------- handlers
134
135fn worktree_topology(ctx: &Context, _: Empty) -> Result<RepositoryTopology, CapabilityError> {
136    service_of(ctx)?.topology(Detail::Full).map_err(refused)
137}
138
139fn worktree_status(ctx: &Context, input: StatusInput) -> Result<StatusReport, CapabilityError> {
140    service_at(ctx, input.path.as_deref())?
141        .status()
142        .map_err(refused)
143}
144
145fn worktree_inspect(ctx: &Context, input: InspectInput) -> Result<InspectReport, CapabilityError> {
146    service_of(ctx)?.inspect(&input.branch).map_err(refused)
147}
148
149fn worktree_migration_plan(ctx: &Context, _: Empty) -> Result<MigrationPlan, CapabilityError> {
150    migrate::plan(&service_of(ctx)?).map_err(refused)
151}

238    /// The declaration is the only place these names exist; every projection derives from
239    /// it. This is the assertion a refactor that dropped an exposure would fail.
240    #[test]
241    fn the_declaration_yields_the_projections_it_claims() {
242        let m = module();
243        assert_eq!(m.id.as_str(), "worktree");
244        let expected: &[(&str, &str, &str)] = &[
245            (
246                "worktree.topology",
247                "majordomus_worktrees",
248                "/api/v1/worktrees",
249            ),
250            (
251                "worktree.status",
252                "majordomus_worktree_status",
253                "/api/v1/worktrees/status",
254            ),
255            (
256                "worktree.inspect",
257                "majordomus_worktree_inspect",
258                "/api/v1/worktrees/inspect",
259            ),
260            (
261                "worktree.migration_plan",
262                "majordomus_worktree_migration_plan",
263                "/api/v1/worktrees/migration",
264            ),
265        ];
266        let ids: Vec<&str> = m
267            .capabilities
268            .iter()
269            .map(|e| e.capability.id.as_str())
270            .collect();
271        let want: Vec<&str> = expected.iter().map(|(id, _, _)| *id).collect();
272        assert_eq!(ids, want);
273        for (executable, (id, tool, path)) in m.capabilities.iter().zip(expected) {
274            let exposure = &executable.capability.exposure;
275            assert_eq!(
276                exposure.mcp.as_ref().and_then(|m| m.tool.as_deref()),
277                Some(*tool),
278                "{id} lost or renamed its MCP tool"
279            );
280            assert_eq!(
281                exposure.http.as_ref().map(|h| h.path.as_str()),
282                Some(*path),
283                "{id} lost or renamed its HTTP route"
284            );
285            assert!(
286                !executable.capability.cache.is_enabled(),
287                "{id} must not cache: the topology changes outside this process"
288            );
289        }
290        let resource = m.capabilities[0]
291            .capability
292            .exposure
293            .mcp
294            .as_ref()
295            .and_then(|m| m.resource.as_ref());
296        assert_eq!(resource.map(|r| r.uri.as_str()), Some(WORKTREES_URI));
297    }
