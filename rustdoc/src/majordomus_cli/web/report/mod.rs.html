1//! The reports this repository generates as web surfaces: the test run, the benchmark run,
2//! and the UI conformance audit that is a section of the first.
3//!
4//! A report is a *rendering* of evidence somebody else produced. The suite writes its
5//! results, the benchmark run writes its own, and this module turns either into a directory
6//! under the generated web root: an `index.html` a person reads, the machine-readable
7//! results beside it, and the `surface.json` that declares what the directory is for. It
8//! renders; it never runs anything and never decides what passed.
9//!
10//! The HTML is deliberately self-contained — its own style, no external asset, and every
11//! link relative — so that the same directory works at `/tests`, at `/reports/tests` or
12//! opened from disk. That is the whole of the "static-safe URL" problem, solved once by not
13//! creating it.
14

30/// What every report writes beside its rendering: the declaration that makes the directory
31/// a surface.
32///
33/// The producer states the intent — its id, where it belongs, what built it — and discovery
34/// reads it back. Nothing else in the repository is told that the surface exists.
35pub fn declare(root: &Path, id: &str, title: &str, producer: &str) -> Result<PathBuf> {
36    let dir = directory(root, id);
37    std::fs::create_dir_all(&dir).map_err(|e| Error::io(dir.display().to_string(), e))?;
38    let declaration = Declaration {
39        schema: DECLARATION_SCHEMA.into(),
40        id: id.into(),
41        mount: format!("/{id}"),
42        title: Some(title.into()),
43        index: Some("index.html".into()),
44        producer: Some(producer.into()),
45        availability: Some(Availability::Both),
46        category: Some(Category::Report),
47        visibility: Some(Visibility::Public),
48        built_from: Origin::read(root).revision,
49    };
50    let path = dir.join(DECLARATION_FILE);
51    let mut body =
52        serde_json::to_string_pretty(&declaration).map_err(|e| Error::InvalidSurface {
53            surface: id.into(),
54            reason: format!("its declaration cannot be serialised: {e}"),
55        })?;
56    body.push('\n');
57    std::fs::write(&path, body).map_err(|e| Error::io(path.display().to_string(), e))?;
58    Ok(dir)
59}

61/// Declare a surface only when nothing has declared it yet.
62///
63/// A report that is a *section* of another's surface still needs that surface to exist —
64/// an unreachable section is not a report — but it must never restate the enclosing
65/// producer's identity over the top of it. So: create when absent, leave alone when there.
66pub fn declare_if_absent(root: &Path, id: &str, title: &str, producer: &str) -> Result<PathBuf> {
67    let dir = directory(root, id);
68    if dir.join(DECLARATION_FILE).is_file() {
69        return Ok(dir);
70    }
71    declare(root, id, title, producer)
72}

74/// Where a report's directory is, absolute.
75pub fn directory(root: &Path, id: &str) -> PathBuf {
76    root.join(GENERATED_ROOT).join(id)
77}

79/// Write one file inside a report's directory, creating what it needs.
80pub fn write(dir: &Path, name: &str, body: &str) -> Result<()> {
81    let path = dir.join(name);
82    if let Some(parent) = path.parent() {
83        std::fs::create_dir_all(parent).map_err(|e| Error::io(parent.display().to_string(), e))?;
84    }
85    std::fs::write(&path, body).map_err(|e| Error::io(path.display().to_string(), e))
86}

88/// Write a value as pretty JSON beside a rendering: the evidence the page was made from,
89/// kept because a page nobody can check is not evidence.
90pub fn write_json<T: Serialize>(dir: &Path, name: &str, value: &T) -> Result<()> {
91    let mut body = serde_json::to_string_pretty(value).map_err(|e| Error::InvalidSurface {
92        surface: name.into(),
93        reason: format!("cannot be serialised: {e}"),
94    })?;
95    body.push('\n');
96    write(dir, name, &body)
97}

99/// What a report says about where it came from, printed on every page.
100#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
101pub struct Origin {
102    /// The commit the evidence was produced from, when git could say.
103    pub revision: Option<String>,
104    /// When the report was rendered, in UTC.
105    pub rendered_at: String,
106    /// The executable that rendered it.
107    pub version: String,
108}
109
110impl Origin {
111    /// Read the origin from the repository, without failing when git is not there: a report
112    /// rendered outside a checkout is still a report.
113    pub fn read(root: &Path) -> Self {
114        let revision = std::process::Command::new("git")
115            .args(["rev-parse", "--short", "HEAD"])
116            .current_dir(root)
117            .output()
118            .ok()
119            .filter(|o| o.status.success())
120            .and_then(|o| String::from_utf8(o.stdout).ok())
121            .map(|s| s.trim().to_string())
122            .filter(|s| !s.is_empty());
123        Origin {
124            revision,
125            rendered_at: crate::web::report::now(),
126            version: crate::VERSION.into(),
127        }
128    }
129}

131/// The current time as `YYYY-MM-DDTHH:MM:SSZ`, without a date dependency.
132fn now() -> String {
133    let secs = std::time::SystemTime::now()
134        .duration_since(std::time::UNIX_EPOCH)
135        .map(|d| d.as_secs())
136        .unwrap_or(0) as i64;
137    let days = secs.div_euclid(86_400);
138    let rest = secs.rem_euclid(86_400);
139    let (year, month, day) = civil_from_days(days);
140    format!(
141        "{year:04}-{month:02}-{day:02}T{:02}:{:02}:{:02}Z",
142        rest / 3600,
143        (rest % 3600) / 60,
144        rest % 60
145    )
146}

148/// Days since the Unix epoch to a civil date (Howard Hinnant's algorithm).
149fn civil_from_days(z: i64) -> (i64, u32, u32) {
150    let z = z + 719_468;
151    let era = z.div_euclid(146_097);
152    let doe = z.rem_euclid(146_097);
153    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
154    let y = yoe + era * 400;
155    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
156    let mp = (5 * doy + 2) / 153;
157    let d = (doy - (153 * mp + 2) / 5 + 1) as u32;
158    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
159    (if m <= 2 { y + 1 } else { y }, m, d)
160}

162#[cfg(test)]
163mod unit {
164    use super::*;

166    #[test]
167    fn a_declaration_makes_a_directory_discoverable() {
168        let tmp = tempfile::tempdir().unwrap();
169        let dir = declare(
170            tmp.path(),
171            "example",
172            "Example report",
173            "the example producer",
174        )
175        .unwrap();
176        assert!(dir.join(DECLARATION_FILE).is_file());
177        let found = crate::web::discover::generated(tmp.path()).unwrap();
178        assert_eq!(found.len(), 1);
179        assert_eq!(found[0].id, "example");
180        assert_eq!(found[0].mount.as_str(), "/example");
181        assert_eq!(found[0].producer, "the example producer");
182    }
183
184    #[test]
185    fn the_epoch_and_a_known_day_render_as_the_dates_they_are() {
186        assert_eq!(civil_from_days(0), (1970, 1, 1));
187        assert_eq!(civil_from_days(19_723), (2024, 1, 1));
188        assert_eq!(civil_from_days(20_608), (2026, 6, 4));
189    }
190
191    #[test]
192    fn a_rendered_time_has_the_shape_every_report_prints() {
193        let stamp = now();
194        assert_eq!(stamp.len(), 20, "{stamp}");
195        assert!(stamp.ends_with('Z'));
196        assert!(stamp.contains('T'));
197    }
