1//! Knowledge: what the deriver left for review and whether it is still writing. Read from
2//! `knowledge_base.status` and `knowledge_base.candidates`.
3//!
4//! The screen shows the queue and names the commands a reviewer runs. It promotes and rejects
5//! nothing: a screen reads and never executes (ADR 0142), and promotion is a person's act.
6//! The table is bounded, because the queue is not: what is left out is counted and said.
7

25/// How many candidates, and how many subjects, the screen draws. The queue has no bound of
26/// its own beyond the policy's cap, and a screen answers within its budget whatever the queue
27/// holds; the newest are drawn and the rest are counted.
28pub const SHOWN: usize = 50;

30/// The commands a reviewer runs, in the order a review takes them. Text, never a control:
31/// the screen is a reading.
32const REVIEW: &[(&str, &str)] = &[
33    (
34        "majordomus knowledge candidates",
35        "the queue, with the branch each record came from",
36    ),
37    (
38        "majordomus knowledge observe --kind <kind> --subject <subject> \"<statement>\"",
39        "what was noticed is written down, so the next derivation has it",
40    ),
41    (
42        "majordomus knowledge record <id>",
43        "one record with every reference it names resolved",
44    ),
45    (

59/// The Knowledge screen.
60pub fn screen(ctx: &Context) -> View {
61    let answers = ask_all(ctx, &[STATUS, CANDIDATES]);
62    let status = &answers[&STATUS];
63    let queue = &answers[&CANDIDATES];
64
65    let top = hero(
66        "sparkles",
67        "orange",
68        "Knowledge",
69        "What the work left behind, awaiting a reviewer.",
70        "Every record the deriver wrote from a closed episode and nobody has promoted or rejected yet, what it is about, and whether the deriver is still writing.",
71        vec![
72            kpi("Awaiting review", &queue.read("/total")),
73            kpi("On this branch", &queue.read("/on_this_branch")),
74            kpi("Unattributed", &queue.read("/unattributed")),
75            kpi("Closed, underived", &status.read("/closed_without_derivation")),
76            kpi_state("Newest episode", &status.read("/freshness")),
77        ],
78    );
79
80    let grid = el("div")
81        .class("mj-ui-grid")
82        .child(panel(
83            "The deriver",
84            "Whether a derivation still follows every closed episode, as the status judges it.",
85            unanswered(status).unwrap_or_else(|| deriver(status)),
86            &[status],
87        ))
88        .child(panel(
89            "Review",
90            "What a reviewer runs. This screen reads the queue; it promotes and rejects nothing.",
91            review(),
92            &[],
93        ))
94        .child(wide(panel(
95            "By subject",
96            "The records derived from observations, grouped by what they are about.",
97            unanswered(queue).unwrap_or_else(|| subjects(queue)),
98            &[queue],
99        )))
100        .child(wide(panel(
101            "Awaiting review",
102            "The newest candidates, each leading to its record.",
103            unanswered(queue).unwrap_or_else(|| candidates(queue)),
104            &[queue],
105        )));
106
107    View::new(
108        "Knowledge",
109        el("div").class("mj-ui-stack").child(top).child(grid),
110    )
111}

113/// The judgement of the deriver: when it last wrote, when an episode last closed, why the
114/// newest episode is judged as it is, and every finding the status reports.
115fn deriver(status: &Answer) -> El {
116    let Some(answer) = status.ok() else {
117        return nothing("The status did not answer.");
118    };
119    let mark = |key: &str| -> String {
120        answer
121            .get(key)
122            .map(|m| {
123                let at = text(m, "ts");
124                let episode = text(m, "episode");
125                if episode.is_empty() {
126                    at
127                } else {
128                    format!("{at}, episode {episode}")
129                }
130            })
131            .filter(|line| !line.is_empty())
132            .unwrap_or_else(|| "never, in this checkout".to_string())
133    };
134    let judged = answer.get("judged").and_then(Value::as_bool);
135    let stopped = answer.get("stopped_writer").and_then(Value::as_bool);
136    let verdict = match (judged, stopped) {
137        (Some(true), Some(true)) => "episodes close and no derivation followed the newest one",
138        (Some(true), _) => "a derivation follows the newest closed episode",
139        (Some(false), _) => "not judged; the findings say why",
140        (None, _) => "the status carries no judgement",
141    };
142    let fact = |name: &str, value: String| {
143        el("div")
144            .class("mj-ui-fact")
145            .child(el("dt").text(name))
146            .child(el("dd").text(value))
147    };
148    let facts = el("dl")
149        .class("mj-ui-facts")
150        .child(fact("Last derived", mark("last_derived")))
151        .child(fact("Last closed", mark("last_closed")))
152        .child(fact("Judgement", verdict.to_string()))
153        .child(fact("Why", text(answer, "freshness_reason")));
154    let findings = list(status, "/findings");
155    let body = el("div").class("mj-ui-stack").child(facts);
156    if findings.is_empty() {
157        return body;
158    }
159    body.child(
160        el("ul").class("mj-ui-prose-list").children(
161            findings
162                .iter()
163                .filter_map(Value::as_str)
164                .map(|finding| el("li").text(finding)),
165        ),
166    )
167}

169/// The review, as the commands it is made of.
170fn review() -> El {
171    el("dl")
172        .class("mj-ui-facts")
173        .children(REVIEW.iter().map(|(command, does)| {
174            el("div")
175                .class("mj-ui-fact")
176                .child(el("dt").child(el("code").class("mj-ui-mono").text(*command)))
177                .child(el("dd").text(*does))
178        }))
179}

181/// The subjects, the first [`SHOWN`] of them, each with how many records await review, how
182/// many were rejected and kept, and how many episodes observed it.
183fn subjects(queue: &Answer) -> El {
184    let groups = list(queue, "/groups");
185    let size = |group: &Value, key: &str| -> String {
186        group
187            .get(key)
188            .and_then(Value::as_array)
189            .map_or(0, Vec::len)
190            .to_string()
191    };
192    let rows = groups
193        .iter()
194        .take(SHOWN)
195        .map(|group| {
196            el("tr")
197                .child(el("td").text(text(group, "about")))
198                .child(el("td").class("mj-ui-dim").text(text(group, "route")))
199                .child(el("td").text(size(group, "candidates")))
200                .child(el("td").text(size(group, "rejected")))
201                .child(el("td").text(size(group, "episodes")))
202        })
203        .collect::<Vec<_>>();
204    let shown = rows.len();
205    let body = el("div").class("mj-ui-stack").child(table(
206        &["About", "Route", "Awaiting", "Rejected", "Episodes"],
207        rows,
208        "No record derived from an observation awaits review.",
209    ));
210    match more(shown, groups.len(), None) {
211        Some(line) => body.child(line),
212        None => body,
213    }
214}

216/// The candidates, the newest [`SHOWN`] of them first. The source orders by id, which begins
217/// with the episode, so the last of its list are the newest; the order is the source's and is
218/// only read from its other end.
219fn candidates(queue: &Answer) -> El {
220    let all = list(queue, "/candidates");
221    let rows = all
222        .iter()
223        .rev()
224        .take(SHOWN)
225        .map(|record| {
226            let id = text(record, "id");
227            let uri = text(record, "uri");
228            let from = record
229                .get("branch")
230                .and_then(Value::as_str)
231                .filter(|branch| !branch.is_empty())
232                .map_or_else(|| "not known in this checkout".to_string(), str::to_string);
233            el("tr")
234                .child(
235                    el("td").child(object(&uri, id)).child(
236                        el("span")
237                            .class("mj-ui-cell-title")
238                            .text(text(record, "title")),
239                    ),
240                )
241                .child(el("td").class("mj-ui-dim").text(text(record, "class")))
242                .child(el("td").class("mj-ui-dim").text(from))
243                .child(
244                    el("td")
245                        .child(state(record, "freshness"))
246                        .child(el("span").class("mj-ui-dim").text(waited(record))),
247                )
248        })
249        .collect::<Vec<_>>();
250    let shown = rows.len();
251    let body = el("div").class("mj-ui-stack").child(table(
252        &["Record", "Class", "From", "Waiting"],
253        rows,
254        "No candidate awaits review.",
255    ));
256    match more(shown, all.len(), None) {
257        Some(line) => body.child(line),
258        None => body,
259    }
260}

262/// How long a record has waited, in the largest whole unit: the source measured it in
263/// minutes, and says in `freshness_reason` when it could not.
264fn waited(record: &Value) -> String {
265    match record.get("age_minutes").and_then(Value::as_i64) {
266        Some(minutes) if minutes >= 2 * 24 * 60 => format!(" {} d", minutes / (24 * 60)),
267        Some(minutes) if minutes >= 2 * 60 => format!(" {} h", minutes / 60),
268        Some(minutes) => format!(" {minutes} min"),
269        None => format!(" {}", text(record, "freshness_reason")),
270    }
271}

280    fn answer(source: Source, value: Value) -> Answer {
281        Answer {
282            source,
283            route: None,
284            exposure: Default::default(),
285            value: Ok(value.into()),
286            took: std::time::Duration::ZERO,
287            standing: Standing::NOW,
288        }
289    }

291    fn candidate(n: usize) -> Value {
292        json!({
293            "id": format!("s-20261013{n:06}-aaaa-{n:012x}"),
294            "uri": format!("majordomus://knowledge/s-20261013{n:06}-aaaa-{n:012x}"),
295            "class": "lesson",
296            "title": format!("lesson {n}"),
297            "branch": if n % 2 == 0 { json!("master") } else { Value::Null },
298            "freshness": "aging",
299            "age_minutes": 90 * n,
300            "freshness_reason": "measured from the derivation",
301        })
302    }

304    /// The screen names both sources, draws every panel, and its review is text: no form,
305    /// no button and no link that would run a command.
306    #[test]
307    fn the_screen_reads_two_sources_and_runs_nothing() {
308        let repo = SyntheticRepository::small().expect("a synthetic repository");
309        let ctx = repo.context().expect("a context");
310        let view = screen(&ctx);
311        assert_eq!(view.title, "Knowledge");
312        let html = view.main.render();
313        for want in [
314            "The deriver",
315            "Review",
316            "By subject",
317            "Awaiting review",
318            "knowledge_base.status",
319            "knowledge_base.candidates",
320            "majordomus knowledge promote",
321        ] {
322            assert!(html.contains(want), "no {want}: {html}");
323        }
324        for control in ["<form", "<button", "method=\"post\""] {
325            assert!(
326                !html.contains(control),
327                "the screen carries {control}: {html}"
328            );
329        }
330    }

332    /// A queue longer than the bound is drawn newest first, to the bound, and says how many
333    /// it left out; each row leads to its record and says where the record came from.
334    #[test]
335    fn a_long_queue_is_bounded_newest_first_and_says_what_it_left_out() {
336        let total = SHOWN + 7;
337        let queue = answer(
338            CANDIDATES,
339            json!({ "total": total, "candidates": (0..total).map(candidate).collect::<Vec<_>>() }),
340        );
341        let html = candidates(&queue).render();
342        assert_eq!(html.matches("<tr>").count(), SHOWN + 1, "{html}");
343        let newest = format!("lesson {}", total - 1);
344        let oldest = "lesson 0<";
345        assert!(html.contains(&newest), "the newest is drawn: {html}");
346        assert!(!html.contains(oldest), "the oldest is left out: {html}");
347        assert!(
348            html.contains(&format!(
349                "The first {SHOWN} of {total}; 7 more not shown here."
350            )),
351            "{html}"
352        );
353        assert!(html.contains("not known in this checkout"), "{html}");
354        assert!(
355            html.contains(&crate::ui::view::object_href(&format!(
356                "majordomus://knowledge/s-20261013{:06}-aaaa-{:012x}",
357                total - 1,
358                total - 1
359            ))),
360            "{html}"
361        );
362    }

364    /// An empty queue is an answer, said in words; a wait is told in its largest whole unit.

366    fn an_empty_queue_says_so_and_a_wait_is_told_in_whole_units() {

379            " the record carries no date"

383    /// The deriver's panel says the judgement in words and every finding the status carries;
384    /// a checkout where it never ran says never, not an empty cell.
385    #[test]
386    fn the_deriver_is_judged_in_words_and_its_findings_are_listed() {
387        let stopped = answer(
388            STATUS,

391                "stopped_writer": true,

399            html.contains("episodes close and no derivation followed the newest one"),
