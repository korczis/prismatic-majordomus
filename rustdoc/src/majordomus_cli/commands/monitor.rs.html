1//! `majordomus monitor`: the terminal rendering of the `monitor` capability (ADR 0133).
2//!
3//! The stream is a fact of the running server's memory — the mesh journal lives there — so
4//! the command finds this checkout's server and asks it over HTTP, like `majordomus mesh`
5//! does. One page is printed with what is for the reader first; `--follow` asks again
6//! with the page's cursor every `--interval` seconds and prints only what is new, so a
7//! reconnect or a restarted server repeats nothing. `--format json` prints the page as the
8//! capability answered it, or, while following, one signal per line (NDJSON).
9//!
10//! Exit codes: 0 answered (or a follow ended by `--count` or `--until`); 10 refused (a bad
11//! cursor, an unknown source) or a follow that reached `--timeout` first.
12

22/// One bounded request to the running server; the GitHub source alone may take seconds.
23const REQUEST_TIMEOUT: Duration = Duration::from_secs(15);
24
25/// The exit code of a refusal, or of a follow that timed out.
26const EXIT_REFUSED: u8 = 10;
27
28/// Run `majordomus monitor`.
29pub fn run(args: MonitorArgs) -> Result<u8> {
30    run_with(args, server_url)
31}

33/// [`run`], finding the server with `discover`: this checkout's ready server in earnest, a
34/// test's own in a test.
35fn run_with(
36    args: MonitorArgs,
37    discover: impl Fn(&MonitorArgs) -> Result<Option<String>>,
38) -> Result<u8> {
39    let color = wants_color(
40        args.query.format,
41        std::io::stdout().is_terminal(),
42        std::env::var_os("NO_COLOR").is_none(),
43    );
44    let reader = args
45        .reader
46        .clone()
47        .or_else(|| std::env::var("MAJORDOMUS_MONITOR_FOR").ok())
48        .filter(|r| !r.trim().is_empty());
49    let started = Instant::now();
50    let mut after = args.after.clone();
51    let mut shown: u64 = 0;
52    let mut warned: Option<String> = None;
53    let mut first = true;
54    loop {
55        let url = discover(&args)?;
56        let page = match url {
57            None => Err(Asked::NoServer),
58            Some(url) => ask(&url, &target(&args, after.as_deref(), reader.as_deref())),
59        };
60        match page {
61            Err(Asked::Refused(message)) => {
62                eprintln!("monitor: {message}");
63                return Ok(EXIT_REFUSED);
64            }
65            Err(Asked::NoServer) if !args.follow => {
66                match args.query.format {
67                    OutputFormat::Json => {
68                        println!("{}", json!({"active": false, "reason": NO_SERVER}))
69                    }
70                    OutputFormat::Text => println!("monitor: {NO_SERVER}"),
71                }
72                return Ok(0);
73            }
74            Err(Asked::NoServer) => {
75                if warned.as_deref() != Some(NO_SERVER) {
76                    eprintln!(
77                        "monitor: {NO_SERVER}; asking again every {}s",
78                        args.interval
79                    );
80                    warned = Some(NO_SERVER.to_string());
81                }
82            }
83            Err(Asked::Unreachable(reason)) | Err(Asked::Other(reason)) if !args.follow => {
84                return Err(Error::Protocol { reason });
85            }
86            Err(Asked::Unreachable(reason)) | Err(Asked::Other(reason)) => {
87                if warned.as_deref() != Some(reason.as_str()) {
88                    eprintln!("monitor: {reason}; asking again every {}s", args.interval);
89                    warned = Some(reason);
90                }
91            }
92            Ok(page) => {
93                warned = None;
94                let signals: Vec<Value> = page["signals"]
95                    .as_array()
96                    .into_iter()
97                    .flatten()
98                    .filter(|s| matches(s, args.pattern.as_deref()))
99                    .cloned()
100                    .collect();
101                if !args.follow {
102                    match args.query.format {
103                        OutputFormat::Json => println!(
104                            "{}",
105                            serde_json::to_string_pretty(&page).unwrap_or_default()
106                        ),
107                        OutputFormat::Text => print!("{}", render_page(&page, &signals, color)),
108                    }
109                    return Ok(0);
110                }
111                if first && args.query.format == OutputFormat::Text {
112                    print!("{}", render_sources(&page, color));
113                }
114                first = false;
115                for signal in &signals {
116                    match args.query.format {
117                        OutputFormat::Json => println!("{signal}"),
118                        OutputFormat::Text => println!("{}", render_line(signal, color)),
119                    }
120                    shown += 1;
121                    let done_count = args.count.is_some_and(|n| shown >= n);
122                    let done_kind = args
123                        .until
124                        .as_deref()
125                        .is_some_and(|k| signal["kind"].as_str() == Some(k));
126                    if done_count || done_kind {
127                        let _ = std::io::stdout().flush();
128                        return Ok(0);
129                    }
130                }
131                let _ = std::io::stdout().flush();
132                if let Some(cursor) = page["cursor"].as_str() {
133                    after = Some(cursor.to_string());
134                }
135                if page["more"] == json!(true) {
136                    continue;
137                }
138            }
139        }
140        if args
141            .timeout
142            .is_some_and(|t| started.elapsed() >= Duration::from_secs(t))
143        {
144            eprintln!("monitor: --timeout reached before --count or --until was met");
145            return Ok(EXIT_REFUSED);
146        }
147        std::thread::sleep(Duration::from_secs(args.interval.max(1)));
148    }
149}

151/// Whether the text is painted: text, to a terminal, and `NO_COLOR` unset.
152fn wants_color(format: OutputFormat, terminal: bool, no_color_unset: bool) -> bool {
153    format == OutputFormat::Text && terminal && no_color_unset
154}

156/// What a reader is told when no server serves this checkout.
157const NO_SERVER: &str = "no running server for this checkout: the stream lives inside the shared server (majordomus serve ensure starts one)";
158
159/// What asking the server came to.
160enum Asked {
161    /// No server serves this checkout.
162    NoServer,
163    /// The capability refused the input: printed, exit 10.
164    Refused(String),
165    /// The server could not be reached.
166    Unreachable(String),
167    /// The server answered something that is not a page.
168    Other(String),
169}

171/// This checkout's ready server, when there is one.
172fn server_url(args: &MonitorArgs) -> Result<Option<String>> {
173    let app = App::load(&args.query.repo)?;
174    app.context
175        .execute("server.status", json!({}))
176        .map(|status| ready_url(&status))
177        .map_err(status_failed)
178}
179
180/// A `server.status` that could not be asked, as the command's error.
181fn status_failed(e: impl std::fmt::Display) -> Error {
182    Error::Protocol {
183        reason: format!("server.status: {e}"),
184    }
185}
186
187/// The address of this checkout's ready server in a `server.status` answer.
188fn ready_url(status: &Value) -> Option<String> {
189    status["servers"]
190        .as_array()
191        .into_iter()
192        .flatten()
193        .find(|s| s["this_checkout"] == json!(true) && s["standing"] == json!("ready"))
194        .and_then(|s| s["lease"]["url"].as_str())
195        .map(str::to_string)
196}

198/// The request target of one page.
199pub(crate) fn target(args: &MonitorArgs, after: Option<&str>, reader: Option<&str>) -> String {
200    let mut pairs = Vec::new();
201    if let Some(after) = after {
202        pairs.push(format!("after={}", encode(after)));
203    }
204    if let Some(limit) = args.limit {
205        pairs.push(format!("limit={limit}"));
206    }
207    if let Some(sources) = &args.sources {
208        pairs.push(format!("sources={}", encode(sources)));
209    }
210    if let Some(reader) = reader {
211        pairs.push(format!("for={}", encode(reader)));
212    }
213    if let Some(since) = args.since {
214        pairs.push(format!("since={since}"));
215    }
216    if pairs.is_empty() {
217        "/api/v1/monitor".into()
218    } else {
219        format!("/api/v1/monitor?{}", pairs.join("&"))
220    }
221}

223fn encode(text: &str) -> String {
224    text.bytes()
225        .map(|b| match b {
226            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
227                (b as char).to_string()
228            }
229            _ => format!("%{b:02X}"),
230        })
231        .collect()
232}
233
234fn ask(url: &str, target: &str) -> std::result::Result<Value, Asked> {
235    let reply = crate::mcp::bridge::request(url, "GET", target, &[], None, REQUEST_TIMEOUT)
236        .map_err(|e| Asked::Unreachable(format!("{url}{target}: {e}")))?;
237    let value: Value = serde_json::from_str(&reply.body).map_err(|e| {
238        Asked::Other(format!(
239            "{url}{target}: status {}, not JSON: {e}",
240            reply.status
241        ))
242    })?;
243    if reply.status == 200 {
244        return Ok(value);
245    }
246    match value.get("error") {
247        Some(error) => Err(Asked::Refused(format!(
248            "{}: {}",
249            error["code"].as_str().unwrap_or("error"),
250            error["message"].as_str().unwrap_or("")
251        ))),
252        None => Err(Asked::Other(format!(
253            "{url}{target}: status {}",
254            reply.status
255        ))),
256    }
257}

259/// Whether a signal's summary contains `pattern`, ignoring case; no pattern keeps everything.
260pub(crate) fn matches(signal: &Value, pattern: Option<&str>) -> bool {
261    let Some(pattern) = pattern.filter(|p| !p.is_empty()) else {
262        return true;
263    };
264    signal["summary"]
265        .as_str()
266        .unwrap_or_default()
267        .to_lowercase()
268        .contains(&pattern.to_lowercase())
269}

271fn paint(text: &str, code: &str, color: bool) -> String {
272    if color {
273        format!("\x1b[{code}m{text}\x1b[0m")
274    } else {
275        text.to_string()
276    }
277}

279/// `HH:MM:SS` of a Unix time, in UTC.
280pub(crate) fn clock(at: u64) -> String {
281    let day = at % 86_400;
282    format!("{:02}:{:02}:{:02}", day / 3600, day % 3600 / 60, day % 60)
283}

285/// One signal as one line: time, source, machine, a mark for its severity and for you, and
286/// the summary.
287pub(crate) fn render_line(signal: &Value, color: bool) -> String {
288    let at = clock(signal["at"].as_u64().unwrap_or(0));
289    let source = signal["source"].as_str().unwrap_or("?");
290    let machine = signal["machine"].as_str().unwrap_or("");
291    let summary = signal["summary"].as_str().unwrap_or("");
292    let (mark, code) = match signal["severity"].as_str() {
293        Some("action") => ("!!", "1;31"),
294        Some("attention") => ("! ", "33"),
295        _ => ("  ", "0"),
296    };
297    let you = if signal["for_you"] == json!(true) {
298        paint("> ", "1;36", color)
299    } else {
300        "  ".to_string()
301    };
302    let tag = format!("{source:<8} {machine:<10.10}");
303    format!(
304        "{} {} {}{} {}",
305        paint(&at, "2", color),
306        paint(&tag, "2", color),
307        you,
308        paint(mark, code, color),
309        paint(summary, code, color && code != "0")
310    )
311}

313/// The sources line: every source's state, with the reason of any that was not read.
314pub(crate) fn render_sources(page: &Value, color: bool) -> String {
315    let parts: Vec<String> = page["sources"]
316        .as_array()
317        .into_iter()
318        .flatten()
319        .map(|s| {
320            let name = s["source"].as_str().unwrap_or("?");
321            match s["state"].as_str() {
322                Some("ok") => format!("{name} ok"),
323                Some(state) => {
324                    let reason = s["reason"].as_str().unwrap_or("");
325                    paint(
326                        &format!("{name} {state} ({reason})"),
327                        "33",
328                        color && state != "skipped",
329                    )
330                }
331                None => name.to_string(),
332            }
333        })
334        .collect();
335    format!("sources  {}\n", parts.join(" · "))
336}

338/// A page for a person: what is for the reader first, then everything, then how much was
339/// folded away, the sources, and how to go on.
340pub(crate) fn render_page(page: &Value, signals: &[Value], color: bool) -> String {
341    let mut out = String::new();
342    let mine: Vec<&Value> = signals
343        .iter()
344        .filter(|s| s["for_you"] == json!(true))
345        .collect();
346    if !mine.is_empty() {
347        out.push_str(&paint(&format!("For you ({})", mine.len()), "1;36", color));
348        out.push('\n');
349        for s in &mine {
350            out.push_str(&render_line(s, color));
351            out.push('\n');
352        }
353        out.push('\n');
354    }
355    if signals.is_empty() {
356        out.push_str("nothing happened in this window\n");
357    }
358    for s in signals {
359        out.push_str(&render_line(s, color));
360        out.push('\n');
361    }
362    out.push('\n');
363    out.push_str(&format!(
364        "{} signal(s), {} folded away as noise\n",
365        signals.len(),
366        page["suppressed"].as_u64().unwrap_or(0)
367    ));
368    out.push_str(&render_sources(page, color));
369    if let Some(cursor) = page["cursor"].as_str().filter(|c| !c.is_empty()) {
370        out.push_str(&format!(
371            "next     majordomus monitor --after '{cursor}'   (or --follow)\n"
372        ));
373    }
374    out
375}

414    /// A server that answers the requests it gets with `replies`, in order, the last one
415    /// for ever after; it lives as long as the test process.
416    fn scripted(replies: Vec<(u16, String)>) -> String {
417        let server = tiny_http::Server::http("127.0.0.1:0").unwrap();
418        let url = format!("http://{}", server.server_addr().to_ip().unwrap());
419        std::thread::spawn(move || {
420            let mut replies = replies.into_iter().peekable();
421            for request in server.incoming_requests() {
422                let (status, body) = match replies.next() {
423                    Some(r) if replies.peek().is_some() => r,
424                    Some(last) => {
425                        replies = vec![last.clone()].into_iter().peekable();
426                        last
427                    }
428                    None => (500, String::new()),
429                };
430                let _ = request
431                    .respond(tiny_http::Response::from_string(body).with_status_code(status));
432            }
433        });
434        url
435    }
