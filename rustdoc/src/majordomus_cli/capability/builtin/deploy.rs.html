1//! The `deploy` module: what this repository's deployments are, and whether they would
2//! work — read-only, from the canonical objects the index already holds.
3//!
4//! Every operation here is a read. A deployment is mutated by the trusted command line and
5//! by CI, never by an HTTP request and never by an MCP client: an operation that could
6//! start or stop a machine has no business behind a surface a browser can reach.
7//!
8//! No capability here contacts the provider. Live provider state costs a network call, and
9//! a network call on a read that a client may poll is not an ordinary read; it belongs to
10//! an explicit operation with its own cost, not behind `deploy.list`.
11

25/// The URI under which the deployments are read as one MCP resource.
26pub const DEPLOYMENTS_URI: &str = "majordomus://deployments";

28/// One deployment as this process reads it, with where it came from.
29#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
30pub struct DeploymentView {
31    /// The repository-relative file the object lives in.
32    pub file: String,
33    /// The object, typed.
34    #[serde(flatten)]
35    pub deployment: Deployment,
36}
37
38/// Every deployment the layer declares.
39#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
40pub struct DeploymentList {
41    /// How many.
42    pub count: usize,
43    /// Each, in identity order.
44    pub deployments: Vec<DeploymentView>,
45    /// Objects of the deployment kind this executable could not read, with why. A
46    /// malformed object is reported here rather than hidden by being skipped.
47    #[serde(default, skip_serializing_if = "Vec::is_empty")]
48    pub unreadable: Vec<Refusal>,
49}

51/// The input of `deploy.get`: which deployment.
52#[derive(Debug, Serialize, Deserialize, JsonSchema)]
53#[serde(deny_unknown_fields)]
54pub struct GetDeploymentInput {
55    /// The deployment's `id`.
56    pub id: String,
57}
58
59/// Whether the deployments this repository declares would work, decided locally: against
60/// the capability registry this process built and the workspace it sits in.
61#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
62pub struct DeploymentCheck {
63    /// True when no deployment earned a refusal.
64    pub ok: bool,
65    /// How many deployments were decided.
66    pub decided: usize,
67    /// Every refusal, in the order the objects and their keys appear.
68    pub refusals: Vec<Refusal>,
69}

71/// The view orders as the deployment it wraps: the file path it adds is provenance, not a
72/// second opinion about the sequence.
73impl crate::order::Ordered for DeploymentView {
74    fn order_key(&self) -> crate::order::OrderKey<'_> {
75        self.deployment.order_key()
76    }
77}

79fn views(ctx: &Context) -> (Vec<DeploymentView>, Vec<Refusal>) {
80    let mut out = Vec::new();
81    let mut bad = Vec::new();
82    for object in ctx.index.objects.iter().filter(|o| o.kind == KIND) {
83        match Deployment::parse(object) {
84            Ok(deployment) => out.push(DeploymentView {
85                file: object.provenance.path.clone(),
86                deployment,
87            }),
88            Err(refusal) => bad.push(refusal),
89        }
90    }
91    crate::order::canonical(&mut out);
92    (out, bad)
93}
94
95fn deploy_list(ctx: &Context, _: Empty) -> Result<DeploymentList, CapabilityError> {
96    let (deployments, unreadable) = views(ctx);
97    Ok(DeploymentList {
98        count: deployments.len(),
99        deployments,
100        unreadable,
101    })
102}
103
104fn deploy_get(ctx: &Context, input: GetDeploymentInput) -> Result<DeploymentView, CapabilityError> {
105    let (deployments, _) = views(ctx);
106    deployments
107        .into_iter()
108        .find(|d| d.deployment.id == input.id)
109        .ok_or_else(|| {
110            CapabilityError::NotFound(format!(
111                "no deployment '{}' (the deployments are read from the layer's deployments section)",
112                input.id
113            ))
114        })
115}
116
117fn deploy_check(ctx: &Context, _: Empty) -> Result<DeploymentCheck, CapabilityError> {
118    let (deployments, unreadable) = views(ctx);
119    let routes = ctx
120        .registry
121        .iter()
122        .filter_map(|c| c.exposure.http.as_ref().map(|h| h.path.clone()))
123        .collect();
124    let ws = Workspace::read(std::path::Path::new(&ctx.index.repository.root), routes);
125    let mut refusals = unreadable;
126    for view in &deployments {
127        refusals.extend(view.deployment.check(&view.file, &ws));
128    }
129    Ok(DeploymentCheck {
130        ok: refusals.is_empty(),
131        decided: deployments.len(),
132        refusals,
133    })
134}

214#[cfg(test)]
215mod tests {
216    use super::*;
217    use crate::capability::model::{CapabilityKind, HttpMethod};
218
219    /// Nothing here mutates. A deployment is changed by the trusted command line and by
220    /// CI; an operation that could start, stop or replace a machine must not be reachable
221    /// from a browser or from an MCP client, and `Query` is what says so to every
222    /// projection at once — read-only in MCP, `GET` in HTTP and OpenAPI.
223    #[test]
224    fn every_deployment_operation_is_a_read() {
225        for c in module().capabilities {
226            assert_eq!(
227                c.capability.kind,
228                CapabilityKind::Query,
229                "{} is not a read",
230                c.capability.id
231            );
232            if let Some(http) = &c.capability.exposure.http {
233                assert_eq!(
234                    http.method,
235                    HttpMethod::Get,
236                    "{} is not a GET",
237                    c.capability.id
238                );
239            }
240        }
241    }
242
243    /// The declaration is the whole registration: the ids, the routes and the MCP names
244    /// exist once, here, and every projection is derived from them. A test that pinned a
245    /// second list would be the defect; this one pins that the module composes what it
246    /// says it does.
247    #[test]
248    fn the_module_declares_the_reads_and_nothing_else() {
249        let ids: Vec<String> = module()
250            .capabilities
251            .iter()
252            .map(|c| c.capability.id.to_string())
253            .collect();
254        assert_eq!(ids, ["deploy.list", "deploy.get", "deploy.check"]);
255    }
256
257    /// A deployment id the layer does not have is refused by name rather than answered
258    /// with an empty result: an absent deployment and a deployment with nothing in it are
259    /// different facts.
260    #[test]
261    fn an_absent_deployment_is_not_found() {
262        let repo = crate::synthetic::SyntheticRepository::new(crate::synthetic::Shape::default())
263            .expect("a synthetic repository");
264        let ctx = repo.context().expect("a context");
265        let e = deploy_get(
266            &ctx,
267            GetDeploymentInput {
268                id: "absent".into(),
269            },
270        )
271        .expect_err("refused");
272        assert!(matches!(e, CapabilityError::NotFound(_)), "{e}");
273    }
274
275    /// A layer that declares no deployment is not a failure: the list is empty and the
276    /// check is satisfied, because there is nothing that could be wrong.
277    #[test]
278    fn a_layer_with_no_deployments_lists_none_and_checks_clean() {
279        let repo = crate::synthetic::SyntheticRepository::new(crate::synthetic::Shape::default())
280            .expect("a synthetic repository");
281        let ctx = repo.context().expect("a context");
282        assert_eq!(deploy_list(&ctx, Empty {}).expect("listed").count, 0);
283        let check = deploy_check(&ctx, Empty {}).expect("checked");
284        assert!(check.ok);
285        assert_eq!(check.decided, 0);
286    }
