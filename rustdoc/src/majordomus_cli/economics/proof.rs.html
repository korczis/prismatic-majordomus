1//! The proof plan: the tests that are to settle what Majordomus is worth, registered before
2//! any of their records exists, and each protocol frozen by its own digest.
3//!
4//! `.ai/repo/benchmarks/proof-plan.yaml` (schema `proof-plan/v1`) names every proof test:
5//! the question it answers, the claims of `docs/CLAIMS.yaml` it settles, the dimension it
6//! measures, the instrument that measures it, and, once it is registered, its protocol. A
7//! protocol is how the test is decided, fixed in advance: a paired benchmark over an
8//! economics suite's registration block, a count of trials, or an observation window.
9//!
10//! A protocol carries `registered_digest`, the [`digest`] of its own canonical form, and a
11//! record counts toward its test only when it carries that digest and was recorded after
12//! `registered_at` ([`counts_toward`]). Each protocol is frozen by its own digest rather
13//! than by the plan file's, so registering a second test's protocol never makes the first
14//! test's records stale, and the plan file is a freshness input of no suite: [`load`]
15//! refuses a suite whose freshness inputs reach it.
16//!
17//! The canonical form is the protocol as the YAML reader reads it, without its
18//! `registered_digest`, with every object's keys sorted and no whitespace. It is the data
19//! as written, not a typed value written back, so a later version of this code that adds a
20//! defaulted field changes no digest and stales no record.
21//!
22//! ```
23//! use majordomus_cli::economics::proof::{load, NO_PLAN};
24//! let dir = tempfile::tempdir().unwrap();
25//! let findings = load(dir.path()).unwrap_err();
26//! assert_eq!(findings[0].kind, "no_plan");
27//! assert!(findings[0].detail.starts_with(NO_PLAN), "an absent plan is not an empty plan");
28//! ```

199/// How a protocol decides its test.
200///
201/// ```
202/// use majordomus_cli::economics::proof::ProofDecision;
203/// let d: ProofDecision = serde_json::from_value(serde_json::json!({
204///     "kind": "trials", "fields": [{ "name": "minutes", "class": "observed" }],
205///     "min_trials": 3, "min_successes": 2, "all_must_pass": false
206/// }))
207/// .unwrap();
208/// assert_eq!(d.kind(), "trials");
209/// ```
210#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
211#[serde(tag = "kind", rename_all = "snake_case", deny_unknown_fields)]
212pub enum ProofDecision {
213    /// A paired benchmark over an economics suite, pre-registered by the suite's own
214    /// registration block: the protocol names the suite and that block's [`digest`].
215    Paired {
216        /// The economics suite's id.
217        suite: String,
218        /// The digest of the suite's `registration` block.
219        registration_digest: String,
220    },
221    /// Independent trials, each a record of the declared fields.
222    Trials {
223        /// The fields every trial record carries.
224        fields: Vec<ProofField>,
225        /// How many trials decide the test.
226        min_trials: u32,
227        /// How many of them must succeed.
228        min_successes: u32,
229        /// Whether one failed trial decides against.
230        all_must_pass: bool,
231    },
232    /// An observation window over admitted work.
233    Window {
234        /// The period observed.
235        period: ProofPeriod,
236        /// The work the window admits.
237        admits: String,
238        /// For each field, the largest share of admitted work it may leave not observed.
239        bounds: Vec<ProofBound>,
240    },
241}

317/// The one digest of the proof plan: SHA-256 over the canonical JSON of `value` (object
318/// keys sorted at every depth, no whitespace), spelled `sha256:<hex>`. A protocol's
319/// `registered_digest` and a paired protocol's `registration_digest` are both this
320/// function, so pre-registration is written once and read the same way everywhere.
321///
322/// ```
323/// use majordomus_cli::economics::proof::digest;
324/// let a = digest(&serde_json::json!({ "b": 1, "a": [1, 2] }));
325/// assert_eq!(a, digest(&serde_json::json!({ "a": [1, 2], "b": 1 })), "key order is not data");
326/// assert_ne!(a, digest(&serde_json::json!({ "a": [2, 1], "b": 1 })), "list order is");
327/// assert!(a.starts_with("sha256:") && a.len() == 7 + 64);
328/// ```
329pub fn digest(value: &Value) -> String {
330    let bytes = crate::mesh::journal::canonical_json(value);
331    format!("sha256:{}", crate::policy::sha256_bytes_hex(&bytes))
332}
333
334/// The digest a protocol registers: [`digest`] of the protocol as written, without its own
335/// `registered_digest`.
336///
337/// ```
338/// use majordomus_cli::economics::proof::{digest, protocol_digest};
339/// let p = serde_json::json!({ "version": 1, "registered_digest": "sha256:stale" });
340/// assert_eq!(protocol_digest(&p), digest(&serde_json::json!({ "version": 1 })));
341/// ```
342pub fn protocol_digest(protocol: &Value) -> String {
343    let mut canonical = protocol.clone();
344    if let Value::Object(map) = &mut canonical {
345        map.remove("registered_digest");
346    }
347    digest(&canonical)
348}

410/// Whether a record counts toward a test, and when it does not, why.
411///
412/// ```
413/// use majordomus_cli::economics::proof::ProofCount;
414/// assert!(ProofCount::Counts.counts());
415/// assert!(!ProofCount::NoProtocol.counts());
416/// ```
417#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
418#[serde(tag = "status", rename_all = "snake_case")]
419pub enum ProofCount {
420    /// It carries the test's protocol digest and postdates its registration.
421    Counts,
422    /// The test has no protocol yet, so nothing counts toward it.
423    NoProtocol,
424    /// It was recorded under another protocol.
425    OtherProtocol {
426        /// The digest the record carries.
427        carried: String,
428        /// The digest the test's protocol registers.
429        registered: String,
430    },
431    /// It was recorded at or before the protocol's registration.
432    BeforeRegistration {
433        /// When the record was made.
434        recorded_at: String,
435        /// When the protocol was registered.
436        registered_at: String,
437    },
438    /// Its time, or the protocol's, is not `YYYY-MM-DDTHH:MM:SSZ`.
439    UnreadableTime {
440        /// The time that could not be read.
441        time: String,
442    },
443}
444
445impl ProofCount {
446    /// Whether the record counts toward its test: true only for [`ProofCount::Counts`].
447    ///
448    /// ```
449    /// use majordomus_cli::economics::proof::ProofCount;
450    /// let other = ProofCount::OtherProtocol { carried: "a".into(), registered: "b".into() };
451    /// assert!(!other.counts());
452    /// ```
453    pub fn counts(&self) -> bool {
454        matches!(self, ProofCount::Counts)
455    }
456
457    /// Why the record does not count, for a list a person reads; `None` when it counts.
458    ///
459    /// ```
460    /// use majordomus_cli::economics::proof::ProofCount;
461    /// assert_eq!(ProofCount::Counts.reason(), None);
462    /// let early = ProofCount::BeforeRegistration {
463    ///     recorded_at: "2026-10-19T00:00:00Z".into(),
464    ///     registered_at: "2026-10-20T00:00:00Z".into(),
465    /// };
466    /// assert!(early.reason().unwrap().contains("before the protocol was registered"));
467    /// ```
468    pub fn reason(&self) -> Option<String> {
469        match self {
470            ProofCount::Counts => None,
471            ProofCount::NoProtocol => Some("the test has no registered protocol".into()),
472            ProofCount::OtherProtocol {
473                carried,
474                registered,
475            } => Some(format!(
476                "recorded under protocol {carried}, and the test registers {registered}"
477            )),
478            ProofCount::BeforeRegistration {
479                recorded_at,
480                registered_at,
481            } => Some(format!(
482                "recorded at {recorded_at}, not after {registered_at}, before the protocol \
483                 was registered"
484            )),
485            ProofCount::UnreadableTime { time } => Some(format!(
486                "{time:?} is not a time of the form YYYY-MM-DDTHH:MM:SSZ"
487            )),
488        }
489    }
490}

492/// Whether `record` counts toward `test`: only when the test has a protocol, the record
493/// carries that protocol's digest, and it was recorded strictly after `registered_at`. A
494/// test's own protocol decides, so another test's registration changes nothing here.
495///
496/// ```
497/// use majordomus_cli::economics::proof::*;
498/// let mut test = ProofTest {
499///     id: "t3".into(), title: "t".into(), question: "q".into(), settles: vec!["c".into()],
500///     dimension: "wall_time".into(), value_words: vec![],
501///     instrument: ProofInstrument { status: "built".into(), issues: vec![] }, protocol: None,
502/// };
503/// let stamp = |d: &str, at: &str| ProofStamp {
504///     protocol_digest: d.into(),
505///     recorded_at: at.into(),
506/// };
507/// let late = stamp("sha256:p", "2026-10-21T00:00:00Z");
508/// assert_eq!(counts_toward(&late, &test), ProofCount::NoProtocol);
509/// test.protocol = Some(ProofProtocol {
510///     version: 1, registered_at: "2026-10-20T00:00:00Z".into(),
511///     changes: vec![ProofChange { version: 1, why: "First.".into() }],
512///     decision: ProofDecision::Paired { suite: "pilot".into(), registration_digest: "r".into() },
513///     registered_digest: "sha256:p".into(),
514/// });
515/// assert!(counts_toward(&late, &test).counts());
516/// assert!(!counts_toward(&stamp("sha256:q", "2026-10-21T00:00:00Z"), &test).counts());
517/// assert!(!counts_toward(&stamp("sha256:p", "2026-10-20T00:00:00Z"), &test).counts());
518/// ```
519pub fn counts_toward(record: &impl ProofRecord, test: &ProofTest) -> ProofCount {
520    let Some(protocol) = &test.protocol else {
521        return ProofCount::NoProtocol;
522    };
523    if record.protocol_digest() != protocol.registered_digest {
524        return ProofCount::OtherProtocol {
525            carried: record.protocol_digest().to_string(),
526            registered: protocol.registered_digest.clone(),
527        };
528    }
529    let time = |t: &str| crate::peers::parse_rfc3339(t).ok_or(t.to_string());
530    match (time(record.recorded_at()), time(&protocol.registered_at)) {
531        (Ok(at), Ok(registered)) if at > registered => ProofCount::Counts,
532        (Ok(_), Ok(_)) => ProofCount::BeforeRegistration {
533            recorded_at: record.recorded_at().to_string(),
534            registered_at: protocol.registered_at.clone(),
535        },
536        (Err(time), _) | (_, Err(time)) => ProofCount::UnreadableTime { time },
537    }
538}

540/// Read and validate the plan under `root`, fail-closed: the plan when no finding stands,
541/// every finding otherwise. An absent file is the finding `no_plan`, never an empty valid
542/// plan. The kinds, each with its own detail:
543///
544/// - `no_plan`, `plan_unreadable`, `schema_unknown`, `no_tests`: the file as a whole.
545/// - `test_malformed`: a test or its protocol does not have the declared shape.
546/// - `test_duplicate`, `dimension_unknown`, `claim_unknown`, `settles_nothing`,
547///   `instrument_unknown`, `instrument_unplanned`: a test's own fields.
548/// - `decision_unknown`, `digest_mismatch`, `registered_at_invalid`, `version_unchanged`:
549///   a protocol.
550/// - `paired_suite_unknown`, `paired_registration_absent`, `paired_registration_mismatch`:
551///   a paired protocol whose suite or registration digest does not resolve.
552/// - `trials_invalid`, `window_invalid`: a trials or window decision that cannot decide.
553/// - `claims_unreadable`, `suite_unreadable`, `suite_duplicate`: an input the checks read, or
554///   two suite files that declare one id.
555/// - `plan_is_freshness_input`: a suite's freshness inputs could select the plan file, read
556///   as `git ls-files` reads a pathspec list, with its wildcards and magic.
557///
558/// ```
559/// use majordomus_cli::economics::proof::{load, PLAN};
560/// let dir = tempfile::tempdir().unwrap();
561/// let plan = dir.path().join(PLAN);
562/// std::fs::create_dir_all(plan.parent().unwrap()).unwrap();
563/// std::fs::write(&plan, "schema: proof-plan/v0\ntests: []\n").unwrap();
564/// let kinds: Vec<String> = load(dir.path()).unwrap_err().into_iter().map(|f| f.kind).collect();
565/// assert_eq!(kinds, ["schema_unknown", "no_tests"]);
566/// ```
567pub fn load(root: &Path) -> Result<ProofPlan, Vec<EconomicsFinding>> {
568    let text = match std::fs::read_to_string(root.join(PLAN)) {
569        Ok(text) => text,
570        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
571            return Err(vec![finding(
572                "no_plan",
573                PLAN,
574                format!("{NO_PLAN}: write {PLAN} (schema {PLAN_SCHEMA})"),
575            )]);
576        }
577        Err(e) => return Err(vec![finding("plan_unreadable", PLAN, e.to_string())]),
578    };
579    let map = yaml::parse_mapping(&text).map_err(|e| vec![finding("plan_unreadable", PLAN, e)])?;
580    validate(root, &map)
581}

592/// An economics suite as written: its file and its parsed mapping.
593struct RawSuite {
594    path: String,
595    map: Map<String, Value>,
596}
597
598/// Every economics suite, by id, read as data, with the files that could not be read. A
599/// suite without an id, or with an id another file already declares, is a finding and is
600/// not keyed, so a paired protocol never resolves against the wrong file.
601fn raw_suites(root: &Path, findings: &mut Vec<EconomicsFinding>) -> BTreeMap<String, RawSuite> {
602    let mut out = BTreeMap::new();
603    for p in super::sorted_entries(&root.join(super::DIR).join("suites")) {
604        if p.extension().and_then(|e| e.to_str()) != Some("yaml") {
605            continue;
606        }
607        let shown = format!(
608            "{}/suites/{}",
609            super::DIR,
610            p.file_name().unwrap_or_default().to_string_lossy()
611        );
612        let parsed = std::fs::read_to_string(&p)
613            .map_err(|e| e.to_string())
614            .and_then(|t| yaml::parse_mapping(&t));
615        match parsed {
616            Ok(map) => match map.get("id").and_then(Value::as_str).map(str::to_string) {
617                None => findings.push(finding("suite_unreadable", &shown, "it has no id".into())),
618                Some(id) => match out.get(&id) {
619                    Some(RawSuite { path, .. }) => findings.push(finding(
620                        "suite_duplicate",
621                        &shown,
622                        format!("it declares suite {id}, which {path} already declares"),
623                    )),
624                    None => {
625                        out.insert(id, RawSuite { path: shown, map });
626                    }
627                },
628            },
629            Err(e) => findings.push(finding("suite_unreadable", &shown, e)),
630        }
631    }
632    out
633}

635/// The plan file and every directory above it: what a pathspec reaches the plan through.
636fn plan_and_ancestors() -> Vec<&'static str> {
637    let mut out = vec![PLAN];
638    let mut at = PLAN;
639    while let Some(i) = at.rfind('/') {
640        at = &PLAN[..i];
641        out.push(at);
642    }
643    out
644}

646/// One git pathspec as `git ls-files` reads it: its magic, then the pattern.
647struct Pathspec<'a> {
648    pattern: &'a str,
649    exclude: bool,
650    literal: bool,
651    icase: bool,
652}
653
654impl<'a> Pathspec<'a> {
655    /// Read the long (`:(glob,icase)x`) and short (`:/x`, `:!x`, `:^x`) magic. A magic word
656    /// this does not know only narrows what git selects, so it is read as no narrowing.
657    fn parse(spec: &'a str) -> Pathspec<'a> {
658        let (mut exclude, mut literal, mut icase) = (false, false, false);
659        let mut rest = spec;
660        if let Some(long) = spec.strip_prefix(":(") {
661            if let Some(end) = long.find(')') {
662                for word in long[..end].split(',').map(str::trim) {
663                    match word {
664                        "exclude" => exclude = true,
665                        "literal" => literal = true,
666                        "icase" => icase = true,
667                        _ => {}
668                    }
669                }
670                rest = &long[end + 1..];
671            }
672        } else if let Some(short) = spec.strip_prefix(':') {
673            let end = short.find(|c| !"/!^".contains(c)).unwrap_or(short.len());
674            exclude = short[..end].contains(['!', '^']);
675            rest = short[end..].strip_prefix(':').unwrap_or(&short[end..]);
676        }
677        while let Some(r) = rest.strip_prefix("./") {
678            rest = r;
679        }
680        Pathspec {
681            pattern: rest.trim_end_matches('/'),
682            exclude,
683            literal,
684            icase,
685        }
686    }
687
688    fn wild(&self) -> bool {
689        !self.literal && self.pattern.contains(['*', '?', '['])
690    }
691
692    /// Whether it selects the plan file, whose path is lower case. `wide` ignores case and
693    /// reads every wildcard as crossing `/`, which selects at least what git selects; without
694    /// it only the exact name of the file or a directory above it counts, which selects at
695    /// most what git selects.
696    fn selects_plan(&self, wide: bool) -> bool {
697        let p = if wide || self.icase {
698            self.pattern.to_lowercase()
699        } else {
700            self.pattern.to_string()
701        };
702        if p.is_empty() || p == "." {
703            return true;
704        }
705        plan_and_ancestors().into_iter().any(|path| {
706            p == path || (wide && self.wild() && wildmatch(p.as_bytes(), path.as_bytes()))
707        })
708    }
709}
710
711/// A wildcard match where `*` crosses `/`, `?` is one byte and `[..]` one byte of a class,
712/// with `**/` also matching no directory: at least what either git pathspec reading selects.
713fn wildmatch(p: &[u8], t: &[u8]) -> bool {
714    match p.first() {
715        None => t.is_empty(),
716        Some(b'*') => {
717            let stars = p.iter().take_while(|&&c| c == b'*').count();
718            let rest = &p[stars..];
719            if stars > 1 && rest.first() == Some(&b'/') && wildmatch(&rest[1..], t) {
720                return true;
721            }
722            (0..=t.len()).any(|i| wildmatch(rest, &t[i..]))
723        }
724        Some(b'?') => !t.is_empty() && wildmatch(&p[1..], &t[1..]),
725        Some(b'[') => match (p[1..].iter().position(|&c| c == b']'), t.first()) {
726            (Some(close), Some(&c)) if close > 0 => {
727                let class = &p[1..=close];
728                let (negated, class) = match class.first() {
729                    Some(b'!' | b'^') => (true, &class[1..]),
730                    _ => (false, class),
731                };
732                let mut hit = false;
733                let mut i = 0;
734                while i < class.len() {
735                    if i + 2 < class.len() && class[i + 1] == b'-' {
736                        hit |= class[i] <= c && c <= class[i + 2];
737                        i += 3;
738                    } else {
739                        hit |= class[i] == c;
740                        i += 1;
741                    }
742                }
743                hit != negated && wildmatch(&p[close + 2..], &t[1..])
744            }
745            _ => t.first() == Some(&b'[') && wildmatch(&p[1..], &t[1..]),
746        },
747        Some(&c) => t.first() == Some(&c) && wildmatch(&p[1..], &t[1..]),
748    }
749}

751/// What carries a suite's freshness inputs to the plan file, read as `git ls-files` reads
752/// the list and failing toward refusal: an including pathspec that could select the plan, or
753/// a list that selects every file because it is empty or only excludes. An excluding
754/// pathspec removes the plan only when it names the file or a directory above it.
755fn reaching_input(specs: &[&str]) -> Option<String> {
756    let parsed: Vec<(&str, Pathspec)> = specs.iter().map(|s| (*s, Pathspec::parse(s))).collect();
757    if parsed
758        .iter()
759        .any(|(_, p)| p.exclude && p.selects_plan(false))
760    {
761        return None;
762    }
763    let mut includes = parsed.iter().filter(|(_, p)| !p.exclude).peekable();
764    if includes.peek().is_none() {
765        return Some(
766            "a freshness_inputs list with no including pathspec, which git reads as every \
767             tracked file,"
768                .into(),
769        );
770    }
771    includes
772        .find(|(_, p)| p.selects_plan(true))
773        .map(|(s, _)| format!("freshness input {s:?}"))
774}

776fn validate(root: &Path, map: &Map<String, Value>) -> Result<ProofPlan, Vec<EconomicsFinding>> {
777    let mut findings = Vec::new();
778    let schema = map.get("schema").and_then(Value::as_str).unwrap_or("");
779    if schema != PLAN_SCHEMA {
780        findings.push(finding(
781            "schema_unknown",
782            PLAN,
783            format!("the plan's schema is {schema:?}; write {PLAN_SCHEMA}"),
784        ));
785    }
786    let raw_tests: Vec<Value> = match map.get("tests") {
787        Some(Value::Array(items)) => items.clone(),
788        _ => Vec::new(),
789    };
790    if raw_tests.is_empty() {
791        findings.push(finding(
792            "no_tests",
793            PLAN,
794            "the plan registers no test; a plan is the list of tests it registers".into(),
795        ));
796    }
797    let claims = match super::claims::claim_ids(root) {
798        Ok(ids) => ids,
799        Err(e) => {
800            findings.push(finding("claims_unreadable", super::claims::CLAIMS, e));
801            None
802        }
803    };
804    let suites = raw_suites(root, &mut findings);
805    for suite in suites.values() {
806        let Some(inputs) = suite.map.get("freshness_inputs").and_then(Value::as_array) else {
807            continue;
808        };
809        let specs: Vec<&str> = inputs.iter().filter_map(Value::as_str).collect();
810        if let Some(what) = reaching_input(&specs) {
811            findings.push(finding(
812                "plan_is_freshness_input",
813                &suite.path,
814                format!(
815                    "{what} reaches {PLAN}: each protocol is frozen by its own digest, so the \
816                     plan is a freshness input of no suite"
817                ),
818            ));
819        }
820    }
821    let mut seen = BTreeSet::new();
822    let mut tests = Vec::new();
823    for (i, raw) in raw_tests.iter().enumerate() {
824        let id = raw
825            .get("id")
826            .and_then(Value::as_str)
827            .map(str::to_string)
828            .unwrap_or(format!("#{}", i + 1));
829        if !seen.insert(id.clone()) {
830            findings.push(finding(
831                "test_duplicate",
832                PLAN,
833                format!("test {id} is registered twice; a test id is unique in the plan"),
834            ));
835        }
836        let kind = raw
837            .pointer("/protocol/decision/kind")
838            .and_then(Value::as_str);
839        let mut raw = raw.clone();
840        if let Some(kind) = kind.filter(|k| !DECISION_KINDS.contains(k)) {
841            findings.push(finding(
842                "decision_unknown",
843                PLAN,
844                format!(
845                    "test {id}: decision kind {kind:?} is not one of {}",
846                    DECISION_KINDS.join(", ")
847                ),
848            ));
849            // the test's own fields are still judged, without the protocol that cannot be read
850            let _ = raw.as_object_mut().map(|test| test.remove("protocol"));
851        }
852        match serde_json::from_value::<ProofTest>(raw.clone()) {
853            Ok(test) => {
854                let protocol = raw.get("protocol").cloned().unwrap_or(Value::Null);
855                check_test(&test, &protocol, claims.as_ref(), &suites, &mut findings);
856                tests.push(test);
857            }
858            Err(e) => findings.push(finding("test_malformed", PLAN, format!("test {id}: {e}"))),
859        }
860    }
861    if findings.is_empty() {
862        Ok(ProofPlan {
863            schema: schema.to_string(),
864            tests,
865        })
866    } else {
867        Err(findings)
868    }
869}

871fn check_test(
872    test: &ProofTest,
873    raw_protocol: &Value,
874    claims: Option<&BTreeSet<String>>,
875    suites: &BTreeMap<String, RawSuite>,
876    findings: &mut Vec<EconomicsFinding>,
877) {
878    let id = &test.id;
879    let mut refuse = |kind: &str, detail: String| {
880        findings.push(finding(kind, PLAN, format!("test {id}: {detail}")));
881    };
882    if !DIMENSIONS.contains(&test.dimension.as_str()) {
883        refuse(
884            "dimension_unknown",
885            format!(
886                "dimension {:?} is not one of {}",
887                test.dimension,
888                DIMENSIONS.join(", ")
889            ),
890        );
891    }
892    if test.settles.is_empty() {
893        refuse(
894            "settles_nothing",
895            "it settles no claim; name the claims of docs/CLAIMS.yaml it decides".into(),
896        );
897    }
898    for claim in &test.settles {
899        if !claims.is_some_and(|ids| ids.contains(claim)) {
900            refuse(
901                "claim_unknown",
902                format!("it settles {claim}, which docs/CLAIMS.yaml does not declare"),
903            );
904        }
905    }
906    match (
907        test.instrument.status.as_str(),
908        test.instrument.issues.is_empty(),
909    ) {
910        ("planned", true) => refuse(
911            "instrument_unplanned",
912            "its instrument is planned and names no issue that builds it".into(),
913        ),
914        ("planned" | "built", _) => {}
915        (other, _) => refuse(
916            "instrument_unknown",
917            format!("instrument status {other:?} is not built or planned"),
918        ),
919    }
920    let Some(protocol) = &test.protocol else {
921        return;
922    };
923    let computed = protocol_digest(raw_protocol);
924    if computed != protocol.registered_digest {
925        refuse(
926            "digest_mismatch",
927            format!(
928                "the protocol registers {} and its canonical form digests to {computed}; a \
929                 changed protocol is a new version with a changes entry and a new digest",
930                protocol.registered_digest
931            ),
932        );
933    }
934    if crate::peers::parse_rfc3339(&protocol.registered_at).is_none() {
935        refuse(
936            "registered_at_invalid",
937            format!(
938                "registered_at {:?} is not YYYY-MM-DDTHH:MM:SSZ",
939                protocol.registered_at
940            ),
941        );
942    }
943    let logged: BTreeSet<u32> = protocol.changes.iter().map(|c| c.version).collect();
944    let missing: Vec<String> = (1..=protocol.version)
945        .filter(|v| !logged.contains(v))
946        .map(|v| v.to_string())
947        .collect();
948    if protocol.version == 0 {
949        refuse(
950            "version_unchanged",
951            "protocol version 0: versions start at 1, and each has a changes entry".into(),
952        );
953    } else if !missing.is_empty() {
954        refuse(
955            "version_unchanged",
956            format!(
957                "protocol version {} has no changes entry for version {}; every version says \
958                 why it changed",
959                protocol.version,
960                missing.join(", ")
961            ),
962        );
963    }
964    match &protocol.decision {
965        ProofDecision::Paired {
966            suite,
967            registration_digest,
968        } => match suites.get(suite).map(|s| s.map.get("registration")) {
969            None => refuse(
970                "paired_suite_unknown",
971                format!("its paired protocol names suite {suite}, which is not declared"),
972            ),
973            Some(None) => refuse(
974                "paired_registration_absent",
975                format!("suite {suite} has no registration block for a paired protocol to name"),
976            ),
977            Some(Some(block)) if digest(block) != *registration_digest => refuse(
978                "paired_registration_mismatch",
979                format!(
980                    "it names registration {registration_digest} of suite {suite}, whose \
981                     registration block digests to {}",
982                    digest(block)
983                ),
984            ),
985            Some(Some(_)) => {}
986        },
987        ProofDecision::Trials {
988            fields,
989            min_trials,
990            min_successes,
991            all_must_pass,
992        } => {
993            let mut problems = Vec::new();
994            if fields.is_empty() {
995                problems.push("it declares no record field".to_string());
996            }
997            let mut names = BTreeSet::new();
998            for f in fields {
999                if !names.insert(f.name.as_str()) {
1000                    problems.push(format!("field {} is declared twice", f.name));
1001                }
1002                if !f.class.is_measurement() {
1003                    problems.push(format!(
1004                        "field {} is {}, and a trial records only observed, counted or \
1005                         derived numbers",
1006                        f.name,
1007                        f.class.word()
1008                    ));
1009                }
1010            }
1011            if *min_trials == 0 || min_successes > min_trials {
1012                problems.push(format!(
1013                    "min_successes {min_successes} of min_trials {min_trials}: at least one \
1014                     trial, and no more successes than trials"
1015                ));
1016            } else if *all_must_pass && min_successes != min_trials {
1017                problems.push(format!(
1018                    "all_must_pass with min_successes {min_successes} of min_trials \
1019                     {min_trials}: when every trial must pass, every trial is a success"
1020                ));
1021            }
1022            if !problems.is_empty() {
1023                refuse("trials_invalid", problems.join("; "));
1024            }
1025        }
1026        ProofDecision::Window {
1027            period,
1028            admits,
1029            bounds,
1030        } => {
1031            let mut problems = Vec::new();
1032            match (
1033                crate::peers::parse_rfc3339(&period.from),
1034                crate::peers::parse_rfc3339(&period.until),
1035            ) {
1036                (Some(from), Some(until)) if from < until => {}
1037                _ => problems.push(format!(
1038                    "period {} to {} is not two YYYY-MM-DDTHH:MM:SSZ times in order",
1039                    period.from, period.until
1040                )),
1041            }
1042            if admits.trim().is_empty() {
1043                problems.push("it admits no work".to_string());
1044            }
1045            if bounds.is_empty() {
1046                problems.push("it bounds no field".to_string());
1047            }
1048            let mut names = BTreeSet::new();
1049            for b in bounds {
1050                if !names.insert(b.field.as_str()) {
1051                    problems.push(format!("field {} is bounded twice", b.field));
1052                }
1053                if b.max_not_observed_bp > 10_000 {
1054                    problems.push(format!(
1055                        "field {} allows {} basis points not observed, more than all of it",
1056                        b.field, b.max_not_observed_bp
1057                    ));
1058                }
1059            }
1060            if !problems.is_empty() {
1061                refuse("window_invalid", problems.join("; "));
1062            }
1063        }
1064    }
1065}
