1//! Who a remote run belongs to, and the record it leaves on its host (I3956).
2//!
3//! On 2026-10-22 a 33-minute `cargo test --doc` kept lundra at load 21, and the only way to
4//! name the session that started it was a process-tree trace by hand; when the owner killed
5//! its cargo and its shell, a `rustc` under them survived for 50 minutes with init as its
6//! parent and a deleted worktree as its working directory, because nothing stopped the group.
7//!
8//! So every offloaded run now carries its owner — the session (its id on the mesh), the board
9//! peer, the issue, the machine and process it was started from — in two places: in the
10//! environment of every process the run starts (`MAJORDOMUS_OWNER_*`, inherited by every
11//! `cargo` and `rustc` under it), and in a run record beside the host's queue
12//! (`<queue directory>.runs/<tag>.run`) that names the run's process group. The run is started
13//! in a process group of its own and stopped as one ([`super::stop`]), and the record is
14//! removed when the run ends; a record that outlives its run is the evidence a leftover is
15//! named by.
16//!
17//! The owner is derived from the slice-1 address model ([`PeerAddress`]: the provider's
18//! session, its pid, this machine) and the board row that address matches; nothing is asked
19//! of a hook or of the model. Every value is validated before it travels: a value that is not
20//! one short word of the characters an id is made of is dropped, never mangled into one, so
21//! no value can become a second line of the record or a word of the host's shell. The
22//! provider's messaging token is a credential and is never read.
23

32/// The variable naming the run itself (its tag) in every process of it: what tells a process
33/// group that is this run's from one whose id the host has since handed to someone else.
34pub const RUN_ENV: &str = "MAJORDOMUS_OWNER_RUN";
35
36/// The first line of a run record.
37pub const RUN_SCHEMA: &str = "majordomus-offload-run/v1";
38
39/// The longest value an owner field may have.
40const MAX_FIELD: usize = 128;
41
42/// One owner value, validated: trimmed, at most 128 characters, every one of them a letter,
43/// a digit or one of `-_.:@/`. Anything else is `None`, so a value carrying a newline, a
44/// space or a quote never reaches a record line or a host's shell.
45pub fn owner_value(v: &str) -> Option<String> {
46    let v = v.trim();
47    let ok = !v.is_empty()
48        && v.chars().count() <= MAX_FIELD
49        && v.chars()
50            .all(|c| c.is_ascii_alphanumeric() || "-_.:@/".contains(c));
51    ok.then(|| v.to_string())
52}

54/// Who a run belongs to: what a person needs to reach the worker that started it.
55#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, JsonSchema)]
56pub struct RunOwner {
57    /// The session's id on the mesh (`claude-code-<id>`), or the one this process inherited.
58    #[serde(skip_serializing_if = "Option::is_none")]
59    pub session: Option<String>,
60    /// The board peer whose address is this worker's, or the one this process inherited.
61    #[serde(skip_serializing_if = "Option::is_none")]
62    pub peer: Option<String>,
63    /// The issue the work is for: the active task's, or the one this process inherited.
64    #[serde(skip_serializing_if = "Option::is_none")]
65    pub issue: Option<String>,
66    /// The machine the run was started from.
67    pub host: String,
68    /// The worker's process on that machine.
69    #[serde(skip_serializing_if = "Option::is_none")]
70    pub pid: Option<u32>,
71}
72
73impl RunOwner {
74    /// The owner an environment describes. What this process inherited as an owner wins — a
75    /// run started from inside another run belongs to the session that started the first —
76    /// and otherwise the address the environment describes ([`PeerAddress::from_env`]: the
77    /// provider's session on the mesh, the worker's pid, `parent` when the provider names
78    /// none), `host`, the active task's issue and the board peer the address matched.
79    pub fn derive(
80        env: impl Fn(&str) -> Option<String>,
81        parent: Option<u32>,
82        host: &str,
83        task_issue: Option<String>,
84        board_peer: Option<String>,
85    ) -> RunOwner {
86        let read = |k: &str| env(k).and_then(|v| owner_value(&v));
87        let address = PeerAddress::from_env(&env, parent, host);
88        let inherited_pid = read("MAJORDOMUS_OWNER_PID").and_then(|p| p.parse::<u32>().ok());
89        RunOwner {
90            session: read("MAJORDOMUS_OWNER_SESSION").or_else(|| {
91                address
92                    .as_ref()
93                    .and_then(PeerAddress::mesh_session)
94                    .and_then(|s| owner_value(&s))
95            }),
96            peer: read("MAJORDOMUS_OWNER_PEER")
97                .or_else(|| board_peer.and_then(|p| owner_value(&p))),
98            issue: read("MAJORDOMUS_OWNER_ISSUE")
99                .or_else(|| task_issue.and_then(|i| owner_value(&i))),
100            host: read("MAJORDOMUS_OWNER_HOST")
101                .or_else(|| owner_value(host))
102                .unwrap_or_default(),
103            pid: inherited_pid.or_else(|| address.and_then(|a| a.pid)),
104        }
105    }
106
107    /// The owner of a run this process starts for the checkout at `worktree`: its
108    /// environment, its parent, this machine, the checkout's active task and the board row
109    /// of the checkout's server whose address is this worker's.
110    pub(crate) fn of_this_process(worktree: &Path) -> RunOwner {
111        let issue = crate::capability::builtin::continuity::read_task(
112            &worktree.join(".ai/local/state/current.yaml"),
113        )
114        .filter(|t| t.outcome == "active")
115        .map(|t| t.issue);
116        let env = |k: &str| std::env::var(k).ok();
117        let parent = Some(std::os::unix::process::parent_id());
118        let host = crate::integration::drain::host();
119        let peer = PeerAddress::from_env(env, parent, &host)
120            .and_then(|a| peer_on_board(&board_of_checkout(worktree), &a));
121        Self::derive(env, parent, &host, issue, peer)
122    }
123
124    /// The five script arguments, in the order the host exports them —
125    /// `MAJORDOMUS_OWNER_SESSION`, `_PEER`, `_ISSUE`, `_HOST`, `_PID`; an unknown value is
126    /// empty.
127    pub fn args(&self) -> [String; 5] {
128        [
129            self.session.clone().unwrap_or_default(),
130            self.peer.clone().unwrap_or_default(),
131            self.issue.clone().unwrap_or_default(),
132            self.host.clone(),
133            self.pid.map(|p| p.to_string()).unwrap_or_default(),
134        ]
135    }
136}

138/// The board peer whose address is `address`: the row carrying the same provider session,
139/// or — for a client that exposes none — the same process on the same machine. An attached
140/// row is preferred over one whose connection has gone. `None` when no row matches.
141pub fn peer_on_board(peers: &[Peer], address: &PeerAddress) -> Option<String> {
142    let same = |p: &Peer| {
143        p.address
144            .as_ref()
145            .is_some_and(|a| match (&a.session, &address.session) {
146                (Some(theirs), Some(ours)) => theirs == ours && a.provider == address.provider,
147                _ => a.host == address.host && a.pid.is_some() && a.pid == address.pid,
148            })
149    };
150    let matching: Vec<&Peer> = peers.iter().filter(|p| same(p)).collect();
151    matching
152        .iter()
153        .find(|p| p.attached)
154        .or_else(|| matching.first())
155        .map(|p| p.id.to_string())
156}

158/// This checkout's own board, as the server its lease names answers it; empty when there is
159/// no lease, no address, no answer or no board — the owner is then simply unnamed by a peer.
160/// One request, bounded by the declared probe timeout, through the route the registry
161/// declares for `peers.list` (never a path typed here).
162pub(crate) fn board_of_checkout(worktree: &Path) -> Vec<Peer> {
163    let half = crate::capability::builtin::server::local_half(worktree);
164    let lease = crate::lease::LeaseFile::read(&crate::lease::lease_file(worktree, &half));
165    let Some(url) = lease.document().and_then(|d| d.url.clone()) else {
166        return Vec::new();
167    };
168    // no route is no answer: the same empty board as a refused request
169    crate::command_graph::load::registry()
170        .and_then(|r| r.get("peers.list").and_then(|c| c.exposure.http.clone()))
171        .and_then(|h| {
172            crate::mcp::bridge::request(
173                &url,
174                "GET",
175                &format!("{}?checkouts=this", h.path),
176                &[],
177                None,
178                crate::lease::timings().probe_timeout,
179            )
180            .ok()
181        })
182        .filter(|r| r.status == 200)
183        .and_then(|r| serde_json::from_str::<serde_json::Value>(&r.body).ok())
184        .and_then(|v| serde_json::from_value(v["peers"].clone()).ok())
185        .unwrap_or_default()
186}

188/// A run record as the host keeps it while the run lives.
189#[derive(Debug, Clone, PartialEq, Eq, Serialize, JsonSchema)]
190pub struct RunRecord {
191    /// The offload tag: the first ten characters of the commit that travelled.
192    pub tag: String,
193    /// The run's shell on the host.
194    pub pid: u32,
195    /// The run's process group; absent where the host could not start one.
196    #[serde(skip_serializing_if = "Option::is_none")]
197    pub pgid: Option<u32>,
198    /// When it started, seconds since the epoch.
199    pub started_at: u64,
200    /// The host it runs on.
201    pub host: String,
202    /// The worktree it runs in.
203    pub cwd: String,
204    /// Who it belongs to.
205    pub owner: RunOwner,
206}
207
208impl RunRecord {
209    /// Read a record; an error says what is missing.
210    pub fn parse(text: &str) -> Result<RunRecord, String> {
211        let mut lines = text.lines();
212        if lines.next().map(str::trim) != Some(RUN_SCHEMA) {
213            return Err(format!(
214                "not a run record: the first line is not {RUN_SCHEMA}"
215            ));
216        }
217        let mut f: BTreeMap<&str, &str> = BTreeMap::new();
218        for line in lines {
219            let (k, v) = line.split_once(' ').unwrap_or((line, ""));
220            f.insert(k.trim(), v.trim());
221        }
222        let text_of = |k: &str| {
223            f.get(k)
224                .copied()
225                .filter(|v| !v.is_empty())
226                .map(str::to_string)
227        };
228        let num = |k: &str| text_of(k).and_then(|v| v.parse::<u32>().ok());
229        let need = |k: &str| text_of(k).ok_or_else(|| format!("the run record has no {k}"));
230        Ok(RunRecord {
231            tag: need("tag")?,
232            pid: num("pid").ok_or("the run record has no pid")?,
233            pgid: num("pgid"),
234            started_at: text_of("started_at")
235                .and_then(|v| v.parse().ok())
236                .unwrap_or(0),
237            host: need("host")?,
238            cwd: text_of("cwd").unwrap_or_default(),
239            owner: RunOwner {
240                session: text_of("owner_session"),
241                peer: text_of("owner_peer"),
242                issue: text_of("owner_issue"),
243                host: text_of("owner_host").unwrap_or_default(),
244                pid: num("owner_pid"),
245            },
246        })
247    }
248}

250/// Where a host keeps its run records: beside the queue's ticket directory, never inside it,
251/// because everything in that directory is read as a ticket.
252pub fn runs_directory(queue_directory: &str) -> String {
253    format!("{}.runs", queue_directory.trim_end_matches('/'))
254}

316    fn board_with(addresses: &[(PeerAddress, bool)]) -> Vec<Peer> {
317        let board = PeerBoard::new();
318        for (a, attached) in addresses {
319            let id = board.attach(Transport::Http);
320            board.locate(&id, a.clone());
321            if !attached {
322                // a peer that announced something is kept, detached, when it goes
323                board.announce(&id, "gone", Vec::new());
324                board.detach(&id);
325            }
326        }
327        board.list()
328    }

468    /// The board the checkout's server answers is read through the route the registry
469    /// declares for `peers.list`, and a row is named by its peer id; an answer that is not a
470    /// board is an empty one.
471    #[test]
472    fn a_checkout_whose_server_answers_has_its_board() {
473        use std::io::{Read, Write};
474        let serve = |body: &'static str, status: &'static str| {
475            let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
476            let port = listener.local_addr().unwrap().port();
477            let handle = std::thread::spawn(move || {
478                let (mut stream, _) = listener.accept().unwrap();
479                let mut buf = [0u8; 4096];
480                let n = stream.read(&mut buf).unwrap();
481                let request = String::from_utf8_lossy(&buf[..n]).into_owned();
482                let reply = format!(
483                    "HTTP/1.1 {status}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
484                    body.len()
485                );
486                stream.write_all(reply.as_bytes()).unwrap();
487                request
488            });
489            (port, handle)
490        };
491        let dir = tempfile::tempdir().unwrap();
492        let half = crate::capability::builtin::server::local_half(dir.path());
493        let file = crate::lease::lease_file(dir.path(), &half);
494        std::fs::create_dir_all(file.parent().unwrap()).unwrap();
495        let lease = |port: u16| {
496            std::fs::write(
497                &file,
498                format!(
499                    r#"{{"schema":"{}","token":"t","root":"/r","url":"http://127.0.0.1:{port}"}}"#,
500                    crate::lease::SCHEMA
501                ),
502            )
503            .unwrap()
504        };
505        let board = PeerBoard::new();
506        let id = board.attach(Transport::Http);
507        let env = |k: &str| (k == "CLAUDE_CODE_SESSION_ID").then(|| "s1".to_string());
508        let me = PeerAddress::from_env(env, Some(7), "mac").unwrap();
509        board.locate(&id, me.clone());
510        let body: &'static str = Box::leak(
511            serde_json::json!({ "peers": board.list() })
512                .to_string()
513                .into_boxed_str(),
514        );
515        let (port, handle) = serve(body, "200 OK");
516        lease(port);
517        let peers = board_of_checkout(dir.path());
518        let request = handle.join().unwrap();
519        assert!(request.contains("checkouts=this"), "{request}");
520        assert_eq!(peer_on_board(&peers, &me).as_deref(), Some("p1"));
521        let (port, handle) = serve("{}", "500 Internal Server Error");
522        lease(port);
523        assert!(
524            board_of_checkout(dir.path()).is_empty(),
525            "a refusal is no board"
526        );
527        handle.join().unwrap();
