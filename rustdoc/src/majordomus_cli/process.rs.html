1//! Starting a program whose file was written a moment ago.
2//!
3//! Linux refuses to execute a file that any process holds open for writing (ETXTBSY). A
4//! program this process has just written is held so by every child forked while the write
5//! was open, until that child execs: a test that writes a stand-in tool and runs it at once
6//! falls into that window whenever another test thread forks at the same moment (CI run
7//! 39049547899). Writing a copy and renaming it into place does not close the window, since
8//! the inherited descriptor holds the inode; waiting does, because the holder lets go as
9//! soon as it execs.
10
11use std::io;
12use std::time::Duration;
13
14/// How often a busy program is asked again, and how long apart: five seconds in all, far
15/// longer than a forked child takes to reach its exec.
16const ATTEMPTS: u32 = 50;
17const PAUSE: Duration = Duration::from_millis(100);
18
19/// Start a program with `start` — a `Command`'s `output`, `status` or `spawn` — and ask
20/// again while the program's file is still open for writing somewhere; the last attempt's
21/// answer is returned, so a program that stays busy fails as it would have.
22pub(crate) fn retrying_busy<T>(mut start: impl FnMut() -> io::Result<T>) -> io::Result<T> {
23    let mut attempts = 1;
24    loop {
25        match start() {
26            Err(e) if e.kind() == io::ErrorKind::ExecutableFileBusy && attempts < ATTEMPTS => {
27                attempts += 1;
28                std::thread::sleep(PAUSE);
29            }
30            other => return other,
31        }
32    }
33}
34
35#[cfg(test)]
36mod tests {
37    use super::*;
38    use std::process::Command;
39
40    /// A program that cannot be started for another reason is answered at once, with that
41    /// reason: only a busy file is waited for.
42    #[test]
43    fn a_program_that_cannot_start_is_not_asked_again() {
44        let mut attempts = 0;
45        let r = retrying_busy(|| {
46            attempts += 1;
47            Command::new("/nonexistent/majordomus-no-such-tool").output()
48        });
49        assert_eq!(r.unwrap_err().kind(), io::ErrorKind::NotFound);
50        assert_eq!(attempts, 1);
51    }
52
53    /// A start that is busy twice is asked a third time and its answer returned, and one that
54    /// stays busy is answered with the busy error after the bound — on every platform, since
55    /// the start is the caller's and the busy error is what the kernel would have said.
56    #[test]
57    fn a_busy_start_is_asked_again_until_it_answers_or_the_bound_is_spent() {
58        let busy = || io::Error::from(io::ErrorKind::ExecutableFileBusy);
59        let mut attempts = 0;
60        let r = retrying_busy(|| {
61            attempts += 1;
62            if attempts < 3 {
63                Err(busy())
64            } else {
65                Ok(attempts)
66            }
67        });
68        assert_eq!(r.unwrap(), 3);
69
70        let mut attempts = 0;
71        let r: io::Result<()> = retrying_busy(|| {
72            attempts += 1;
73            Err(busy())
74        });
75        assert_eq!(r.unwrap_err().kind(), io::ErrorKind::ExecutableFileBusy);
76        assert_eq!(
77            attempts, ATTEMPTS,
78            "a start that stays busy is asked exactly the bound"
79        );
80    }
81
82    /// Linux alone refuses to exec a file held open for writing, so this holds there: the
83    /// program is busy while the writer holds it, and runs once the writer lets go.
84    #[cfg(target_os = "linux")]
85    #[test]
86    fn a_program_still_open_for_writing_runs_once_its_writer_lets_go() {
87        use std::os::unix::fs::PermissionsExt;
88        let dir = tempfile::tempdir().unwrap();
89        let tool = dir.path().join("tool");
90        std::fs::write(&tool, "#!/bin/sh\necho ran\n").unwrap();
91        std::fs::set_permissions(&tool, std::fs::Permissions::from_mode(0o755)).unwrap();
92        let held = std::fs::OpenOptions::new().write(true).open(&tool).unwrap();
93        let busy = Command::new(&tool).output().unwrap_err();
94        assert_eq!(busy.kind(), io::ErrorKind::ExecutableFileBusy, "{busy}");
95        let release = std::thread::spawn(move || {
96            std::thread::sleep(Duration::from_millis(300));
97            drop(held);
98        });
99        let mut attempts = 0;
100        let out = retrying_busy(|| {
101            attempts += 1;
102            Command::new(&tool).output()
103        });
104        release.join().unwrap();
105        let out = out.expect("the tool runs once its writer lets go");
106        assert!(out.status.success(), "{out:?}");
107        assert_eq!(String::from_utf8_lossy(&out.stdout), "ran\n");
108        assert!(attempts > 1, "the busy tool was asked only once");
109    }
110}
