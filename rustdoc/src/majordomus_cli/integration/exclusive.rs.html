1//! One integration executor per repository and base branch, across machines (ADR 0101 §7).
2//!
3//! [`IntegrationLease`] is a `flock` in the common git directory: it serialises every
4//! worktree of one clone and nothing beyond it, so two clones — two machines — would each
5//! take their own lease and both merge. When this checkout's shared server runs the
6//! repository's mesh, the executor therefore also holds an exclusive mesh claim on
7//! [`scope`] — `integration/<repository>/<base>` — which every linked runtime admits
8//! against: a second machine's executor is refused, naming the holder, before it takes its
9//! lease or acts. Where no mesh runs the lease alone guards, and the lease record — what
10//! `prs brief` and the Cockpit read — says so by carrying no claim.
11//!
12//! The claim is taken before the clone's lease, so the lease record is written once, with
13//! the claim's key in it; a claim whose lease is then refused is released as it is dropped,
14//! and a held claim is released after the lease it belongs to. A claim an executor of this
15//! checkout left behind when it was killed carries this checkout's [`session`], which the
16//! mesh never refuses to its own session: the next executor here takes the lease and then
17//! releases the leftover, so a crash costs other machines a wait until this checkout runs
18//! its executor again, never a claim nobody can give back.
19//!
20//! The claim lives by the executor's progress through its queue, not by its process existing
21//! (I3991): it is beaten once as it is taken and again each time the executor advances its
22//! queue ([`MeshClaim::beat`]), so an executor that holds it and does nothing has it suspended
23//! by the mesh once `cooperation.session_ttl_seconds` passes, and another machine's executor
24//! is admitted. Before every write to the forge the executor asks whether it still holds it
25//! ([`MeshClaim::holds`]); one admitted over meanwhile is released by the mesh with
26//! [`crate::mesh::cooperation::OVERTAKEN`] at that beat, and the executor acts on nothing.
27

37const TIMEOUT: Duration = Duration::from_secs(10);

39/// What an executor claims: one base branch of one forge repository.
40pub fn scope(repository: &str, base: &str) -> String {
41    format!("integration/{repository}/{base}")
42}

44/// The mesh session every executor of one checkout and base claims as. The same each time,
45/// so that a claim a killed executor left behind is recognisably this checkout's own.
46pub fn session(root: &Path, base: &str) -> String {
47    use sha2::{Digest, Sha256};
48    let root = root.canonicalize().unwrap_or_else(|_| root.to_path_buf());
49    let digest = Sha256::digest(format!("{}\0{base}", root.display()));
50    format!("integration-{}", &format!("{digest:x}")[..12])
51}

53/// The server a claim is asked of: this checkout's own ([`ServerMesh`]), or a test's.
54pub trait Mesh: Send {
55    /// `method target` with a JSON body: the HTTP status and the JSON answer.
56    fn ask(&self, method: &str, target: &str, body: Option<Value>) -> Result<(u16, Value), String>;
57}

59/// This checkout's shared server, over HTTP.
60pub struct ServerMesh {
61    url: String,
62}
63
64impl ServerMesh {
65    /// The server at `url`.
66    pub fn at(url: impl Into<String>) -> Self {
67        ServerMesh { url: url.into() }
68    }
69}
70
71impl Mesh for ServerMesh {
72    fn ask(&self, method: &str, target: &str, body: Option<Value>) -> Result<(u16, Value), String> {
73        let payload = body.map(|b| b.to_string());
74        let reply = crate::mcp::bridge::request(
75            &self.url,
76            method,
77            target,
78            &[("Content-Type", "application/json")],
79            payload.as_deref(),
80            TIMEOUT,
81        )
82        .map_err(|e| format!("{}{target}: {e}", self.url))?;
83        let value = serde_json::from_str(&reply.body).map_err(|e| {
84            format!(
85                "{}{target}: status {}, not JSON: {e}",
86                self.url, reply.status
87            )
88        })?;
89        Ok((reply.status, value))
90    }
91}

93/// This checkout's shared server, when one is ready to answer for it.
94pub fn server_of(root: &Path) -> Option<ServerMesh> {
95    use crate::capability::builtin::server::{local_half, standing_at, ServerStanding};
96    let (standing, _, lease) = standing_at(root, &local_half(root));
97    let url = lease.document().and_then(|d| d.url.clone());
98    url.filter(|_| standing == ServerStanding::Ready)
99        .map(ServerMesh::at)
100}

102/// An exclusive mesh claim, released when dropped.
103pub struct MeshClaim {
104    mesh: Box<dyn Mesh>,
105    key: String,
106    session: String,
107    /// The mesh said the claim ended — released, or no longer known — so the drop has
108    /// nothing to give back.
109    ended: AtomicBool,
110    /// A refused beat was said once; the rest are not, so an executor of a mesh with the
111    /// progress rule off does not say it at every step.
112    beat_refusal_said: AtomicBool,
113}
114
115impl std::fmt::Debug for MeshClaim {
116    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
117        f.debug_struct("MeshClaim")
118            .field("key", &self.key)
119            .field("session", &self.session)
120            .finish()
121    }
122}
123
124impl MeshClaim {
125    fn of(mesh: Box<dyn Mesh>, key: String, session: &str) -> Self {
126        MeshClaim {
127            mesh,
128            key,
129            session: session.to_string(),
130            ended: AtomicBool::new(false),
131            beat_refusal_said: AtomicBool::new(false),
132        }
133    }
134
135    /// `<stream>/<claim>`, as the mesh answered it.
136    pub fn key(&self) -> &str {
137        &self.key
138    }
139
140    /// Release every other claim this checkout's session still holds: what an executor of
141    /// this checkout left behind when it was killed. Called only by the holder of the
142    /// clone's lease, so no live executor of this checkout holds any of them. Best effort:
143    /// what could not be released is said and keeps excluding until it is.
144    pub fn release_leftovers(&self) {
145        // the mesh keys a session by the stream that wrote it, as it keys the claim: this
146        // runtime's stream is the one this claim was written to
147        let own = match self.key.rsplit_once('/') {
148            Some((stream, _)) => format!("{stream}/{}", self.session),
149            None => self.session.clone(),
150        };
151        let held: Vec<String> = match live_claims(self.mesh.as_ref()) {
152            Ok(claims) => claims
153                .iter()
154                .filter(|c| c["state"]["state"] == json!("held"))
155                .filter(|c| c["session"] == json!(own))
156                .filter_map(|c| c["key"].as_str())
157                .filter(|k| *k != self.key)
158                .map(str::to_string)
159                .collect(),
160            Err(e) => {
161                eprintln!("majordomus: the mesh's claims could not be read: {e}");
162                return;
163            }
164        };
165        for key in held {
166            if let Err(e) = release(self.mesh.as_ref(), &key) {
167                eprintln!("majordomus: a leftover integration claim {key} stays held: {e}");
168            }
169        }
170    }
171
172    /// Beat this claim's session with `phase`: the executor advanced its queue (I3991). The
173    /// mesh writes at most one beat a quarter of its ttl, so this is cheap to ask. A mesh that
174    /// refuses it — the progress rule off, a server older than the beat — keeps the claim by
175    /// its stream alone, as before: said once, and never a reason to stop.
176    pub(crate) fn beat(&self, phase: &str) {
177        let body = json!({
178            "session": self.session,
179            "phase": phase,
180            "pid": std::process::id(),
181        });
182        let refused = match self.mesh.ask("POST", "/api/v1/mesh/beats", Some(body)) {
183            Ok((200, _)) => return,
184            Ok((status, v)) => format!("status {status}: {v}"),
185            Err(e) => e,
186        };
187        if !self.beat_refusal_said.swap(true, Ordering::SeqCst) {
188            eprintln!(
189                "majordomus: the integration claim {} was not beaten ({refused}); it lives by its \
190                 runtime's beat alone",
191                self.key
192            );
193        }
194    }
195
196    /// Whether this claim still holds its scope, asked before `act` reaches the forge. It
197    /// beats first, so that the mesh re-arms a claim suspended while nobody was admitted over
198    /// it, and releases one another session was admitted over meanwhile
199    /// ([`crate::mesh::cooperation::OVERTAKEN`]); then it reads the claim. Held passes;
200    /// suspended, conflicted, released, expired or unknown is refused, and so is a mesh that
201    /// cannot be read: this checkout's server not answering is its runtime not beating, and
202    /// its claims expire on every other machine.
203    pub(crate) fn holds(&self, act: &str) -> Result<(), String> {
204        self.beat(act);
205        let claims = claims_of(
206            self.mesh.as_ref(),
207            &format!("/api/v1/mesh/state?view=full&session={}", self.session),
208        )
209        .map_err(|e| {
210            format!(
211                "the integration claim {} could not be read on the mesh: {e}",
212                self.key
213            )
214        })?;
215        let Some(claim) = claims.iter().find(|c| c["key"] == json!(self.key)) else {
216            self.ended.store(true, Ordering::SeqCst);
217            return Err(format!(
218                "the mesh no longer knows the integration claim {}",
219                self.key
220            ));
221        };
222        let state = claim["state"]["state"].as_str().unwrap_or("unreadable");
223        if state == "held" {
224            return Ok(());
225        }
226        if state == "released" {
227            self.ended.store(true, Ordering::SeqCst);
228        }
229        let why = claim["state"]["detail"]
230            .as_str()
231            .map(|d| format!(" ({d})"))
232            .unwrap_or_default();
233        Err(format!(
234            "the integration claim {} is {state} on the mesh{why}: another executor may hold \
235             its scope now",
236            self.key
237        ))
238    }
239}

241/// The mesh's live claims, from the full view of its state.
242fn live_claims(mesh: &dyn Mesh) -> Result<Vec<Value>, String> {
243    claims_of(mesh, "/api/v1/mesh/state?view=full&claims=live")
244}
245
246/// The claims of the state `query` asks for. A server older than the full view refuses the
247/// query as invalid input and answers the whole state when asked bare, so the one refusal is
248/// asked again without it: every claim asked about is in the whole state too.
249fn claims_of(mesh: &dyn Mesh, query: &str) -> Result<Vec<Value>, String> {
250    let answer = match mesh.ask("GET", query, None)? {
251        (400, _) => mesh.ask("GET", "/api/v1/mesh/state", None)?,
252        other => other,
253    };
254    match answer {
255        (200, v) => Ok(v["state"]["claims"].as_array().cloned().unwrap_or_default()),
256        (status, v) => Err(format!("status {status}: {v}")),
257    }
258}
259
260fn release(mesh: &dyn Mesh, key: &str) -> Result<(), String> {
261    match mesh.ask(
262        "POST",
263        "/api/v1/mesh/claims/release",
264        Some(json!({ "claim": key })),
265    )? {
266        (200, _) => Ok(()),
267        (status, v) => Err(format!("status {status}: {v}")),
268    }
269}
270
271impl Drop for MeshClaim {
272    fn drop(&mut self) {
273        if self.ended.load(Ordering::SeqCst) {
274            return;
275        }
276        if let Err(e) = release(self.mesh.as_ref(), &self.key) {
277            // a destructor cannot refuse; it says so where a person reads
278            eprintln!(
279                "majordomus: the integration claim {} was not released and excludes other \
280                 machines until it is (majordomus mesh release {}): {e}",
281                self.key, self.key
282            );
283        }
284    }
285}

287/// How far an executor's exclusivity reaches.
288#[derive(Debug)]
289pub enum Reach {
290    /// Every machine of the repository's mesh: the claim it holds.
291    Repository(MeshClaim),
292    /// This clone alone: no mesh runs for this checkout.
293    CloneOnly,
294}
295
296/// Take the exclusive claim on `scope` as `session` from `mesh`.
297///
298/// A mesh that is not running here is no refusal — the lease still guards the clone, and
299/// [`Reach::CloneOnly`] says it guards no more. Another session's live claim is: the
300/// refusal names its key, its session and what it said it was for.
301pub fn claim(
302    mesh: Box<dyn Mesh>,
303    session: &str,
304    scope: &str,
305    intent: &str,
306) -> Result<Reach, String> {
307    let body = json!({
308        "session": session,
309        "client": "majordomus prs",
310        "scope": [scope],
311        "intent": intent,
312        "mode": "exclusive",
313    });
314    let (status, answer) = mesh.ask("POST", "/api/v1/mesh/claims", Some(body))?;
315    if status == 200 {
316        let key = answer["key"]
317            .as_str()
318            .ok_or_else(|| format!("the mesh answered a claim without a key: {answer}"))?
319            .to_string();
320        return Ok(Reach::Repository(MeshClaim::of(mesh, key, session)));
321    }
322    let message = answer["error"]["message"].as_str().unwrap_or_default();
323    if message.starts_with("not_active") {
324        return Ok(Reach::CloneOnly);
325    }
326    if message.starts_with("claim_conflict") {
327        return Err(format!(
328            "another integration executor holds {scope} on the mesh{}: {message}",
329            holder(mesh.as_ref(), scope)
330        ));
331    }
332    Err(format!(
333        "the mesh refused the integration claim on {scope} (status {status}): {answer}"
334    ))
335}

337/// ", held by session S (intent)" for the live claims on `scope`, read from the mesh's
338/// state; empty when the state cannot be read, the refusal itself still naming the key.
339fn holder(mesh: &dyn Mesh, scope: &str) -> String {
340    let Ok(claims) = live_claims(mesh) else {
341        return String::new();
342    };
343    let named: Vec<String> = claims
344        .iter()
345        .filter(|c| c["state"]["state"] == json!("held"))
346        .filter(|c| {
347            c["scope"]
348                .as_array()
349                .is_some_and(|s| s.contains(&json!(scope)))
350        })
351        .map(|c| {
352            format!(
353                "session {} ({})",
354                c["session"].as_str().unwrap_or("?"),
355                c["intent"].as_str().unwrap_or("no intent given")
356            )
357        })
358        .collect();
359    if named.is_empty() {
360        String::new()
361    } else {
362        format!(", held by {}", named.join("; "))
363    }
364}

366/// Take the lease of `base` for this checkout's executor, with the repository-wide claim
367/// when this checkout's server runs the mesh. The one way an executor (`prs drain`,
368/// `cleanup --apply`) takes its lease.
369pub fn acquire(root: &Path, base: &str) -> Result<IntegrationLease, String> {
370    acquire_through(
371        root,
372        base,
373        server_of(root).map(|m| Box::new(m) as Box<dyn Mesh>),
374    )
375}
376
377/// [`acquire`], asking `mesh` (or none) rather than finding this checkout's server.
378pub fn acquire_through(
379    root: &Path,
380    base: &str,
381    mesh: Option<Box<dyn Mesh>>,
382) -> Result<IntegrationLease, String> {
383    let reach = match mesh {
384        None => Reach::CloneOnly,
385        Some(mesh) => {
386            let repository = super::load_observation(root)
387                .ok()
388                .flatten()
389                .map(|o| o.repository)
390                .unwrap_or_default();
391            let intent = format!(
392                "the integration executor of {} (pid {}) on {}",
393                root.display(),
394                std::process::id(),
395                super::drain::host()
396            );
397            claim(
398                mesh,
399                &session(root, base),
400                &scope(&repository, base),
401                &intent,
402            )?
403        }
404    };
405    match reach {
406        Reach::Repository(claim) => {
407            let lease = IntegrationLease::acquire_holding(root, base, Some(claim))?;
408            // the lease holds the claim it was given, so this releases exactly once
409            lease.mesh_claim().into_iter().for_each(|claim| {
410                claim.release_leftovers();
411                // from its first beat the claim lives by the executor's progress (I3991)
412                claim.beat("took the integration lease");
413            });
414            Ok(lease)
415        }
416        Reach::CloneOnly => IntegrationLease::acquire(root, base),
417    }
418}

428    /// A mesh that answers from a script and remembers what it was asked.
429    #[derive(Clone)]
430    struct Scripted {
431        answers: Arc<Mutex<Vec<Answer>>>,
432        asked: Arc<Mutex<Vec<Asked>>>,
433    }
434
435    impl Scripted {
436        fn new(answers: Vec<Result<(u16, Value), String>>) -> Self {
437            Scripted {
438                answers: Arc::new(Mutex::new(answers)),
439                asked: Arc::new(Mutex::new(Vec::new())),
440            }
441        }
442        fn asked(&self) -> Vec<(String, String, Option<Value>)> {
443            self.asked.lock().unwrap().clone()
444        }
445    }
446
447    impl Mesh for Scripted {
448        fn ask(
449            &self,
450            method: &str,
451            target: &str,
452            body: Option<Value>,
453        ) -> Result<(u16, Value), String> {
454            self.asked
455                .lock()
456                .unwrap()
457                .push((method.into(), target.into(), body));
458            let mut answers = self.answers.lock().unwrap();
459            if answers.is_empty() {
460                return Ok((200, json!({})));
461            }
462            answers.remove(0)
463        }
464    }

467        Ok((

470        ))
471    }

473    fn state(claims: Value) -> Result<(u16, Value), String> {
474        Ok((200, json!({ "state": { "claims": claims } })))
475    }

847    fn a_claim_read_suspended_or_released_is_not_held() {
848        let ok = || Ok((200, json!({})));
849        let claim = |standing: Value| {
850            state(json!([
851                { "key": "s1/c-1", "session": "s1/integration-me", "state": standing },
852                { "key": "s2/c-9", "session": "s2/integration-me", "state": { "state": "held" } },
853            ]))
854        };
855        let held_by = |answers: Vec<Answer>| {
856            let mesh = Scripted::new(answers);
857            let held = MeshClaim::of(Box::new(mesh.clone()), "s1/c-1".into(), "integration-me");
858            let verdict = held.holds("merging #4");
859            (mesh, held, verdict)
860        };
861
862        // held: the beat first, with the act, then the claim read by its session
863        let (mesh, held, verdict) = held_by(vec![ok(), claim(json!({ "state": "held" }))]);
864        verdict.expect("held");
865        let asked = mesh.asked();
866        assert_eq!(asked[0].1, "/api/v1/mesh/beats");
867        assert_eq!(asked[0].2.clone().unwrap()["phase"], "merging #4");
868        assert_eq!(
869            asked[1].1,
870            "/api/v1/mesh/state?view=full&session=integration-me"
871        );
872        drop(held);
873        assert_eq!(mesh.asked().pop().unwrap().1, "/api/v1/mesh/claims/release");
874
875        // suspended: refused, naming why, and still given back when dropped
876        let (mesh, held, verdict) = held_by(vec![
877            ok(),
878            claim(json!({ "state": "suspended",
879                          "detail": "its holder stopped making progress: no beat for over 8s" })),
880        ]);
881        let e = verdict.unwrap_err();
882        assert!(
883            e.contains("suspended") && e.contains("no beat for over 8s"),
884            "{e}"
885        );
886        drop(held);
887        assert_eq!(mesh.asked().pop().unwrap().1, "/api/v1/mesh/claims/release");
888
889        // released by the mesh at that beat (overtaken): refused, and not released again
890        let (mesh, held, verdict) = held_by(vec![
891            Ok((
892                400,
893                json!({ "error": { "code": "invalid_input", "message": "no beat was written" } }),
894            )),
895            claim(json!({ "state": "released" })),
896        ]);
897        assert!(verdict.unwrap_err().contains("is released on the mesh"));
898        drop(held);
899        assert_eq!(
900            mesh.asked().len(),
901            2,
902            "a released claim is not released twice"
903        );
904
905        // unknown to the mesh: refused, and nothing to give back
906        let (mesh, held, verdict) = held_by(vec![ok(), state(json!([]))]);
907        assert!(verdict.unwrap_err().contains("no longer knows"));
908        drop(held);
909        assert_eq!(mesh.asked().len(), 2);
910
911        // a mesh that cannot be read is refused: its claims expire everywhere else
912        let (_, held, verdict) = held_by(vec![ok(), Err("connection refused".into())]);
913        assert!(verdict.unwrap_err().contains("could not be read"));
914        std::mem::forget(held);
915
916        // an older server answers the whole state bare, where the claim is found as well
917        let (mesh, held, verdict) = held_by(vec![
918            Ok((404, json!({ "error": "no such route" }))),
919            Ok((400, json!({ "error": { "code": "invalid_input" } }))),
920            claim(json!({ "state": "held" })),
921        ]);
922        verdict.expect("held, read from the whole state");
923        assert_eq!(mesh.asked()[2].1, "/api/v1/mesh/state");
924        std::mem::forget(held);
