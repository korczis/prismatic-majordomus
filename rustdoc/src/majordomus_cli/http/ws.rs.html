1//! WebSocket, by hand: the handshake and the frames of RFC 6455, and nothing else.
2//!
3//! There is no dependency here for the same reason there is none for base64 or for
4//! percent-encoding: what this server needs of the protocol is a handshake, text frames it
5//! writes, a ping it writes and a close it writes, and that is less code than the seam an
6//! async runtime would introduce into a synchronous server. `Socket.IO` would be a second
7//! protocol on top of this one, with its own framing, its own reconnection semantics and
8//! its own client to ship, in exchange for nothing this control plane needs.
9//!
10//! What is deliberately absent is reading. The server this crate runs is
11//! [`tiny_http`], whose upgrade hands over one `Read + Write` value that cannot be split;
12//! a reader thread blocking on it would have to hold the same lock the writer needs. So
13//! the stream this module writes is one-way by design: everything a client wants to say —
14//! which executions to follow, and from which sequence — it says in the request target,
15//! and the server ignores whatever the client sends afterwards. A client that goes away is
16//! noticed by the next write failing, which the heartbeat guarantees will happen.
17
18use std::io::Write;
19
20/// The constant RFC 6455 appends to the client's key before hashing it.
21const GUID: &str = "258EAFA5-E914-47DA-95CA-C5AB0DC85B11";
22
23/// The largest frame this server writes in one piece.
24pub const MAX_FRAME_BYTES: usize = 256 * 1024;
25
26/// Is this request asking to become a WebSocket?
27///
28/// ```
29/// use majordomus_cli::http::{ws, Request};
30/// let plain = Request::parse_target("GET", "/events", vec![]);
31/// assert!(!ws::is_upgrade(&plain));
32/// let asking = plain.clone().with_headers(vec![
33///     ("Upgrade".into(), "websocket".into()),
34///     ("Connection".into(), "Upgrade".into()),
35/// ]);
36/// assert!(ws::is_upgrade(&asking));
37/// ```
38pub fn is_upgrade(req: &super::Request) -> bool {
39    req.header("upgrade")
40        .is_some_and(|v| v.eq_ignore_ascii_case("websocket"))
41        && req
42            .header("connection")
43            .is_some_and(|v| v.to_ascii_lowercase().contains("upgrade"))
44}

46/// What a handshake needs from the request, or why it cannot be done.
47#[derive(Debug, Clone, PartialEq, Eq)]
48pub enum Handshake {
49    /// The value of `Sec-WebSocket-Accept` to answer with.
50    Accept(String),
51    /// The request cannot become a WebSocket, and this is what to tell the client.
52    Refuse {
53        /// The status to answer with.
54        status: u16,
55        /// Why, for a person reading a network panel.
56        reason: String,
57    },
58}
59
60/// Read the handshake headers and compute the answer.
61///
62/// ```
63/// use majordomus_cli::http::{ws::{handshake, Handshake}, Request};
64/// // the example key of RFC 6455 §1.3 and the accept value it specifies
65/// let req = Request::parse_target("GET", "/events", vec![]).with_headers(vec![
66///     ("Upgrade".into(), "websocket".into()),
67///     ("Connection".into(), "Upgrade".into()),
68///     ("Sec-WebSocket-Key".into(), "dGhlIHNhbXBsZSBub25jZQ==".into()),
69///     ("Sec-WebSocket-Version".into(), "13".into()),
70/// ]);
71/// assert_eq!(
72///     handshake(&req),
73///     Handshake::Accept("s3pPLMBiTxaQ9kYGzzhZRbK+xOo=".into())
74/// );
75/// ```
76pub fn handshake(req: &super::Request) -> Handshake {
77    if req.method != "GET" {
78        return Handshake::Refuse {
79            status: 405,
80            reason: "a WebSocket is opened with GET".into(),
81        };
82    }
83    match req.header("sec-websocket-version") {
84        Some("13") => {}
85        Some(other) => {
86            return Handshake::Refuse {
87                status: 400,
88                reason: format!("Sec-WebSocket-Version {other} is not 13"),
89            }
90        }
91        None => {
92            return Handshake::Refuse {
93                status: 400,
94                reason: "no Sec-WebSocket-Version header".into(),
95            }
96        }
97    }
98    let Some(key) = req.header("sec-websocket-key") else {
99        return Handshake::Refuse {
100            status: 400,
101            reason: "no Sec-WebSocket-Key header".into(),
102        };
103    };
104    Handshake::Accept(accept_for(key))
105}

107/// The `Sec-WebSocket-Accept` value for a client key.
108pub fn accept_for(key: &str) -> String {
109    crate::cockpit::base64(&sha1(format!("{key}{GUID}").as_bytes()))
110}

112/// The opcodes this server writes.
113#[derive(Debug, Clone, Copy, PartialEq, Eq)]
114pub enum Opcode {
115    /// A UTF-8 payload.
116    Text,
117    /// A heartbeat the client answers automatically.
118    Ping,
119    /// The end of the conversation.
120    Close,
121}
122
123impl Opcode {
124    fn bits(self) -> u8 {
125        match self {
126            Opcode::Text => 0x1,
127            Opcode::Ping => 0x9,
128            Opcode::Close => 0x8,
129        }
130    }
131}
132
133/// One unfragmented frame, from the server, unmasked as RFC 6455 requires of a server.
134///
135/// ```
136/// use majordomus_cli::http::ws::{frame, Opcode};
137/// assert_eq!(frame(Opcode::Text, b"hi"), vec![0x81, 0x02, b'h', b'i']);
138/// assert_eq!(&frame(Opcode::Ping, b"")[..2], &[0x89, 0x00]);
139/// // a payload of 126 bytes or more carries an extended length
140/// assert_eq!(&frame(Opcode::Text, &[b'x'; 200])[..4], &[0x81, 126, 0, 200]);
141/// ```
142pub fn frame(opcode: Opcode, payload: &[u8]) -> Vec<u8> {
143    let mut out = Vec::with_capacity(payload.len() + 10);
144    out.push(0x80 | opcode.bits());
145    match payload.len() {
146        n if n < 126 => out.push(n as u8),
147        n if n <= u16::MAX as usize => {
148            out.push(126);
149            out.extend_from_slice(&(n as u16).to_be_bytes());
150        }
151        n => {
152            out.push(127);
153            out.extend_from_slice(&(n as u64).to_be_bytes());
154        }
155    }
156    out.extend_from_slice(payload);
157    out
158}

160/// Write one text frame, refusing a payload over [`MAX_FRAME_BYTES`].
161///
162/// The refusal is not a truncation: a client that received half a JSON document would
163/// have to guess, and a bounded producer is why this cannot happen — every event this
164/// server writes is already bounded by the store's own limits.
165pub fn write_text(stream: &mut dyn Write, text: &str) -> std::io::Result<()> {
166    if text.len() > MAX_FRAME_BYTES {
167        return Err(std::io::Error::other(format!(
168            "a frame of {} bytes is over the {MAX_FRAME_BYTES}-byte bound",
169            text.len()
170        )));
171    }
172    stream.write_all(&frame(Opcode::Text, text.as_bytes()))?;
173    stream.flush()
174}
175
176/// Write a ping with no payload.
177pub fn write_ping(stream: &mut dyn Write) -> std::io::Result<()> {
178    stream.write_all(&frame(Opcode::Ping, b""))?;
179    stream.flush()
180}

182/// Write a close frame carrying a status code and a reason, then flush.
183///
184/// 1000 is a normal close; 1001 is "going away", which is what a server that is shutting
185/// down means.
186pub fn write_close(stream: &mut dyn Write, code: u16, reason: &str) -> std::io::Result<()> {
187    let mut payload = code.to_be_bytes().to_vec();
188    payload.extend_from_slice(&reason.as_bytes()[..reason.len().min(120)]);
189    stream.write_all(&frame(Opcode::Close, &payload))?;
190    stream.flush()
191}

193/// SHA-1 of a byte string, for the one thing RFC 6455 specifies it for.
194///
195/// The handshake is the only place this algorithm appears in this crate, and it is not
196/// used for anything that needs to be hard to forge — the accept value proves the peer
197/// read the request, nothing more. Everything this executable hashes for real (the
198/// registry fingerprint, the index, the artifacts) is SHA-256.
199///
200/// ```
201/// use majordomus_cli::http::ws::sha1;
202/// let digest: String = sha1(b"abc").iter().map(|b| format!("{b:02x}")).collect();
203/// assert_eq!(digest, "a9993e364706816aba3e25717850c26c9cd0d89d");
204/// let empty: String = sha1(b"").iter().map(|b| format!("{b:02x}")).collect();
205/// assert_eq!(empty, "da39a3ee5e6b4b0d3255bfef95601890afd80709");
206/// ```
207pub fn sha1(message: &[u8]) -> [u8; 20] {
208    let mut h: [u32; 5] = [
209        0x6745_2301,
210        0xEFCD_AB89,
211        0x98BA_DCFE,
212        0x1032_5476,
213        0xC3D2_E1F0,
214    ];
215    let mut padded = message.to_vec();
216    padded.push(0x80);
217    while padded.len() % 64 != 56 {
218        padded.push(0);
219    }
220    padded.extend_from_slice(&((message.len() as u64) * 8).to_be_bytes());
221
222    for block in padded.chunks_exact(64) {
223        let mut w = [0u32; 80];
224        for (i, word) in block.chunks_exact(4).enumerate() {
225            w[i] = u32::from_be_bytes([word[0], word[1], word[2], word[3]]);
226        }
227        for i in 16..80 {
228            w[i] = (w[i - 3] ^ w[i - 8] ^ w[i - 14] ^ w[i - 16]).rotate_left(1);
229        }
230        let [mut a, mut b, mut c, mut d, mut e] = h;
231        for (i, word) in w.iter().enumerate() {
232            let (f, k) = match i {
233                0..=19 => ((b & c) | ((!b) & d), 0x5A82_7999),
234                20..=39 => (b ^ c ^ d, 0x6ED9_EBA1),
235                40..=59 => ((b & c) | (b & d) | (c & d), 0x8F1B_BCDC),
236                _ => (b ^ c ^ d, 0xCA62_C1D6),
237            };
238            let temp = a
239                .rotate_left(5)
240                .wrapping_add(f)
241                .wrapping_add(e)
242                .wrapping_add(k)
243                .wrapping_add(*word);
244            e = d;
245            d = c;
246            c = b.rotate_left(30);
247            b = a;
248            a = temp;
249        }
250        h[0] = h[0].wrapping_add(a);
251        h[1] = h[1].wrapping_add(b);
252        h[2] = h[2].wrapping_add(c);
253        h[3] = h[3].wrapping_add(d);
254        h[4] = h[4].wrapping_add(e);
255    }
256    let mut out = [0u8; 20];
257    for (i, word) in h.iter().enumerate() {
258        out[i * 4..i * 4 + 4].copy_from_slice(&word.to_be_bytes());
259    }
260    out
261}

272    #[test]
273    fn sha1_matches_the_published_vectors() {
274        assert_eq!(hex(&sha1(b"")), "da39a3ee5e6b4b0d3255bfef95601890afd80709");
275        assert_eq!(
276            hex(&sha1(b"abc")),
277            "a9993e364706816aba3e25717850c26c9cd0d89d"
278        );
279        assert_eq!(
280            hex(&sha1(
281                b"abcdbcdecdefdefgefghfghighijhijkijkljklmklmnlmnomnopnopq"
282            )),
283            "84983e441c3bd26ebaae4aa1f95129e5e54670f1",
284            "a message that crosses a block boundary"
285        );
286        assert_eq!(
287            hex(&sha1(&[b'a'; 1000])),
288            "291e9a6c66994949b57ba5e650361e98fc36b1ba"
289        );
290        // exactly one block of padding overflow: 56 bytes needs a second block
291        assert_eq!(sha1(&[b'x'; 56]).len(), 20);
292        assert_eq!(sha1(&[b'x'; 64]).len(), 20);
293    }

303    #[test]
304    fn a_handshake_is_refused_with_a_reason_a_person_can_read() {
305        let base = Request::parse_target("GET", "/events", vec![]);
306        let refuse =
307            |headers: Vec<(String, String)>| match handshake(&base.clone().with_headers(headers)) {
308                Handshake::Refuse { status, reason } => (status, reason),
309                Handshake::Accept(_) => panic!("accepted"),
310            };
311        assert_eq!(refuse(vec![]).0, 400);
312        assert!(refuse(vec![]).1.contains("Version"));
313        assert!(refuse(vec![("Sec-WebSocket-Version".into(), "8".into())])
314            .1
315            .contains("not 13"));
316        assert!(refuse(vec![("Sec-WebSocket-Version".into(), "13".into())])
317            .1
318            .contains("Key"));
319        let post = Request::parse_target("POST", "/events", vec![]);
320        assert_eq!(
321            match handshake(&post) {
322                Handshake::Refuse { status, .. } => status,
323                Handshake::Accept(_) => 0,
324            },
325            405

344    #[test]
345    fn frames_carry_the_lengths_the_protocol_specifies() {
346        assert_eq!(frame(Opcode::Text, b""), vec![0x81, 0x00]);
347        assert_eq!(frame(Opcode::Close, b""), vec![0x88, 0x00]);
348        let medium = frame(Opcode::Text, &[b'x'; 65_535]);
349        assert_eq!(&medium[..4], &[0x81, 126, 0xff, 0xff]);
350        assert_eq!(medium.len(), 65_535 + 4);
351        let large = frame(Opcode::Text, &[b'x'; 70_000]);
352        assert_eq!(large[1], 127);
353        assert_eq!(
354            u64::from_be_bytes(large[2..10].try_into().unwrap()),
355            70_000,
356            "the 64-bit length"
357        );
358        // no mask bit is ever set: a server frame is never masked
359        for f in [
360            frame(Opcode::Text, b"a"),
361            frame(Opcode::Ping, b""),
362            frame(Opcode::Close, b"x"),
363        ] {
364            assert_eq!(f[1] & 0x80, 0, "a server frame carries no mask");
365            assert_eq!(f[0] & 0x80, 0x80, "and is never fragmented");
366        }
367    }

369    #[test]
370    fn writing_refuses_an_oversized_frame_rather_than_cutting_it() {
371        let mut sink: Vec<u8> = Vec::new();
372        let big = "x".repeat(MAX_FRAME_BYTES + 1);
373        assert!(write_text(&mut sink, &big).is_err());
374        assert!(sink.is_empty(), "nothing half-written reaches the wire");
375        assert!(write_text(&mut sink, "{\"a\":1}").is_ok());
376        assert_eq!(&sink[2..], b"{\"a\":1}");
377        sink.clear();
378        assert!(write_ping(&mut sink).is_ok());
379        assert_eq!(sink, vec![0x89, 0x00]);
380        sink.clear();
381        assert!(write_close(&mut sink, 1001, "going away").is_ok());
382        assert_eq!(sink[0], 0x88);
383        assert_eq!(u16::from_be_bytes([sink[2], sink[3]]), 1001);
384        assert_eq!(&sink[4..], b"going away");
385    }
