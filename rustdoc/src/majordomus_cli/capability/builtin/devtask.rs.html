1//! The `devtask` module: an issue and a milestone read as executable development scopes.
2//!
3//! # Why this is a module and not two more `plan.*` capabilities
4//!
5//! Every capability is composed into exactly one Rust module, and a module's id is the
6//! namespace of everything in it, so `plan.task` would have to live inside
7//! `builtin/plan.rs`. It does not belong there: `plan` is the projection of
8//! [`crate::plan`], which is held to byte equality with `lib/project.awk` by
9//! `test/cases/99_plan_capabilities.sh`, and a capability that joins the plan to git and to
10//! the session records has no counterpart in the awk to be equal to. Keeping them apart
11//! keeps that equality checkable.
12//!
13//! What it must not be is a second project model, and it is not:
14//!
15//! * the records are the canonical ones, read through the index like every other object;
16//! * the status, wave, dependents, edges and findings are [`crate::plan`]'s, unchanged —
17//!   `the_status_never_disagrees_with_the_plan` is the assertion;
18//! * the branches and commits are `worktree::trace`'s, through the same [`Tracer`]
19//!   `trace.issue` uses;
20//! * the readiness vocabulary is a projection of the canonical statuses and refines three
21//!   of them without replacing any — [`crate::devtask::readiness`] states which and why;
22//! * the external projection stays `scripts/github-sync`'s. Nothing here writes, and
23//!   nothing here restates its reconciliation vocabulary.
24//!
25//! [`Tracer`]: crate::worktree::trace::Tracer
26//!
27//! # Why the milestone graph does not touch git
28//!
29//! `devtask.milestone` is a pure function of the index, and deliberately: a graph whose
30//! ordering depended on `git for-each-ref` would be a graph that differs between a full
31//! clone and a shallow one, and this repository has already been bitten by a derivation
32//! that was machine-dependent. `devtask.issue` consults git because a single issue's
33//! branches are the question being asked; the graph does not, because they are not.

59/// Which issue to read as a development task.
60pub struct DevTaskInput {
61    /// The issue id, as the canonical model spells it (`I0901`). An id the model does not
62    /// declare is answered with `declared: false` and every canonical field `unknown`,
63    /// never refused and never answered with empty strings that read as authored.
64    pub issue: String,
65    #[serde(default, skip_serializing_if = "Option::is_none")]
66    /// Consult git for the branches and commits that realised it. Default: yes. Set it to
67    /// false for a deterministic answer that depends on the records alone — which is what a
68    /// generated document or a comparison across machines wants.
69    pub git: Option<bool>,
70}

118/// The first object of a kind, in URI order. A benchmark case's input, and nothing else: a
119/// capability never picks an object for a caller.
120fn first(index: &Index, kind: &str) -> Option<String> {
121    index
122        .objects
123        .iter()
124        .find(|o| o.kind == kind)
125        .map(|o| o.identity.clone())
126}
127
128/// One canonical record of a kind, by identity: its repository-relative path and its parsed
129/// metadata. The index has already discovered, parsed and schema-validated it, so this is
130/// the model itself rather than a second reading of the directory it lives in.
131fn record<'a>(index: &'a Index, kind: &str, identity: &str) -> Option<(&'a str, &'a Value)> {
132    index
133        .objects
134        .iter()
135        .find(|o| o.kind == kind && o.identity == identity)
136        .map(|o| (o.provenance.path.as_str(), &o.metadata))
137}
138
139/// Every issue id the canonical model declares, in id order. What the tracer needs in order
140/// to read an issue out of a branch name.
141fn issue_ids(index: &Index) -> Vec<String> {
142    let mut ids: Vec<String> = index
143        .objects
144        .iter()
145        .filter(|o| o.kind == ISSUE)
146        .map(|o| o.identity.clone())
147        .collect();
148    crate::order::canonical(&mut ids);
149    ids
150}
151
152/// The session records that reach this issue, each carrying how strongly.
153///
154/// Two relations, and they are deliberately not merged. A record's own `issues` key is what
155/// `share/schemas/majordomus/session-record` calls "the issues it moved, by id — derived
156/// from the ledger's own events for this episode; never authored": a canonical link. Its
157/// branch name is a rule about a string — [`crate::worktree::state::issue_of`]'s, not
158/// restated here — which a rename can break. The first is derived, the second inferred, and
159/// [`SessionRef::declares_issue`] is which, so that the field carrying each can state its
160/// own provenance instead of one of them borrowing the other's.
161fn sessions_of(index: &Index, issue: &str) -> Vec<SessionRef> {
162    let ids = vec![issue.to_string()];
163    let mut out: Vec<SessionRef> = index
164        .objects
165        .iter()
166        .filter(|o| o.kind == "session")
167        .filter_map(|o| {
168            let branch = o
169                .metadata
170                .get("branch")
171                .and_then(Value::as_str)
172                .unwrap_or_default();
173            let declares_issue = o
174                .metadata
175                .get("issues")
176                .and_then(Value::as_array)
177                .is_some_and(|items| items.iter().any(|i| i.as_str() == Some(issue)));
178            // a record that neither declares the issue nor carries a branch naming it is
179            // not this issue's session, and is left out rather than reported weakly
180            if !declares_issue && crate::worktree::state::issue_of(branch, &ids).is_none() {
181                return None;
182            }
183            Some(SessionRef {
184                session_id: o.identity.clone(),
185                branch: branch.to_string(),
186                created_at: o
187                    .metadata
188                    .get("created_at")
189                    .and_then(Value::as_str)
190                    .unwrap_or_default()
191                    .to_string(),
192                path: o.provenance.path.clone(),
193                declares_issue,
194            })
195        })
196        .collect();
197    out.sort_by(|a, b| {
198        crate::order::natural_cmp(&a.created_at, &b.created_at)
199            .then_with(|| crate::order::natural_cmp(&a.session_id, &b.session_id))
200    });
201    out
202}

204/// The repository the external projection targets, as `project.yaml` declares it.
205fn repository_of(plan: &Plan) -> Option<&str> {
206    Some(plan.project.repository.as_str()).filter(|r| !r.is_empty())
207}
208
209fn refused(e: WorktreeError) -> CapabilityError {
210    match e.exit_code() {
211        crate::worktree::EXIT_MISSING => CapabilityError::NotFound(e.to_string()),
212        crate::worktree::EXIT_REFUSED => CapabilityError::Refused(e.to_string()),
213        _ => CapabilityError::Internal(e.to_string()),
214    }
215}
216
217/// What git says about one issue, or nothing when git could not be opened at all.
218///
219/// A repository this process cannot read as a git repository is not an error here: the
220/// canonical half of the answer is still the whole canonical half, and the execution group
221/// reports `git_consulted: false` rather than an empty list that would read as "nothing
222/// realised this issue".
223fn trace_of(ctx: &Context, issue: &str) -> Result<Option<IssueTrace>, CapabilityError> {
224    let tracer = match Tracer::open(Path::new(&ctx.index.repository.root)) {
225        Ok(t) => t.with_issues(issue_ids(&ctx.index)),
226        Err(e) if e.exit_code() == crate::worktree::EXIT_MISSING => return Ok(None),
227        Err(e) => return Err(refused(e)),
228    };
229    tracer.trace(issue).map(Some).map_err(refused)
230}

234fn devtask_issue(ctx: &Context, input: DevTaskInput) -> Result<DevTask, CapabilityError> {
235    let plan = Plan::build(&ctx.index);
236    let trace = if input.git.unwrap_or(true) {
237        trace_of(ctx, &input.issue)?
238    } else {
239        None
240    };
241    let record = record(&ctx.index, ISSUE, &input.issue);
242    let model = DevTaskModel {
243        plan: &plan,
244        record: record.map(|(path, metadata)| RecordRef { path, metadata }),
245        trace: trace.as_ref(),
246        sessions: sessions_of(&ctx.index, &input.issue),
247        repository: repository_of(&plan),
248    };
249    Ok(DevTask::build(&model, &input.issue))
250}
251
252fn devtask_milestone(
253    ctx: &Context,
254    input: DevMilestoneInput,
255) -> Result<MilestoneGraph, CapabilityError> {
256    let plan = Plan::build(&ctx.index);
257    let record = record(&ctx.index, MILESTONE, &input.milestone)
258        .map(|(path, metadata)| RecordRef { path, metadata });
259    Ok(MilestoneGraph::build(&plan, &input.milestone, record))
260}

314    /// The declaration is the only place these names exist; every projection derives from

385    /// The one property that makes this a composition rather than a parallel registry: for
386    /// every issue of this repository, the task's status *is* the plan's status, and its
387    /// readiness names that same status back.
388    #[test]
389    fn the_status_never_disagrees_with_the_plan() {
390        let repo = crate::synthetic::SyntheticRepository::small().expect("synthetic repository");
391        let index = repo.index().expect("index");
392        let plan = Plan::build(&index);
393        for issue in &plan.issues {
394            let record = record(&index, ISSUE, &issue.id)
395                .map(|(path, metadata)| RecordRef { path, metadata });
396            let model = DevTaskModel {
397                plan: &plan,
398                record,
399                trace: None,
400                sessions: Vec::new(),
401                repository: repository_of(&plan),
402            };
403            let task = DevTask::build(&model, &issue.id);
404            assert_eq!(
405                task.position.status.text(),
406                issue.status,
407                "{} disagrees with the plan",
408                issue.id
409            );
410            assert_eq!(
411                task.readiness.canonical_status, issue.status,
412                "{} readiness names a status the plan did not derive",
413                issue.id
414            );
415            assert_eq!(task.position.status.provenance, FieldProvenance::Derived);
416            assert!(task.is_consistent());
417        }
418    }
