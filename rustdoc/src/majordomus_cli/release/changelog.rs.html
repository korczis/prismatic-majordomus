1//! Composing the changelog out of what the repository already states.
2//!
3//! Nothing here is authored. A section per release record, newest first, with an unreleased
4//! section leading when there is work after the last one:
5//!
6//! ```text
7//!   release record  ──►  version, tag, date, commit, artifacts
8//!   adr objects     ──►  the decisions dated inside this release's window
9//!   git log A..B    ──►  the changes, as conventional commits
10//! ```
11//!
12//! The window of a release is `(previous release's date, this release's date]` for the
13//! decisions and `previous..this` for the commits. Both are the same interval said in the
14//! two vocabularies the two sources have — an ADR carries a date and no commit, and a commit
15//! carries no date the layer indexes.
16

31/// The repository the links point into, as the crate manifest declares it.
32///
33/// Never a literal: `about::REPOSITORY` is `CARGO_PKG_REPOSITORY`, so a fork or a move
34/// carries every link with it and nothing here has to be told. A URL that is not a forge
35/// this understands yields no links at all rather than a guess — a wrong link is worse than
36/// no link, because a reader cannot tell it is wrong until they follow it.
37fn forge() -> Option<&'static str> {
38    let url = crate::about::REPOSITORY.trim_end_matches('/');
39    if url.starts_with("https://github.com/") && url.split('/').count() == 5 {
40        Some(url)
41    } else {
42        None
43    }
44}

61/// Compose the changelog, unreleased section included.
62///
63/// `objects` is the index's own view of the layer; `root` is the repository git reads. A
64/// repository with no records yields one unreleased section, which is the right answer for a
65/// project that has never published rather than an error.
66///
67/// This is what a live reader wants — the API, MCP and the Cockpit answer at request time
68/// and HEAD is whatever it is then. It is not what a committed artifact may hold: see
69/// [`compose_published`].
70pub fn compose(root: &Path, objects: &[Object]) -> Changelog {
71    compose_with(root, objects, true)
72}
73
74/// Compose the changelog of what has been published, and nothing after it.
75///
76/// The one form a committed artifact may take. The unreleased section is
77/// `<last release>..HEAD`, and a file inside a commit cannot describe the commit it is in:
78/// every commit made the committed changelog stale by construction, the pre-commit gate
79/// refused until a full derive had run, and every session paid several minutes per commit
80/// or worked around the gate. A document that depends only on the release records and the
81/// history up to the newest of them is the same bytes at every commit that changes neither,
82/// which is what "current" has to mean for a file under `docs/generated/`.
83///
84/// A repository with no records yields no section at all here — there is nothing published
85/// to describe — where [`compose`] yields the unreleased one.
86pub fn compose_published(root: &Path, objects: &[Object]) -> Changelog {
87    compose_with(root, objects, false)
88}

90fn compose_with(root: &Path, objects: &[Object], unreleased: bool) -> Changelog {
91    let mut diagnostics = Vec::new();
92
93    let mut records: Vec<Record> = objects
94        .iter()
95        .filter(|o| o.kind == RELEASE_KIND)
96        .filter_map(|o| record_of(&o.metadata, &mut diagnostics))
97        .collect();
98    // Newest first, by the date the record carries: the order a changelog is read in.
99    records.sort_by(|a, b| b.date.cmp(&a.date));
100
101    let decisions = decisions_of(root, objects);
102    // Which commits each release's tree holds, read once per release for every decision
103    // asked about it (see `Ancestry`).
104    let mut ancestry = Ancestry::new(root);
105
106    let mut sections = Vec::new();
107
108    // The unreleased section: everything after the newest release, when there is any, and
109    // only for a reader that asked for it — a committed document never does (see
110    // `compose_published`).
111    let newest = records.first();
112    let unreleased_range = match newest {
113        Some(r) => format!("{}..HEAD", r.commit),
114        None => "HEAD".to_string(),
115    };
116    let unreleased_changes = if unreleased {
117        commits::in_range(root, &unreleased_range, objects)
118    } else {
119        Vec::new()
120    };
121    if !unreleased || (newest.is_some() && unreleased_changes.is_empty()) {
122        // nothing since the last release, or nobody asked; no section rather than an empty one
123    } else {
124        sections.push(ReleaseSection {
125            version: "unreleased".into(),
126            tag: None,
127            date: None,
128            commit: None,
129            unreleased: true,
130            // Nothing is published yet, so there are no notes; the range is what has landed
131            // since the last release, which is exactly what this section lists.
132            notes_url: None,
133            compare_url: forge()
134                .zip(newest)
135                .map(|(base, r)| format!("{base}/compare/{}...master", r.tag)),
136            tree_url: forge().map(|base| format!("{base}/tree/master")),
137            decisions: decisions_after(
138                &mut ancestry,
139                &decisions,
140                newest.map(|r| r.commit.as_str()),
141            ),
142            groups: grouped(unreleased_changes),
143            artifacts: Vec::new(),
144        });
145    }
146
147    for (i, r) in records.iter().enumerate() {
148        let previous = records.get(i + 1);
149        let range = match previous {
150            Some(p) => format!("{}..{}", p.commit, r.commit),
151            // The first release: everything up to it. A clone without that history answers
152            // nothing, which the diagnostics below make visible rather than silent.
153            None => r.commit.clone(),
154        };
155        let changes = commits::in_range(root, &range, objects);
156        if changes.is_empty() {
157            diagnostics.push(format!(
158                "no commit was readable for {} ({}); the clone may not carry that history",
159                r.version, range
160            ));
161        }
162        sections.push(ReleaseSection {
163            version: r.version.clone(),
164            tag: Some(r.tag.clone()),
165            date: Some(r.date.clone()),
166            commit: Some(r.commit.clone()),
167            unreleased: false,
168            // The record names its own notes; the range and the tree are the forge's own
169            // addresses for facts the record already carries, so neither is authored.
170            notes_url: r
171                .notes
172                .clone()
173                .or_else(|| forge().map(|base| format!("{base}/releases/tag/{}", r.tag))),
174            compare_url: forge().map(|base| match previous {
175                Some(p) => format!("{base}/compare/{}...{}", p.tag, r.tag),
176                None => format!("{base}/commits/{}", r.tag),
177            }),
178            tree_url: forge().map(|base| format!("{base}/tree/{}", r.tag)),
179            decisions: decisions_between(
180                &mut ancestry,
181                &decisions,
182                previous.map(|p| p.commit.as_str()),
183                &r.commit,
184            ),
185            groups: grouped(changes),
186            artifacts: r.artifacts.clone(),
187        });
188    }
189
190    Changelog {
191        schema: CHANGELOG_SCHEMA.into(),
192        current: version::declared(root).unwrap_or_else(|| "unknown".into()),
193        sections,
194        diagnostics,
195        // Filled by whoever answers with it: the composer does not know which surface asked.
196        produced_by: None,
197    }
198}

200/// The changes of a section, grouped by kind and ordered by the rank each kind carries.
201///
202/// Done here rather than in each renderer: the Markdown, the site and any other reader get
203/// the same order because they are given it, not because they each reimplemented it.
204///
205/// One group per kind, exactly. The rank is a *presentation* order and two kinds may share
206/// one — `Ci` and `Build` are both rank 6, `Chore` and `Style` both rank 7 — so a sort keyed
207/// on the rank alone leaves equal-ranked kinds interleaved in the order their commits
208/// arrived, and `dedup`, which removes only *adjacent* repeats, then leaves the same kind
209/// standing several times. That is what put six `chore` groups and five `style` ones on the
210/// changelog page, each rendering all of its changes again under an id the page had already
211/// used, and what repeated their headings in the Markdown. The key is therefore total: the
212/// rank first, then the kind's own order, so equal ranks are adjacent and one survives.
213fn grouped(mut changes: Vec<Change>) -> Vec<ChangeGroup> {
214    if let Some(base) = forge() {
215        for c in &mut changes {
216            c.url = Some(format!("{base}/commit/{}", c.commit));
217        }
218    }
219    let mut kinds: Vec<_> = changes.iter().map(|c| c.kind).collect();
220    kinds.sort_by_key(|k| (k.rank(), *k));
221    kinds.dedup();
222    kinds
223        .into_iter()
224        .map(|kind| ChangeGroup {
225            kind,
226            heading: kind.heading().to_string(),
227            rank: kind.rank(),
228            changes: changes.iter().filter(|c| c.kind == kind).cloned().collect(),
229        })
230        .collect()
231}

233/// One release record, from its metadata.
234fn record_of(metadata: &Value, diagnostics: &mut Vec<String>) -> Option<Record> {
235    let version = metadata.get("version")?.as_str()?.to_string();
236    let tag = metadata
237        .get("tag")
238        .and_then(|v| v.as_str())
239        .unwrap_or(&version)
240        .to_string();
241    let date = metadata
242        .get("published_at")
243        .and_then(|v| v.as_str())
244        .unwrap_or_default()
245        .to_string();
246    let Some(commit) = metadata.get("commit").and_then(|v| v.as_str()) else {
247        diagnostics.push(format!(
248            "the record for {version} names no commit, so its changes cannot be read"
249        ));
250        return None;
251    };
252    let artifacts = metadata
253        .get("artifacts")
254        .and_then(|v| v.as_array())
255        .map(|a| {
256            a.iter()
257                .filter_map(|v| {
258                    Some(Artifact {
259                        target: v.get("target")?.as_str()?.to_string(),
260                        name: v.get("name")?.as_str()?.to_string(),
261                        sha256: v
262                            .get("sha256")
263                            .and_then(|s| s.as_str())
264                            .unwrap_or_default()
265                            .to_string(),
266                    })
267                })
268                .collect()
269        })
270        .unwrap_or_default();
271    Some(Record {
272        version,
273        tag,
274        date,
275        commit: commit.to_string(),
276        notes: metadata
277            .get("notes_url")
278            .and_then(|v| v.as_str())
279            .map(|s| s.to_string()),
280        artifacts,
281    })
282}

284/// When the file that states a decision was added, as git knows it.
285///
286/// The date in an ADR's front matter is when the decision was *made*, which is not the same
287/// question as which release carried it — and using it put two decisions written on 2026-09-09
288/// into a release published on 2026-09-08, because both ends were compared as calendar days.
289/// The commits half of this module already asks git; the decisions half asks git now too.
290fn added_at(root: &Path, path: &str) -> Option<String> {
291    let out = std::process::Command::new("git")
292        .arg("-C")
293        .arg(root)
294        .args([
295            "log",
296            "--diff-filter=A",
297            "--format=%H",
298            "--max-count=1",
299            "--",
300            path,
301        ])
302        .output()
303        .ok()?;
304    if !out.status.success() {
305        return None;
306    }
307    let sha = String::from_utf8_lossy(&out.stdout).trim().to_string();
308    if sha.is_empty() {
309        None
310    } else {
311        Some(sha)
312    }
313}

315/// When each of `paths` was added, as [`added_at`] answers it, in the order given.
316///
317/// One `git log` per decision walks the whole history each time, and asked one after the
318/// other they were 43% of `generate --check` — some ten of the twenty seconds the pre-commit
319/// gate is bounded by (case 56). A single walk over every path is not the same question:
320/// git simplifies history per pathspec, so a merge that leaves one decision untouched but
321/// changes another prunes a side branch for the first and follows it for the second, and on
322/// this repository's history three of 95 answers moved. The question is unchanged here; the
323/// processes run side by side instead, and they mostly wait on git rather than on this
324/// process.
325fn added_at_each(root: &Path, paths: &[&str]) -> Vec<Option<String>> {
326    // as many at once as the machine runs, and no more: it runs other sessions' work too
327    let threads = std::thread::available_parallelism()
328        .map(|n| n.get())
329        .unwrap_or(4)
330        .clamp(1, 16)
331        .min(paths.len());
332    if threads < 2 {
333        return paths.iter().map(|p| added_at(root, p)).collect();
334    }
335    let next = AtomicUsize::new(0);
336    let answered: Vec<(usize, Option<String>)> = std::thread::scope(|scope| {
337        let workers: Vec<_> = (0..threads)
338            .map(|_| {
339                scope.spawn(|| {
340                    let mut mine = Vec::new();
341                    loop {
342                        let i = next.fetch_add(1, Ordering::Relaxed);
343                        let Some(path) = paths.get(i) else {
344                            return mine;
345                        };
346                        mine.push((i, added_at(root, path)));
347                    }
348                })
349            })
350            .collect();
351        workers
352            .into_iter()
353            // `added_at` has no panic site, so a worker always returns: no panic to carry on
354            .flat_map(|w| w.join().expect("added_at does not panic"))
355            .collect()
356    });
357    let mut out = vec![None; paths.len()];
358    for (i, added) in answered {
359        out[i] = added;
360    }
361    out
362}

364/// Which commits a tree already holds, asked of git once per tree.
365///
366/// The question the two window functions ask is "was the commit that added this decision
367/// already in that release's tree?", for every decision against every release, twice for
368/// most. Asked as `git merge-base --is-ancestor` it was one process per pair: some six hundred
369/// at 62 decisions and six releases, about 6.5 of the 13.7 seconds an unoptimised
370/// `generate --check` spent, and so of the pre-commit gate that runs it on every commit, and
371/// it grows with the product of the two. Here each tree's history is listed once
372/// (`git rev-list <tree>`) and every question about it is a lookup: the same answer, since a
373/// commit is an ancestor of a tree exactly when that list names it (the tree's own commit
374/// included, as `--is-ancestor` includes it), and a name git cannot resolve holds nothing,
375/// as `--is-ancestor` refuses it.
376struct Ancestry<'a> {
377    root: &'a Path,
378    held: HashMap<String, HashSet<String>>,
379}
380
381impl<'a> Ancestry<'a> {
382    fn new(root: &'a Path) -> Self {
383        Ancestry {
384            root,
385            held: HashMap::new(),
386        }
387    }
388
389    /// Whether `commit` — a full commit name, as [`added_at`] gives it — was already in the
390    /// tree `of`: whether it is `of` or one of its ancestors.
391    fn holds(&mut self, of: &str, commit: &str) -> bool {
392        let root = self.root;
393        self.held
394            .entry(of.to_string())
395            .or_insert_with(|| history(root, of))
396            .contains(commit)
397    }
398}

400/// Every commit `of` reaches, by full name; nothing when git cannot resolve it.
401fn history(root: &Path, of: &str) -> HashSet<String> {
402    let Ok(out) = std::process::Command::new("git")
403        .arg("-C")
404        .arg(root)
405        .args(["rev-list", of, "--"])
406        .output()
407    else {
408        return HashSet::new();
409    };
410    if !out.status.success() {
411        return HashSet::new();
412    }
413    String::from_utf8_lossy(&out.stdout)
414        .lines()
415        .map(str::to_string)
416        .collect()
417}

419/// Every decision the layer holds, newest first.
420fn decisions_of(root: &Path, objects: &[Object]) -> Vec<Decision> {
421    let adrs: Vec<(&Object, String)> = objects
422        .iter()
423        .filter(|o| o.kind == ADR_KIND)
424        .filter_map(|o| Some((o, o.metadata.get("id")?.as_str()?.to_string())))
425        .collect();
426    let paths: Vec<&str> = adrs
427        .iter()
428        .map(|(o, _)| o.provenance.path.as_str())
429        .collect();
430    let mut out: Vec<Decision> = adrs
431        .into_iter()
432        .zip(added_at_each(root, &paths))
433        .map(|((o, id), added)| {
434            Decision {
435                title: o
436                    .title
437                    .clone()
438                    .or_else(|| {
439                        o.metadata
440                            .get("title")
441                            .and_then(|v| v.as_str())
442                            .map(|s| s.to_string())
443                    })
444                    .unwrap_or_else(|| id.clone()),
445                status: o
446                    .metadata
447                    .get("status")
448                    .and_then(|v| v.as_str())
449                    .unwrap_or("unknown")
450                    .to_string(),
451                date: o
452                    .metadata
453                    .get("date")
454                    .and_then(|v| v.as_str())
455                    .unwrap_or_default()
456                    .to_string(),
457                url: forge().map(|base| format!("{base}/blob/master/{}", o.provenance.path)),
458                // Which release carried it is a question for git, not for the front matter:
459                // the date says when the decision was made, and two ADRs written the day
460                // after a release were claimed by it while this read that field instead.
461                added,
462                id,
463            }
464        })
465        .collect();
466    out.sort_by(|a, b| b.date.cmp(&a.date).then_with(|| b.id.cmp(&a.id)));
467    out
468}

470/// The decisions not yet in any release: their file was added after the newest release's tree.
471fn decisions_after(
472    ancestry: &mut Ancestry<'_>,
473    decisions: &[Decision],
474    newest: Option<&str>,
475) -> Vec<Decision> {
476    decisions
477        .iter()
478        .filter(|d| match (newest, d.added.as_deref()) {
479            // added, and not already in the last release's tree
480            (Some(rel), Some(added)) => !ancestry.holds(rel, added),
481            // nothing released yet, or git could not say when it was added: an unreleased
482            // section that omits a decision is worse than one that shows an early arrival
483            _ => true,
484        })
485        .cloned()
486        .collect()
487}

489/// The decisions this release carried: in its tree, and not in the one before it.
490fn decisions_between(
491    ancestry: &mut Ancestry<'_>,
492    decisions: &[Decision],
493    previous: Option<&str>,
494    release: &str,
495) -> Vec<Decision> {
496    decisions
497        .iter()
498        .filter(|d| {
499            let Some(added) = d.added.as_deref() else {
500                return false;
501            };
502            ancestry.holds(release, added) && previous.is_none_or(|p| !ancestry.holds(p, added))
503        })
504        .cloned()
505        .collect()
506}

508/// The `YYYY-MM-DD` of an ISO-8601 date or timestamp.
509fn day(text: &str) -> &str {
510    text.split('T').next().unwrap_or(text)
511}

513/// The changelog as Markdown, which is what the generated document and the site render.
514pub fn render(changelog: &Changelog) -> String {
515    let mut out = String::new();
516    out.push_str("# Changelog\n\n");
517    out.push_str(
518        "Every entry below is derived: a section per release the layer records, its decisions \
519         the ADRs dated inside that release's window, its changes the conventional commits in \
520         its range, its artifacts the record's own evidence. Nothing here is written by hand, \
521         and `majordomus generate changelog --check` fails when it stops matching the tree.\n\n",
522    );
523    out.push_str(&format!("Current version: **{}**\n", changelog.current));
524
525    for section in &changelog.sections {
526        out.push('\n');
527        if section.unreleased {
528            out.push_str("## Unreleased\n\n");
529        } else {
530            let date = section.date.as_deref().unwrap_or("");
531            out.push_str(&format!(
532                "## {} — {}\n\n",
533                section.tag.as_deref().unwrap_or(&section.version),
534                day(date)
535            ));
536        }
537
538        if !section.decisions.is_empty() {
539            out.push_str("### Decisions\n\n");
540            for d in &section.decisions {
541                out.push_str(&format!(
542                    "- **{}** {} _({})_\n",
543                    d.id.to_uppercase(),
544                    d.title,
545                    d.status
546                ));
547            }
548            out.push('\n');
549        }
550
551        // The groups the document carries, in the order it carries them: this renderer no
552        // longer decides the order, it reads it, which is what lets the site agree with it.
553        for group in &section.groups {
554            out.push_str(&format!("### {}\n\n", group.heading));
555            for c in &group.changes {
556                let scope = c
557                    .scope
558                    .as_ref()
559                    .map(|s| format!("**{s}**: "))
560                    .unwrap_or_default();
561                let breaking = if c.breaking { "**BREAKING** " } else { "" };
562                out.push_str(&format!(
563                    "- {breaking}{scope}{} (`{}`)\n",
564                    c.subject, c.commit
565                ));
566            }
567            out.push('\n');
568        }
569
570        if !section.artifacts.is_empty() {
571            out.push_str("### Published\n\n");
572            for a in &section.artifacts {
573                out.push_str(&format!(
574                    "- `{}` — {} (`{}`)\n",
575                    a.target,
576                    a.name,
577                    &a.sha256[..a.sha256.len().min(12)]
578                ));
579            }
580            out.push('\n');
581        }
582    }
583
584    if !changelog.diagnostics.is_empty() {
585        out.push_str("\n## What could not be read\n\n");
586        for d in &changelog.diagnostics {
587            out.push_str(&format!("- {d}\n"));
588        }
589    }
590    out
591}

719    /// Listing a tree's history once answers what `git merge-base --is-ancestor` answered one
720    /// process per question: for every pair of commits of a history with a branch and a merge
721    /// in it — a commit holds itself, a side branch's commit is held from the merge on and not
722    /// before — and a tree git cannot resolve, which is what a clone without that history
723    /// has, holds nothing. And each tree is listed once however often it is asked about, which
724    /// is the whole of the change.
725    #[test]
726    fn a_tree_holds_exactly_the_commits_merge_base_calls_its_ancestors() {
727        let dir = tempfile::tempdir().expect("a temp dir");
728        let root = dir.path();
729        git(root, &["init", "-q", "-b", "main"]);
730        let step = |message: &str| {
731            git(root, &["commit", "-q", "--allow-empty", "-m", message]);
732            git(root, &["rev-parse", "HEAD"])
733        };
734        let a = step("a");
735        git(root, &["checkout", "-q", "-b", "side"]);
736        let s = step("s");
737        git(root, &["checkout", "-q", "main"]);
738        let b = step("b");
739        git(root, &["merge", "-q", "--no-ff", "-m", "m", "side"]);
740        let m = git(root, &["rev-parse", "HEAD"]);
741        let c = step("c");
742        let all = [a, s, b, m, c];
743
744        let mut ancestry = Ancestry::new(root);
745        let mut held = 0;
746        for of in &all {
747            for commit in &all {
748                let asked = std::process::Command::new("git")
749                    .arg("-C")
750                    .arg(root)
751                    .args(["merge-base", "--is-ancestor", commit, of])
752                    .status()
753                    .expect("git runs")
754                    .success();
755                assert_eq!(ancestry.holds(of, commit), asked, "is {commit} in {of}?");
756                held += usize::from(asked);
757            }
758        }
759        // both answers occur, so the agreement is not vacuous
760        assert!(held > 0 && held < all.len() * all.len(), "{held} held");
761        let unknown = "0".repeat(40);
762        assert!(!ancestry.holds(&unknown, &all[0]));
763        assert_eq!(ancestry.held.len(), all.len() + 1, "one listing per tree");
764    }

807    /// A throwaway repository with `n` commits, the first of them tagged as a release the
808    /// records name. What every test of the two compositions needs: a real HEAD that moves.
809    fn repo_with_commits(n: usize) -> (tempfile::TempDir, String) {
810        let dir = tempfile::tempdir().expect("a temp dir");
811        let root = dir.path();
812        let git = |args: &[&str]| {
813            let out = std::process::Command::new("git")
814                .arg("-C")
815                .arg(root)
816                .args(args)
817                .env("GIT_AUTHOR_NAME", "t")
818                .env("GIT_AUTHOR_EMAIL", "t@t")
819                .env("GIT_COMMITTER_NAME", "t")
820                .env("GIT_COMMITTER_EMAIL", "t@t")
821                .output()
822                .expect("git runs");
823            assert!(
824                out.status.success(),
825                "git {args:?}: {}",
826                String::from_utf8_lossy(&out.stderr)
827            );
828            String::from_utf8_lossy(&out.stdout).trim().to_string()
829        };
830        git(&["init", "-q"]);
831        git(&[
832            "commit",
833            "-q",
834            "--allow-empty",
835            "-m",
836            "feat(a): the release",
837        ]);
838        let release = git(&["rev-parse", "HEAD"]);
839        for i in 1..n {
840            git(&[
841                "commit",
842                "-q",
843                "--allow-empty",
844                "-m",
845                &format!("fix(b): after the release {i}"),
846            ]);
847        }
848        (dir, release)
849    }
