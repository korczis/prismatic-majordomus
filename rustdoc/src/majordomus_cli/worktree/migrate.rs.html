1//! Bringing misplaced worktrees home, transactionally and losslessly.
2//!
3//! A plan is computed from the topology and changes nothing. Applying it recomputes the plan
4//! under the repository lock, then for each movable step takes a
5//! [`WorktreeFingerprint`] of the worktree, asks git to move it, takes another at the
6//! destination and refuses to call the step a success unless the two are equal. A dirty
7//! worktree is not an error: it is the case this exists for, and `git worktree move` moves
8//! the directory — modified, staged and untracked files included — with one rename.
9//!
10//! Two shapes need more than a rename. A worktree that occupies the container path itself
11//! is moved out to a staging path beside the container, the container is created, and the
12//! worktree is moved into it. A move that crosses filesystems, which `rename` cannot do, is
13//! made by copying the tree, repairing git's administrative link, verifying the copy —
14//! including a manifest of every entry of the directory tree — and only then removing the
15//! original. Nothing is ever reset, stashed, cleaned or checked out.
16

33/// How a step moves its worktree.
34#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
35#[serde(rename_all = "snake_case")]
36pub enum MigrationAction {
37    /// One `git worktree move`.
38    Move,
39    /// The worktree is the container itself: out to a staging path, then into the container.
40    MoveViaStaging,
41    /// The move crossed devices: copy, `git worktree repair`, verify, remove the original.
42    CopyAndRepair,
43    /// Nothing: the step is blocked.
44    None,
45}
46
47/// What became of a step.
48#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
49#[serde(rename_all = "snake_case")]
50pub enum StepOutcome {
51    /// Planned and not applied.
52    Planned,
53    /// Not carried out, with the blockers on the step.
54    Blocked,
55    /// Moved, and the fingerprint after equals the one before.
56    Moved,
57    /// Attempted and not completed, or completed and not verified; the message says which.
58    Failed,
59}

61/// One worktree to bring home.
62#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
63pub struct MigrationStep {
64    /// The branch it holds.
65    pub branch: String,
66    /// Where it is.
67    pub from: String,
68    /// Where it belongs.
69    pub to: String,
70    #[serde(default, skip_serializing_if = "Option::is_none")]
71    /// The commit checked out.
72    pub head: Option<String>,
73    /// How it moves.
74    pub action: MigrationAction,
75    /// Its uncommitted work, which moves with it.
76    pub dirty: DirtyState,
77    /// Why it cannot move, when it cannot.
78    pub blockers: Vec<TopologyDiagnostic>,
79    /// What happened.
80    pub outcome: StepOutcome,
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    /// What happened, in words, when it was not simply moved.
83    pub message: Option<String>,
84    #[serde(default, skip_serializing_if = "Option::is_none")]
85    /// The fingerprint before the move.
86    pub before: Option<WorktreeFingerprint>,
87    #[serde(default, skip_serializing_if = "Option::is_none")]
88    /// The fingerprint after it.
89    pub after: Option<WorktreeFingerprint>,
90    /// What differs between the two; empty when verified.
91    pub differences: Vec<String>,
92    #[serde(default, skip_serializing_if = "Option::is_none")]
93    /// What became of its `.envrc` under direnv once it was at its new path: the primary
94    /// checkout's approval carried there, or why it was not. Only on a moved step.
95    pub envrc: Option<EnvrcApproval>,
96}

98/// A migration, planned or applied.
99#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
100pub struct MigrationPlan {
101    /// [`SCHEMA`].
102    pub schema: String,
103    /// The container everything moves into.
104    pub container: String,
105    /// One step per misplaced worktree with a branch, container occupants first.
106    pub steps: Vec<MigrationStep>,
107    /// Worktrees the migration cannot address by design: detached, missing, the primary
108    /// checkout off the trunk. Each says what a person does about it.
109    pub exceptions: Vec<TopologyDiagnostic>,
110    /// Steps that can be carried out as things stand.
111    pub movable: usize,
112    /// Steps that cannot.
113    pub blocked: usize,
114    /// Steps carried out and verified.
115    pub moved: usize,
116    /// Steps attempted and not verified.
117    pub failed: usize,
118    /// Whether anything was changed.
119    pub applied: bool,
120    #[serde(default, skip_serializing_if = "Option::is_none")]
121    /// The new path of the worktree the command was run from, when that one moved.
122    pub moved_current: Option<String>,
123}

125/// What an apply may do beyond a rename.
126#[derive(Debug, Clone, Default)]
127pub struct MigrationOptions {
128    /// Fall back to copy, repair, verify and remove when a move crosses devices.
129    pub allow_copy: bool,
130    /// Only these branches.
131    pub only: Vec<String>,
132    /// Also move the scratch checkouts of sessions (`Standing::Ephemeral`), which are
133    /// otherwise reported and left alone.
134    pub include_ephemeral: bool,
135}

137/// The plan: every misplaced worktree with a branch, and what stands in its way. Changes
138/// nothing. Ephemeral worktrees are exceptions, not steps.
139pub fn plan(service: &WorktreeService) -> Result<MigrationPlan> {
140    plan_with(service, false)
141}

143/// The plan, optionally counting ephemeral worktrees as steps.
144///
145/// A plan states the uncommitted work of the work trees it moves, and of no others — every
146/// other line of it is read out of a standing, which costs no subprocess. So the topology
147/// is read without the per-work-tree `git status` first, and measured only when that
148/// reading shows there is a step to carry work. On a repository whose work trees are all
149/// where they belong that is the difference between 129 `git status` subprocesses and
150/// none, and the Cockpit's worktree page renders this beside the topology: it was paying
151/// for the same measurement twice.
152pub fn plan_with(service: &WorktreeService, include_ephemeral: bool) -> Result<MigrationPlan> {
153    let fast = service.topology(Detail::Fast)?;
154    let moves_something = fast
155        .worktrees
156        .iter()
157        .any(|w| is_step(w.standing, include_ephemeral));
158    let topology = if moves_something {
159        service.topology(Detail::Full)?
160    } else {
161        fast
162    };
163    Ok(plan_from(&topology, include_ephemeral))
164}

166/// Is a work tree with this standing one the migration would move?
167fn is_step(standing: Standing, include_ephemeral: bool) -> bool {
168    match standing {
169        Standing::Misplaced => true,
170        Standing::Ephemeral => include_ephemeral,
171        _ => false,
172    }
173}

175/// The plan a topology implies.
176///
177/// Pure: every step, every exception and every blocker is read out of the work trees the
178/// topology already judged, and the container it names is the topology's own. The
179/// migration plan is a projection of the topology and of nothing else, and this is the one
180/// place that is written.
181pub fn plan_from(topology: &RepositoryTopology, include_ephemeral: bool) -> MigrationPlan {
182    let container = topology.container.path.clone();
183    let mut steps = Vec::new();
184    let mut exceptions = Vec::new();
185    for w in &topology.worktrees {
186        match w.standing {
187            Standing::Ephemeral if !include_ephemeral => {
188                exceptions.extend(w.diagnostics.iter().cloned());
189            }
190            Standing::Misplaced | Standing::Ephemeral => {
191                let (Some(branch), Some(to)) = (&w.branch, &w.expected_path) else {
192                    exceptions.extend(w.diagnostics.iter().cloned());
193                    continue;
194                };
195                let occupies_container = w
196                    .diagnostics
197                    .iter()
198                    .any(|d| d.code == DiagnosticCode::ContainerOccupied);
199                let blockers: Vec<TopologyDiagnostic> = w
200                    .diagnostics
201                    .iter()
202                    .filter(|d| {
203                        matches!(
204                            d.code,
205                            DiagnosticCode::Locked
206                                | DiagnosticCode::DestinationConflict
207                                | DiagnosticCode::InvalidBranchName
208                                | DiagnosticCode::PathEscape
209                        )
210                    })
211                    .cloned()
212                    .collect();
213                steps.push(MigrationStep {
214                    branch: branch.clone(),
215                    from: w.path.clone(),
216                    to: to.clone(),
217                    head: w.head.clone(),
218                    action: if !blockers.is_empty() {
219                        MigrationAction::None
220                    } else if occupies_container {
221                        MigrationAction::MoveViaStaging
222                    } else {
223                        MigrationAction::Move
224                    },
225                    dirty: w.dirty.clone().unwrap_or_default(),
226                    outcome: if blockers.is_empty() {
227                        StepOutcome::Planned
228                    } else {
229                        StepOutcome::Blocked
230                    },
231                    blockers,
232                    message: None,
233                    before: None,
234                    after: None,
235                    differences: Vec::new(),
236                    envrc: None,
237                });
238            }
239            Standing::Missing | Standing::Detached => {
240                exceptions.extend(w.diagnostics.iter().cloned());
241            }
242            Standing::Primary => {
243                exceptions.extend(
244                    w.diagnostics
245                        .iter()
246                        .filter(|d| d.code == DiagnosticCode::PrimaryOnNonTrunk)
247                        .cloned(),
248                );
249            }
250            Standing::Canonical => {}
251        }
252    }
253    // The container's occupant moves first: every other destination is inside the
254    // container, and while a worktree *is* the container those destinations would be
255    // created inside that worktree. If it cannot move, nothing can.
256    steps.sort_by_key(|s| s.action != MigrationAction::MoveViaStaging);
257    let occupant_blocked = steps
258        .iter()
259        .any(|s| s.outcome == StepOutcome::Blocked && s.from == container);
260    if occupant_blocked {
261        for s in steps.iter_mut().filter(|s| s.from != container) {
262            s.blockers.push(TopologyDiagnostic {
263                code: DiagnosticCode::ContainerOccupied,
264                severity: Severity::Error,
265                path: Some(s.from.clone()),
266                branch: Some(s.branch.clone()),
267                expected: Some(s.to.clone()),
268                message: format!(
269                    "the container {} is occupied by a worktree that cannot move yet",
270                    container
271                ),
272                remedy: "unblock the container's occupant first".into(),
273            });
274            s.action = MigrationAction::None;
275            s.outcome = StepOutcome::Blocked;
276        }
277    }
278    let movable = steps
279        .iter()
280        .filter(|s| s.outcome == StepOutcome::Planned)
281        .count();
282    MigrationPlan {
283        schema: SCHEMA.to_string(),
284        container,
285        blocked: steps.len() - movable,
286        movable,
287        moved: 0,
288        failed: 0,
289        steps,
290        exceptions,
291        applied: false,
292        moved_current: None,
293    }
294}

296/// Carry out the plan. Recomputed under the lock, never replayed from a plan the caller is
297/// holding: the tree may have changed since it was printed.
298pub fn apply(service: &WorktreeService, options: &MigrationOptions) -> Result<MigrationPlan> {
299    let _lock = WorktreeLock::acquire(&service.identity().git_common_dir().path)?;
300    let mut plan = plan_with(service, options.include_ephemeral)?;
301    let current = service.identity().current_worktree().clone();
302    let primary = service.identity().primary_worktree().path.clone();
303    let container = service.container().path.clone();
304
305    for step in plan.steps.iter_mut() {
306        if step.outcome != StepOutcome::Planned {
307            continue;
308        }
309        if !options.only.is_empty() && !options.only.contains(&step.branch) {
310            step.outcome = StepOutcome::Blocked;
311            step.message = Some("not selected".into());
312            continue;
313        }
314        // Prove the destination is still free immediately before moving: planning and
315        // applying are not atomic with respect to anything outside this lock.
316        let to = PathBuf::from(&step.to);
317        if step.action != MigrationAction::MoveViaStaging && std::fs::symlink_metadata(&to).is_ok()
318        {
319            step.outcome = StepOutcome::Failed;
320            step.message = Some(format!(
321                "{} appeared before the move could be made",
322                step.to
323            ));
324            continue;
325        }
326        if !ResolvedPath::of(&to).is_inside(service.container()) {
327            step.outcome = StepOutcome::Failed;
328            step.message = Some(format!("{} is not inside {}", step.to, display(&container)));
329            continue;
330        }
331        let from = PathBuf::from(&step.from);
332        let was_current = ResolvedPath::of(&from).same_as(&current);
333        match execute(service, &primary, &container, step, &from, &to, options) {
334            Ok(()) => {
335                if was_current {
336                    plan.moved_current = Some(step.to.clone());
337                }
338            }
339            Err(e) => {
340                step.outcome = StepOutcome::Failed;
341                step.message = Some(e.to_string());
342            }
343        }
344    }
345    plan.applied = true;
346    plan.moved = plan
347        .steps
348        .iter()
349        .filter(|s| s.outcome == StepOutcome::Moved)
350        .count();
351    plan.failed = plan
352        .steps
353        .iter()
354        .filter(|s| s.outcome == StepOutcome::Failed)
355        .count();
356    plan.blocked = plan
357        .steps
358        .iter()
359        .filter(|s| s.outcome == StepOutcome::Blocked)
360        .count();
361    plan.movable = 0;
362    Ok(plan)
363}

365fn execute(
366    service: &WorktreeService,
367    primary: &Path,
368    container: &Path,
369    step: &mut MigrationStep,
370    from: &Path,
371    to: &Path,
372    options: &MigrationOptions,
373) -> Result<()> {
374    let before = fingerprint::capture(from, false)?;
375    step.before = Some(before.clone());
376    match step.action {
377        MigrationAction::Move => match worktree_move(primary, from, to) {
378            Ok(()) => {}
379            Err(e) if is_cross_device(&e) => {
380                if !options.allow_copy {
381                    return Err(WorktreeError::MigrationUnsafe {
382                        path: from.to_path_buf(),
383                        reason: format!(
384                            "{e}; the destination is on another filesystem, and a copy was not \
385                             permitted (majordomus worktree migrate --allow-copy)"
386                        ),
387                    });
388                }
389                step.action = MigrationAction::CopyAndRepair;
390                let before_tree = fingerprint::capture(from, true)?;
391                step.before = Some(before_tree.clone());
392                copy_and_repair(primary, from, to)?;
393                let after_tree = fingerprint::capture(to, true)?;
394                let differences = fingerprint::differences(&before_tree, &after_tree);
395                step.after = Some(after_tree);
396                step.differences = differences.clone();
397                if !differences.is_empty() {
398                    step.outcome = StepOutcome::Failed;
399                    step.message = Some(format!(
400                        "the copy at {} does not match the original at {}, which was kept; \
401                         git points at the copy (git worktree repair {} points it back)",
402                        display(to),
403                        display(from),
404                        display(from)
405                    ));
406                    return Ok(());
407                }
408                // verified: the original is now an unregistered duplicate, and this is the one
409                // place a directory is deleted — after the proof, never before it
410                let mut identity = service.identity().clone();
411                identity.refresh()?;
412                if identity.record_at(&ResolvedPath::of(from)).is_some() {
413                    return Err(WorktreeError::MigrationUnsafe {
414                        path: from.to_path_buf(),
415                        reason: "git still registers the original after the repair; both copies were kept".into(),
416                    });
417                }
418                std::fs::remove_dir_all(from).map_err(|e| WorktreeError::io(from, &e))?;
419                step.outcome = StepOutcome::Moved;
420                step.message =
421                    Some("moved by copy across filesystems, verified, original removed".into());
422                step.envrc = Some(direnv::approve(primary, to));
423                return Ok(());
424            }
425            Err(e) => return Err(e),
426        },
427        MigrationAction::MoveViaStaging => {
428            let staging = staging_path(container);
429            worktree_move(primary, from, &staging)?;
430            if std::fs::symlink_metadata(container).is_ok() {
431                return Err(WorktreeError::MigrationUnsafe {
432                    path: staging,
433                    reason: format!(
434                        "after moving the occupant out, {} still exists; the worktree waits at the staging path and the next migration moves it home",
435                        display(container)
436                    ),
437                });
438            }
439            service.ensure_container()?;
440            if let Some(parent) = to.parent() {
441                std::fs::create_dir_all(parent).map_err(|e| WorktreeError::io(parent, &e))?;
442            }
443            worktree_move(primary, &staging, to)?;
444        }
445        MigrationAction::CopyAndRepair | MigrationAction::None => {
446            return Err(WorktreeError::MigrationUnsafe {
447                path: from.to_path_buf(),
448                reason: "the step was not planned as a move".into(),
449            })
450        }
451    }
452    let after = fingerprint::capture(to, false)?;
453    let differences = fingerprint::differences(&before, &after);
454    step.after = Some(after);
455    step.differences = differences.clone();
456    let mut identity = service.identity().clone();
457    identity.refresh()?;
458    let registered = identity.record_at(&ResolvedPath::of(to)).is_some();
459    if !differences.is_empty() || !registered {
460        step.outcome = StepOutcome::Failed;
461        step.message = Some(if registered {
462            format!(
463                "moved to {}, and the fingerprint differs: {}. A rename cannot lose content, so something wrote to the worktree during the move; inspect it there",
464                display(to),
465                differences.join("; ")
466            )
467        } else {
468            format!("git reported the move and does not list {}", display(to))
469        });
470        return Ok(());
471    }
472    step.outcome = StepOutcome::Moved;
473    // A moved `.envrc` is a new path to direnv, and the approval it had is gone with the old
474    // one; carry it, and say what happened beside the fingerprint that says the move was
475    // faithful.
476    step.envrc = Some(direnv::approve(primary, to));
477    let dangling = dangling_relative_links(to);
478    if !dangling.is_empty() {
479        step.message = Some(format!(
480            "relative symbolic links that no longer resolve from the new path: {}",
481            dangling.join(", ")
482        ));
483    }
484    Ok(())
485}

487/// Top-level symbolic links of a worktree whose relative target does not resolve any more:
488/// a `node_modules -> ../sibling/node_modules` written for the old path. Reported, never
489/// rewritten; the link is the person's and the move did not change it.
490fn dangling_relative_links(worktree: &Path) -> Vec<String> {
491    let Ok(entries) = std::fs::read_dir(worktree) else {
492        return Vec::new();
493    };
494    let mut out = Vec::new();
495    for entry in entries.filter_map(|e| e.ok()) {
496        let path = entry.path();
497        let Ok(meta) = std::fs::symlink_metadata(&path) else {
498            continue;
499        };
500        if !meta.file_type().is_symlink() {
501            continue;
502        }
503        let Ok(target) = std::fs::read_link(&path) else {
504            continue;
505        };
506        if target.is_relative() && !worktree.join(&target).exists() {
507            out.push(format!(
508                "{} -> {}",
509                entry.file_name().to_string_lossy(),
510                target.display()
511            ));
512        }
513    }
514    out.sort();
515    out
516}

518/// `git worktree move`, with absolute paths, from the primary checkout. Git resolves a
519/// relative path against its working directory and matches a bare name against the suffix
520/// of any registered worktree; absolute paths are the only form that says what it means.
521fn worktree_move(primary: &Path, from: &Path, to: &Path) -> Result<()> {
522    if let Some(parent) = to.parent() {
523        std::fs::create_dir_all(parent).map_err(|e| WorktreeError::io(parent, &e))?;
524    }
525    git::run(
526        primary,
527        &[
528            "worktree".to_string(),
529            "move".to_string(),
530            "--".to_string(),
531            absolute(from),
532            absolute(to),
533        ],
534    )?;
535    Ok(())
536}

538fn absolute(p: &Path) -> String {
539    if p.is_absolute() {
540        display(p)
541    } else {
542        display(
543            &std::env::current_dir()
544                .map(|c| c.join(p))
545                .unwrap_or_else(|_| p.to_path_buf()),
546        )
547    }
548}

550/// Where the container's occupant waits between its two moves.
551fn staging_path(container: &Path) -> PathBuf {
552    let name = container
553        .file_name()
554        .map(|n| n.to_string_lossy().to_string())
555        .unwrap_or_else(|| "worktrees".into());
556    container
557        .parent()
558        .unwrap_or(container)
559        .join(format!("{name}.migrating-{}", std::process::id()))
560}

562fn is_cross_device(e: &WorktreeError) -> bool {
563    match e {
564        WorktreeError::GitCommandFailed { stderr, .. } => {
565            let s = stderr.to_ascii_lowercase();
566            s.contains("cross-device") || s.contains("exdev")
567        }
568        _ => false,
569    }
570}

572/// Copy the tree, point git at the copy, and leave the original in place for the caller to
573/// verify against and remove.
574fn copy_and_repair(primary: &Path, from: &Path, to: &Path) -> Result<()> {
575    if let Some(parent) = to.parent() {
576        std::fs::create_dir_all(parent).map_err(|e| WorktreeError::io(parent, &e))?;
577    }
578    copy_tree(from, to)?;
579    git::run(
580        primary,
581        &["worktree".to_string(), "repair".to_string(), absolute(to)],
582    )?;
583    Ok(())
584}

586/// A faithful copy: directories, files with their permissions, symbolic links as links.
587/// Nothing is followed and nothing is skipped.
588pub fn copy_tree(from: &Path, to: &Path) -> Result<()> {
589    std::fs::create_dir(to).map_err(|e| WorktreeError::io(to, &e))?;
590    let mut stack = vec![(from.to_path_buf(), to.to_path_buf())];
591    while let Some((src, dst)) = stack.pop() {
592        let read = std::fs::read_dir(&src).map_err(|e| WorktreeError::io(&src, &e))?;
593        for entry in read {
594            let entry = entry.map_err(|e| WorktreeError::io(&src, &e))?;
595            let s = entry.path();
596            let d = dst.join(entry.file_name());
597            let meta = std::fs::symlink_metadata(&s).map_err(|e| WorktreeError::io(&s, &e))?;
598            if meta.file_type().is_symlink() {
599                let target = std::fs::read_link(&s).map_err(|e| WorktreeError::io(&s, &e))?;
600                #[cfg(unix)]
601                std::os::unix::fs::symlink(&target, &d).map_err(|e| WorktreeError::io(&d, &e))?;
602                #[cfg(not(unix))]
603                std::fs::copy(&s, &d).map_err(|e| WorktreeError::io(&d, &e))?;
604            } else if meta.is_dir() {
605                std::fs::create_dir(&d).map_err(|e| WorktreeError::io(&d, &e))?;
606                std::fs::set_permissions(&d, meta.permissions())
607                    .map_err(|e| WorktreeError::io(&d, &e))?;
608                stack.push((s, d));
609            } else {
610                std::fs::copy(&s, &d).map_err(|e| WorktreeError::io(&d, &e))?;
611                std::fs::set_permissions(&d, meta.permissions())
612                    .map_err(|e| WorktreeError::io(&d, &e))?;
613            }
614        }
615    }
616    Ok(())
