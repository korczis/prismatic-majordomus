1//! The repository environment: one typed snapshot of what this checkout is right now —
2//! the project, the repository, version control, the toolchains it declares, what the
3//! layer holds, the workflows a person can run, the provider projections and the local
4//! services — with a provenance entry for every fact in it.
5//!
6//! It is the canonical source for every surface that says any of those things. The direnv
7//! banner, `majordomus env`, `GET /api/v1/environment`, the MCP resource
8//! `majordomus://environment`, the Cockpit's overview and the documentation are
9//! renderings of this one value; none of them discovers anything of its own. That is the
10//! whole point: before it existed, "what branch is this and what is running" was a shell
11//! pipeline in whoever's dotfiles, and a second one in the next surface that wanted it.
12//!
13//! # Two resolutions, one model
14//!
15//! Building the index costs seconds, and the banner runs on every `cd`. So a snapshot is
16//! resolved in one of two modes ([`Resolution`]), and both produce the same type:
17//!
18//! - [`Resolution::Full`] reads everything, including the index, and writes the cache.
19//! - [`Resolution::Fast`] reads only what is cheap — one `git` call, one `just` call, a
20//!   few file reads — and takes the rest from the cache written by the last full
21//!   resolution. What no cache can supply is reported as unknown, never guessed and never
22//!   defaulted: a count that is not a count is worse than no count.
23//!
24//! The modes differ in what they may read, never in what they mean. `fast_and_full_agree`
25//! holds them to that.
26//!
27//! # What is canonical, and what is derived
28//!
29//! | Fact | Canonical source |
30//! |---|---|
31//! | project name, version, licence, summary | the crate manifest and [`crate::about`], at compile time |
32//! | repository root, layer sections | `.ai/manifest.yaml`, through [`crate::repository::Repository`] |
33//! | version control | one `git status --porcelain=v2 --branch` |
34//! | toolchains | the manifest that declares each one |
35//! | what the layer holds | the index, through the capability registry |
36//! | workflows | `just --dump --dump-format json` |
37//! | provider projections | the policy's `projections[]` |
38//! | services | the capability registry's routes and the shared server's lease |
39//!
40//! Nothing in this module writes to the repository outside `.ai/local/`, and nothing in it
41//! opens a socket to anything but the loopback address the lease names.

66/// The schema identity of [`RepositoryEnvironment`]. It travels in the snapshot itself, so

75/// How completely a snapshot was resolved. Carried in the snapshot because a consumer must
76/// be able to tell "there is no server running" from "nobody looked".
77#[derive(
78    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
79)]
80#[serde(rename_all = "snake_case")]
81pub enum Resolution {
82    /// Only what is cheap enough for a shell prompt; the rest from the cache.
83    Fast,
84    /// Everything, including the index. Writes the cache.
85    Full,
86}
87
88impl Resolution {
89    /// The word as serialised.
90    ///
91    /// ```
92    /// use majordomus_cli::environment::Resolution;
93    /// assert_eq!(Resolution::Fast.as_str(), "fast");
94    /// ```
95    pub fn as_str(self) -> &'static str {
96        match self {
97            Resolution::Fast => "fast",
98            Resolution::Full => "full",
99        }
100    }
101}

103/// Where a tier of the snapshot came from. A tier the active resolution cannot reach is
104/// [`TierState::Unavailable`] with the reason, and its values are absent rather than zero.
105#[derive(
106    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
107)]
108#[serde(rename_all = "snake_case")]
109pub enum TierState {
110    /// Read from its canonical source during this resolution.
111    Resolved,
112    /// Taken from the cache written by an earlier full resolution.
113    Cached,
114    /// Neither available; the values are absent and the reason is a diagnostic.
115    Unavailable,
116}
117
118impl TierState {
119    /// Did the tier produce values?
120    pub fn is_known(self) -> bool {
121        !matches!(self, TierState::Unavailable)
122    }
123
124    /// The word as serialised.
125    pub fn as_str(self) -> &'static str {
126        match self {
127            TierState::Resolved => "resolved",
128            TierState::Cached => "cached",
129            TierState::Unavailable => "unavailable",
130        }
131    }
132}

229/// What the layer holds, counted per kind, plus the registry the executable composes.
230/// Every number here comes from the index — the same objects every other surface serves —
231/// and never from counting files that match a pattern: a file that does not parse is not
232/// a rule, and a count that says otherwise is a lie a person acts on.
233#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
234pub struct LayerSummary {
235    /// Where these numbers came from.
236    pub state: TierState,
237    /// One entry per kind present, sorted by kind.
238    #[serde(default, skip_serializing_if = "Vec::is_empty")]
239    pub kinds: Vec<KindCount>,
240    /// How many objects the index holds.
241    #[serde(skip_serializing_if = "Option::is_none")]
242    pub objects: Option<usize>,
243    /// How many capabilities the registry holds.
244    #[serde(skip_serializing_if = "Option::is_none")]
245    pub capabilities: Option<usize>,
246    /// How many files the layer declared that did not become objects.
247    #[serde(skip_serializing_if = "Option::is_none")]
248    pub invalid: Option<usize>,
249    /// Which ones, and what decided it: the index's own error diagnostics, each naming the
250    /// file, the code and the constraint it failed. A count says something is wrong; only
251    /// this says what and where, and a reader that has the count and not the list has to go
252    /// looking through a log for the answer the index already produced.
253    #[serde(default, skip_serializing_if = "Vec::is_empty")]
254    pub refused: Vec<crate::model::Diagnostic>,
255    /// Whether the layer read cleanly.
256    #[serde(skip_serializing_if = "Option::is_none")]
257    pub degraded: Option<bool>,
258}

289// The kinds line ranks by count, largest first, and the workflow list reads by name. Both

531/// The snapshot: what this checkout is, right now.
532///
533/// Every surface that reports any of this renders this value. Collections are in a
534/// documented, stable order — kinds and workflows by name, services and providers in the
535/// order their canonical source declares them, diagnostics in the order they were found —
536/// so that two snapshots of the same repository serialise identically apart from
537/// [`RepositoryEnvironment::generated_at`].
538#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
539pub struct RepositoryEnvironment {
540    /// The contract this document follows, `majordomus/repository-environment/v1`.
541    pub schema: String,
542    /// When it was taken, RFC 3339 in UTC.
543    pub generated_at: String,
544    /// How completely it was resolved.
545    pub resolution: Resolution,
546    /// What the project is.
547    pub project: ProjectIdentity,
548    /// What the checkout is.
549    pub repository: RepositoryIdentity,
550    /// What version control says.
551    pub vcs: VcsState,
552    /// The toolchains the repository declares, sorted by id.
553    pub toolchains: Vec<ToolchainState>,
554    /// What the layer holds.
555    pub layer: LayerSummary,
556    /// The workflows a person can run.
557    pub workflows: WorkflowCatalogue,
558    /// The provider projections the policy declares, in the policy's order.
559    pub providers: Vec<ProviderState>,
560    /// The local services, in the order the service table declares them.
561    pub services: Vec<ServiceState>,
562    /// Everything that went wrong or is worth knowing, in the order it was found.
563    pub diagnostics: Vec<Diagnostic>,
564    /// Where every field came from.
565    pub provenance: Vec<FieldSource>,
566}
567
568impl RepositoryEnvironment {
569    /// The schema identity a document of this shape carries.
570    pub fn schema_id() -> String {
571        format!("{SCHEMA}/v{SCHEMA_VERSION}")
572    }
573
574    /// The provenance of one field, by its dotted name.
575    ///
576    /// ```
577    /// use majordomus_cli::environment::{FieldSource, RepositoryEnvironment};
578    /// # fn demo(env: &RepositoryEnvironment) {
579    /// let _ = env.explain("project.version");
580    /// # }
581    /// ```
582    pub fn explain(&self, field: &str) -> Option<&FieldSource> {
583        self.provenance.iter().find(|p| p.field == field)
584    }
585
586    /// A digest of everything a reader would notice, excluding the moment it was taken.
587    /// Two snapshots with the same digest say the same thing, which is how the banner
588    /// decides whether a person has already seen this.
589    pub fn digest(&self) -> String {
590        let mut stable = self.clone();
591        stable.generated_at = String::new();
592        // The resolution is not part of what a reader sees: the same repository resolved
593        // fast and full must digest the same, or every warm banner would look like news.
594        stable.resolution = Resolution::Full;
595        stable.provenance.clear();
596        let text = serde_json::to_string(&stable).unwrap_or_default();
597        crate::policy::sha256_hex(&text)
598    }
599
600    /// The digest that decides whether a banner is news: [`Self::digest`] without the
601    /// services.
602    ///
603    /// The entry file watches the server's lease, so a server coming up re-evaluates entry
604    /// twice more — once when it claims the lease, once when it publishes its address — and
605    /// a digest that counted the services made each of those a first look, drawing the whole
606    /// box again at a person who had just seen it. A server arriving or leaving is not a
607    /// different repository; the short form names the Cockpit when it answers.
608    ///
609    /// ```
610    /// use majordomus_cli::environment::RepositoryEnvironment;
611    /// # fn demo(env: &RepositoryEnvironment) {
612    /// // a server arriving or leaving is not a different repository, so clearing the
613    /// // services changes nothing this digest can see
614    /// let mut without = env.clone();
615    /// without.services.clear();
616    /// assert_eq!(env.news_digest(), without.news_digest());
617    /// // and with no services left, the two digests are the same value
618    /// assert_eq!(without.news_digest(), without.digest());
619    /// # }
620    /// ```
621    pub fn news_digest(&self) -> String {
622        let mut stable = self.clone();
623        stable.services.clear();
624        stable.digest()
625    }
626
627    /// The service with this id.
628    pub fn service(&self, id: &str) -> Option<&ServiceState> {
629        self.services.iter().find(|s| s.id == id)
630    }

632    /// The worst thing in the snapshot worth telling a person about, when there is one.
633    pub fn headline_diagnostic(&self) -> Option<&Diagnostic> {
634        self.diagnostics
635            .iter()
636            .max_by_key(|d| d.severity)
637            .filter(|d| d.severity != crate::model::Severity::Info)
638    }
