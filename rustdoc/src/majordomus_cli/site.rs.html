1//! The site's registry dataset: what GitHub Pages renders about the executable, derived
2//! from the registry, the index, the command line, the benchmark projection and the
3//! accepted baselines, and nothing else. The site has no content model of its own for
4//! any of it; its templates read `site/data/registry/registry.json` and lay it out, and
5//! the shell site generator reads `docs/generated/registry.json` for the ids it turns into
6//! routes.
7//!
8//! Deterministic: same repository tree and same executable, same bytes. No timestamps of
9//! its own (a baseline's `finished_at` is observed evidence, copied as committed), no
10//! absolute paths, no git state (the site's `source.json` carries the commit; it is
11//! written at build time by the site generator and is not compared by `generate --check`).
12

133/// A page of the crate's reference: the web surface that publishes it and the page within
134/// it, as the rustdoc check derives it from the crate's own inventory.
135///
136/// The mount is deliberately not here. Whether the reference is composed into a build, and
137/// at which mount, is the build's fact — `scripts/site-build` records every surface it
138/// composed in `data/build.json` — so a page links this one only when the build that
139/// renders it carries the surface, and at the mount it was composed at.
140///
141/// ```
142/// use majordomus_cli::site::ReferencePage;
143/// use majordomus_cli::web::discover::RUSTDOC;
144/// let page = ReferencePage {
145///     surface: RUSTDOC,
146///     module: "majordomus_cli::capability::builtin::quality".into(),
147///     path: "majordomus_cli/capability/builtin/quality/index.html".into(),
148/// };
149/// let v = serde_json::to_value(&page).unwrap();
150/// assert_eq!(v["surface"], "rustdoc");
151/// // relative to the surface: the mount is the build's, joined by the page that links it
152/// assert!(!v["path"].as_str().unwrap().starts_with('/'));
153/// ```
154#[derive(Debug, Clone, Serialize)]
155pub struct ReferencePage {
156    /// The web surface the page is published under: [`crate::web::discover::RUSTDOC`].
157    pub surface: &'static str,
158    /// The Rust module's crate path: `majordomus_cli::capability::builtin::quality`.
159    pub module: String,
160    /// The page, relative to the surface's root:
161    /// `majordomus_cli/capability/builtin/quality/index.html`.
162    pub path: String,
163}

431/// The crate's exported modules, by crate path, each with the page its reference documents
432/// it on: `quality::rustdoc::module_routes`, the derivation `majordomus quality rustdoc`
433/// holds the published tree to, read here rather than restated. It is read from the crate's
434/// source, so the dataset is the same on every machine whether or not the tree was built;
435/// a repository without the crate has no reference, and no module links one.
436fn reference_pages(root: &Path) -> Result<BTreeMap<String, ReferencePage>> {
437    let mut pages = BTreeMap::new();
438    let Some(dir) = crate::capability::builtin::quality::crate_dir(root) else {
439        return Ok(pages);
440    };
441    for route in crate::quality::rustdoc::module_routes(&dir)? {
442        // a module rustdoc documents twice (under two re-exports) is linked at the first
443        // page in canonical order, which is the order the routes are answered in
444        pages
445            .entry(route.path.clone())
446            .or_insert_with(|| ReferencePage {
447                surface: crate::web::discover::RUSTDOC,
448                module: route.path,
449                path: route.route,
450            });
451    }
452    Ok(pages)
453}

455/// Build the dataset.
456pub fn dataset(
457    ctx: &Context,
458    schema: &KindSchema,
459    policy: &LoadedPolicy,
460    repo: &Repository,
461) -> Result<SiteRegistry> {
462    let registry = &ctx.registry;
463    let index: &Index = &ctx.index;
464    let mut builtin: Vec<CapabilityView> = registry
465        .iter()
466        .filter(|c| matches!(c.provenance, Provenance::Builtin { .. }))
467        .map(|c| CapabilityView {
468            capability: c.clone(),
469            source_path: c.provenance.source_path(),
470        })
471        .collect();
472    crate::order::canonical(&mut builtin);
473    let reference = reference_pages(repo.root())?;
474    let mut modules: Vec<ModuleView> = registry
475        .modules()
476        .map(|m| {
477            let mine: Vec<&Capability> = registry
478                .iter()
479                .filter(|c| {
480                    c.module.as_str() == m.id.as_str()
481                        && matches!(c.provenance, Provenance::Builtin { .. })
482                })
483                .collect();
484            let rust_module = mine.first().and_then(|c| match &c.provenance {
485                Provenance::Builtin { module } => Some(module.as_str()),
486                Provenance::Declarative { .. } => None,
487            });
488            ModuleView {
489                id: m.id.to_string(),
490                title: m.title.clone(),
491                description: m.description.clone(),
492                stability: m.stability,
493                source: enum_name(m.source),
494                capabilities: registry
495                    .iter()
496                    .filter(|c| c.module.as_str() == m.id.as_str())
497                    .count(),
498                capability_ids: mine.iter().map(|c| c.id.to_string()).collect(),
499                source_path: (m.source != ModuleSource::Declarative)
500                    .then(|| mine.first().map(|c| c.provenance.source_path()))
501                    .flatten(),
502                rustdoc: rust_module.and_then(|module| reference.get(module).cloned()),
503            }
504        })
505        .collect();
506    crate::order::canonical(&mut modules);
507
508    let mut objects: Vec<ObjectView> = index
509        .objects
510        .iter()
511        .map(|o| ObjectView {
512            uri: o.uri.clone(),
513            kind: o.kind.clone(),
514            identity: o.identity.clone(),
515            title: o.title.clone(),
516            description: o.description.clone(),
517            path: o.provenance.path.clone(),
518            section: o.provenance.section.clone(),
519            source_class: o.provenance.source_class.clone(),
520            bytes: o.provenance.bytes,
521            tags: o.tags().into_iter().map(str::to_string).collect(),
522        })
523        .collect();
524    objects.sort_by(|a, b| a.uri.cmp(&b.uri));
525    let by_kind: BTreeMap<String, usize> = index
526        .kinds()
527        .into_iter()
528        .map(|(k, n)| (k.to_string(), n))
529        .collect();
530    let mut diagnostics: BTreeMap<String, usize> = BTreeMap::new();
531    for d in &index.diagnostics {
532        *diagnostics
533            .entry(format!("{:?}", d.severity).to_lowercase())
534            .or_default() += 1;
535    }
536
537    let mut kinds: Vec<KindView> = schema
538        .kinds()
539        .map(|(name, spec)| KindView {
540            name: name.clone(),
541            format: format!("{:?}", spec.format).to_lowercase(),
542            front_matter: format!("{:?}", spec.front_matter).to_lowercase(),
543            schema: spec.schema.clone(),
544            identity: spec.identity.clone(),
545            objects: by_kind.get(name).copied().unwrap_or(0),
546        })
547        .collect();
548    crate::order::canonical(&mut kinds);
549
550    let projections = policy
551        .policy
552        .projections
553        .iter()
554        .map(|p| ProjectionView {
555            provider: p.provider.clone(),
556            target: p.target.clone(),
557            mode: format!("{:?}", p.mode).to_lowercase(),
558            always_loaded: p.always_loaded,
559        })
560        .collect();
561
562    let mcp = McpView {
563        server_name: crate::mcp::protocol::SERVER_NAME,
564        protocol_versions: crate::mcp::protocol::PROTOCOL_VERSIONS,
565        tools: registry
566            .iter()
567            .filter_map(|c| {
568                let name = c.exposure.mcp.as_ref()?.tool.clone()?;
569                Some(ToolView {
570                    name,
571                    id: c.id.to_string(),
572                    title: c.title.clone(),
573                    description: c.description.clone(),
574                    kind: c.kind,
575                    read_only: c.kind.is_read_only(),
576                    input: c.input.name.clone().unwrap_or_else(|| "object".into()),
577                    output: c.output.name.clone().unwrap_or_else(|| "object".into()),
578                })
579            })
580            .collect(),
581        resources: registry
582            .iter()
583            .filter(|c| matches!(c.provenance, Provenance::Builtin { .. }))
584            .filter_map(|c| {
585                let r = c.exposure.mcp.as_ref()?.resource.as_ref()?;
586                Some(ResourceView {
587                    uri: r.uri.clone(),
588                    name: r.name.clone(),
589                    id: c.id.to_string(),
590                    title: c.title.clone(),
591                    kind: c.kind,
592                })
593            })
594            .collect(),
595        resource_template: "majordomus://<kind>/<identity>",
596    };
597
598    let http = HttpView {
599        routes: registry
600            .iter()
601            .filter_map(|c| {
602                let h = c.exposure.http.as_ref()?;
603                Some(RouteView {
604                    method: h.method.as_str().to_string(),
605                    path: h.path.clone(),
606                    id: c.id.to_string(),
607                    kind: c.kind,
608                })
609            })
610            .collect(),
611        infrastructure: openapi::infrastructure_routes(),
612        openapi_path: format!("{}/openapi.json", crate::generate::OUT_DIR),
613    };
614
615    let projection = BenchmarkProjection::from_context(ctx);
616    let coverage = Coverage::compute(ctx, &projection);
617    let targets = projection
618        .targets
619        .iter()
620        .map(|t| match &t.kind {
621            TargetKind::Capability {
622                id,
623                module,
624                transport,
625                case,
626                cache,
627                tool,
628                route,
629                ..
630            } => TargetView {
631                key: t.key.clone(),
632                kind: "capability",
633                transport: transport.name().to_string(),
634                id: Some(id.clone()),
635                module: module.clone(),
636                case: Some(case.clone()),
637                cached: !matches!(cache, crate::capability::CachePolicy::Disabled),
638                tool: tool.clone(),
639                route: route.as_ref().map(|(m, p)| format!("{m} {p}")),
640            },
641            TargetKind::System { target } => TargetView {
642                key: t.key.clone(),
643                kind: "system",
644                transport: target.transport().name().to_string(),
645                id: None,
646                module: "system".into(),
647                case: None,
648                cached: false,
649                tool: None,
650                route: None,
651            },

707/// Every `baseline.<platform>.json` beside the policy, by platform, as committed. A
708/// baseline that does not parse is an error: a stale evidence file is not silently
709/// dropped from the page that presents it.
710fn load_baselines(repo: &Repository, policy_path: &Path) -> Result<Vec<BaselineView>> {
711    let Some(dir) = policy_path.parent() else {
712        return Ok(Vec::new());
713    };
714    let Ok(entries) = std::fs::read_dir(dir) else {
715        return Ok(Vec::new());
716    };
717    let mut names: Vec<String> = entries
718        .filter_map(|e| e.ok())
719        .filter_map(|e| e.file_name().to_str().map(str::to_string))
720        .filter(|n| n.starts_with("baseline.") && n.ends_with(".json"))
721        .collect();
722    crate::order::canonical(&mut names);
723    let mut out = Vec::new();
724    for name in names {
725        let path = dir.join(&name);
726        let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
727        let doc: ResultDocument =
728            serde_json::from_str(&text).map_err(|e| Error::InvalidManifest {
729                path: path.clone(),
730                reason: format!("not a benchmark result document: {e}"),
731            })?;
732        let platform = name
733            .trim_start_matches("baseline.")
734            .trim_end_matches(".json")
735            .to_string();
736        out.push(BaselineView {
737            platform,
738            path: relative(repo.root(), &path),
739            profile: doc.profile.clone(),
740            finished_at: doc.finished_at.clone(),
741            provenance: doc.provenance.clone(),
742            results: doc
743                .results
744                .iter()
745                .map(|r| MeasurementView {
746                    key: r.key.clone(),
747                    cache_mode: match r.cache_mode {
748                        CacheMode::Uncached => "uncached",
749                        CacheMode::Cold => "cold",
750                        CacheMode::Warm => "warm",
751                        CacheMode::NotApplicable => "n/a",
752                    }
753                    .to_string(),
754                    samples: r.stats.samples,
755                    p50_us: r.stats.p50_us,
756                    p95_us: r.stats.p95_us,
757                    p99_us: r.stats.p99_us,
758                    handler_invocations: r.handler_invocations,
759                })
760                .collect(),
761        });
762    }
763    Ok(out)
764}

799/// A graph the site commits as data, as the artifact at `file` under the site's data
800/// directory: derived whole or refused ([`crate::graph::derive_whole`]), never a prefix of
801/// the repository, and carrying the provenance members every generated document carries (the
802/// graph is a value of the domain and has none of its own). The one place both site graphs
803/// are taken.
804fn graph_artifact(
805    ctx: &Context,
806    id: &str,
807    file: &str,
808    artifact: &str,
809    source: &str,
810) -> Result<crate::generate::Artifact> {
811    crate::graph::derive_whole(id, &ctx.registry, &ctx.index)
812        .map_err(|reason| Error::Protocol { reason })
813        .map(|graph| {
814            // a graph serializes to an object; its members and the provenance are one map
815            let mut document: serde_json::Map<String, serde_json::Value> =
816                serde_json::to_value(&graph)
817                    .ok()
818                    .and_then(|v| v.as_object().cloned())
819                    .unwrap_or_default();
820            document.insert(
821                "generated".into(),
822                serde_json::Value::String(crate::generate::json_banner(source)),
823            );
824            document.insert(
825                "generator".into(),
826                serde_json::json!({ "id": "majordomus-cli", "version": crate::VERSION }),
827            );
828            crate::generate::Artifact::verbatim(
829                format!("{}/{file}", crate::generate::SITE_DATA_DIR),
830                artifact,
831                crate::generate::ArtifactFormat::Json,
832                None,
833                source,
834                render_json(&serde_json::Value::Object(document)),
835            )
836        })
837}

839/// The Why catalogue and its graph, as the site's templates read them:
840/// `site/data/registry/why.json` and `site/data/registry/why-graph.json`.
841///
842/// Both are projections of [`crate::why::Catalogue`], which the context already built.
843/// Nothing here re-reads a file, and nothing here is a second opinion about what a moment
844/// says: the site generator writes one page per entry from this document, and the
845/// templates read it for every listing, filter, count, backlink and questionnaire.
846///
847/// The bodies are not in it: a page's prose is read from the file the record names in
848/// `source`, so the dataset stays the metadata every listing, filter, count and backlink
849/// needs and never becomes a second copy of the writing.
850///
851/// Deterministic and index-independent: the payload depends on the catalogue's own
852/// sources and on nothing else, so `scripts/derive` — which runs `generate`, then the
853/// site generator, then `generate` again over the tree that left — produces the same
854/// bytes in both passes.
855pub fn why_artifacts(ctx: &Context) -> Result<Vec<crate::generate::Artifact>> {
856    let by_cli = |path: &[&str]| -> Option<String> {
857        ctx.registry
858            .by_cli(&path.iter().map(|w| w.to_string()).collect::<Vec<_>>())
859            .map(|c| c.id.to_string())
860    };
861    let run = |path: &[&str], input: serde_json::Value| -> Result<serde_json::Value> {
862        let id = by_cli(path).ok_or_else(|| Error::Protocol {
863            reason: format!(
864                "no capability is exposed as `majordomus {}`",
865                path.join(" ")
866            ),
867        })?;
868        ctx.execute(&id, input).map_err(|e| Error::Protocol {
869            reason: e.to_string(),
870        })
871    };
872
873    // the catalogue, then one detail per moment: the site needs the derived relations and
874    // the body, and asking the capability for them is what keeps this from being a second
875    // reading of the same files
876    let catalogue = run(&["why", "list"], serde_json::json!({ "status": "any" }))?;
877    let mut moments = Vec::new();
878    for m in catalogue["moments"].as_array().into_iter().flatten() {
879        let id = m["id"].as_str().unwrap_or_default();
880        let mut detail = run(&["why", "show"], serde_json::json!({ "id": id }))?;
881        // The prose stays where it was authored. Every page the site generator writes
882        // takes the body from the file this record names in `source`, so the dataset is
883        // the metadata a template reads and not a second copy of the writing.
884        if let Some(o) = detail.as_object_mut() {
885            o.remove("body");
886        }
887        moments.push(detail);
888    }
889    let validation = run(&["why", "validate"], serde_json::json!({}))?;
890
891    // the same for the two taxonomies: their pages take the prose from their own files
892    let strip = |v: &serde_json::Value| -> Vec<serde_json::Value> {
893        v.as_array()
894            .into_iter()
895            .flatten()
896            .map(|e| {
897                let mut e = e.clone();
898                if let Some(o) = e.as_object_mut() {
899                    o.remove("body");
900                }
901                e
902            })
903            .collect()
904    };
905    let audiences = strip(&catalogue["audiences"]);
906    let areas = strip(&catalogue["areas"]);
907
908    // the questionnaire's index: every signal with the moment that owns it, flat, so a
909    // template renders it without joining two lists and a script never learns a mapping
910    let mut signals = Vec::new();
911    for m in &moments {
912        if m["status"] != "stable" {
913            continue;
914        }
915        for s in m["signals"].as_array().into_iter().flatten() {
916            signals.push(serde_json::json!({
917                "id": s["id"],
918                "text": s["text"],
919                "moment": m["id"],
920            }));
921        }
922    }
923
924    let document = serde_json::json!({
925        "schema": WHY_SCHEMA,
926        "generated": crate::generate::json_banner(WHY_SOURCE),
927        "generator": { "id": "majordomus-cli", "version": crate::VERSION },
928        "fingerprint": catalogue["fingerprint"],
929        "route": crate::why::ROUTE,
930        "counts": catalogue["counts"],
931        "facets": catalogue["facets"],
932        "audiences": audiences,
933        "areas": areas,
934        "signals": signals,
935        "moments": moments,
936        "valid": validation["valid"],
937    });
938
939    graph_artifact(ctx, "why", "why-graph.json", "site-why-graph", GRAPH_SOURCE).map(|graph| {
940        vec![
941            crate::generate::Artifact::verbatim(
942                format!("{}/why.json", crate::generate::SITE_DATA_DIR),
943                "site-why",
944                crate::generate::ArtifactFormat::Json,
945                Some(WHY_SCHEMA.to_string()),
946                WHY_SOURCE,
947                render_json(&document),
948            ),
949            graph,
950        ]
951    })
952}

955    let mut s = serde_json::to_string_pretty(v).unwrap_or_default();
956    s.push('\n');
957    s
958}

1038/// Where the site's evidence verdicts come from, named in the dataset beside them.
1039pub const EVIDENCE_PRODUCER: &str = "majordomus evidence show";
1040
1041/// The evidence verdict of every claim, as the site renders it beside the declared status:
1042/// the `evidence` section of `product.json`.
1043///
1044/// A projection of the `evidence show` report, never a second opinion about it. A claim is
1045/// `supported` when it declares `guaranteed` and the report has no finding against it. The
1046/// state is the report's own word, hyphenated the way the design vocabulary spells it, with
1047/// one deliberate weakening: `proven` is published as `inputs-unchanged`. This dataset is
1048/// committed, and committing it is itself a change after any recorded run, so a verdict it
1049/// carries can never be proof of the tree it is read from.
1050///
1051/// `available` is false when the ledger holds nothing or the index could not read the
1052/// whole matrix: the verdicts are then about nothing, and a page says `unknown`.
1053///
1054/// ```
1055/// use majordomus_cli::site::claim_evidence;
1056/// let report = serde_json::json!({
1057///     "subject": {"complete": true},
1058///     "ledger": {"path": "l.json", "present": true, "executions": 2, "newest": "2026-09-01"},
1059///     "claims": [
1060///         {"id": "a", "status": "guaranteed", "state": "proven"},
1061///         {"id": "b", "status": "guaranteed", "state": "not_run"},
1062///         {"id": "c", "status": "advisory", "state": "no_test"}
1063///     ],
1064///     "findings": [{"claim": "b"}]
1065/// });
1066/// let e = claim_evidence(&report);
1067/// assert_eq!(e["available"], true);
1068/// assert_eq!(e["declared"], 2);
1069/// assert_eq!(e["supported"], 1);
1070/// assert_eq!(e["claims"]["a"]["state"], "inputs-unchanged", "never published as proven");
1071/// assert_eq!(e["claims"]["b"]["supported"], false);
1072/// assert_eq!(e["claims"]["c"]["supported"], false, "only a guarantee is judged");
1073/// ```
1074pub fn claim_evidence(report: &serde_json::Value) -> serde_json::Value {
1075    let unsupported: std::collections::BTreeSet<&str> = report["findings"]
1076        .as_array()
1077        .into_iter()
1078        .flatten()
1079        .filter_map(|f| f["claim"].as_str())
1080        .collect();
1081    let complete = report["subject"]["complete"].as_bool() == Some(true);
1082    let present = report["ledger"]["present"].as_bool() == Some(true);
1083    let mut claims = serde_json::Map::new();
1084    let (mut declared, mut supported) = (0usize, 0usize);
1085    for c in report["claims"].as_array().into_iter().flatten() {
1086        let Some(id) = c["id"].as_str() else { continue };
1087        let guaranteed = c["status"].as_str() == Some("guaranteed");
1088        let is_supported = complete && guaranteed && !unsupported.contains(id);
1089        declared += usize::from(guaranteed);
1090        supported += usize::from(is_supported);
1091        let state = match c["state"].as_str().unwrap_or("unknown") {
1092            "proven" => "inputs-unchanged".to_string(),
1093            s => s.replace('_', "-"),
1094        };
1095        claims.insert(
1096            id.to_string(),
1097            serde_json::json!({
1098                "status": c["status"],
1099                "state": state,
1100                "label": state.replace('-', " "),
1101                "supported": is_supported,
1102            }),
1103        );
1104    }
1105    serde_json::json!({
1106        "producer": EVIDENCE_PRODUCER,
1107        "available": complete && present,
1108        "complete": complete,
1109        "ledger": {
1110            "path": report["ledger"]["path"],
1111            "executions": report["ledger"]["executions"],
1112            "newest": report["ledger"]["newest"],
1113        },
1114        "declared": declared,
1115        "supported": supported,
1116        "claims": claims,
1117    })
1118}

1120/// The fields of `v` an allow-list names, and nothing else: what reaches the published site is

1243        // what every checkout has, as web.json lists it: a report a producer wrote here is
1244        // served here and counted nowhere, or the count would follow who ran the suite
1245        "web_surfaces": crate::generate::surfaces_every_checkout_has(&ctx.web).count(),
1246    });
1247
1248    // The doctrine the product rests on: every rule a public feature names, once, carrying the
1249    // class and the enforcement the rule itself declares. Deduplicated here and not in a
1250    // template, because which rules the product rests on is a fact of the model, and a page
1251    // that had to work it out would be a page inferring domain structure.
1252    let mut by_rule: BTreeMap<String, serde_json::Value> = BTreeMap::new();
1253    for f in &features {
1254        if f["status"].as_str() == Some("draft") {
1255            continue;
1256        }
1257        for r in f["rule_refs"].as_array().into_iter().flatten() {
1258            if let Some(id) = r["id"].as_str() {
1259                by_rule.entry(id.to_string()).or_insert_with(|| r.clone());
1260            }
1261        }
1262    }
1263    let rules: Vec<serde_json::Value> = by_rule.into_values().collect();
1264
1265    let cockpit_areas: Vec<serde_json::Value> = crate::cockpit::nav::areas()
1266        .iter()
1267        .map(|a| serde_json::json!({ "id": a.id, "title": a.label, "route": a.href }))
1268        .collect();
1269
1270    // What the recorded evidence says about each claim, from the one derivation of it
1271    // (`majordomus evidence show`), so that a page can put the declared status beside the
1272    // verdict and never render the first as the second.
1273    let evidence = claim_evidence(&run(&["evidence", "show"], serde_json::json!({}))?);
1274
1275    let document = serde_json::json!({
1276        "schema": PRODUCT_SCHEMA,
1277        "generated": crate::generate::json_banner(PRODUCT_SOURCE),
1278        "generator": { "id": "majordomus-cli", "version": crate::VERSION },
1279        "fingerprint": list["fingerprint"],
1280        "route": crate::product::ROUTE,
1281        "domain_route": crate::product::DOMAIN_ROUTE,
1282        "counts": list["counts"],
1283        "surfaces": list["surfaces"],
1284        "domains": domains,
1285        "features": features,
1286        "matrix": { "rows": matrix["rows"], "modules": matrix["modules"], "commands": matrix["commands"], "kinds": matrix["kinds"] },
1287        "providers": providers["providers"],
1288        "rules": rules,
1289        "cockpit_areas": cockpit_areas,
1290        "telemetry": telemetry,
1291        "evidence": evidence,
1292        "valid": validation["valid"],
1293    });
1294
1295    graph_artifact(
1296        ctx,
1297        "product",
1298        "product-graph.json",
1299        "site-product-graph",
1300        "the features, what they are made of, and the interfaces that follow, as the derived `product` graph",
1301    )
1302    .map(|graph| {
1303        vec![
1304            crate::generate::Artifact::verbatim(
1305                format!("{}/product.json", crate::generate::SITE_DATA_DIR),
1306                "site-product",
1307                crate::generate::ArtifactFormat::Json,
1308                Some(PRODUCT_SCHEMA.to_string()),
1309                PRODUCT_SOURCE,
1310                render_json(&document),
1311            ),
1312            graph,
1313        ]
1314    })

1340    /// A repository holding the crate at the path this repository's crate lives, with the
1341    /// given sources under `src/`.
1342    fn repository_with(files: &[(&str, &str)]) -> tempfile::TempDir {
1343        let root = tempfile::tempdir().unwrap();
1344        let dir = root.path().join(crate::capability::model::CRATE_DIR);
1345        std::fs::create_dir_all(dir.join("src")).unwrap();
1346        std::fs::write(
1347            dir.join("Cargo.toml"),
1348            "[package]\nname = \"majordomus-cli\"\nversion = \"0.1.0\"\n",
1349        )
1350        .unwrap();
1351        for (path, text) in files {
1352            let p = dir.join("src").join(path);
1353            std::fs::create_dir_all(p.parent().unwrap()).unwrap();
1354            std::fs::write(p, text).unwrap();
1355        }
1356        root
1357    }

1359    #[test]
1360    fn a_module_links_the_reference_page_only_when_the_crate_exports_it() {
1361        let repo = repository_with(&[
1362            ("lib.rs", "//! Root.\npub mod capability;\n"),
1363            ("capability/mod.rs", "//! Capabilities.\npub mod builtin;\n"),
1364            (
1365                "capability/builtin/mod.rs",
1366                "//! Builtin.\npub mod open;\npub(crate) mod crated;\nmod closed;\n",
1367            ),
1368            ("capability/builtin/open.rs", "//! Open.\n"),
1369            ("capability/builtin/crated.rs", "//! Crate-visible.\n"),
1370            ("capability/builtin/closed.rs", "//! Private.\n"),
1371        ]);
1372        let pages = reference_pages(repo.path()).unwrap();
1373        // a module the crate exports is linked at the page rustdoc documents it on, under the
1374        // surface the topology publishes the reference as
1375        let open = &pages["majordomus_cli::capability::builtin::open"];
1376        assert_eq!(
1377            open.path,
1378            "majordomus_cli/capability/builtin/open/index.html"
1379        );
1380        assert_eq!(open.module, "majordomus_cli::capability::builtin::open");
1381        assert_eq!(open.surface, crate::web::discover::RUSTDOC);
1382        // one rustdoc gives no page is never linked: `pub(crate)` and private alike
1383        assert!(!pages.contains_key("majordomus_cli::capability::builtin::crated"));
1384        assert!(!pages.contains_key("majordomus_cli::capability::builtin::closed"));
1385
1386        // a repository without the crate has no reference and links nothing
1387        let bare = tempfile::tempdir().unwrap();
1388        assert!(reference_pages(bare.path()).unwrap().is_empty());
1389    }
