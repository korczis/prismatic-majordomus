1//! `majordomus design kit`: the repository's user interface against a kit profile, through
2//! the registry's `design.kit`, and `design kit adopt`, through `design.kit.adopt`
3//! (I4334, ADR 0230).
4//!
5//! The command renders the capability's answer and chooses the exit code from it; the
6//! enumeration, the judgement and the verdict are the capability's, so the terminal, the
7//! `--format json` document, MCP and HTTP cannot disagree about whether the kit holds.
8//!
9//! One path does not go through the registry: a `--repo` that Majordomus does not
10//! supervise has no layer to load and so no registry to ask, and the owner's mandate covers
11//! those repositories too. There the command calls the same judgement the capability calls,
12//! [`crate::design::kit::inspect`], with the distribution's share directory, and answers
13//! with the same document.
14

26/// The exit code when a finding stands under `--check`.
27pub const EXIT_FINDINGS: u8 = 10;
28/// The exit code of a policy that cannot be read.
29pub const EXIT_BAD_POLICY: u8 = 2;

31/// Run `majordomus design`: `kit` here, `licence` in [`super::design_licence`].
32pub fn run(args: DesignArgs) -> Result<u8> {
33    if matches!(args.command, DesignCommand::Licence(_)) {
34        return super::design_licence::run(args);
35    }
36    run_to(args, &mut std::io::stdout().lock())
37}

39fn run_to(args: DesignArgs, w: &mut dyn Write) -> Result<u8> {
40    let DesignCommand::Kit(kit_args) = args.command else {
41        return Err(Error::Protocol {
42            reason: "design kit was handed another design subcommand".to_string(),
43        });
44    };
45    // Each arm folds its steps into one result, so a failure anywhere leaves by one exit.
46    let (value, text, code) = match kit_args.action {
47        None => judged(&args.repo, kit_args.evidence.as_deref()).and_then(|value| {
48            answer::<KitReport>("design.kit", &value)
49                .map(|report| (render(&report), exit_code(&report, kit_args.check)))
50                .map(|(text, code)| (value, text, code))
51        }),
52        Some(KitAction::Adopt { profile, dry_run }) => App::load(&args.repo)
53            .and_then(|app| {
54                let input = json!({ "profile": profile.name(), "dry_run": dry_run });
55                execute(&app.context, &["design", "kit", "adopt"], input)
56            })
57            .and_then(|value| {
58                answer::<AdoptReport>("design.kit.adopt", &value)
59                    .map(|report| render_adoption(&report))
60                    .map(|text| (value, text, 0))
61            }),
62    }?;
63    emit(w, args.format, &value, text).map(|()| code)
64}

66/// The judgement, from the registry when the repository is supervised and from the same
67/// function the capability calls when it is not. `evidence` is the kit-evidence/v1 file
68/// `--evidence` names, handed over as the caller named it.
69fn judged(repo: &RepoArgs, evidence: Option<&Path>) -> Result<Value> {
70    match App::load(repo) {
71        Ok(app) => {
72            let input = match evidence {
73                Some(e) => json!({ "evidence": e.to_string_lossy() }),
74                None => json!({}),
75            };
76            execute(&app.context, &["design", "kit"], input)
77        }
78        Err(Error::RepositoryNotFound { start }) => {
79            let root = toplevel(&start);
80            let share = crate::share::Share::locate(repo.share.as_deref(), &root)?;
81            let report =
82                kit::inspect(&root, share.dir(), evidence).map_err(|reason| Error::Protocol {
83                    reason: format!("design kit: {reason}"),
84                })?;
85            serde_json::to_value(&report).map_err(protocol)
86        }
87        Err(other) => Err(other),
88    }
89}

91/// The root of the git work tree holding `start`, or `start` itself outside one.
92fn toplevel(start: &Path) -> PathBuf {
93    let start = start.canonicalize().unwrap_or_else(|_| start.to_path_buf());
94    start
95        .ancestors()
96        .find(|d| d.join(".git").exists())
97        .map(Path::to_path_buf)
98        .unwrap_or(start)
99}

101fn execute(ctx: &crate::capability::Context, words: &[&str], input: Value) -> Result<Value> {
102    let words: Vec<String> = words.iter().map(|w| w.to_string()).collect();
103    let id = ctx
104        .registry
105        .by_cli(&words)
106        .map(|c| c.id.as_str())
107        .ok_or_else(|| Error::Protocol {
108            reason: format!(
109                "no capability is exposed as `majordomus {}`",
110                words.join(" ")
111            ),
112        })?;
113    ctx.execute(id, input).map_err(refusal)
114}

116/// A capability's error as this command's: each refusal keeps its exit code, and anything
117/// else is a protocol error carrying the capability's words.
118fn refusal(e: CapabilityError) -> Error {
119    match e {
120        CapabilityError::Refused(reason) => Error::Refused { code: 15, reason },
121        CapabilityError::NotFound(reason) => Error::Refused { code: 12, reason },
122        CapabilityError::InvalidInput(reason) => Error::Refused { code: 2, reason },
123        other => protocol(other),
124    }
125}

127/// A capability's answer as the report this command renders, or a protocol error naming
128/// the capability whose answer it cannot read.
129fn answer<T: serde::de::DeserializeOwned>(capability: &str, value: &Value) -> Result<T> {
130    serde_json::from_value(value.clone()).map_err(|e| {
131        protocol(format!(
132            "{capability} answered with something this command cannot read: {e}"
133        ))
134    })
135}
136
137fn protocol(reason: impl std::fmt::Display) -> Error {
138    Error::Protocol {
139        reason: reason.to_string(),
140    }
141}

143fn emit(w: &mut dyn Write, format: OutputFormat, value: &Value, text: String) -> Result<()> {
144    let out = match format {
145        OutputFormat::Json => format!("{value:#}"),
146        OutputFormat::Text => text,
147    };
148    writeln!(w, "{out}").map_err(Error::Transport)
149}

151/// The exit code a report earns: a bad policy always refuses, a finding refuses only under
152/// `--check`, and no interface is a pass.
153fn exit_code(report: &KitReport, check: bool) -> u8 {
154    match report.verdict {
155        KitVerdict::BadPolicy => EXIT_BAD_POLICY,
156        KitVerdict::Findings if check => EXIT_FINDINGS,
157        _ => 0,
158    }
159}

161/// The report as a person reads it: every finding with its remedy, every exemption with
162/// its reason, then the verdict.
163fn render(report: &KitReport) -> String {
164    let mut out = Vec::new();
165    let policy = if report.policy_declared {
166        "declared in .ai/repo/policy.yaml"
167    } else {
168        "no ui.kit; the default profile"
169    };
170    out.push(format!(
171        "design kit: {} ({policy}), {}",
172        report.profile, report.profile_source
173    ));
174    for e in &report.policy_errors {
175        out.push(format!("POLICY  {e}"));
176    }
177    for f in &report.findings {
178        let path = if f.path.is_empty() {
179            "."
180        } else {
181            f.path.as_str()
182        };
183        out.push(format!("FAIL  {:?}  {path}  {}", f.signal, f.evidence));
184        out.push(format!("      remedy: {}", f.remedy));
185    }
186    for x in &report.exempted {
187        out.push(format!(
188            "EXEMPT  {:?}  {}  {}  because: {}",
189            x.finding.signal, x.finding.path, x.finding.evidence, x.because
190        ));
191    }
192    for n in &report.reported {
193        out.push(format!("NOTE  {n}"));
194    }
195    for n in &report.notices {
196        out.push(format!("NOTICE  {n}"));
197    }
198    if let Some(l) = &report.licence {
199        out.push(format!(
200            "licence: {:?}, {:?} ({}){}{}",
201            l.variant,
202            l.state,
203            l.evidence,
204            if l.markers.is_empty() {
205                String::new()
206            } else {
207                format!("; licensed markers: {}", l.markers.join("; "))
208            },
209            l.reason
210                .as_ref()
211                .map(|r| format!("; {r}"))
212                .unwrap_or_default()
213        ));
214    }
215    for s in &report.surfaces {
216        let dir = if s.dir.is_empty() {
217            "."
218        } else {
219            s.dir.as_str()
220        };
221        out.push(format!(
222            "surface {dir} ({:?}): {}",
223            s.kind,
224            if s.usage.is_empty() {
225                "no usage marker".to_string()
226            } else {
227                s.usage.join("; ")
228            }
229        ));
230    }
231    if report.ui {
232        out.push(format!(
233            "interface: {}; stacks: {}; layouts: {}",
234            report.ui_evidence.join("; "),
235            if report.stacks.is_empty() {
236                "none".to_string()
237            } else {
238                report.stacks.join(" ")
239            },
240            if report.layouts.is_empty() {
241                "none".to_string()
242            } else {
243                report.layouts.join(" ")
244            }
245        ));
246    }
247    out.push(match report.verdict {
248        KitVerdict::NoUi => "design kit: no user interface; the rule holds vacuously".to_string(),
249        KitVerdict::Clean => format!(
250            "design kit: clean ({} exemption(s) covering {} finding(s))",
251            report.exemptions.len(),
252            report.exempted.len()
253        ),
254        KitVerdict::Findings => format!("design kit: {} finding(s)", report.findings.len()),
255        KitVerdict::BadPolicy => format!(
256            "design kit: the policy cannot be read ({} problem(s)); nothing was judged",
257            report.policy_errors.len()
258        ),
259    });
260    out.join("\n")
261}

263/// The adoption as a person reads it.
264fn render_adoption(report: &AdoptReport) -> String {
265    let mut out = Vec::new();
266    for w in &report.writes {
267        let action = match w.action {
268            AdoptAction::Created => "create",
269            AdoptAction::Updated => "update",
270            AdoptAction::Unchanged => "unchanged",
271            AdoptAction::Kept => "kept",
272        };
273        out.push(format!("{action:<10} {}  {}", w.path, w.detail));
274    }
275    for n in &report.notes {
276        out.push(format!("note       {n}"));
277    }
278    out.push(format!(
279        "design kit adopt: {} ({}){}",
280        report.profile,
281        report.rule,
282        if report.dry_run {
283            ", dry run: nothing written"
284        } else {
285            ""
286        }
287    ));
288    out.join("\n")
289}

597    #[test]
598    fn what_the_command_cannot_read_is_a_protocol_error_and_each_refusal_keeps_its_code() {
599        // `design licence` is dispatched before `run_to`; handed to it, it is refused
600        let a = args(&["majordomus", "design", "licence"]);
601        let err = run_to(a, &mut Vec::new()).unwrap_err();
602        assert!(
603            matches!(&err, Error::Protocol { reason } if reason.contains("another design subcommand")),
604            "{err:?}"
605        );
606        // an answer that is not the capability's report names the capability
607        for capability in ["design.kit", "design.kit.adopt"] {
608            let err = answer::<KitReport>(capability, &json!({ "verdict": 3 })).unwrap_err();
609            assert!(
610                matches!(&err, Error::Protocol { reason }
611                    if reason.starts_with(&format!("{capability} answered with something"))),
612                "{err:?}"
613            );
614        }
615        // each refusal keeps its exit code; anything else is the capability's own words
616        for (e, code) in [
617            (CapabilityError::Refused("r".into()), 15),
618            (CapabilityError::NotFound("n".into()), 12),
619            (CapabilityError::InvalidInput("i".into()), 2),
620        ] {
621            assert!(
622                matches!(refusal(e), Error::Refused { code: c, .. } if c == code),
623                "{code}"
624            );
625        }
626        assert!(matches!(
627            refusal(CapabilityError::Internal("broke".into())),
628            Error::Protocol { reason } if reason == "internal: broke"
629        ));
630        // words no capability is exposed as are refused before anything runs
631        let repo = crate::synthetic::SyntheticRepository::small().unwrap();

728                "kit",

733                "--format",

735            ]);
