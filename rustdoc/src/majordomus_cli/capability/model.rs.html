1//! The descriptor: everything a projection may say about a capability, and nothing else.
2//!
3//! # What this module owns
4//!
5//! [`Capability`] is the value the whole executable is a projection of. It carries identity
6//! ([`CapabilityId`], [`ModuleId`]), what the thing is ([`CapabilityKind`]), the canonical
7//! schemas of its input and output, where it came from ([`Provenance`]), where it is
8//! projected ([`Exposure`]), where it means anything ([`Availability`]), who it is for
9//! ([`Visibility`]), where it stands ([`Stability`]), and the two operational policies a
10//! call is subject to ([`CachePolicy`], [`BenchmarkPolicy`]).
11//!
12//! # What it deliberately does not own
13//!
14//! There is no handler here and no transport type. A descriptor is plain data: it
15//! serialises, it round-trips, and it can be read by something that cannot execute
16//! anything — which is what lets the generated reference, the site dataset and the OpenAPI
17//! document be built from the same value the executor dispatches on. Execution lives in
18//! [`super::handler`]; the collection and its invariants live in [`super::registry`].
19//!
20//! # Declared against classified
21//!
22//! Two fields are *classified* rather than declared: [`Availability::classify`] and
23//! [`Visibility::classify`] compute their values from the kind and the exposures the
24//! declaration already carries. Asking each declaration to restate them would be the same
25//! knowledge written twice, and the second copy is the one that goes wrong. Both matches are
26//! exhaustive with no fallback arm on purpose, so a new kind or a new transport is a compile
27//! error here rather than a silent default in a page that then links to nothing.
28//!
29//! # Invariants
30//!
31//! Nothing in this module enforces anything: a descriptor can be built wrong and the
32//! registry is what refuses it. What lives here is the vocabulary each check is written
33//! against — the grammar of an id, the shape of a route, the states a policy may be in —
34//! each with its own validator returning the reason it failed rather than a boolean.
35//!
36//! ```
37//! use majordomus_cli::capability::{
38//!     Availability, CapabilityId, CapabilityKind, CliExposure, Exposure, Visibility,
39//! };
40//!
41//! let id = CapabilityId::parse("repository.info").unwrap();
42//! assert_eq!(id.namespace(), "repository");
43//!
44//! // a capability offered only on the command line is developer-facing, and needs a
45//! // process to answer at all — neither of which its declaration had to say
46//! let exposure = Exposure {
47//!     cli: Some(CliExposure { path: vec!["scope".into()] }),
48//!     ..Default::default()
49//! };
50//! assert_eq!(Visibility::classify(&exposure), Visibility::Developer);
51//! assert_eq!(
52//!     Availability::classify(CapabilityKind::Query, &exposure),
53//!     Availability::Runtime
54//! );
55//! ```

64/// A stable, globally meaningful identity: a namespace, a dot, and a local part.
65/// `repository.info` and `objects.get` for executables; `<kind>.<identity>` for a
66/// declarative object (`rule.majordomus.scope-integrity@1`, `document.docs/CLI.md`,
67/// `policy..ai/repo/policy.yaml`).
68///
69/// Grammar: the namespace matches `[a-z][a-z0-9_-]*`; the local part is non-empty and
70/// carries no whitespace or control character, any other Unicode included, because it is
71/// opaque: a path, a versioned identity, or a name, as the kind's identity rule produced it.
72#[derive(
73    Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
74)]
75#[serde(transparent)]
76pub struct CapabilityId(String);
77
78impl CapabilityId {
79    /// Parse and validate.
80    ///
81    /// ```
82    /// use majordomus_cli::capability::CapabilityId;
83    /// let id = CapabilityId::parse("rule.majordomus.scope-integrity@1").unwrap();
84    /// assert_eq!(id.namespace(), "rule");
85    /// assert!(CapabilityId::parse("Repository.info").is_err(), "namespace is lowercase");
86    /// assert!(CapabilityId::parse("repository").is_err(), "a dot is required");
87    /// assert!(CapabilityId::parse("document.docs/Příručka.md").is_ok(), "the local part is opaque");
88    /// ```
89    pub fn parse(text: &str) -> Result<Self, String> {
90        let bytes = text.as_bytes();
91        if bytes.is_empty() {
92            return Err("empty".into());
93        }
94        if text.chars().any(|c| c.is_whitespace() || c.is_control()) {
95            return Err("carries whitespace or a control character".into());
96        }
97        let Some((namespace, local)) = text.split_once('.') else {
98            return Err("needs a namespace, a dot, and a local part".into());
99        };
100        let mut chars = namespace.chars();
101        let ok_namespace = chars.next().is_some_and(|c| c.is_ascii_lowercase())
102            && chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_' || c == '-');
103        if !ok_namespace {
104            return Err(format!("namespace '{namespace}' is not [a-z][a-z0-9_-]*"));
105        }
106        if local.is_empty() {
107            return Err("the local part after the dot is empty".into());
108        }
109        Ok(CapabilityId(text.to_string()))
110    }
111
112    /// For descriptors written in code (the `capability!` macro); validated when the
113    /// registry is built, never before.
114    pub fn unchecked(text: &str) -> Self {
115        CapabilityId(text.to_string())
116    }
117
118    /// The id as text.
119    pub fn as_str(&self) -> &str {
120        &self.0
121    }
122
123    /// The first segment.
124    pub fn namespace(&self) -> &str {
125        self.0.split('.').next().unwrap_or_default()
126    }
127}
128
129impl fmt::Display for CapabilityId {
130    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
131        f.write_str(&self.0)
132    }
133}

135/// A module identity: the namespace of every capability the module composes, matching
136/// `[a-z][a-z0-9_-]*`. Builtin modules declare theirs in `module!`; a declarative
137/// object's module is its kind.
138#[derive(
139    Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize, JsonSchema,
140)]
141#[serde(transparent)]
142pub struct ModuleId(String);
143
144impl ModuleId {
145    /// Parse and validate.
146    ///
147    /// ```
148    /// use majordomus_cli::capability::ModuleId;
149    /// assert!(ModuleId::parse("repository").is_ok());
150    /// assert!(ModuleId::parse("Repository").is_err());
151    /// assert!(ModuleId::parse("").is_err());
152    /// ```
153    pub fn parse(text: &str) -> Result<Self, String> {
154        let mut chars = text.chars();
155        let ok = chars.next().is_some_and(|c| c.is_ascii_lowercase())
156            && chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_' || c == '-');

158            Ok(ModuleId(text.to_string()))
159        } else {
160            Err(format!("module id '{text}' is not [a-z][a-z0-9_-]*"))
161        }
162    }

165    pub fn unchecked(text: &str) -> Self {
166        ModuleId(text.to_string())
167    }
168
169    /// The id as text.
170    pub fn as_str(&self) -> &str {
171        &self.0
172    }
173}

181/// Why an executable capability is not benchmarked. Typed, so that a waiver is a
182/// reviewable statement and never a convenience; `not_executable` is the registry's own
183/// reason for resources and is never written by hand.
184#[derive(
185    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
186)]
187#[serde(rename_all = "snake_case")]
188pub enum WaiverReason {
189    /// A resource: read, never executed; nothing to time but `objects.get`, which is.
190    NotExecutable,
191    /// The capability changes something outside this process and cannot be run in a loop.
192    Destructive,
193    /// The capability talks to something the benchmark host cannot provide.
194    ExternalDependency,
195    /// The capability starts, or answers about, work that exists only while it is running.
196    /// A benchmark host cannot stage an execution to read, and running the operation in a
197    /// loop would measure the work rather than the operation.
198    TransientState,
199    /// The capability answers about what this repository has *published*, so its input is a
200    /// release: a version tag, the commit it points at, and the projection committed there.
201    /// A benchmark fixture is a repository that has published nothing, and the honest answer
202    /// in one is a refusal — so the case would time an error path and call it coverage.
203    /// Distinct from [`WaiverReason::ExternalDependency`]: nothing outside this process is
204    /// consulted, and distinct from [`WaiverReason::TransientState`]: the history is durable,
205    /// it simply does not exist yet where the benchmark runs.
206    PublishedHistory,
207}

209/// What a repository must hold before a conditional benchmark requirement applies to it.
210/// Typed, like a waiver's reason, so that the condition is a reviewable statement on the
211/// descriptor and never a provider quietly returning nothing.
212///
213/// ```
214/// use majordomus_cli::capability::BenchmarkPrecondition;
215/// let p: BenchmarkPrecondition = serde_json::from_str(r#""plan_holds_an_issue""#).unwrap();
216/// assert_eq!(p, BenchmarkPrecondition::PlanHoldsAnIssue);
217/// ```
218#[derive(
219    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
220)]
221#[serde(rename_all = "snake_case")]
222pub enum BenchmarkPrecondition {
223    /// The plan holds at least one issue. A command whose every honest case is a refusal
224    /// chosen from an issue's own status has no such refusal in a plan with no issue: the
225    /// only answer there is a lookup miss, which times an error path and is not the
226    /// operation.
227    PlanHoldsAnIssue,
228}
229
230impl BenchmarkPrecondition {
231    /// Does the repository this index describes hold what the precondition names?
232    ///
233    /// It is answered from the index, so the same capability can be a requirement in one
234    /// repository and inapplicable in another without any provider returning nothing.
235    ///
236    /// ```
237    /// use majordomus_cli::capability::BenchmarkPrecondition;
238    /// use majordomus_cli::synthetic::SyntheticRepository;
239    ///
240    /// let repo = SyntheticRepository::small().unwrap();
241    /// let p = BenchmarkPrecondition::PlanHoldsAnIssue;
242    ///
243    /// // a repository whose plan holds nothing: the requirement does not apply here
244    /// assert!(!p.holds(&repo.index().unwrap()));
245    ///
246    /// // declare where issues live, as `majordomus init` does, and put one issue in the plan
247    /// let sources = repo.root().join(".ai/repo/knowledge/sources.yaml");
248    /// let mut text = std::fs::read_to_string(&sources).unwrap();
249    /// text.push_str("  - id: issue\n    kind: issue\n    discovery: vcs\n    pathspec: ':(glob).ai/repo/project/issues/*.yaml'\n    required: false\n");
250    /// std::fs::write(&sources, text).unwrap();
251    /// let issues = repo.root().join(".ai/repo/project/issues");
252    /// std::fs::create_dir_all(&issues).unwrap();
253    /// std::fs::write(
254    ///     issues.join("I0001.yaml"),
255    ///     "id: I0001\nmilestone: M000\ntitle: A first issue\nslug: a-first-issue\n\
256    ///      priority: p1\nprofile: implementation\nobjective: \"Do the first thing.\"\n\
257    ///      scope:\n  - lib\nacceptance_criteria:\n  - It is done\nvalidation:\n  - \"true\"\n\
258    ///      evidence_required:\n  - proof\n",
259    /// )
260    /// .unwrap();
261    ///
262    /// // the same precondition now holds, answered from the index alone
263    /// assert!(p.holds(&repo.index().unwrap()));
264    /// ```
265    pub fn holds(self, index: &crate::index::Index) -> bool {
266        match self {
267            Self::PlanHoldsAnIssue => index.objects.iter().any(|o| o.kind == crate::plan::ISSUE),
268        }
269    }
270}

272/// Whether the capability is a benchmark target. `Required` is the default and the norm:
273/// every executable capability is timed directly and through every transport it is
274/// exposed on, with the cases its input type provides.
275#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
276#[serde(tag = "policy", rename_all = "snake_case")]
277pub enum BenchmarkPolicy {
278    /// Timed directly and through every exposure; coverage fails without a case.
279    Required,
280    /// Required exactly like `Required` in a repository where the precondition holds, and
281    /// reported as inapplicable — never covered, never waived — where it does not. Not a
282    /// waiver: with the precondition met and no case, the requirement is missing.
283    RequiredWhen {
284        /// What the repository must hold.
285        precondition: BenchmarkPrecondition,
286    },
287    /// Not timed, for the typed reason; coverage reports it as waived, never as covered.
288    Waived {
289        /// Why.
290        reason: WaiverReason,
291    },
292}
293
294impl BenchmarkPolicy {
295    /// Is the capability a benchmark target at all — required, unconditionally or on a
296    /// precondition? Only a waiver says no.
297    ///
298    /// A conditional requirement is a target even where its precondition does not hold:
299    /// being inapplicable to one repository is not the same as being waived, and coverage
300    /// only decides applicability after this has said the capability is timed at all.
301    ///
302    /// ```
303    /// use majordomus_cli::capability::{BenchmarkPolicy, BenchmarkPrecondition, WaiverReason};
304    ///
305    /// assert!(BenchmarkPolicy::Required.is_target());
306    ///
307    /// // conditional, and still a target: the precondition decides applicability, not this
308    /// let conditional = BenchmarkPolicy::RequiredWhen {
309    ///     precondition: BenchmarkPrecondition::PlanHoldsAnIssue,
310    /// };
311    /// assert!(conditional.is_target());
312    ///
313    /// // only a waiver takes a capability out of the benchmark set
314    /// let waived = BenchmarkPolicy::Waived { reason: WaiverReason::PublishedHistory };
315    /// assert!(!waived.is_target());
316    /// ```
317    pub fn is_target(self) -> bool {
318        !matches!(self, Self::Waived { .. })
319    }
320}

322/// Whether, and how, the executor keeps results of this capability. Cache lives in the
323/// executor and nowhere else, so MCP, HTTP and the command line share one; the key is the
324/// canonical id, the normalised input and the registry fingerprint, so a changed
325/// repository never answers from an old entry.
326#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
327#[serde(tag = "policy", rename_all = "snake_case")]
328pub enum CachePolicy {
329    /// Every call runs the handler.
330    Disabled,
331    /// Results are kept in this process's memory, bounded, for equal inputs.
332    Process {
333        /// The most entries kept for this capability; the oldest is evicted first.
334        max_entries: usize,
335        /// Seconds an entry stays valid; `None` for the life of the process.
336        #[serde(skip_serializing_if = "Option::is_none")]
337        ttl_seconds: Option<u64>,
338    },
339}
340
341impl CachePolicy {
342    /// Is the policy well-formed? The reason when it is not.
343    ///
344    /// ```
345    /// use majordomus_cli::capability::CachePolicy;
346    /// assert!(CachePolicy::Disabled.validate().is_ok());
347    /// assert!(CachePolicy::Process { max_entries: 0, ttl_seconds: None }.validate().is_err());
348    /// ```
349    pub fn validate(&self) -> Result<(), String> {
350        match self {
351            CachePolicy::Disabled => Ok(()),
352            CachePolicy::Process { max_entries: 0, .. } => {
353                Err("a process cache with max_entries 0 keeps nothing".into())
354            }
355            CachePolicy::Process {
356                ttl_seconds: Some(0),
357                ..
358            } => Err("a process cache with ttl_seconds 0 keeps nothing".into()),
359            CachePolicy::Process { .. } => Ok(()),
360        }
361    }
362
363    /// Does the policy keep anything?
364    pub fn is_enabled(&self) -> bool {
365        !matches!(self, CachePolicy::Disabled)
366    }
367}

369/// What a capability is. Three kinds exist because three semantics exist: something that
370/// is executed and changes nothing, something that is executed and changes this process's
371/// own memory, and something that is read. A command may also write the repository's own
372/// tracked records, which its kind does not say and its execution policy does
373/// ([`ExecutionPolicy::writes_repository`]).
374///
375/// How *long* a call takes is not a kind. A read that walks every file of the layer is
376/// still a read, and the thing that makes it worth watching — that it reports as it goes
377/// and stops when it is asked to — is one property of its handler, declared with
378/// [`crate::capability::Executable::cancellable`] and carried on [`ExecutionPolicy`].
379#[derive(
380    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
381)]
382#[serde(rename_all = "lowercase")]
383pub enum CapabilityKind {
384    /// Executable and read-only: a typed handler, an input schema, an output schema.
385    Query,
386    /// Executable and changing something: a typed handler, bound to `POST` over HTTP, and
387    /// announced to MCP clients as not read-only. Most reach no further than this process's
388    /// own memory (a peer announcing itself); one that writes the repository's tracked
389    /// records says so on its execution policy, which is where a projection reads it.
390    Command,
391    /// Declarative content the repository holds: read as it is, never executed.
392    Resource,
393}
394
395impl CapabilityKind {
396    /// Is a capability of this kind called, with a handler, rather than read?
397    ///
398    /// ```
399    /// use majordomus_cli::capability::CapabilityKind;
400    /// assert!(CapabilityKind::Query.is_executable() && CapabilityKind::Command.is_executable());
401    /// assert!(!CapabilityKind::Resource.is_executable());
402    /// ```
403    pub fn is_executable(self) -> bool {
404        !matches!(self, CapabilityKind::Resource)
405    }
406
407    /// Does a call of this kind leave the process as it found it?
408    ///
409    /// ```
410    /// use majordomus_cli::capability::CapabilityKind;
411    /// assert!(CapabilityKind::Query.is_read_only());
412    /// assert!(!CapabilityKind::Command.is_read_only());
413    /// ```
414    pub fn is_read_only(self) -> bool {
415        !matches!(self, CapabilityKind::Command)
416    }
417
418    /// The word as serialised.
419    pub fn as_str(self) -> &'static str {
420        match self {
421            CapabilityKind::Query => "query",
422            CapabilityKind::Command => "command",
423            CapabilityKind::Resource => "resource",
424        }
425    }
426}

428/// What running a capability changes outside the caller.
429///
430/// Classified, never declared: it follows from the kind, which is the field a declaration
431/// already carries. A projection reads this to decide whether to ask before running
432/// something — the Cockpit's confirmation is derived from it — instead of naming
433/// capabilities it must treat carefully, which is a list that goes stale the day after it
434/// is written.
435#[derive(
436    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
437)]
438#[serde(rename_all = "snake_case")]
439// the component namespace of the OpenAPI document is flat, and `deploy` already has a
440// `Concurrency`; each says which it is
441#[schemars(rename = "ExecutionEffect")]
442pub enum Effect {
443    /// Nothing changes. Every query and every resource of this executable.
444    Read,
445    /// This process's own memory changes, and nothing outside it.
446    ProcessState,
447    /// The repository changes.
448    ///
449    /// For most of this executable's life nothing classified to this, because no capability
450    /// of any kind wrote to the repository. That day arrived with [ADR 0040]: development
451    /// semantics are capabilities of this runtime, and the first of them — `plan.transition`
452    /// — stamps a field into an issue's own tracked record. The variant was put on the model
453    /// in advance so that when one did, it would say so *here*, where a projection already
454    /// reads it and a client already asks before running it, rather than in whichever page
455    /// happens to render its button.
456    ///
457    /// It is not classified from the kind, because the kind cannot know it: a command that
458    /// announces a peer and a command that writes a tracked file are the same kind. A
459    /// handler that writes the repository says so with
460    /// [`ExecutionPolicy::writes_repository`], the way one that stops says so with
461    /// [`ExecutionPolicy::stoppable`].
462    ///
463    /// [ADR 0040]: https://github.com/korczis/prismatic-majordomus/blob/master/.ai/repo/adrs/0040-development-semantics-are-capabilities-of-one-runtime.md
464    RepositoryMutation,
465}

467/// Whether two executions of one capability may overlap.
468#[derive(
469    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
470)]
471#[serde(rename_all = "snake_case")]
472#[schemars(rename = "ExecutionConcurrency")]
473pub enum Concurrency {
474    /// Any number at once. Every read is one of these: the index and the registry are
475    /// immutable for the life of the process, so concurrent readers cannot interfere.
476    Unrestricted,
477    /// One at a time. A second execution of the same capability waits for the first, which
478    /// is what a capability that changes anything — this process's own memory included —
479    /// needs in order to be reasoned about at all.
480    Serial,
481}

483/// What running a capability as an execution means: what it changes, whether asking it to
484/// stop achieves anything, and whether two of them may overlap.
485///
486/// The effect and the concurrency are classified from the kind by
487/// [`ExecutionPolicy::classify`], for the same reason [`Availability`] is: the facts are
488/// already on the declaration, and asking each `capability!` block to restate them would be
489/// one more thing that can disagree with itself.
490///
491/// Cancellability is the one thing the kind cannot decide, because it is a fact about the
492/// handler: whether it looks at its cancellation flag and stops. A handler that does says
493/// so with [`crate::capability::Executable::cancellable`], and a client is then told
494/// whether a Cancel button will achieve anything instead of being given one that lies.
495#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
496pub struct ExecutionPolicy {
497    /// What it changes.
498    pub effect: Effect,
499    /// Whether asking it to stop does anything. A task looks at its cancellation flag; a
500    /// query and a command do not, and a client is told so rather than being given a
501    /// button that lies.
502    pub cancellable: bool,
503    /// Whether two of them may overlap.
504    pub concurrency: Concurrency,
505}
506
507impl ExecutionPolicy {
508    /// The policy of a kind.
509    ///
510    /// ```
511    /// use majordomus_cli::capability::{CapabilityKind, Concurrency, Effect, ExecutionPolicy};
512    /// let query = ExecutionPolicy::classify(CapabilityKind::Query);
513    /// assert_eq!(query.effect, Effect::Read);
514    /// assert_eq!(query.concurrency, Concurrency::Unrestricted);
515    /// assert!(!query.cancellable, "until its handler says it looks at the flag");
516    /// assert!(query.stoppable().cancellable);
517    /// let command = ExecutionPolicy::classify(CapabilityKind::Command);
518    /// assert_eq!(command.effect, Effect::ProcessState);
519    /// assert_eq!(command.concurrency, Concurrency::Serial);
520    /// ```
521    pub fn classify(kind: CapabilityKind) -> Self {
522        match kind {
523            // an immutable index and an immutable registry: readers cannot interfere
524            CapabilityKind::Query | CapabilityKind::Resource => ExecutionPolicy {
525                effect: Effect::Read,
526                cancellable: false,
527                concurrency: Concurrency::Unrestricted,
528            },
529            // it changes this process's memory, so two of them are made to take turns
530            CapabilityKind::Command => ExecutionPolicy {
531                effect: Effect::ProcessState,
532                cancellable: false,
533                concurrency: Concurrency::Serial,
534            },
535        }
536    }
537
538    /// The same policy, for a handler that looks at its cancellation flag and stops.
539    ///
540    /// ```
541    /// use majordomus_cli::capability::{CapabilityKind, ExecutionPolicy};
542    /// let p = ExecutionPolicy::classify(CapabilityKind::Query).stoppable();
543    /// assert!(p.cancellable);
544    /// assert_eq!(p.effect, ExecutionPolicy::classify(CapabilityKind::Query).effect);
545    /// ```
546    pub fn stoppable(self) -> Self {
547        ExecutionPolicy {
548            cancellable: true,
549            ..self
550        }
551    }
552
553    /// The same policy, for a handler that writes the repository's own tracked files.
554    ///
555    /// The second and last thing the kind cannot decide. Two commands are the same kind
556    /// whether one announces a peer into this process's memory and the other stamps a field
557    /// into a record a commit will carry, and the difference is the whole of what a caller
558    /// needs to know before saying yes. It is stated by the handler because only the handler
559    /// knows it.
560    ///
561    /// This is a widening and never a narrowing: a query cannot reach for it, because the
562    /// registry accepts a declared policy only when it is the kind's own or one of these two
563    /// documented additions to it.
564    ///
565    /// ```
566    /// use majordomus_cli::capability::{CapabilityKind, Effect, ExecutionPolicy};
567    /// let p = ExecutionPolicy::classify(CapabilityKind::Command).writes_repository();
568    /// assert_eq!(p.effect, Effect::RepositoryMutation);
569    /// assert!(p.needs_confirmation());
570    /// ```
571    pub fn writes_repository(self) -> Self {
572        ExecutionPolicy {
573            effect: Effect::RepositoryMutation,
574            ..self
575        }
576    }
577
578    /// Should a client ask before running this? True for anything that changes something.
579    ///
580    /// ```
581    /// use majordomus_cli::capability::{CapabilityKind, ExecutionPolicy};
582    /// assert!(!ExecutionPolicy::classify(CapabilityKind::Query).needs_confirmation());
583    /// assert!(ExecutionPolicy::classify(CapabilityKind::Command).needs_confirmation());
584    /// ```
585    pub fn needs_confirmation(self) -> bool {
586        !matches!(self.effect, Effect::Read)
587    }
588
589    /// What a caller may assume about a call before making it, classified from the effect.
590    ///
591    /// A projection that tells a client how careful to be — MCP's tool annotations, the
592    /// site's table of tools, the Cockpit's — renders this value and states none of its
593    /// own. Until it existed the MCP projection wrote `destructiveHint: false` and
594    /// `idempotentHint: true` as constants, which was true of every tool on the day they
595    /// were written and false of the first one that wrote the repository: a sweep that
596    /// unlinks files was announced as safe to repeat and unable to destroy anything.
597    ///
598    /// The classification is conservative in the one direction a caller can survive.
599    /// Anything that changes something is not assumed repeatable, because whether a second
600    /// call is a no-op is a fact about the handler and no handler has declared it; and
601    /// anything that writes the repository may overwrite or remove what was there.
602    /// A change to this process's own memory destroys nothing outside it.
603    ///
604    /// ```
605    /// use majordomus_cli::capability::{CapabilityKind, ExecutionPolicy};
606    /// let read = ExecutionPolicy::classify(CapabilityKind::Query).hints();
607    /// assert!(read.read_only && read.idempotent && !read.destructive);
608    /// let memory = ExecutionPolicy::classify(CapabilityKind::Command).hints();
609    /// assert!(!memory.read_only && !memory.idempotent && !memory.destructive);
610    /// let writer = ExecutionPolicy::classify(CapabilityKind::Command).writes_repository().hints();
611    /// assert!(!writer.read_only && !writer.idempotent && writer.destructive);
612    /// assert!(!writer.open_world, "no handler of this executable reaches beyond the machine");
613    /// ```
614    pub fn hints(self) -> Hints {
615        match self.effect {
616            Effect::Read => Hints {
617                read_only: true,
618                destructive: false,
619                idempotent: true,
620                open_world: false,
621            },
622            Effect::ProcessState => Hints {
623                read_only: false,
624                destructive: false,
625                idempotent: false,
626                open_world: false,
627            },
628            Effect::RepositoryMutation => Hints {
629                read_only: false,
630                destructive: true,
631                idempotent: false,
632                open_world: false,
633            },
634        }
635    }
636}

638/// What a caller may assume about a call before making it: the four facts MCP names tool
639/// annotations, held once on the model so that no projection states its own.
640///
641/// Classified by [`ExecutionPolicy::hints`], never declared.
642///
643/// ```
644/// use majordomus_cli::capability::{CapabilityKind, ExecutionPolicy, Hints};
645/// let read: Hints = ExecutionPolicy::classify(CapabilityKind::Query).hints();
646/// assert_eq!(
647///     serde_json::to_value(read).unwrap(),
648///     serde_json::json!({ "read_only": true, "destructive": false, "idempotent": true, "open_world": false })
649/// );
650/// ```
651#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
652#[schemars(rename = "ExecutionHints")]
653pub struct Hints {
654    /// The call changes nothing, anywhere.
655    pub read_only: bool,
656    /// The call may overwrite or remove something that outlives this process.
657    pub destructive: bool,
658    /// A second identical call changes nothing the first did not.
659    pub idempotent: bool,

662    pub open_world: bool,
663}

665/// Where a capability stands, in the repository's own vocabulary for claims. A capability
666/// that is `Planned` or `Unsupported` may be listed but is never executable through any
667/// projection; the registry refuses to build otherwise.
668#[derive(
669    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,

671#[serde(rename_all = "snake_case")]
672pub enum Stability {

674    Implemented,
675    /// Implemented and proved by a behavioural test.

677    /// Implemented, executable, and expected to change.
678    Experimental,

725/// Where this crate lives in the repository, so that a builtin descriptor's Rust module
726/// path names a file a reader can open. A fact of the tree's layout; the site generator
727/// refuses a path that does not exist.
728pub const CRATE_DIR: &str = "apps/majordomus-cli";
729
730impl Provenance {
731    /// The repository-relative path of the source: a declarative object's file, or the
732    /// Rust file a builtin descriptor was composed in, by the crate's layout
733    /// (`majordomus_cli::capability::builtin::objects` is
734    /// `apps/majordomus-cli/src/capability/builtin/objects.rs`).
735    ///
736    /// ```
737    /// use majordomus_cli::capability::Provenance;
738    /// let p = Provenance::Builtin { module: "majordomus_cli::capability::builtin::objects".into() };
739    /// assert_eq!(p.source_path(), "apps/majordomus-cli/src/capability/builtin/objects.rs");
740    /// ```
741    pub fn source_path(&self) -> String {
742        match self {
743            Provenance::Builtin { module } => {
744                let inner = module
745                    .strip_prefix("majordomus_cli::")
746                    .unwrap_or(module.as_str());
747                format!("{CRATE_DIR}/src/{}.rs", inner.replace("::", "/"))
748            }
749            Provenance::Declarative { path, .. } => path.clone(),
750        }
751    }
752}

754impl fmt::Display for Provenance {
755    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
756        match self {
757            Provenance::Builtin { module } => write!(f, "builtin {module}"),
758            Provenance::Declarative {
759                path,
760                member: Some(m),
761                ..
762            } => write!(f, "{path}#{m}"),
763            Provenance::Declarative {
764                path, member: None, ..
765            } => write!(f, "{path}"),
766        }
767    }

826/// How a capability appears over HTTP. `GET` binds every top-level input property as a
827/// query parameter; `POST` binds the input as the JSON request body. Paths are absolute
828/// and live under [`HttpExposure::PREFIX`].
829#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
830pub struct HttpExposure {
831    /// The method.
832    pub method: HttpMethod,
833    /// The absolute path, under [`HttpExposure::PREFIX`].
834    pub path: String,
835}
836
837impl HttpExposure {
838    /// Every capability route starts here; the version is part of the contract.
839    pub const PREFIX: &'static str = "/api/v1/";
840
841    /// Is the path one this projection can serve? The reason when it is not.
842    ///
843    /// ```
844    /// use majordomus_cli::capability::{HttpExposure, HttpMethod};
845    /// let ok = HttpExposure { method: HttpMethod::Get, path: "/api/v1/objects".into() };
846    /// assert!(ok.validate().is_ok());
847    /// let bad = HttpExposure { method: HttpMethod::Get, path: "/objects".into() };
848    /// assert!(bad.validate().unwrap_err().contains("/api/v1/"));
849    /// ```
850    pub fn validate(&self) -> Result<(), String> {
851        let p = &self.path;
852        if !p.starts_with(Self::PREFIX) {
853            return Err(format!("path '{p}' is not under {}", Self::PREFIX));
854        }
855        if p.ends_with('/') || p.contains("//") || p.contains('?') || p.contains('#') {
856            return Err(format!(
857                "path '{p}' has a trailing slash, an empty segment, or a query or fragment"
858            ));
859        }
860        if !p
861            .bytes()
862            .all(|b| b.is_ascii_alphanumeric() || b"/-_.".contains(&b))
863        {
864            return Err(format!(
865                "path '{p}' carries a character outside [A-Za-z0-9/-_.]"
866            ));
867        }
868        Ok(())
869    }
870}

901/// Where a capability means anything: the environment a caller must be in for it to
902/// answer at all.
903///
904/// The published site and the running server are genuinely different places. Without this
905/// on the model, every template grows its own idea of what works where — and the usual
906/// shape that takes is a condition on the page's own address, which is a rule hidden
907/// where nobody will find it and nothing can test it. This is the only thing a projection
908/// may ask.
909///
910/// It is classified rather than declared: the facts that decide it — what kind of thing
911/// this is and which transports it is projected through — are already on the descriptor,
912/// and asking each declaration to restate them would be the same knowledge written twice.
913/// [`Availability::classify`] is the one place the rule lives.
914#[derive(
915    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
916)]
917#[serde(rename_all = "snake_case")]
918// Three things in this executable answer to `Availability` — a capability's, a
919// surface's, and whether a surface is answering right now. The schema component namespace
920// is flat, so each says which it is.
921#[schemars(rename = "CapabilityAvailability")]
922pub enum Availability {
923    /// True in every environment, a published page with no server included: the layer's
924    /// own content, which a build renders and a process serves from the same index.
925    Always,
926    /// A process must be running to answer: everything with a handler, whether it is
927    /// reached over HTTP, over MCP or from the command line.
928    Runtime,
929    /// A value captured when the site was generated, rendered afterwards as the capture
930    /// it is. Nothing classifies to this yet; the static projection of the graph is what
931    /// will declare it, and it is on the model so that a captured value can be labelled
932    /// as captured instead of being shown as current.
933    BuildTime,
934    /// A process must be running and the caller must be one it has authenticated. Nothing
935    /// in this repository authenticates a caller yet; a surface that does will say so
936    /// here rather than in the template that renders its link.
937    Authenticated,
938}
939
940impl Availability {
941    /// Classify from what the descriptor already declares.
942    ///
943    /// The match is exhaustive on purpose and has no fallback arm: a new kind, or a
944    /// transport that changes what an environment can offer, is a compile error here
945    /// rather than a silent `Always` in a page that then links to nothing.
946    ///
947    /// ```
948    /// use majordomus_cli::capability::{Availability, CapabilityKind, Exposure};
949    /// let nowhere = Exposure::default();
950    /// assert_eq!(Availability::classify(CapabilityKind::Resource, &nowhere), Availability::Always);
951    /// assert_eq!(Availability::classify(CapabilityKind::Query, &nowhere), Availability::Runtime);
952    /// ```
953    pub fn classify(kind: CapabilityKind, _exposure: &Exposure) -> Availability {
954        match kind {
955            // the content is the layer's, and the index that holds it is read the same way
956            // by a build and by a process
957            CapabilityKind::Resource => Availability::Always,
958            // a handler answers, and a handler needs a process to run in — the transport
959            // decides who may call it, not whether anything can
960            CapabilityKind::Query | CapabilityKind::Command => Availability::Runtime,
961        }
962    }
963}

965/// Who a capability is for, and whether anything offers it.
966///
967/// Internal is a statement, not an omission: a capability nothing projects is invisible
968/// either way, and the difference between deliberate and forgotten is exactly what this
969/// records.
970#[derive(
971    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Serialize, Deserialize, JsonSchema,
972)]
973#[serde(rename_all = "snake_case")]
974#[schemars(rename = "CapabilityVisibility")]
975pub enum Visibility {
976    /// Offered to anyone who can reach the process: an HTTP route or an MCP entry.
977    Public,
978    /// Offered to whoever runs the executable, and to nobody over a network.
979    Developer,
980    /// Projected nowhere. It exists, it is listed as existing, and no surface offers it.
981    Internal,
982}
983
984impl Visibility {
985    /// Classify from the transports the descriptor declares.
986    ///
987    /// ```
988    /// use majordomus_cli::capability::{CliExposure, Exposure, Visibility};
989    /// assert_eq!(Visibility::classify(&Exposure::default()), Visibility::Internal);
990    /// let cli = Exposure { cli: Some(CliExposure { path: vec!["scope".into()] }), ..Default::default() };
991    /// assert_eq!(Visibility::classify(&cli), Visibility::Developer);
992    /// ```
993    pub fn classify(exposure: &Exposure) -> Visibility {
994        if exposure.mcp.is_some() || exposure.http.is_some() {
995            Visibility::Public
996        } else if exposure.cli.is_some() {
997            Visibility::Developer
998        } else {
999            Visibility::Internal
1000        }
1001    }
1002}
