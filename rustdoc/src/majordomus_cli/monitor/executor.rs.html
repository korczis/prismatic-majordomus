1//! The executor's trail, folded into signals.
2//!
3//! The trail records every step of the pull-request lane: each observation, each selection,
4//! each attempt before its outcome. A reader wants the outcomes — merged, refreshed, closed,
5//! repaired, failed — and who holds the lane; the steps that decided nothing are counted as
6//! suppressed. The trail has no sequence of its own, so a position is how many of its lines
7//! were read; a trail shorter than the position (rewritten) reads as having nothing new.
8
9use crate::integration::drain::{IntegrationAction, IntegrationEvent};
10use crate::peers::parse_rfc3339;
11
12use super::{MonitorSeverity, MonitorSignal, MonitorSource, Reader, SourceRead};
13
14/// Fold `events` (the whole trail, oldest first) after `after` lines into signals; without
15/// a position, only events at or after `since` (Unix seconds) are shown.
16pub(crate) fn fold(
17    events: &[IntegrationEvent],
18    after: Option<usize>,
19    since: u64,
20    reader: &Reader,
21) -> SourceRead<usize> {
22    let mut read = SourceRead::default();
23    let start = after.unwrap_or(0).min(events.len());
24    for (offset, event) in events.iter().enumerate().skip(start) {
25        let position = offset + 1;
26        read.high = Some(position);
27        let at = parse_rfc3339(&event.at).unwrap_or(0);
28        if after.is_none() && at < since {
29            continue;
30        }
31        match signal(event, at, reader) {
32            Some(s) => read.signals.push((position, s)),
33            None => read.suppressed += 1,
34        }
35    }
36    if after.is_some() && read.high.is_none() {
37        read.high = after.map(|a| a.min(events.len()));
38    }
39    read
40}

42/// The signal an executor event is, if it is an outcome.
43fn signal(event: &IntegrationEvent, at: u64, reader: &Reader) -> Option<MonitorSignal> {
44    use IntegrationAction as A;
45    use MonitorSeverity::{Attention, Info};
46    let pr = event.pr.map(|n| format!("#{n}")).unwrap_or_default();
47    let (kind, severity, text) = match event.action {
48        A::MergeSucceeded => ("merged", Info, format!("merged {pr}")),
49        A::MergeFailed => ("merge_failed", Attention, format!("could not merge {pr}")),
50        A::VerificationFailed => (
51            "verification_failed",
52            Attention,
53            format!("{pr} failed verification"),
54        ),
55        A::Refreshed => ("refreshed", Info, format!("refreshed {pr} onto master")),
56        A::RefreshFailed => (
57            "refresh_failed",
58            Attention,
59            format!("could not refresh {pr}"),
60        ),
61        A::ClosedRedundant => (
62            "closed",
63            Info,
64            format!("closed {pr}: its work is on master"),
65        ),
66        A::ClosedSuperseded => ("closed", Info, format!("closed {pr}: superseded")),
67        A::CloseFailed => ("close_failed", Attention, format!("could not close {pr}")),
68        A::Repaired => ("repaired", Info, format!("repaired {pr}")),
69        A::RepairRefused => (
70            "repair_refused",
71            Attention,
72            format!("refused to repair {pr}"),
73        ),
74        A::LeaseAcquired => ("lane_taken", Info, "took the integration lane".to_string()),
75        A::LeaseReleased => (
76            "lane_released",
77            Info,
78            "released the integration lane".to_string(),
79        ),
80        A::ObserveFailed => (
81            "observe_failed",
82            Attention,
83            "could not observe the forge".to_string(),
84        ),
85        _ => return None,
86    };
87    let actor = event
88        .actor
89        .split(" (")
90        .next()
91        .unwrap_or(&event.actor)
92        .to_string();
93    let detail = if event.detail.is_empty() {
94        String::new()
95    } else {
96        format!(": {}", event.detail)
97    };
98    let mut s = MonitorSignal::new(
99        at,
100        MonitorSource::Executor,
101        kind,
102        severity,
103        &format!("executor {actor} {text}{detail}"),
104    );
105    s.pr = event.pr;
106    s.for_you = reader.named_in(&event.detail) || reader.named_in(&event.actor);
107    s.actor = Some(actor);
108    if let Some(n) = event.pr {
109        s.link = Some(format!("/cockpit/integration#pr-{n}"));
110    }
111    Some(s)
112}

128    #[test]
129    fn outcomes_are_signals_and_steps_are_suppressed() {
130        let trail = vec![
131            event("2026-10-10T20:00:00Z", "observed", None, ""),
132            event("2026-10-10T20:00:01Z", "merge_attempted", Some(868), ""),
133            event(
134                "2026-10-10T20:00:02Z",
135                "merge_succeeded",
136                Some(868),
137                "pinned to its own run",
138            ),
139            event(
140                "2026-10-10T20:00:03Z",
141                "refresh_failed",
142                Some(885),
143                "conflict in serve.rs; ask 1b",
144            ),
145        ];
146        let read = fold(&trail, None, 0, &Reader::named(Some("1b")));
147        let kinds: Vec<&str> = read.signals.iter().map(|(_, s)| s.kind.as_str()).collect();
148        assert_eq!(kinds, vec!["merged", "refresh_failed"]);
149        assert_eq!(read.suppressed, 2);
150        assert_eq!(read.high, Some(4));
151        let merged = &read.signals[0].1;
152        assert_eq!(
153            merged.summary,
154            "executor Tomas Korcak merged #868: pinned to its own run"
155        );
156        assert_eq!(merged.pr, Some(868));
157        assert_eq!(merged.link.as_deref(), Some("/cockpit/integration#pr-868"));
158        assert!(!merged.for_you);
159        let failed = &read.signals[1].1;
160        assert_eq!(failed.severity, MonitorSeverity::Attention);
161        assert!(failed.for_you, "the detail names the reader");
162    }

166        let table = [

168            ("merge_failed", "merge_failed", true, "could not merge #9"),
169            (
170                "verification_failed",
171                "verification_failed",
172                true,
173                "#9 failed verification",

175            ("refreshed", "refreshed", false, "refreshed #9 onto master"),
176            (
177                "refresh_failed",
178                "refresh_failed",
179                true,
180                "could not refresh #9",
181            ),
182            (
183                "closed_redundant",
184                "closed",
185                false,
186                "closed #9: its work is on master",
187            ),
188            (
189                "closed_superseded",
190                "closed",
191                false,
192                "closed #9: superseded",
193            ),
194            ("close_failed", "close_failed", true, "could not close #9"),
195            ("repaired", "repaired", false, "repaired #9"),
196            (
197                "repair_refused",
198                "repair_refused",
199                true,
200                "refused to repair #9",
201            ),
202            (
203                "lease_acquired",
204                "lane_taken",
205                false,
206                "took the integration lane",
207            ),
208            (
209                "lease_released",
210                "lane_released",
211                false,
212                "released the integration lane",
213            ),
214            (
215                "observe_failed",
216                "observe_failed",
217                true,
218                "could not observe the forge",
219            ),
220        ];
221        for (action, kind, attention, words) in table {
222            let pr = (!action.starts_with("lease") && action != "observe_failed").then_some(9);
223            let trail = vec![event("2026-10-10T20:00:00Z", action, pr, "")];
224            let read = fold(&trail, None, 0, &Reader::default());
225            assert_eq!(read.signals.len(), 1, "{action} is an outcome");
226            let s = &read.signals[0].1;
227            assert_eq!(s.kind, kind, "{action}");
228            let want = if attention {
229                MonitorSeverity::Attention
230            } else {
231                MonitorSeverity::Info
232            };
233            assert_eq!(s.severity, want, "{action}");
234            assert_eq!(
235                s.summary,
236                format!("executor Tomas Korcak {words}"),
237                "{action}"
238            );
239            assert_eq!(
240                s.link.is_some(),
241                pr.is_some(),
242                "{action}: a link only with a PR"
243            );
244        }
245        let steps = [
246            "observed",
247            "selected",
248            "merge_attempted",
249            "repair_attempted",
250            "idle",
251        ];
252        for action in steps {
253            let read = fold(
254                &[event("2026-10-10T20:00:00Z", action, None, "")],
255                None,
256                0,
257                &Reader::default(),
258            );
259            assert!(read.signals.is_empty(), "{action} is a step, not news");
260            assert_eq!(read.suppressed, 1, "{action}");
261        }
262    }

264    #[test]
265    fn a_position_resumes_and_a_rewritten_trail_reads_nothing_new() {
266        let trail = vec![
267            event("2026-10-10T20:00:00Z", "lease_acquired", None, ""),
268            event("2026-10-10T20:00:01Z", "lease_released", None, ""),
269        ];
270        let reader = Reader::default();
271        let read = fold(&trail, Some(1), 0, &reader);
272        assert_eq!(read.signals.len(), 1);
273        assert_eq!(read.signals[0].1.kind, "lane_released");
274        let past = fold(&trail, Some(9), 0, &reader);
275        assert!(past.signals.is_empty());
276        assert_eq!(past.high, Some(2));
277        let none = fold(&trail, Some(2), 0, &reader);
278        assert_eq!(none.high, Some(2));
279    }

282    fn a_fresh_read_shows_only_the_window() {
283        let trail = vec![
284            event("2026-10-10T20:00:00Z", "merge_succeeded", Some(1), ""),

290        assert_eq!(read.signals[0].1.pr, Some(2));
