1//! The releases this repository published, and the public metadata derived from them.
2//!
3//! A record under `.ai/repo/releases/` is evidence: the release pipeline writes it from
4//! what it actually uploaded, and every projection reads it rather than GitHub's API. The
5//! installer resolves two files and nothing else — `releases/<tag>.json` for a pinned
6//! version, `releases/latest.json` for the stable pointer — and both are rendered here.
7

33/// Which releases an unpinned installation may resolve to.
34#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
35#[serde(rename_all = "lowercase")]
36pub enum Channel {
37    /// Resolved by `latest`.
38    Stable,
39    /// Published, addressable by its exact tag, never resolved by default.
40    Prerelease,
41}

43/// One published artifact.
44#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
45#[serde(deny_unknown_fields)]
46pub struct ReleaseArtifact {
47    /// The id of the target in the distribution model.
48    pub target: String,
49    /// The artifact's file name, as the naming function derives it.
50    pub name: String,
51    /// Where it is downloaded from.
52    pub url: String,
53    /// The digest the installer verifies before it extracts anything.
54    pub sha256: String,
55    /// The size in bytes, as published.
56    pub size: u64,
57}

59/// One published release.
60#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
61#[serde(deny_unknown_fields)]
62pub struct Release {
63    /// The contract this record follows.
64    pub schema: String,
65    /// The semantic version, without a leading `v`.
66    pub version: String,
67    /// The git tag: `v` and the version.
68    pub tag: String,
69    /// Which releases an unpinned installation may resolve to.
70    pub channel: Channel,
71    /// The commit the artifacts were built from.
72    pub commit: String,
73    /// When it was published, UTC.
74    pub published_at: String,
75    /// The release notes a person reads.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub notes_url: Option<String>,
78    /// True when the release must no longer be resolved.
79    #[serde(default)]
80    pub yanked: bool,
81    /// The target ids the model published when this release was made — the set the release
82    /// had to be complete over, recorded rather than recomputed.
83    ///
84    /// This is a historical snapshot and the only field here that could in principle be
85    /// derived, which is exactly why it is not. Judging a record against the *current*
86    /// model means adding a platform retroactively makes every release ever published
87    /// incomplete, and `v0.3.1` cannot grow a `riscv64` artifact however the model changes.
88    /// Written by `scripts/release-record` from the model it was published against.
89    ///
90    /// Absent on records written before the field existed; those are judged on internal
91    /// consistency alone, and `.ai/repo/releases/README.md` says so.
92    #[serde(default, skip_serializing_if = "Option::is_none")]
93    pub required_targets: Option<Vec<String>>,
94    /// One entry per built target.
95    pub artifacts: Vec<ReleaseArtifact>,
96}
97
98impl Release {
99    /// Parse one record.
100    pub fn parse(text: &str) -> std::result::Result<Self, String> {
101        let release: Release = yaml::parse_into(text)?;
102        if release.schema != SCHEMA {
103            return Err(format!("schema is `{}`, not `{SCHEMA}`", release.schema));
104        }
105        if release.tag != format!("v{}", release.version) {
106            return Err(format!(
107                "tag `{}` is not `v` followed by version `{}`",
108                release.tag, release.version
109            ));
110        }
111        Ok(release)
112    }
113
114    /// Is this release a candidate for the stable pointer?
115    pub fn is_stable_candidate(&self) -> bool {
116        self.channel == Channel::Stable && !self.yanked
117    }
118
119    /// The artifact of a target id.
120    pub fn artifact(&self, target: &str) -> Option<&ReleaseArtifact> {
121        self.artifacts.iter().find(|a| a.target == target)
122    }
123
124    /// Every way this record disagrees with the model it claims to have been built from:
125    /// a missing supported target, an artifact of a target that is not published, a name
126    /// the naming function does not derive, a URL from another repository.
127    pub fn findings(&self, model: &Model) -> Vec<String> {
128        let mut out = Vec::new();
129        let prefix = model.project.download_prefix();
130        // Completeness is judged against the set this release was published against, not
131        // against the model as it stands now. A target added to the model is a promise about
132        // the *next* release; it cannot be a finding against one that is already published,
133        // whose artifacts are immutable and whose tag someone may have pinned. The check that
134        // a release is complete when it is made lives in `scripts/release-record`, which
135        // writes the record from the artifacts that were actually uploaded and refuses a
136        // partial one — that is the enforcement point, and it sees the model of its own day.
137        //
138        // A record written before this field existed carries no set, and is judged on
139        // internal consistency alone rather than against a model it never saw.
140        if let Some(required) = &self.required_targets {
141            for id in required {
142                if self.artifact(id).is_none() {
143                    out.push(format!(
144                        "{}: no artifact for `{}`, which it was published against; a partial release is not a release",
145                        self.tag, id
146                    ));
147                }
148            }
149        }
150        let mut seen: Vec<&str> = Vec::new();
151        for a in &self.artifacts {
152            if seen.contains(&a.target.as_str()) {
153                out.push(format!("{}: two artifacts for `{}`", self.tag, a.target));
154            }
155            seen.push(&a.target);
156            let Some(t) = model.target(&a.target) else {
157                out.push(format!(
158                    "{}: artifact for `{}`, which the model does not declare",
159                    self.tag, a.target
160                ));
161                continue;
162            };
163            if !t.status.is_published() {
164                out.push(format!(
165                    "{}: artifact for `{}`, which the model does not publish",
166                    self.tag, a.target
167                ));
168            }
169            let expected = t.artifact_name(&model.project, &model.archive, &self.tag);
170            if a.name != expected {
171                out.push(format!(
172                    "{}: artifact for `{}` is named `{}`; the naming function derives `{expected}`",
173                    self.tag, a.target, a.name
174                ));
175            }
176            if !a.url.starts_with(&prefix) {
177                out.push(format!(
178                    "{}: artifact for `{}` is served from `{}`, not from {prefix}",
179                    self.tag, a.target, a.url
180                ));
181            }
182            if !a.url.ends_with(&format!("/{}/{}", self.tag, a.name)) {
183                out.push(format!(
184                    "{}: the URL of `{}` does not end with its own tag and name",
185                    self.tag, a.target
186                ));
187            }
188            if a.sha256.len() != 64 || !a.sha256.bytes().all(|b| b.is_ascii_hexdigit()) {
189                out.push(format!(
190                    "{}: the digest of `{}` is not 64 hexadecimal characters",
191                    self.tag, a.target
192                ));
193            }
194            if a.size == 0 {
195                out.push(format!("{}: the size of `{}` is zero", self.tag, a.target));
196            }
197        }
198        out
199    }
200
201    /// The public metadata a pinned installation reads. Rendered by hand rather than by
202    /// `serde_json` so that every artifact is exactly one line: the installer parses this
203    /// file with `sed`, in a shell that has no JSON parser, and
204    /// `test/cases/85_installer.sh` holds both halves to that shape.
205    pub fn public_json(&self, model: &Model) -> String {
206        let mut s = String::from("{\n");
207        s.push_str(&format!("  \"schema\": \"{}\",\n", SCHEMA));
208        // Generated like everything else this tool writes, and it says so in the member the
209        // typed-artifact check reads. Without it `majordomus generate` refuses to write the
210        // file the moment a record exists — which is to say, from the first release onwards.
211        s.push_str(&format!(
212            "  \"generated\": \"{}\",\n",
213            crate::generate::json_banner(&format!("{DIR}/{}.yaml", self.tag))
214        ));
215        s.push_str(&format!("  \"version\": \"{}\",\n", self.version));
216        s.push_str(&format!("  \"tag\": \"{}\",\n", self.tag));
217        s.push_str(&format!(
218            "  \"channel\": \"{}\",\n",
219            match self.channel {
220                Channel::Stable => "stable",
221                Channel::Prerelease => "prerelease",
222            }
223        ));
224        s.push_str(&format!("  \"commit\": \"{}\",\n", self.commit));
225        s.push_str(&format!("  \"published_at\": \"{}\",\n", self.published_at));
226        if let Some(url) = &self.notes_url {
227            s.push_str(&format!("  \"notes_url\": \"{url}\",\n"));
228        }
229        s.push_str("  \"artifacts\": {\n");
230        // keyed by the Rust target triple: what the installer resolves the machine to
231        let rows: Vec<String> = self
232            .artifacts
233            .iter()
234            .filter_map(|a| model.target(&a.target).map(|t| (t, a)))
235            .map(|(t, a)| {
236                format!(
237                    "    \"{}\": {{ \"name\": \"{}\", \"url\": \"{}\", \"sha256\": \"{}\", \"size\": {} }}",
238                    t.rust_target, a.name, a.url, a.sha256, a.size
239                )
240            })
241            .collect();
242        s.push_str(&rows.join(",\n"));
243        s.push_str("\n  }\n}\n");
244        s
245    }
246}

248/// Every record this repository holds, newest version first.
249#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, JsonSchema)]
250pub struct Releases {
251    /// The records, ordered by version, highest first.
252    pub releases: Vec<Release>,
253}
254
255impl Releases {
256    /// Read every record under `<root>/.ai/repo/releases/`. A repository that has
257    /// published nothing has no directory, and that is not an error.
258    pub fn load(root: &Path) -> Result<Self> {
259        let dir = root.join(DIR);
260        let mut releases = Vec::new();
261        if dir.is_dir() {
262            let mut paths: Vec<std::path::PathBuf> = std::fs::read_dir(&dir)
263                .map_err(|e| Error::io(&dir, e))?
264                .filter_map(|e| e.ok().map(|e| e.path()))
265                .filter(|p| p.extension().and_then(|e| e.to_str()) == Some("yaml"))
266                .collect();
267            paths.sort();
268            for path in paths {
269                let text = std::fs::read_to_string(&path).map_err(|e| Error::io(&path, e))?;
270                let release = Release::parse(&text).map_err(|reason| Error::InvalidRelease {
271                    path: path.display().to_string(),
272                    reason,
273                })?;
274                let stem = path
275                    .file_stem()
276                    .and_then(|s| s.to_str())
277                    .unwrap_or_default();
278                if stem != release.tag {
279                    return Err(Error::InvalidRelease {
280                        path: path.display().to_string(),
281                        reason: format!("the file name is `{stem}` and the tag is `{}`; a record is named by its tag", release.tag),
282                    });
283                }
284                releases.push(release);
285            }
286        }
287        Ok(Self::ordered(releases))
288    }
289
290    /// The same, from the repository's index, which is what a capability sees.
291    pub fn from_index(index: &Index) -> std::result::Result<Self, String> {
292        let mut releases = Vec::new();
293        for o in index.objects.iter().filter(|o| o.kind == KIND) {
294            let release: Release = serde_json::from_value(o.metadata.clone())
295                .map_err(|e| format!("{}: {e}", o.provenance.path))?;
296            releases.push(release);
297        }
298        Ok(Self::ordered(releases))
299    }
300
301    pub(crate) fn ordered(mut releases: Vec<Release>) -> Self {
302        // Highest version first, and deliberately not the canonical order: `latest_stable`
303        // below takes the first stable record, so descending version is what makes that
304        // answer correct. A label order would make it return an arbitrary release.
305        releases.sort_by_key(|r| std::cmp::Reverse(version_key(&r.version)));
306        Releases { releases }
307    }
308
309    /// The release an unpinned installation resolves to: the highest version among the
310    /// stable, unyanked records. Derived, never authored.
311    pub fn latest_stable(&self) -> Option<&Release> {
312        self.releases.iter().find(|r| r.is_stable_candidate())
313    }
314
315    /// A release by tag; `None` when no record carries it.
316    ///
317    /// Test support, and compiled only for tests: this module's own suite names one record
318    /// among several with it, and nothing in the shipped binary asks the question that way.
319    #[cfg(test)]
320    pub(crate) fn by_tag(&self, tag: &str) -> Option<&Release> {
321        self.releases.iter().find(|r| r.tag == tag)
322    }
323
324    /// Is there any release at all?
325    pub fn is_empty(&self) -> bool {
326        self.releases.is_empty()
327    }
328
329    /// Every way the records disagree with the model or with each other.
330    pub fn findings(&self, model: &Model) -> Vec<String> {
331        let mut out = Vec::new();
332        let mut tags: Vec<&str> = Vec::new();
333        for r in &self.releases {
334            if tags.contains(&r.tag.as_str()) {
335                out.push(format!("two records claim the tag `{}`", r.tag));
336            }
337            tags.push(&r.tag);
338            out.extend(r.findings(model));
339        }
340        out
341    }
342}

344/// A comparable key for a semantic version: the three numbers, then whether it is a final
345/// release (a pre-release sorts below its own final), then the pre-release text.
346fn version_key(version: &str) -> (u64, u64, u64, u8, String) {
347    let (core, pre) = match version.split_once('-') {
348        Some((c, p)) => (c, p.to_string()),
349        None => (version, String::new()),
350    };
351    let mut parts = core.split('.').map(|p| p.parse::<u64>().unwrap_or(0));
352    (
353        parts.next().unwrap_or(0),
354        parts.next().unwrap_or(0),
355        parts.next().unwrap_or(0),
356        u8::from(pre.is_empty()),
357        pre,
358    )
359}

361/// The stable pointer, rendered from the record it points at. Identical in shape to a
362/// pinned release's metadata, so that the installer has one parser and not two.
363pub fn latest_json(release: &Release, model: &Model) -> String {
364    release.public_json(model)
