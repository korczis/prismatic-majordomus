1//! The recorder: turning a run that happened into evidence that lasts.
2//!
3//! The runs already write their results down. `test/run.sh` writes a TSV row per case when
4//! `MJ_TEST_REPORT` is set — name, result, seconds, phase — and CI already sets it.
5//! `cargo test` prints a `Running <binary>` line and a `test result:` line per test binary.
6//! Neither is durable, neither carries a commit, and neither is joined to anything.
7//!
8//! This module reads what those runs wrote, carries into each result what the run measured
9//! about its own checkout (which commit, which tree) with the digest, the time and the
10//! origin, and merges it into the ledger.
11//!
12//! # It records, it does not decide
13//!
14//! A case that failed is a case the runner said failed. Nothing here re-judges a result,
15//! and nothing here can produce one: a test id that appears in no report is simply not
16//! recorded, and the report will say `not run` for it, which is true.
17//!
18//! # A crate binary states only what ran
19//!
20//! "Records what the runner said" needs care with `cargo test`, whose result line starts
21//! with `ok` for a binary that ran nothing at all: every test ignored, or every test
22//! filtered out of a run that named one. So [`read_crate_output`] reads the counts, not
23//! only the word. A binary that ran no test, or only a filtered subset, is a skip; one with
24//! a failed test is a failure whatever its word says; one whose result line carries no
25//! count, or that printed no result line at all, is an error, because the harness could not
26//! say what ran. Colour codes are stripped before any line is matched, so a coloured run and
27//! a plain one read the same. What the output held that no claim can name yet, the crate's
28//! own unit-test binary and its doctests, is listed in [`RecordOutcome::dropped`] rather
29//! than ignored.
30//!
31//! # The run measures; the recorder carries
32//!
33//! The recorder does not measure its own checkout. A run measures the checkout it left with
34//! `majordomus evidence stamp` ([`super::provenance::stamp`]): the commit, and the tree with
35//! the evidence ledger ignored and the run's own untracked outputs excluded. The recorder
36//! carries that measurement into every execution of the report it names, and refuses one of
37//! another commit or of other bytes. A report recorded without a measurement carries the
38//! tree `unknown`: the recorder does not vouch for a tree it did not see, and such a row
39//! never reads `proven`. When the measured tree was clean, the digest is taken from the
40//! commit, which holds the bytes the run executed.
41//!
42//! A stamp measures the checkout when it is taken, after the report was written, and not
43//! the tree while the run executed: a run on a dirty tree that is cleaned before the stamp
44//! reads clean.
45//!
46//! # A run, recorded
47//!
48//! A repository with one case, a report the runner wrote, and the execution that comes out
49//! of joining the two:
50//!
51//! ```
52//! use majordomus_cli::evidence::{
53//!     record, stamp, EvidenceProducer, Ledger, Origin, Outcome, RecordRequest, StampRequest,
54//! };
55//! use std::process::Command;
56//!
57//! let root = tempfile::tempdir().unwrap();
58//! let git = |args: &[&str]| {
59//!     Command::new("git").arg("-C").arg(root.path()).args(args).output().unwrap()
60//! };
61//! git(&["init", "-q"]);
62//! git(&["config", "user.email", "t@example.com"]);
63//! git(&["config", "user.name", "t"]);
64//! std::fs::create_dir_all(root.path().join("test/cases")).unwrap();
65//! std::fs::write(root.path().join("test/cases/07_scope.sh"), "echo scope\n").unwrap();
66//! git(&["add", "-A"]);
67//! git(&["commit", "-qm", "init"]);
68//!
69//! // what `test/run.sh` already writes when `MJ_TEST_REPORT` is set; kept outside the
70//! // work tree, so the recorded tree state describes the run and not the report
71//! let reports = tempfile::tempdir().unwrap();
72//! let tsv = reports.path().join("run.tsv");
73//! std::fs::write(&tsv, "07_scope\tok\t12\tparallel\n").unwrap();
74//!
75//! // the run measures its own checkout, naming its report
76//! let stamp_req = StampRequest {
77//!     producer: Some(EvidenceProducer::Suite),
78//!     report: Some(tsv.clone()),
79//!     ..StampRequest::default()
80//! };
81//! let measured = stamp(root.path(), &stamp_req).unwrap();
82//!
83//! let mut request = RecordRequest { suite: Some(tsv), ..RecordRequest::new(Origin::Ci) };
84//! request.provenance.insert(EvidenceProducer::Suite, measured);
85//! let outcome = record(root.path(), &request).unwrap();
86//! assert_eq!(outcome.recorded, 1);
87//! assert_eq!(outcome.passed, 1);
88//! assert_eq!(outcome.commit.len(), 40);
89//! assert_eq!(outcome.working_tree, "clean", "the tree the run's own measurement stated");
90//!
91//! // and it is in the ledger, stamped, with the command that produces it again
92//! let ledger = Ledger::load(root.path()).unwrap();
93//! let execution = ledger.latest("suite:07_scope").unwrap();
94//! assert_eq!(execution.outcome, Outcome::Pass);
95//! assert_eq!(execution.seconds, 12);
96//! assert_eq!(execution.origin, Origin::Ci);
97//! assert_eq!(execution.command, "bash test/run.sh 07_scope");
98//! assert_eq!(execution.digest_matches(root.path()), Some(true));
99//! ```

137/// What to record, and where from.
138///
139/// Both sources are optional and independent — a run of the suite, a run of the crate, or
140/// one invocation recording both — but a request that names neither is refused rather than
141/// recording an empty run as if nothing had failed.
142///
143/// ```
144/// use majordomus_cli::evidence::{record, Origin, RecordRequest};
145///
146/// let nothing = RecordRequest::default();
147/// let root = tempfile::tempdir().unwrap();
148/// let refused = record(root.path(), &nothing).unwrap_err().to_string();
149/// assert!(refused.contains("nothing to record"), "{refused}");
150///
151/// let suite_run =
152///     RecordRequest { suite: Some("tmp/run.tsv".into()), ..RecordRequest::new(Origin::Ci) };
153/// assert_eq!(suite_run.origin, Origin::Ci);
154/// assert_eq!(suite_run.suite.as_deref(), Some(std::path::Path::new("tmp/run.tsv")));
155/// ```
156#[derive(Debug, Clone)]
157/// # Example
158///
159/// ```
160/// use majordomus_cli::evidence::{Origin, RecordRequest};
161/// let r = RecordRequest::new(Origin::Ci);
162/// assert_eq!(r.origin, Origin::Ci);
163/// ```
164pub struct RecordRequest {
165    /// The runner's TSV report, when a suite run is being recorded.
166    pub suite: Option<std::path::PathBuf>,
167    /// `cargo test`'s output, when a crate run is being recorded.
168    pub crate_output: Option<std::path::PathBuf>,
169    /// Where the run happened.
170    pub origin: Origin,
171    /// The continuous-integration run to stamp every execution with, when there is one.
172    pub run: Option<super::RunRef>,
173    /// What each report's run measured about its checkout, by producer: values, not paths,
174    /// so a runner can pass the measurement it just took.
175    pub provenance: BTreeMap<EvidenceProducer, EvidenceProvenance>,
176    /// A coverage summary (`scripts/rust-coverage --summary-json`), when one is recorded.
177    pub coverage: Option<std::path::PathBuf>,
178    /// The ledger the executions are merged into.
179    pub ledger: LedgerTarget,
180    /// Where to write the run record as well, when asked.
181    pub run_record: Option<std::path::PathBuf>,
182}

215/// What a recording did: how much of the run reached the ledger, how much of it passed,
216/// the provenance every execution was stamped with, and what the run named that this
217/// repository does not have.
218///
219/// The last is the one worth reading. A report naming a test no runner here owns is
220/// counted in [`RecordOutcome::unknown`] and recorded for nobody, because an execution of
221/// a test that does not exist is evidence for nothing — and dropping it silently would
222/// hide a runner and a matrix that have drifted apart.
223///
224/// ```
225/// # use std::process::Command;
226/// # let root = tempfile::tempdir().unwrap();

228/// #     Command::new("git").arg("-C").arg(root.path()).args(args).output().unwrap()
229/// # };

283///      test result: ok. 2 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; \
284///      finished in 0.10s\n\
285///         Doc-tests majordomus_cli\n\
286///      test result: ok. 5 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; \

292///     RecordRequest { suite: None, crate_output: Some(log), ..RecordRequest::new(Origin::Local) };

294/// assert_eq!((outcome.recorded, outcome.passed), (1, 1), "the binary under tests/");
295/// assert_eq!(outcome.dropped.len(), 1, "the doctests, listed");
296/// assert_eq!(outcome.dropped[0].producer, EvidenceProducer::Crate);
297/// assert_eq!(outcome.dropped[0].what, "doc-tests majordomus_cli");
298/// ```
299pub struct RecordOutcome {
300    /// How many executions were written.
301    pub recorded: usize,
302    /// How many of them passed.
303    pub passed: usize,
304    /// The commit they were recorded against.
305    pub commit: String,
306    /// The weakest tree the reports' own measurements stated: `clean`, `dirty`, or
307    /// `unknown` when a report came without one. Read from [`RecordOutcome::run_record`].
308    pub working_tree: String,
309    /// Reports the run named that no runner in this repository owns; recorded for nobody
310    /// and named here rather than dropped.
311    pub unknown: Vec<String>,
312    /// What the reports held that no claim can name yet, in the order the reports held it:
313    /// the crate's own unit-test binary, its doctests, a binary outside `tests/`, a result
314    /// line no binary owned. Listed rather than ignored, so a run is never silently smaller
315    /// than it was. The same list as the run record's.
316    pub dropped: Vec<EvidenceDropped>,
317    /// The recording as a typed run.
318    pub run_record: EvidenceRunRecord,
319}

321/// The instant, RFC 3339, UTC, to whole seconds.
322///
323/// Written out rather than taken from a date crate: the crate has no time dependency, and
324/// one timestamp does not justify one. This is the civil-date algorithm, which is exact.
325pub(crate) fn now_rfc3339() -> String {
326    let secs = std::time::SystemTime::now()
327        .duration_since(std::time::UNIX_EPOCH)
328        .map(|d| d.as_secs() as i64)
329        .unwrap_or(0);
330    let (days, rem) = (secs.div_euclid(86_400), secs.rem_euclid(86_400));
331    let (h, mi, s) = (rem / 3600, (rem % 3600) / 60, rem % 60);
332    // days since 1970-01-01 to a civil date (Howard Hinnant's civil_from_days)
333    let z = days + 719_468;
334    let era = z.div_euclid(146_097);
335    let doe = z.rem_euclid(146_097);
336    let yoe = (doe - doe / 1_460 + doe / 36_524 - doe / 146_096) / 365;
337    let y = yoe + era * 400;
338    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
339    let mp = (5 * doy + 2) / 153;
340    let d = doy - (153 * mp + 2) / 5 + 1;
341    let m = if mp < 10 { mp + 3 } else { mp - 9 };
342    let y = if m <= 2 { y + 1 } else { y };
343    format!("{y:04}-{m:02}-{d:02}T{h:02}:{mi:02}:{s:02}Z")
344}

346/// One integration test binary of a `cargo test` run, as its result line stated it.
347///
348/// `name` is the binary's source under `tests/` without the extension, never the hashed
349/// file cargo built, which changes between builds. The counts are the result line's own
350/// (a count the line did not carry is 0), and `outcome` is decided from them by the rule
351/// [`read_crate_output`] states: a binary that ran nothing is not a pass.
352///
353/// ```
354/// use majordomus_cli::evidence::{read_crate_output, CrateBinary, Outcome};
355///
356/// let read = read_crate_output(
357///     "     Running tests/why.rs (target/debug/deps/why-1)\n\
358///      test result: ok. 1 passed; 0 failed; 2 ignored; 0 measured; 4 filtered out; \
359///      finished in 0.60s\n",
360/// );
361/// let why: &CrateBinary = &read.binaries[0];
362/// assert_eq!(why.name, "why");
363/// assert_eq!((why.passed, why.failed, why.ignored, why.filtered_out), (1, 0, 2, 4));
364/// assert_eq!(why.seconds, 1, "0.60s rounds to the nearest second");
365/// assert_eq!(why.outcome, Outcome::Skip, "a filtered subset is not the binary's proof");
366/// ```
367#[derive(Debug, Clone, PartialEq, Eq)]
368pub struct CrateBinary {
369    /// The binary's name: `tests/<name>.rs` without the directory and the extension.
370    pub name: String,
371    /// What the binary's result line, read by its counts, says happened.
372    pub outcome: Outcome,
373    /// `finished in` rounded to whole seconds; 0 when the line carried none.
374    pub seconds: u64,
375    /// Tests that ran and passed.
376    pub passed: u64,
377    /// Tests that ran and failed.
378    pub failed: u64,
379    /// Tests that were compiled in and not run.
380    pub ignored: u64,
381    /// Tests a name filter left out of this run.
382    pub filtered_out: u64,
383}

385/// What `cargo test`'s output held: the integration binaries a claim can name, and what it
386/// cannot, listed in [`CrateRead::dropped`] in the order the output held it.
387///
388/// ```
389/// use majordomus_cli::evidence::{read_crate_output, CrateRead};
390///
391/// let read: CrateRead = read_crate_output(
392///     "   Doc-tests majordomus_cli\n\
393///      test result: ok. 5 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; \
394///      finished in 2.00s\n",
395/// );
396/// assert!(read.binaries.is_empty(), "no claim can name a doctest yet");
397/// assert_eq!(read.dropped[0].what, "doc-tests majordomus_cli");
398/// ```
399#[derive(Debug, Clone, Default, PartialEq, Eq)]
400pub struct CrateRead {
401    /// Each binary under `tests/` that the output named, in order, with its outcome.
402    pub binaries: Vec<CrateBinary>,
403    /// What the output held that no claim can name yet, with the reason for each.
404    pub dropped: Vec<EvidenceDropped>,
405}

407/// The entry a `Running` or `Doc-tests` line opened, waiting for its result line.
408enum Pending {
409    /// An integration binary under `tests/`: a claim can name it.
410    Binary(String),
411    /// Anything else: listed, with why, when it closes.
412    Dropped(String, &'static str),
413}
414
415impl CrateRead {
416    /// Close the entry that was waiting for a result line and got none before the next
417    /// `Running`, the next `Doc-tests` or the end of the output. A binary that started and
418    /// never reported is an error, because the harness could not say what ran; anything
419    /// else is listed as it is.
420    fn close(&mut self, pending: Option<Pending>) {
421        match pending {
422            None => {}
423            Some(Pending::Binary(name)) => self.binaries.push(CrateBinary {
424                name,
425                outcome: Outcome::Error,
426                seconds: 0,
427                passed: 0,
428                failed: 0,
429                ignored: 0,
430                filtered_out: 0,
431            }),
432            Some(Pending::Dropped(what, reason)) => self.dropped.push(dropped(what, reason)),
433        }
434    }
435}

446/// Read `cargo test`'s output: each `Running tests/<name>.rs` line joined to the
447/// `test result:` line that follows it, judged by its counts, and everything else it ran
448/// listed rather than ignored.
449///
450/// `cargo test`'s machine format is still unstable, so this reads what every cargo prints.
451/// The `Running` line names the binary the following result belongs to; without it a run of
452/// forty binaries is forty anonymous result lines, which is how a per-test-binary outcome
453/// becomes an untraceable total. Escape sequences are stripped from every line before it is
454/// matched, so a coloured run reads as a plain one.
455///
456/// A result line is judged by the first of these that holds:
457///
458/// | The result line | Outcome |
459/// |---|---|
460/// | a failed count above zero | fail |
461/// | the word `FAILED` | fail |
462/// | the word `ok` and no `passed` count | error |
463/// | the word `ok` and nothing passed (every test ignored, say) | skip |
464/// | the word `ok` and a filtered-out count above zero (a subset) | skip |
465/// | the word `ok` | pass |
466/// | any other word | error |
467///
468/// A binary whose `Running` line is followed by no result line is an error. The crate's own
469/// unit-test binary (`Running unittests <path>`), its doctests (`Doc-tests <crate>`), a
470/// binary outside `tests/` and a result line with no `Running` line before it are listed in
471/// [`CrateRead::dropped`] with the reason, and never credited to a neighbour.
472///
473/// ```
474/// use majordomus_cli::evidence::{read_crate_output, Outcome};
475///
476/// let read = read_crate_output(
477///     "     Running tests/all_ignored.rs (target/debug/deps/all_ignored-1)\n\
478///      test result: ok. 0 passed; 0 failed; 2 ignored; 0 measured; 0 filtered out; \
479///      finished in 0.00s\n\
480///           Running tests/broken.rs (target/debug/deps/broken-1)\n\
481///      test result: ok. 2 passed; 1 failed; 0 ignored; 0 measured; 0 filtered out; \
482///      finished in 0.20s\n\
483///           Running tests/silent.rs (target/debug/deps/silent-1)\n",
484/// );
485/// let outcomes: Vec<(&str, Outcome)> =
486///     read.binaries.iter().map(|b| (b.name.as_str(), b.outcome)).collect();
487/// assert_eq!(
488///     outcomes,
489///     [("all_ignored", Outcome::Skip), ("broken", Outcome::Fail), ("silent", Outcome::Error)]
490/// );
491/// ```
492pub fn read_crate_output(text: &str) -> CrateRead {
493    let mut read = CrateRead::default();
494    let mut pending: Option<Pending> = None;
495    for line in text.lines() {
496        let plain = strip_ansi(line);
497        let t = plain.trim();
498        if let Some(rest) = t.strip_prefix("Running ") {
499            read.close(pending.take());
500            // `tests/why.rs (target/debug/deps/why-1a2b)` — the source path is the name,
501            // never the hashed binary, which changes between builds
502            let mut words = rest.split_whitespace();
503            let first = words.next().unwrap_or("");
504            pending = Some(if first == "unittests" {
505                let path = words.next().unwrap_or("");
506                Pending::Dropped(format!("unittests {path}"), UNIT_TESTS)
507            } else if let Some(name) = first
508                .strip_prefix("tests/")
509                .and_then(|p| p.strip_suffix(".rs"))
510            {
511                Pending::Binary(name.to_string())
512            } else {
513                Pending::Dropped(first.to_string(), NOT_AN_INTEGRATION_BINARY)
514            });
515            continue;
516        }
517        if let Some(rest) = t.strip_prefix("Doc-tests ") {
518            read.close(pending.take());
519            let krate = rest.split_whitespace().next().unwrap_or("");
520            pending = Some(Pending::Dropped(format!("doc-tests {krate}"), DOC_TESTS));
521            continue;
522        }
523        if let Some(rest) = t.strip_prefix("test result: ") {
524            match pending.take() {
525                None => read.dropped.push(dropped("a result line", UNATTRIBUTED)),
526                Some(Pending::Dropped(what, reason)) => read.dropped.push(dropped(what, reason)),
527                Some(Pending::Binary(name)) => read.binaries.push(judge(name, rest)),
528            }
529        }
530    }
531    read.close(pending.take());
532    read
533}

535/// A binary's result line, `<word>. <counts>; finished in <S>s`, judged by the table of
536/// [`read_crate_output`].
537fn judge(name: String, rest: &str) -> CrateBinary {
538    let (word, counts) = rest.split_once('.').unwrap_or((rest, ""));
539    let word = word.trim();
540    let (mut passed, mut failed, mut ignored, mut filtered_out) = (None, 0, 0, 0);
541    let mut seconds = 0;
542    for part in counts.split(';').map(str::trim) {
543        if let Some(s) = part.strip_prefix("finished in ") {
544            seconds = s
545                .trim_end_matches('s')
546                .trim()
547                .parse::<f64>()
548                .map(|f| f.round() as u64)
549                .unwrap_or(0);
550            continue;
551        }
552        let Some((n, label)) = part.split_once(' ') else {
553            continue;
554        };
555        let Ok(n) = n.parse::<u64>() else { continue };
556        match label.trim() {
557            "passed" => passed = Some(n),
558            "failed" => failed = n,
559            "ignored" => ignored = n,
560            "filtered out" => filtered_out = n,
561            _ => {}
562        }
563    }
564    let outcome = if failed > 0 || word == "FAILED" {
565        Outcome::Fail
566    } else if word == "ok" {
567        match passed {
568            // the line said `ok` and not what ran: nothing here can say it passed
569            None => Outcome::Error,
570            // it ran nothing, or only a subset a filter chose: a skip, not the proof
571            Some(0) => Outcome::Skip,
572            Some(_) if filtered_out > 0 => Outcome::Skip,
573            Some(_) => Outcome::Pass,
574        }
575    } else {
576        Outcome::Error
577    };
578    CrateBinary {
579        name,
580        outcome,
581        seconds,
582        passed: passed.unwrap_or(0),
583        failed,
584        ignored,
585        filtered_out,
586    }
587}

589/// A line with its terminal escape sequences removed: `ESC [` up to its final byte (`@`
590/// to `~`), `ESC ]` up to BEL or `ESC \`, `ESC` with intermediate bytes (space to `/`) up to
591/// the final byte after them, and any other `ESC` with the character after it.
592/// `cargo test` colours its output when it thinks a person is watching, and a coloured
593/// `Running` or `ok` is the same line as a plain one. The test harness resets its colour
594/// through the terminal's own `sgr0`, which on an xterm is `ESC ( B ESC [ m`: the `( B`
595/// selects the character set, and reading it as a two-character pair would leave its `B`
596/// behind and turn `ok` into `okB`.
597fn strip_ansi(line: &str) -> String {
598    const ESC: char = '\u{1b}';
599    let mut out = String::with_capacity(line.len());
600    let mut chars = line.chars().peekable();
601    while let Some(c) = chars.next() {
602        if c != ESC {
603            out.push(c);
604            continue;
605        }
606        match chars.next() {
607            Some('[') => {
608                for c in chars.by_ref() {
609                    if ('@'..='~').contains(&c) {
610                        break;
611                    }
612                }
613            }
614            Some(']') => {
615                let mut after_esc = false;
616                for c in chars.by_ref() {
617                    if c == '\u{7}' || (after_esc && c == '\\') {
618                        break;
619                    }
620                    after_esc = c == ESC;
621                }
622            }
623            // intermediate bytes, then the one final byte that ends the sequence
624            Some(' '..='/') => {
625                while chars.next_if(|c| (' '..='/').contains(c)).is_some() {}
626                chars.next();
627            }
628            Some(_) | None => {}
629        }
630    }
631    out
632}

634/// One `Running tests/<name>.rs` / `test result:` pair per test binary, joined: the
635/// binaries of [`read_crate_output`] as `(name, outcome, seconds)`, without what it listed.
636///
637/// ```
638/// use majordomus_cli::evidence::parse_crate_binaries;
639/// let out = "   Running tests/why.rs (target/debug/deps/why-1a2b)\n\
640///             test result: ok. 12 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; \
641///             finished in 0.41s\n\
642///                Running tests/product.rs (target/debug/deps/product-3c4d)\n\
643///             test result: FAILED. 3 passed; 1 failed; 0 ignored; 0 measured\n";
644/// let got = parse_crate_binaries(out);
645/// assert_eq!(got.len(), 2);
646/// assert_eq!(got[0].0, "why");
647/// assert!(got[0].1.proves());
648/// assert_eq!(got[1].0, "product");
649/// assert!(!got[1].1.proves());
650/// ```
651/// ```
652/// use majordomus_cli::evidence::parse_crate_binaries;
653/// assert!(parse_crate_binaries("nothing to see here").is_empty());
654/// ```
655/// The outputs of several lanes are joined end to end, so a binary whose run was cut short
656/// must not take the next result line, a doctest run's least of all: it stands as an error
657/// that ran nothing, and the doctest result belongs to no binary.
658/// ```
659/// use majordomus_cli::evidence::{parse_crate_binaries, Outcome};
660/// let joined = "     Running tests/cut.rs (target/debug/deps/cut-1a2b)\n\
661///                  Doc-tests majordomus_cli\n\
662///               test result: ok. 900 passed; 0 failed; finished in 600.00s\n";
663/// assert_eq!(parse_crate_binaries(joined), [("cut".to_string(), Outcome::Error, 0)]);
664/// ```
665pub fn parse_crate_binaries(text: &str) -> Vec<(String, Outcome, u64)> {
666    read_crate_output(text)
667        .binaries
668        .into_iter()
669        .map(|b| (b.name, b.outcome, b.seconds))
670        .collect()
671}

673/// The runner's TSV: `name <TAB> result <TAB> seconds <TAB> phase`, one case per line.
674///
675/// A malformed line is refused rather than skipped, for the same reason the renderer
676/// refuses one: a recording that silently dropped a case would leave that case's claim
677/// reporting `not run` after a run that did run it, which is a lie in the safe direction
678/// and still a lie.
679fn parse_suite(text: &str) -> Result<Vec<(String, Outcome, u64)>> {
680    let mut out = Vec::new();
681    for (n, line) in text.lines().enumerate() {
682        if line.trim().is_empty() {
683            continue;
684        }
685        let f: Vec<&str> = line.split('\t').collect();
686        if f.len() < 4 {
687            return Err(Error::InvalidSurface {
688                surface: "evidence".into(),
689                reason: format!(
690                    "line {} of the run report has {} field(s); it is \
691                     name<TAB>result<TAB>seconds<TAB>phase",
692                    n + 1,
693                    f.len()
694                ),
695            });
696        }
697        out.push((
698            f[0].to_string(),
699            Outcome::parse(f[1]),
700            f[2].parse().unwrap_or(0),
701        ));
702    }
703    Ok(out)
704}

706/// Read the runs a request names, stamp each result with provenance, and merge into the
707/// ledger of `root`.
708///
709/// It decides nothing: a case that failed is a case the runner said failed. Every refusal
710/// comes before any write: a request naming no report, a tree with no commit, a
711/// measurement of a report not given, of another producer, of another commit or of other
712/// bytes, a local or release measurement that names no report, a CI report without a
713/// measurement, a malformed report (refused rather than partly read), a coverage summary
714/// of another schema, and a run-record path that is a directory or a ledger.
715///
716/// Each execution carries its report's measured tree, or `unknown` without one, and the
717/// recording is returned as an [`EvidenceRunRecord`] in [`RecordOutcome::run_record`].
718///
719/// Because it merges, a later partial run updates only what it ran:
720///
721/// ```
722/// # use std::process::Command;
723/// # let root = tempfile::tempdir().unwrap();
724/// # let git = |args: &[&str]| {
725/// #     Command::new("git").arg("-C").arg(root.path()).args(args).output().unwrap()
726/// # };
727/// # git(&["init", "-q"]);
728/// # git(&["config", "user.email", "t@example.com"]);
729/// # git(&["config", "user.name", "t"]);
730/// # std::fs::create_dir_all(root.path().join("test/cases")).unwrap();
731/// # std::fs::write(root.path().join("test/cases/07_scope.sh"), "echo scope\n").unwrap();
732/// # std::fs::write(root.path().join("test/cases/08_other.sh"), "echo other\n").unwrap();
733/// # git(&["add", "-A"]);
734/// # git(&["commit", "-qm", "init"]);
735/// use majordomus_cli::evidence::{record, Ledger, Origin, Outcome, RecordRequest};
736///
737/// let reports = tempfile::tempdir().unwrap();
738/// let suite = |name: &str, rows: &str| {
739///     let p = reports.path().join(name);
740///     std::fs::write(&p, rows).unwrap();
741///     RecordRequest { suite: Some(p), crate_output: None, ..RecordRequest::new(Origin::Local) }
742/// };
743///
744/// let both = suite("all.tsv", "07_scope\tok\t1\tparallel\n08_other\tok\t2\tparallel\n");
745/// assert_eq!(record(root.path(), &both).unwrap().recorded, 2);
746///
747/// let again = suite("one.tsv", "07_scope\tFAIL\t9\tserial\n");
748/// assert_eq!(record(root.path(), &again).unwrap().recorded, 1);
749///
750/// let ledger = Ledger::load(root.path()).unwrap();
751/// assert_eq!(ledger.latest("suite:07_scope").unwrap().outcome, Outcome::Fail);
752/// assert_eq!(
753///     ledger.latest("suite:08_other").unwrap().outcome,
754///     Outcome::Pass,
755///     "the second run must not erase evidence it never measured",
756/// );
757///
758/// // a malformed line is refused, not skipped: an under-recorded run would report
759/// // `not run` for a test that ran
760/// let short = suite("bad.tsv", "07_scope\tok\t12\n");
761/// let refused = record(root.path(), &short).unwrap_err().to_string();
762/// assert!(refused.contains("field(s)"), "{refused}");
763/// ```
764pub fn record(root: &Path, req: &RecordRequest) -> Result<RecordOutcome> {
765    use EvidenceProducer as P;
766    if req.suite.is_none() && req.crate_output.is_none() && req.coverage.is_none() {
767        return Err(refused(
768            "nothing to record: give --suite, --crate-output, --coverage or several",
769        ));
770    }
771
772    // the commit as a full id, once: every row and the coverage record carry this one
773    let commit_id = match crate::git::inspect(root) {
774        crate::git::GitState::Available(i) => i.head,
775        crate::git::GitState::Unavailable { .. } => None,
776    }
777    .and_then(|h| CommitId::parse(&h).ok())
778    .ok_or_else(|| {
779        refused(
780            "this is not a git work tree with a commit, so a run recorded here would carry no \
781             provenance and prove nothing",
782        )
783    })?;
784    let commit = commit_id.as_str().to_string();
785
786    let report_of = |p: EvidenceProducer| match p {
787        P::Suite => req.suite.as_deref(),
788        P::Crate => req.crate_output.as_deref(),
789        P::Coverage => req.coverage.as_deref(),
790    };
791    for (&p, prov) in &req.provenance {
792        let w = p.as_str();
793        if report_of(p).is_none() {
794            return Err(refused(format!(
795                "a measurement of the {w} report was given, and no {w} report"
796            )));
797        }
798        if let Some(x) = prov.producer.filter(|x| *x != p) {
799            return Err(refused(format!(
800                "the {w} measurement measures the {} report, not the {w} one",
801                x.as_str()
802            )));
803        }
804        if CommitId::parse(&prov.commit).is_err() || prov.commit != commit {
805            return Err(refused(format!(
806                "the {w} report was measured at {}, and this checkout is at {}: record it in a \
807                 checkout of the commit it ran",
808                short12(&prov.commit),
809                short12(&commit)
810            )));
811        }
812        if req.origin != Origin::Ci && prov.report.is_none() {
813            return Err(refused(format!(
814                "the {w} measurement names no report, so it cannot say which run it measured: \
815                 stamp with --report <file>"
816            )));
817        }
818    }
819    if req.origin == Origin::Ci {
820        for p in [P::Suite, P::Crate] {
821            if report_of(p).is_some() && !req.provenance.contains_key(&p) {
822                return Err(refused(format!(
823                    "a CI report is recorded only with the measurement its own run made: give \
824                     --provenance {}=<file> from `majordomus evidence stamp`",
825                    p.as_str()
826                )));
827            }
828        }
829    }
830    // the bytes a report holds, refused when its measurement describes other bytes
831    let read_report = |p: EvidenceProducer, path: &Path| -> Result<Vec<u8>> {
832        let bytes = std::fs::read(path).map_err(Error::Transport)?;
833        if let Some(artifact) = req.provenance.get(&p).and_then(|m| m.report.as_ref()) {
834            let actual = digest_of(&bytes);
835            if artifact.digest != actual {
836                return Err(refused(format!(
837                    "the measurement describes another report ({} ≠ {})",
838                    short12(artifact.digest.trim_start_matches("sha256:")),
839                    short12(actual.trim_start_matches("sha256:"))
840                )));
841            }
842        }
843        Ok(bytes)
844    };
845    let tree_of = |p: EvidenceProducer| {
846        req.provenance
847            .get(&p)
848            .map_or(TreeState::Unknown, |m| m.working_tree)
849    };
850
851    let mut results: Vec<(TestId, Outcome, u64)> = Vec::new();
852    let mut unknown = Vec::new();
853    let mut absent = Vec::new();
854
855    if let Some(p) = &req.suite {
856        let bytes = read_report(P::Suite, p)?;
857        let text = String::from_utf8_lossy(&bytes);
858        let before = results.len();
859        for (name, outcome, seconds) in parse_suite(&text)? {
860            // the runner names a case by its stem, which is exactly the suite test id
861            let id = TestId {
862                runner: Runner::Suite,
863                name: name.clone(),
864            };
865            if root.join(id.source()).exists() {
866                results.push((id, outcome, seconds));
867            } else {
868                unknown.push(format!("suite:{name}"));
869            }
870        }
871        if results.len() == before {
872            absent.push(absence(
873                P::Suite,
874                "the report named no test this repository has",
875            ));
876        }
877    } else {
878        absent.push(absence(P::Suite, "no suite report was given"));
879    }
880
881    let mut dropped = Vec::new();
882    if let Some(p) = &req.crate_output {
883        let bytes = read_report(P::Crate, p)?;
884        let read = read_crate_output(&String::from_utf8_lossy(&bytes));
885        if read.binaries.is_empty() {
886            absent.push(absence(
887                P::Crate,
888                "the output named no integration test binary",
889            ));
890        }
891        for binary in read.binaries {
892            let id = TestId {
893                runner: Runner::Crate,
894                name: binary.name.clone(),
895            };
896            if root.join(id.source()).exists() {
897                results.push((id, binary.outcome, binary.seconds));
898            } else {
899                unknown.push(format!("crate:{}", binary.name));
900            }
901        }
902        // what the output held that no claim can name yet, in the order it held it
903        dropped = read.dropped;
904    } else {
905        absent.push(absence(P::Crate, "no crate report was given"));
906    }
907
908    let coverage = match &req.coverage {
909        Some(p) => {
910            let bytes = read_report(P::Coverage, p)?;
911            let floors = EvidenceCoverageFloors::read(root);
912            let text = String::from_utf8_lossy(&bytes);
913            let cov =
914                EvidenceCoverage::from_summary(&text, commit_id, tree_of(P::Coverage), floors)
915                    .map_err(|e| refused(format!("{}: {e}", p.display())))?;
916            Some(cov)
917        }
918        None => {
919            absent.push(absence(P::Coverage, "no coverage report was given"));
920            None
921        }
922    };
923
924    if let Some(out) = &req.run_record {
925        let is_ledger = [LEDGER_PATH, LOCAL_LEDGER_PATH].iter().any(|l| {
926            let ledger = root.join(l);
927            out == &ledger
928                || matches!(
929                    (std::fs::canonicalize(out), std::fs::canonicalize(&ledger)),
930                    (Ok(a), Ok(b)) if a == b
931                )
932        });
933        if out.as_os_str().is_empty() || out.is_dir() || is_ledger {
934            return Err(refused(format!(
935                "`{}` cannot hold a run record: it is empty, a directory or a ledger",
936                out.display()
937            )));
938        }
939    }
940
941    let at = now_rfc3339();
942    let mut executions = Vec::with_capacity(results.len());
943    let mut passed = 0;
944    for (id, outcome, seconds) in results {
945        if outcome.proves() {
946            passed += 1;
947        }
948        let producer = match id.runner {
949            Runner::Suite => P::Suite,
950            Runner::Crate => P::Crate,
951        };
952        let tree = tree_of(producer);
953        let source = id.source();
954        let digest = digest_ran(root, &commit, &source, tree);
955        executions.push(Execution {
956            test: id.as_string(),
957            runner: id.runner,
958            source,
959            outcome,
960            seconds,
961            commit: commit.clone(),
962            working_tree: tree.as_str().into(),
963            digest,
964            at: at.clone(),
965            origin: req.origin,
966            command: id.reproduce(),
967            run: req.run.clone(),
968        });
969    }
970
971    let mut ledger = Ledger::load_from(root, req.ledger)?;
972    let recorded = ledger.merge(executions.clone());
973    // A run that recorded nothing writes nothing. Saving here would create a ledger
974    // holding no executions out of a report that named only tests this repository does
975    // not have — a file that says "evidence was recorded" where none was. The malformed
976    // report above is refused before any write for the same reason, and the two paths
977    // must not disagree about what an empty recording leaves behind.
978    if recorded > 0 {
979        ledger.save_to(root, req.ledger)?;
980    }
981
982    crate::order::canonical_strings(&mut unknown);
983    unknown.dedup();
984    let given = [P::Suite, P::Crate, P::Coverage]
985        .into_iter()
986        .filter(|p| report_of(*p).is_some());
987    let working_tree = given.fold(TreeState::Clean, |t, p| t.weaker(tree_of(p)));
988    let run_record = EvidenceRunRecord {
989        schema: RUN_RECORD_SCHEMA,
990        id: run_id(req.origin, req.run.as_ref(), &commit, &at),
991        origin: req.origin,
992        commit: commit.clone(),
993        working_tree,
994        recorded_at: at,
995        ledger: req.ledger,
996        run: req.run.clone(),
997        provenance: req.provenance.values().cloned().collect(),
998        totals: EvidenceRunTotals::of(&executions),
999        absent,
1000        dropped,
1001        unknown,
1002        coverage,
1003        executions,
1004    };
1005
1006    if let Some(out) = &req.run_record {
1007        write_run_record(out, &run_record).map_err(|e| {
1008            refused(format!(
1009                "the executions were recorded into {}, and the run record could not be \
1010                 written to {}: {e}",
1011                req.ledger.path(),
1012                out.display()
1013            ))
1014        })?;
1015    }
1016
1017    Ok(RecordOutcome {
1018        recorded,
1019        passed,
1020        commit,
1021        working_tree: run_record.working_tree.as_str().to_string(),
1022        unknown: run_record.unknown.clone(),
1023        dropped: run_record.dropped.clone(),
1024        run_record,
1025    })
1026}

1047/// The digest of the bytes the run executed: the commit's blob when the run measured a
1048/// clean tree, else the working copy, else `sha256:unreadable`.
1049fn digest_ran(root: &Path, commit: &str, source: &str, tree: TreeState) -> String {
1050    if tree == TreeState::Clean {
1051        let blob = crate::git::read_only(root)
1052            .args(["cat-file", "blob", &format!("{commit}:{source}")])
1053            .output()
1054            .ok()
1055            .filter(|out| out.status.success());
1056        if let Some(out) = blob {
1057            return digest_of(&out.stdout);
1058        }
1059    }
1060    std::fs::read(root.join(source))
1061        .map(|b| digest_of(&b))
1062        .unwrap_or_else(|_| "sha256:unreadable".into())
1063}

1065fn write_run_record(out: &Path, run_record: &EvidenceRunRecord) -> std::io::Result<()> {
1066    // a bare file name's parent is the empty path, which create_dir_all accepts as is
1067    out.parent().map_or(Ok(()), std::fs::create_dir_all)?;
1068    // plain data with string keys: serialising it cannot fail, so there is no error to carry
1069    let mut text = serde_json::to_string_pretty(run_record).unwrap_or_default();
1070    text.push('\n');
1071    std::fs::write(out, text)
1072}
